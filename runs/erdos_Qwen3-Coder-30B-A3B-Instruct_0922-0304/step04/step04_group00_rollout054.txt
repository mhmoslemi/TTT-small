import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 overlap integral for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    n_particles = 50
    max_iter = 10000
    w = 0.7  # inertia weight
    c1 = 1.5  # cognitive parameter
    c2 = 1.5  # social parameter
    
    # Initialize particles
    particles = []
    velocities = []
    
    for i in range(n_particles):
        if i == 0 and 'initial_h_values' in globals():
            particle_h = h.copy()
        else:
            particle_h = h + np.random.normal(0, 0.05, n_points)
            particle_h = np.clip(particle_h, 0, 1)
            particle_h = particle_h * (n_points / 2) / np.sum(particle_h)
            
        particles.append(particle_h)
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # Store best positions for each particle and global best
    p_best = [p.copy() for p in particles]
    p_best_values = [compute_c5(p, dx) for p in particles]
    g_best = particles[np.argmin(p_best_values)].copy()
    g_best_value = min(p_best_values)
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < p_best_values[i]:
                p_best[i] = particles[i].copy()
                p_best_values[i] = current_c5
                
                # Update global best
                if current_c5 < g_best_value:
                    g_best = particles[i].copy()
                    g_best_value = current_c5
                    
        # Early exit if target achieved
        if g_best_value <= 0.3807:
            break
            
        # Progress check for timeout prevention
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 1:
                break
    
    return (g_best.tolist(), g_best_value, n_points)
