import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    pop_size = 100
    num_generations = 1000
    elite_rate = 0.2
    mutation_strength = 0.1
    adaptive_mutation = True
    
    # Initialize population
    population = []
    fitness = []
    
    # Create initial population
    for _ in range(pop_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
        fitness.append(evaluate_c5(individual, dx))
    
    # ES iterations
    for generation in range(num_generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Selection
        elite_count = int(pop_size * elite_rate)
        elite = population[:elite_count]
        
        # Reproduction with mutation
        new_population = elite.copy()
        
        # Generate offspring
        for _ in range(pop_size - elite_count):
            # Select parent from top 50%
            parent_idx = np.random.randint(0, pop_size // 2)
            parent = population[parent_idx]
            
            # Mutate
            mutated = parent.copy()
            
            # Mutate about 50% of elements
            num_mutations = max(1, n_points // 2)
            mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
            for idx in mutation_indices:
                mutated[idx] += np.random.normal(0, mutation_strength)
            
            # Repair
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated * (n_points / 2) / np.sum(mutated)
            
            new_population.append(mutated)
        
        # Evaluate new population
        population = new_population
        fitness = [evaluate_c5(ind, dx) for ind in population]
        
        # Adaptive mutation strength
        if adaptive_mutation and generation > 0 and generation % 10 == 0:
            improvement = np.mean(fitness[:elite_count]) - np.mean(fitness[elite_count:])
            if improvement < 0.0001:
                mutation_strength *= 1.1
            else:
                mutation_strength *= 0.95
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s * 0.95 and best_c5 > 0.3807:
        from scipy.optimize import minimize
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
