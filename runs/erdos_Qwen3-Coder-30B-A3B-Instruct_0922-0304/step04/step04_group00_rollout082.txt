import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 bound for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add some noise to avoid stuck
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop with coordinate-wise updates
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Save original
            original_h_i = h[i]
            
            # Try different values for h[i]
            best_h_i = original_h_i
            best_local_c5 = current_c5
            
            # Evaluate several candidates
            candidates = [0, 0.25, 0.5, 0.75, 1.0]

                h[i] = candidate
                # Re-normalize to maintain sum constraint
                current_sum = np.sum(h)

            h = np.clip(h, 0, 1)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
