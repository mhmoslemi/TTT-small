import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ES parameters
    population_size = 100
    generations = 1000
    elite_size = population_size // 5
    sigma = 0.1
    mutation_rate = 0.5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = particle * (n_points / 2) / np.sum(particle)
        population.append(particle)
    
    # ES iterations
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate population
        fitness = []
        for particle in population:
            fit = evaluate_c5(particle, dx)
            fitness.append(fit)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Selection (elitism)
        elite = population[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(population_size - elite_size):
            # Select parent
            parent = elite[np.random.randint(0, elite_size)]
            
            # Mutation
            child = parent.copy()
            mask = np.random.random(n_points) < mutation_rate
            child[mask] += np.random.normal(0, sigma, np.sum(mask))
            
            # Repair
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            offspring.append(child)
        
        # Create new population
        population = elite + offspring
        
        # Adapt sigma
        if gen % 10 == 0:
            improvement = np.mean(fitness[:elite_size]) - np.mean(fitness[elite_size:elite_size*2])
            if improvement > 0:
                sigma *= 0.95
            else:
                sigma *= 1.05
            
            sigma = np.clip(sigma, 0.001, 0.5)
    
    # Refinement with SLSQP
    if time.time() - start_time < budget_s * 0.98:
        from scipy.optimize import minimize
        
        # Constraints for SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function
        def objective(h):
            return evaluate_c5(h, dx)
        
        # Refine with SLSQP
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
