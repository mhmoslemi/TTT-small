import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    population_size = 30
    max_generations = 2000
    mutation_scale = 0.1
    elite_ratio = 0.25
    success_threshold = 0.1
    adaptation_factor = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population
    for i in range(1, population_size):
        h = np.random.normal(0.5, 0.1, n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_c5(ind, dx) for ind in population]
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    # Evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = int(elite_ratio * population_size)
        elite_indices = sorted_indices[:elite_size]
        elite_individuals = [population[i] for i in elite_indices]
        elite_fitness = [fitness[i] for i in elite_indices]
        
        # Update best
        if elite_fitness[0] < best_c5:
            best_h = elite_individuals[0].copy()
            best_c5 = elite_fitness[0]
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create offspring
        offspring = []
        for _ in range(population_size):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, mutation_scale, n_points)
            child = parent + noise
            child = project_feasible(child, n_points)
            offspring.append(child)
            
        # Evaluate offspring
        offspring_fitness = [evaluate_c5(ind, dx) for ind in offspring]
        
        # Combine and select next generation
        all_individuals = elite_individuals + offspring
        all_fitness = elite_fitness + offspring_fitness
        
        # Sort and keep top population_size
        sorted_indices = np.argsort(all_fitness)
        selected_indices = sorted_indices[:population_size]
        population = [all_individuals[i] for i in selected_indices]
        fitness = [all_fitness[i] for i in selected_indices]
        
        # Adapt mutation scale
        recent_improvements = len([f for f in offspring_fitness if f < best_c5])
        if recent_improvements / len(offspring_fitness) > success_threshold:
            mutation_scale *= (1 + adaptation_factor)
        else:
            mutation_scale *= (1 - adaptation_factor)
        mutation_scale = np.clip(mutation_scale, 0.01, 0.5)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
