import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_feasible(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 30
    max_iter = 2000
    elite_ratio = 0.25
    mutation_scale = 0.1
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    else:
        # Triangular function
        h0 = np.zeros(n_points)
        for j in range(n_points):
            h0[j] = min(1, 2 * abs(j - n_points/2) / n_points)
        h0 = project_feasible(h0, n_points)
        population.append(h0.copy())
    
    # Fill remaining population with random feasible solutions
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_feasible(h, n_points)
        population.append(h.copy())
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # ES parameters
    sigma = mutation_scale
    success_count = 0
    success_window = 20
    adaptive_factor = 1.2
    
    # Main evolutionary loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        fitness = []
        for h in population:
            fitness.append(evaluate_c5(h, dx))
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_fitness = [fitness[i] for i in sorted_indices]
        sorted_population = [population[i] for i in sorted_indices]
        
        # Update global best
        if sorted_fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = sorted_fitness[0]
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection: keep top 25%
        elite_size = max(1, int(pop_size * elite_ratio))
        elite = sorted_population[:elite_size]
        
        # Generate offspring
        offspring = []
        for i in range(pop_size):
            # Select parent from elite
            parent_idx = np.random.randint(0, elite_size)
            parent = elite[parent_idx]
            
            # Mutate
            noise = np.random.normal(0, sigma, n_points)
            child = parent + noise
            
            # Make feasible
            child = project_feasible(child, n_points)
            
            offspring.append(child)
        
        # Update population
        population = offspring
        
        # Adapt sigma based on success
        if iter_num % success_window == 0 and iter_num > 0:
            if success_count > success_window * 0.7:
                sigma *= adaptive_factor
            elif success_count < success_window * 0.3:
                sigma /= adaptive_factor
            success_count = 0
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
