import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Haar wavelet basis construction
    def haar_wavelets(n):
        # Generate all Haar wavelets up to scale log2(n)
        wavelets = []
        # Scaling function (constant)
        wavelets.append(np.ones(n))
        
        # Haar wavelets
        scale = 1
        while scale < n:
            for k in range(scale):
                psi = np.zeros(n)
                start = 2 * k * scale
                end1 = start + scale
                end2 = start + 2 * scale
                if end1 <= n and end2 <= n:
                    psi[start:end1] = 1
                    psi[end1:end2] = -1
                    wavelets.append(psi)
            scale *= 2
        return np.array(wavelets[:n])  # limit to n basis functions
    
    # Try to use initial construction if available
    try:
        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    except NameError:
        # Fallback: simple step function
        h_initial = np.zeros(n_points)
        h_initial[:n_points//2] = 1
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)

    # Formulate optimization problem
    # We want to minimize max_k \int h(x)(1 - h(x+k)) dx
    # which is max_k \int h(x) dx - \int h(x) h(x+k) dx

    # Try convex formulation with wavelet basis to represent h
    wavelets = haar_wavelets(n_points)
    
    # Create Haar wavelet coefficients as variables
    coeffs = cp.Variable(n_points)
    
    # Wavelet representation of h(x)
    h_vec = wavelets.T @ coeffs
    
    # Normalize to integral=1
    integral_constraint = cp.sum(h_vec) * dx == 1
    
    # Bounding constraints
    bound_constraints = [h_vec >= 0, h_vec <= 1]
    
    # Various optimizations to minimize overlap
    # This is a complex constrained optimization problem,
    # better to use gradient-based optimization with provided tools
    
    # Quick gradient-based approach on top of wavelet basis
    
    def calculate_c5(h_vec):
        # Cast as array and evaluate overlap
        h = np.array(h_vec)
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    # Use local optimization for speed
    from scipy.optimize import minimize
    
    # Warm start with initial_h_values or step function
    h0 = h_initial.copy()
    h0 = h0 * (n_points / 2) / np.sum(h0)  # ensure integral = 1
    h0 = np.clip(h0, 0, 1)
    
    # Precompute required convolutions 
    conv_matrices = {}
    for k in range(n_points):
        # create kernel for shifting by k
        shift_kernel = np.zeros(n_points)
        shift_kernel[k] = 1
        conv_matrices[k] = shift_kernel
    
    # Optimize
    def objective(h):
        # Internally compute C5 based on given h
        # This implements the overlap integral using fft
        h_arr = np.array(h)
        result = correlate(h_arr, 1 - h_arr, mode="full")
        return np.max(result) * dx
    
    # Various bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    result = minimize(objective, h0, method='L-BFGS-B', bounds=bounds, 
                      options={'maxiter': 3000, 'ftol': 1e-10})
    
    if time.time() - start_time > budget_s - 1:
        pass
    
    # Extract best solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)  # Re-normalize
    
    final_c5 = calculate_c5(best_h)
    
    # Return best solution
    return (best_h.tolist(), final_c5, n_points)
