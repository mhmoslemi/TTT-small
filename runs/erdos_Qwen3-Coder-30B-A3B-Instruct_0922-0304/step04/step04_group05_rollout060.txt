import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_to_feasible(h, target_sum):
    # Project to [0,1] bounds
    h = np.clip(h, 0, 1)
    # Adjust to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    max_iter = 1000
    
    # Initialize
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
    else:
        h = np.zeros(n_points)
        midpoint = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - 2 * abs(i - midpoint) / n_points)
        h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximations
        # For now: use simple finite differences in projected gradient descent
        # But prefer the linearized Frank-Wolfe approach
        
        # Compute the shift correlation (this gives us overlapping terms)
        # For a method that explicitly handles changes:
        h_copy = h.copy()
        current_c5 = evaluate_c5(h_copy, dx)
        
        # Simple gradient-based update - approximate since we're not doing
        # the exact gradient derivation (non-sparse case)
        # Instead, we do a Frank-Wolfe inspired projection approach
        
        # Direct update using a Frank-Wolfe-like projection:
        # Grab the direction that most improves the worst overlap
        
        # Compute overlap for all shifts
        overlaps = correlate(h, 1 - h, mode="full")
        max_shift = len(overlaps) - n_points
        k_idx = np.argmax(overlaps)
        
        # Get that shift index
        k = k_idx - (n_points - 1)
        
        # We'll generate a direct update direction from gradient estimation
        # Compute exact gradient to maximum overlap component
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                gradient[i] += 1 - 2 * h[i+k]  # We assume h is fixed in context
            gradient[i] = 1 - 2 * h[i]
        
        # Use a line search approach with gradient descent
        # Start with an approximate gradient
        grad = np.zeros(n_points)
        # Around the max shift we're focusing on
        point = 0
        if k_idx >= n_points - 1:
            point = k_idx - (n_points - 1)
        else:
            point = 0
        
        # Get full shift envelope
        B = np.zeros(n_points)
        for i in range(n_points):
            # ليلا تحسين تطبيق تدرج الدالة
            B[i] = 1 - 2 * h[i]
        
        # Update using projected gradient:
        # We want to go in the direction that reduces C5
        # Thus direction is -B as the gradient is approximately -B 
        
        alpha = 0.01
        h_new = h - alpha * B
        h_new = project_to_feasible(h_new, target_sum)
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

        if iter_num % 3 == 0:  # Every few iterations, refine
            alpha_start = 0.01

                h_trial = h - test_alpha * B
                h_trial = project_to_feasible(h_trial, target_sum)

        # Early stopping if we're close enough
        if best_c5 <= 0.3807:

    # Re-evaluate final best
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
