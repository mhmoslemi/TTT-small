import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 80
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordiante-wise optimization with normalization
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise update
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute impact of changing h[i]
            h_orig = h[i]
            h_temp = h.copy()
            
            # Try a small step
            step = 0.01
            h_temp[i] = min(1.0, max(0.0, h_orig + step))
            
            # Normalize
            sum_h = np.sum(h_temp)
            h_temp = h_temp * (n_points / 2) / sum_h
            
            c5_temp = compute_c5(h_temp)
            if c5_temp < current_c5:
                h = h_temp.copy()
                continue
                
            # Try negative step
            h_temp[i] = min(1.0, max(0.0, h_orig - step))
            sum_h = np.sum(h_temp)
            h_temp = h_temp * (n_points / 2) / sum_h
            
            c5_temp = compute_c5(h_temp)
            if c5_temp < current_c5:
                h = h_temp.copy()
                continue
                
            # No improvement, restore original
            h[i] = h_orig
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
