import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 75
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to triangular construction
        h = np.zeros(n_points)
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0, 1 - abs(i - mid) / (n_points / 2))
        h = h * (n_points / 2) / np.sum(h)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Augmented Lagrangian parameters
    lambda_ = 0.0
    mu = 1.0
    nu = 1.0
    
    max_iter = 20000
    adaptive_step = 0.002
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient and current value
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx] * dx
        
        # Gradient approximation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - max_overlap) / eps
        
        # Update step
        h_new = h - adaptive_step * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check constraint fulfillment and update Lagrange multipliers
        sum_violation = np.sum(h_new) - n_points / 2
        h_new = h_new - lambda_ * 0.01 * sum_violation / n_points
        
        # Update Lagrange multipliers
        lambda_ += 0.01 * sum_violation
        
        h = h_new
        
        # Adaptive step size adjustment
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            adaptive_step = min(adaptive_step * 1.05, 0.02)
        else:
            adaptive_step *= 0.95
            
        # Exploratory perturbations
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
