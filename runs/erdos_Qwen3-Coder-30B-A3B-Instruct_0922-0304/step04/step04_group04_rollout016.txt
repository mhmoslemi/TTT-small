import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Define Haar wavelet basis (simplified version for computational efficiency)
    def haar_wavelet(j, k, x):
        # Haar wavelet at scale j, position k
        if j == 0:
            if 0 <= x < 1:
                return 1
            else:
                return 0
        else:
            t = 2**j * x - k
            if 0 <= t < 0.5:
                return 1
            elif 0.5 <= t < 1:
                return -1
            else:
                return 0

    # Generate Haar wavelet basis up to scale 5 (for 32 basis functions)
    basis_functions = []
    num_basis = 32
    
    for j in range(6):
        for k in range(2**j):
            if j == 0:
                basis_functions.append(lambda x, j=j, k=k: 1 if 0 <= x < 1 else 0)
            else:
                basis_functions.append(lambda x, j=j, k=k: haar_wavelet(j, k, x))
    
    # Truncate to desired basis size
    basis_functions = basis_functions[:num_basis]
    
    # Create matrix of basis evaluations on grid points
    X = np.zeros((n_points, num_basis))
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    for i in range(n_points):
        for j in range(num_basis):
            X[i, j] = basis_functions[j](x_vals[i])
    
    # Initialize with parent construction if available
    h_initial = np.ones(n_points)
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    
    # Convert to wavelet space
    c_initial = np.linalg.lstsq(X, h_initial, rcond=None)[0]
    
    # Apply convex optimization with constraints
    c = Variable(num_basis)
    
    # Integral constraint
    integral_constr = X.T @ c == 1
    
    # Relaxation of h(x) in [0, 1]
    h_bound_constr = [c >= 0, c <= 1]
    
    # Create McCormick envelopes for products c_i * c_j 
    z_vars = []
    z_constrs = []
    for i in range(num_basis):
        for j in range(i, num_basis):
            z = Variable()
            z_vars.append(z)
            # McCormick envelopes: z >= c_i + c_j - 1, z <= c_i, z <= c_j
            z_constrs.append(z >= c[i] + c[j] - 1)
            z_constrs.append(z <= c[i])
            z_constrs.append(z <= c[j])
    
    # Observe that we can easier handle the maximum overlap integral by reparameterizing
    # We'll reformulate the problem using the dual approach and optimize iteratively
    
    # Objective: Minimize the maximum of overlap integrals
    # For reasonably sized problems, we can solve it approximately
    
    # Use a tailored quasi-Newton method for this convex problem
    # Initialize with parent and fine-tune via iterative scheme
    
    def objective_func(coeff):
        h = X @ coeff
        h = np.clip(h, 0, 1)
        # Ensure constraint
        integral_val = np.sum(h) * dx
        h = h * (1 / integral_val)
        return evaluate_c5(h, dx)
    
    # Optimize using scipy to avoid expensive CVXPY solves
    from scipy.optimize import minimize
    
    bounds = [(0, 1) for _ in range(num_basis)]
    result = minimize(objective_func, 
                      c_initial, 
                      bounds=bounds, 
                      method='L-BFGS-B',
                      options={'maxiter': 3000})
    
    if not result.success:
        c_opt = c_initial
    else:
        c_opt = result.x
    
    # Back to h(x)
    h = X @ c_opt
    h = np.clip(h, 0, 1)
    integral_val = np.sum(h) * dx
    h = h * (1 / integral_val)
    
    # Recomputing C5
    final_c5 = evaluate_c5(h, dx)
    
    # Final sanity checks
    integral_check = np.sum(h) * dx
    assert abs(integral_check - 1.0) < 1e-4, "Integral constraint violated"
    assert np.all(h >= 0) and np.all(h <= 1), "Bounds violated"
    
    return (h.tolist(), final_c5, n_points)
