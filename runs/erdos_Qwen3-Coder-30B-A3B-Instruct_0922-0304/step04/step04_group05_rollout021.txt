import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization with alternating updates
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Fixed support optimization
        # Use gradient descent to adjust h values
        h_old = h.copy()
        
        # Compute gradient approximation
        # Simple finite difference approach
        epsilon = 1e-4
        grad = np.zeros(n_points)
        
        # Compute gradient by finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1, h[i] + epsilon)
            h_minus[i] = max(0, h[i] - epsilon)
            
            # If we're at the bounds, skip
            if h_plus[i] == h[i] or h_minus[i] == h[i]:
                continue
                
            c_plus = evaluate_c5(h_plus, dx)
            c_minus = evaluate_c5(h_minus, dx)
            grad[i] = (c_plus - c_minus) / (2 * epsilon)
            
        # Update h using gradient descent with adaptive step
        learning_rate = 0.005
        h_new = h - learning_rate * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
