import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple symmetric two-interval construction
        h = np.zeros(n_points)
        w = 0.5  # width of each interval
        a = 0    # starting point of first interval
        d = 0.5  # distance between intervals
        i0 = int(a / dx)
        i1 = int((a + w) / dx)
        i2 = int((a + w + d) / dx)
        i3 = int((a + w + d + w) / dx)
        h[i0:i1+1] = 1
        h[i2:i3+1] = 1
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Grid search on interval spacing
    d_candidates = np.arange(0.0, 1.01, 0.01)
    
    for d in d_candidates:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create the step function with two intervals
        h_temp = np.zeros(n_points)
        w = 0.5
        a = 0
        i0 = int(a / dx)
        i1 = int((a + w) / dx)
        i2 = int((a + w + d) / dx)
        i3 = int((a + w + d + w) / dx)
        h_temp[i0:i1+1] = 1
        h_temp[i2:i3+1] = 1
        # Normalize
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        c5 = evaluate_c5(h_temp, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
    
    # Refine around best d if time allows
    if time.time() - start_time < budget_s - 2:
        # Find the best d from previous search
        best_d = 0
        for d in d_candidates:
            h_temp = np.zeros(n_points)
            w = 0.5
            a = 0
            i0 = int(a / dx)
            i1 = int((a + w) / dx)
            i2 = int((a + w + d) / dx)
            i3 = int((a + w + d + w) / dx)
            h_temp[i0:i1+1] = 1
            h_temp[i2:i3+1] = 1
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            c5 = evaluate_c5(h_temp, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_d = d
        
        # Do finer search around the best d
        if best_d > 0 and best_d < 1:
            d_candidates_fine = np.arange(max(0, best_d - 0.05), 
                                          min(1, best_d + 0.05), 0.001)
            for d in d_candidates_fine:
                h_temp = np.zeros(n_points)
                w = 0.5
                a = 0
                i0 = int(a / dx)
                i1 = int((a + w) / dx)
                i2 = int((a + w + d) / dx)
                i3 = int((a + w + d + w) / dx)
                h_temp[i0:i1+1] = 1
                h_temp[i2:i3+1] = 1
                h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
                c5 = evaluate_c5(h_temp, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_temp.copy()
    
    return (best_h.tolist(), best_c5, n_points)
