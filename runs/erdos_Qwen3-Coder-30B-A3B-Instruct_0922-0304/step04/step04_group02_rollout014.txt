import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolutionary Strategy parameters
    population_size = 100
    elite_size = population_size // 5
    mutation_strength = 0.1
    max_generations = 1000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        new_h = h + np.random.normal(0, mutation_strength, n_points)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # Track improvement
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    # Evolutionary loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            fit = evaluate_c5(individual, dx)
            fitness.append(fit)
        
        # Sort by fitness (lower is better)
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt mutation strength
        if no_improvement_count > 50:
            mutation_strength *= 1.2
            no_improvement_count = 0
        elif no_improvement_count < 10 and mutation_strength > 0.01:
            mutation_strength *= 0.9
            
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = population[:elite_size]  # Keep elites
        
        # Generate offspring through mutation
        while len(new_population) < population_size:
            # Select two parents using tournament selection
            parent1 = population[np.random.randint(0, elite_size)]
            parent2 = population[np.random.randint(0, elite_size)]
            
            # Crossover (uniform)
            crossover_mask = np.random.rand(n_points) > 0.5
            child = np.where(crossover_mask, parent1, parent2)
            
            # Mutation
            mutation_mask = np.random.rand(n_points) < 0.3
            noise = np.random.normal(0, mutation_strength, n_points)
            child = child + noise * mutation_mask
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:]  # Copy the list
    
    # SLSQP refinement on best solution
    if time.time() - start_time < budget_s * 0.95:
        from scipy.optimize import minimize
        
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        def objective(h):
            return evaluate_c5(h, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                         constraints=cons, options={'maxiter': 500, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
