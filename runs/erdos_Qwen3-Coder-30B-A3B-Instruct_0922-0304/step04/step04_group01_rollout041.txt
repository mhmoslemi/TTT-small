import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Create symmetric triangular window
        h_init = np.ones(n_points)
        mid = n_points // 2
        h_init[:mid] = np.linspace(0, 1, mid)
        h_init[mid:] = np.linspace(1, 0, n_points - mid)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize particles around the initial solution
    for i in range(n_particles):
        if i == 0:
            h = h_init.copy()
        else:
            # Add small random perturbations
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
    
    # PSO parameters
    w = 0.9  # Inertia weight
    c1 = 2.0  # Cognitive parameter
    c2 = 2.0  # Social parameter
    max_iter = 2000
    
    # Initialize personal bests and global best
    personal_bests = [particle.copy() for particle in particles]
    personal_best_c5 = [evaluate_c5(particle, dx) for particle in particles]
    
    gbest_idx = np.argmin(personal_best_c5)
    gbest = personal_bests[gbest_idx].copy()
    gbest_c5 = personal_best_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_bests[i] - particles[i])
                           + c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            c5 = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_bests[i] = particles[i].copy()
                personal_best_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
                    
                    if gbest_c5 < best_c5:
                        best_c5 = gbest_c5
                        best_h = gbest.copy()
            
            # Early stopping
            if gbest_c5 <= 0.3807:
                break
                
        # Periodically reinitialize some particles
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                # Reinitialize with small perturbation of gbest
                particles[idx] = gbest + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
    
    # Ensure final solution is valid
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
