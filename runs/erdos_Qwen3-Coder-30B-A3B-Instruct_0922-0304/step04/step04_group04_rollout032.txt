import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    basis_size = 20
    
    # Generate basis functions: sin and cos terms
    x = np.linspace(0, 2, n_points)
    basis = []
    for i in range(1, basis_size + 1):
        if i % 2 == 1:
            basis.append(np.sin(np.pi * i * x / 2))
        else:
            basis.append(np.cos(np.pi * i * x / 2))
    basis = np.array(basis)
    
    # Precompute integrals of basis functions
    int_basis = np.sum(basis, axis=1) * dx
    
    # Initial guess from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Compute Fourier coefficients
        c = (basis.T @ h_init) * dx
        # Adjust to satisfy constraint
        c0 = np.sum(c * int_basis)
        c[0] = c[0] - (c0 - 1) / int_basis[0] if abs(int_basis[0]) > 1e-12 else 0
    else:
        # Simple initialization
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        c = (basis.T @ h_init) * dx
        c[0] = c[0] - (np.sum(c * int_basis) - 1) / int_basis[0] if abs(int_basis[0]) > 1e-12 else 0
    
    # Compute kernel matrix
    K = np.zeros((basis_size, basis_size, 2 * n_points - 1))
    for lag in range(2 * n_points - 1):
        for i in range(basis_size):
            for j in range(basis_size):
                # Compute integral of basis_i(x) * basis_j(x+lag)
                lag_idx = lag
                if lag_idx >= n_points:
                    lag_idx = 2 * n_points - 2 - lag_idx
                val = np.sum(basis[i] * np.roll(basis[j], -lag_idx)) * dx
                K[i, j, lag] = val
                
    # Optimize
    # Decision variables
    c_vars = Variable(basis_size)
    
    # Constraints
    constraints = []
    
    # Objective: minimize max overlap
    max_overlap = Variable()
    constraints.append(max_overlap >= 0)
    
    abs_sum = 0
    for lag in range(2 * n_points - 1):
        temp_expr = 0
        for i in range(basis_size):
            for j in range(basis_size):
                temp_expr += c_vars[i] * c_vars[j] * K[i, j, lag]
        constraints.append(temp_expr <= max_overlap)
        constraints.append(-temp_expr <= max_overlap)
    
    # Convex formulation setup
    # Enforce ∫h dx = 1:
    constraints.append(sum(c_vars[i] * int_basis[i] for i in range(basis_size)) == 1)
    
    # Ensure h(x) in [0,1] by bounds on c
    for i in range(basis_size):
        constraints.append(c_vars[i] >= 0)
    
    # Implicit constraint that h(x) = c0 + sum(c_i * basis_i(x)) <= 1
    # And h(x) >= 0, which should be enforced via c_i >= 0 
    # The rest of the constraint is already imposed via optimization
    # We can use a simple approach to verify final h
    objective = Minimize(max_overlap)
    prob = Problem(objective, constraints)
    
    # Use SCS solver with high iteration limit
    try:
        prob.solve(solver=SCS, max_iters=10000, verbose=False)
        if prob.status in [OPTIMAL, APPROX_OPTIMAL]:
            c_sol = c_vars.value
        else:
            # Fall-back if optimization fails
            c_sol = c
    except:
        c_sol = c
    
    # Construct h using the solution
    h_sol = basis.T @ c_sol
    h_sol = h_sol + c_sol[0]  # Add constant offset
    # Normalize for integral constraint
    integral = np.sum(h_sol) * dx
    h_sol = h_sol * (n_points / 2) / integral
    h_sol = np.clip(h_sol, 0, 1)
    
    # Final check
    c5_bound = evaluate_c5(h_sol, dx)
    
    return (h_sol.tolist(), c5_bound, n_points)
