import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral for all shifts
    n = len(h)
    max_overlap = 0
    for k in range(2 * n - 1):
        overlap = 0
        for i in range(n):
            j = i + k
            if j < n:
                overlap += h[i] * (1 - h[j]) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Create a reasonable starting guess: piecewise constant function
        h0 = np.zeros(n_points)
        # Create two intervals of height 1 with some separation
        mid = n_points // 2
        left_width = n_points // 4
        right_width = n_points // 4
        h0[mid-left_width:mid] = 1
        h0[mid:mid+right_width] = 1
        h0 = h0 * (n_points / 2) / np.sum(h0)
        
    # Try to improve the initial solution using gradient-based method
    def objective(x):
        # x contains n_points values for h
        h = np.clip(x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Optimization using Nelder-Mead with multiple random restarts
    n_restarts = 10
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Start with slightly perturbed version of h0
        x0 = h0 + np.random.normal(0, 0.01, n_points)
        x0 = np.clip(x0, 0, 1)
        x0 = x0 * (n_points / 2) / np.sum(x0)
        
        try:
            res = minimize(objective, x0, method='Nelder-Mead', 
                          options={'maxiter': 1000, 'disp': False})
            if res.success:
                h_candidate = np.clip(res.x, 0, 1)
                h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
                c5_candidate = evaluate_c5(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()
        except:
            pass
        
        # Occasionally do a more global search by adding noise
        if restart % 3 == 0:
            h0 = np.random.rand(n_points) * 0.5  # Random distribution
            h0 = h0 * (n_points / 2) / np.sum(h0)
            
    # If no improvement was found, return the initial solution
    if best_h is None:
        best_h = h0
        
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
