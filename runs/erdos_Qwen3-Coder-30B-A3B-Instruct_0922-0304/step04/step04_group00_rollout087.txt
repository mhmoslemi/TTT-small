import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1e-4
    alpha = 0.99
    T = T0
    max_iter = 5000
    step_size = 0.05
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Save current state
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate neighbor by perturbing a random index
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        delta = np.random.uniform(-step_size, step_size)
        h_new[idx] += delta
        h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Renormalize to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject the new solution
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            
        # Cool down
        T *= alpha
        step_size *= alpha
        
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
