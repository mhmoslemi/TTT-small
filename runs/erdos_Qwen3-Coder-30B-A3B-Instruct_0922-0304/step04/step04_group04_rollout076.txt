import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    n_basis = 20
    
    # Create Fourier basis functions
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Basis functions: sine and cosine
    def fourier_basis(k):
        if k == 0:
            return np.ones(n_points)
        elif k % 2 == 1:
            return np.sin(np.pi * k * x)
        else:
            return np.cos(np.pi * k * x)
    
    # Precompute basis integrals
    basis_integrals = []
    for k in range(n_basis):
        basis_func = fourier_basis(k)
        integral = np.sum(basis_func) * dx
        basis_integrals.append(integral)
    
    # Project initial solution onto basis
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
    
    # Normalize initial solution
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize coefficients using FFT
    coeffs = np.zeros(n_basis)
    for k in range(n_basis):
        basis_func = fourier_basis(k)
        coeffs[k] = np.sum(h_init * basis_func) * dx
    
    # Set up convex optimization problem
    # Variables: Fourier coefficients
    c = cp.Variable(n_basis)
    
    # Constraints
    # Integral constraint
    integral_constraint = cp.sum(cp.multiply(c, basis_integrals)) == 1
    
    # Bounds and McCormick envelopes for h(x) = sum(c_k * phi_k(x))
    # We want to compute max_k int h(x)(1-h(x+k)) dx
    # For each k, we compute:
    # int h(x) * (1 - h(x+k)) dx = 
    # int sum(c_i*phi_i(x)) * (1 - sum(c_j*phi_j(x+k))) dx
    # = sum(c_i*c_j)* int phi_i(x) * (1 - phi_j(x+k)) dx
    
    # Set up the maximum over all lags k
    # For efficiency, we avoid storing all terms and instead consider all necessary convolutions
    max_overlap = cp.Variable()
    
    # Compute overlap efficiently using FFT (this is not standard convex optimization
    # but we approximate it with convex constraints)
    
    # Use McCormick envelopes to represent h(x)*h(x+k) products:
    # Let z_i_j_l = c_i * c_j where i+j=k (mod 2n_points)
    # This would be very large, so let's relax to smaller basis
    
    # Simplified approach: optimize over coefficients directly using local search
    # since convex optimization won't work easily for this non-convex problem
    
    # Use L-BFGS-B on the function to refine
    
    def evaluate_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        return max_overlap
    
    # Initialize with best from previous code
    current_h = h_init.copy()
    current_c5 = evaluate_c5(current_h)
    
    # Optimization using gradient-based method
    from scipy.optimize import minimize
    
    # Define objective for optimization
    def obj_func(coeffs_flat):
        # Reconstruct h(x) from coefficients
        h = np.zeros(n_points)
        for k in range(n_basis):
            basis_func = fourier_basis(k)
            h += coeffs_flat[k] * basis_func
            
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        # Enforce integral = 1
        h = h * (n_points / 2) / np.sum(h)
        
        return evaluate_c5(h)
    
    def constraint_func(coeffs_flat):
        # Integral constraint
        integral = 0
        for k in range(n_basis):
            integral += coeffs_flat[k] * basis_integrals[k]
        return integral - 1
    
    def grad_func(coeffs_flat):
        # Approximate gradient
        eps = 1e-6
        grad = np.zeros_like(coeffs_flat)
        for i in range(len(coeffs_flat)):
            coeffs_plus = coeffs_flat.copy()
            coeffs_minus = coeffs_flat.copy()
            coeffs_plus[i] += eps
            coeffs_minus[i] -= eps
            grad[i] = (obj_func(coeffs_plus) - obj_func(coeffs_minus)) / (2 * eps)
        return grad
    
    # Make bounds for coefficients using the idea that h(x) must be in [0,1]
    # This leads to restrictions on the coefficient bounds
    lb = np.full(n_basis, -100)
    ub = np.full(n_basis, 100)
    
    # Run optimization
    max_iter = 5000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different starting points around current solution
        starts = [current_h.copy()]
        for _ in range(3):
            perturbation = np.random.normal(0, 0.03, n_points)
            h_pert = current_h + perturbation
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            starts.append(h_pert)
        
        best_start = starts[0]
        best_c5_val = current_c5
        for start_h in starts:
            # Optimization step: local search
            # We actually simulate the optimization around this point in coefficient space
            
            # Just truncate the existing solution and try that for the local search
            coeffs_try = np.zeros(n_basis)
            for k in range(n_basis):
                basis_func = fourier_basis(k)
                coeffs_try[k] = np.sum(start_h * basis_func) * dx
            
            # Use minimize on this truncated space
            res = minimize(obj_func, coeffs_try, method='L-BFGS-B', bounds=list(zip(lb, ub)), 
                          options={'maxiter': 50}, callback=lambda x: None)
            new_coeff = res.x
            
            # Validate this new state
            h_new = np.zeros(n_points)
            for k in range(n_basis):
                basis_func = fourier_basis(k)
                h_new += new_coeff[k] * basis_func
            h_new = np.clip(h_new, 0, 1) 
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_c5(h_new)
            
            if new_c5 < best_c5_val:
                best_c5_val = new_c5
                current_c5 = best_c5_val
                coeffs = new_coeff.copy()
                current_h = h_new.copy()
            
        # Check if we achieved target
        if current_c5 <= 0.3807:
            break
             
    # Final conversion to pointwise values from coefficients
    h_final = np.zeros(n_points)
    for k in range(n_basis):
        basis_func = fourier_basis(k)
        h_final += coeffs[k] * basis_func
    
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), current_c5, n_points)
