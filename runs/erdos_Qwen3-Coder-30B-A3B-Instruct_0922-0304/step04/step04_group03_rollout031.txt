import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, dx, target_sum):
    """Project h to satisfy the integral constraint"""
    current_sum = np.sum(h) * dx
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a symmetric tent-shaped function
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            h[i] = max(0, 1 - abs(x - 1))
    
    # Normalize to meet sum constraint
    h = project_h(h, dx, target_sum)
    
    # Gradient descent parameters
    max_iter = 10000
    learning_rate = 0.01
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps and find maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_k = max_idx - (n_points - 1)  # Convert index to shift
        
        # Compute gradient for the worst-case overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # dO_k/dh_i for shift k
            j_plus = (i + max_k) % n_points
            j_minus = (i - max_k) % n_points
            gradient[i] = (1 - h[j_plus]) - h[j_minus]
        
        # Update step
        h_new = h - learning_rate * gradient
        
        # Project to feasible region
        h_new = project_h(h_new, dx, target_sum)
        
        # Check for improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        h = h_new.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Adaptive learning rate
        if iter_num > 100 and iter_num % 100 == 0:
            learning_rate *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
