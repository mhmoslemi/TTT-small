import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def solve_qp(H, c, A, b, delta_low, delta_high):
    """
    Solve quadratic program using cvxpy.
    min 0.5 * delta^T * H * delta + c^T * delta
    subject to A * delta = b
           delta_low <= delta <= delta_high
    """
    n = len(c)
    delta = cp.Variable(n)
    
    # QP objective
    obj = cp.Minimize(0.5 * cp.quad_form(delta, H) + c.T @ delta)
    
    # Constraints
    constraints = [A @ delta == b]
    constraints += [delta >= delta_low]
    constraints += [delta <= delta_high]
    
    # Problem
    prob = cp.Problem(obj, constraints)
    
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return delta.value
        else:
            return None
    except:
        return None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create triangular function
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1 - 2 * abs(i - n_points/2) / n_points
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # QP parameters
    max_iter = 500
    reg = 1e-5
    
    # Constraint matrix for sum = 0
    A = np.ones((1, n_points))
    b = np.array([0.0])
    
    # Low/high bounds for delta (relative to h limits)
    delta_low = -h
    delta_high = 1 - h
    
    # Main SCP loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values for current h
        corr_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr_full)
        max_val = corr_full[max_idx]
        
        # Compute gradient and Hessian
        grad = np.zeros(n_points)
        h_shifted = np.roll(h, 1)
        grad = dx * (1 - 2 * h - h_shifted)
        
        # Hessian - approximate second derivative
        hessian = np.zeros((n_points, n_points))
        for i in range(n_points):
            for j in range(n_points):
                if i == j:
                    hessian[i, j] = -2 * dx
                elif abs(i - j) == 1:
                    hessian[i, j] = dx
                    
        hessian += reg * np.eye(n_points)
        
        # Solve QP
        delta = solve_qp(hessian, grad, A, b, delta_low, delta_high)
        
        if delta is None:
            # If optimization fails, try increasing regularization
            reg *= 10
            hessian = np.zeros((n_points, n_points))
            for i in range(n_points):
                for j in range(n_points):
                    if i == j:
                        hessian[i, j] = -2 * dx
                    elif abs(i - j) == 1:
                        hessian[i, j] = dx
            hessian += reg * np.eye(n_points)
            delta = solve_qp(hessian, grad, A, b, delta_low, delta_high)
            if delta is None:
                # Resort to random perturbation
                delta = np.random.normal(0, 0.01, n_points)
                delta = delta * (0.5/n_points)
        
        if delta is not None:
            # Line search
            alpha = 1.0
            step_size = 0.5
            while alpha > 0.01:
                new_h = h + alpha * delta
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                
                new_c5 = evaluate_c5(new_h, dx)
                if new_c5 < best_c5:
                    h = new_h.copy()
                    best_c5 = new_c5
                    break
                alpha *= step_size
            else:
                # Accept the change but do not update best
                h = np.clip(h + delta, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Early termination
        if best_c5 <= 0.38070:
            break
        
        # Update bounds for the next iteration
        delta_low = -h
        delta_high = 1 - h
        
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
