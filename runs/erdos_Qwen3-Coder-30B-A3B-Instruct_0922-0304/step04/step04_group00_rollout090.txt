import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting plane approach
    constraints_list = []
    epsilon = 0.001
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst shift
        overlap = correlate(h, 1 - h, mode="full")
        max_shift = np.argmax(overlap)
        max_overlap = overlap[max_shift] * dx
        
        # Add constraint
        if max_shift < len(h):
            # Left shift case: h[i] * (1 - h[i + shift])
            shift = max_shift
            # Constraint: sum h[i] * (1 - h[i + shift]) <= max_overlap - epsilon
            constraint_expr = 0
            for i in range(len(h) - shift):
                constraint_expr += h[i] * (1 - h[i + shift])
            constraint_expr *= dx
        else:
            # Right shift case
            shift = max_shift - len(h) + 1
            constraint_expr = 0
            for i in range(shift, len(h)):
                constraint_expr += h[i] * (1 - h[i - shift])
            constraint_expr *= dx
            
        if iteration % 5 == 0:
            constraints_list.append(constraint_expr <= max_overlap - epsilon)
            
        # Convex optimization solver
        try:
            # Create variables
            h_vars = cp.Variable(n_points)
            
            # Create the objective (minimize C5 approximately)
            # We'll pretend h is fixed and update it based on constraints
            # Our goal is just to make h satisfy the constraints and reduce overlap
            
            # Constraints
            constraints = [h_vars[i] >= 0 for i in range(n_points)]
            constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
            constraints.append(cp.sum(h_vars) == n_points / 2)
            for constraint in constraints_list:
                constraints.append(constraint)
            
            # This is tricky to make convexly minimizing max overlap
            # So instead we solve an inner optimization problem
            
            # Perform Douglas-Rachford splitting or basic projection
            # For now update with alternating projection
            new_h = h.copy()
            for _ in range(5):
                # Project to sum constraint
                total = np.sum(new_h)
                new_h = new_h * (n_points / 2) / total
                
                # Project to bounds
                new_h = np.clip(new_h, 0, 1)
                
                # Recompute and update
                if iteration % 10 == 0:
                    break  # Avoids too for training
            
        except Exception:
            pass
            
        # Make small stochastic updates to escape local optima  
        if iteration % 30 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Vanilla update
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
