import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a symmetric two-interval construction
        h = np.zeros(n_points)
        width = 0.5  # Each interval width
        i0 = 0
        i1 = int(width / dx)
        # Try different separations
        best_c5 = float('inf')
        best_h = None
        
        d_candidates = np.arange(0.0, 1.0 + 0.001, 0.01)
        for d in d_candidates:
            if 0.5 + d > 2.0:
                continue
            i2 = int((0.5 + d) / dx)
            i3 = int((1.0 + d) / dx)
            if i2 <= i1:
                continue
            # Create h
            h_temp = np.zeros(n_points)
            h_temp[i0:i1+1] = 1.0
            h_temp[i2:i3+1] = 1.0
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            c5 = evaluate_c5(h_temp, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
                
        h = best_h if best_h is not None else h
        h = np.clip(h, 0, 1)
    
    # Refine with local search
    max_iter = 1000
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Take a small random perturbation
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        c5_new = evaluate_c5(h_perturbed, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_perturbed.copy()
            h = h_perturbed.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
