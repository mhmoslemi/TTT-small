import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()

    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coordinate-wise optimization
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Try updating each coordinate
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:

            # Store current h
            h_old = h.copy()
            
            # Try different values for h[i]

                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                h = np.clip(h, 0, 1)
                # Check if normalization preserved sum
                h_sum = np.sum(h)
                if abs(h_sum - n_points / 2) > 1e-10:
                    h = h * (n_points / 2) / h_sum

                c5 = compute_c5(h, dx)

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
