import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h):
    """Compute the maximum overlap integral for given step function h."""
    dx = 2.0 / len(h)
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 50
    max_iter = 10000
    
    # Initialize particles around the initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    else:
        # Fallback: create a simple initial construction
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Initialize population
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for i in range(population_size):
        # Create a perturbed variant of initial_h
        h = initial_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.01, 0.01, n_points))
        p_best.append(h.copy())
        p_best_vals.append(compute_c5(h))
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Main PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(population_size):
            h = particles[i]
            v = velocities[i]
            p_best_h = p_best[i]
            
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            v = w * v + c1 * r1 * (p_best_h - h) + c2 * r2 * (g_best - h)
            
            # Clamp velocity to [-0.01, 0.01]
            v = np.clip(v, -0.01, 0.01)
            
            # Update position
            h = h + v
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate new position
            current_c5 = compute_c5(h)
            
            # Update personal best
            if current_c5 < p_best_vals[i]:
                p_best[i] = h.copy()
                p_best_vals[i] = current_c5
                
                # Update global best if needed
                if current_c5 < g_best_val:
                    g_best = h.copy()
                    g_best_val = current_c5
                    
                    # Early exit if target is achieved
                    if g_best_val <= 0.3807:
                        break
                        
            # Update particle
            particles[i] = h.copy()
            velocities[i] = v.copy()
    
    return (g_best.tolist(), g_best_val, n_points)
