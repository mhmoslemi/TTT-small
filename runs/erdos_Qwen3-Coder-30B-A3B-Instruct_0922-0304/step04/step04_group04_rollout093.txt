import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Fallback: simple step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Optimization using CVXPY with Haar wavelets
    # Wavelet-based approach using built-in Haar basis
    # We'll formulate a convex relaxation

    # For improved performance, use gradient-based method with constraints
    # Start with theinitial point
    h_opt = h_init.copy()
    best_c5 = evaluate_c5(h_opt, dx)
    
    # Try different optimization strategies:
    # 1. Gradient-based local search
    # 2. Randomized perturbations to escape local optima
    # 3. Basin-hopping style approach
    
    # Local optimization using scipy
    from scipy.optimize import minimize
    
    def objective(h_vec):
        return evaluate_c5(h_vec, dx)
    
    def constraint_integral(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    cons = {'type': 'eq', 'fun': constraint_integral}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Try multiple random restarts
    n_restarts = 20
    n_local_iter = 300
    
    for restart in range(n_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random perturbation around current best
        h_perturbed = h_opt + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Local minimization
        result = minimize(objective, h_perturbed, 
                         method='SLSQP',
                         bounds=bounds,
                         constraints=cons,
                         options={'maxiter': n_local_iter, 'ftol': 1e-6})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                h_opt = h_candidate.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    # Final refinement using a few iterations of particle swarm with projection
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Simple gradient descent style improvement  
    for i in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = np.zeros(n_points)
        h = h_opt.copy()
        
        # Simple finite difference gradient approximation for the overlap integral
        # This is a 1D gradient approximation
        eps = 1e-4
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            
            grad[j] = (evaluate_c5(h_plus, dx) - evaluate_c5(h_minus, dx)) / (2 * eps)
            
        # Update step
        step_size = 0.001
        h_opt = h_opt - step_size * grad
        h_opt = project_h(h_opt)
        
        c5_current = evaluate_c5(h_opt, dx)
        if c5_current < best_c5:
            best_c5 = c5_current
    
    # Force the final solution to satisfy constraints
    h_opt = project_h(h_opt)
    final_c5 = evaluate_c5(h_opt, dx)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (h_opt.tolist(), best_c5, n_points)
