import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create triangular initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Sequential Convex Programming (SCP) approach
    max_iter = 1000
    alpha = 0.1  # Initial step size
    damp_factor = 0.5
    mu = 1e-3  # Barrier parameter
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # For a smooth approximation, use a finite difference approach
        # Simplified: use current h to estimate gradient direction from local change
        grad = np.zeros(n_points)
        eps = 1e-6
        current_c5 = evaluate_c5(h, dx)
        
        # Numerical gradient approximation
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Renormalize
            c5_plus = evaluate_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Form convex subproblem using Hessian approximation
        # For simplicity, use identity matrix to form convex approximation
        H = np.eye(n_points)
        
        # Add small regularization to ensure stability
        H += 1e-8 * np.eye(n_points)
        
        # Define optimization variable
        delta = cp.Variable(n_points)
        
        # Objective: quadratic approximation
        obj = cp.Minimize(cp.quad_form(delta, H) + grad.T @ delta)
        
        # Constraints
        constraints = [
            cp.sum(delta) == 0,  # Maintain integral constraint
            delta >= -h,  # Ensure lower bound
            delta <= 1 - h,  # Ensure upper bound
        ]
        
        # Create and solve problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.OSQP, verbose=False, 
                      eps_abs=1e-6, eps_rel=1e-6)
            
            if prob.status == cp.OPTIMAL:
                update = delta.value
            else:
                update = np.zeros(n_points)
        except:
            update = np.zeros(n_points)
        
        # Update step
        h_new = h + alpha * update
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # Renormalize
        
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept update if better, else reduce step size
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            alpha = min(alpha * 1.2, 0.5)  # Increase step size if improving
        else:
            alpha *= damp_factor  # Reduce step size if no improvement
            
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
