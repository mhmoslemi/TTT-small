import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    n_points = 80
    dx = 2.0 / n_points

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Coordinate-wise optimization
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:

            # Save current value
            old_h_i = h[i]
            
            # Try to improve h[i] by optimizing its contribution to C5
            # We compute how changing h[i] affects C5 for all possible k values
            # For the current h[i], compute the effect on overlap integral
            
            # Current sum before change
            old_sum = np.sum(h)
            
            # Try a small step (positive and negative) - coordinate-wise optimization
            step_size = 0.01
            
            # Try two steps: increase and decrease
            candidates = []
            for sign in [-1, 1]:
                new_h = h.copy()
                new_h[i] += sign * step_size
                new_h[i] = max(0, min(1, new_h[i]))
                
                # Re-normalize
                new_sum = np.sum(new_h)

                    h = best_candidate_h.copy()
        
        # Periodic re-normalization
        if iteration % 10 == 0:
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
