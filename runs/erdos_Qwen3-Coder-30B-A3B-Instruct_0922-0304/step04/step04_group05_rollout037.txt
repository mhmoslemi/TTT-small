import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 70

    target_sum = n_points / 2

        h = np.clip(h, 0, 1)
        h = project_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = min(1, 2 * abs(i - n_points/2) / n_points)
        h = project_h(h, target_sum)
    
    current_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Frank-Wolfe style optimization
    gamma = 0.1
    prev_c5 = current_c5 + 1
    stagnation_count = 0
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using correlation
        # Gradient ∇C_k ≈ dx * (1 - 2*h - shift_average)
        gradient = np.zeros(n_points)
        
        # For all shifts, we'll approximate the gradient
        # Start with gradient approximation via convolution
        h1 = 1 - h  # We're computing overlap with (1 - h)
        # Perform correlation of h with h1 (which gives us the overlap)  
        # And use it to derive gradient-like updates
        
        # More efficient way to build gradient approx
        grad_temp = np.zeros(n_points)
        for k in range(n_points):
            # This is partial derivative of overlap w.r.t. h[i] 
            # C_k = ∑ h[i] * (1 - h[i+k]) over all i
            # ∂C_k/∂h[i] = 1 - h[i+k] - h[i] + (if shift was from right: an extra contribution)
            # Approximate gradient using current solution
            if k < len(h):
                mean_term = np.mean(h[max(0,k):min(len(h),k+n_points)])
                grad_temp += h1[k:] * (1 if k < n_points else 0)  # approximate
            else:

        # Simplified approach: do direct gradient estimate
        # C_k = ∑_i h[i] * (1 - h[i + k]) for valid i
        # ∂C_k/∂h[i] = 1 - h[i + k] (for valid shifts)
        
        # Compute gradient with sliding windows
        grad = np.zeros(n_points)
        for k in range(n_points):
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
                
        # Approximate by looking at all shifts, add small noise
        grad = np.zeros(n_points)
        for k in range(n_points):
            if k < n_points:
                grad[:n_points-k] += 1 - h[k:]
                
        # We also need to consider term - h[i] from the overlap
        # Gradients: 
        # ∂/∂h_i ∫( h(x) (1 - h(x+k)) )dx 
        # = 1 - h[i+k] - h[i] (when h[i+k] contributes)
        # But we run a scale to prevent gradient-based instability in max

        # Compute update direction (Frank-Wolfe style)
        direction = -gamma * grad
        
        # Apply update and project
        h_candidate = h + direction
        
        # Project back to feasible set
        h_candidate = project_h(h_candidate, target_sum)

        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
