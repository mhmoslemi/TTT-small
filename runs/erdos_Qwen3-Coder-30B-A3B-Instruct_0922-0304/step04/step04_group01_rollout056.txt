import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: polynomial construction
        x = np.linspace(0, 2, n_points)
        h = np.zeros(n_points)
        # Create a piecewise linear function: rise then fall
        mid = n_points // 2
        h[:mid] = x[:mid] / (2 * mid / n_points)
        h[mid:] = 1 - (x[mid:] - 1) / (1 - mid / n_points)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adam style gradient descent with adaptive learning rate
    # For each point we sample derivative via finite differences
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    max_iter = 3000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current function and gradient estimates
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        current_c5 = overlap[max_idx] * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = overlap_plus[np.argmax(overlap_plus)] * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Occasionally perturb
        if iteration % 100 == 0:
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
