import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: balanced step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-start with gradient descent
    n_starts = 50
    n_steps = 200
    
    # Helper function for constraint
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Helper for evaluation
    def objective(h):
        return evaluate_c5(h, dx)
    
    # Constraints for SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Generate random starting points
    random_starts = []
    for _ in range(n_starts - 1):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        random_starts.append(h_rand)
    
    # Add initial point
    random_starts.append(h)
    
    # Multi-start loop
    for i, h_start in enumerate(random_starts):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        try:
            # SLSQP optimization
            result = minimize(objective, h_start, method='SLSQP', bounds=bounds, 
                            constraints=cons, options={'maxiter': n_steps, 'ftol': 1e-9})
            
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_c5(refined_h, dx)
                
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
                    
                if best_c5 <= 0.3807:
                    break
        except:
            pass
    
    # Final refinement
    if time.time() - start_time < budget_s * 0.98:
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, 
                            constraints=cons, options={'maxiter': 100, 'ftol': 1e-10})
            
            if result.success:
                refined_h = result.x
                refined_c5 = evaluate_c5(refined_h, dx)
                
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
