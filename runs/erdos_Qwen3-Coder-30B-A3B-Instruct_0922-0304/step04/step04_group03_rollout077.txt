import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        # Create a triangular shape as initial guess
        h_init = np.zeros(n_points)
        for i in range(n_points):
            h_init[i] = 1.0 - 2.0 * abs(i - n_points/2) / n_points
        h_init = np.clip(h_init, 0, 1)
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Define objective function
    def obj(vars):
        # vars: [a, d] - starting position and distance between intervals
        a, d = vars
        
        # Create step function with two intervals
        h = np.zeros(n_points)
        
        # First interval [a, a+w] where w = 0.5 
        w = 0.5
        
        # Clip a to valid range
        a = np.clip(a, 0, 1 - d - w)
        
        # Set first interval
        start1 = int(a / dx)
        end1 = int((a + w) / dx)
        if end1 > n_points:
            end1 = n_points
        if start1 < n_points:
            h[start1:end1] = 1.0
            
        # Set second interval [a+w+d, a+w+d+w]
        start2 = int((a + w + d) / dx)
        end2 = int((a + w + d + w) / dx)
        if end2 > n_points:
            end2 = n_points
        if start2 < n_points:
            h[start2:end2] = 1.0
        
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate C5
        return evaluate_c5(h, dx)
    
    # Optimization
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Start with initial configuration
    a0 = 0.25
    d0 = 0.25
    try:
        res = minimize(obj, [a0, d0], method='Nelder-Mead', 
                       options={'maxiter': 500, 'xatol': 1e-4, 'fatol': 1e-6})
        if res.success:
            a, d = res.x
            h = np.zeros(n_points)
            w = 0.5
            a = np.clip(a, 0, 1 - d - w)
            start1 = int(a / dx)
            end1 = int((a + w) / dx)
            if end1 > n_points:
                end1 = n_points
            if start1 < n_points:
                h[start1:end1] = 1.0
            start2 = int((a + w + d) / dx)
            end2 = int((a + w + d + w) / dx)
            if end2 > n_points:
                end2 = n_points
            if start2 < n_points:
                h[start2:end2] = 1.0
            h = h * (n_points / 2) / np.sum(h)
            c5 = evaluate_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    except:
        pass
    
    # Grid search in vicinity of good regions
    for a_try in np.linspace(0.1, 0.4, 20):
        for d_try in np.linspace(0.1, 0.5, 20):
            if time.time() - start_time > budget_s - 1:
                break
            try:
                h = np.zeros(n_points)
                w = 0.5
                a_try = np.clip(a_try, 0, 1 - d_try - w)
                start1 = int(a_try / dx)
                end1 = int((a_try + w) / dx)
                if end1 > n_points:
                    end1 = n_points
                if start1 < n_points:
                    h[start1:end1] = 1.0
                start2 = int((a_try + w + d_try) / dx)
                end2 = int((a_try + w + d_try + w) / dx)
                if end2 > n_points:
                    end2 = n_points
                if start2 < n_points:
                    h[start2:end2] = 1.0
                h = h * (n_points / 2) / np.sum(h)
                c5 = evaluate_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except:
                continue
    
    # Fine tune with PSO like approach but simple
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        # Random perturbation
        h_pert = best_h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        c5_pert = evaluate_c5(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
    
    return (best_h.tolist(), best_c5, n_points)
