import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 50
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_ref = np.array(initial_h_values)
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    else:
        # Create symmetric triangular window
        h_ref = np.zeros(n_points)
        mid = n_points // 2
        h_ref[:mid] = np.linspace(0, 1, mid)
        h_ref[mid:] = np.linspace(1, 0, n_points - mid)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    
    # Initialize particles around reference
    for i in range(n_particles):
        particles[i] = h_ref + np.random.normal(0, 0.01, n_points)
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        velocities[i] = np.random.uniform(-0.01, 0.01, n_points)
    
    # PSO parameters
    w = 0.9
    phi_p = 2.0
    phi_g = 2.0
    inertia_damping = 0.99
    
    # Track best positions
    personal_best = particles.copy()
    personal_best_scores = np.array([evaluate_c5(p, dx) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Optimization loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r_p = np.random.random(n_points)
            r_g = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                           phi_p * r_p * (personal_best[i] - particles[i]) + 
                           phi_g * r_g * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new position
            new_score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best_scores[i] = new_score
                personal_best[i] = particles[i].copy()
                
                # Update global best
                if new_score < global_best_score:
                    global_best_score = new_score
                    global_best = particles[i].copy()
        
        # Periodic reinitialization to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            for i in range(n_particles // 10):  # Reinitialize 10% of particles
                idx = np.random.randint(0, n_particles)
                particles[idx] = h_ref + np.random.normal(0, 0.01, n_points)
                particles[idx] = np.clip(particles[idx], 0, 1)
                particles[idx] = particles[idx] * (n_points / 2) / np.sum(particles[idx])
                velocities[idx] = np.random.uniform(-0.01, 0.01, n_points)
                personal_best_scores[idx] = evaluate_c5(particles[idx], dx)
            
            # Decay inertia weight periodically
            w *= inertia_damping
            
        # Early termination
        if global_best_score <= 0.3807:
            break
    
    return (global_best.tolist(), global_best_score, n_points)
