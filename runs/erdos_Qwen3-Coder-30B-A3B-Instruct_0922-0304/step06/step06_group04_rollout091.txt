import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient-based optimization with line search
    max_iter = 1000
    step_size = 0.01
    beta = 0.9
    eps = 1e-3
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        h = best_individual.copy()
        
        # Compute the full correlation for gradient
        corr_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr_full)
        max_val = corr_full[max_idx]
        
        # Construct surrogate loss
        def surrogate_loss(h_test):
            corr_test = correlate(h_test, 1 - h_test, mode="full")
            max_corr = np.max(corr_test)
            # Add a small regularization term to encourage smoothness
            smooth_reg = eps * np.sum(np.diff(h_test)**2)
            return max_corr * dx + smooth_reg
        
        # Estimating gradient via finite differences
        grad = np.zeros_like(h)
        epsilon = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = normalize_h(h_minus, target_sum)
            
            grad[i] = (surrogate_loss(h_plus) - surrogate_loss(h_minus)) / (2 * epsilon)
        
        # Line search for step size
        temp_h = h - step_size * grad
        temp_h = np.clip(temp_h, 0, 1)
        temp_h = normalize_h(temp_h, target_sum)
        
        # Update h
        h = temp_h.copy()
        
        # Evaluate the new solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
        
        # Early exit check
        if best_c5 <= 0.3807:
            break
            
        # Update step size (backtracking)
        if iteration % 10 == 0:
            step_size *= beta
        iteration += 1

    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
