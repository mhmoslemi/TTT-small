import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the maximum overlap integral for the given step function h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the specified target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Probability of using gradient step
    grad_prob = 0.7
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        h = best_h.copy()
        
        # Random perturbation
        if np.random.random() < grad_prob:
            # Gradient direction estimation
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
        else:
            # Random uniform perturbation
            noise = np.random.uniform(-0.1, 0.1, n_points)
            h = h + noise
            
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization using gradient descent with line search
        max_iter = 50
        for i in range(max_iter):
            # Estimate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = normalize_h(h_plus, target_sum)
                
                c5_plus = compute_c5(h_plus, dx)
                grad[j] = (c5_plus - compute_c5(h, dx)) / eps
            
            # Update step
            step_size = 0.01
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            if compute_c5(h_new, dx) < compute_c5(h, dx):
                h = h_new
            else:
                break
                
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
