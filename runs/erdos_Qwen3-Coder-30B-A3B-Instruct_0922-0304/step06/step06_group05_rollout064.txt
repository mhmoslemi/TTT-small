import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Binary search for optimal C5
    c_low, c_high = 0.0, 1.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a gradual refinement approach
    # for efficiency, we'll do direct binary search with fewer iterations
    max_iters = 30
    iter_count = 0
    
    while time.time() - start_time < budget_s - 2 and iter_count < max_iters:
        c_mid = (c_low + c_high) / 2.0
        
        # Constraint: h[i] in [0,1], sum(h) = n_points/2  
        # For each k: sum h[i] * (1 - h[(i+k)%n]) <= c_mid / dx
        # This is non-convex; we use a heuristic evaluation to check if a feasible solution exists
        
        # We start the search from current h and try gradient-based refinement
        local_h = h.copy()
        local_c5 = compute_c5(local_h)
        
        # Try a local optimization step
        for _ in range(200):  # limited steps for speed
            if time.time() - start_time > budget_s - 2:
                break
                
            g = np.zeros(n_points)
            eps_val = 1e-5
            
            # Compute gradient approximation
            for i in range(n_points):
                h_plus = local_h.copy()
                h_minus = local_h.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                ov_plus = correlate(h_plus, 1 - h_plus, mode="full")
                ov_minus = correlate(h_minus, 1 - h_minus, mode="full")
                
                g[i] = (ov_plus.max() - ov_minus.max()) / (2 * eps_val) * dx

            local_h = local_h - 0.005 * g
            local_h = np.clip(local_h, 0, 1)
            local_h = local_h * (n_points / 2) / np.sum(local_h)
            
            # Early stopping if idle
            if np.allclose(local_h, h, atol=1e-6):
                break
                
        new_c5 = compute_c5(local_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = local_h.copy()
        
        # Binary search based on whether mid is feasible
        if new_c5 <= c_mid:
            # Safe: smaller C
            c_high = c_mid
        else:
            # Too high, need larger C
            c_low = c_mid
            
        iter_count += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final refinement to get a stable solution
    final_h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 2:
            break

        g = np.zeros(n_points)
        eps_val = 1e-5
        
        # Central differences
        for i in range(n_points):
            h_plus = final_h.copy()
            h_minus = final_h.copy()
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            ov_plus = correlate(h_plus, 1 - h_plus, mode="full")
            ov_minus = correlate(h_minus, 1 - h_minus, mode="full")
            
            g[i] = (ov_plus.max() - ov_minus.max()) / (2 * eps_val) * dx

        final_h = final_h - 0.002 * g
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
