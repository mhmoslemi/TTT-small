import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / len(h)
    target_sum = len(h) / 2
    
    # Normalize to correct sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Discrete values for h
    discrete_values = [0.0, 0.25, 0.5, 0.75, 1.0]
    
    # Iterative refinement using gradient-based method with discrete steps
    max_iter = 50000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Random perturbation to explore neighborhood
        perturbation = np.random.normal(0, 0.02, len(h))
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        
        # Ensure sum constraint
        new_h = new_h * (target_sum / np.sum(new_h))
        
        # Evaluate the candidate solution
        new_c5 = compute_c5(new_h, dx)
        
        # Accept the candidate if it improves the solution
        if new_c5 < c5:
            h = new_h
        
        # Occasionally project onto discrete values
        if iteration % 5 == 0:
            discrete_h = np.array([discrete_values[np.argmin(np.abs(x - np.array(discrete_values)))] for x in h])
            discrete_h = discrete_h * (target_sum / np.sum(discrete_h))
            discrete_c5 = compute_c5(discrete_h, dx)
            if discrete_c5 < c5:
                h = discrete_h
                
        iteration += 1
    
    # Final refinement pass
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))
