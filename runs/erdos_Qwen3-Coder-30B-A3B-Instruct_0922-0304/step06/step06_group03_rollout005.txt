import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available, otherwise create a pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple step function as initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    # Particle Swarm Optimization (PSO) parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8  # inertia weight
    c1 = c2 = 2.0  # acceleration coefficients
    
    # Initialize particles
    particles = []
    velocities = []
    personal_best = []
    personal_best_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        c5 = compute_c5(particle, dx)
        personal_best.append(particle.copy())
        personal_best_c5.append(c5)
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    best_h = global_best.copy()
    best_c5 = global_best_c5
    
    # PSO iteration loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            p_best = personal_best[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (p_best - particle) + 
                   c2 * r2 * (global_best - particle))
            
            # Update position
            particle = particle + vel
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = vel
            
            # Evaluate fitness
            c5 = compute_c5(particle, dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = particle.copy()
                personal_best_c5[i] = c5
                
                # Update global best if needed
                if c5 < global_best_c5:
                    global_best = particle.copy()
                    global_best_c5 = c5
        
        # Periodic adjustments for diversity
        if iteration % 50 == 0:
            # Apply frequency domain filtering to some particles
            for i in range(n_particles // 2):
                if np.random.rand() < 0.3:  # 30% chance to apply filter
                    particle = particles[i]
                    # FFT transform
                    freq_domain = np.fft.fft(particle)
                    # Apply high-pass filter
                    freq_domain[10:20] = 0  # Suppress low frequencies
                    filtered = np.real(np.fft.ifft(freq_domain))
                    filtered = np.clip(filtered, 0, 1)
                    filtered = normalize_h(filtered, target_sum)
                    particles[i] = filtered
        
        # Gradient refinement
        if iteration % 100 == 0:
            # Refine top particles with local optimization
            top_indices = np.argsort(personal_best_c5)[:10]
            for i in top_indices:
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, personal_best[i], method='SLSQP', 
                                  bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined_h = result.x
                    refined_h = np.clip(refined_h, 0, 1)
                    refined_h = normalize_h(refined_h, target_sum)
                    c5 = compute_c5(refined_h, dx)
                    if c5 < personal_best_c5[i]:
                        personal_best[i] = refined_h
                        personal_best_c5[i] = c5
                        if c5 < global_best_c5:
                            global_best = refined_h
                            global_best_c5 = c5
        
        # Update best solution
        if global_best_c5 < best_c5:
            best_c5 = global_best_c5
            best_h = global_best.copy()
        
        if best_c5 <= 0.3807:
            break
    
    # Final computation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
