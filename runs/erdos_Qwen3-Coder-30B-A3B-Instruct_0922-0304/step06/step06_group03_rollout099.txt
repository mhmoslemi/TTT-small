import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Discrete values for step function
    discrete_values = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    # Local search with discrete steps and random restarts
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create a mutated version
        h_new = h.copy()
        
        # Randomly choose indices to modify
        num_modifications = max(1, n_points // 20)
        indices_to_modify = np.random.choice(n_points, size=num_modifications, replace=False)
        
        for i in indices_to_modify:
            # Perturb towards discrete values
            current_val = h_new[i]
            # Choose a nearest discrete value
            closest_value = discrete_values[np.argmin(np.abs(discrete_values - current_val))]
            # Small random perturbation
            perturbation = np.random.normal(0, 0.02)
            h_new[i] = closest_value + perturbation
            h_new[i] = np.clip(h_new[i], 0, 1)
            
        # Normalize to target sum
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Accept the new solution with some probability
        # This helps escape local minima
        if np.random.random() < 0.3 or compute_c5(h_new, dx) < c5:
            h = h_new
            
        iteration += 1
            
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
