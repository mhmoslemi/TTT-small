import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    discretization_levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Discretize to nearest levels
        h_discrete = np.zeros(n_points)
        for i in range(n_points):
            h_discrete[i] = discretization_levels[np.argmin(np.abs(discretization_levels - h[i]))]
        h = h_discrete
    else:
        # Initialize with a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize initial solution
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Small perturbation to escape local optima
        perturbation_magnitude = 0.01 + 0.02 * np.random.random()  # Adaptive magnitude
        random_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
        h[random_indices] += np.random.normal(0, perturbation_magnitude, len(random_indices))
        
        # Project h back to feasible region and discretize
        h = np.clip(h, 0, 1)
        # Discretize to nearest allowed value
        h_discrete = np.zeros(n_points)
        for i in range(n_points):
            h_discrete[i] = discretization_levels[np.argmin(np.abs(discretization_levels - h[i]))]
        h = h_discrete
        
        # Renormalize to target sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
