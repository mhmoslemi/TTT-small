import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, epsilon=1e-5):
    """Compute gradient of C5 w.r.t h using finite differences."""
    n_points = len(h)
    grad = np.zeros(n_points)
    base_c5 = compute_c5(h, dx)
    
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = normalize_h(h_plus, len(h) / 2)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = normalize_h(h_minus, len(h) / 2)
        
        grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * epsilon)
    
    return grad

def line_search(h, grad, dx, alpha_max=1.0, max_iter=50):
    """Perform backtracking line search to find suitable alpha."""
    alpha = alpha_max
    c5_current = compute_c5(h, dx)
    rho = 0.5
    c = 1e-4
    
    for _ in range(max_iter):
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h) / 2)
        
        c5_new = compute_c5(h_new, dx)
        
        if c5_new <= c5_current - c * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= rho
    
    return alpha, h
