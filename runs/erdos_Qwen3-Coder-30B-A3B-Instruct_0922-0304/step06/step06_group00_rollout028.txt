import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2, rate=0.8):
    """Perform uniform crossover between two parents."""
    if np.random.rand() > rate:
        return parent1.copy(), parent2.copy()
    
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, strength=0.1):
    """Mutate an individual with Gaussian noise."""
    noise = np.random.normal(0, strength, len(individual))
    individual = individual + noise
    return np.clip(individual, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 100
    
    # Initialize population
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        base_h = normalize_h(base_h, target_sum)
    
    population = []
    for _ in range(population_size):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    max_generations = 60
    mutation_strength = 0.2
    mutation_decay = 0.99
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_strength)
            child2 = mutate(child2, mutation_strength)
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            new_population.extend([child1, child2])
        
        population = new_population
        mutation_strength *= mutation_decay
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to base solution if no improvement
        base_c5 = compute_c5(base_h, dx)
        return (base_h.tolist(), base_c5, n_points)
