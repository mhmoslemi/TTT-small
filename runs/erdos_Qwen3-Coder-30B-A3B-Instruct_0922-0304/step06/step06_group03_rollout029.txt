import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, std_dev):
    """Perturb h with Gaussian noise and maintain bounds."""
    h = h + np.random.normal(0, std_dev, len(h))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Generate diverse initial guesses
    initial_guesses = []
    
    # 1. Original based
    h1 = h.copy()
    initial_guesses.append(h1)
    
    # 2. Comb pattern with 2 peaks
    h2 = np.zeros(n_points)
    h2[10:20] = 1
    h2[50:60] = 1
    h2 = normalize_h(h2, target_sum)
    initial_guesses.append(h2)
    
    # 3. Random pattern
    h3 = np.random.random(n_points)
    h3 = normalize_h(h3, target_sum)
    initial_guesses.append(h3)
    
    # 4. Alternating pattern
    h4 = np.zeros(n_points)
    h4[::2] = 1
    h4 = normalize_h(h4, target_sum)
    initial_guesses.append(h4)
    
    # 5. Low-discrepancy pattern (approximation)
    h5 = np.zeros(n_points)
    for i in range(n_points):
        h5[i] = (i * 0.618033988749895) % 1
    h5 = normalize_h(h5, target_sum)
    initial_guesses.append(h5)
    
    iteration = 0
    max_iter = 10000
    std_dev = 0.02
    std_dev_decay = 0.95
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try each initial guess
        for h_guess in initial_guesses:
            h = h_guess.copy()
            
            # SLSQP optimization
            def objective(h):
                return compute_c5(h, dx)

            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h = result.x
                else:
                    h = perturb_h(h, std_dev * 2)
            except:
                h = perturb_h(h, std_dev * 2)
            
            # Apply more aggressive perturbations
            if iteration % 10 == 0:
                h = perturb_h(h, std_dev)
                h = normalize_h(h, target_sum)
            
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                return (best_h.tolist(), best_c5, n_points)
        
        # Adaptive perturbation
        if iteration % 100 == 0 and std_dev > 0.001:
            std_dev *= std_dev_decay
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
