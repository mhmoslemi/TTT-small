import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Make a random perturbation
        h_candidate = h.copy()
        
        # Select two indices to modify
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure we don't violate bounds
        if h_candidate[i] >= 1 or h_candidate[j] <= 0:
            continue
            
        # Determine the maximum delta we can make
        delta = min(1 - h_candidate[i], h_candidate[j])
        if delta <= 0:
            continue
            
        # Adjust values
        delta_change = np.random.uniform(0, delta)
        h_candidate[i] += delta_change
        h_candidate[j] -= delta_change
        
        # Renormalize to satisfy sum constraint
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute new C5
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            best_h = h.copy()
            stagnation_count = 0
        else:
            # Metropolis criterion
            delta_c5 = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                h = h_candidate
                
            stagnation_count += 1
            
        # Cooling schedule
        if stagnation_count > max_stagnation:
            T *= 1.2  # Increase T to escape local minima
            stagnation_count = 0
        else:
            T *= alpha
            
        if T < min_T:
            break
            
        # Occasionally perturb to escape local minima
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            # Add a small perturbation to escape
            perturbation = np.random.normal(0, 0.005, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
