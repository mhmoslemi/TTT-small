import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with local search
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    # Adaptation: compute exact gradient for the maximum k
    def compute_exact_gradient(h, k_star):
        grad = np.zeros(n_points)
        # i is the index in h
        # j = (i + k_star) % n_points is the index in (1-h)
        for i in range(n_points):
            j = (i + k_star) % n_points
            if j < n_points:
                grad[i] = (1 - h[j]) * dx
            else:
                grad[i] = 0
        return grad
    
    # Line search function
    def line_search(h, grad, max_iter=5):
        alpha = 0.1  # Acceptance threshold
        beta = 0.5   # Reduction factor
        eta = 0.01   # Initial step size
        old_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            h_new = h - eta * grad
            # Project to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = compute_c5(h_new)
            
            # Accept if improvement
            if new_c5 < old_c5 - alpha * eta * np.dot(grad, grad):
                return eta, h_new
            else:
                eta *= beta
                
        # Return default step if no improvement
        return eta, h
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient
        g = compute_exact_gradient(h, max_k)
        
        # Perform line search
        eta, h_new = line_search(h, g)
        
        # Update h with line search result
        h = h_new
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
