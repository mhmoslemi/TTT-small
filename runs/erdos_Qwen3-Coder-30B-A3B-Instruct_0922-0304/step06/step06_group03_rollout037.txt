import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    
    # Adjust heights to maintain total mass
    total_tooth_mass = len(tooth_positions) * tooth_height
    tooth_mass = (total_tooth_mass * dx) / (n_points / 2)
    
    for pos in tooth_positions:
        h[pos] = tooth_mass
    
    # Normalize to target sum
    h = normalize_h(h, n_points / 2)
    return h

def add_perturbations(h, sigma=0.02):
    """Add small random perturbations."""
    noise = np.random.normal(0, sigma, len(h))
    h_perturbed = h + noise
    return np.clip(h_perturbed, 0, 1)

def optimize_pattern(h, target_sum, dx):
    """Optimize a single pattern using SLSQP."""
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            return np.clip(result.x, 0, 1)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default comb initialization
        h = create_comb_pattern(n_points, 3)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Combine optimization with comb patterns
    spacing_list = [2, 3, 4, 5, 6]
    
    # Gradient-free comb exploration
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Choose a random comb pattern
        spacing = np.random.choice(spacing_list)
        h_comb = create_comb_pattern(n_points, spacing)
        
        # Add small perturbations
        h_comb_perturbed = add_perturbations(h_comb, 0.02)
        h_comb_perturbed = np.clip(h_comb_perturbed, 0, 1)
        
        # Optimize
        h_opt = optimize_pattern(h_comb_perturbed, target_sum, dx)
        h = h_opt
        
        # Occasionally try a higher resolution refinement
        if iteration % 200 == 0:
            n_points_high = 140
            dx_high = 2.0 / n_points_high
            h_high = np.zeros(n_points_high)
            spacing_rescaled = spacing * 2  # since we are upscaling
            
            # Interpolate the pattern
            for i in range(n_points):
                for j in range(int(spacing_rescaled)):
                    idx = i * 2 + j
                    if idx < n_points_high:
                        h_high[idx] += h[i] / spacing_rescaled
                        
            h_high = normalize_h(h_high, n_points_high / 2)
            h_high_opt = optimize_pattern(h_high, n_points_high / 2, dx_high)
            
            # Transfer back to lower resolution
            h_down = np.zeros(n_points)
            for i in range(n_points):
                h_down[i] = h_high_opt[i * 2]
            h_down = normalize_h(h_down, target_sum)
            
            # Check if this is better
            temp_c5 = compute_c5(h_down, dx)
            if temp_c5 < best_c5:
                best_c5 = temp_c5
                best_h = h_down.copy()
            h = h_down
                
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
