import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def barrier_gradient(h, target_sum, mu):
    """Compute gradient of barrier function."""
    grad = np.zeros_like(h)
    # Gradient of -log(h) - log(1-h) 
    grad += -1.0 / (h + 1e-12) + 1.0 / (1 - h + 1e-12) 
    # Gradient of -log(sum(h) - target_sum)
    grad += -1.0 / (np.sum(h) - target_sum + 1e-12)
    return mu * grad

def gradient_approx(h, dx, mu, eps=1e-5):
    """Approximate gradient of C5 with barrier term."""
    base_c5 = compute_c5(h, dx)
    grad = np.zeros_like(h)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = normalize_h(h_plus, len(h) / 2)
        
        c5_plus = compute_c5(h_plus, dx)
        grad[i] = (c5_plus - base_c5) / eps
    
    barrier_grad = barrier_gradient(h, len(h) / 2, mu)
    return grad + barrier_grad

def backtracking_line_search(h, grad, mu, dx, alpha=1.0, rho=0.5, c=1e-4):
    """Perform backtracking line search."""
    while True:
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h) / 2)
        
        c5_new = compute_c5(h_new, dx)
        c5_old = compute_c5(h, dx)
        lhs = c5_new
        rhs = c5_old - c * alpha * np.dot(grad, grad)
        
        if lhs <= rhs:
            return h_new, alpha
        alpha *= rho

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create starting point
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    mu = 1.0
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        grad = gradient_approx(h, dx, mu)
        
        # If gradient is too small, decrease mu
        if np.linalg.norm(grad) < 1e-6:
            mu *= 0.5
            if mu < 1e-6:
                mu = 1e-6
                
        # Line search
        h_new, alpha = backtracking_line_search(h, grad, mu, dx)
        h = h_new
        
        # Evaluate new solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
        
        # Decrease mu periodically
        if iter_count % 50 == 0:
            mu *= 0.5
            if mu < 1e-6:
                mu = 1e-6
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
