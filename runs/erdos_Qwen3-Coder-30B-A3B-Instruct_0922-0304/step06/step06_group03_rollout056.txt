import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize PSO
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        else:
            particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle, dx))
    
    # Global best
    global_best_idx = np.argmin(p_best_values)
    g_best = p_best[global_best_idx].copy()
    g_best_value = p_best_values[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            fitness = compute_c5(particle, dx)
            
            # Update personal best
            if fitness < p_best_values[i]:
                p_best[i] = particle.copy()
                p_best_values[i] = fitness
                
                # Update global best
                if fitness < g_best_value:
                    g_best = particle.copy()
                    g_best_value = fitness
                    
        # Termination check
        if g_best_value <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Clip and normalize
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
    
    # Gradient refinement
    if time.time() - start_time < budget_s - 2:
        top_particles = np.argsort(p_best_values)[:10]
        for i in top_particles:
            particle = p_best[i].copy()
            def objective(h):
                return compute_c5(h, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            result = minimize(objective, particle, method='SLSQP', bounds=bounds, options={'maxiter': 100})
            
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = normalize_h(refined, target_sum)
                ref_value = compute_c5(refined, dx)
                if ref_value < p_best_values[i]:
                    p_best[i] = refined
                    p_best_values[i] = ref_value
                    
                    if ref_value < g_best_value:
                        g_best = refined.copy()
                        g_best_value = ref_value
    
    # Final evaluation
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
