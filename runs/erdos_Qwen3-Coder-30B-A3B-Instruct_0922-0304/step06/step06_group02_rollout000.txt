import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the given construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use particle swarm optimization with local refinement
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_c5.append(float('inf'))
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_iter = 100
    
    gbest = h.copy()
    gbest_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Evaluate current particle
            current_c5 = compute_c5(particle, dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = current_c5
                
            # Update global best
            if current_c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = current_c5
                
            # Update best solution
            if gbest_c5 < best_c5:
                best_c5 = gbest_c5
                best_h = gbest.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
            
            # Update velocity and position
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            vel = (w * vel + 
                   c1 * r1 * (pbest[i] - particle) + 
                   c2 * r2 * (gbest - particle))
            
            particles[i] = particle + vel
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
        iteration += 1
    
    # Local refinement using L-BFGS
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Refine the best solution found
    if 'gbest' in locals():
        result = minimize(objective, gbest, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
                
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
