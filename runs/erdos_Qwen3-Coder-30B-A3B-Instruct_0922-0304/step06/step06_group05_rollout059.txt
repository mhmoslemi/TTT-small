import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal pattern
        h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adaptive gradient descent with momentum
    max_iter = 10000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current max_k
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            if i - max_k >= 0:
                g[i] -= h[i - max_k] * dx
        
        # Update with momentum and adaptive step size
        v = momentum * g_prev + g
        new_h = h - step_size * v
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        g_prev = g
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Adaptive step size and momentum decay
        if iteration % 1000 == 0:
            step_size *= 0.8
            momentum = max(0.5, momentum * 0.95)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
