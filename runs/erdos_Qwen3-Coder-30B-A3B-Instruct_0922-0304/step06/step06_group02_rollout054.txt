import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Better initial construction
        h = np.zeros(n_points)
        # Place a step at the center with appropriate height
        mid = n_points // 2
        h[mid-5:mid+5] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Track iterations and time
    iteration = 0
    max_iter = 10000
    
    # Use an alternating optimization strategy
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Adaptive perturbation - vary strength with time
        elapsed = time.time() - start_time
        time_ratio = elapsed / budget_s
        perturb_strength = max(0.001, 0.05 * (1 - time_ratio))
        
        # Random perturbation with normalized step sizes
        noise = np.random.normal(0, perturb_strength, n_points)
        h_noisy = h + noise
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        # Optimization using L-BFGS-B
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective, 
            h_noisy, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=constraint, 
            options={'maxiter': 100}
        )
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 * 0.99999:
                h = h_candidate.copy()
                
        # Periodic global perturbation
        if iteration % 20 == 0:
            perturb = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
