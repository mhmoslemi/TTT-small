import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement using cutting-plane method
    iteration = 0
    max_iter = 5000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lag
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the worst lag
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_lag = max_lag_idx - (len(h) - 1)  # Convert indices to actual lag values
        
        # Optimization to minimize C with constraint for the worst lag
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def constraint_worst_lag(h, lag=max_lag):
            # Compute the overlap at this lag for current h
            # We use correlate in same way as when we compute C5
            corr = correlate(h, 1 - h, mode="full")
            lag_idx = lag + len(h) - 1  # Convert to full array index
            return corr[lag_idx] - (best_c5 / dx)  # Enforce it's at most best_c5
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
            {'type': 'ineq', 'fun': constraint_worst_lag}
        ]
        
        # Try to improve with SLSQP
        h_noisy = h + np.random.normal(0, 0.01, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        def objective(h):
            return compute_c5(h, dx)
            
        result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraints, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Check improvement using a simple criterion
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 * 0.99999:
                # Accept the improvement
                h = h_candidate.copy()
        
        # Add some randomness to escape local minima
        if iteration % 10 == 0:
            noise = np.random.normal(0, 0.01, len(h))
            h = np.clip(h + noise, 0, 1)
            h = normalize_h(h, target_sum)
        
        iteration += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
