import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple diverse initial guesses."""
    guesses = []
    
    # 1. Initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_normalize(h, target_sum)
        guesses.append(h.copy())
    
    # 2. Comb pattern
    h = np.zeros(n_points)
    for i in range(0, n_points, 5):
        h[i] = 1.0
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # 3. Random pattern
    h = np.random.random(n_points) * 2
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # 4. Alternating pattern
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # 5. Block pattern
    h = np.zeros(n_points)
    h[n_points//4:3*n_points//4] = 1.0
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    iter_count = 0
    max_iter = 20000
    
    for guess_idx, h in enumerate(initial_guesses):
        current_h = h.copy()
        current_c5 = compute_c5(current_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimization loop
        for iteration in range(1000):
            if time.time() - start_time >= budget_s - 1:
                break
                
            # SLSQP optimization step
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, current_h, method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 10, 'ftol': 1e-8})
                if result.success:
                    current_h = result.x
                else:
                    raise ValueError("Optimization failed")
            except:
                # Fallback to perturbation if optimization fails
                pass
            
            # Apply systematic perturbations to escape local optima
            if iteration % 5 == 0:
                perturbation_std = max(0.01, 0.02 * np.exp(-iteration / 1000.0))
                perturbation = np.random.normal(0, perturbation_std, n_points)
                current_h += perturbation
                current_h = project_and_normalize(current_h, target_sum)
                
                # Occasionally apply larger perturbations
                if iteration % 25 == 0:
                    perturbation = np.random.normal(0, 0.05, n_points)
                    current_h += perturbation
                    current_h = project_and_normalize(current_h, target_sum)
            
            # Update best if improved
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            iter_count += 1
            if iter_count >= max_iter:
                break
                
        if best_c5 <= 0.3807:
            break
    
    if best_h is None:
        # Fallback to initial_h_values or generate default
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = project_and_normalize(best_h, target_sum)
        else:
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h = project_and_normalize(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
