import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x) in time domain."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs))[:, None] * np.pi * x),
        axis=0
    )
    return np.clip(h, 0, 1)

def evaluate_h(h, dx):
    """Evaluate h and return C5."""
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial h from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple rectangular pulse for initial solution
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
        
    # Normalize
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Fourier representation: use first 15 coefficients
    n_freq = 15
    
    def objective_fourier(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h_norm = h * (target_sum / np.sum(h))
        return evaluate_h(h_norm, dx)
    
    # Bounds for Fourier coefficients: [a0] and [ak, bk] for k=1..n_freq
    bounds = [(-10, 10)] + [(-10, 10)] * (2 * n_freq)
    
    # If initial_h_values is available, use it to initialize population for DE
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        # Compute Fourier coefficients
        h_normalized = h_init * (target_sum / np.sum(h_init))
        # Use DCT-based approximation for initial coefficients
        coeffs_init = np.fft.fft(h_normalized)
        coeffs_init = np.real(coeffs_init)
        initial_guess = np.concatenate(([coeffs_init[0].real], coeffs_init[1:n_freq+1].real, coeffs_init[1:n_freq+1].imag))
    else:
        # Initial simple guess
        initial_guess = np.zeros(2 * n_freq + 1)
    
    # Run DE optimization
    result = differential_evolution(
        objective_fourier, 
        bounds, 
        seed=seed, 
        maxiter=300, 
        popsize=70, 
        mutation=(0.5, 1.0), 
        recombination=0.7,
        init=[initial_guess]
    )
    
    # Extract best coefficients
    best_coeffs = result.x
    best_h = fourier_to_h(best_coeffs, n_points)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    # Local refinement - SLSQP
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    con = {'type': 'eq', 'fun': constraint_sum}
    
    result_slsqp = minimize(
        lambda x: evaluate_h(x, dx),
        best_h.copy(),
        method='SLSQP',
        bounds=bounds,
        constraints=con,
        options={'maxiter': 100}
    )
    
    if result_slsqp.success:
        best_h = np.clip(result_slsqp.x, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))
    
    # Final check
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
