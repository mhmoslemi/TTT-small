import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize sum to target
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Set up the QP problem
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_vars) == target_sum,
        h_vars >= 0,
        h_vars <= 1
    ]
    
    # C5 constraints: for each lag k, sum of h[i](1-h[i+k]) <= C*dx
    # Generate full correlation constraints
    for k in range(-n_points + 1, n_points):
        # Find valid indices where both h[i] and h[i+k] are defined
        if k >= 0:
            valid_indices = range(n_points - k)
            # Valid i where i+k < n_points
            # So i ranges from 0 to n_points - 1 - k
        else:
            valid_indices = range(-k, n_points)
            #Valid i where i+k >= 0, i from -k to n_points - 1
        # Now sum h[i] * (1 - h[i+k]) for i in valid_indices
        expr = 0
        for i in valid_indices:
            i_k = i + k
            expr += h_vars[i] * (1 - h_vars[i_k])
        constraints.append(expr <= C * dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create and solve the CVXPY problem
    prob = cp.Problem(objective, constraints)
    
    # Try a few iterations with warm start
    max_iterations = 50
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        try:
            # Initialize with current h
            h_vars.value = h
            prob.solve(solver=cp.MOSEK, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_vars.value
                h_new = np.clip(h_new, 0, 1)
                
                # Re-normalize
                h_sum_new = np.sum(h_new)
                if h_sum_new > 0:
                    h_new = h_new * (target_sum / h_sum_new)
                
                # Compute C5
                current_c5 = compute_c5(h_new, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_new.copy()
                
                # Early exit if good enough
                if best_c5 <= 0.3807:
                    break
                
                # Re-initialize for next iteration
                h = h_new
            else:
                # Fallback: just trim and normalize old h
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (target_sum / h_sum)
        except Exception:
            # Fallback if optimization fails
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (target_sum / h_sum)
    
    # Final evaluation    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
