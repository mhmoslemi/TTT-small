import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    max_iter = 500
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.random.rand(n_points) * 0.1)
        pbest.append(particle.copy())
        pbest_scores.append(compute_c5(particle, dx))
    
    # Initialize gbest
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    if gbest_score < best_c5:
        best_c5 = gbest_score
        best_h = gbest.copy()
        
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Adaptive inertia
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = score
                
                # Early exit if target achieved
                if gbest_score <= 0.3807:
                    break
                
                if gbest_score < best_c5:
                    best_c5 = gbest_score
                    best_h = gbest.copy()
        
        # Periodic local refinement
        if iter_num % 10 == 0:
            # Local refinement using L-BFGS-B
            refined_particles = []
            for particle in particles[:10]:  # Refine some particles
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    return compute_c5(h, dx)
                
                # Use L-BFGS-B for each particle
                result = minimize(objective, particle, method='L-BFGS-B', 
                                bounds=bounds, options={'maxiter': 50})
                
                if result.success:
                    refined = np.clip(result.x, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    refined_particles.append(refined)
                else:
                    refined_particles.append(particle)
            
            # Replace with refined particles
            particles[:10] = refined_particles
            
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
