import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_gradient(h, k_star):
        g = np.zeros(n_points)
        for i in range(n_points):
            # g[i] = (1 - h[(i + k_star) % n_points]) * dx
            if (i + k_star) < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                g[i] = -h[i - k_star] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    constant_sigma = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using the current best lag
        g = compute_gradient(h, max_k)
        
        # Adaptive line search with backtracking
        alpha = 0.1
        beta = 0.5
        eta = step_size
        
        for _ in range(10):  # At most 10 line search steps
            h_new = h - eta * g
            h_new = project_h(h_new)
            c5_new = compute_c5(h_new)
            
            if c5_new <= max_overlap - alpha * eta * np.dot(g, g):
                h = h_new
                break
            else:
                eta *= beta

        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            pert = np.random.normal(0, constant_sigma, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
