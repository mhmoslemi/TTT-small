import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    discrete_levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():

        # Discretize to nearest allowed levels
        h_discrete = np.zeros_like(h)
        for i in range(len(h)):
            h_discrete[i] = discrete_levels[np.argmin(np.abs(h[i] - discrete_levels))]
        h = h_discrete
    else:
        # Default initialize to a step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    h = h * (target_sum / np.sum(h))
    
    # Discretize to closest allowed levels
    for i in range(len(h)):
        h[i] = discrete_levels[np.argmin(np.abs(h[i] - discrete_levels))]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Helper function to generate neighborhood for local search
    def get_neighbor(h, idx, delta=0.25):
        h_new = h.copy()
        h_new[idx] = max(0, min(1, h_new[idx] + delta))
        # Re-discretize to nearest level
        h_new[idx] = discrete_levels[np.argmin(np.abs(h_new[idx] - discrete_levels))]
        return h_new
    
    # Try different starting configurations with random searches
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization
        h_try = np.random.choice(discrete_levels, n_points)
        # Normalize
        h_try = h_try * (target_sum / np.sum(h_try))
        
        # Local search using neighborhood perturbations
        local_best = h_try.copy()
        local_best_c5 = compute_c5(local_best, dx)
        
        iter_count = 0
        while time.time() - start_time < budget_s - 1 and iter_count < 200:
            if local_best_c5 < best_c5:
                best_c5 = local_best_c5
                best_h = local_best.copy()
            if best_c5 <= 0.3807:
                break
            
            # Try small perturbations
            rand_idx = np.random.randint(0, n_points)
            delta = 0.25 * (np.random.random() - 0.5)
            h_new = get_neighbor(local_best, rand_idx, delta)
            
            # Enforce sum constraint by adjusting one element
            diff = target_sum - np.sum(h_new)
            if abs(diff) > 1e-6:
                # Adjust the element with the smallest difference in levels
                min_idx = np.argmin(np.abs(h_new - discrete_levels))
                h_new[min_idx] = max(0, min(1, h_new[min_idx] + diff))
                h_new[min_idx] = discrete_levels[np.argmin(np.abs(h_new[min_idx] - discrete_levels))]
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < local_best_c5:
                local_best = h_new
                local_best_c5 = new_c5
            elif np.random.random() < 0.1:  # Sometimes accept worse moves
                local_best = h_new
                local_best_c5 = new_c5
                
            iter_count += 1
            
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best.copy()
            
        iteration += 1
    
    # Final optimization using a more refined local search
    final_h = best_h.copy()
    final_c5 = best_c5
    
    if final_c5 > 0.3807:
        # Perform a exhaustive search on a subset of variables
        for i in range(0, n_points, 5):  # check every 5th element
            if time.time() - start_time > budget_s - 1:
                break
            for val in discrete_levels:
                if abs(val - final_h[i]) < 1e-3:
                    continue
                h_new = final_h.copy()
                h_new[i] = val
                # Re-discretize other elements nearby to keep sum constraint
                diff = target_sum - np.sum(h_new)
                adjustment = diff / (n_points - 1)  # spread error across others
                for j in range(n_points):
                    if j != i:
                        h_new[j] = max(0, min(1, h_new[j] + adjustment))
                        h_new[j] = discrete_levels[np.argmin(np.abs(h_new[j] - discrete_levels))]
            
                new_c5 = compute_c5(h_new, dx)
                if new_c5 < final_c5:
                    final_h = h_new
                    final_c5 = new_c5
    
    return (final_h.tolist(), final_c5, n_points)
