import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    tooth_indices = tooth_positions
    h[tooth_indices] = tooth_height
    return h

def comb_pattern_search(n_points, dx, target_sum, initial_h, start_time, budget_s):
    """Search comb patterns and optimize them."""
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Combinations of tooth spacing and heights
    spacings = [2, 3, 4, 5, 6]
    tooth_heights = [0.5, 1.0, 1.5]
    
    num_combinations = len(spacings) * len(tooth_heights)
    comb_configs = [(s, th) for s in spacings for th in tooth_heights]
    
    for i, (spacing, tooth_height) in enumerate(comb_configs):
        h = create_comb_pattern(n_points, spacing, tooth_height)
        h = normalize_h(h, target_sum)
        
        # Optimize this comb pattern
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Perturb a bit before optimization
        perturbation = np.random.normal(0, 0.02, n_points)
        h = np.clip(h + perturbation, 0, 1)
        h = normalize_h(h, target_sum)
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            h_opt = result.x
        else:
            h_opt = h
            
        c5 = compute_c5(h_opt, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            
        if time.time() - start_time > budget_s - 10:
            break
            
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from precomputed values if available

    else:
        # Fallback: create a basic comb pattern
        h = create_comb_pattern(n_points, 4, 1.0)
        h = normalize_h(h, target_sum)

    # Multi-resolution refinement
    for resolution_factor in [1, 2]:  # Start with current resolution, then increase
        if resolution_factor == 2:
            n_points = 140
            dx = 2.0 / n_points
            target_sum = n_points / 2
            # Interpolate from previous solution
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
            h = normalize_h(h, target_sum)
        
        # Combine comb search with optimization
        h, c5 = comb_pattern_search(n_points, dx, target_sum, h, start_time, budget_s)
        
        if time.time() - start_time >= budget_s - 10:
            break
        
        # Additional refinement using local optimization
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Local refinement using SLSQP
        for _ in range(5):
            # Small perturbations to help escape local minima
            perturbation = np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h + perturbation, 0, 1)
            h_pert = normalize_h(h_pert, target_sum)
            
            result = minimize(objective, h_pert, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_try = result.x
            else:
                h_try = h_pert
            
            c5_try = compute_c5(h_try, dx)
