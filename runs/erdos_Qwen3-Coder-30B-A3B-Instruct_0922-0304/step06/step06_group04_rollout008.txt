import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    gen = 0
    no_improvement_count = 0
    mut_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        gen += 1
        # Evaluate population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Track best
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_individual = population[fitness.index(min_fitness)].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        elite_size = max(2, population_size // 5)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Crossover and mutation
        new_population = elite.copy()
        
        # Elitism: add one best from previous
        new_population.append(best_individual.copy())
        
        # Generate remaining individuals
        while len(new_population) < population_size:
            # Selection by tournament
            parent1 = population[np.random.choice(sorted_indices[:population_size//2])]
            parent2 = population[np.random.choice(sorted_indices[:population_size//2])]
            
            # Crossover
            if np.random.rand() < 0.8:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.rand() < mut_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child = child + noise
                    child = np.clip(child, 0, 1)
                    child = normalize_h(child, target_sum)
                new_population.append(child)
                if len(new_population) >= population_size:
                    break
        
        # Ensure population size
        population = new_population[:population_size]
        
        # Adjust mutation rate
        if no_improvement_count > 20:
            mut_rate = min(0.5, mut_rate * 1.2)
        if no_improvement_count > 50:
            mut_rate = 0.1
            
        # Increase population size if stuck
        if no_improvement_count > 20 and len(population) < 40:
            population_size = 40
            # Reinitialize with more diversity
            population = []
            for _ in range(population_size):
                h = np.random.rand(n_points)
                h = normalize_h(h, target_sum)
                population.append(h)
    
    # Final evaluation
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        # Fallback to initial if nothing was found
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
        else:
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
        best_individual = normalize_h(best_individual, target_sum)
        final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
