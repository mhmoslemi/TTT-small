import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    max_iter = 1000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        particles.append(particle.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(compute_c5(particle, dx))
    
    # Global best
    gbest_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_value = pbest_values[gbest_index]
    
    # PSO hyperparameters
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        w = w_start - (w_start - w_end) * (iter_count / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest_positions[i] - particles[i]) +
                             c2 * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if fitness < gbest_value:
                gbest_value = fitness
                gbest_position = particles[i].copy()
                
        # Periodic local refinement with L-BFGS-B
        if iter_count % 10 == 0:
            for i in range(n_particles):
                # Local refinement with bounds and constraint
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    return compute_c5(h, dx)
                
                # Use L-BFGS-B for local search
                result = minimize(objective, particles[i], method='L-BFGS-B',
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 50})
                
                if result.success:
                    refined = np.clip(result.x, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    refined_fitness = compute_c5(refined, dx)
                    
                    if refined_fitness < pbest_values[i]:
                        pbest_values[i] = refined_fitness
                        pbest_positions[i] = refined.copy()
                    
                    if refined_fitness < gbest_value:
                        gbest_value = refined_fitness
                        gbest_position = refined.copy()
                        
        # Update global best
        gbest_index = np.argmin(pbest_values)
        if pbest_values[gbest_index] < gbest_value:
            gbest_value = pbest_values[gbest_index]
            gbest_position = pbest_positions[gbest_index].copy()
            
        # Early termination
        if gbest_value <= 0.3807:
            break
            
        iter_count += 1
    
    # Final evaluation
    final_c5 = gbest_value
    final_h = gbest_position
    
    return (final_h.tolist(), final_c5, n_points)
