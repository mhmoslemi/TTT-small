import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    h = h * (target_sum / np.sum(h))
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Warm start will improve convergence
    # CVXPY QP approach for global optimization
    # Define variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Initialize with best current
    h_var.value = h.copy()
    
    # Avoiding repeated computation by tracking group of constraints
    # For each lag k, we compute the overlap constraint
    # Lag indices from -n_points+1 to n_points-1
    
    # Create constraint list
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == target_sum)
    
    # Bound constraints
    constraints.extend([h_var[i] >= 0 for i in range(n_points)])
    constraints.extend([h_var[i] <= 1 for i in range(n_points)])
    
    # Max overlap constraints for all lags
    # We can limit to non-negative lags for efficiency since the problem is symmetric
    for k in range(1, n_points):  # k from 1 to n_points-1
        # For lag k, indices i such that i+k < n_points
        idx = np.arange(n_points - k)
        # h_var[i] * (1 - h_var[i+k]) for valid indices
        expr = cp.sum(cp.multiply(h_var[idx], 1 - h_var[idx + k])) * dx
        constraints.append(expr <= C)
    
    # The symmetric part for negative lags is handled by symmetry of correlation
    # But realistic in practice, we must check negative lags carefully
    
    # To handle negative lags correctly, loop through all valid shifts
    for k in range(-n_points + 1, 0):  # k = -n+1 to -1
        # i such that i+k >= 0 and i < n_points
        # so i:-k range, but also i < n_points
        idx = np.arange(max(0, -k), min(n_points, n_points - k))
        # Sum h_var[i] * (1 - h_var[i + k]) * dx for valid indices
        if len(idx) > 0:
            expr = cp.sum(cp.multiply(h_var[idx], 1 - h_var[idx - k])) * dx
            constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with CVXPY
    solver = cp.GUROBI  # Available
    try:
        prob.solve(solver=solver, verbose=False, TimeLimit=budget_s/2.0)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h_var.value
            h_result = np.clip(h_result, 0, 1)
            h_result = h_result * (target_sum / np.sum(h_result))
            c5_result = C.value
            if c5_result is not None:
                best_c5 = c5_result
                best_h = h_result.copy()
    except Exception as e:
        # Fall back to heuristics if QP fails
        pass
    
    # Fine tune using local search with SLSQP if not time expired
    if time.time() - start_time < budget_s - 1 and best_c5 > 0.3807:
        h = best_h.copy()
        def constraint_sum(h):
            return np.sum(h) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
            
        # Try a slightly better local init
        h = np.clip(h + np.random.normal(0, 0.01, len(h)), 0, 1)
        h = normalize_h(h, target_sum)
        
        for iter_ in range(100):  # Limited local search
            if time.time() - start_time > budget_s - 1:
                break
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()
                h = h_candidate.copy()
            else:
                break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

from scipy.optimize import minimize
