```python
import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import MOSEK

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
        h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Bilinear optimization approach with CVXPY
    iteration = 0
    max_iter = 30  # Reduced for time budget
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try to solve bilinear program
        try:
            h_var = cp.Variable(n_points)
            
            # Define objective
            z_vars = []
            for k in range(2 * n_points - 1):
                z_k = cp.Variable()
                z_vars.append(z_k)
            
            # Constraints
            constraints = [
                0 <= h_var,
                h_var <= 1,
                cp.sum(h_var) == target_sum
            ]
            
            # Bilinear constraints
            for k in range(2 * n_points - 1):
                # Index range of valid correlation
                start_i = max(0, k - n_points + 1)
                end_i = min(n_points, n_points - k)
                if start_i >= end_i:
                    continue
                    
                corr_sum = 0
                for i in range(start_i, end_i):
                    corr_sum += h_var[i] * (1 - h_var[i + k])
                constraints.append(z_vars[k] == corr_sum)
            
            # Objective function
            obj = cp.Minimize(cp.max(z_vars) * dx)
            prob = cp.Problem(obj, constraints)
            
            # Solve
            prob.solve(solver=MOSEK, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (target_sum / np.sum(h_new))
                    c5 = compute_c5(h_new, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_new.copy()
            
        except Exception as e:
            pass
        
        # Local perturbation if needed
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (target_sum / np.sum(h_pert))
        
        c5_local = compute_c5(h_pert, dx)
        if c5_local < best_c5:
            best_c5 = c5_local
            best_h = h_pert.copy()
        
        iteration += 1
        
        # Early exit if target achieved 
        if best_c5 <= 0.3807:
            break
        
        # Time check
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation for exact value
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
