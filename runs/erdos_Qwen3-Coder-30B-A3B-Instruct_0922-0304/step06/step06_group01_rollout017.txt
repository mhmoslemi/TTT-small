import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct h as a comb with specified parameters."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        tooth_start = i * (tooth_width + spacing)
        tooth_end = tooth_start + tooth_width
        
        # Indices for the tooth
        start_idx = int(np.floor(tooth_start / dx))
        end_idx = int(np.ceil(tooth_end / dx))
        
        # Make sure indices are valid
        start_idx = max(0, start_idx)
        end_idx = min(n_points, end_idx)
        
        for j in range(start_idx, end_idx):
            x = j * dx
            # Compute distance from tooth center
            if x < tooth_start:
                dist = tooth_start - x
            elif x > tooth_end:
                dist = x - tooth_end
            else:
                dist = 0
            
            # Apply transition shape
            if dist <= slope:
                h[j] += max(0, 1 - dist / slope)
    
    # Normalize to sum to n_points/2
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    
    return h

def optimize_comb_parameters(h, n_points, dx, initial_params):
    """Optimize comb parameters to minimize C5."""
    def objective(params):
        n_teeth, tooth_width, spacing, slope = params
        n_teeth = int(n_teeth)
        # Ensure bounds
        n_teeth = max(2, min(20, n_teeth))
        tooth_width = max(0.01, min(0.5, tooth_width))
        spacing = max(0.001, min(0.5, spacing))
        slope = max(0.01, min(0.5, slope))
        
        h_new = construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope)
        return compute_c5(h_new, dx)
    
    # Bounds for optimization
    bounds = [(2, 20), (0.01, 0.5), (0.001, 0.5), (0.01, 0.5)]
    
    # Initial guess
    x0 = [initial_params[0], initial_params[1], initial_params[2], initial_params[3]]
    
    # Optimize with L-BFGS-B
    result = minimize(objective, x0, bounds=bounds, method='L-BFGS-B', options={'maxiter': 100})
    
    if result.success:
        return result.x
    
    return initial_params

        h = normalize_h(h, target_sum)
    
    # Set up initial comb parameters
    initial_comb_params = [7, 2.0/7, (2.0-7*(2.0/7))/(7+1), (2.0/7)/2.0]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    max_iterations = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iterations:
        # Optimize comb parameters
        if iter_count % 5 == 0:
            comb_params = optimize_comb_parameters(h, n_points, dx, initial_comb_params)
            n_teeth, tooth_width, spacing, slope = comb_params
            n_teeth = int(n_teeth)
            h = construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope)
        
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)

        return np.sum(h) - target_sum

        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        best_c5 = min(best_c5, final_c5)
