import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple initialization: rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1

    # Normalize to target sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    # Ensure h is always within constraints with small epsilon for numerical stability
    h = np.clip(h, 1e-6, 1 - 1e-6)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Meta-heuristic approach: Simulated Annealing
    # Since we are trying to improve the previous result,
    # we will try to avoid local minima with a more diverse search
    
    # For this problem, let's use a hybrid scheme:
    # 1. Slightly randomize h with phase changes
    # 2. Evaluate and update best
    # 3. Step size adaptation
    
    # We optimize with variability in attempts, not just a single local optimization
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Create a variation around the current solution
        h_new = h.copy()
        
        # Add randomness with a small magnitude
        noise = np.random.normal(0, 0.02, n_points)
        h_new = h_new + noise
        
        # Clip and re-normalize
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        c5 = compute_c5(h_new, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        # Occasionally make a radical change if stuck
        if iteration % 300 == 0 and iteration > 0:
            # Randomize with a chance of large perturbation
            h_new = np.random.rand(n_points)
            #Ensure sum matches
            h_new = h_new * (target_sum / np.sum(h_new))
            h_new = np.clip(h_new, 0, 1)
            noise = np.random.normal(0, 0.02, n_points)
            h = h_new + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (target_sum / current_sum)
        else:
            h = h_new
            
        iteration += 1
    
    # Final local refinement
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        current_sum = np.sum(final_h)
        final_h = final_h * (target_sum / current_sum)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
