import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to target sum
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        particle = h.copy()
        particle += np.random.normal(0, 0.05, n_points)
        particle = project_to_constraints(particle, target_sum)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest_positions.append(particle.copy())
        pbest_values.append(compute_c5(particle, dx))
    
    # Global best
    best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[best_idx].copy()
    gbest_value = pbest_values[best_idx]
    
    iterations = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iterations / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest_positions[i] - particles[i]) +
                c2 * r2 * (gbest_position - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to constraints
            particles[i] = project_to_constraints(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
            
            # Update global best
            if current_c5 < gbest_value:
                gbest_value = current_c5
                gbest_position = particles[i].copy()
        
        # Early termination
        if gbest_value <= 0.3807:
            break
            
        iterations += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(gbest_position, dx)
    
    return (gbest_position.tolist(), final_c5, n_points)
