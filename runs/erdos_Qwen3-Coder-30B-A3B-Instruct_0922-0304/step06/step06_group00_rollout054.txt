```python
import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate(h, mutation_strength):
    """Apply Gaussian mutation to h."""
    h = h + np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover(h1, h2, crossover_rate=0.8):
    """Perform crossover between two h vectors."""
    if random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    # Single-point crossover
    point = random.randint(1, len(h1) - 1)
    new_h1 = np.concatenate([h1[:point], h2[point:]])
    new_h2 = np.concatenate([h2[:point], h1[point:]])
    return new_h1, new_h2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_indices = random.sample(range(len(population)), tournament_size)
    best_index = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Build population
    population_size = 100
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = mutate(h.copy(), 0.05)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    generation = 0
    max_generations = 500
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitnesses = []
        for individual in population:
            fitness = compute_c5(individual, dx)
            fitnesses.append(fitness)
        
        # Track best solution in current generation
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            mutation_strength = max(0.001, 0.1 * np.exp(-generation / 100.0))
            child1 = mutate(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            child2 = mutate(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        generation += 1
    
    # Final refinement of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
