import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate=0.05, target_sum=35.0):
    """Mutate h with constraint maintenance."""
    h_mut = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.05)
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover_h(h1, h2, crossover_rate=0.7):
    """Crossover two h vectors."""
    h_new = h1.copy()
    for i in range(len(h1)):
        if np.random.random() < crossover_rate:
            h_new[i] = h2[i]
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic algorithm parameters
    population_size = 30
    num_generations = 1000
    elite_size = 5
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    population = [h.copy()]
    for i in range(1, population_size):
        h_perturbed = mutate_h(h.copy(), mutation_rate=0.02, target_sum=target_sum)
        population.append(h_perturbed)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < num_generations:
        # Evaluate population
        fitnesses = []
        for h_ind in population:
            fitness = compute_c5(h_ind, dx)
            fitnesses.append(fitness)
        
        # Update best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        
        # Elitism: keep top individuals
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = sorted_population[np.random.randint(0, elite_size)]
            parent2 = sorted_population[np.random.randint(0, elite_size)]
            
            # Crossover
            offspring = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            offspring = mutate_h(offspring, mutation_rate, target_sum)
            
            new_population.append(offspring)
        
        population = new_population
        
        # Local refinement of best solutions
        for i in range(min(elite_size, len(population))):
            h_refined = population[i].copy()
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            # SLSQP local optimization
            try:
                result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = normalize_h(h_refined, target_sum)
                    
                    # Update if improvement
                    if compute_c5(h_refined, dx) < compute_c5(population[i], dx):
                        population[i] = h_refined
            except:
                pass  # Continue if optimization fails
        
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
