import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with alternating optimization
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try to improve via local optimization
        # Constraint function
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective function: minimize C5
        def objective(h):
            return compute_c5(h, dx)
            
        # Solve with SLSQP
        h_noisy = h + np.random.normal(0, 0.01, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        try:
            result = minimize(objective, h_noisy, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 50})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5 * 0.999999:
                    h = h_candidate.copy()
        except:
            pass
        
        # Occasionally do global perturbations
        if iteration % 10 == 0:
            # Random walk on some fraction of elements
            mask = np.random.random(n_points) < 0.3
            h[mask] += np.random.normal(0, 0.02, np.sum(mask))
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
