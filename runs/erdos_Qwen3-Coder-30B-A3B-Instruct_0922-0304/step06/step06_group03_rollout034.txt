import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a reasonable initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        pbest.append(particle.copy())
        pbest_values.append(compute_c5(particle, dx))
    
    # Global best initialization
    global_best_idx = np.argmin(pbest_values)
    gbest = pbest[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update all particles
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # PSO update with inertia and acceleration
            w = 0.8
            c1, c2 = 2.0, 2.0
            
            r1, r2 = np.random.random(n_points), np.random.random(n_points)
            velocity = w * velocity + \
                       c1 * r1 * (pbest[i] - particle) + \
                       c2 * r2 * (gbest - particle)
            
            particle += velocity
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            particles[i] = particle
            
            # Evaluate new fitness
            current_c5 = compute_c5(particle, dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particle.copy()
                pbest_values[i] = current_c5
                
                # Update global best if needed
                if current_c5 < gbest_value:
                    gbest = particle.copy()
                    gbest_value = current_c5
                    
            # Occasionally apply frequency filtering
            if iteration % 50 == 0 and iteration > 0:
                # Apply high-pass filtering in frequency domain
                filtered = np.zeros_like(particle)
                for j in range(n_points):
                    if j < n_points // 3:  # Low frequency components
                        filtered[j] = 0
                    else:
                        filtered[j] = particle[j]
                
                # Inverse transform and normalize
                filtered = np.fft.ifft(np.fft.fft(filtered)).real
                filtered = np.clip(filtered, 0, 1)
                filtered = normalize_h(filtered, target_sum)
                particles[i] = filtered
            
            # Occasionally apply gradient descent refinement
            if iteration % 100 == 0 and iteration > 0:
                # Refine top particles using gradient descent
                top_indices = np.argsort(pbest_values)[:10]
                for idx in top_indices:
                    def objective(h):
                        return compute_c5(h, dx)
                    def constraint_sum(h):
                        return np.sum(h) - target_sum
                    bounds = [(0, 1) for _ in range(n_points)]
                    constraint = {'type': 'eq', 'fun': constraint_sum}
                    result = minimize(objective, pbest[idx], method='SLSQP', 
                                    bounds=bounds, constraints=constraint, options={'maxiter': 100})
                    if result.success:
                        pbest[idx] = result.x
                        pbest_values[idx] = compute_c5(pbest[idx], dx)
                
                # Update global best after refinement
                global_best_idx = np.argmin(pbest_values)
                if pbest_values[global_best_idx] < gbest_value:
                    gbest = pbest[global_best_idx].copy()
                    gbest_value = pbest_values[global_best_idx]
        
        # Early stopping condition
        if gbest_value <= 0.3807:
            break
    
    # Final computation with the best found solution
    final_c5 = gbest_value
    
    # Make sure we return a valid solution
    if final_c5 > 1e10 or np.isnan(final_c5):
        final_c5 = compute_c5(h, dx)
        h = h.copy()
    
    return (gbest.tolist(), final_c5, n_points)
