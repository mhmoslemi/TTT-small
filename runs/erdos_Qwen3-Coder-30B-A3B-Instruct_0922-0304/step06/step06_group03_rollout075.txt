import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_comb_initial(n_points, spacing=5):
    """Generate a comb pattern with high values."""
    h = np.zeros(n_points)
    for i in range(0, n_points, spacing):
        if i < n_points:
            h[i] = 1.0
    return h

def generate_halton_initial(n_points, base=2):
    """Generate a low-discrepancy sequence pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        # Simple Halton-type sequence using base 2
        value = 0
        temp = i + 1
        power = 1.0
        while temp > 0:
            if temp % 2 == 1:
                value += power
            temp //= 2
            power *= 0.5
        h[i] = value
    return h

def generate_alternating_initial(n_points):
    """Generate alternating high/low pattern."""
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h[i] = 1.0
        else:
            h[i] = 0.0
    return h

def project_and_normalize(h, target_sum):
    """Project h to [0,1] bounds and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Collect initial guesses
    initial_guesses = []
    
    # Use provided initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_guesses.append(h.copy())
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_guesses.append(h.copy())
        
    # Generate additional initial guesses
    initial_guesses.append(generate_comb_initial(n_points, 4))
    initial_guesses.append(generate_halton_initial(n_points))
    initial_guesses.append(generate_alternating_initial(n_points))
    
    # Uniform random patterns with sum constraint
    for _ in range(3):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        initial_guesses.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    max_iter = 20000
    
    # Track time
    time_limit = start_time + budget_s - 1
    
    for guess_idx, h in enumerate(initial_guesses):
        # Reset h for each initial guess
        h = project_and_normalize(h, target_sum)
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Optimization loop with alternating step sizes
        step_size = 0.02
        inner_iteration = 0
        
        while time.time() < time_limit and iteration < max_iter:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # L-BFGS-based local search with SLSQP
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Try to optimize
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200, 'ftol': 1e-6})
                if result.success:
                    h = result.x
                else:
                    # Fallback to projection and normalization
                    h = project_and_normalize(h + np.random.normal(0, step_size, n_points), target_sum)
            except:
                # Can fall back to projection if optimization fails
                h = project_and_normalize(h + np.random.normal(0, step_size, n_points), target_sum)
            
            # Apply systematic perturbations
            if inner_iteration % 10 == 0:
                h = project_and_normalize(h + np.random.normal(0, step_size, n_points), target_sum)
                
            # Periodically decrease perturbation size
            if inner_iteration % 100 == 0:
                step_size *= 0.95
                
            iteration += 1
            inner_iteration += 1
            
            # Occasionally do more aggressive perturbations
            if inner_iteration % 50 == 0:
                h = project_and_normalize(h + np.random.normal(0, step_size * 2, n_points), target_sum)
    
    # Final recomputation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        h = initial_guesses[0]
        final_c5 = compute_c5(h, dx)
        best_h = h.copy()
    
    return (best_h.tolist(), final_c5, n_points)
