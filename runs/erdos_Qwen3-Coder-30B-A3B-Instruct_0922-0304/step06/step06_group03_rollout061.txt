import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, dx, target_sum):
    """Create a comb pattern with given spacing."""
    h = np.zeros(n_points)
    tooth_width = 1
    for i in range(0, n_points, spacing):
        end = min(i + tooth_width, n_points)
        h[i:end] = 1.0
    h = normalize_h(h, target_sum)
    return h

def perturb_pattern(h, sigma):
    """Add small random perturbations to pattern."""
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Initialize with known solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Start with comb patterns
    spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    for spacing in spacings:
        comb_h = create_comb_pattern(n_points, spacing, dx, target_sum)
        comb_patterns.append(comb_h.copy())
    
    # Try all comb patterns
    for comb_h in comb_patterns:
        c5 = compute_c5(comb_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = comb_h.copy()
    
    # Optimization loop
    iteration = 0
    max_iter = 50000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try perturbed versions of current best solution
        test_h = best_h.copy()
        test_h = perturb_pattern(test_h, 0.02)
        test_h = normalize_h(test_h, target_sum)
        
        # Optimize
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, test_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        if result.success:
            optimized_h = result.x
            optimized_c5 = compute_c5(optimized_h, dx)
            if optimized_c5 < best_c5:
                best_c5 = optimized_c5
                best_h = optimized_h.copy()
    
        # Occasionally try new comb patterns
        if iteration % 50 == 0:
            for spacing in spacings:
                new_h = create_comb_pattern(n_points, spacing, dx, target_sum)
                c5 = compute_c5(new_h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = new_h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final optimization
    if best_h is not None:
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 1000})
        if result.success:
            best_h = result.x
            best_c5 = compute_c5(best_h, dx)
    
    # Ensure normalization
    best_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
