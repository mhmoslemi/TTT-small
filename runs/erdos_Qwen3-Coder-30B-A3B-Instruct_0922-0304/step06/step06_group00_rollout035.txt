import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint while preserving bounds."""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    elif current_sum == 0:
        return np.full_like(h, 1.0 / len(h)) * target_sum
    else:
        # Proportional adjustment
        factor = target_sum / current_sum
        h = h * factor
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Renormalize to exact sum
        h = normalize_h(h, target_sum)
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create base function with structure
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        base_h = normalize_h(base_h, target_sum)
        base_h += np.random.normal(0, 0.02, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize each particle with slightly perturbed base solution
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Keep track of best solution
    gbest_h = None
    gbest_c5 = float('inf')
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Evaluate current particle
            c5 = compute_c5(h, dx)
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_h = h.copy()
            
            # Update particle personal best if necessary
            # For simplicity, we don't track personal best in this implementation
            # We just use global best for all particles
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (gbest_h - h) + 
                 c2 * r2 * (gbest_h - h))
            
            # Clamp velocity
            v = np.clip(v, -0.1, 0.1)
            
            # Update position
            h_new = h + v
            h_new = np.clip(h_new, 0, 1)
            h_new = project_to_constraint(h_new, target_sum)
            
            particles[i] = h_new
            velocities[i] = v
        
        if gbest_c5 <= 0.3807:
            break
        iteration += 1
    
    if gbest_h is None:
        # Fall back to base solution if no particles were evaluated
        gbest_h = base_h.copy()
        gbest_c5 = compute_c5(gbest_h, dx)
    
    return (gbest_h.tolist(), gbest_c5, n_points)
