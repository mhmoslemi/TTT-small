import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def fft_based_h(x, coeffs):
    """Reconstruct h(x) from Fourier coefficients."""
    n_freq = len(coeffs) // 2
    a0, a_coeffs, b_coeffs = coeffs[0], coeffs[1:n_freq+1], coeffs[n_freq+1:]
    
    h = a0 / 2.0
    for k in range(1, n_freq + 1):
        h += a_coeffs[k-1] * np.cos(k * np.pi * x) + b_coeffs[k-1] * np.sin(k * np.pi * x)
    return h

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_candidate(coeffs, n_points, dx, target_sum):
    """Evaluate a candidate solution in Fourier space."""
    x = np.linspace(0, 2, n_points)
    h = fft_based_h(x, coeffs)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    c5 = compute_c5(h, dx)
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using the provided construction or create a basic one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial construction
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        
    # Fourier representation for n_freq = 15 (this is half the number of coefficients)
    n_freq = 15
    n_coeffs = 2 * n_freq + 1  # a0, a_k, b_k
    
    # Initialize coefficients from h
    x_vals = np.linspace(0, 2, n_points)
    h_vals = np.clip(h_init, 0, 1)
    h_vals = normalize_h(h_vals, target_sum)
    
    # Compute initial Fourier coefficients
    a0 = np.mean(h_vals)
    a_coeffs = []
    b_coeffs = []
    for k in range(1, n_freq + 1):
        a_k = 2 * np.mean(h_vals * np.cos(k * np.pi * x_vals))
        b_k = 2 * np.mean(h_vals * np.sin(k * np.pi * x_vals))
        a_coeffs.append(a_k)
        b_coeffs.append(b_k)
    initial_coeffs = [a0] + a_coeffs + b_coeffs
    
    # Define bounds for differential evolution
    bounds = [(0, 1)] + [(-2, 2)] * (n_coeffs - 1)  # a0 in [0,1] others are unrestricted
    
    # Define objective function for DE
    def objective(coeffs):
        if time.time() - start_time > budget_s - 1:
            return float('inf')
        return evaluate_candidate(coeffs, n_points, dx, target_sum)
        
    # Differential evolution to search the Fourier space
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False
    )
    
    final_coeffs = de_result.x
    
    # Refine once with local optimizer (SLSQP)
    x = np.linspace(0, 2, n_points)
    h_final = fft_based_h(x, final_coeffs)
    h_final = np.clip(h_final, 0, 1)
    h_final = normalize_h(h_final, target_sum)
    
    # Local refinement using SLSQP
    def refine_objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(refine_objective, h_final, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    
    # Double-check final c5
    return (h_final.tolist(), final_c5, n_points)
