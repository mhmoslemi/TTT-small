import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple projection by scaling
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    else:
        h0 = np.random.rand(n_points)
        h0 = normalize_h(h0, target_sum)
    
    # Initialize swarm
    for _ in range(n_particles):
        h = h0.copy()
        # Add some noise
        if 'initial_h_values' in globals():
            h += np.random.normal(0, 0.02, n_points)
        else:
            h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest_positions.append(h.copy())
        pbest_values.append(compute_c5(h, dx))
    
    gbest_position = pbest_positions[np.argmin(pbest_values)].copy()
    gbest_value = min(pbest_values)
    
    # PSO parameters
    inertia_w_start = 0.9
    inertia_w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    max_iter = 1000
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        inertia_w = inertia_w_start - (inertia_w_start - inertia_w_end) * (iterations / max_iter)
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest_positions[i] - h)
            social = c2 * r2 * (gbest_position - h)
            v = inertia_w * v + cognitive + social
            
            # Clamp velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h = h + v
            
            # Apply bounds and constraint
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Store updated particle
            particles[i] = h
            velocities[i] = v
            
            # Evaluate
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = h.copy()
                
                # Update global best
                if c5 < gbest_value:
                    gbest_value = c5
                    gbest_position = h.copy()
            
            # Early termination
            if gbest_value <= 0.3807:
                break
                
        iterations += 1
    
    # Return best solution
    final_c5 = compute_c5(gbest_position, dx)
    return (gbest_position.tolist(), final_c5, n_points)
