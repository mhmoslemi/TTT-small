import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project(h):
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
        return h

    def compute_gradient(h, k_star):
        # Compute gradient for the current maximum lag k_star
        g = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at lag k_star, we only need h[i] * (1 - h[(i+k_star)%n_points])
            # and (1 - h[i]) * h[(i-k_star)%n_points] for the contribution to the gradient
            # But since we look at max overlap, we focus on the contribution to that specific term
            # Let's compute exact gradient via finite difference, but we speed it up with direct formula
            # This is tricky for discrete case, apply formula:
            # For overlap at lag k*, we have:
            # overlap[k*] = sum h[i] * (1 - h[(i+k*)%n])
            # So partial derivative wrt h[i] gives:
            # d/dh[i] = 1 - h[(i+k*)%n] 
            # But we are trying to minimize max overlap, so we consider this at k* which 
            # achieved max in the current step
            
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                # wrap around for circular convolution
                wrapped_idx = i + k_star - n_points
                g[i] = (1 - h[wrapped_idx]) * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 50000
    iteration = 0
    base_step_size = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap values for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient
        g = compute_gradient(h, max_k)
        
        # Update using momentum-based gradient descent with adaptive step size
        step_size = base_step_size * (0.99 ** (iteration // 1000))
        
        # Update with momentum
        v = momentum * g_prev + g
        h = h - step_size * v
        
        # Apply projection to maintain constraints
        h = project(h)
        
        # Update momentum term
        g_prev = v
        
        # Occasionally perturb to avoid local minima
        if iteration % 100 == 0 and np.random.rand() < 0.05:
            pert = np.random.normal(0, 0.002, n_points)
            h = h + pert
            h = project(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
