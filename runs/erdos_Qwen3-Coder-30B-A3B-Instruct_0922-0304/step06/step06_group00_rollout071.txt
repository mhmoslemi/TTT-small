```python
import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Define and solve SDP relaxation
    try:
        # Variables
        h_var = Variable(n_points)
        C5_var = Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Generate shift indices
        shifts = range(-n_points + 1, n_points)
        
        # Add overlap constraints
        for k in shifts:
            # Construct matrix representation of shift k
            shift_expr = []
            for i in range(n_points):
                j = (i - k) % n_points
                shift_expr.append(h_var[j])
            
            # Overlap constraint
            overlap_expr = cp.multiply(h_var, 1 - cp.hstack(shift_expr))
            constraints.append(cp.sum(overlap_expr) * dx <= C5_var)
        
        # Objective
        objective = Minimize(C5_var)
        prob = Problem(objective, constraints)
        
        # Solve with MOSEK
        prob.solve(solver=MOSEK, verbose=False)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_bound = C5_var.value
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h_opt
                    
    except Exception as e:
        pass
    
    # Refinement with local optimization
    iterations = 0
    max_iter = 1000
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Define constraint function for optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        # Bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Local optimization with SLSQP
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            best_h = h_refined
            
        # Evaluate after refinement
        current_c5 = compute_c5(best_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            # Random perturbation
            mutation_strength = max(0.001, 0.05 * np.exp(-iterations / 100))
            best_h += np.random.normal(0, mutation_strength, n_points)
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
        
        iterations += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h
```
