import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def build_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Build a comb-like h function with specified parameters."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        pos_start = i * (tooth_width + spacing)
        pos_end = pos_start + tooth_width
        
        # Find indices
        idx_start = int(pos_start / dx)
        idx_end = int(pos_end / dx)
        
        if idx_start >= n_points:
            break
            
        idx_end = min(idx_end, n_points)
        if idx_end <= idx_start:
            continue
            
        # Create triangular transition
        x_vals = np.arange(idx_start, idx_end) * dx
        x_vals = x_vals - pos_start - tooth_width/2.0
        
        # Triangle shape
        triangle = 1.0 - np.abs(x_vals) / slope
        triangle = np.clip(triangle, 0, 1)
        
        # Apply to h
        h[idx_start:idx_end] = triangle[:len(h[idx_start:idx_end])]
    
    return h

def optimize_comb_parameters(n_points, max_iter=1000, seed=42):
    """Optimize comb parameters to minimize C5."""
    np.random.seed(seed)
    dx = 2.0 / n_points
    
    best_c5 = float('inf')
    best_h = None
    best_params = None
    
    # Sweeping search over parameters
    param_search = [
        {'n_teeth': 5, 'tooth_width': 0.2, 'spacing': 0.04, 'slope': 0.1},
        {'n_teeth': 6, 'tooth_width': 0.2, 'spacing': 0.03, 'slope': 0.1},
        {'n_teeth': 7, 'tooth_width': 0.2, 'spacing': 0.028, 'slope': 0.1},
        {'n_teeth': 8, 'tooth_width': 0.2, 'spacing': 0.02, 'slope': 0.1},
    ]
    
    # Add some random variations
    for _ in range(100):
        n_teeth = np.random.randint(5, 10)
        tooth_width = np.random.uniform(0.1, 0.3)
        spacing = np.random.uniform(0.01, 0.05)
        slope = np.random.uniform(tooth_width/5, tooth_width/2)
        
        param_search.append({
            'n_teeth': n_teeth,
            'tooth_width': tooth_width,
            'spacing': spacing,
            'slope': slope
        })
    
    for params in param_search:
        h = build_comb_h(n_points, **params)
        # Normalize
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points/2) / h_sum
        
        h = np.clip(h, 0, 1)
        
        # Evaluate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_params = params.copy()
            
    return best_h, best_c5, best_params

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Check for initial construction
    if 'initial_h_values' in globals():

        h, _, _ = optimize_comb_parameters(n_points, seed=seed)
    
    # Normalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points/2) / h_sum
        
    h = np.clip(h, 0, 1)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iteration = 0
    max_iter = 1000
    
    # Local optimization with SLSQP
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Refine using local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - n_points/2
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds,
                          constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if h_new_sum > 0:
                h_new = h_new * (n_points/2) / h_new_sum
            h = h_new.copy()

        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Perturb for escape
        if iteration % 5 == 0:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points/2) / h_sum

        iteration += 1
    
    # Final optimization with refined structure
    if time.time() - start_time < budget_s - 1:
        h_final, c5_final, _ = optimize_comb_parameters(n_points, seed=seed + 1)
        if h_final is not None:
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
