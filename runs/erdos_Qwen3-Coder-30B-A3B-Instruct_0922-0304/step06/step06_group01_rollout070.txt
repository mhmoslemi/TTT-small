import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Use a more sophisticated optimization approach with genetic search
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES inspired population-based search
    population_size = min(140, 2 * n_points)
    lambda_ = population_size // 2
    
    # Initialize the population with random variations around h
    population = []
    for _ in range(population_size):
        h_pert = h + np.random.normal(0, 0.02, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = normalize_h(h_pert, target_sum)
        population.append(h_pert)
    
    iteration = 0
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute C5 for each individual
        c5_values = [compute_c5(individual, dx) for individual in population]
        
        # Select best individuals for recombination
        sorted_indices = np.argsort(c5_values)
        elite = [population[i] for i in sorted_indices[:lambda_]]
        elite_c5 = [c5_values[i] for i in sorted_indices[:lambda_]]
        
        # Update best solution
        min_c5 = min(elite_c5)
        if min_c5 < best_c5:
            best_c5 = min_c5
            best_h = elite[0].copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Create new population through recombination and mutation
        new_population = []
        
        # Keep the best solution
        new_population.append(elite[0])
        
        # Generate offspring through mutation of elites
        for _ in range(population_size - 1):
            parent = elite[np.random.randint(0, lambda_)]
            
            # Create offspring with mutation
            offspring = parent + np.random.normal(0, 0.05, n_points) 
            offspring = np.clip(offspring, 0, 1)
            offspring = normalize_h(offspring, target_sum)
            new_population.append(offspring)
            
        population = new_population
        
        # Local refinement
        if iteration % 5 == 0:
            # Select a good solution and refine it locally
            ref_h = elite[0].copy()
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, ref_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                ref_h = result.x
                ref_h = np.clip(ref_h, 0, 1)
                ref_h = normalize_h(ref_h, target_sum)
                
                c5_ref = compute_c5(ref_h, dx)
                if c5_ref < best_c5:
                    best_c5 = c5_ref
                    best_h = ref_h.copy()
        
        iteration += 1
    
    # Final refinement of the best solution
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
