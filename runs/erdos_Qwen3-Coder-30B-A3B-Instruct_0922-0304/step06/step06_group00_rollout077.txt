import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate, noise_std):
    """Mutate an individual by adding Gaussian noise."""
    noise = np.random.normal(0, noise_std, len(h))
    h_mut = h + noise
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create base function
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        base_h = normalize_h(base_h, target_sum)
    
    # Generate initial population
    for _ in range(population_size):
        if np.random.random() < 0.7:
            h = mutate_individual(base_h, 0.2, 0.05)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    max_generations = 600
    mutation_rate = 0.2
    noise_std = 0.05
    
    gen = 0
    while time.time() - start_time < budget_s - 1 and gen < max_generations:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_rate, noise_std)
            child2 = mutate_individual(child2, mutation_rate, noise_std)
            
            # Normalize with sum constraint
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to population size
        population = new_population[:population_size]
        
        # Update parameters
        mutation_rate *= 0.999
        noise_std *= 0.999
        
        gen += 1
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        final_c5 = compute_c5(population[0], dx)
    
    # Return best solution found within budget
    return (best_h.tolist() if best_h is not None else population[0].tolist(), 
            final_c5, n_points)
