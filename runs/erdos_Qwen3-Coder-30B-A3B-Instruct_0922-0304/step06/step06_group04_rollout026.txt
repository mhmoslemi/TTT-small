import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize multiple starting points
    initial_solutions = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        initial_solutions.append(base_h.copy())
    
    # Add diverse starting points
    for _ in range(10):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        initial_solutions.append(h.copy())
    
    # Add a step-like solution
    step_h = np.zeros(n_points)
    step_h[:n_points//2] = 1.0
    step_h = normalize_h(step_h, target_sum)
    initial_solutions.append(step_h.copy())
    
    # Add a smooth solution
    x = np.linspace(0, 2, n_points)
    smooth_h = 0.5 + 0.5 * np.sin(np.pi * x - np.pi/2)
    smooth_h = np.clip(smooth_h, 0, 1)
    smooth_h = normalize_h(smooth_h, target_sum)
    initial_solutions.append(smooth_h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    
    # Optimization loop
    for i, h_start in enumerate(initial_solutions):
        h = h_start.copy()
        iteration = 0
        
        while time.time() - start_time < budget_s - 1 and iteration < 50:
            # Local optimization
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 100})
                
                if result.success:
                    h = result.x
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    c5 = compute_c5(h, dx)
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_individual = h.copy()
            except:
                pass
            
            # Perturb and try again
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            iteration += 1
            
            if best_c5 <= 0.3807:
                break
    
    # If we didn't find anything better, return the initial solution
    if best_individual is None:
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
        else:
            # Default case
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
            best_individual = normalize_h(best_individual, target_sum)
        
        best_c5 = compute_c5(best_individual, dx)
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
