import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a rectangular step function
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    
    sa_iterations = 100000
    sa_step = 0
    
    # Local refinement settings
    local_refine_freq = 10000
    local_refine_iter = 200
    
    while time.time() - start_time < budget_s - 1:
        # Perturb h: modify about 10% of values
        num_modify = max(1, n_points // 10)
        indices_to_modify = np.random.choice(n_points, num_modify, replace=False)
        h_new = h.copy()
        for idx in indices_to_modify:
            h_new[idx] += np.random.normal(0, 0.02)
        
        # Ensure h remains within bounds and sum is correct
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate the new solution
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on SA criterion
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Accept with probability based on temperature
            delta = new_c5 - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
        
        # Decrease temperature
        if sa_step % 100 == 0:
            T = max(min_T, T * cooling_rate)
        
        sa_step += 1
        
        # Occasionally perform local refinement with SLSQP
        if sa_step % local_refine_freq == 0:
            # Define optimization bounds and constraints
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
            
            def objective(h_vec):
                return compute_c5(h_vec, dx)
            
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Local optimization
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': local_refine_iter})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                refined_c5 = compute_c5(h_refined, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = h_refined.copy()
                    h = h_refined.copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
