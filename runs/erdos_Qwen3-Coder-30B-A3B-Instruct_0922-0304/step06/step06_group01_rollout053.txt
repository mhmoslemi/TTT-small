import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Use a different construction: 2 ramps
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    # Generate initial particles
    for _ in range(n_particles):
        # Random Fourier coefficients
        coeffs = np.random.uniform(-0.5, 0.5, 2 * n_freq + 1)
        particles.append(coeffs)
        velocities.append(np.random.uniform(-0.1, 0.1, 2 * n_freq + 1))
        pbest.append(coeffs.copy())
        pbest_vals.append(float('inf'))

    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        for i in range(n_particles):
            # Convert to function values
            coeffs = particles[i]
            h_func = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            h_func = coeffs[0] + np.sum([
                coeffs[2*k+1] * np.cos(k * np.pi * x) + 
                coeffs[2*k+2] * np.sin(k * np.pi * x)
                for k in range(1, n_freq + 1)
            ], axis=0)
            
            # Apply nonlinear constraints
            h_func = np.clip(h_func, 0, 1)
            h_func = normalize_h(h_func, target_sum)
            
            # Compute C5
            c5_val = compute_c5(h_func, dx)
            
            # Update personal best
            if c5_val < pbest_vals[i]:
                pbest[i] = coeffs.copy()
                pbest_vals[i] = c5_val
            
            # Update global best
            if c5_val < gbest_val:
                gbest = coeffs.copy()
                gbest_val = c5_val
                
            # Update best solution found so far
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_func.copy()
                
            # Early exit
            if best_c5 <= 0.3807:
                break
                
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            
        iter_count += 1
        
        # Local refinement of global best
        if iter_count % 10 == 0 and gbest is not None:
            try:
                coeffs = gbest.copy()
                x = np.linspace(0, 2, n_points)
                h_func = coeffs[0] + np.sum([
                    coeffs[2*k+1] * np.cos(k * np.pi * x) + 
                    coeffs[2*k+2] * np.sin(k * np.pi * x)
                    for k in range(1, n_freq + 1)
                ], axis=0)
                h_func = np.clip(h_func, 0, 1)
                h_func = normalize_h(h_func, target_sum)
                
                def objective(h):
                    return compute_c5(h, dx)
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
                
                result = minimize(objective, h_func.copy(), method='SLSQP', 
                                  bounds=bounds, constraints=constraint, options={'maxiter': 50})
                
                if result.success:
                    h_func = result.x
                    h_func = np.clip(h_func, 0, 1)
                    h_func = normalize_h(h_func, target_sum)
                    c5_val = compute_c5(h_func, dx)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_func.copy()
            except:
                pass
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
