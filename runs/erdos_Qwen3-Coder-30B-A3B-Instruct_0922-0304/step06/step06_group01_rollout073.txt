import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES inspired optimization with adaptive sampling
    lambda_ = 2 * n_points
    mu = lambda_ // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Initialize GA parameters
    sigma = 0.1
    C = np.eye(n_points)
    mean = h.copy()
    p_c = np.zeros(n_points)
    p_s = np.zeros(n_points)
    c_1 = 0.1
    c_mu = 0.2
    d_sigma = 1 + 2 * max(1, np.sqrt((n_points - 1) / (n_points + 1))) + 2 * weights[0]
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample new population
        population = []
        for _ in range(lambda_):
            h_new = np.random.multivariate_normal(mean, sigma * C)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            population.append(h_new)
        
        # Evaluate population
        fitness = [compute_c5(h, dx) for h in population]
        sorted_indices = np.argsort(fitness)
        
        # Update evolutionary paths
        h_best = population[sorted_indices[0]]
        best_c5_current = fitness[sorted_indices[0]]
        
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = h_best.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Update mean
        h_new_mean = np.zeros(n_points)
        for i in range(mu):
            h_new_mean += weights[i] * population[sorted_indices[i]]
        mean = h_new_mean
        
        # Update covariance matrix
        C = (1 - c_1 - c_mu) * C
        for i in range(mu):
            v = population[sorted_indices[i]] - h_best
            C += c_mu * weights[i] * np.outer(v, v)
            
        # Update step-size
        p_s = (1 - c_1) * p_s + np.sqrt(c_1 * (2 - c_1)) * np.random.normal(0, 1, n_points)
        sigma = sigma * np.exp((c_1 / d_sigma) * (np.linalg.norm(p_s) / np.sqrt(n_points) - 1))
        
        # Local SLSQP refinement of the best solution
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                c5_refined = compute_c5(h_refined, dx)
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
        except:
            pass
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
