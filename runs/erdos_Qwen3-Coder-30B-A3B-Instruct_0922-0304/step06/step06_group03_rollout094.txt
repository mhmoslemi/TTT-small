import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, dx, target_sum, high_overlap_regions):
    """Apply perturbations to h, focusing on high overlap regions."""
    n_points = len(h)
    perturbation = np.zeros(n_points)
    
    # Apply larger perturbation to high overlap regions
    for idx in high_overlap_regions:
        if 0 <= idx < n_points:
            perturbation[idx] += np.random.normal(0, 0.03)
    
    # Apply some random noise to other regions
    mask = np.random.random(n_points) > 0.7
    perturbation += mask * np.random.normal(0, 0.01, n_points)
    
    h = h + perturbation
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    return h

def identify_high_overlap_regions(h, dx):
    """Identify regions that contribute most to overlap."""
    # Compute full cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # Find the top 10% of peaks
    peak_indices = np.argpartition(corr, -len(corr)//10)[-len(corr)//10:]
    # Get only the indices that are within the meaningful range of overlap
    valid_indices = [i for i in peak_indices if 0 <= i < len(h)]
    return list(set(valid_indices))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with available parent if it exists
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization: comb pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000
    
    # Perturbation amplitude control
    pert_amp = 0.03
    pert_decay = 0.95
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Identify high overlap regions
        high_overlap_regions = identify_high_overlap_regions(h, dx)
        
        # Apply perturbation
        h = perturb_h(h, dx, target_sum, high_overlap_regions)
        
        # Local optimization with SLSQP
        def objective(h_var):
            return compute_c5(h_var, dx)
            
        def constraint_sum(h_var):
            return np.sum(h_var) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            h = result.x
        else:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Occasionally do more aggressive search
        if iteration % 100 == 0:
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Adjust perturbation amplitude
        pert_amp *= pert_decay
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
