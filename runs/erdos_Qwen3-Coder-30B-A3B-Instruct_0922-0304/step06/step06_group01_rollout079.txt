import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try a mix of optimization strategies
    strategies = [
        # Strategy 1: Gradient-based with refined bounds
        lambda h, dx, target_sum, best_h, best_c5: (
            minimize(
                lambda x: compute_c5(np.clip(x, 0, 1), dx),
                np.clip(h, 0, 1),
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum},
                options={'maxiter': 200, 'ftol': 1e-6}
            ).x if time.time() - start_time < budget_s - 5 else h
        ),
        # Strategy 2: Simple random perturbations
        lambda h, dx, target_sum, best_h, best_c5: (
            np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            if time.time() - start_time < budget_s - 5 else h
        )
    ]
    
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Select a random strategy
        strategy = np.random.choice(strategies)
        
        h_new = strategy(h, dx, target_sum, best_h, best_c5)
        
        # Normalize and clip the new h to ensure constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Compute the new C5 value
        c5 = compute_c5(h_new, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Update h to the new one
        h = h_new.copy()
        
        iteration += 1

    # Final refinement with more precise method if time allows
    if time.time() - start_time < budget_s - 2:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, np.clip(best_h, 0, 1), method='SLSQP', 
                         bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            final_c5 = compute_c5(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
