import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x)."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0  # DC component
    for k in range(1, len(coeffs) // 2):
        h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    # DC component
    a0 = np.sum(h) * 2.0 / len(h)
    coeffs = [a0]
    for k in range(1, n_freq):
        ak = 2.0 * np.sum(h * np.cos(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        bk = 2.0 * np.sum(h * np.sin(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        coeffs.extend([ak, bk])
    return coeffs

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Frequency domain representation size

    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    
    # Set up bounds for differential evolution
    bounds = [(-2.0, 2.0) for _ in range(2 * n_freq)]  # Fourier coefficients
    
    def objective_fourier(coeffs):
        # Convert to h
        h = fourier_to_h(coeffs, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return compute_c5(h, dx)

    # Initialize differential evolution
    if initial_h is not None:
        # Convert initial_h to Fourier coefficients to use as initial guess
        init_coeffs = h_to_fourier(initial_h, n_freq)
        # Slightly perturb initial guess to avoid local minima
        init_coeffs = [c + np.random.normal(0, 0.05) for c in init_coeffs]
    else:
        # Default initialization
        init_coeffs = [0.0] * (2 * n_freq)
        init_coeffs[0] = 1.0  # Ensure DC component is non-zero
        
    try:
        # Differential Evolution
        de_result = differential_evolution(
            objective_fourier, 
            bounds,
            maxiter=300,
            popsize=75,
            mutation=(0.5, 1),
            recombination=0.7,
            seed=seed,
            callback=lambda x, convergence: None  # Placeholder callback
        )
        
        h_final = fourier_to_h(de_result.x, n_points)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
        
        # Local refinement
        def objective_h(h):
            return compute_c5(h, dx)
        
        bounds_h = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
        
        result = minimize(
            objective_h,
            h_final.copy(),
            method='SLSQP',
            bounds=bounds_h,
            constraints=constraint,
            options={'maxiter': 100}
        )
        
        if result.success:
            h_final = result.x.copy()
        else:
            h_final = h_final.copy()
            
        final_c5 = compute_c5(h_final, dx)
        
        return (h_final.tolist(), final_c5, n_points)
        
    except Exception as e:
        # Fallback to initial_h or a simple heuristic
        if initial_h is not None:
            h_final = initial_h.copy()
        else:
            h_final = np.ones(n_points)
            h_final[n_points//2:] = 0
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (target_sum / np.sum(h_final))
        
        final_c5 = compute_c5(h_final, dx)
        return (h_final.tolist(), final_c5, n_points)
