import time
from scipy.signal import correlate

    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    stagnation_count = 0
    stagnation_limit = 500
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Make a small perturbation
        h_candidate = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Ensure valid indices for the perturbation
        if h[i] >= 1 or h[j] <= 0:
            continue
            
        # Compute delta to move
        delta = min(1 - h[i], h[j])
        delta *= np.random.rand()  # Random fraction
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Renormalize
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Compute C5 for candidate
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            best_h = h.copy()
            stagnation_count = 0
        else:
            # Accept with Metropolis probability
            if np.random.rand() < np.exp(-(c5_candidate - best_c5) / T):
                h = h_candidate
            stagnation_count += 1
            
        # Adaptive cooling
        if stagnation_count > stagnation_limit:
            T *= 1.2  # Increase temperature to escape local minima
            stagnation_count = 0
        else:
            T *= alpha
            
        if T < 1e-5:
            break
            
        # Occasionally restore the best solution to avoid getting stuck
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            h = best_h.copy()
            
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
