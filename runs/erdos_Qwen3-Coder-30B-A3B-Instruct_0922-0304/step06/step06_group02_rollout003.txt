import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with potential better starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a better starting point based on insight of minimizing overlap
        h = np.zeros(n_points)
        # Place a peak at center and taper edges to reduce correlation
        mid = n_points // 2
        for i in range(n_points):
            dist = abs(i - mid)
            h[i] = max(0, 1.0 - dist / (n_points / 4))
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use hybrid approach: alternating between global (PSO-like) and local search
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Apply perturbations with adaptive strength
        perturbation_strength = max(0.01, 0.05 * (1 - (time.time() - start_time) / budget_s))
        
        # Local refinement using L-BFGS-B
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
            
        # Local search with perturbation add
        h_perturbed = h + np.random.normal(0, perturbation_strength, len(h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        try:
            result = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5:
                    h = h_candidate.copy()
        except:
            pass
            
        # Periodic global search to escape local minima
        if iteration % 50 == 0:
            # Random perturbation to explore diversity
            h = np.clip(h + np.random.normal(0, 0.02, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            # Occasional full restart leaning towards initial pattern
            if iteration % 200 == 0:
                h = h * 0.8 + (np.array(initial_h_values) if 'initial_h_values' in globals() else np.clip(np.random.rand(n_points), 0, 1)) * 0.2
                h = normalize_h(h, target_sum)
                
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
