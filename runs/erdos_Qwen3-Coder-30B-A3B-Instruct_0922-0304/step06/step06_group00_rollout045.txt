import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Precompute shift indices
    shift_indices = []
    for k in range(-n_points + 1, n_points):
        shift_indices.append(k)
    
    # Try to solve via SDP
    try:
        # Create variables
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C5)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h) == target_sum)
        
        # Bounds
        constraints.extend([h[i] >= 0 for i in range(n_points)])
        constraints.extend([h[i] <= 1 for i in range(n_points)])
        
        # Overlap constraints for all shifts
        for k in shift_indices:
            # Create shifted indices (handle wrap-around)
            shifted_indices = [(i - k) % n_points for i in range(n_points)]
            # Compute sum: h[i] * (1 - h[(i-k) % n_points])
            overlap_sum = 0
            for i in range(n_points):
                overlap_sum += h[i] * (1 - h[shifted_indices[i]])
            constraints.append(overlap_sum * dx <= C5)
        
        # Form and solve problem
        prob = cp.Problem(objective, constraints)
        
        # Try to solve with different solvers
        solver = cp.MOSEK
        prob.solve(solver=solver, verbose=False, time_limit=budget_s - 10)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C5.value
            # Double-check solution
            if h_opt is not None and c5_opt is not None:
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                final_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode="full")) * dx
                return (h_opt.tolist(), final_c5, n_points)
    except:
        pass

    # Fallback to previous method if SDP fails
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    best_h = h.copy()

    iterations = 0
    max_iter = 2000
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Local optimization
        def objective(h):
            overlap = correlate(h, 1 - h, mode="full")
            return -np.max(overlap) * dx
        
        # Constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            h = h_new
        
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Mutation
        mutation_strength = max(0.001, 0.05 * np.exp(-iterations / 300.0))
        h += np.random.normal(0, mutation_strength, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iterations += 1
    
    return (best_h.tolist(), best_c5, n_points)
