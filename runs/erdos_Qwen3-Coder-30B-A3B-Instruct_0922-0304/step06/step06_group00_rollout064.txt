import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: adjust by proportionally scaling
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create base function that's structured
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        # Some small perturbation to start
        h += np.random.normal(0, 0.01, n_points)
        h = project_to_constraints(h, target_sum)
    
    # PSO parameters
    num_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(num_particles):
        if 'initial_h_values' in globals():
            particle = h.copy()
        else:
            particle = np.random.rand(n_points)
            particle = project_to_constraints(particle, target_sum)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(particle.copy())
        pbest_values.append(compute_c5(particle, dx))
    
    # Global best tracking
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * (iterations / max_iter)
        
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_constraints(particles[i], target_sum)
            
            # Evaluate
            current_value = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
            
            # Update global best
            if current_value < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_value
                
        # Early termination
        if gbest_value <= 0.3807:
            break
            
        iterations += 1
    
    final_c5 = compute_c5(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
