import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiple initial guesses
    initial_guesses = []
    
    # Current best
    initial_guesses.append(h.copy())
    
    # Comb-like pattern
    comb_h = np.zeros(n_points)
    for i in range(0, n_points, 3):
        if i < n_points:
            comb_h[i] = 1.0
    comb_h = normalize_h(comb_h, target_sum)
    initial_guesses.append(comb_h)
    
    # Random with constraint
    random_h = np.random.rand(n_points)
    random_h = normalize_h(random_h, target_sum)
    initial_guesses.append(random_h)
    
    # Alternating pattern
    alt_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            alt_h[i] = 1.0
    alt_h = normalize_h(alt_h, target_sum)
    initial_guesses.append(alt_h)
    
    # Low discrepancy
    halton_h = np.array([((i * 1.618033988749895) % 1) for i in range(n_points)])
    halton_h = normalize_h(halton_h, target_sum)
    initial_guesses.append(halton_h)
    
    max_iter = 10000
    iter_count = 0
    perturbation_std = 0.02
    decay_factor = 0.95
    
    # Optimization with several initial points and adaptive perturbations
    for trial in range(len(initial_guesses)):
        h = initial_guesses[trial].copy()
        
        for i in range(max_iter):
            if time.time() - start_time >= budget_s - 1:
                break
                
            iter_count += 1
            
            # Evaluate current solution
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
            # Use strategy with multiple solver calls with different perturbations
            if i % 20 == 0:
                perturbation_std = max(0.001, perturbation_std * decay_factor)
                
            # Small perturbations often improve things
            h_new = h + np.random.normal(0, perturbation_std, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h = h_new
            
            # SLSQP optimization with bounds and constraints
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                h = h_new
            
            # Occasionally do more aggressive search
            if i % 100 == 0 and perturbation_std > 0.005:
                h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
    # Final evaluation for best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
