import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Make small perturbations to escape local minima
        h_perturbed = h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Re-normalize
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
            
        # Evaluate the perturbed solution
        perturbed_c5 = compute_c5(h_perturbed, dx)
        if perturbed_c5 < current_c5:
            h = h_perturbed.copy()
            
        # Occasionally do global search
        if iteration % 50 == 0:
            # Try a different initialization approach
            h_new = np.zeros(n_points)
            # Place a bump in the middle
            mid = n_points // 2
            half_width = n_points // 6
            h_new[mid - half_width:mid + half_width] = 1.0
            # Normalize
            current_sum = np.sum(h_new)
            if current_sum > 0:
                h_new = h_new * (target_sum / current_sum)
            h_new = np.clip(h_new, 0, 1)
            
            # Evaluate
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new.copy()
                
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
