import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Discretization levels
    levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def discretize(h_vals):
        return np.array([levels[np.argmin(np.abs(h_vals[i] - levels))] for i in range(len(h_vals))])
    
    # Local search with discrete steps
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try local perturbations
        h_new = h.copy()
        
        # Randomly select some indices to modify
        n_modify = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_modify, replace=False)
        
        for i in indices:
            # Try to change the value
            current_val = h_new[i]
            # Create a neighbor value (discrete)
            neighbors = levels[np.abs(levels - current_val) > 1e-6]
            if len(neighbors) > 0:
                new_val = np.random.choice(neighbors)
                h_new[i] = new_val
                
        # Ensure normalized sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        h_new = np.clip(h_new, 0, 1)
        
        # Discretize the result
        h_new = discretize(h_new)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        h = h_new
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
