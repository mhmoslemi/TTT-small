import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region respecting bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust values to meet sum exactly
        diff = target_sum - current_sum
        # Find indices where we can add diff without violating bounds
        indices = np.argsort(h)  # sort by value to add to smallest first
        h_copy = h.copy()
        for i in indices:
            if diff > 0:
                can_add = min(diff, 1.0 - h_copy[i])
                h_copy[i] += can_add
                diff -= can_add
            elif diff < 0:
                can_subtract = min(-diff, h_copy[i])
                h_copy[i] -= can_subtract
                diff += can_subtract
            else:
                break
        h = h_copy
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # PSO parameters
    n_particles = 50
    max_iter = 1000
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1

    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    else:
        h0 = np.random.rand(n_points)
        h0 = normalize_h(h0, target_sum)
    
    # Create initial particles around h0
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))

    # Initialize personal bests and global best
    pbests = [p.copy() for p in particles]
    pbest_values = [compute_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_values)
    gbest = pbests[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]

    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (pbests[i] - h) + 
                 c2 * r2 * (gbest - h))
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            h_new = h + v
            
            # Project to feasible set
            h_new = project_to_feasible(h_new, target_sum)
            
            # Evaluate new solution
            new_value = compute_c5(h_new, dx)
            
            # Update personal best
            if new_value < pbest_values[i]:
                pbest_values[i] = new_value
                pbests[i] = h_new.copy()
                
                # Update global best if needed
                if new_value < gbest_value:
                    gbest_value = new_value
                    gbest = h_new.copy()
            
            # Update particle and velocity
            particles[i] = h_new
            velocities[i] = v
            
            # Early stopping
            if gbest_value <= 0.3807:
                break
        
        if gbest_value <= 0.3807:
            break
    
    # Final check with best solution
    final_c5 = gbest_value
    return (gbest.tolist(), final_c5, n_points)
