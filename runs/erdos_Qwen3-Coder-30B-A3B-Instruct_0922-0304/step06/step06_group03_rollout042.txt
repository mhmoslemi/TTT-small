import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a piecewise constant function with a step
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameter for perturbation amplitude
    perturbation_amplitude = 0.02
    iteration = 0
    max_iter = 30000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Periodically adjust perturbation amplitude
        if iteration % 500 == 0 and iteration > 0:
            perturbation_amplitude *= 0.95
            
        # Perturb based on cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # Identify the locations of peaks in the cross-correlation
        peak_indices = np.argsort(corr)[-10:]  # Top 10 peaks
        
        # Create mask that influences high overlap regions
        mask = np.zeros(n_points)
        for idx in peak_indices:
            # Expand around the peak to influence nearby regions
            start = max(0, idx - 2)
            end = min(n_points, idx + 2)
            mask[start:end] += 1
            
        # Normalize the mask
        mask = mask / (np.max(mask) + 1e-8)
        
        # Add noise influenced by the mask
        noise = np.random.normal(0, perturbation_amplitude, n_points)
        noise = noise * mask
        
        # Apply to current h
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization step
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Occasionally do random perturbations
        if iteration % 20 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # SLSQP optimization
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else:
                # If optimization failed, perturb and try again
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        except:
            # Fallback if optimization fails
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
