import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple step function as initialization
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    initial_temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    
    # SA loop
    iteration = 0
    sa_steps = 100000
    sa_step_count = 0
    
    temp = initial_temp
    current_c5 = compute_c5(h, dx)
    
    while time.time() - start_time < budget_s - 1 and sa_step_count < sa_steps:
        # Perturb h
        h_new = h.copy()
        num_perturb = max(1, int(0.1 * n_points))  # Perturb about 10% of entries
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for idx in indices:
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.05)
            h_new[idx] += perturbation
        
        # Ensure h_new is in [0, 1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint by proportional redistribution
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < math.exp(-delta / temp):
                h = h_new
                current_c5 = new_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stop condition
        if best_c5 <= 0.3807:
            break
            
        # Cool down temperature
        temp = max(min_temp, temp * cooling_rate)
        sa_step_count += 1
        
        # Every 100 steps, log progress
        if sa_step_count % 100 == 0:
            if time.time() - start_time > budget_s - 2:
                break
    
    # Local refinement with SLSQP after SA
    if time.time() - start_time < budget_s - 2:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Refine with SLSQP
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, options={'maxiter': 200})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = refined_h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
