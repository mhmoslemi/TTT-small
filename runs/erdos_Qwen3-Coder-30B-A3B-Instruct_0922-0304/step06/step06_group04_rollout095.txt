import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create a reasonable starting point
        base_h = np.zeros(n_points)
        # Place a spike in the middle with appropriate normalization
        mid = n_points // 2
        base_h[mid-5:mid+5] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Alternating optimization approach
    iteration = 0
    max_iter = 1000
    sigma = 0.03
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Perturb solution
        h = best_individual.copy()
        noise = np.random.normal(0, sigma, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Optimization to reduce C5
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200, 'ftol': 1e-6})
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h.copy()
                
                # Cool down the perturbation
                sigma = max(0.001, sigma * 0.995)
            else:
                # If optimization fails, try a simpler approach
                sigma = min(0.1, sigma * 1.1)
        except:
            sigma = min(0.1, sigma * 1.1)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
