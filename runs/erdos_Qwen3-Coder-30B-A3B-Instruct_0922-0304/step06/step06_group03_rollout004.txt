import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guess(n_points, method="comb"):
    """Generate initial guess based on method."""
    target_sum = n_points / 2
    
    if method == "comb":
        h = np.zeros(n_points)
        step = max(1, n_points // 10)
        for i in range(0, n_points, step):
            h[i] = 1.0
        h = normalize_h(h, target_sum)
    elif method == "random":
        h = np.random.random(n_points)
        h = normalize_h(h, target_sum)
    elif method == "alternating":
        h = np.ones(n_points)
        h[1::2] = 0
        h = normalize_h(h, target_sum)
    elif method == "halton":
        # Simple quasi-random using Halton-like approach
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = (i * 0.618033988749895) % 1  # golden ratio
        h = normalize_h(h, target_sum)
    else:
        # Default: uniform
        h = np.ones(n_points)
        h = normalize_h(h, target_sum)
    
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Best solution found so far
    best_c5 = float('inf')
    best_h = None
    
    # Try multiple initializations
    initial_methods = ["comb", "random", "alternating", "halton"]
    
    for method in initial_methods:
        if time.time() - start_time > budget_s - 1:
            break
          
        # Initialize h
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            h = generate_initial_guess(n_points, method)
        
        # Optimization loop
        iteration = 0
        max_iter = 5000
        perturb_std = 0.02
        perturb_decay = 0.95
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
            # SLSQP optimization step
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h = result.x
                else:
                    # If optimization fails, use perturbation
                    h = h + np.random.normal(0, perturb_std, n_points)
            except:
                # Fallback
                h = h + np.random.normal(0, perturb_std, n_points)
                
            # Project h back into feasible region
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Occasionally apply larger perturbations
            if iteration % 50 == 0:
                h = np.clip(h + np.random.normal(0, perturb_std * 2, n_points), 0, 1)
                h = normalize_h(h, target_sum)
            
            # Gradually decrease perturbation magnitude
            if iteration % 100 == 0:
                perturb_std *= perturb_decay
                
            iteration += 1
    
    # If no good solution found, fall back to initial_h_values or uniform
    if best_h is None:
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.ones(n_points)
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
