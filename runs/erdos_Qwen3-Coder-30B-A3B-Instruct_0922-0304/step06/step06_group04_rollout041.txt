import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    # Store best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with line search
    max_iter = 1000
    step_size = 0.01
    max_backtrack = 10
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient estimate via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update step
        new_h = h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        
        # Backtracking line search
        c5_new = compute_c5(new_h, dx)
        alpha = 1.0
        beta = 0.5
        c1 = 1e-4
        while alpha > 1e-10:
            test_h = h - alpha * grad
            test_h = np.clip(test_h, 0, 1)
            test_h = normalize_h(test_h, target_sum)
            test_c5 = compute_c5(test_h, dx)
            
            if test_c5 <= base_c5 - c1 * alpha * np.dot(grad, grad):
                new_h = test_h
                c5_new = test_c5
                break
            alpha *= beta
        
        # Accept new point if better
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = new_h.copy()
        
        # Move to new point
        h = new_h
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
