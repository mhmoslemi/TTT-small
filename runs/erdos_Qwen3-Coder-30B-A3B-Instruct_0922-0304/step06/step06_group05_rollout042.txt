import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_limit = 500
    stagnation_count = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Make a perturbation
        h_candidate = h.copy()
        
        # Select two indices for adjustment
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Determine amount to move
        delta = min(1 - h_candidate[i], h_candidate[j])
        if delta > 0:
            # Apply perturbation
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize to sum to n_points / 2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            c5_new = compute_c5(h_candidate)
            
            # Compute current C5
            c5_current = compute_c5(h)
            
            # Metropolis acceptance criterion
            if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
                h = h_candidate
                c5_current = c5_new
            
            # Update best solution
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
            
            # Adaptive cooling
            if stagnation_count > stagnation_limit:
                T *= 1.2  # Increase temperature to escape local minima
                stagnation_count = 0
                
            # Decrease temperature
            T *= alpha
            if T < min_T:
                break
                
        # Occasionally perturb even if not accepted to keep exploring
        if iteration % 100 == 0:
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
