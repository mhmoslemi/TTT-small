import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, offset=0):
    """Create a comb pattern with given spacing and offset."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_width = dx
    tooth_height = 1.0
    
    # Determine number of teeth that fit in [0, 2]
    max_teeth = int(2.0 / (spacing * dx)) + 1
    
    # Place teeth
    for i in range(max_teeth):
        pos = offset + i * spacing * dx
        if pos < 2.0:
            start_idx = int(pos / dx)
            if start_idx < n_points:
                h[start_idx] = tooth_height
    
    # Normalize
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try predefined comb patterns
    spacings = [2, 3, 4, 5, 6]
    for spacing in spacings:
        # Try different offsets
        for offset in [0, 0.5, 1.0, 1.5]:
            h_comb = create_comb_pattern(n_points, spacing, offset)
            c5 = compute_c5(h_comb, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_comb.copy()
            if best_c5 <= 0.3807:
                break
        if best_c5 <= 0.3807:
            break
    
    # Fine-tune the best comb pattern
    original_comb = best_h.copy()
    
    # Some local search with SLSQP
    max_iter = 5000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Run SLSQP optimization on comb pattern
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda h: np.sum(h) - target_sum}
        
        # Try optimizing with small random perturbations
        attempt_h = original_comb + np.random.normal(0, 0.01, n_points)
        attempt_h = np.clip(attempt_h, 0, 1)
        attempt_h = normalize_h(attempt_h, target_sum)
        
        def objective(h):
            return compute_c5(h, dx)
        
        result = minimize(objective, attempt_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            attempt_h = result.x
        else:
            attempt_h = np.clip(attempt_h + np.random.normal(0, 0.01, n_points), 0, 1)
            attempt_h = normalize_h(attempt_h, target_sum)
        
        c5 = compute_c5(attempt_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = attempt_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
