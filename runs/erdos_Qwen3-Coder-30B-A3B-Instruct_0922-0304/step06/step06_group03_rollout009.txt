import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Start from perturbed version of best solution
            particle = h + np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        c5 = compute_c5(particle, dx)
        p_best.append(particle.copy())
        p_best_c5.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = particle.copy()
    
    # PSO iterations
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (best_h - particle))
            
            # Update position
            particle = particle + vel
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = vel
            
            # Evaluate fitness
            c5 = compute_c5(particle, dx)
            
            # Update personal best
            if c5 < p_best_c5[i]:
                p_best[i] = particle.copy()
                p_best_c5[i] = c5
                
                # Update global best
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = particle.copy()
            
            if best_c5 <= 0.3807:
                break
        
        # Periodic frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            for i in range(n_particles):
                h = particles[i]
                # FFT-based filtering
                H = np.fft.fft(h)
                # Apply frequency filter (low pass filter)
                freqs = np.fft.fftfreq(n_points, dx)
                filt = np.abs(freqs) > 1.0
                H_filtered = H * filt
                h_filtered = np.real(np.fft.ifft(H_filtered))
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = normalize_h(h_filtered, target_sum)
                
                # Replace worst performing particle
                worst_idx = np.argmax([compute_c5(p, dx) for p in particles])
                if compute_c5(h_filtered, dx) < compute_c5(particles[worst_idx], dx):
                    particles[worst_idx] = h_filtered
        
        # Periodic gradient refinement
        if iteration % 100 == 0 and iteration > 0:
            # Select top 10 particles
            particle_scores = [compute_c5(p, dx) for p in particles]
            top_indices = np.argsort(particle_scores)[:10]
            
            for idx in top_indices:
                p = particles[idx]
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    result = minimize(objective, p, method='SLSQP', bounds=bounds, 
                                    constraints=constraint, options={'maxiter': 100})
                    if result.success:
                        refined = result.x
                        refined = np.clip(refined, 0, 1)
                        refined = normalize_h(refined, target_sum)
                        c5_refined = compute_c5(refined, dx)
                        c5_old = particle_scores[idx]
                        
                        if c5_refined < c5_old:
                            particles[idx] = refined
                            particle_scores[idx] = c5_refined
                            
                            if c5_refined < best_c5:
                                best_c5 = c5_refined
                                best_h = refined.copy()
                except:
                    pass
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
