import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Fallback: create smooth step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = project_h(h, n_points)
    
    # Adaptive parameters
    initial_eta = 0.01
    momentum = 0.9
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For momentum
    g_prev = np.zeros(n_points)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap and max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for max lag k*
        g = np.zeros(n_points)
        for i in range(n_points):
            # Contribution to derivative at position i
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            if i >= max_k:
                g[i] -= h[i - max_k] * dx

        # Adaptive step size
        eta = initial_eta * (0.5 ** (iteration // 1000))
        
        # Gradient update with momentum
        g_momentum = g + momentum * g_prev
        new_h = h - eta * g_momentum
        
        # Apply constraints
        new_h = project_h(new_h, n_points)
        
        h = new_h
        g_prev = g.copy()
        
        # Occasionally random perturbation to escape local optima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_h(h, n_points)

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
