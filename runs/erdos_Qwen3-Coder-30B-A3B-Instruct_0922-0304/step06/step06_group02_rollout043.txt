import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.05, target_sum=35.0):
    """Mutate an individual h with constraint enforcement."""
    h_mut = h.copy()
    for i in range(len(h_mut)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.05)
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def evaluate_population(population):
    """Evaluate all individuals in the population."""
    return [compute_c5(individual, dx) for individual in population]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population from initial_h_values or default
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population
    population_size = 30
    population = [base_h.copy()]
    for _ in range(population_size - 1):
        individual = mutate_individual(base_h, 0.05, target_sum)
        population.append(individual)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = evaluate_population(population)
        
        # Find best individual
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Select top 50% for reproduction
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        selected = [population[i] for i in top_indices]
        
        # Create new population
        new_population = selected.copy()
        
        # Generate offspring through crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate_individual(child1, 0.05, target_sum)
            child2 = mutate_individual(child2, 0.05, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        # Local refinement using SLSQP for top 10%
        num_refine = max(1, population_size // 10)
        top_refine_indices = sorted_indices[:num_refine]
        
        for i in top_refine_indices:
            individual = population[i]
            def objective(h):
                return compute_c5(h, dx)
            def constraint_sum(h):
                return np.sum(h) - target_sum
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Use SLSQP to refine
            result = minimize(objective, individual, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined = np.clip(result.x, 0, 1)
                refined = normalize_h(refined, target_sum)
                current_fitness = fitness[i]
                new_fitness = compute_c5(refined, dx)
                if new_fitness < current_fitness:
                    new_population[i] = refined
        
        population = new_population
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
