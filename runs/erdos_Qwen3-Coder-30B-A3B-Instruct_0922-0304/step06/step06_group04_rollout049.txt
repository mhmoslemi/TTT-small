import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Random initialization with some structure to break symmetry
        base_h = np.random.rand(n_points)
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Alternating optimization variables
    g = np.zeros(2 * n_points - 1)
    
    # Alternate between optimizing h and updating g
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Step 1: Optimize h while fixing g
        h = best_individual.copy()
        
        # Create constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            # Approximate by minimizing max g{k}
            return np.max(g)  # Use fixed g as proxy
            
        # Optimize h
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Step 2: Compute g based on this new h
            new_g = np.zeros(2 * n_points - 1)
            for k in range(2 * n_points - 1):
                overlap = 0.0
                for i in range(n_points):
                    j = i + k - (n_points - 1)  # shift index
                    if 0 <= j < n_points:
                        overlap += h[i] * (1 - h[j])
                new_g[k] = overlap
            
            # Update g with maximum values
            g = np.maximum(g, new_g)
            
            # Evaluate C5
            c5 = np.max(g) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
                
        # Perturb to escape local minima
        if iteration % 10 == 0:
            h = best_individual.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            # Re-evaluate g and c5
            new_g = np.zeros(2 * n_points - 1)
            for k in range(2 * n_points - 1):
                overlap = 0.0
                for i in range(n_points):
                    j = i + k - (n_points - 1)
                    if 0 <= j < n_points:
                        overlap += h[i] * (1 - h[j])
                new_g[k] = overlap
            g = np.maximum(g, new_g)
            c5 = np.max(g) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
