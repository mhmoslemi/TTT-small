import time

    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # Sum constraint
    sum_constraint = cp.Equality(cp.sum(h_vars), target_sum)
    
    # Lag constraints
    constraints = [sum_constraint, C >= 0]
    
    # Generate all valid lag constraints
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        if lag >= 0:
            valid_indices = range(n_points - lag)
            expr = cp.sum(cp.multiply(h_vars[valid_indices], 
                                    1 - h_vars[valid_indices + lag])) * dx
        else:
            valid_indices = range(-lag, n_points)
            expr = cp.sum(cp.multiply(h_vars[valid_indices],

    # Add bound constraints
    for i in range(n_points):
        constraints.append(h_vars[i] >= 0)
        constraints.append(h_vars[i] <= 1)

    objective = cp.Minimize(C)

    prob = cp.Problem(objective, constraints)

                solver_used = solver
                break

                h = h * (target_sum / current_sum)
