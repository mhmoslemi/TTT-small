import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to satisfy sum(h) = n_points/2
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Simple adjustment: scale all values
            h = h * (n_points / 2) / current_sum
        return h

    def compute_gradient(h, k_star):
        # Compute gradient for the maximum overlap lag k_star
        g = np.zeros_like(h)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                # Wrap around
                wrap_index = i + k_star - n_points
                g[i] = (1 - h[wrap_index]) * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically
        g = compute_gradient(h, max_k)
        
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply constraints via projection
        new_h = project_h(new_h)
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = project_h(h)
            
        # Adaptive step size
        if iteration % 1000 == 0:
            step_size *= 0.5

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
