import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum to target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        # Create starting function with structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Perturb slightly for diversity
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier optimization parameters
    mu = 1.0
    mu_decay = 0.5
    mu_min = 1e-8
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    # Gradient-based optimization
    eps = 1e-5
    
    def compute_gradient(h, dx, mu):
        """Compute gradient of C5 with barrier term."""
        # Compute C5 gradient via finite differences
        grad_c5 = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad_c5[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Barrier gradient: -1/h[i] - 1/(1-h[i]) - 1/(sum_h - target_sum)
        barrier_grad = np.zeros_like(h)
        h_sum = np.sum(h)
        for i in range(n_points):
            if h[i] > 0 and h[i] < 1 and abs(h_sum - target_sum) > 1e-10:
                barrier_grad[i] = -1.0/h[i] - 1.0/(1.0-h[i]) - 1.0/(h_sum - target_sum)
        
        # Combine
        return grad_c5 + mu * barrier_grad
    
    def line_search(h, grad, dx, mu, alpha_init=1.0):
        """Backtracking line search for sufficient decrease."""
        alpha = alpha_init
        c1 = 1e-4
        rho = 0.5
        f0 = compute_c5(h, dx) + mu * np.sum(np.log(h) + np.log(1-h) + np.log(abs(np.sum(h)-target_sum)))
        
        while alpha > 1e-10:
            h_new = h - alpha * grad
            h_new = project_to_feasible(h_new, target_sum)
            f_new = compute_c5(h_new, dx) + mu * np.sum(np.log(h_new) + np.log(1-h_new) + np.log(abs(np.sum(h_new)-target_sum)))
            if f_new <= f0 + c1 * alpha * np.dot(grad, -grad):
                return alpha, h_new
            alpha *= rho
            
        return alpha, h
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute gradient
        try:
            grad = compute_gradient(h, dx, mu)
        except:
            # Reset if gradient computation fails
            h = project_to_feasible(h, target_sum)
            mu = max(mu * 0.5, mu_min)
            continue
            
        # Line search
        alpha, h_new = line_search(h, grad, dx, mu)
        
        if alpha > 1e-10:
            h = h_new
            current_c5 = compute_c5(h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            # If line search fails, reduce mu and retry
            mu = max(mu * 0.5, mu_min)
            no_improve_count += 1
            
        # Adaptively decrease mu
        if iter_count % 50 == 0:
            if mu > mu_min:
                mu = max(mu * mu_decay, mu_min)
                
        # Restart if stuck
        if no_improve_count > max_no_improve:
            # Start with best solution and random perturbation
            h = best_h.copy()
            h += np.random.normal(0, 0.02, n_points)
            h = project_to_feasible(h, target_sum)
            no_improve_count = 0
            mu = 1.0
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
