import numpy as np
from scipy.signal import correlate
import time

def project_h(h, n_points):
    """Project h onto the feasible set: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def compute_c5(h, dx):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with a sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x)
        h = project_h(h, n_points)
    
    # Parameters
    max_iter = 10000
    eta = 0.01
    mu = 0.9
    best_c5 = float('inf')
    best_h = h.copy()
    
    g_prev = np.zeros(n_points)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find k* that maximizes it
        overlap = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(overlap)
        
        # Compute exact gradient for this k_star
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                g[i] = -h[i - (n_points - k_star)] * dx
        
        # Update h with momentum and adaptive step size
        new_h = h - eta * (g + mu * g_prev)
        g_prev = g
        
        # Project onto feasible set
        h = project_h(new_h, n_points)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Adaptive step size and momentum
        if iteration % 1000 == 0:
            eta *= 0.5
            mu *= 0.95
            
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
