import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()

    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # PSO initialization
    positions = np.random.uniform(-1, 1, (n_particles, 2*n_freq))
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, 2*n_freq))
    pbest_positions = positions.copy()
    pbest_values = np.full(n_particles, float('inf'))
    gbest_position = None
    gbest_value = float('inf')

    # Convert Fourier representation to function and evaluate
    def fourier_to_h(fourier_coeffs):
        h_vals = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_freq):
            if i == 0:
                h_vals += fourier_coeffs[i]
            else:
                h_vals += fourier_coeffs[i] * np.cos(i * np.pi * x) + \
                         fourier_coeffs[i + n_freq] * np.sin(i * np.pi * x)
        return np.clip(h_vals, 0, 1)

    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        for i in range(n_particles):
            # Convert Fourier coefficients to h function
            h_vals = fourier_to_h(positions[i])
            h_vals = normalize_h(h_vals, target_sum)
            
            # Compute C5 value
            c5 = compute_c5(h_vals, dx)
            
            # Update personal best
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = positions[i].copy()
            
            # Update global best
            if c5 < gbest_value:
                gbest_value = c5
                gbest_position = positions[i].copy()

            # Update particle position using PSO rules
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_positions[i] - positions[i]) + 
                            c2 * r2 * (gbest_position - positions[i]))
            positions[i] += velocities[i]
            
            # Keep coefficients within bounds
            positions[i] = np.clip(positions[i], -1, 1)

        # Early exit
        if gbest_value <= 0.3807:
            break

    # Final refinement using local optimization
    if gbest_position is not None:
        h_final = fourier_to_h(gbest_position)
        h_final = normalize_h(h_final, target_sum)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vals):
            return compute_c5(h_vals, dx)
            
        def constraint_sum(h_vals):
            return np.sum(h_vals) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h_final.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            
        final_c5 = compute_c5(h_final, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()

    return (best_h.tolist(), best_c5, n_points)
