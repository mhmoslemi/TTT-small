import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while staying in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: adjust values proportionally
    if current_sum > 0:
        factor = target_sum / current_sum
        h = h * factor
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    else:
        h0 = np.random.rand(n_points)
        h0 = normalize_h(h0, target_sum)
    
    for _ in range(n_particles):
        h = h0 + np.random.normal(0, 0.02, n_points)
        h = project_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Initialize personal bests and global best
    pbests = [p.copy() for p in particles]
    pbest_scores = [compute_c5(p, dx) for p in particles]
    
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbests[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_iter = 1000
    velocity_clamp = 0.1
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * iterations / max_iter
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Velocity update
            v = (w * v + 
                 c1 * r1 * (pbests[i] - h) + 
                 c2 * r2 * (gbest - h))
            
            # Clamp velocity
            v = np.clip(v, -velocity_clamp, velocity_clamp)
            
            # Update position
            h_new = h + v
            
            # Apply constraints
            h_new = project_h(h_new, target_sum)
            
            # Evaluate new position
            new_score = compute_c5(h_new, dx)
            
            # Update personal best
            if new_score < pbest_scores[i]:
                pbest_scores[i] = new_score
                pbests[i] = h_new.copy()
                
                # Update global best
                if new_score < gbest_score:
                    gbest_score = new_score
                    gbest = h_new.copy()
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
        
        # Early stopping
        if gbest_score <= 0.3807:
            break
            
        iterations += 1
    
    # Final refinement of the best solution
    h_final = gbest
    c5_final = compute_c5(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
