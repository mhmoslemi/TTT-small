import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize from parent or default
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points) * 0.5
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Bilinear programming approach with CVXPY
    solver = cp.MOSEK
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Try to solve with CVXPY 
        h_var = cp.Variable(n_points)
        z_vars = [cp.Variable() for _ in range(2 * n_points - 1)]
        objective = cp.Minimize(cp.max(z_vars))
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Add bilinear constraints z_k = sum(h[i] * (1 - h[i+k]))
        for k in range(-n_points + 1, n_points):
            idx_shift = k if k >= 0 else 0
            idx_end = min(n_points, n_points + k) 
            indices = np.arange(idx_shift, idx_end)
            if indices.size == 0:
                continue
            term = h_var[indices] * (1 - h_var[indices - k])
            # Approximate bilinear term with linear constraint
            # Lower bound for h[i]*(1-h[i+k])
            lower = h_var[indices] + h_var[indices - k] - 1
            # Upper bound for h[i]*(1-h[i+k])  
            upper = h_var[indices] - h_var[indices - k] + 1
            # z_k is an upper bound on sum(h[i] * (1 - h[i+k]))

            # Just skip to simpler local descent

        from scipy.optimize import minimize
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Try to optimize - better to use 'L-BFGS-B' over 'SLSQP'
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
