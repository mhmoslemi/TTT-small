import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with local search and adaptive step sizes
    iteration = 0
    max_iter = 10000
    step_size = 0.02
    c_reduction_factor = 0.999
    improvement_threshold = 1e-6
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try adaptive step-size local search
        noise_factor = np.random.uniform(0.005, 0.02) * step_size
        h_noisy = h + np.random.normal(0, noise_factor, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        # Minimize using L-BFGS-B which handles bounds well
        def objective(h):
            return compute_c5(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Slightly perturbed start
        result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 - improvement_threshold:
                h = h_candidate.copy()
                step_size *= 1.05  # Increase step size when successful
            else:
                step_size *= 0.95  # Decrease when no improvement
        else:
            # If L-BFGS fails, try global perturbation
            h = np.clip(h + np.random.normal(0, step_size, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Periodic global perturbation
        if iteration % 30 == 0:
            h = np.clip(h + np.random.normal(0, step_size * 2, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
