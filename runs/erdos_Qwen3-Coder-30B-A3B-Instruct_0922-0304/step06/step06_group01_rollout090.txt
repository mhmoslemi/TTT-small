import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope):
    """Construct h using comb-like teeth with smooth transitions."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_height = 1.0  # will be normalized after construction
    
    # Calculate actual tooth positions
    for i in range(n_teeth):
        tooth_start = i * (tooth_width + spacing)
        tooth_end = tooth_start + tooth_width
        
        # Convert to indices
        idx_start = int(tooth_start / dx)
        idx_end = int(tooth_end / dx)
        
        # Ensure indices are within bounds
        idx_start = max(0, idx_start)
        idx_end = min(n_points, idx_end)
        
        if idx_start >= n_points or idx_end <= 0:
            continue
            
        # Add the tooth
        for j in range(idx_start, idx_end):
            # Position in tooth coordinate system
            pos = j * dx
            if pos < tooth_start:
                # Left transition
                transition_pos = (tooth_start - pos) / slope
                h[j] += max(0, 1 - transition_pos)
            elif pos > tooth_end:
                # Right transition
                transition_pos = (pos - tooth_end) / slope
                h[j] += max(0, 1 - transition_pos)
            else:
                h[j] += 1.0
    
    # Normalize to maintain target integral
    h = normalize_h(h, n_points / 2)
    
    return h

def optimize_comb_parameters(h, dx, params_bounds):
    """Optimize comb parameters to minimize C5"""
    def objective(params):
        n_teeth, tooth_width, spacing, slope = params
        # Clip parameters to valid ranges
        n_teeth = max(2, min(20, int(n_teeth)))  # bounded
        tooth_width = max(0.01, min(0.5, tooth_width))
        spacing = max(0.0, min(1.0, spacing))
        slope = max(0.01, min(tooth_width, slope))
        
        h_new = construct_comb_h(len(h), n_teeth, tooth_width, spacing, slope)
        return compute_c5(h_new, dx)
    
    # Initial guess from current parameters
    n_teeth = 7
    tooth_width = 2 / len(h)
    spacing = (2 - n_teeth * tooth_width) / (n_teeth + 1)
    slope = tooth_width / 2
    
    x0 = [n_teeth, tooth_width, spacing, slope]
    bounds = [
        (2, 20),          # n_teeth
        (0.01, 0.5),      # tooth_width
        (0.0, 1.0),       # spacing
        (0.01, 0.5)       # slope
    ]
    
    # Minimize with SLSQP
    try:
        result = minimize(objective, x0, method='SLSQP', bounds=bounds, options={'maxiter': 100})
        if result.success:
            return result.x
    except:
        pass
    return x0
