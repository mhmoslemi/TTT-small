import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Alternating optimization: optimize h and g in alternating fashion
    iteration = 0
    g = np.zeros(2 * n_points - 1)  # surrogate overlap values
    
    while time.time() - start_time < budget_s - 1:
        # Alternate optimization
        
        # Update g based on current h
        h = best_individual.copy()
        # Compute true overlaps
        for k in range(2 * n_points - 1):
            overlap = 0.0
            for i in range(n_points):
                j = i + k - (n_points - 1)  # shift
                if 0 <= j < n_points:
                    overlap += h[i] * (1 - h[j])
            g[k] = max(g[k], overlap)
        
        # Optimize h with fixed g
        def objective(h):
            # Approximate max of overlaps
            max_g = np.max(g)
            return max_g

        # Constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum

        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Perturb and optimize
        h_perturbed = h + np.random.normal(0, 0.03, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Optimize using SLSQP
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200, 'ftol': 1e-6})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_individual = h_new.copy()
        except Exception:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
