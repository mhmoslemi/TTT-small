import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate diverse starting points for optimization."""
    guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern
    h_comb = np.zeros(n_points)
    for i in range(0, n_points, 4):  # Adjust spacing as needed
        if i < n_points:
            h_comb[i] = 1.0
    h_comb = normalize_h(h_comb, target_sum)
    guesses.append(h_comb.copy())
    
    # Random pattern
    h_random = np.random.random(n_points)
    h_random = normalize_h(h_random, target_sum)
    guesses.append(h_random.copy())
    
    # Alternating pattern
    h_alt = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_alt[i] = 1.0
    h_alt = normalize_h(h_alt, target_sum)
    guesses.append(h_alt.copy())
    
    # Low-discrepancy sequence (Halton-like)
    h_halton = np.zeros(n_points)
    for i in range(n_points):
        # Simple pseudo-Halton-like sequence
        h_halton[i] = (i * 0.618033988749895) % 1.0
    h_halton = normalize_h(h_halton, target_sum)
    guesses.append(h_halton.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    for guess_idx, h in enumerate(initial_guesses):
        h = h.copy()
        
        # Within each guess, perform iterative optimization
        iteration = 0
        max_iter = 10000  # Reduced for quick testing
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
            
            # Adaptive perturbation strategy
            perturb_strength = 0.02 * (0.95 ** (iteration // 100))
            
            # Gradient descent step (if available)
            def objective(h):
                return compute_c5(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                # Try SLSQP optimization
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h = result.x
                else:
                    raise Exception("Optimization failed")
            except:
                # Fallback to simple perturbations if optimization fails
                h = h + np.random.normal(0, perturb_strength, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
            
            # Large perturbations occasionally
            if iteration % 50 == 0:
                h = h + np.random.normal(0, perturb_strength * 2, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                
            iteration += 1
            
            # Adjusting time pressure based on progress
            if iteration % 500 == 0 and best_c5 <= 0.38075:
                break
    
    if best_h is None:
        # Fallback to default if no improvement
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
            # normalize to sum = 35
            best_h = normalize_h(best_h, target_sum)
        else:
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
