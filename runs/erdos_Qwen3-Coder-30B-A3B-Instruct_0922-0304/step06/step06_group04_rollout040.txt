import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Use a smooth approximation to avoid local minima
    # Implement crude gradient descent with backtracking line search
    def smooth_max(arr, epsilon=1e-5):
        # Smooth approximation of max
        return np.log(np.sum(np.exp(arr / epsilon))) * epsilon

    def surrogate_loss(h, dx, epsilon=1e-3):
        overlap = correlate(h, 1 - h, mode="full")
        # Smooth approximation for serverity
        return smooth_max(overlap, epsilon) * dx

    def gradient_surrogate(h, dx, epsilon=1e-3):
        # Gradients are computed numerically for robustness
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        # Define perturbation for finite difference
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            loss_plus = surrogate_loss(h_plus, dx, epsilon)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = normalize_h(h_minus, target_sum)
            loss_minus = surrogate_loss(h_minus, dx, epsilon)
            
            grad[i] = (loss_plus - loss_minus) / (2 * eps)
        
        return grad

    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        h = best_individual.copy()
        
        # Gradient based update with backtracking line search
        grad = gradient_surrogate(h, dx)
        # Simple backtracking line search
        alpha = 0.1
        beta = 0.9
        max_iter = 100
        
        for _ in range(max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            loss_old = surrogate_loss(h, dx)
            loss_new = surrogate_loss(h_new, dx)
            
            # Armijo condition
            if loss_new <= loss_old - 0.5 * alpha * np.sum(grad**2):
                h = h_new
                break
            alpha *= beta
        
        # Evaluate actual C5
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
