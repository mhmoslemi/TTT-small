import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.05):
    """Apply Gaussian mutation to an individual while maintaining bounds."""
    h = h + np.random.normal(0, mutation_rate, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def genetic_algorithm(initial_h, n_points, target_sum, dx, population_size=30, generations=50):
    """Perform genetic algorithm optimization."""
    # Initialize population around the initial solution
    population = []
    for _ in range(population_size):
        h = mutate_individual(initial_h, 0.02)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    for generation in range(generations):
        # Evaluate all individuals in the population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        top_half = sorted_indices[:population_size // 2]
        
        # Create new population through crossover and mutation
        new_population = [population[i].copy() for i in top_half]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = population[np.random.choice(top_half)]
            parent2 = population[np.random.choice(top_half)]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate_individual(child1, 0.05)
            child2 = mutate_individual(child2, 0.05)
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    return best_h, best_c5

        local_candidates = [best_h.copy()]
        
        # Generate additional candidates for refinement
        for _ in range(5):
            h_candidate = mutate_individual(best_h, 0.02)
            h_candidate = normalize_h(h_candidate, target_sum)
            local_candidates.append(h_candidate)
        
        # Refine each candidate using local optimization
        for candidate in local_candidates:
            def objective(h_refine):
                return compute_c5(h_refine, dx)
            
            def constraint_sum(h_refine):
                return np.sum(h_refine) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
