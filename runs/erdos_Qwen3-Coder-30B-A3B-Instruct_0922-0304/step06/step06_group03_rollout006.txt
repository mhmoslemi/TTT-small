import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the parent construction or a simple step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Simple step function initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle = np.clip(particle, 0, 1)
            # Add noise around initial
            noise = np.random.normal(0, 0.02, n_points)
            particle = np.clip(particle + noise, 0, 1)
            particle = normalize_h(particle, target_sum)
        else:
            particle = np.random.rand(n_points)
            particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        score = compute_c5(particle, dx)
        p_best.append(particle.copy())
        p_best_scores.append(score)
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    best_c5 = g_best_score
    best_h = g_best.copy()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if best_c5 <= 0.3807:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate particle
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particles[i].copy()
                p_best_scores[i] = score
                
                # Update global best
                if score < g_best_score:
                    g_best = particles[i].copy()
                    g_best_score = score
            
            # Periodic operations
            if iteration % 50 == 0:
                # Apply frequency domain filter to reduce noise
                if iteration > 0:
                    # Convert to frequency domain
                    fft_h = np.fft.fft(particles[i])
                    # Create high-pass filter
                    freqs = np.fft.fftfreq(n_points, dx)
                    # Low frequency components are attenuated
                    mask = np.abs(freqs) > 0.3 * np.pi
                    fft_h *= mask
                    # Back to spatial domain
                    filtered = np.real(np.fft.ifft(fft_h))
                    particles[i] = np.clip(filtered, 0, 1)
                    particles[i] = normalize_h(particles[i], target_sum)
            
            if iteration % 100 == 0 and iteration > 0:
                # Gradient refinement on top performing particles
                # Top 10 particles selected
                top_indices = np.argsort(p_best_scores)[:10]
                for idx in top_indices:
                    def obj_func(h_vec):
                        return compute_c5(h_vec, dx)
                    
                    def constraint_sum(h_vec):
                        return np.sum(h_vec) - target_sum
                    
                    bounds = [(0, 1) for _ in range(n_points)]
                    constraint = {'type': 'eq', 'fun': constraint_sum}
                    
                    # Run SLSQP
                    result = minimize(obj_func, p_best[idx], method='SLSQP', 
                                    bounds=bounds, constraints=constraint, 
                                    options={'maxiter': 100})
                    if result.success:
                        updated_h = result.x
                        updated_h = np.clip(updated_h, 0, 1)
                        updated_h = normalize_h(updated_h, target_sum)
                        new_score = compute_c5(updated_h, dx)
                        
                        # Update personal best if better
                        if new_score < p_best_scores[idx]:
                            p_best[idx] = updated_h
                            p_best_scores[idx] = new_score
                            
                            # Update global best if necessary
                            if new_score < g_best_score:
                                g_best = updated_h.copy()
                                g_best_score = new_score

        # Track best solution
        if g_best_score < best_c5:
            best_c5 = g_best_score
            best_h = g_best.copy()
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
