import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1, strength=0.1):
    """Apply mutation to an individual."""
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + np.random.normal(0, strength, len(h)) * mask
    return np.clip(h, 0, 1)

def local_optimize(h, dx, target_sum, max_iter=50):
    """Use local optimization to refine an individual."""
    bounds = [(0, 1) for _ in range(len(h))]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': max_iter})
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = normalize_h(h_opt, target_sum)
        return h_opt
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population with diversity
    pop_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = normalize_h(h_base, target_sum)
    else:
        # Base with ramps
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1
        h_base = normalize_h(h_base, target_sum)
    
    # Add base with slight noise
    for _ in range(pop_size):
        h = h_base + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Main GA loop
    generation = 0
    max_generations = 100
    mutation_rate = 0.2
    min_mutation_rate = 0.01
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create next generation
        new_population = []
        
        # Elitism - keep best individual
        new_population.append(population[current_best_idx].copy())
        
        # Generate rest through selection, crossover, mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate, 0.1)
            child2 = mutate(child2, mutation_rate, 0.1)
            
            # Local optimization
            child1 = local_optimize(child1, dx, target_sum)
            child2 = local_optimize(child2, dx, target_sum)
            
            # Normalize
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        mutation_rate = max(min_mutation_rate, mutation_rate * 0.99)
        generation += 1
    
    # Final refinement of best
    if best_h is not None:
        best_h = local_optimize(best_h, dx, target_sum)
        final_c5 = compute_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    if best_h is None:
        # Fallback to random with normalization
        h_fallback = np.random.rand(n_points)
        h_fallback = np.clip(h_fallback, 0, 1)
        h_fallback = normalize_h(h_fallback, target_sum)
        best_h = local_optimize(h_fallback, dx, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
