import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guess(n_points, method='initial'):
    """Generate different initial guess patterns."""
    target_sum = n_points / 2
    if method == 'initial' and 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        return h
    elif method == 'comb':
        h = np.zeros(n_points)
        stride = max(1, n_points // 10)
        for i in range(0, n_points, stride):
            h[i] = 1.0
        h = normalize_h(h, target_sum)
        return h
    elif method == 'random':
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        return h
    elif method == 'alternating':
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = normalize_h(h, target_sum)
        return h
    else:
        h = np.ones(n_points)
        h = normalize_h(h, target_sum)
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Try multiple initializations
    initial_methods = ['initial', 'comb', 'random', 'alternating']
    
    for method in initial_methods:
        # Generate initial guess
        h = generate_initial_guess(n_points, method)
        
        iteration = 0
        max_iter = 10000
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            # Compute current c5
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
                
            # Optimizer step with constraints
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # SLSQP with random perturbations
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-8})
                if result.success:
                    h = result.x
                else:
                    # Perturb if optimization fails
                    h = h + np.random.normal(0, 0.01, n_points)
            except:
                h = h + np.random.normal(0, 0.01, n_points)
            
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Periodic perturbations
            if iteration % 20 == 0:
                noise = np.random.normal(0, 0.02, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                
            iteration += 1
            
            # Adaptive step size decay
            if iteration % 100 == 0:
                # No need to do anything special here since we haven't changed parameters
    
    # Final check
    if best_h is None:
        # Fallback to known initial
        best_h = generate_initial_guess(n_points, 'initial')
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
