import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with line search
    max_iter = 10000
    step_size = 0.01
    decay = 0.99
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (approximate)
        c5_val = compute_c5(h, dx)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Small random perturbation to estimate gradient
        pert = np.random.normal(0, 0.001, n_points)
        pert = normalize_h(pert, target_sum)
        
        h_plus = h + pert
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = normalize_h(h_plus, target_sum)
        
        h_minus = h - pert
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = normalize_h(h_minus, target_sum)
        
        c5_plus = compute_c5(h_plus, dx)
        c5_minus = compute_c5(h_minus, dx)
        
        grad = (c5_plus - c5_minus) / (2 * 0.001) * pert
        
        # Update step
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Line search
        alpha = 1.0
        c5_new = compute_c5(h_new, dx)
        while alpha > 1e-6:
            h_try = h - alpha * grad
            h_try = np.clip(h_try, 0, 1)
            h_try = normalize_h(h_try, target_sum)
            c5_try = compute_c5(h_try, dx)
            if c5_try < c5_new:
                h_new = h_try
                c5_new = c5_try
                break
            alpha *= 0.5
        
        h = h_new
        step_size *= decay
        
        # Periodic check
        if iteration % 100 == 0:
            c5_current = compute_c5(h, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
