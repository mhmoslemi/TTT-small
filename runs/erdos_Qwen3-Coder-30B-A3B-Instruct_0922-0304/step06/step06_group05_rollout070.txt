import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def project_h(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet constraint sum(h) = n_points/2
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_exact_gradient(h, k_star):
        # Compute gradient for the specific lag k_star
        n = len(h)
        g = np.zeros(n)
        
        # For each position i, compute partial derivative of overlap with respect to h[i]
        for i in range(n):
            # Gradient contribution from overlap at lag k_star
            # This is the chain rule: d/dh[i] (h[i] * (1-h[(i+k_star)%n]))
            # which is 1 - h[(i+k_star)%n] if i + k_star < n, 
            # and -h[i-k_star] if i - k_star >= 0
            j = (i + k_star) % n
            g[i] = (1 - h[j]) * dx
            
        return g
    
    def line_search(h, g, max_iter=10):
        # Backtracking line search for optimal step size
        eta = 0.01
        c1 = 0.1  # Armijo condition parameter
        beta = 0.5  # Reduction factor
        
        current_c5 = compute_c5(h)
        for _ in range(max_iter):
            # Try step eta along gradient
            h_new = h - eta * g
            h_new = project_h(h_new)
            new_c5 = compute_c5(h_new)
            
            # Armijo condition: f(x + eta*d) <= f(x) + c1 * eta * g^T * d
            # In our case, we minimize C5, so we want improvement
            if new_c5 <= current_c5 - c1 * eta * np.dot(g, g):
                return h_new, eta
            else:
                eta *= beta
                
        return h, eta
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    max_iter = int(1e6)  # Prevent infinite loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # Store the best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= target_c5:
            break
            
        # Compute exact gradient for the max lag
        g = compute_exact_gradient(h, max_k)
        
        # Adaptive line search with backtracking
        h, step_size = line_search(h, g)
        
        # Project back onto feasible set
        h = project_h(h)
        
        # Perturb periodically to escape local minima
        if iteration % 500 == 0:
            sigma = 0.01 * math.exp(-iteration/1000)
            pert = np.random.normal(0, sigma, n_points)
            h = h + pert
            h = project_h(h)
        
        # Occasionally add more noise for exploration
        if iteration % 2000 == 0:
            h = h + np.random.normal(0, 0.01, n_points)
            h = project_h(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
