import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    max_iter = 100
    inertia = 0.7
    inertia_decay = 0.99
    cognitive_weight = 1.5
    social_weight = 1.5
    
    # Initialize particles
    particles = [h.copy() + np.random.normal(0, 0.01, n_points) for _ in range(n_particles)]
    for p in particles:
        p = np.clip(p, 0, 1)
        p = normalize_h(p, target_sum)
    
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    pbest = [p.copy() for p in particles]
    pbest_values = [compute_c5(p, dx) for p in particles]
    gbest = h.copy()
    gbest_value = compute_c5(h, dx)
    
    iteration = 0
    
    # PSO with local refinement
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            cognitive_weight * r1 * (pbest[i] - particles[i]) + 
                            social_weight * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
            
            # Update global best
            if current_c5 < gbest_value:
                gbest = particles[i].copy()
                gbest_value = current_c5
                
        # Store best
        if gbest_value < best_c5:
            best_c5 = gbest_value
            best_h = gbest.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Local refinement of gbest
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # L-BFGS-B refinement
        result = minimize(objective, gbest, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            refined = result.x
            refined = np.clip(refined, 0, 1)
            refined = normalize_h(refined, target_sum)
            refined_c5 = compute_c5(refined, dx)
            if refined_c5 < gbest_value:
                gbest = refined.copy()
                gbest_value = refined_c5
            
        # Update inertia
        inertia *= inertia_decay
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
