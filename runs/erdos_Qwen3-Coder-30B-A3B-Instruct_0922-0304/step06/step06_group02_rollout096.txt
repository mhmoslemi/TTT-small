import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Identify the worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        worst_value = overlap[worst_lag_idx] * dx
        
        # Create a new optimization problem to reduce the worst overlap
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        def constraint_worst_lag(h_vec):
            # Let's only target a specific shift for now
            # Compute convolution of h with (1-h) and target worst lag
            conv = correlate(h_vec, 1 - h_vec, mode="full")
            return conv[worst_lag_idx] * dx - worst_value
            
        # While using explicit constraint is difficult, we approach via SLSQP
        # by solving a local optimization that might improve worst case
        
        # Try random perturbation, then local optimization
        h_new = h.copy()
        
        # Random small perturbations
        noise = np.random.normal(0, 0.01, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Local optimization around current solution
        def objective_local(h_vec):
            return compute_c5(h_vec, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective_local, h_new, method='SLSQP', 
                             bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5:
                    h = h_candidate.copy()
        except:
            pass
            
        # Periodic large perturbation
        if iteration % 30 == 0:
            h = np.clip(h + np.random.normal(0, 0.05, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
