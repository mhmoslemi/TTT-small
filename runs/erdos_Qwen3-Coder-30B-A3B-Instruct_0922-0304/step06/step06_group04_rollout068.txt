import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient descent with line search
    max_iter = 1000
    alpha = 0.1
    beta = 0.9
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h = best_individual.copy()
        
        # Compute gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        current_loss = evaluate_individual(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            loss_plus = evaluate_individual(h_plus, dx)
            grad[i] = (loss_plus - current_loss) / eps
        
        # Update with gradient descent
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Line search
        while True:
            h_test = h - alpha * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = normalize_h(h_test, target_sum)
            test_loss = evaluate_individual(h_test, dx)
            
            if test_loss < current_loss or alpha < 1e-8:
                h_new = h_test
                break
            alpha *= beta
            
        # Check improvement
        new_c5 = evaluate_individual(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_individual = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adaptive step size
        if iteration % 10 == 0:
            alpha *= 0.99
        
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
