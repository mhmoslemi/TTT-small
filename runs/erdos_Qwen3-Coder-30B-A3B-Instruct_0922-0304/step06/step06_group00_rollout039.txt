import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple projection: rescale to target sum
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    else:
        # If sum is zero, make all elements equal to target_sum/n_points
        h = np.full_like(h, target_sum / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_iter = 1000
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    gbest = None
    gbest_c5 = float('inf')
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Random initialization
            particle = np.random.rand(n_points) * 0.5 + 0.25  # 0.25 to 0.75
            particle = project_to_constraint(particle, target_sum)
            
        particles.append(particle)
        velocities.append(np.random.rand(n_points) * 0.05 - 0.025)  # -0.025 to 0.025
        pbest.append(particle.copy())
        pbest_c5.append(compute_c5(particle, dx))
        
        if pbest_c5[-1] < gbest_c5:
            gbest = particle.copy()
            gbest_c5 = pbest_c5[-1]
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Linearly decrease inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            current_c5 = pbest_c5[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (w * velocity + 
                       c1 * r1 * (pbest[i] - particle) + 
                       c2 * r2 * (gbest - particle))
            
            # Clamp velocity
            velocity = np.clip(velocity, -0.1, 0.1)
            
            # Update position
            particle = particle + velocity
            
            # Project to valid domain [0,1] and enforce sum constraint
            particle = np.clip(particle, 0, 1)
            particle = project_to_constraint(particle, target_sum)
            
            # Evaluate new particle
            new_c5 = compute_c5(particle, dx)
            
            # Update personal best
            if new_c5 < current_c5:
                pbest[i] = particle.copy()
                pbest_c5[i] = new_c5
                
                # Update global best
                if new_c5 < gbest_c5:
                    gbest = particle.copy()
                    gbest_c5 = new_c5
        
        # Early termination if target achieved
        if gbest_c5 <= 0.3807:
            break
    
    # Final check of the best solution
    final_c5 = compute_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
