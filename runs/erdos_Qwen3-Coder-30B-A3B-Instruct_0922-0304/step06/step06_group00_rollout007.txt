import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a better initial function
        h = np.zeros(n_points)
        # Use a piecewise constant with some variation
        mid = n_points // 2
        h[:mid] = 1
        h[mid:] = 0
        # Add slight variation to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier method parameters
    mu = 1.0
    decrease_mu_every = 50
    mu_decay = 0.5
    
    # Optimizer parameters
    max_iter = 10000
    gradient_eps = 1e-5
    learning_rate = 0.01
    max_line_search = 50
    
    def barrier(h):
        """Log barrier function"""
        penalty = 0
        # h > 0 and h < 1
        penalty += -np.sum(np.log(np.maximum(h, 1e-8)))
        penalty += -np.sum(np.log(np.maximum(1 - h, 1e-8)))
        # sum constraint
        penalty += -np.log(np.maximum(target_sum - np.sum(h), 1e-8))
        return penalty
    
    def gradient_barrier(h):
        """Gradient of barrier function"""
        grad = np.zeros_like(h)
        grad += -1 / (h + 1e-8)
        grad += 1 / (1 - h + 1e-8)
        # constraint gradient
        grad += 1 / (target_sum - np.sum(h) + 1e-8)
        return grad
    
    def eval_obj(h):
        return compute_c5(h, dx) + mu * barrier(h)
    
    def compute_gradient(h):
        # Numerical gradient of C5 using central differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += gradient_eps
            h_minus[i] -= gradient_eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            h_minus = normalize_h(h_minus, target_sum)
            grad[i] = (eval_obj(h_plus) - eval_obj(h_minus)) / (2 * gradient_eps)
        return grad
    
    iterations = 0
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute gradient
        grad = compute_gradient(h)
        barrier_grad = gradient_barrier(h)
        full_grad = grad + mu * barrier_grad
        
        # Line search
        alpha = 1.0
        step_size = alpha
        
        for _ in range(max_line_search):
            h_new = h - step_size * full_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            
            if eval_obj(h_new) < eval_obj(h) + 1e-8:
                h = h_new
                break
            else:
                step_size *= 0.5
        
        # Update mu
        if iterations % decrease_mu_every == 0 and iterations > 0:
            mu = max(mu * mu_decay, 1e-6)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        iterations += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
