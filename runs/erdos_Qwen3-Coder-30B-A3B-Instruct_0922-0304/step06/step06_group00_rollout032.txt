import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

        h = np.zeros(n_points)
        # Create a piecewise linear shape
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        # Add some smoothed noise
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iterations = 0

        # Gradient approximation using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = normalize_h(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update using gradient descent with adaptive step size
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Adaptive step size reduction
            step_size *= 0.5
            h = h_new.copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Mutation with decreasing strength
        mutation_strength = max(0.001, 0.05 * np.exp(-iterations / 300.0))
        if np.random.rand() < 0.3:  # 30% chance to mutate
            h += np.random.normal(0, mutation_strength, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iterations += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
