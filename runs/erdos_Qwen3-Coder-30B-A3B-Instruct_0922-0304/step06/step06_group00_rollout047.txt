import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_solution(h, dx):
    """Compute C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple but structured initial solution
        h = np.zeros(n_points)
        # Initialize step function with gradual ramp
        mid = n_points // 2
        h[:mid] = 1
        h[mid:] = 0.5
        # Add some noise for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = evaluate_solution(h, dx)
    best_h = h.copy()
    
    # Try to refine using more direct optimization
    # Create optimization variables
    h_var = Variable(n_points)
    C5_var = Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        sum(h_var) == target_sum,
        C5_var >= 0
    ]
    
    # All shift constraints - convert to SDP-like form
    # Build shift matrices that capture the constraint structure
    full_length = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    
    # Compute full set of constraints
    # For each lag k, we want sum(h[i] * (1 - h[(i-k) % n_points])) <= C5
    # This is done by careful construction of matrices and terms
    
    # Instead of full explicit formulation, try direct optimization approach
    
    # Approach: local search with gradient descent 
    # We'll manually do a simple gradient descent style search
    
    def compute_shifted_overlap(h_shift, k):
        # Compute overlap for lag k
        # This is: sum(h[i] * (1 - h[(i-k) % n_points]))
        n = len(h_shift)
        shift = np.roll(h_shift, k)
        return np.dot(h_shift, 1 - shift)
    
    def evaluate_objective(h):
        max_overlap = -np.inf
        for k in range(-n_points + 1, n_points):
            overlap = compute_shifted_overlap(h, k)
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    max_iter = 1000
    lr = 0.01
    patience = 20
    
    best_h = h.copy()
    best_c5 = evaluate_objective(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_objective(best_h)
        if current_c5 <= 0.3807:
            break
        
        # Approximate gradient with finite differences
        grad_h = np.zeros_like(best_h)
        eps = 1e-5
        
        # Gradient estimation
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))  # renormalize
            
            h_minus = best_h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))  # renormalize
            
            # Note: We need to compute change in maximum overlap
            # For efficiency, we only compute the gradient direction
            # This is a simplified version - gradient descent step
            grad_h[j] = (evaluate_objective(h_plus) - evaluate_objective(h_minus)) / (2 * eps)
        
        # Apply gradient descent step
        new_h = best_h - lr * grad_h
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (target_sum / np.sum(new_h))  # renormalize
        
        # Evaluate new solution
        new_c5 = evaluate_objective(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final refinement
    final_eval = evaluate_solution(best_h, dx)
    
    return (best_h.tolist(), final_eval, n_points)
