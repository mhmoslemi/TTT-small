import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform uniform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    mask = np.random.rand(len(parent1)) < 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, mutation_rate, strength=0.1):
    """Mutate an individual."""
    mask = np.random.rand(len(individual)) < mutation_rate
    noise = np.random.normal(0, strength, len(individual))
    individual = individual + mask * noise
    return np.clip(individual, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Create initial population
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.01, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Genetic algorithm loop
    max_generations = 60
    mutation_rate = 0.2
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Track best individual
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Select parents
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate(child2, mutation_rate)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        population = new_population
        mutation_rate *= 0.98  # Decrease mutation rate over time
    
    # Final refinement with gradient descent for best individual
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    from scipy.optimize import minimize
    
    for _ in range(10):  # Run a few rounds of local refinement
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            best_h = result.x
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
