import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
        h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Core optimization loop
    while time.time() - start_time < budget_s - 1:
        # Try to improve using bilinear programming approach
        h = h_base.copy()
        
        # Optimization using CVXPY
        try:
            # Set up bilinear optimization
            h_var = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
            
            # Objective: minimize max of overlaps
            # Create auxiliary variables for each shift
            max_val = cp.Variable()
            # Would normally use nonlinear constraints but we'll estimate this
            
            # Quick local search by small perturbations
            h_perturbed = h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
            
            c5_candidate = compute_c5(h_perturbed, dx)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_perturbed.copy()
                
            # Try another random perturbation
            h_perturbed2 = h + np.random.normal(0, 0.005, n_points) 
            h_perturbed2 = np.clip(h_perturbed2, 0, 1)
            h_perturbed2 = h_perturbed2 * (target_sum / np.sum(h_perturbed2))
            
            c5_candidate2 = compute_c5(h_perturbed2, dx)
            if c5_candidate2 < best_c5:
                best_c5 = c5_candidate2
                best_h = h_perturbed2.copy()
                
        except:
            pass
        
        # Alternate heuristic: try gradient-free hill climbing
        # Start from the best value found so far and evolute
        h_current = best_h.copy()
        # Sample some iterations of gradient-free improvement
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb
            h_test = h_current + np.random.normal(0, 0.01, n_points)
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (target_sum / np.sum(h_test))
            
            c5_test = compute_c5(h_test, dx)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h_test.copy()
                h_current = h_test.copy()
        
        # If we've made progress, continue
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
