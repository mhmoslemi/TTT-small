import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize particle swarm
    n_particles = 50
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.02, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
    
    # Initialize personal bests
    p_best = [p.copy() for p in particles]
    p_best_vals = [compute_c5(p, dx) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate all particles
        for i in range(n_particles):
            current_c5 = compute_c5(particles[i], dx)
            if current_c5 < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_c5
            
            if current_c5 < g_best_val:
                g_best = particles[i].copy()
                g_best_val = current_c5
                
        # Terminate if we've found a good enough solution
        if g_best_val <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
        # Every 50 iterations: apply frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            # Sort by performance
            sorted_indices = np.argsort(p_best_vals)
            worst_25 = sorted_indices[-(n_particles//4):]
            
            # Apply frequency domain filtering
            for idx in worst_25:
                particle = p_best[idx].copy()
                # Apply FFT and high-pass filter
                fft_vals = np.fft.fft(particle)
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter with cutoff around 0.3
                filter_mask = np.abs(freqs) > 0.3
                filtered_fft = fft_vals * filter_mask
                filtered = np.real(np.fft.ifft(filtered_fft))
                # Renormalize
                filtered = np.clip(filtered, 0, 1)
                filtered = normalize_h(filtered, target_sum)
                p_best[idx] = filtered
                p_best_vals[idx] = compute_c5(filtered, dx)
                
        # Every 100 iterations: gradient-based refinement
        if iteration % 100 == 0 and iteration > 0:
            # Take top 10 particles
            sorted_indices = np.argsort(p_best_vals)
            top_indices = sorted_indices[:10]
            for i in top_indices:
                # More aggressive optimization for top particles
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, p_best[i], method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 100})
                if result.success:
                    refined = result.x
                    refined = np.clip(refined, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    new_val = compute_c5(refined, dx)
                    if new_val < p_best_vals[i]:
                        p_best[i] = refined
                        p_best_vals[i] = new_val
                        if new_val < g_best_val:
                            g_best = refined.copy()
                            g_best_val = new_val
            
        iteration += 1
    
    # Final evaluation of global best
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
