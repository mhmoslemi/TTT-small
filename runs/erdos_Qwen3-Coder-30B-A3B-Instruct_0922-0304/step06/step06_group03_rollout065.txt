import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a comb-like initial pattern that balances spread and concentration
        h = np.zeros(n_points)
        h[:n_points//4] = 1
        h[n_points//4:3*n_points//4] = 0.5
        h[3*n_points//4:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    noise_amp = 0.02
    max_iter = 20000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Cross-correlation analysis to identify high-overlap regions
        corr = correlate(h, 1 - h, mode="full")
        peaks = np.argsort(corr)[::-1][:n_points//10]  # Top 10% peaks
        peak_positions = peaks - (len(corr) // 2 - 1)
        
        # Create a perturbation mask: focus on regions near high overlap
        mask = np.zeros(n_points)
        for pos in peak_positions:
            if 0 <= pos < n_points:
                mask[pos] = 1
                
        # Apply perturbation with gradient adjustment
        perturbation = np.random.normal(0, noise_amp, n_points)
        perturbation *= mask
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally do a more refined local optimization
        if iteration % 20 == 0:
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else:
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
        # Reduce noise amplitude every 500 iterations
        if iteration % 500 == 0 and noise_amp > 0.001:
            noise_amp *= 0.95
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
