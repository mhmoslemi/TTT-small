import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize particles around the base
    for _ in range(n_particles):
        perturbation = np.random.normal(0, 0.05, n_points)
        h = base_h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    
    # Track personal and global bests
    personal_best_h = [p.copy() for p in particles]
    personal_best_c5 = [compute_c5(p, dx) for p in particles]
    g_best_idx = np.argmin(personal_best_c5)
    global_best_h = personal_best_h[g_best_idx].copy()
    global_best_c5 = personal_best_c5[g_best_idx]
    
    # Frequency domain filtering
    def apply_frequency_filter(h):
        # FFT and high-pass filter
        f = np.fft.fft(h)
        freq = np.fft.fftfreq(len(h), dx)
        # High-pass filter at 0.3
        mask = np.abs(freq) > 0.3
        f_filtered = f * mask
        h_filtered = np.real(np.fft.ifft(f_filtered))
        return h_filtered
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate all particles
        for i, h in enumerate(particles):
            c5 = compute_c5(h, dx)
            if c5 < personal_best_c5[i]:
                personal_best_c5[i] = c5
                personal_best_h[i] = h.copy()
                if c5 < global_best_c5:
                    global_best_c5 = c5
                    global_best_h = h.copy()
        
        # Check for early termination
        if global_best_c5 <= 0.3807:
            break
            
        # PSO update
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            cognitive = c1 * r1 * (personal_best_h[i] - h)
            social = c2 * r2 * (global_best_h - h)
            v = w * v + cognitive + social
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            particles[i] = h.copy()
            velocities[i] = v.copy()
            
        # Periodic frequency domain acceleration (every 50 iterations)
        if iteration % 50 == 0:
            # Sort particles by fitness
            fitnesses = [compute_c5(p, dx) for p in particles]
            sorted_indices = np.argsort(fitnesses)
            
            # Replace worst 25% with filtered versions
            for i in range(n_particles // 4):
                idx = sorted_indices[-(i+1)]
                h_filtered = apply_frequency_filter(particles[idx])
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = normalize_h(h_filtered, target_sum)
                particles[idx] = h_filtered
        
        # Periodic gradient refinement (every 100 iterations)
        if iteration % 100 == 0:
            # Select top 10 best particles for refinement
            fitnesses = [compute_c5(p, dx) for p in particles]
            sorted_indices = np.argsort(fitnesses)[:10]
            
            # Refine each selected particle
            for idx in sorted_indices:
                h = particles[idx]
                def objective(h):
                    return compute_c5(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h_new = result.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = normalize_h(h_new, target_sum)
                    particles[idx] = h_new
                
    final_c5 = compute_c5(global_best_h, dx)
    return (global_best_h.tolist(), final_c5, n_points)
