import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Binary search to find minimal C5
    low = 0.0
    high = 1.0
    best_c5 = float('inf')
    best_h = h.copy()

    while time.time() - start_time < budget_s - 1 and (high - low) > 1e-5:
        c_test = (low + high) / 2.0
        
        # Solve feasibility problem
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints for all non-zero lags
        for k in range(1, n_points):
            # Compute overlap manually with convolution
            # This is a simplified check; the actual optimization
            # would involve quadratic constraints but we will
            # just heuristically verify the maximum overlap.
            pass  # We check feasibility via direct computation later

        # Use a relaxation approach with heuristics for large systems
        # Since CVXPY struggles with quadratic constraints, we approximate
        # with iterative approach and local refinement based on gradient
        
        # Compute the maximum overlap for current h and c_test
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap) * dx
        
        if max_overlap <= c_test:
            high = c_test
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
        else:
            low = c_test
            
        if best_c5 <= target_c5:
            break

        # Local optimization heuristic every few iterations
        if (high - low) < 0.01 or np.random.rand() < 0.1:
            # Gradient-based adjustment
            g = np.zeros(n_points)
            eps_val = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            step_size = 0.05
            new_h = h - step_size * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            h = new_h
            
            # Occasionally perturb to escape local minima
            if np.random.rand() < 0.1:
                pert = np.random.normal(0, 0.005, n_points)
                h = h + pert
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)

    # Final refinement step
    refined_h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = refined_h.copy()
            h_minus = refined_h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        step_size = 0.02
        new_h = refined_h - step_size * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        refined_h = new_h

    final_c5 = compute_c5(refined_h)
    return (refined_h.tolist(), final_c5, n_points)
