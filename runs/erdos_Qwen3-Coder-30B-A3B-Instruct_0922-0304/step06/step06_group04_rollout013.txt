import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
    
    # Normalize
    current_sum = np.sum(base_h)
    if current_sum > 0:
        base_h = base_h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    def solve_bilinear(h0, max_time=30):
        h = cp.Variable(n_points)
        z = cp.Variable(2 * n_points - 1)
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        
        # Bilinear constraints
        for k in range(2 * n_points - 1):
            start_i = max(0, k - n_points + 1)
            end_i = min(n_points, n_points + k - n_points + 1)
            if start_i < end_i:
                # sum_{i: valid} h[i] * (1 - h[i + k])
                expr = 0
                for i in range(start_i, end_i):
                    expr += h[i] * (1 - h[i + k - start_i])
                constraints.append(z[k] == expr)
        
        # Objective: minimize max of z * dx
        objective = cp.Minimize(cp.max(z) * dx)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=max_time)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value
        except:
            pass
        return h0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Try to improve via bilinear programming
        h_new = solve_bilinear(best_h.copy(), max_time=30)
        
        # Ensure feasible and normalize
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        # Apply small random perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h_perturbed = best_h + noise
        h_perturbed = np.clip(h_perturbed, 0, 1)
        current_sum = np.sum(h_perturbed)
        if current_sum > 0:
            h_perturbed = h_perturbed * (target_sum / current_sum)
        
        # Re-evaluate and update if better
        c5_perturbed = compute_c5(h_perturbed, dx)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final verification
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
