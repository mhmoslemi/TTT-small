import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct h as a comb with specified parameters."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0

    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        start_pos = i * (tooth_width + spacing)
        end_pos = start_pos + tooth_width
        
        # Create indices for this tooth
        start_idx = int(np.floor(start_pos / dx))
        end_idx = int(np.ceil(end_pos / dx))
        
        # Ensure bounds
        start_idx = max(0, start_idx)
        end_idx = min(n_points, end_idx)
        
        # Set tooth heights
        for j in range(start_idx, end_idx):
            x = j * dx
            # Distance from left edge of tooth
            dist_from_start = x - start_pos
            # Distance from right edge of tooth
            dist_from_end = end_pos - x
            
            # Determine how much of the tooth contributes at this point
            if dist_from_start >= 0 and dist_from_end >= 0:
                # Inside the tooth
                h[j] = 1.0
                
            elif dist_from_start >= 0 and dist_from_end < 0:
                # Near right edge
                rel_pos = dist_from_end / (-slope)
                if rel_pos >= 0:
                    h[j] = max(0, 1 - rel_pos)
                    
            elif dist_from_start < 0 and dist_from_end >= 0:
                # Near left edge
                rel_pos = dist_from_start / slope
                if rel_pos <= 0:
                    h[j] = max(0, 1 + rel_pos)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_c5 = float('inf')
    best_h = None
    
    # Initialize using the predefined construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)

    # Parameter sweeps
    sweep_params = [
        {'n_teeth': 5, 'tooth_width': 2/5, 'spacing': 0.05, 'slope': 0.1},
        {'n_teeth': 6, 'tooth_width': 2/6, 'spacing': 0.03, 'slope': 0.07},
        {'n_teeth': 7, 'tooth_width': 2/7, 'spacing': 0.02, 'slope': 0.05},
        {'n_teeth': 8, 'tooth_width': 2/8, 'spacing': 0.01, 'slope': 0.04},
    ]
    
    # Evaluate different comb structures
    for params in sweep_params:
        if time.time() - start_time > budget_s - 10:
            break
            
        h_temp = construct_comb_h(n_points, **params)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = normalize_h(h_temp, target_sum)
        
        c5 = compute_c5(h_temp, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_temp.copy()
    
    # Local optimization with SLSQP on the best initial guess
    if best_h is not None:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            c5_final = compute_c5(h_new, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_new.copy()
    
    # If we didn't improve, return the comb initialization
    if best_h is None:
        # Default comb
        h_default = construct_comb_h(n_points, n_teeth=7)
        h_default = np.clip(h_default, 0, 1)
        h_default = normalize_h(h_default, target_sum)
        best_h = h_default.copy()
        best_c5 = compute_c5(best_h, dx)
    
    # Final SLSQP refinement
    if time.time() - start_time < budget_s - 10:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 300})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            c5_final = compute_c5(h_new, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_new.copy()

    return (best_h.tolist(), best_c5, n_points)
