import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate a single individual."""
    h += np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(h1, h2, crossover_rate=0.8):
    """Crossover two parents to produce offspring."""
    if np.random.rand() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1) - 1)
    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill population with diverse samples
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Params
    max_generations = 600
    mutation_strength_start = 0.1
    mutation_strength_decay = 0.99
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            
        # Track best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[current_best_idx])
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutate
            mutation_strength = mutation_strength_start * (mutation_strength_decay ** generation)
            child1 = mutate_individual(child1, mutation_strength)
            child2 = mutate_individual(child2, mutation_strength)
            
            # Normalize
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:population_size]
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
