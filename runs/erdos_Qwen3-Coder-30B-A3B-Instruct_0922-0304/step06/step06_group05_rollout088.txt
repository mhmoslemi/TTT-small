import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a reasonable starting guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search approach for C5
    c_low = 0.0
    c_high = 1.0
    tolerance = 1e-4
    max_iterations = 50
    iteration = 0
    
    while time.time() - start_time < budget_s - 2 and iteration < max_iterations:
        iteration += 1
        c_mid = (c_low + c_high) / 2.0
        
        # Convex feasibility check using CVXPY
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        for k in range(1, n_points):
            # For each lag k, compute the constraint
            # sum h[i] * (1 - h[(i+k) % n]) <= c_mid / dx
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= c_mid / dx)
        
        # Minimize a dummy objective for feasibility checking
        objective = cp.Minimize(cp.sum(h_vars))  # Dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Check actual overlap with the computed solution
                actual_c5 = compute_c5(h_sol)
                
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_sol.copy()
                    
                if actual_c5 <= 0.3807:
                    break
                    
                # Update search bounds
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid
    
    # Final optimization of the best solution found
    if time.time() - start_time < budget_s - 5:
        # Use gradient-based refinement
        step_size = 0.005
        max_inner_iter = 20000
        inner_iter = 0
        
        while time.time() - start_time < budget_s - 2 and inner_iter < max_inner_iter:
            inner_iter += 1
            
            overlap = correlate(best_h, 1 - best_h, mode="full")
            max_k = np.argmax(overlap)
            max_overlap = overlap[max_k] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                
            if max_overlap <= 0.3807:
                break
                
            # Compute gradient
            g = np.zeros(n_points)
            eps_val = 1e-4
            for i in range(n_points):
                h_plus = best_h.copy()
                h_minus = best_h.copy()
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Update with gradient descent
            new_h = best_h - step_size * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            best_h = new_h

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
