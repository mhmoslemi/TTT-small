import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h = h + np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Create offspring from two parents via crossover."""
    if np.random.rand() > crossover_rate:
        return p1.copy(), p2.copy()
    
    # Single-point crossover
    crossover_point = np.random.randint(1, len(p1) - 1)
    offspring1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    offspring2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    return offspring1, offspring2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Generate initial diverse population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
        
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    max_generations = 60
    mutation_strength_init = 0.1
    mutation_strength_decay = 0.97
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (C5)
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Select parents and create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[min_fitness_idx])
        
        # Generate offspring
        for _ in range(population_size - 1):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            offspring1, offspring2 = crossover_parents(parent1, parent2)
            
            # Mutate offspring
            mutation_strength = mutation_strength_init * (mutation_strength_decay ** generation)
            offspring1 = mutate_individual(offspring1, mutation_strength)
            offspring2 = mutate_individual(offspring2, mutation_strength)
            
            # Normalize to meet sum constraint
            offspring1 = normalize_h(offspring1, target_sum)
            offspring2 = normalize_h(offspring2, target_sum)
            
            new_population.extend([offspring1, offspring2])
            
        # Trim population to exact size
        population = new_population[:population_size]
        
        # Occasionally add new diversity
        if generation % 10 == 0:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                # Replace worst individual
                worst_idx = np.argmax(fitnesses)
                population[worst_idx] = h.copy()
    
    if best_h is None:
        # Fallback to final population
        fitnesses = [compute_c5(h, dx) for h in population]
        min_fitness_idx = np.argmin(fitnesses)
        best_c5 = fitnesses[min_fitness_idx]
        best_h = population[min_fitness_idx]
    
    return (best_h.tolist(), best_c5, n_points)
