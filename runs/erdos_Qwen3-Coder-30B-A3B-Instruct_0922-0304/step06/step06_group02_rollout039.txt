import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to desired sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try CVXPy approach with tighter formulation
    solver_timeout = 100  # seconds
    
    # Set up constraints properly
    max_lag = n_points - 1
    num_constraints = 2 * max_lag + 1
    
    # Solve using CVXPy if possible
    try:
        # Define variables
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            C_var >= 0
        ]
        
        # Add overlap constraints for all lags
        for k in range(-max_lag, max_lag + 1):
            # For lag k, compute sum_{i} h[i] * (1 - h[i+k]) 
            # i ranges from 0 to n_points - 1 - |k| (in case of positive k)
            # or 0 to n_points - 1 + k (in case of negative k)
            if k >= 0:
                # Sum over i where both h[i] and h[i+k] are valid
                valid_indices = range(n_points - k)
                expr = 0
                for i in valid_indices:
                    expr += h_var[i] * (1 - h_var[i + k])
            else:
                valid_indices = range(-k, n_points)
                expr = 0
                for i in valid_indices:
                    expr += h_var[i] * (1 - h_var[i + k])
            constraints.append(expr * dx <= C_var)
        
        # Objective
        objective = cp.Minimize(C_var)
        prob = cp.Problem(objective, constraints)
        
        # Attempt to solve
        prob.solve(solver=cp.GUROBI, verbose=False, time_limit=solver_timeout)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h_var.value
            if h_result is not None:
                h_result = np.clip(h_result, 0, 1)
                # Ensure sum constraint is met
                h_result = h_result * (target_sum / np.sum(h_result))
                test_c5 = compute_c5(h_result, dx)
                if test_c5 < best_c5:
                    best_c5 = test_c5
                    best_h = h_result.copy()
                print(f"CVXPy solution: C5 = {best_c5:.6f}")
    
    except Exception as e:
        pass  # Fall back to other methods if needed
    
    # Fine-tune with local search
    max_iter = 20000
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current value
        current_c5 = compute_c5(best_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Store best_h only if better
            best_h = best_h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Local optimization step via scipy
        # Prepare constraints for optimization
        def constraint_sum(x):
            return np.sum(x) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(x):
            return compute_c5(x, dx)
            
        # Random perturbation around best_h
        noise = np.random.normal(0, 0.005, n_points)
        h_noisy = best_h + noise
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = h_noisy * (target_sum / np.sum(h_noisy))
        
        # Optimize
        from scipy.optimize import minimize
        try:
            result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5:
                    best_h = h_candidate.copy()
        except:
            pass
            
        iteration += 1
        
        # Perturb occasionally
        if iteration % 50 == 0:
            # Add small noise and renormalize
            noise = np.random.normal(0, 0.01, n_points)
            best_h = best_h + noise
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (target_sum / np.sum(best_h))
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
