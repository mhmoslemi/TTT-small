import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create initial guess with structured pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try SDP-based approach for better solution
    if budget_s > 100:
        try:
            # Try to solve this as a convex optimization problem
            h_var = cp.Variable(n_points)
            C5_var = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_var) == target_sum,
                h_var >= 0,
                h_var <= 1
            ]
            
            # Generate all shift constraints
            for k in range(-n_points + 1, n_points):
                # Create the corresponding sum for shift k
                shift_sum = 0
                for i in range(n_points):
                    j = (i - k) % n_points
                    shift_sum += h_var[i] * (1 - h_var[j])
                constraints.append(shift_sum * dx <= C5_var)
            
            # Objective
            objective = cp.Minimize(C5_var)
            prob = cp.Problem(objective, constraints)
            
            # Solve with MOSEK or CVXOPT
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sdp = h_var.value
                c5_sdp = C5_var.value
                if c5_sdp < best_c5:
                    best_c5 = c5_sdp
                    best_h = np.array(h_sdp)
        except:
            pass
    
    # Second stage: gradient-based refinement
    if time.time() - start_time < budget_s - 10:
        # Simple gradient-based refinement with simulated annealing
        iterations = 0
        max_iter = 1000
        temp = 1.0
        
        while time.time() - start_time < budget_s - 1 and iterations < max_iter:
            # Create small perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h + perturbation * temp
            
            # Clip and normalize
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if current_sum > 0:
                h_new = h_new * (target_sum / current_sum)
            
            # Compute new C5
            new_c5 = compute_c5(h_new, dx)
            
            # Accept or reject
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                h = h_new.copy()
            elif np.random.rand() < np.exp(-(new_c5 - best_c5) / temp):
                h = h_new.copy()
            
            # Cooling schedule
            temp *= 0.995
            iterations += 1
    
    # Final safety check on solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
