import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, rate=0.05, target_sum=35):
    """Apply mutation to h while maintaining constraints."""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < rate
    h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover_h(h1, h2, rate=0.7):
    """Perform crossover between two h vectors."""
    h_child = h1.copy()
    mask = np.random.random(len(h1)) < rate
    h_child[mask] = h2[mask]
    return h_child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 10
    num_generations = 10000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_perturbed = mutate_h(h, mutation_rate, target_sum)
        population.append(h_perturbed)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main GA loop
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tournament_size = 3
                tournament_indices = np.random.choice(len(population), tournament_size)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(population[winner_idx])
            
            # Crossover
            child = crossover_h(parents[0], parents[1], crossover_rate)
            
            # Mutation
            child = mutate_h(child, mutation_rate, target_sum)
            
            new_population.append(child)
            
        population = new_population
        
        # Local refinement with SLSQP for top 10%
        top_k = min(elite_size, len(population))
        for i in range(top_k):
            idx = np.argmin([compute_c5(p, dx) for p in population])
            h_local = population[idx]
            
            def constraint_sum(x):
                return np.sum(x) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(x):
                return compute_c5(x, dx)
                
            try:
                result = minimize(objective, h_local, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined = np.clip(result.x, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    population[idx] = refined
            except:
                pass
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
