import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    for i in range(num_particles):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Initialize best
    best_h = base_h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    w = 0.8
    c1 = c2 = 2.0
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update particles
        for i in range(num_particles):
            h = particles[i]
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            if best_c5 <= 0.3807:
                break
        
        if best_c5 <= 0.3807:
            break
            
        # Update positions
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            
            # PSO update
            r1, r2 = np.random.rand(2)
            p_best = particles[i]  # Placeholder, actual value will be set...
            g_best = best_h
            
            v = w * v + c1 * r1 * (p_best - h) + c2 * r2 * (g_best - h)
            h = h + v
            
            # Apply constraints
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            particles[i] = h
            velocities[i] = v
            
        # Frequency domain acceleration every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            # Find worst particles
            particle_scores = [compute_c5(p, dx) for p in particles]
            worst_indices = np.argsort(particle_scores)[-num_particles//4:]
            
            for idx in worst_indices:
                # FFT-based filtering
                h = particles[idx]
                H = np.fft.fft(h)
                # High-pass filter
                freqs = np.fft.fftfreq(n_points, dx)
                mask = np.abs(freqs) > 0.3
                H_filtered = H * mask
                h_filtered = np.real(np.fft.ifft(H_filtered))
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = normalize_h(h_filtered, target_sum)
                particles[idx] = h_filtered
                
        # Gradient refinement every 100 iterations
        if iteration % 100 == 0 and iteration > 0:
            particle_scores = [compute_c5(p, dx) for p in particles]
            best_particle_indices = np.argsort(particle_scores)[:10]
            
            for idx in best_particle_indices:
                h = particles[idx].copy()
                
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                # SLSQP optimization
                try:
                    result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                    constraints=constraint, options={'maxiter': 100})
                    if result.success:
                        h_refined = result.x
                        h_refined = np.clip(h_refined, 0, 1)
                        h_refined = normalize_h(h_refined, target_sum)
                        particles[idx] = h_refined
                except:
                    pass
                
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
