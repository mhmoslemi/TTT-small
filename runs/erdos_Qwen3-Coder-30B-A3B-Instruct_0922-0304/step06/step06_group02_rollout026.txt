import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    max_iter = 100
    inertia = 0.7
    inertia_decay = (0.4 / inertia) ** (1.0 / max_iter)
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            ph = np.array(initial_h_values)
            ph = np.clip(ph, 0, 1)
            ph = normalize_h(ph, target_sum)
        else:
            ph = np.ones(n_points)
            ph[n_points//2:] = 0
            ph = np.clip(ph, 0, 1)
            ph = normalize_h(ph, target_sum)
        
       particles.append(ph.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(ph.copy())
        p_best_c5.append(compute_c5(ph, dx))
    
    global_best = h.copy()
    global_best_c5 = compute_c5(h, dx)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                           cognitive * r1 * (p_best[i] - particles[i]) + 
                           social * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = current_c5
                
                if current_c5 < global_best_c5:
                    global_best = particles[i].copy()
                    global_best_c5 = current_c5
                    
        inertia *= inertia_decay
        iteration += 1
        
        # Periodic local refinement using L-BFGS-B
        if iteration % 10 == 0:
            best_refined = global_best.copy()
            best_refined_c5 = global_best_c5
            
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Try several local optimizations
            for _ in range(5):
                noise = np.random.normal(0, 0.005, n_points)
                candidate = np.clip(global_best + noise, 0, 1)
                candidate = normalize_h(candidate, target_sum)
                
                result = minimize(objective, candidate, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
                
                if result.success:
                    refined_candidate = result.x
                    refined_candidate = np.clip(refined_candidate, 0, 1)
                    refined_candidate = normalize_h(refined_candidate, target_sum)
                    refined_c5 = compute_c5(refined_candidate, dx)
                    
                    if refined_c5 < best_refined_c5:

                        best_refined_c5 = refined_c5

                global_best = best_refined.copy()
                global_best_c5 = best_refined_c5
                
        if global_best_c5 <= 0.3807:

    final_c5 = compute_c5(global_best, dx)
    return (global_best.tolist(), final_c5, n_points)
