import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x) using cosine basis."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs)) * np.pi * x[:, None]),
        axis=1
    )
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    
    # Convert to Fourier representation
    h_fft = np.fft.fft(h_init)
    initial_coeffs = np.concatenate([np.real(h_fft[:n_freq+1]), np.imag(h_fft[1:n_freq])])
    
    # Objective function for differential evolution
    def de_objective(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        c5 = compute_c5(h, dx)
        return c5
    
    # Bounds for coefficients
    bounds = [(-1.0, 1.0) for _ in range(2 * n_freq)]
    
    # Differential evolution to search global minima
    de_result = differential_evolution(
        de_objective,
        bounds,
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        disp=False,
        tol=1e-6
    )
    
    # Extract best coefficients
    best_coeffs = de_result.x
    h_best = fourier_to_h(best_coeffs, n_points)
    h_best = np.clip(h_best, 0, 1)
    h_best = normalize_h(h_best, target_sum)
    
    # Local refinement with SLSQP
    def local_objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds_local = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(local_objective, h_best, method='SLSQP', bounds=bounds_local, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h_best = result.x
        h_best = np.clip(h_best, 0, 1)
        h_best = normalize_h(h_best, target_sum)
    
    final_c5 = compute_c5(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
