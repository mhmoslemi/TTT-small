import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutation(h, strength):
    """Apply mutation to h with given strength."""
    noise = np.random.normal(0, strength, len(h))
    h_mut = h + noise
    return np.clip(h_mut, 0, 1)

def crossover(parent1, parent2, rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > rate:
        return parent1.copy()
    
    point = np.random.randint(1, len(parent1)-1)
    child = np.concatenate([parent1[:point], parent2[point:]])
    return child

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill population with diverse individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial population
    fitness = []
    for h in population:
        c5 = compute_c5(h, dx)
        fitness.append(c5)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # GA loop
    generations = 0
    max_gens = 60  # Each generation ~10s, so 60 gens = 600 sec budget
    
    # Mutation parameters
    max_mutation_strength = 0.2
    min_mutation_strength = 0.005
    
    while time.time() - start_time < budget_s - 1 and generations < max_gens:
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = [sorted_population[0].copy(), sorted_population[1].copy()]  # Elitism
        
        # Generate offspring
        for _ in range(population_size - 2):
            parent1 = tournament_selection(sorted_population, sorted_fitness)
            parent2 = tournament_selection(sorted_population, sorted_fitness)
            
            child = crossover(parent1, parent2)
            mutation_strength = max_mutation_strength * np.exp(-generations / 20)
            child = mutation(child, mutation_strength)
            child = np.clip(child, 0, 1)
            child = normalize_h(child, target_sum)
            new_population.append(child)
        
        # Update population and fitness
        population = new_population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        generations += 1
    
    return (best_h.tolist(), best_c5, n_points)
