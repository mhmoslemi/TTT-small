import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    # Initialize best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use an evolutionary strategy-like approach with local refinement
    iter_count = 0
    max_iter = 10000
    
    # CMA-ES inspired local search with multiple randomizations
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Generate new candidate by adding noise to current h
        noise_scale = 0.02
        h_new = h + np.random.normal(0, noise_scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Local optimization step: try SLSQP refinement
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h_new.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
        except:
            pass  # If optimization fails, fallback to the previous candidate
            
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        # Early exit if we've hit our target
        if best_c5 <= 0.3807:
            break
            
        # Update current solution and add some mutation
        h = h_new.copy()
        h += np.random.normal(0, 0.001, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iter_count += 1
        
        # Occasionally restart from the best solution
        if iter_count % 200 == 0:
            h = best_h.copy()
    
    # Final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = normalize_h(final_h, target_sum)
            final_c5 = compute_c5(final_h, dx)
            if final_c5 < best_c5:
                best_h = final_h.copy()
                best_c5 = final_c5
    except:
        pass
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
