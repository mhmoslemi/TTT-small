import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Main optimization loop adapted from enhanced perturbation approach
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Cross-correlation based perturbation
        overlap = correlate(h, 1 - h, mode="full")
        # Identify high overlap regions
        peak_indices = np.argsort(overlap)[-10:]  # top 10%
        
        # Create a perturbation mask focused on these regions
        mask = np.zeros(n_points)
        # Periodically update mask with high overlap regions
        if iteration % 10 == 0:
            mask = np.zeros(n_points)
            for peak_idx in peak_indices:
                # Get the center of the peak
                center = peak_idx - n_points + 1
                # Set a focal region around this peak
                for i in range(n_points):
                    # A Gaussian-like influence with decreasing strength
                    dist = abs(i - (center + n_points//2))
                    influence = np.exp(-dist**2 / (2 * (n_points//10)**2))
                    mask[i] += influence
                
        # Apply perturbation
        noise = np.random.normal(0, 0.02 * (0.95**(iteration // 500)), n_points)
        # Only add noise in high overlap regions
        perturbation = mask * noise
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Run optimization with constraints
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-9})
        if result.success:
            h = result.x
        else:
            # If optimization failed, do a random perturbation
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
