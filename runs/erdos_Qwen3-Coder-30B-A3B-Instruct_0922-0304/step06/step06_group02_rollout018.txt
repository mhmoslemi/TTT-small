import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def genetic_search(initial_h, n_points, dx, target_sum, budget_s):
    """Perform genetic algorithm search for improved solution."""
    start_time = time.time()
    
    pop_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = max(1, pop_size // 10)
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = initial_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = [compute_c5(h, dx) for h in population]
        
        # Track best
        for i, f in enumerate(fitness):
            if f < best_c5:
                best_c5 = f
                best_h = population[i].copy()
                
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # New population
        new_population = population[:elite_size].copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            parents = []
            for _ in range(2):
                tourn_size = 3
                tourn_indices = np.random.choice(pop_size, tourn_size, replace=False)
                tourn_fitness = [fitness[i] for i in tourn_indices]
                winner_idx = tourn_indices[np.argmin(tourn_fitness)]
                parents.append(population[winner_idx])
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parents[0][:crossover_point], parents[1][crossover_point:]])
                child2 = np.concatenate([parents[1][:crossover_point], parents[0][crossover_point:]])
            else:
                child1, child2 = parents[0].copy(), parents[1].copy()
                
            # Mutation with constraints
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
                    
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Local refinement with SLSQP 
        if generation % 5 == 0:  # Refine top 10%
            top_indices = sorted_indices[:elite_size]
            for idx in top_indices:
                h = population[idx].copy()
                
                def objective(h):
                    return compute_c5(h, dx)
                    
                bounds = [(0, 1) for _ in range(n_points)]
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                # SLSQP optimization
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                
                if result.success:
                    refined = result.x.copy()
                    refined = np.clip(refined, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    population[idx] = refined
                    
        generation += 1
        
    return best_h, best_c5

    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Try genetic algorithm with local search
    best_h, best_c5 = genetic_search(h, n_points, dx, target_sum, budget_s)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
