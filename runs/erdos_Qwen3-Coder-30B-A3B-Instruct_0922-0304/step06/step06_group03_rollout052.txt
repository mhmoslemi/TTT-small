import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple initial guesses."""
    guesses = []
    
    # Use existing initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern: high values at regular intervals
    h = np.zeros(n_points)
    step = max(1, n_points // 10)
    for i in range(0, n_points, step):
        h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Random pattern with constraint
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.ones(n_points)
    h[1::2] = 0.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Low-high pattern
    h = np.full(n_points, 0.5)
    h[:n_points//4] = 1.0
    h[3*n_points//4:] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Provide fallback with the initial construction
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
        c5 = compute_c5(h0, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h0.copy()
    
    iteration = 0
    max_iter = 15000
    perturb_std = 0.02
    
    # Optimization loop over initial guesses
    for initial_h in initial_guesses:
        h = initial_h.copy()
        local_best_h = h.copy()
        local_best_c5 = compute_c5(h, dx)
        
        # Fine tune with SLSQP for a while
        for i in range(max_iter // len(initial_guesses) + 1):
            if time.time() - start_time >= budget_s - 1:
                break
                
            iteration += 1
            
            # Update best solution seen so far
            if local_best_c5 < best_c5:
                best_c5 = local_best_c5
                best_h = local_best_h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Store current best and update with optimization
            current_h = h.copy()
            
            # Optimization with constraints
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 30})
                if result.success:
                    h = result.x
                else:
                    h = current_h.copy()
            except:
                h = current_h.copy()
            
            # Apply deterministic perturbation
            noise = np.random.normal(0, perturb_std, n_points)
            h = np.clip(h + noise, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Check if this is a new local best
            new_c5 = compute_c5(h, dx)
            if new_c5 < local_best_c5:
                local_best_c5 = new_c5
                local_best_h = h.copy()
                
            # Reduce perturbation over time
            if iteration % 100 == 0:
                perturb_std *= 0.95
        
        if local_best_c5 < best_c5:
            best_c5 = local_best_c5
            best_h = local_best_h.copy()
    
    if best_h is None:
        # Fallback: return the initial guess
        best_h = initial_guesses[0] if initial_guesses else np.ones(n_points) * 0.5
        best_h = normalize_h(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
