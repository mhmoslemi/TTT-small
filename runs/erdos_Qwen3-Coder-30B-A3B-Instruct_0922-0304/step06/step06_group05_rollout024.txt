import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with line search
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    def project(h):
        """Project h onto feasible set: [0,1] and sum = n/2"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def line_search(h, grad, max_iter=5):
        """Perform backtracking line search to find good step size"""
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        current_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            new_h = project(h - eta * grad)
            new_c5 = compute_c5(new_h)
            if new_c5 <= current_c5 - alpha * eta * np.dot(grad, grad):
                return eta, new_h
            eta *= beta
        return eta, h
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically using max_k
        g = np.zeros(n_points)
        for i in range(n_points):
            # We compute partial derivative of max_k overlap w.r.t. h[i]
            # The gradient direction for current max_k
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            else:
                g[i] = -h[i - (n_points - max_k)] * dx

        # Perform line search
        eta, new_h = line_search(h, g)
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
