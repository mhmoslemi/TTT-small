import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple initial guesses."""
    guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern
    h = np.zeros(n_points)
    step = max(1, n_points // 10)
    for i in range(0, n_points, step):
        h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Random pattern
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 if i % 2 == 0 else 0.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Sine pattern
    x = np.linspace(0, 2, n_points)
    h = 0.5 + 0.5 * np.sin(2 * np.pi * x)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Optimization settings
    max_iter_per_guess = 5000
    perturb_freq = 10
    coarse_perturb_freq = 50
    
    # Process each initial guess
    for i, h in enumerate(initial_guesses):
        h = h.copy()
        iteration = 0
        last_improvement = 0
        perturb_std = 0.02
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter_per_guess:
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
            
            if best_c5 <= 0.3807:
                break
                
            # Perturb based on iteration
            if iteration % coarse_perturb_freq == 0:
                # Coarse perturbation
                h = np.clip(h + np.random.normal(0, perturb_std * 2, n_points), 0, 1)
            elif iteration % perturb_freq == 0:
                # Regular perturbation
                h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
            
            h = normalize_h(h, target_sum)
            
            # Use a hybrid approach: sometimes do gradient step, sometimes just perturb
            if iteration % 3 == 0:
                # Try to optimize
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 10})
                    if result.success:
                        h = result.x
                    else:
                        # If optimization failed, just perturb
                        h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
                except:
                    # Fallback to perturbation if optimization fails
                    h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
                
                h = normalize_h(h, target_sum)
            
            iteration += 1
            # Decrease perturbation size over time
            if iteration > 0 and iteration % 100 == 0:
                perturb_std *= 0.95
    
    # If no valid solution found, return the best possible fallback
    if best_h is None:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        best_c5 = compute_c5(h, dx)
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
