import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h values."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for k in range(n_freq):
        if k == 0:
            h += coeffs[k]
        else:
            h += coeffs[k*2] * np.cos(k * np.pi * x) + coeffs[k*2+1] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h values to Fourier coefficients."""
    coeffs = np.zeros(n_freq * 2)
    x = np.linspace(0, 2, len(h))
    for k in range(n_freq):
        if k == 0:
            coeffs[k] = np.mean(h)
        else:
            coeffs[k*2] = 2 * np.mean(h * np.cos(k * np.pi * x))
            coeffs[k*2+1] = 2 * np.mean(h * np.sin(k * np.pi * x))
    return coeffs

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initialize with Fourier representation
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_coeffs = h_to_fourier(h, n_freq)
    else:
        # Start with simple rectangular pulse
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_coeffs = h_to_fourier(h, n_freq)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = np.random.uniform(-1, 1, (n_particles, n_freq * 2))
    velocities = np.zeros((n_particles, n_freq * 2))
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, float('inf'))
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    # Set initial particle positions to Fourier coeffs of initial h
    particles[0] = initial_coeffs
    
    for iter_idx in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        for i in range(n_particles):
            # Convert coefficients to h
            h = fourier_to_h(particles[i], n_points, n_freq)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Compute C5
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < personal_best_fitness[i]:
                personal_best_fitness[i] = c5
                personal_best[i] = particles[i].copy()
                
            # Update global best
            if c5 < global_best_fitness:
                global_best_fitness = c5
                global_best_idx = i
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
    # Get final best solution
    final_coeffs = personal_best[global_best_idx]
    h = fourier_to_h(final_coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    final_c5 = compute_c5(h, dx)
    
    # Local SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        h = result.x
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        final_c5 = compute_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
