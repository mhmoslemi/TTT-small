import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to satisfy integral constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Simple adjustment - scale to meet constraint
            scale = (n_points / 2) / current_sum
            h = h * scale
        return h

    def compute_gradient(h, k_star):
        # Compute gradient for the specific lag k_star
        g = np.zeros(n_points)
        for i in range(n_points):
            # Using analytical derivative of the contribution to max_overlap
            # Overlap at lag k* is sum_i h[i] * (1 - h[(i + k_star) % n_points])
            if (i + k_star) < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                g[i] = -h[i - k_star] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 10000
    iteration = 0
    step_size = 0.01
    step_decay = 0.5
    step_decay_freq = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically
        g = compute_gradient(h, max_k)
        
        # Update h using gradient descent
        new_h = h - step_size * g
        
        # Apply projection to maintain constraints
        new_h = project_h(new_h)
        h = new_h
        
        # Adaptive step size
        if iteration % step_decay_freq == 0:
            step_size *= step_decay
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
