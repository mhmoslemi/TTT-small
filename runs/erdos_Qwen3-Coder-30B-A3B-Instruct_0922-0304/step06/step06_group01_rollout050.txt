import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Use a simple initial setup
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize h
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO-inspired approach with particle positions in Fourier space
    n_freq = 15  # Frequency components for representation
    n_particles = 50
    max_iter = 500
    
    # Initialize particles with random Fourier coefficients
    particles = np.random.rand(n_particles, 2*n_freq + 1) * 2 - 1  # [-1, 1]
    velocities = np.random.rand(n_particles, 2*n_freq + 1) * 0.1
    pbest_positions = particles.copy()
    pbest_c5 = [float('inf')] * n_particles
    gbest_position = None
    gbest_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Generate the x values for reconstruction (we'll use linear spacing)
    x = np.linspace(0, 2, n_points)

    def reconstruct_h(coeffs):
        """Reconstruct h from Fourier coefficients."""
        h = np.zeros(n_points)
        # Includes constant term (k=0)
        h += coeffs[0] * np.ones(n_points)
        # Add sine and cosine terms
        for i in range(1, n_freq + 1):
            k = i
            h += coeffs[2*i-1] * np.cos(k * np.pi * x)
            h += coeffs[2*i] * np.sin(k * np.pi * x)
        return h

    def project_and_normalize(h):
        """Project h to [0,1] and normalize to target sum."""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        return h

    # PSO iteration
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for p in range(n_particles):
            # Reconstruct h for particle p
            coeffs = particles[p]
            h = reconstruct_h(coeffs)
            h = project_and_normalize(h)
            
            # Evaluate
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_c5[p]:
                pbest_c5[p] = c5
                pbest_positions[p] = coeffs.copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_position = coeffs.copy()
                
            # Update best solution
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Early exit if we've hit our target
            if best_c5 <= 0.3807:
                break

        # Update velocities and positions for all particles
        if gbest_position is not None:
            for p in range(n_particles):
                r1, r2 = np.random.rand(2)
                velocities[p] = (w * velocities[p] + 
                                c1 * r1 * (pbest_positions[p] - particles[p]) + 
                                c2 * r2 * (gbest_position - particles[p]))
                particles[p] += velocities[p]
                
        if best_c5 <= 0.3807:
            break
    
    # Post-refinement with local optimization
    if gbest_position is not None:
        refinement_h = reconstruct_h(gbest_position)
        refinment_h = project_and_normalize(refinment_h)
        
        # Define objective and constraints for SLSQP
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Local optimization
        result = minimize(objective, refinment_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})

        if result.success:
            refinment_h = np.clip(result.x, 0, 1)
            refinment_h = project_and_normalize(refinment_h)
            c5_refined = compute_c5(refinment_h, dx)
            if c5_refined < best_c5:
                best_c5 = c5_refined
                best_h = refinment_h.copy()
    
    # Final re-evaluation to ensure correctness
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
