import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, offset=0):
    """Create a comb pattern with given spacing and offset."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_width = dx
    for i in range(0, n_points, spacing):
        start_idx = (i + offset) % n_points
        end_idx = min(start_idx + 1, n_points)
        h[start_idx:end_idx] = 1.0
    return h

def comb_pattern_optimization(initial_h, dx, target_sum, n_points):
    """Optimize a comb pattern using local search."""
    h = initial_h.copy()
    h = normalize_h(h, target_sum)
    
    def objective(h):
        return evaluate_h(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h = result.x
    else:
        # If optimization failed, perturb and try again
        h = h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    return h

        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Explore combs with different spacings
    spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    
    # Generate comb patterns
    for s in spacings:
        for offset in [0, s//2]:
            if offset < s:
                comb = create_comb_pattern(n_points, s, offset)
                comb_patterns.append(comb)
    
    # Optimization loop
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate initial solution
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Explore comb patterns
        best_comb_c5 = float('inf')
        best_comb_h = h.copy()
        
        for comb in comb_patterns:
            # Optimize comb pattern
            optimized_comb = comb_pattern_optimization(comb, dx, target_sum, n_points)
            comb_c5 = evaluate_h(optimized_comb, dx)
            
            if comb_c5 < best_comb_c5:
