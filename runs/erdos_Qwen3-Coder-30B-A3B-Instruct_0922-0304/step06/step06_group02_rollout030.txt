import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use adaptive perturbation and gradient-based local search
    iteration = 0
    max_iter = 10000
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Vary learning rate and noise levels
    lr = 0.01
    noise_level = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Perturb h
        h_perturbed = h + np.random.normal(0, noise_level, len(h))
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_perturbed, dx)
        
        # Accept if better
        if new_c5 < current_c5:
            h = h_perturbed.copy()
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                
                # Reduce learning rate and noise for fine tuning
                lr = max(0.001, lr * 0.99)
                noise_level = max(0.001, noise_level * 0.99)
        else:
            # Occasionally accept worse moves to escape local optima
            if np.random.random() < 0.05:
                h = h_perturbed.copy()
                current_c5 = new_c5
        
        # Periodic update to force exploration
        if iteration % 50 == 0:
            # Add a larger random perturbation
            h = np.clip(h + np.random.normal(0, 0.05, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            current_c5 = compute_c5(h, dx)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
