import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a more sophisticated initial construction
        h = np.zeros(n_points)
        # Create a smooth step function
        mid_point = n_points // 2
        for i in range(n_points):
            if i < mid_point:
                h[i] = min(1.0, i / (mid_point / 2))
            else:
                h[i] = max(0.0, 1.0 - (i - mid_point) / (mid_point / 2))
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize large perturbation mask
    noise_amplitude = 0.02
    
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to identify high overlap regions
        corr = correlate(h, 1 - h, mode="full")
        corr_indices = np.argsort(corr)[-10:]  # Top 10% peaks
        
        # Create perturbation mask
        perturbation_mask = np.zeros(n_points)
        for idx in corr_indices:
            if idx < n_points:
                # Focus on contributing parts
                left = max(0, idx - 2)
                right = min(n_points, idx + 3)
                perturbation_mask[left:right] = 1
            
        # Apply perturbation with selective focus
        noise = np.random.normal(0, noise_amplitude, n_points)
        perturbed_h = h + perturbation_mask * noise
        
        # Add some random noise to escape local minima
        perturbed_h = perturbed_h + np.random.normal(0, 0.01, n_points)
        
        h = np.clip(perturbed_h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally perform SLSQP refinement
        if iteration % 10 == 0 and iteration > 0:
            # Define constraints for optimization
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Restart SLSQP if we're close to convergence
            try:
                result = minimize(lambda x: compute_c5(x, dx), h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h = result.x
                else:
                    # Fallback random perturbation
                    h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
                    h = normalize_h(h, target_sum)
            except:
                h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
                h = normalize_h(h, target_sum)
        
        noise_amplitude *= 0.995  # Gradually reduce noise
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
