import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with cutting-plane approach
    iteration = 0
    max_iter = 5000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lag
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst lag
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        worst_lag = max_idx - (len(h) - 1)
        
        # Solve optimization to tighten constraint for worst lag
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_overlap(h):
            # Constraint for the worst lag
            # This is a simplified approach: we compute the overlap for the
            # worst lag and ensure it doesn't exceed the current maximum
            overlap_lag = 0
            if worst_lag >= 0:
                for i in range(len(h) - worst_lag):
                    overlap_lag += h[i] * (1 - h[i + worst_lag])
            else:
                for i in range(-worst_lag, len(h)):
                    overlap_lag += h[i] * (1 - h[i + worst_lag])
            return overlap_lag * dx - current_c5
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'ineq', 'fun': constraint_overlap}
        
        # Try to adjust h to reduce overlap at worst lag
        h_initial = h.copy()
        # Small random perturbation to avoid local minima
        noise = np.random.normal(0, 0.01, len(h))
        h_noisy = h_initial + noise
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        # Try solving with the constraint
        result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, 
                          constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Check if this improves
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 and candidate_c5 < best_c5 * 0.9999:
                h = h_candidate.copy()
        else:
            # If optimization fails, try direct perturbation
            h_perturbed = h + np.random.normal(0, 0.02, len(h))
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
            h = h_perturbed.copy()
            
        iteration += 1
    
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
