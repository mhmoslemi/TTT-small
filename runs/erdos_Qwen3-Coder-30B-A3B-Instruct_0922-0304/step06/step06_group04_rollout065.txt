import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Convex optimization with line search
    def surrogate_loss(h, epsilon=1e-3):
        # Compute the overlap integral
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        # Add small regularization to make it differentiable
        reg = epsilon * np.sum(h**2)
        return max_overlap * dx - reg
    
    def gradient(h):
        # Approximate gradient based on finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        base_loss = surrogate_loss(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = normalize_h(h_plus, target_sum)
            h_plus = np.clip(h_plus, 0, 1)
            loss_plus = surrogate_loss(h_plus)
            grad[i] = (loss_plus - base_loss) / eps
        return grad
    
    # Line search to find good step size
    def line_search(h, grad, alpha_init=0.1, beta=0.9):
        alpha = alpha_init
        for _ in range(20):  # limit iterations
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            if surrogate_loss(h_new) < surrogate_loss(h):
                return h_new, alpha
            alpha *= beta
        return h, alpha_init
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        h = best_individual.copy()
        
        # Gradient descent step
        grad = gradient(h)
        h_new, _ = line_search(h, grad)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h_new.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
