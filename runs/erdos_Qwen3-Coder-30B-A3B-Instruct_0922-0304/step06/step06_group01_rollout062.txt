import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    sa_steps = 0
    max_sa_steps = 100000
    
    # Local refinement parameters
    local_refine_freq = 10000
    local_refine_iter = 200
    
    while time.time() - start_time < budget_s - 1 and sa_steps < max_sa_steps:
        # Current C5 value
        current_c5 = compute_c5(h, dx)
        
        # Perturb h: randomly change 10% of the values
        num_perturbed = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, num_perturbed, replace=False)
        h_new = h.copy()
        for i in indices:
            h_new[i] = np.random.random()
        
        # Normalize to maintain sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5 value
        new_c5 = compute_c5(h_new, dx)
        
        # Metropolis acceptance criterion
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new.copy()
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Check for early termination
        if best_c5 <= 0.3807:
            break
            
        sa_steps += 1
        
        # Cool down temperature
        if sa_steps % 100 == 0:
            T = max(min_T, T * cooling_rate)
            
        # Occasional local refinement
        if sa_steps % local_refine_freq == 0:
            # Use SLSQP for local refinement
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h_new = result.x
                    h_new = np.clip(h_new, 0, 1)
                    h_new = normalize_h(h_new, target_sum)
                    h = h_new.copy()
                    c5 = compute_c5(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
            except:
                pass  # If optimization fails, continue with current solution
    
    # Final local refinement
    try:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            final_c5 = compute_c5(h_final, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final.copy()
    except:
        pass  # If optimization fails, proceed with best_h
    
    return (best_h.tolist(), best_c5, n_points)
