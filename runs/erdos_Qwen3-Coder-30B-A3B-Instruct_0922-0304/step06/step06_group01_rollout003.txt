import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use particle swarm optimization approach
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles in Fourier space
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        # Sample random Fourier coefficients
        a = np.random.normal(0, 0.5, n_freq)
        b = np.random.normal(0, 0.5, n_freq)
        particles.append((a, b))
        velocities.append((np.zeros(n_freq), np.zeros(n_freq)))
    
    # Initialize global best
    g_best_c5 = float('inf')
    g_best_params = None
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, (a, b) in enumerate(particles):
            # Generate h from Fourier components
            x = np.linspace(0, 2, n_points)
            h_trial = np.zeros(n_points)
            for k in range(n_freq):
                h_trial += a[k] * np.cos(k * np.pi * x) + b[k] * np.sin(k * np.pi * x)
            
            # Apply non-negativity and normalization
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial * (target_sum / np.sum(h_trial))
            
            # Compute C5
            c5 = compute_c5(h_trial, dx)
            
            # Update particle best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_trial.copy()
                
            # Update global best
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best_params = (a.copy(), b.copy())
            
            # Early termination
            if g_best_c5 <= 0.3807:
                break
                
        # Update particles
        for i in range(n_particles):
            a, b = particles[i]
            va, vb = velocities[i]
            pa, pb = g_best_params
            
            # Update velocity
            r1, r2 = np.random.rand(2, n_freq)
            va = w * va + c1 * r1 * (pa - a) + c2 * r2 * (g_best_params[0] - a)
            vb = w * vb + c1 * r1 * (pb - b) + c2 * r2 * (g_best_params[1] - b)
            
            # Update position
            a += va
            b += vb
            
            particles[i] = (a, b)
            velocities[i] = (va, vb)
    
    # Final SLSQP refinement of best solution
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    if time.time() - start_time < budget_s - 1:
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * (target_sum / np.sum(h_final))
            final_c5 = compute_c5(h_final, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final.copy()
    
    return (best_h.tolist(), best_c5, n_points)
