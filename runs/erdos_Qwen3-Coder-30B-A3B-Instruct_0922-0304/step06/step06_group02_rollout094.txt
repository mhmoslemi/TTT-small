import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, rate=0.05, target_sum=35.0):
    """Mutate h with constraint maintenance."""
    h = h.copy()
    mask = np.random.random(len(h)) < rate
    h[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def crossover_h(h1, h2, rate=0.7):
    """Crossover two h vectors."""
    h1, h2 = h1.copy(), h2.copy()
    mask = np.random.random(len(h1)) < rate
    h1[mask] = h2[mask]
    return h1

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Population-based genetic algorithm
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Create initial population
    population = []
    for _ in range(population_size):
        h_candidate = h + np.random.normal(0, 0.02, len(h))
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = normalize_h(h_candidate, target_sum)
        population.append(h_candidate)
    
    population = np.array(population)
    
    iteration = 0
    max_iter = 2000  # Enough iterations for GA to converge
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitnesses = []
        for h_i in population:
            c5 = compute_c5(h_i, dx)
            fitnesses.append(c5)
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 2]
        
        # Create new population
        new_population = [population[i] for i in top_indices]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            parent1_idx, parent2_idx = np.random.choice(top_indices, 2, replace=False)
            parent1, parent2 = population[parent1_idx], population[parent2_idx]
            
            # Crossover
            if np.random.random() < crossover_rate:
                child = crossover_h(parent1, parent2, crossover_rate)
            else:
                child = parent1.copy()
            
            # Mutation
            child = mutate_h(child, mutation_rate, target_sum)
            new_population.append(child)
        
        # Ensure exact size
        population = np.array(new_population[:population_size])
        
        # Local refinement with SLSQP for top 10%
        top_refine = population_size // 10
        if top_refine > 0:
            for i in range(top_refine):
                idx = sorted_indices[i]
                h_refined = population[idx].copy()
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    return compute_c5(h, dx)
                
                result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                
                if result.success:
                    refined_candidate = result.x
                    refined_candidate = np.clip(refined_candidate, 0, 1)
                    refined_candidate = normalize_h(refined_candidate, target_sum)
                    population[idx] = refined_candidate
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
