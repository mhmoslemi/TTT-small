import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple staircase-like construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Discretize h to allowed values: [0, 0.25, 0.5, 0.75, 1]
    allowed_values = np.array([0, 0.25, 0.5, 0.75, 1.0])
    
    def discrete_project(h_vec):
        """Project h onto discrete allowed values."""
        result = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Find closest allowed value
            diffs = np.abs(h_vec[i] - allowed_values)
            result[i] = allowed_values[np.argmin(diffs)]
        return result
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    h = discrete_project(h)
    current_sum = np.sum(h)
    if current_sum > target_sum:
        # Reduce values to sum to target
        diff = current_sum - target_sum
        # Find positions with largest values and reduce them
        idxs = np.argsort(h)[::-1]
        reduced = 0
        for idx in idxs:
            if reduced >= diff:
                break
            reduce_amount = min(h[idx], diff - reduced)
            h[idx] -= reduce_amount
            reduced += reduce_amount
    elif current_sum < target_sum:
        # Increase values to sum to target
        diff = target_sum - current_sum
        idxs = np.argsort(h)
        increased = 0
        for idx in idxs:
            if increased >= diff:
                break
            increase_amount = min(1 - h[idx], diff - increased)
            h[idx] += increase_amount
            increased += increase_amount
    
    # Ensure sum constraint exactly
    current_sum = np.sum(h) 
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust arbitrarily to meet constraint
        # Find smallest value to reduce
        smallest_idx = np.argmin(h)
        h[smallest_idx] = max(0, h[smallest_idx] - (current_sum - target_sum))
    
    # Evaluate initial solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Search strategy: gradient-free local search with random perturbations and discrete projection
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if best_c5 <= 0.3807:
            break
        
        # Random perturbation
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Discretize
        h_proj = discrete_project(h_perturbed)
        
        # Re-normalize
        current_sum = np.sum(h_proj)
        if current_sum > 0:
            h_proj = h_proj * (target_sum / current_sum)
        
        # Make sure sum is exactly target_sum
        h_proj = h_proj * (target_sum / np.sum(h_proj))
        
        # Ensure values are within discrete set
        h_proj = discrete_project(h_proj)
        
        # Final re-normalization if needed
        current_sum = np.sum(h_proj)
        h_proj = h_proj * (target_sum / current_sum)
        
        c5 = compute_c5(h_proj, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_proj.copy()            
        
        # Modify h for next iteration
        h = h_proj.copy()
        
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
