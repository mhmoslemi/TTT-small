import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n/2"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Adjust towards mean to maintain sum
            adjustment = (n_points / 2 - current_sum) / n_points
            h += adjustment
            h = np.clip(h, 0, 1)
        return h
    
    def compute_gradient(h, k_star):
        """Compute analytical gradient for the maximum overlap lag k_star"""
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
        return g
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with adaptive step size
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute overlap and find maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(overlap)
        max_overlap = overlap[k_star] * dx
        
        # Store best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient at the maximum overlap lag
        g = compute_gradient(h, k_star)
        
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Project to feasible set
        new_h = project_h(new_h)
        h = new_h
        
        # Adapt step size
        if iteration % 1000 == 0 and iteration > 0:
            step_size *= 0.5
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.2:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_h(h)
            
        iteration += 1

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
