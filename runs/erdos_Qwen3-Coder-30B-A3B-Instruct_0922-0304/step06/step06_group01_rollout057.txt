import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x) using inverse DFT."""
    # coeffs: [a0, a1, b1, a2, b2, ...]
    n_freq = (len(coeffs) - 1) // 2
    
    # Reconstruct the function values on the grid
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initialize with DC component
    h = coeffs[0] / 2.0 * np.ones_like(x)
    
    # Add cosine and sine terms
    for k in range(1, n_freq + 1):
        a_k = coeffs[2*k - 1]
        b_k = coeffs[2*k]
        h += a_k * np.cos(k * np.pi * x) + b_k * np.sin(k * np.pi * x)
        
    return h

def fit_fourier_coeffs(h_values, n_freq):
    """Fit Fourier coefficients to given h(x)."""
    # Use FFT to get coefficients
    n_points = len(h_values)
    # Pad with zeros for proper DFT
    h_padded = np.concatenate([h_values, np.zeros(n_points)])
    fft_coeffs = np.fft.fft(h_padded)
    
    # Get real part as a0, a1, b1, ...
    coeffs = np.zeros(2 * n_freq + 1)
    coeffs[0] = 2 * np.real(fft_coeffs[0]) / n_points
    
    for k in range(1, n_freq + 1):
        coeffs[2*k - 1] = 2 * np.real(fft_coeffs[k]) / n_points
        coeffs[2*k] = 2 * np.imag(fft_coeffs[k]) / n_points
    return coeffs

def objective_fourier(coeffs, n_points, target_sum, dx):
    """Objective function for DE in Fourier space."""
    h = fourier_to_h(coeffs, n_points)
    h_clipped = np.clip(h, 0, 1)
    h_normal = h_clipped * (target_sum / np.sum(h_clipped))
    return compute_c5(h_normal, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = np.clip(h_start, 0, 1)
    else:
        # Initialize with a rectangular pulse
        h_start = np.zeros(n_points)
        h_start[:n_points//2] = 1.0
        h_start = np.clip(h_start, 0, 1)
    
    # Normalize initial h
    h_start = h_start * (target_sum / np.sum(h_start))
    
    # Fourier representation: 15 frequencies (a0, a1, b1, a2, b2, ...)
    n_freq = 15
    start_coeffs = fit_fourier_coeffs(h_start, n_freq)
    
    # Define bounds for DE: a0, a1, b1, a2, b2, ...
    bounds = []
    bounds.append((-1.0, 1.0))  # a0
    for k in range(1, n_freq + 1):
        bounds.append((-1.0, 1.0))  # a_k
        bounds.append((-1.0, 1.0))  # b_k
        
    # Global optimization with Differential Evolution
    de_result = differential_evolution(
        objective_fourier,
        bounds,
        args=(n_points, target_sum, dx),
        maxiter=300,
        popsize=50,
        seed=seed,
        callback=lambda x: None,
        disp=False,
        polish=True
    )
    
    # Convert best coefficients to h
    h_best = fourier_to_h(de_result.x, n_points)
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (target_sum / np.sum(h_best))
    
    # Local optimization with SLSQP to refine
    def objective_local(h):
        return compute_c5(h, dx)
        
    bounds_local = [(0, 1) for _ in range(n_points)]
    
    def constraint_local(h):
        return np.sum(h) - target_sum
        
    cons = {'type': 'eq', 'fun': constraint_local}
    
    result_refine = minimize(
        objective_local,
        h_best,
        method='SLSQP',
        bounds=bounds_local,
        constraints=cons,
        options={'maxiter': 100}
    )
    
    if result_refine.success:
        h_refined = result_refine.x
    else:
        h_refined = h_best
    
    h_final = np.clip(h_refined, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    c5_final = compute_c5(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
