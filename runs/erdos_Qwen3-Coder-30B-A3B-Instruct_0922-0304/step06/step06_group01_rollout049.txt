```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cma

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization: rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)

    # Set up CMA-ES
    bounds = [[0, 1] for _ in range(n_points)]
    initial_sigma = 0.1
    popsize = 2 * n_points  # Use 2*n_points population size
    opts = cma.CMAOptions()
    opts.set('verbose', -1)  # Silent mode
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use CMA-ES to search for better h
    es = cma.CMAEvolutionStrategy(h, initial_sigma, opts)
    
    while not es.stop() and time.time() - start_time < budget_s - 1:
        # Generate new population
        solutions = es.ask()
        fitness_values = []
        
        for sol in solutions:
            # Ensure all values in [0,1]
            sol = np.clip(sol, 0, 1)
            # Normalize to meet sum constraint
            sol = normalize_h(sol, target_sum)
            c5 = compute_c5(sol, dx)
            fitness_values.append(c5)
            
        # Tell CMA-ES about fitness
        es.tell(solutions, fitness_values)
        
        # Update best solution
        current_best_idx = np.argmin(fitness_values)
        current_best = solutions[current_best_idx]
        current_c5 = fitness_values[current_best_idx]
        
        current_best = np.clip(current_best, 0, 1)
        current_best = normalize_h(current_best, target_sum)
        current_c5 = compute_c5(current_best, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
    
    # Apply local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum

    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return compute_c5(h, dx)
        
    result = minimize(objective, best_h.copy(), method='SLSQP', 
                      bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        refined_c5 = compute_c5(refined_h, dx)
        if refined_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
