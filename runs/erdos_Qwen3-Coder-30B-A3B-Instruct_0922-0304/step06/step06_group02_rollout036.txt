import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        
    # Normalize
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use convex optimization approach - reformulate as quadratic program
    # Variables: h, and C (scalar)
    h_vars = cp.Variable(n_points)
    C_var = cp.Variable()
    
    # Total sum constraint
    sum_constraint = cp.Equality(cp.sum(h_vars), target_sum)
    
    # All overlap constraints
    constraints = [sum_constraint]
    constraints.append(C_var >= 0)
    
    # For each lag k, add overlap constraint
    for k in range(-n_points+1, n_points):
        # Compute valid indices i where both i and i+k are in range
        if k >= 0:
            i_indices = np.arange(0, n_points - k)
            j_indices = np.arange(k, n_points)
        else:
            i_indices = np.arange(-k, n_points)
            j_indices = np.arange(0, n_points + k)
        
        if len(i_indices) > 0:
            # Sum over valid indices
            valid_sum = cp.sum(cp.multiply(h_vars[i_indices], 1 - h_vars[j_indices])) * dx
            constraints.append(valid_sum <= C_var)
    
    # Objective
    objective = cp.Minimize(C_var)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s-1)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_vars.value
            if h_opt is not None:
                h_opt = np.clip(h_opt, 0, 1)
                # Normalize to target sum
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                c5_opt = C_var.value
                if c5_opt is not None and c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
    except:
        pass
    
    # If no valid convex solution, fallback to local search
    if best_c5 == float('inf'):
        # Fine-tune using local search
        max_iter = 10000
        iteration = 0
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            current_c5 = evaluate_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Perform local search step
            # Generate a noisy version
            h_noisy = h + np.random.normal(0, 0.02, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            # Re-normalize
            h_noisy = h_noisy * (target_sum / np.sum(h_noisy))
            
            # Simple gradient-like adjustment
            h_new = h_noisy.copy()
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            new_c5 = evaluate_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new.copy()
                
            iteration += 1
            
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
