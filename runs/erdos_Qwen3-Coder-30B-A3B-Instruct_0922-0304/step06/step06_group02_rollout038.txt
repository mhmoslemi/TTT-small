import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Create CVXPY variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Add overlap constraints for all lags
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i in [0, n_points - k - 1]
            expr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
        else:
            # h[i] * (1 - h[i + k]) for i in [0, n_points + k - 1]
            expr = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:-k]))
        constraints.append(expr * dx <= C)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solver settings
    solver_opts = {
        'max_iter': 10000,
        'time_limit': budget_s - 1
    }
    
    # Iterative refinement
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        try:
            # Solve the QP
            prob.solve(solver=cp.GUROBI, **solver_opts)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (target_sum / np.sum(h_new))
                    c5_new = C.value
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        
                    # Early exit
                    if best_c5 <= 0.3807:
                        break
                        
                    # Update h for next iteration
                    h = h_new.copy()
                else:
                    # Fallback to local search if optimization fails
                    h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                    h = h * (target_sum / np.sum(h))
            else:
                # Fallback to local search
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h * (target_sum / np.sum(h))
                
        except Exception:
            # Fallback to local search
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (target_sum / np.sum(h))
            
        iteration += 1
    
    # Final check of best solution
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    
    return (best_h.tolist(), final_c5, n_points)
