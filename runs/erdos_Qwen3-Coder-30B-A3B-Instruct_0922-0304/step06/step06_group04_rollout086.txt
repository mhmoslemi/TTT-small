import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with best known construction or simple one
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
        h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = compute_c5(h_base, dx)
    best_h = h_base.copy()
    
    # Try different approaches
    # 1. Direct optimization with CVXPY for bilinear constraints
    
    attempts = 0
    while time.time() - start_time < budget_s - 1 and attempts < 30:
        # Try optimization using CVXPY
        try:
            h_var = cp.Variable(n_points)
            objective = cp.Minimize(cp.max(cp.conv(h_var, 1 - h_var)[n_points-1:2*n_points-1]))
            
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == target_sum
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                c5_opt = compute_c5(h_opt, dx)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                
                if best_c5 <= 0.3807:
                    break
                    
        except:
            pass
            
        # Local perturbation 
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (target_sum / np.sum(h_pert))
        c5_pert = compute_c5(h_pert, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_pert.copy()
            
        attempts += 1
        
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
