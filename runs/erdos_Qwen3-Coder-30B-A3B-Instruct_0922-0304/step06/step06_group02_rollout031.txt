import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple starting function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 5000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Stop if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try to improve using gradient-based method with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # SLSQP with multiple restarts
        best_local_c5 = float('inf')
        best_local_h = h.copy()
        
        for _ in range(5):  # Multiple restarts
            # Add small random perturbation
            h_noisy = h + np.random.normal(0, 0.01, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            try:
                result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 200, 'ftol': 1e-8})
                if result.success:
                    h_candidate = result.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = normalize_h(h_candidate, target_sum)
                    candidate_c5 = compute_c5(h_candidate, dx)
                    
                    if candidate_c5 < best_local_c5:
                        best_local_c5 = candidate_c5
                        best_local_h = h_candidate.copy()
            except:
                continue
                
        if best_local_c5 < current_c5:
            h = best_local_h.copy()
        else:
            # Apply small random perturbation if no progress
            h = np.clip(h + np.random.normal(0, 0.005, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
