import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Use a smooth approximation: ramp up then down
        x = np.linspace(0, 2, n_points)
        base_h = np.maximum(0, 1 - np.abs(x - 1))
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use particle swarm optimization with periodic boundary conditions
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles in frequency domain 
    particles = np.random.uniform(-1, 1, (n_particles, n_freq * 2))
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, float('inf'))
    
    # Global best
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    def fourier_to_h(fourier_coeffs):
        # Map Fourier coefficients to step function
        coeffs_a = fourier_coeffs[:n_freq]
        coeffs_b = fourier_coeffs[n_freq:]
        x = np.linspace(0, 2, n_points)
        h_val = 0.5 * coeffs_a[0]  # DC component
        for k in range(1, n_freq):
            h_val += coeffs_a[k] * np.cos(k * np.pi * x) + coeffs_b[k] * np.sin(k * np.pi * x)
        # Ensure h is in [0,1] and sum is correct
        h_val = np.clip(h_val, 0, 1)
        h_val = normalize_h(h_val, target_sum)
        return h_val
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            h_val = fourier_to_h(particles[i])
            c5 = compute_c5(h_val, dx)
            
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
            if c5 < gbest_c5:
                gbest_idx = i
                gbest_c5 = c5
                best_h = h_val.copy()
                
        # Early stopping
        if gbest_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        r1, r2 = np.random.rand(n_particles, 1), np.random.rand(n_particles, 1)
        velocities = (w * velocities + 
                      c1 * r1 * (pbest - particles) + 
                      c2 * r2 * (gbest_c5 * np.ones((n_particles, 1)) - particles))
        
        particles += velocities
        
        # Periodic boundaries (wrap around)
        particles = np.mod(particles, 2) - 1
    
    # Local refinement at best solution
    best_h_refined = best_h.copy()
    
    # Now try a local optimization as a final refinement
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h_refined, method='SLSQP', 
                      bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        best_h_refined = result.x.copy()
        best_h_refined = np.clip(best_h_refined, 0, 1)
        best_h_refined = normalize_h(best_h_refined, target_sum)
        
    final_c5 = compute_c5(best_h_refined, dx)
    
    return (best_h_refined.tolist(), final_c5, n_points)
