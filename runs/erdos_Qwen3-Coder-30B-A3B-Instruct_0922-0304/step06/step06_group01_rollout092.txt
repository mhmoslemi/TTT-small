import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Generate a comb-like h function."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        tooth_start = i * (tooth_width + spacing)
        tooth_end = tooth_start + tooth_width
        
        # Discretize tooth
        for j in range(n_points):
            x = j * dx
            if tooth_start <= x < tooth_end:
                h[j] = 1.0
            elif tooth_start - slope <= x < tooth_start:
                h[j] = max(0, 1 - (x - (tooth_start - slope)) / slope)
            elif tooth_end <= x < tooth_end + slope:
                h[j] = max(0, 1 - (x - tooth_end) / slope)
    
    return h

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None

    # Try different comb configurations
    for n_teeth in range(5, 11):
        # Sweep tooth parameters
        tooth_widths = np.linspace(2.0/n_teeth * 0.7, 2.0/n_teeth * 1.3, 5)
        slopes = np.linspace(2.0/n_teeth * 0.2, 2.0/n_teeth * 0.8, 5)
        
        for tooth_width in tooth_widths:
            for slope in slopes:
                # Generate comb function
                h = generate_comb_h(n_points, n_teeth, tooth_width, None, slope)
                h = normalize_h(h, target_sum)
                
                # Compute C5
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break

            slope = tooth_width / 2.0
            
            for i in range(n_teeth):
                tooth_start = i * (tooth_width + spacing)
                tooth_end = tooth_start + tooth_width

                    x = j * dx
                    if tooth_start <= x < tooth_end:
