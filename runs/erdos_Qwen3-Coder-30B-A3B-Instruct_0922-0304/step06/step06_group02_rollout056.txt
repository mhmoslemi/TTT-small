import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with direction-aware improvements
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use a more sophisticated local search
        constraints = []
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Add sum constraint
        def constraint_sum(h_in):
            return np.sum(h_in) - target_sum
        constraints.append({'type': 'eq', 'fun': constraint_sum})
        
        def objective(h_in):
            return compute_c5(h_in, dx)
            
        # Try to improve by sampling from local neighborhood
        best_improvement = 0
        best_candidate = h.copy()
        
        for _ in range(5):  # Multiple trials
            # Add noise and project to feasible region
            h_noisy = h + np.random.normal(0, 0.02, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            # This ensures local minimization by avoiding invalid explorations
            try:
                result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, 
                                constraints=constraints, options={'maxiter': 200, 'ftol': 1e-8})
                if result.success:
                    h_candidate = result.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = normalize_h(h_candidate, target_sum)
                    
                    candidate_c5 = compute_c5(h_candidate, dx)
                    improvement = current_c5 - candidate_c5
                    if improvement > best_improvement:
                        best_improvement = improvement
                        best_candidate = h_candidate.copy()
            except:
                continue
                
        # Accept improvement or adjust exploration
        if best_improvement > 1e-8:
            h = best_candidate.copy()
        else:
            # Exploratory perturbation
            noise = np.random.normal(0, 0.04, n_points)
            h = np.clip(h + noise, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
