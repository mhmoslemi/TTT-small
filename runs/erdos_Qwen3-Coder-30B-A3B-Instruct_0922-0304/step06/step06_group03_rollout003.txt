import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple initial guesses."""
    guesses = []
    
    # Initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern
    h = np.zeros(n_points)
    for i in range(0, n_points, 4):
        if i < n_points:
            h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Random
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Low-discrepancy like pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i / n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0

    # Try each initial guess

        if current_c5 < best_c5:
            best_c5 = current_c5

        iter_count = 0
        while time.time() - start_time < budget_s - 1 and iter_count < 5000:

            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # SLSQP optimization with smaller step
            try:
                result = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50, 'ftol': 1e-8})
