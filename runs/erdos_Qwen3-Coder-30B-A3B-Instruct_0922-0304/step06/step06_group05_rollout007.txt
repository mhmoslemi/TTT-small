import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with momentum and adaptive step size
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    velocity = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current max lag
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            g[i] = (1 - h[j]) * dx
        
        # Update velocity with momentum
        velocity = momentum * velocity + step_size * g
        
        # Update h
        new_h = h - velocity
        
        # Project to feasible region
        h = project_h(new_h)
        
        # Adjust step size and momentum
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum = max(0.5, momentum * 0.98)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
