import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.05, target_sum=35):
    """Mutate an individual with constraint enforcement."""
    h_mut = h + np.random.normal(0, 0.02, len(h))
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Single-point crossover
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial population from `initial_h_values` if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    
    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        individual = mutate_individual(initial_h, target_sum=target_sum)
        population.append(individual)

    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    generation = 0
    max_generations = 1000
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate all individuals in population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = ind.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        top_indices = sorted_indices[:population_size // 2]
        top_individuals = [population[i] for i in top_indices]

        # Create new population
        new_population = []
        # Elitism: keep top individuals
        new_population.extend(top_individuals)
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(top_individuals, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate_individual(child1, target_sum=target_sum)
            child2 = mutate_individual(child2, target_sum=target_sum)
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]

        # Use SLSQP to refine top individuals
        top_refine = min(10, len(population) // 2)
        for idx in range(top_refine):
            individual = population[idx]

            def constraint_sum(h):
                return np.sum(h) - target_sum

            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}

            def objective(h):
                return compute_c5(h, dx)

            # SLSQP optimization
            try:
                result = minimize(objective, individual, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined = result.x
                    refined = np.clip(refined, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    population[idx] = refined
            except:
                pass

        generation += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
