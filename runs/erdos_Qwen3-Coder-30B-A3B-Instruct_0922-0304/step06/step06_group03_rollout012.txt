import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    
    # Distribute teeth evenly across the range
    tooth_height = tooth_height * (n_points / 2) / len(tooth_positions)
    for pos in tooth_positions:
        h[pos] = tooth_height
    
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbations to the pattern."""
    perturbed = h + np.random.normal(0, sigma, len(h))
    return np.clip(perturbed, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base comb pattern
        h = create_comb_pattern(n_points, 4)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution search with combs
    spacing_list = [2, 3, 4, 5, 6]
    max_iter = 50000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try different comb structures
        if iteration % 50 == 0:
            best_spacing = 4
            best_spacing_c5 = float('inf')
            for spacing in spacing_list:
                comb_h = create_comb_pattern(n_points, spacing)
                comb_h = normalize_h(comb_h, target_sum)
                c5 = compute_c5(comb_h, dx)
                if c5 < best_spacing_c5:
                    best_spacing_c5 = c5
                    best_spacing = spacing
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = comb_h.copy()
                    if best_c5 <= 0.3807:
                        break
            
            # Use best spacing and perturb
            new_h = create_comb_pattern(n_points, best_spacing)
            new_h = normalize_h(new_h, target_sum)
            h = perturb_pattern(new_h, sigma=0.02)
            h = normalize_h(h, target_sum)
            
        else:
            # Use SLSQP local optimization on current pattern
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Optimize using SLSQP 
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = result.x
            else:
                # If optimization fails, do a random perturbation
                h = perturb_pattern(h, sigma=0.01)
                h = normalize_h(h, target_sum)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
