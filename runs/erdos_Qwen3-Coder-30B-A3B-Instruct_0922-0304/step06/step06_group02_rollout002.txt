import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-temperature optimization parameters
    max_iter = 5000
    temperature = 1.0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Simulated annealing with adaptive temperature
        temp_factor = 0.9995
        temperature = max(0.001, temperature * temp_factor)
        
        # Generate candidate
        h_candidate = h.copy()
        
        # Apply small random perturbations
        noise_scale = 0.05 * temperature
        perturbation = np.random.normal(0, noise_scale, n_points)
        h_candidate += perturbation
        
        # Ensure feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = normalize_h(h_candidate, target_sum)
        
        # Evaluate
        candidate_c5 = compute_c5(h_candidate, dx)
        
        # Accept or reject
        if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / temperature):
            h = h_candidate
        
        # Periodic structural perturbations
        if iteration % 50 == 0:
            # Add a small structured perturbation
            perturb = np.zeros(n_points)
            peak_idx = np.random.randint(0, n_points)
            for i in range(n_points):
                dist = abs(i - peak_idx)
                if dist < n_points // 6:
                    perturb[i] = 0.02 * (1.0 - dist / (n_points // 6))
            
            h = np.clip(h + perturb, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Occasionally do a local optimization
        if iteration % 100 == 0:
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            # Use L-BFGS-B for local refinement
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                h_local = result.x
                h_local = np.clip(h_local, 0, 1)
                h_local = normalize_h(h_local, target_sum)
                
                local_c5 = compute_c5(h_local, dx)
                if local_c5 < current_c5:
                    h = h_local
                    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
