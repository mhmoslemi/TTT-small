import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    best_c5 = compute_c5(h)
    best_h = h.copy()
    current_h = h.copy()
    current_c5 = best_c5
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 2000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb the solution
        h_candidate = current_h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure i != j
        while i == j:
            j = np.random.randint(0, n_points)
            
        # Move mass from j to i
        delta = min(1 - h_candidate[i], h_candidate[j])
        if delta > 1e-10:  # Avoid negligible changes
            # Random amount of mass to move (0 to delta)
            move_amount = np.random.uniform(0, delta)
            h_candidate[i] += move_amount
            h_candidate[j] -= move_amount
            
            # Renormalize to satisfy constraint sum(h) = n_points/2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            candidate_c5 = compute_c5(h_candidate)
            
            # Accept or reject based on Metropolis criterion
            if candidate_c5 < current_c5 or np.random.rand() < np.exp(-(candidate_c5 - current_c5) / T):
                current_h = h_candidate
                current_c5 = candidate_c5
                
                # Update best solution if improved
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h.copy()
                    stagnation_count = 0  # Reset stagnation count
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        
        # Cooling schedule
        if T > min_T:
            T *= alpha
            
        # If no improvement for a while, increase temperature to escape local minima
        if stagnation_count > max_stagnation:
            T *= 1.2  # Increase temperature
            stagnation_count = 0  # Reset stagnation count
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
