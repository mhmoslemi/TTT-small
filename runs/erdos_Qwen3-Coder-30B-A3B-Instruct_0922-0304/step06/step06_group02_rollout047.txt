import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Simple binary step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Use a gradient-based optimization approach with cutting plane method
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Identify worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        worst_value = overlap[worst_lag_idx] * dx
        
        # Optimization using SLSQP with more focus on improving worst case
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Try to improve with a different starting point to avoid local minima
        h_trial = h + np.random.normal(0, 0.01, len(h))
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = normalize_h(h_trial, target_sum)
        
        result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 and candidate_c5 < 0.99999 * current_c5:
                h = h_candidate.copy()
                
        # Periodic global perturbation to escape local minima
        if iteration % 10 == 0:
            # Mutate
            mutation_strength = 0.03
            noise = np.random.normal(0, mutation_strength, n_points)
            h = np.clip(h + noise, 0, 1)
            # Renormalize
            h = normalize_h(h, target_sum)
        
        iteration += 1

    # Final check and refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
