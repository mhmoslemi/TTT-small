import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try SDP approach for better solution
    try:
        # Define SDP variables
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Define constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # For each shift k, compute the overlap
            if k >= 0:
                # h[i] * (1 - h[i-k]) for i = k to n_points-1
                # h[i] * (1 - h[i-k]) for i = 0 to k-1 -> 0 (h[i-k] out of bounds)
                # -> h[i] * (1 - h[i-k]) for i = k to n_points-1
                expr = 0
                for i in range(k, n_points):
                    expr += h_var[i] * (1 - h_var[i-k])
                # The same for indices beyond: h[i] * (1 - h[i-k])  for i = 0 to k-1
                # but error in the above because it was a direct dot product in the correlation
                # In full convolution mode, we compute all possible shifts correctly
                total_expr = 0
                for i in range(n_points):
                    j = (i - k) % n_points
                    total_expr += h_var[i] * (1 - h_var[j])
                constraints.append(total_expr <= c5_var)
            else:
                expr = 0
                for i in range(0, n_points + k):
                    expr += h_var[i] * (1 - h_var[i-k])
                #     rest when i >= n_points + k: nop because that's out of bounds
                #  we must compute correctly all auto-correlations
                # In fact why not just use the NumPy correlate function
                # and use the right formula for correlation with a shifted function
                
        # Formulate the SDP program
        objective = cp.Minimize(c5_var)
        prob = cp.Problem(objective, constraints)
        
        # Solve it
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sdp = h_var.value
                if h_sdp is not None:
                    h_sdp = np.clip(h_sdp, 0, 1)
                    current_sum_sdp = np.sum(h_sdp)
                    h_sdp = h_sdp * (target_sum / current_sum_sdp)
                    c5_sdp = compute_c5(h_sdp, dx)
                    if c5_sdp < best_c5:
                        best_h = h_sdp
                        best_c5 = c5_sdp
        except Exception as e:
            pass  # Fallback to optimization if SDP fails
            
    except Exception:
        pass  # Ignore SDP-related errors
        
    # Post-SDP: Local search around the best solution
    iterations = 0
    max_iter = 1000
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Define constraint function for optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        # Bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Local optimization with SLSQP
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            best_h = h_refined
            
        # Evaluate after refinement
        current_c5 = compute_c5(best_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
        else:
            # Random perturbation
            mutation_strength = max(0.001, 0.03 * np.exp(-iterations / 200.0))
            best_h += np.random.normal(0, mutation_strength, n_points)
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
        
        iterations += 1
        
    return (best_h.tolist(), best_c5, n_points)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

from scipy.optimize import minimize
