```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Initialize noise amplitude
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Identify regions of high overlap
        if iteration % 10 == 0:
            corr = correlate(h, 1 - h, mode="full")
            # Focus on top 20% of peaks
            peaks = np.sort(corr)[-int(len(corr) * 0.2):]
            peak_threshold = np.mean(peaks) if len(peaks) > 0 else 0
            
            # Create masking regions
            mask = np.zeros_like(h)
            for i in range(len(corr)):
                if corr[i] >= peak_threshold:
                    # Map to positions
                    k = i - (len(h) - 1)
                    if 0 <= k < len(h):
                        mask[k] = 1
                    if 0 <= len(h) - 1 - k < len(h):
                        mask[len(h) - 1 - k] = 1
        
        # Apply perturbation with temporal decay
        if iteration % 5 == 0:
            pert = np.random.normal(0, noise_amplitude, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Sample optimization with constraint handling
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP for local optimization
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        if result.success:
            h = result.x
        else:
            # If optimization failed, refine and try again
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Occasionally do more aggressive search
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            noise_amplitude *= 0.95
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
