import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h(x) in time domain."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0  # DC component
    for k in range(1, n_freq + 1):
        h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    # Compute Fourier coefficients via DFT (simplified using real FFT for real signal)
    # Using a simple rectangular approximation for first few harmonics
    # This approach directly uses cosine/sine integral approximations
    coeffs = [np.sum(h) / 2.0]  # DC component
    for k in range(1, n_freq + 1):
        ak = 2.0 * np.sum(h * np.cos(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        bk = 2.0 * np.sum(h * np.sin(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        coeffs.extend([ak, bk])
    return coeffs

def constraint_sum(coeffs, n_points, target_sum):
    """Constraint for sum of h to equal target_sum."""
    h = fourier_to_h(coeffs, n_points, n_freq)
    return np.sum(h) - target_sum

def objective_fourier(coeffs, n_points, n_freq, dx):
    """Objective function for optimization in Fourier domain."""
    h = fourier_to_h(coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    n_freq = 15  # Number of frequency components (excluding DC)
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Helper functions for optimization bounds and constraints
    def constraint_sum_func(coeffs):
        return constraint_sum(coeffs, n_points, target_sum)
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_coeffs = h_to_fourier(h, n_freq)
    else:
        # Initialize with a simple approximation
        initial_coeffs = [1.0] + [0.0]*n_freq*2

    # Bounds for coefficients: [-2, 2] is a reasonable range
    bounds = [(-2.0, 2.0)] * (n_freq * 2 + 1)
    
    # Define optimization constraints
    constraints = [{'type': 'eq', 'fun': constraint_sum_func}]
    
    # Optimization function in Fourier domain
    def objective(coeffs):
        return objective_fourier(coeffs, n_points, n_freq, dx)
    
    # DE parameters
    population_size = 60
    mutation_rate = 0.8
    crossover_rate = 0.9
    
    best_c5 = float('inf')
    best_coeffs = initial_coeffs.copy()
    
    # Differential Evolution optimization
    iters = 0
    max_iter = 300
    
    de = differential_evolution(
        objective, 
        bounds, 
        strategy='best1bin',
        mutation=(mutation_rate, 1.0),
        recombination=crossover_rate,
        maxiter=max_iter,
        popsize=population_size,
        tol=1e-6,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )
    
    # Convert best solution to h and refine with local optimization
    best_h = fourier_to_h(de.x, n_points, n_freq)
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    
    # Local refinement with SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    def objective_slsqp(h_vec):
        return compute_c5(h_vec, dx)
    
    constraint = {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - target_sum}
    
    result_slsqp = minimize(
        objective_slsqp,
        best_h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    if result_slsqp.success:
        final_h = result_slsqp.x
        final_h = np.clip(final_h, 0, 1)
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
    else:
        final_h = best_h
        final_c5 = compute_c5(final_h, dx)
        
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
