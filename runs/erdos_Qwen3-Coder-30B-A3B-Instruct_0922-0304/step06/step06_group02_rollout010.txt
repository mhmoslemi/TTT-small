```python
import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize initial h
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Prepare CVXPY variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h_var) == target_sum,
        h_var >= 0,
        h_var <= 1,
        C >= 0
    ]
    
    # Compute maximum lag
    max_lag = n_points - 1
    
    # Add overlap constraints for all lags
    for k in range(-max_lag, max_lag + 1):
        # For lag k, compute the sum: h[i] * (1 - h[i+k]) for valid indices
        # We need i in [0, n_points - 1 - |k|]
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i = 0..(n_points-1-k)
            valid_indices = range(n_points - k)
            expr = sum(h_var[i] * (1 - h_var[i+k]) for i in valid_indices)
        else:
            # h[i] * (1 - h[i+k]) for i = -k..(n_points-1)
            valid_indices = range(-k, n_points)
            expr = sum(h_var[i] * (1 - h_var[i+k]) for i in valid_indices)
        
        constraints.append(expr * dx <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Create problem and solve with repeated attempts
    prob = cp.Problem(objective, constraints)
    
    iteration = 0
    max_iter = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        try:
            # Solve
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_var.value
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (target_sum / np.sum(h_solution))
                current_c5 = compute_c5(h_solution, dx)
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_solution.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
                    
            # Random perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        except Exception:
            # Fallback to simple local search if optimization fails
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h = h * (target_sum / np.sum(h))
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
