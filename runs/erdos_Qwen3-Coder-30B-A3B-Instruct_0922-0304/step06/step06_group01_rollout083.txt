import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(freq_coeffs, n_points):
    """Convert Fourier coefficients to h(x)"""
    # freq_coeffs: [a0, a1, b1, a2, b2, ...]
    n_freq = (len(freq_coeffs) - 1) // 2
    
    # Reconstruct signal in frequency domain
    # We'll use the real FFT convention: [real[0], real[1], imag[1], real[2], imag[2], ...]
    # For complex FFT, we need the full complex spectrum
    # F = [freq_coeffs[0]/2] + sum_{k=1}^{n_freq} (freq_coeffs[2*k-1]*cos(k*pi*x) + freq_coeffs[2*k]*sin(k*pi*x))
    
    # We build a complex spectrum to compute the real signal
    # Real signal from complex FFT coefficients where we assume conjugate symmetry for real output
    # We start from half spectrum
    # freq_coeffs[0] is a0/2, so we double it for the full a0
    full_freq = np.zeros(n_points, dtype=complex)
    full_freq[0] = freq_coeffs[0] * 2  # We'll use FFT convention with reordering
    
    # Build spectrum for DFT
    # for k=1..n_freq, our coeffs are [a_k, b_k] for real terms
    # In FFT, for a cosine component with amplitude a_k and frequency kπ
    # it contributes (a_k/2) to frequency k and (a_k/2) to -k
    # for a sine component with amplitude b_k
    # it contributes (-i*b_k/2) to frequency k and (i*b_k/2) to -k
    # Let's handle this properly:
    
    # Construct spatial points (note: we'll build a full real signal)
    # We simplify by constructing the DFT result vector properly to avoid messy frequency indexing
    
    # Build forward spectrum from Symmetric real FFT coefficients
    # freq_coeffs[0] is a0/2
    # freq_coeffs[1], freq_coeffs[2] are a1, b1
    # etc.
    
    # The DFT general formula.
    # For a real signal with N points, we need spatial frequencies from k=0..N-1
    # We work within the real FFT constraint
    
    # Let's go with a clean approach:
    # We'll create real FFT-like spectrum directly:
    # s[0] = a0/2 (first coefficient)
    # s[1] = a1 - i*b1
    # s[2] = a2 - i*b2
    # ...
    # s[N//2] = a_{N//2} (real for even N) 
    # s[N//2+1] = a_{N//2+1} - i*b_{N//2+1}
    # ...
    # s[N-1] = a_{N//2+1} + i*b_{N//2+1} (conjugate)
    # (but we keep just first N/2+1 for real FFT)
    
    # We can only properly do it if generating the frequencies properly
    # let's construct the spectrum 
    # Create the frequency domain representation for FFT
    # For a real signal of length n_points, the frequency domain representation has n_points/2 + 1 unique points. 
    # So lengths must match:
    # freq_coeffs has 2*n_freq + 1 terms (a0, a1, b1, ..., an, bn)
    
    spectrum = np.zeros(n_points, dtype=complex)
    spectrum[0] = freq_coeffs[0]  # DC component

    # pad for an odd number of frequencies
    # we want to make a mapping from (a0, a1, b1, ...) to the DFT inputs
    
    # Direct mapping to DFT components 
    # Real signal requires:
    # C[k] ~ a_k * cos(k*pi*t) + b_k * sin(k*pi*t) 
    # So we want to compute this via the DFT convention:
    # Real parts: a_k, and imaginary parts: -b_k (sign issue depending on how cos/sin expansion chosen)
    # In terms of ITF:
    # For k=1..n-1, real part: a_k, imag part: -b_k in freq domain
    # But we must be careful with indexing: t=0,1,...n-1 
    # And since FFT uses freq 0,1,...n-1, not 0,1,2,... 
    # Nominal mapping:
    # [a0, a1,..., an, b1, ..., bn]  -> fft components
    
    # CRT: construct signal from real FFT coefficients
    # Handle even number of points
    half_n = n_points // 2
    # Create the first half with our coefficients
    # F[0] = a0
    # F[1] = a1 - i*b1, F[2] = a2 - i*b2, ..., F[n//2] = a[n//2] (real)
    # But we have   [a0, a1, b1, a2, b2, ..., an, bn]
    
    # So we have:
    # [a0],  [a1, b1], [a2, b2], ..., [an, bn]
    # There are 1 + n_freq pairs, total 1 + 2*n_freq elements
    # Total length should be 2*n_freq + 1
    
    if len(freq_coeffs) != 2 * n_freq + 1:
        # Clip or fill appropriately
        if len(freq_coeffs) > 2 * n_freq + 1:
            freq_coeffs = freq_coeffs[:2 * n_freq + 1]
        else:
            freq_coeffs = np.pad(freq_coeffs, (0, 2 * n_freq + 1 - len(freq_coeffs)), mode='constant')
    
    for k in range(1, n_freq + 1):
        if 2*k < n_points:
            # Fill k + 1, n_points - k along with conjugate
            # F[k] = a_k - i * b_k
            # F[n_points - k] = a_k + i * b_k (conjugate)
            spectrum[k] = freq_coeffs[2*k-1] - 1j * freq_coeffs[2*k]
            if 2*k < n_points//2:
                spectrum[n_points - k] = freq_coeffs[2*k-1] + 1j * freq_coeffs[2*k]
    
    spectrum[n_points // 2] = freq_coeffs[0]
    # Use FFT to synthesize signal
    h_signal = np.real(ifft(spectrum))  # This may have scaling issues, so...)
    h_signal /= n_points  # Scale back
    return h_signal

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients"""
    # Use DFT to get inverse, but map to real coefficients
    # Specifically, we want the real coefficients [a0, a1, b1, a2, b2, ..., an, bn]
    # from a signal.
    # For real signal, DFT[0] -> real, DFT[1] -> complex, ..., DFT[n/2] -> real
    # DFT[n/2+1] -> complex, ..., DFT[n-1] -> conjugate of DFT[1]
    
    # We get a full spectrum
    spec = fft(h)
    # [a0, a1, b1, a2, b2, ..., an, bn]   
    # We will compute this
    coeffs = [np.real(spec[0])]  # a0/2
    for i in range(1, min(n_freq + 1, len(spec))):
        if i < n_points // 2:
            a = np.real(spec[i])
            b = -np.imag(spec[i])  # fix sign
            coeffs.append(a)
            coeffs.append(b)
        else:
            break
    
    if len(coeffs) % 2 == 1:
        # 0 frequency only
        coeffs.append(0.0)
    return coeffs
    
def project_to_feasible(h, target_sum):
    """Project h to feasible set [0,1] with sum = target_sum"""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def model_performance(freq_coeffs, n_points, target_sum, dx):
    """Compute C5 for given Fourier coeffs"""
    try:
        h = fourier_to_h(freq_coeffs, n_points)
        h = project_to_feasible(h, target_sum)
        c5 = compute_c5(h, dx)
        return c5
    except Exception as e:
        return float('inf')

def run(seed=42, budget_s=600, **kwargs):
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of harmonics
    
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from initial_h_values if exists
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Start from square wave pattern
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1.0
    
    # Convert to Fourier representation
    freq_coeffs_init = h_to_fourier(h_init, n_freq)
    
    # Define bounds: [-1, 1] for each coeff
    bounds = [(-1.0, 1.0)] * (2 * n_freq + 1)
    
    def objective(freq_coeffs):
        return model_performance(freq_coeffs, n_points, target_sum, dx)
        
    # Differential evolution optimization
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=50,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None,
        workers=1
    )
    
    best_freq_coeffs = de_result.x
    best_h = fourier_to_h(best_freq_coeffs, n_points)
    best_h = project_to_feasible(best_h, target_sum)
    best_c5 = compute_c5(best_h, dx)
    
    # Final local optimization
    def opt_objective(h):
        return compute_c5(h, dx)
    
    bounds_h = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        opt_objective,
        best_h,
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    if result.success:
        final_h = project_to_feasible(result.x, target_sum)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h
            
    return (best_h.tolist(), best_c5, n_points)
