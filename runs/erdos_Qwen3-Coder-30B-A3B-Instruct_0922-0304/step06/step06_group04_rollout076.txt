import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try multiple random starts to avoid local optima
    num_starts = 20
    for start in range(num_starts):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Initialize with different random distributions
        h_init = np.random.rand(n_points)
        h_init = normalize_h(h_init, target_sum)
        
        # Clip to [0,1] to ensure bounds
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        
        # Local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                              constraints=constraint, options={'maxiter': 200})
            if result.success:
                h_opt = np.clip(result.x, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final verification
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
