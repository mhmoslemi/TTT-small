import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(freq_coeffs, n_points, dx):
    """Convert Fourier coefficients to h(x) with normalization."""
    # freq_coeffs: [a0, a1, b1, a2, b2, ...]
    a0 = freq_coeffs[0]
    a_coeffs = freq_coeffs[1::2]
    b_coeffs = freq_coeffs[2::2]
    
    x = np.linspace(0, 2, n_points, endpoint=False)
    h = a0 / 2.0 + np.zeros_like(x)
    
    for k, (ak, bk) in enumerate(zip(a_coeffs, b_coeffs), 1):
        h += ak * np.cos(k * np.pi * x) + bk * np.sin(k * np.pi * x)
    
    # Clip to [0,1] and normalize
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * (1.0 / current_sum)
    
    return h

def objective_fourier(freq_coeffs, n_points, dx):
    """Objective function for differential evolution."""
    h = fourier_to_h(freq_coeffs, n_points, dx)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    n_freq = 15  # Number of frequency components (a0, a1, b1, a2, b2, ...)
    
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (1.0 / np.sum(h0)) * n_points * dx  # Normalize for dx
    else:
        # Initialize with a step function
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (1.0 / np.sum(h0)) * n_points * dx

    # Convert initial h to Fourier coefficients
    # This using the standard DFT computation
    # However, for efficiency we go straight to mean
    a0 = np.mean(h0) * 2
    # Set some initial coefficients based on the shape
    freq_coeffs_init = [a0] + [0.0] * (2 * n_freq - 1)
    
    best_c5 = float('inf')
    best_freq_coeffs = freq_coeffs_init.copy()
    
    # Bounds for coefficients
    bounds = [(-2.0, 2.0)] + [(-1.0, 1.0)] * (2 * n_freq - 1)
    
    # Use differential evolution for global search
    de_result = differential_evolution(
        objective_fourier,
        bounds,
        args=(n_points, dx),
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        callback=lambda x, convergence: None if time.time() - start_time > budget_s - 2 else None,
        disp=False
    )
    
    if de_result.success:
        best_freq_coeffs = de_result.x
    else:
        # Fallback to initial if DE fails
        best_freq_coeffs = freq_coeffs_init.copy()
    
    # Final refinement with SLSQP
    def objective_slsqp(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        # Ensure normalization
        h = h * (1.0 / np.sum(h)) * n_points * dx
        return compute_c5(h, dx)
    
    h_refined = fourier_to_h(best_freq_coeffs, n_points, dx)
    # Normalize to correct sum
    h_refined = h_refined * (1.0 / np.sum(h_refined)) * n_points * dx
    
    # Final SLSQP optimization with linear constraint
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_flat):
        return np.sum(h_flat) * dx - 1.0
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run SLSQP
    result_slsqp = minimize(
        objective_slsqp,
        h_refined.flatten(),
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraint,
        options={'maxiter': 100}
    )
    
    if result_slsqp.success:
        h_final = result_slsqp.x.reshape(-1)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (1.0 / np.sum(h_final)) * n_points * dx
    else:
        h_final = h_refined
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
