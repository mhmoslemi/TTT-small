import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x)."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum([
        coeffs[2*i+1] * np.cos(i * np.pi * x) + 
        coeffs[2*i+2] * np.sin(i * np.pi * x) 
        for i in range(1, len(coeffs)//2)
    ], axis=0)
    return np.clip(h, 0, 1)

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    # Use discrete cosine transform for approximation
    # Extract cosine terms (including DC component)
    coeffs = np.zeros(2*n_freq + 1)
    coeffs[0] = np.mean(h)
    for i in range(1, n_freq+1):
        coeffs[2*i-1] = 2 * np.mean(h * np.cos(i * np.pi * np.linspace(0, 2, len(h))))
        coeffs[2*i] = 2 * np.mean(h * np.sin(i * np.pi * np.linspace(0, 2, len(h))))
    return coeffs

def evaluate_solution(coeffs, n_points, dx):
    """Evaluate solution from Fourier coefficients."""
    h = fourier_to_h(coeffs, n_points)
    # Ensure normalization
    h = h * (1.0 / np.sum(h)) * n_points * dx
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    n_freq = 15
    target_sum = n_points / 2

    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        # Convert to Fourier
        initial_coeffs = h_to_fourier(initial_h, n_freq)
    else:
        # Initialize with simple pulse in frequency domain
        initial_coeffs = np.zeros(2*n_freq + 1)
        initial_coeffs[0] = 0.5  # DC component
        initial_coeffs[1] = 0.2  # First cosine component
        
    def objective(coeffs):
        return evaluate_solution(coeffs, n_points, dx)

    # Bounds for differential evolution (fourier coefficients)
    bounds = [(-1, 1) for _ in range(2*n_freq + 1)]

    # Differential evolution optimization
    result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        callback=lambda x, convergence: None,
        disp=False
    )

    best_coeffs = result.x
    best_h = fourier_to_h(best_coeffs, n_points)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    # Local refinement with SLSQP
    def refine_objective(h):
        return compute_c5(h, dx)
    
    bounds_h = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    cons = {'type': 'eq', 'fun': constraint_sum}
    result_local = minimize(
        refine_objective,
        best_h,
        method='SLSQP',
        bounds=bounds_h,
        constraints=cons,
        options={'maxiter': 100}
    )
    
    if result_local.success:
        best_h = result_local.x
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (target_sum / np.sum(best_h))

    final_c5 = compute_c5(best_h, dx)
    
    # Final check and refinement to ensure feasibility and minimum
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (target_sum / np.sum(best_h))
    
    return (best_h.tolist(), final_c5, n_points)
