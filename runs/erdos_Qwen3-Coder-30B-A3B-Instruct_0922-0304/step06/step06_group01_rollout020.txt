import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function: 1 for first half, 0 for second half
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    popsize = 2 * n_points
    lambda_ = popsize // 2
    iteration = 0
    max_iter = 500
    
    # Initialize CMA-ES parameters
    mu = n_points // 2
    sigma = 0.1
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Covariance matrix
    C = np.eye(n_points)
    
    # Mean and evolution paths
    mean = h.copy()
    pc = np.zeros(n_points)
    ps = np.zeros(n_points)
    
    # Adaptation parameters
    c1 = 2.0 / (n_points + 2.0)**2
    cmu = min(1 - c1, 2 * (mu - 2) / (n_points + 2)**2)
    damps = 1 + 2 * max(0, np.sqrt((mu - 1) / (n_points + 1)) - 1) + c1
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Sample new population
        population = []
        for _ in range(popsize):
            # Sample from multivariate normal
            z = np.random.randn(n_points)
            sample = mean + sigma * C @ z
            # Clip and normalize
            sample = np.clip(sample, 0, 1)
            sample = normalize_h(sample, target_sum)
            population.append(sample)
        
        # Evaluate population
        fitness = []
        for ind in population:
            c5_val = compute_c5(ind, dx)
            fitness.append(c5_val)
        
        # Rank individuals
        indices = np.argsort(fitness)
        best_indices = indices[:lambda_]
        
        # Update evolution paths
        z_w = np.sum([weights[i] * (population[indices[i]] - mean) for i in range(lambda_)], axis=0)
        ps = (1 - c1) * ps + np.sqrt(c1 * (2 - c1)) * z_w / sigma
        pc = (1 - cmu) * pc + cmu * z_w
        
        # Update covariance matrix
        C = (1 - c1 - cmu) * C + c1 * (pc @ pc.T) + cmu * np.sum([weights[i] * np.outer(population[indices[i]] - mean, population[indices[i]] - mean) for i in range(lambda_)], axis=0)
        
        # Update mean
        new_mean = mean + sigma * z_w
        mean = new_mean
        
        # Update step size according to CMA-ES rule
        sigma *= np.exp((c1 / damps) * (np.linalg.norm(ps) / np.sqrt(n_points) - 1))
        
        # Keep the best individual
        current_best = population[indices[0]]
        current_c5 = fitness[indices[0]]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        
    # Final refinement
    def objective(h):
        return compute_c5(h, dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Mirror best solution for SLSQP
    test_h = best_h.copy()
    result = minimize(objective, test_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h
    
    return (best_h.tolist(), best_c5, n_points)
