import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_to_feasible(h, target_sum)
    else:
        # Start with a reasonable construction
        h = np.ones(n_points) * 0.5
        # Add structured randomness
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h += np.random.normal(0, 0.02, n_points)
        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier method parameters
    mu = 1.0
    decay_factor = 0.5
    linesearch_max_iter = 50
    min_mu = 1e-8
    
    # Precompute the indices of the full correlation
    full_length = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and mu > min_mu:
        # Compute current gradient with finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        
        # Compute correlation matrix
        cor1 = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(cor1)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            cor_plus = correlate(h_plus, 1 - h_plus, mode='full')
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_to_feasible(h_minus, target_sum)
            cor_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            # Approximate gradient of correlation at max point
            grad[i] = (cor_plus[max_idx] - cor_minus[max_idx]) / (2 * epsilon)
        
        # Compute barrier gradient
        barrier_grad = np.zeros(n_points)
        for i in range(n_points):
            barrier_grad[i] = -1.0/h[i] - 1.0/(1.0 - h[i]) 
        barrier_grad += 1.0 / (target_sum - np.sum(h))
        
        # Combined gradient
        combined_grad = grad + mu * barrier_grad
        
        # Line search
        alpha = 1.0
        direction = -combined_grad
        
        for _ in range(linesearch_max_iter):
            h_new = h + alpha * direction
            h_new = project_to_feasible(h_new, target_sum)
            
            c5_new = compute_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                break
            alpha *= 0.5
        
        # Update barrier parameter
        if iteration % 50 == 0:
            mu *= decay_factor
            if mu < min_mu:
                mu = min_mu
                
        iteration += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
