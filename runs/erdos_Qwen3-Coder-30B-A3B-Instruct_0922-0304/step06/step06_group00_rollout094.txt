import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitness[i] for i in idxs])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
        
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, mut_strength=0.1):
    """Mutate an individual."""
    for i in range(len(individual)):
        if np.random.rand() < mutation_rate:
            individual[i] += np.random.normal(0, mut_strength)
    return individual

def gradient_refinement(h, target_sum, dx):
    """Refine h using gradient-based method."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum

    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})
    
    if result.success:
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = normalize_h(h_refined, target_sum)
        return h_refined
    else:
        return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with some structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    pop_size = 100
    num_generations = 60  # Each generation ~10 seconds within budget
    
    # Initialize population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        h_mut = h + np.random.normal(0, 0.05, n_points)
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = normalize_h(h_mut, target_sum)
        population.append(h_mut)
    
    # Track best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # GA loop
    for gen in range(num_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = [population[0].copy()]  # Keep best (elitism)
        
        # Tournament selection, crossover, and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2)
            
            mut_rate = 0.2 * np.exp(-gen / 20.0)
            child1 = mutate(child1, mut_rate)
            child2 = mutate(child2, mut_rate)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            # Local refinement
            child1 = gradient_refinement(child1, target_sum, dx)
            child2 = gradient_refinement(child2, target_sum, dx)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
    
    # Final refinement of best solution
    final_h = gradient_refinement(best_h, target_sum, dx)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
