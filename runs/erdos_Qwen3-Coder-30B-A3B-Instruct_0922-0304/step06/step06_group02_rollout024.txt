import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        worst_c5 = overlap[worst_lag_idx] * dx
        
        # Use projected gradient to fix worst lag
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return np.max(correlate(h, 1 - h, mode="full")) * dx
            
        def constraint_worst_lag(h):
            # This is a soft constraint that encourages the worst lag to improve
            # We compute an approximation to the worst lag quantization and impose a limit
            # This approach penalizes violations of the current worst lag's overlap
            overlap_vals = correlate(h, 1 - h, mode="full")
            return worst_c5 - np.max(overlap_vals) * dx
            
        constraint = {'type': 'ineq', 'fun': constraint_worst_lag}
        
        # SLSQP optimization with small perturbation
        noise_factor = 0.01
        h_noisy = h + np.random.normal(0, noise_factor, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        result = minimize(
            objective, 
            h_noisy, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=constraint, 
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Ensure significant improvement
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5:
                h = h_candidate.copy()
                
        # Periodic global perturbation to escape local minima
        if iteration % 20 == 0:
            noise_magnitude = 0.02
            h = np.clip(h + np.random.normal(0, noise_magnitude, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
