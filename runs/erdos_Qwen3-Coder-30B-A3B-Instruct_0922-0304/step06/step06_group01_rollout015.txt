import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def build_comb(h, n_teeth, tooth_width, spacing, slope, n_points, dx):
    """Construct h based on comb parameters."""
    h.fill(0.0)
    tooth_height = 1.0
    
    # Calculate tooth positions
    for i in range(n_teeth):
        start = i * (tooth_width + spacing)
        end = start + tooth_width
        # Ensure we don't go beyond bounds
        if start >= 2.0:
            break
        # Set tooth values
        x = np.arange(n_points) * dx
        # Apply triangle shape
        for j in range(n_points):
            x_pos = j * dx
            if start <= x_pos < end:
                h[j] += tooth_height
            elif start - slope <= x_pos < start:
                h[j] += tooth_height * (1 - (x_pos - (start - slope)) / slope)
            elif end <= x_pos < end + slope:
                h[j] += tooth_height * (1 - (x_pos - end) / slope)
    
    # Normalize sum
    h = normalize_h(h, n_points / 2)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Check for initial_h_values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default comb construction
        h = np.zeros(n_points)
        n_teeth = 7
        tooth_width = 2.0 / n_teeth
        spacing = 0.028
        slope = tooth_width / 2.0
        h = build_comb(h, n_teeth, tooth_width, spacing, slope, n_points, dx)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different comb configurations
    param_search_space = [
        (5, 0.2857, 0.028, 0.1428),
        (6, 0.3333, 0.0333, 0.1666),
        (7, 0.2857, 0.028, 0.1428),
        (8, 0.25, 0.03125, 0.125),
        (9, 0.2222, 0.02469, 0.1111),
        (10, 0.2, 0.02, 0.1)
    ]
    
    for n_teeth, tooth_width, spacing, slope in param_search_space:
        # Construct specific comb
        h_temp = np.zeros(n_points)
        build_comb(h_temp, n_teeth, tooth_width, spacing, slope, n_points, dx)
        
        # Optimization step
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h_temp.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                c5 = compute_c5(h_new, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
        except:
            pass
            
        # Early exit if threshold met
        if best_c5 <= 0.3807 and time.time() - start_time < budget_s * 0.9:
            break
    
    # Final global refinement
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                c5_refined = compute_c5(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
        except:
            pass
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
