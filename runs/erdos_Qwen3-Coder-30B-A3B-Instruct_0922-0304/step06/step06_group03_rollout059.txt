import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points)
            particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle, dx))
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    best_c5 = g_best_value
    best_h = g_best.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particle) + 
                            c2 * r2 * (g_best - particle))
            
            # Update position
            particle = particle + velocities[i]
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Evaluate fitness
            fitness = compute_c5(particle, dx)
            
            # Update personal best
            if fitness < p_best_values[i]:
                p_best[i] = particle.copy()
                p_best_values[i] = fitness
            
            # Update global best
            if fitness < g_best_value:
                g_best = particle.copy()
                g_best_value = fitness
                
                if g_best_value <= 0.3807:
                    break
                    
            # Store updated particle
            particles[i] = particle
        
        # Periodic FFT-based acceleration (every 50 iterations)
        if iter_count % 50 == 0 and iter_count > 0:
            # Identify worst 25% of particles
            worst_indices = np.argsort(p_best_values)[-n_particles//4:]
            
            # Apply frequency domain filtering on worst particles
            for idx in worst_indices:
                freq_domain = np.fft.fft(p_best[idx])
                # High-pass filter
                freq_domain[0] = 0  # Remove DC component
                # Apply some smoothing in frequency domain
                freq_domain = freq_domain * np.exp(-0.1 * np.arange(len(freq_domain))**2)
                filtered = np.fft.ifft(freq_domain).real
                filtered = np.clip(filtered, 0, 1)
                filtered = normalize_h(filtered, target_sum)
                particles[idx] = filtered
                p_best[idx] = filtered.copy()
                p_best_values[idx] = compute_c5(filtered, dx)
                
                if p_best_values[idx] < g_best_value:
                    g_best = filtered.copy()
                    g_best_value = p_best_values[idx]
        
        # Gradient refinement (every 100 iterations)
        if iter_count % 100 == 0 and iter_count > 0:
            top_indices = np.argsort(p_best_values)[:10]
            for i in top_indices:
                h_current = p_best[i].copy()
                
                # Local SLSQP optimization
                def objective(h):
                    return compute_c5(h, dx)
                    
                bounds = [(0, 1) for _ in range(n_points)]
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h_current, method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 100})
                if result.success:
                    refined = normalize_h(result.x, target_sum)
                    new_fitness = compute_c5(refined, dx)
                    if new_fitness < p_best_values[i]:
                        p_best[i] = refined.copy()
                        p_best_values[i] = new_fitness
                        
                        # Update global best if needed
                        if new_fitness < g_best_value:
                            g_best = refined.copy()
                            g_best_value = new_fitness
        
        if g_best_value <= 0.3807:
            break
            
        iter_count += 1
    
    # Final marking of best solution
    final_c5 = g_best_value
    return (g_best.tolist(), final_c5, n_points)
