import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to function h(x)"""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs))[:, None] * np.pi * x),
        axis=0
    )
    return np.clip(h, 0, 1)

def compute_c5(h, dx):
    """Compute the C5 value for given h"""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def objective_function(coeffs, n_points, dx, target_sum):
    """Objective function to minimize"""
    h = fourier_to_h(coeffs, n_points)
    h = normalize_h(h, target_sum)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initialize coefficients
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Convert to Fourier coefficients
        coeffs_init = np.fft.fft(h_init)[:n_freq]
        coeffs_init = np.real(coeffs_init)
        coeffs = coeffs_init.copy()
    else:
        # Random initial coefficients
        coeffs = np.random.rand(n_freq) * 0.1
    
    # Define bounds for each coefficient (a0, a1, ..., an, b1, ..., bn)
    bounds = [(0, 1) for _ in range(n_freq)]
    
    def de_objective(coeffs):
        return objective_function(coeffs, n_points, dx, target_sum)
    
    # Differential Evolution optimization
    result = differential_evolution(
        de_objective,
        bounds,
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False,
        atol=1e-6,
        rtol=1e-6
    )
    
    # Refine with local optimization
    if result.success:
        coeffs = result.x
        h = fourier_to_h(coeffs, n_points)
        h = normalize_h(h, target_sum)
        
        def local_objective(h_vec):
            h = np.clip(h_vec, 0, 1)
            h = normalize_h(h, target_sum)
            return compute_c5(h, dx)
            
        bounds_h = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        local_result = minimize(
            local_objective,
            h,
            method='SLSQP',
            bounds=bounds_h,
            constraints=constraint,
            options={'maxiter': 100}
        )
        
        if local_result.success:
            h = np.clip(local_result.x, 0, 1)
            h = normalize_h(h, target_sum)
    
    final_c5 = compute_c5(h, dx)
    return (h.tolist(), final_c5, n_points)
