import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    for _ in range(num_particles):
        perturbation = np.random.normal(0, 0.05, n_points)
        h = np.clip(base_h + perturbation, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    
    # Personal and global bests
    pbest = [p.copy() for p in particles]
    pbest_c5 = [compute_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    best_h = gbest.copy()
    best_c5 = gbest_c5
    
    # PSO iterations
    max_iter = 3000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # PSO iteration
        for i in range(num_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (pbest[i] - h) + c2 * r2 * (gbest - h)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            particles[i] = h
            velocities[i] = v
            
            # Evaluate fitness
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = h.copy()
                    gbest_c5 = c5
                    
                    if gbest_c5 <= 0.3807:
                        break
                        
        # Frequency domain acceleration every 50 iterations
        if iter_count % 50 == 0 and iter_count > 0:
            for i in range(num_particles):
                h = particles[i]
                # FFT
                H = np.fft.fft(h)
                # High-pass filter
                freqs = np.fft.fftfreq(n_points, dx)
                mask = np.abs(freqs) > 0.3
                H_filtered = H * mask
                h_filtered = np.real(np.fft.ifft(H_filtered))
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = normalize_h(h_filtered, target_sum)
                particles[i] = h_filtered
        
        # Gradient refinement every 100 iterations
        if iter_count % 100 == 0 and iter_count > 0:
            # Find top 10 particles
            top_indices = np.argsort(pbest_c5)[:10]
            for i in top_indices:
                h = pbest[i]
                
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h_refined = result.x
                    c5_refined = compute_c5(h_refined, dx)
                    if c5_refined < pbest_c5[i]:
                        pbest[i] = h_refined
                        pbest_c5[i] = c5_refined
                        
                        if c5_refined < gbest_c5:
                            gbest = h_refined
                            gbest_c5 = c5_refined
        
        iter_count += 1
        
        # Update global best if needed
        current_gbest_idx = np.argmin(pbest_c5)
        if pbest_c5[current_gbest_idx] < gbest_c5:
            gbest = pbest[current_gbest_idx].copy()
            gbest_c5 = pbest_c5[current_gbest_idx]
        
        if gbest_c5 <= 0.3807:
            break
            
    # Final evaluation
    final_c5 = compute_c5(gbest, dx)
    return (gbest.tolist(), final_c5, n_points)
