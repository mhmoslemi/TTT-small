import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing and tooth height."""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    tooth_positions = np.arange(0, 2, spacing * dx)
    for pos in tooth_positions:
        idx = int(pos / dx)
        if 0 <= idx < n_points:
            h[idx] = tooth_height
    return h

def optimize_pattern(h, target_sum, dx):
    """Optimize pattern using SLSQP."""
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            return np.clip(result.x, 0, 1)
    except:
        pass
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    n_combs_tried = 0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create comb patterns
        spacing_options = [2, 3, 4, 5, 6]
        candidates = []
        for spacing in spacing_options:
            htemp = create_comb_pattern(n_points, spacing)
            htemp = normalize_h(htemp, target_sum)
            candidates.append(htemp)
    
    # Optimization loop
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if 'candidates' in locals() and iteration < len(candidates):
            h = candidates[iteration]
            iteration += 1
        else:
            # Generate new comb pattern
            spacing = np.random.choice([2, 3, 4, 5, 6])
            h = create_comb_pattern(n_points, spacing)
            h = normalize_h(h, target_sum)
        
        # Perturb the pattern slightly
        if iteration > 0:
            perturbation = np.random.normal(0, 0.02, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        
        # Try optimized version
        h_opt = optimize_pattern(h, target_sum, dx)
        c5 = compute_c5(h_opt, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Occasionally do more aggressive search
        if iteration % 100 == 0:
            # Try binary pattern
            h = np.zeros(n_points)
            if iteration % 200 == 0:
                h[:n_points//2] = 1
            else:
                h[n_points//2:] = 1
            h = normalize_h(h, target_sum)
            h_opt = optimize_pattern(h, target_sum, dx)
            c5 = compute_c5(h_opt, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
    
    # Final refinement with high resolution if needed
    if n_points == 70 and best_h is not None:
        h = best_h
        dx = 2.0 / 140
        target_sum = 70
        # Refine the best pattern
        h = np.interp(np.linspace(0, 2, 140), np.linspace(0, 2, 70), h)
        h = normalize_h(h, target_sum)
        h = np.clip(h, 0, 1)
        h = optimize_pattern(h, target_sum, dx)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    if best_h is None:
        # Fallback
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        best_c5 = compute_c5(h, dx)
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
