import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    # Initialize parameters
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    learning_rate = 0.01
    decay = 0.995
    min_lr = 0.001
    
    # Precompute indices for efficiency
    lag_indices = np.arange(-n_points + 1, n_points)
    
    iterations = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps

            learning_rate = min_lr

    final_c5 = compute_c5(best_h, dx)
