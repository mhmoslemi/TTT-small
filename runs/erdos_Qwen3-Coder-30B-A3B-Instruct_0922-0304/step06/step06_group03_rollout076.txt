import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple initial guesses"""
    guesses = []
    
    # 1. Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # 2. Comb pattern
    h = np.zeros(n_points)
    step = max(2, n_points // 10)
    for i in range(0, n_points, step):
        h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 3. Random pattern with sum constraint
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 4. Alternating high/low pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 5. Sine wave pattern
    x = np.linspace(0, 2, n_points)
    h = np.sin(np.pi * x) + 1.5
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    max_iter = 15000
    
    # For tracking and perturbations
    last_improvement = 0
    perturbation_std = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try each initial guess
        for h in initial_guesses:
            # Apply optimization if it's the first iteration of this guess
            if iteration % 1000 == 0:
                # SLSQP optimization with constraints
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h, method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 50})
                
                if result.success:
                    h = result.x
            
            # Perturbation
            if iteration % 500 == 0:
                h = np.clip(h + np.random.normal(0, perturbation_std, n_points), 0, 1)
                h = normalize_h(h, target_sum)
            
            # Apply random perturbation
            if iteration % 20 == 0:
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
                
        # Adjust perturbation if not improving
        if iteration - last_improvement > 1000:
            perturbation_std *= 0.95
            last_improvement = iteration
            
        # Early termination if good enough
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final recomputation of best solution
    final_c5 = compute_c5(best_h, dx) if best_h is not None else float('inf')
    return (best_h.tolist(), final_c5, n_points)
