import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Multi-start optimization with diverse initializations
    n_starts = 20
    for start_idx in range(n_starts):
        # Use different initialization strategies
        if start_idx == 0:
            # Original initialization
            h = base_h.copy()
        elif start_idx == 1:
            # Random noise around uniform
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h, target_sum)
        elif start_idx == 2:
            # Gaussian perturbation
            h = np.random.normal(0.5, 0.2, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            # More randomized strategies
            h = np.random.exponential(1, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        
        # Local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Adaptive maxiter
        maxiter = 200
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': maxiter})
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h_opt.copy()
        except:
            pass
        
        if time.time() - start_time > budget_s - 1:
            break
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
