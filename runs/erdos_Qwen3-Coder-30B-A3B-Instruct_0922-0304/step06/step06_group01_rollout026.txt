import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple starting construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        # Random frequency coefficients
        coeffs = np.random.uniform(-1, 1, 2 * n_freq)
        particles.append(coeffs)
        velocities.append(np.random.uniform(-0.1, 0.1, 2 * n_freq))
        pbest.append(coeffs.copy())
        pbest_vals.append(float('inf'))
    
    # Global best
    gbest = particles[0].copy()
    gbest_val = float('inf')
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Recompute h from coeffs
            h_temp = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            for k in range(n_freq):
                h_temp += particles[i][k] * np.cos(k * np.pi * x) + \
                         particles[i][k + n_freq] * np.sin(k * np.pi * x)
            h_temp = np.clip(h_temp, 0, 1)
            
            # Normalize
            current_sum = np.sum(h_temp)
            if current_sum > 0:
                h_temp = h_temp * (target_sum / current_sum)
            
            # Compute C5
            c5 = compute_c5(h_temp, dx)
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                if c5 < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = c5
                    
            # Update global best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
                
            # Early stopping
            if best_c5 <= 0.3807:
                break
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            particles[i] += velocities[i]
            
            # Clip particles to avoid divergence
            particles[i] = np.clip(particles[i], -2, 2)
            
        if best_c5 <= 0.3807:
            break
    
    # Final local refinement
    def objective(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Use best solution from PSO as starting point
    h_refined = best_h.copy()
    
    result = minimize(objective, h_refined, method='SLSQP',
                      bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        current_sum = np.sum(h_final)
        if current_sum > 0:
            h_final = h_final * (target_sum / current_sum)
        final_c5 = compute_c5(h_final, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_c5, n_points)
