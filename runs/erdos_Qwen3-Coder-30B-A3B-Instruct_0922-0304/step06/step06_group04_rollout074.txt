import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = base_h * (target_sum / np.sum(base_h))
    else:
        base_h = np.ones(n_points) * 0.5
    
    # Perturb to get a good starting point
    noise = np.random.normal(0, 0.05, n_points)
    base_h = base_h + noise
    base_h = np.clip(base_h, 0, 1)
    base_h = base_h * (target_sum / np.sum(base_h))
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    iteration = 0
    # Alternating optimization
    while time.time() - start_time < budget_s - 1:
        # Optimize h with fixed g
        h = best_individual.copy()
        noise = np.random.normal(0, 0.03, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Optimization with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-6})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h.copy()
        except:
            pass
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
