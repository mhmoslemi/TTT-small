import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with local search
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Analytical gradient computation with respect to max_k
        g = np.zeros(n_points)
        k = max_k - (n_points - 1)  # convert full convolution index to lag
        
        # Calculate gradient: g_i = d/dh_i [h_i * (1 - h_{i+k})] if i+k < n
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                g[i] = (1 - h[i + k]) * dx
            elif i - k >= 0 and i - k < n_points:
                g[i] = -h[i - k] * dx
            else:
                g[i] = 0

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Reduce step size periodically
        if iteration % 1000 == 0:
            step_size *= 0.5
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
