import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    swarm = []
    velocities = []
    p_best = []
    p_best_values = []
    g_best = h.copy()
    g_best_value = compute_c5(h, dx)
    
    # Initialize swarm
    for _ in range(n_particles):
        particle = h + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        swarm.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle, dx))
        
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update particle positions and find new bests
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - swarm[i]) + 
                           c2 * r2 * (g_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            swarm[i] = np.clip(swarm[i], 0, 1)
            swarm[i] = normalize_h(swarm[i], target_sum)
            
            # Evaluate
            current_value = compute_c5(swarm[i], dx)
            
            # Update personal best
            if current_value < p_best_values[i]:
                p_best[i] = swarm[i].copy()
                p_best_values[i] = current_value
                
            # Update global best
            if current_value < g_best_value:
                g_best = swarm[i].copy()
                g_best_value = current_value
                
        # Periodic frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            # Filter worst 25% particles
            particle_values = [compute_c5(p, dx) for p in swarm]
            worst_indices = np.argsort(particle_values)[-n_particles//4:]
            
            for idx in worst_indices:
                # FFT-based filtering
                fft_h = np.fft.fft(swarm[idx])
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter
                cutoff = 0.3
                mask = np.abs(freqs) > cutoff
                filtered_fft = fft_h * mask
                filtered_h = np.real(np.fft.ifft(filtered_fft))
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                swarm[idx] = filtered_h
                
            # Reevaluate
            for idx in worst_indices:
                p_best_values[idx] = compute_c5(swarm[idx], dx)
                if p_best_values[idx] < g_best_value:
                    g_best = swarm[idx].copy()
                    g_best_value = p_best_values[idx]
                    
        # Occasionally refine top particles with SLSQP
        if iteration % 100 == 0 and iteration > 0:
            particle_values = [compute_c5(p, dx) for p in swarm]
            top_indices = np.argsort(particle_values)[:10]
            
            for idx in top_indices:
                h0 = swarm[idx].copy()
                bounds = [(0, 1) for _ in range(n_points)]
                def objective(h):
                    return compute_c5(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h0, method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 50})
                if result.success:
                    new_h = result.x
                    new_h = np.clip(new_h, 0, 1)
                    new_h = normalize_h(new_h, target_sum)
                    swarm[idx] = new_h
                    new_value = compute_c5(new_h, dx)
                    if new_value < p_best_values[idx]:
                        p_best[idx] = new_h.copy()
                        p_best_values[idx] = new_value
                        if new_value < g_best_value:
                            g_best = new_h.copy()
                            g_best_value = new_value
                            
        iteration += 1
        if g_best_value <= 0.3807:
            break
            
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
