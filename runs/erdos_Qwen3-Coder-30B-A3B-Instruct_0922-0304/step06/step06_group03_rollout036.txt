import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    
    # Set tooth heights
    h[tooth_positions] = tooth_height
    
    # Normalize to correct sum
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbation."""
    perturbation = np.random.normal(0, sigma, len(h))
    h_perturbed = h + perturbation
    return np.clip(h_perturbed, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try comb patterns with different spacings
    spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    
    for s in spacings:
        comb_h = create_comb_pattern(n_points, s)
        comb_patterns.append(comb_h)
        
    # Evaluate initial patterns
    for comb_h in comb_patterns:
        c5 = compute_c5(comb_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = comb_h.copy()
    
    # Optimization loop
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Occasionally do more aggressive search
        if iteration % 20 == 0:
            # Try to improve comb pattern from best_h
            # Search different spacings around best_h
            best_spaced_h = None
            best_spaced_c5 = float('inf')
            
            for spacing in [2, 3, 4, 5, 6]:
                if spacing == 1:
                    continue
                    
                comb_pattern = create_comb_pattern(n_points, spacing)
                # Try random variations
                for _ in range(5):
                    h_var = perturb_pattern(comb_pattern, 0.01)
                    h_var = normalize_h(h_var, target_sum)
                    c5 = compute_c5(h_var, dx)
                    if c5 < best_spaced_c5:
                        best_spaced_c5 = c5
                        best_spaced_h = h_var.copy()
                
                if best_spaced_c5 < best_c5:
                    best_c5 = best_spaced_c5
                    best_h = best_spaced_h.copy()
                    if best_c5 <= 0.3807:
                        break
        
        # Local optimization with SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Small random perturbation to escape local minima
        h_perturbed = perturb_pattern(best_h, 0.01)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        result = minimize(objective, h_perturbed, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_opt = result.x
        else:
            h_opt = h_perturbed
            
        # Possibly accept improvement
        c5_opt = compute_c5(h_opt, dx)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            if best_c5 <= 0.3807:
                break
        
        iteration += 1
    
    # Final refinement with higher resolution (if time permits)
    if time.time() - start_time < budget_s - 10:
        n_points_high = 140
        dx_high = 2.0 / n_points_high
        target_sum_high = n_points_high / 2
        
        # Initialize from best_h at higher resolution
        def interpolate_h(h_low, n):
            h_high = np.interp(np.linspace(0, 2, n), np.linspace(0, 2, len(h_low)), h_low)
            return h_high
        
        h_high = interpolate_h(best_h, n_points_high)
        h_high = normalize_h(h_high, target_sum_high)
        
        # Optimize at higher resolution
        def objective_high(h):
            return compute_c5(h, dx_high)
            
        def constraint_sum_high(h):
            return np.sum(h) - target_sum_high
            
        bounds_high = [(0, 1) for _ in range(n_points_high)]
        constraint_high = {'type': 'eq', 'fun': constraint_sum_high}
        
        result = minimize(objective_high, h_high, method='SLSQP', bounds=bounds_high, constraints=constraint_high, options={'maxiter': 100})
        if result.success:
            h_final = result.x
        else:
            h_final = h_high
            
        c5_final = compute_c5(h_final, dx_high)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
