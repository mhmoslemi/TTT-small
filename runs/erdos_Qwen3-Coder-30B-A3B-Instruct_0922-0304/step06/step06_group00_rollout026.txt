import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual with Gaussian noise."""
    h += np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    crossover_point = np.random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        # Create base solutions
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Add diverse individuals to the population
    for _ in range(population_size - 1):
        h = np.random.rand(n_points) * 0.5
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    
    # Track generation for mutation strength decay
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutation
            mutation_strength = max(0.001, 0.2 * np.exp(-generation / 50))
            child1 = mutate_individual(child1, mutation_strength)
            child2 = mutate_individual(child2, mutation_strength)
            
            # Ensure constraints
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim population to exact size
        population = new_population[:population_size]
        generation += 1
    
    # Final validation
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback: return the best of the initial population
        fitnesses = [compute_c5(h, dx) for h in population]
        best_idx = np.argmin(fitnesses)
        return (population[best_idx].tolist(), fitnesses[best_idx], n_points)
