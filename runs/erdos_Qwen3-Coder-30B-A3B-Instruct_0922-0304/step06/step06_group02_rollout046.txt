import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from known good construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with focus on worst overlaps
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lag
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Focus optimization on the most problematic lag
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_lag = max_lag_idx - (n_points - 1)  # Convert to signed lag
        
        # Try to decrease the overlap at this lag by optimization
        def objective(h):
            c5 = compute_c5(h, dx)
            return c5
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def constraint_worst_lag(h):
            # Create correlation for this specific lag
            h_extended = np.concatenate([h, [0] * (n_points - 1)])
            h_flip = np.concatenate([[0] * (n_points - 1), 1 - h])
            
            # Specific cross-correlation
            conv = np.convolve(h_extended, h_flip, mode="valid")
            return conv[n_points - 1 + max_lag] * dx - best_c5 * 0.999
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': constraint_sum},
        ]
        
        # Use bounds constrained optimization
        h_noisy = h + np.random.normal(0, 0.01, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = h_noisy * (target_sum / np.sum(h_noisy))
        
        try:
            result = minimize(objective, h_noisy, method='SLSQP', 
                             bounds=bounds, constraints=constraints,
                             options={'maxiter': 50})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5 and candidate_c5 < best_c5 * 0.999:
                    h = h_candidate.copy()
        except:
            pass
            
        # Add stochastic perturbations
        if iteration % 10 == 0:
            perturb = np.random.normal(0, 0.02, len(h))
            h = np.clip(h + perturb, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
