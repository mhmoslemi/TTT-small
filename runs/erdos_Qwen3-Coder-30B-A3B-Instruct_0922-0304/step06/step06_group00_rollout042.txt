import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection."""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitness[i] for i in idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, sigma=0.05):
    """Gaussian mutation."""
    mask = np.random.rand(len(individual)) < mutation_rate
    noise = np.random.normal(0, sigma, len(individual))
    individual = individual + mask * noise
    return np.clip(individual, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population parameters
    population_size = 100
    max_generations = 60
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Fill population with random individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # GA loop
    best_c5 = float('inf')
    best_h = None
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = np.array([compute_c5(h, dx) for h in population])
        
        # Track best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = [population[current_best_idx].copy()]
        
        # Create offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, 0.2 * np.exp(-generation/20), 0.05)
            child2 = mutate(child2, 0.2 * np.exp(-generation/20), 0.05)
            
            # Normalize
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        # Local optimization for top individuals
        top_indices = np.argsort(fitness)[:10]
        for idx in top_indices:
            h = population[idx]
            # Simple gradient descent step
            # This is a naive local refinement
            step_size = 0.01
            # Approximate gradient using finite differences
            h_plus = h.copy()
            h_plus[0] += step_size
            h_plus = normalize_h(h_plus, target_sum)
            c_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[0] -= step_size
            h_minus = normalize_h(h_minus, target_sum)
            c_minus = compute_c5(h_minus, dx)
            
            grad = (c_plus - c_minus) / (2 * step_size)
            h[0] -= grad * step_size
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            population[idx] = h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
