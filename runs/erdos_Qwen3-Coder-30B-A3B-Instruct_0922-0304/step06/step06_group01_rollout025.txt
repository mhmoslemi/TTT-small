import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles with Fourier coefficients
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        coeffs = np.random.normal(0, 0.1, 2 * n_freq)
        particles.append(coeffs)
        velocities.append(np.random.normal(0, 0.01, 2 * n_freq))
    
    # Global best
    gbest_coeffs = particles[0].copy()
    gbest_c5 = float('inf')
    
    # PSO optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, coeffs in enumerate(particles):
            # Convert coefficients to h
            h_candidate = np.zeros(n_points)
            x = np.linspace(0, 2, n_points)
            
            for k in range(n_freq):
                if k == 0:
                    h_candidate += coeffs[0]
                else:
                    h_candidate += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
            
            # Clip and normalize
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Compute C5
            c5 = compute_c5(h_candidate, dx)
            
            # Update personal best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_coeffs = coeffs.copy()
                
            # Update particle velocity and position
            r1, r2 = np.random.random(2)
            for j in range(len(velocities[i])):
                velocities[i][j] = (w * velocities[i][j] + 
                                  c1 * r1 * (coeffs[j] - particles[i][j]) + 
                                  c2 * r2 * (gbest_coeffs[j] - particles[i][j]))
                particles[i][j] += velocities[i][j]
                
        # Early exit
        if gbest_c5 <= 0.3807:
            break
    
    # Final refinement of best solution with SLSQP
    def objective(coeffs):
        h_candidate = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        
        for k in range(n_freq):
            if k == 0:
                h_candidate += coeffs[0]
            else:
                h_candidate += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
        
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = normalize_h(h_candidate, target_sum)
        return compute_c5(h_candidate, dx)
    
    def constraint_sum(coeffs):
        h_candidate = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        
        for k in range(n_freq):
            if k == 0:
                h_candidate += coeffs[0]
            else:
                h_candidate += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
        
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = normalize_h(h_candidate, target_sum)
        return np.sum(h_candidate) - target_sum
    
    # Refine the best solution found
    bounds = [(-2, 2) for _ in range(2 * n_freq)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, gbest_coeffs.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        final_coeffs = result.x
    else:
        final_coeffs = gbest_coeffs
    
    # Generate final h from optimized coefficients
    h_final = np.zeros(n_points)
    x = np.linspace(0, 2, n_points)
    
    for k in range(n_freq):
        if k == 0:
            h_final += final_coeffs[0]
        else:
            h_final += final_coeffs[2*k-1] * np.cos(k * np.pi * x) + final_coeffs[2*k] * np.sin(k * np.pi * x)
    
    h_final = np.clip(h_final, 0, 1)
    h_final = normalize_h(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
