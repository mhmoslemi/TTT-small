import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    perturb_fraction = 0.1
    
    sa_steps = 100000
    sa_step = 0
    
    # Local optimization parameters
    local_refine_steps = 200
    
    while time.time() - start_time < budget_s - 1 and sa_step < sa_steps:
        # Make perturbation
        h_new = h.copy()
        n_perturb = int(n_points * perturb_fraction)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] = np.random.uniform(0, 1)
        
        # Re-normalize to maintain sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Check if we should accept this new h
        current_c5 = compute_c5(h, dx)
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
        else:
            # Accept with probability
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temp):
                h = h_new
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
        
        sa_step += 1
        
        # Cool down
        if sa_step % 100 == 0:
            temp *= cooling_rate
            
        # Early exit if we have a good enough solution
        if best_c5 <= 0.3807:
            break
    
    # Local refinement with SLSQP if time allows
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': local_refine_steps})
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            final_c5 = compute_c5(h_final, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_final.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
