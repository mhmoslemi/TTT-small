import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Simple projection: adjust middle elements
            diff = (n_points / 2 - current_sum)
            # Distribute adjustment to middle elements
            mid_idx = n_points // 2
            # Adjust first few and last few elements to meet constraint
            adjust_start = max(0, mid_idx - 10)
            adjust_end = min(n_points, mid_idx + 10)
            # Check current adjustment range
            for i in range(adjust_start, adjust_end):
                if diff > 0:
                    available = 1 - h[i]
                    to_add = min(available, diff)
                    h[i] += to_add
                    diff -= to_add
                else:
                    available = h[i]
                    to_subtract = min(available, -diff)
                    h[i] -= to_subtract
                    diff += to_subtract
                if abs(diff) < 1e-10:
                    break
        return h
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with local search
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically using the maximum lag
        n = len(h)
        g = np.zeros(n)
        k_star = max_k - (n - 1)  # Convert from full to lag index
        
        # Gradient computation for overlap at k_star
        for i in range(n):
            # Handle wrap-around properly
            if i + k_star < n and i + k_star >= 0:
                g[i] = (1 - h[i + k_star]) * dx
            elif i + k_star >= n:
                g[i] = -h[i + k_star - n] * dx
            elif i + k_star < 0:
                g[i] = -h[i + k_star + n] * dx
            else:
                g[i] = 0
                
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply projection to maintain constraints
        new_h = project_h(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_h(h)
            
        # Adjust step size and perturbation
        if iteration % 1000 == 0:
            step_size *= 0.5
        if iteration % 500 == 0:
            pert_std = max(0.001, 0.01 * (1.0 - iteration / 10000))
            pert = np.random.normal(0, pert_std, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
