import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_individual = None
    best_c5 = float('inf')
    stagnation_counter = 0
    max_stagnation = 20
    
    gen = 0
    while time.time() - start_time < budget_s - 1 and stagnation_counter < max_stagnation:
        # Evaluate current population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        elite_size = max(1, population_size // 5)
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.8 and len(new_population) + 1 < population_size:
                idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
                h1, h2 = population[idx1], population[idx2]
                crossover_point = np.random.randint(1, n_points - 1)
                offspring1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                offspring2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
                new_population.append(offspring1)
                if len(new_population) < population_size:
                    new_population.append(offspring2)
            else:
                # Mutation
                idx = np.random.randint(elite_size)
                h = population[idx].copy()
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population.append(h)
        
        population = new_population
        
        # Adjust mutation rate based on progress
        if gen > 0 and gen % 10 == 0:
            if stagnation_counter > 0:
                for i in range(len(population)):
                    noise = np.random.normal(0, 0.03, n_points)
                    population[i] = population[i] + noise
                    population[i] = np.clip(population[i], 0, 1)
                    population[i] = normalize_h(population[i], target_sum)
        
        # Check for stagnation
        if gen > 0 and gen % 10 == 0:
            if abs(fitness[0] - fitness[-1]) < 1e-8:
                stagnation_counter += 1
            else:
                stagnation_counter = 0

        gen += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
