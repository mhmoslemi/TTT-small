import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct a comb-like h based on specified parameters."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
        
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        start_x = i * (tooth_width + spacing)
        end_x = start_x + tooth_width
        start_idx = int(start_x / dx)
        end_idx = int(end_x / dx)
        if end_idx > n_points:
            end_idx = n_points
            
        for j in range(start_idx, end_idx):
            x = j * dx
            if x < start_x:
                h[j] = max(0, 1 - (start_x - x) / slope)
            elif x > end_x:
                h[j] = max(0, 1 - (x - end_x) / slope)
            else:
                h[j] = 1.0
    
    return h

def optimize_comb_parameters(h, n_points, dx, n_iter=100):
    """Optimize comb parameters."""
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Parameter ranges
    n_teeth_range = range(5, 11)
    tooth_width_range = np.linspace(0.1, 0.5, 10)
    slope_range = np.linspace(0.05, 0.2, 10)
    
    for _ in range(n_iter):
        n_teeth = np.random.choice(n_teeth_range)
        tooth_width = np.random.choice(tooth_width_range)
        slope = np.random.choice(slope_range)
        
        # Try to construct new h
        h_new = construct_comb_h(n_points, n_teeth, tooth_width, None, slope)
        
        # Normalize and check
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points/2) / h_new_sum
            
        # Compute C5
        c5 = compute_c5(h_new, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
    return best_h, best_c5

    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000

    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
