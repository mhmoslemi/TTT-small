```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate(h, mutation_rate):
    """Mutate h with given rate."""
    mask = np.random.random(len(h)) < mutation_rate
    delta = np.random.normal(0, 0.05, len(h))
    h = h + mask * delta
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2, crossover_rate):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitness = [fitness[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitness)]
    return population[winner_index].copy()

def local_optimize(h, target_sum, dx):
    """Perform local optimization using scipy minimize."""
    from scipy.optimize import minimize
    
    def constraint_sum(x):
        return np.sum(x) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(x):
        return compute_c5(x, dx)
    
    result = minimize(objective, h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = normalize_h(h_opt, target_sum)
        return h_opt
    else:
        return h.copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    population_size = 100
    population = []
    fitness = []
    
    # Initialize population
    for _ in range(population_size):
        individual = mutate(h.copy(), 0.1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
        fitness.append(compute_c5(individual, dx))
    
    best_c5 = min(fitness)
    best_h = population[np.argmin(fitness)].copy()
    
    # GA iterations
    generations = 60
    mutation_rate = 0.2
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break

        # Create new population
        new_population = [population[0].copy()]  # Elitism
        new_fitness = [fitness[0]]
        
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, 0.8)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Normalize
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            # Local optimization
            child1 = local_optimize(child1, target_sum, dx)
            child2 = local_optimize(child2, target_sum, dx)
            
            new_population.extend([child1, child2])
            new_fitness.extend([compute_c5(child1, dx), compute_c5(child2, dx)])
        
        # Trim to population size
        population = new_population[:population_size]
        fitness = new_fitness[:population_size]
        
        mutation_rate *= 0.95  # Decrease mutation rate
    
    # Final refine best solution
    final_h = local_optimize(best_h, target_sum, dx)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
