import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple diverse initial guesses."""
    guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern: high values at regular intervals
    h = np.zeros(n_points)
    step = max(2, n_points // 10)
    for i in range(0, n_points, step):
        h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Random pattern
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.zeros(n_points)
    h[::2] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Low-discrepancy-like pattern (simplified)
    h = np.sin(np.arange(n_points) * 2 * np.pi / n_points) + 1
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Constant 1s
    h = np.ones(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def perturb_solution(h, perturbation_std):
    """Apply perturbation to solution."""
    h = h + np.random.normal(0, perturbation_std, len(h))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    max_iter = 10000
    
    perturbation_std = 0.02
    perturbation_decay = 0.95
    perturbation_min = 1e-5
    
    # Evaluate initial solutions to see if any is good
    for i, h in enumerate(initial_guesses):
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Now proceed with optimization
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Always ensure we keep at least the best solution found before
        
        if best_c5 <= 0.3807:
            break
            
        # Check if it's time to do a more aggressive perturbation
        if iteration % 50 == 0 and perturbation_std > perturbation_min:
            perturbation_std *= perturbation_decay
            
        # Try to improve the current best solution
        h = best_h.copy()
        h = perturb_solution(h, perturbation_std)
        h = normalize_h(h, target_sum)
        
        # Simplified optimization: mostly perturbation work
        # Only run if time permits
        if time.time() - start_time < budget_s - 5:
            # Objective: minimize c5
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-6})
                if result.success:
                    h = result.x
                else:
                    # If optimization fails, perturb anyway but make it more aggressive
                    h = perturb_solution(h, perturbation_std * 2)
            except:
                h = perturb_solution(h, perturbation_std * 2)
            
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Compute the C5 for current h
        c5 = compute_c5(h, dx)
        
        # Update best if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
        
        # Occasionally do more aggressive search by starting from scratch
        if iteration % 100 == 0:
            h = best_h.copy()
            h = perturb_solution(h, 0.05)
            h = normalize_h(h, target_sum)
            
            # Run local optimization on this one too
            try:
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            except:
                pass
        
    # Final validation
    if best_h is None:
        # Fallback to initial guess if no better solution
        best_h = initial_guesses[0].copy()
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
