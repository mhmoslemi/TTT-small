import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles from initial_h_values if available
    n_particles = 50
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
    else:
        h0 = np.random.rand(n_points)
        h0 = normalize_h(h0, target_sum)
    
    # Initialize PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = np.tile(h0, (n_particles, 1))
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    
    # Add some diversity
    for i in range(n_particles):
        particles[i] += np.random.normal(0, 0.01, n_points)
        particles[i] = project_to_feasible(particles[i], target_sum)
    
    # Initialize personal bests
    personal_best_positions = particles.copy()
    personal_best_values = np.array([compute_c5(p, dx) for p in particles])
    
    # Global best
    global_best_idx = np.argmin(personal_best_values)
    global_best_position = personal_best_positions[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO iteration
    max_iter = 1000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iter_num / max_iter)
        
        # Update each particle
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best_position - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Position update
            particles[i] += velocities[i]
            
            # Project to feasible bounds
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best_position = particles[i].copy()
        
        # Early termination
        if global_best_value <= 0.3807:
            break
    
    # Return best solution found
    return (global_best_position.tolist(), global_best_value, n_points)
