import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with specified spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_width = 1.0  # fixed tooth width in units of dx

    for i in range(0, n_points, spacing):
        start_idx = i
        end_idx = min(i + int(tooth_width / dx), n_points)
        if start_idx < n_points:
            h[start_idx:end_idx] = tooth_height
    
    # Normalize to sum = n_points/2
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small Gaussian noise to pattern."""
    noise = np.random.normal(0, sigma, len(h))
    h_pert = h + noise
    h_pert = np.clip(h_pert, 0, 1)
    return h_pert
