import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))

    # Variables
    h_var = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [sum(h_var) == target_sum]
    constraints += [0 <= h_var[i] <= 1 for i in range(n_points)]
    
    # Overlap constraints for all lags
    for k in range(-n_points+1, n_points):
        # Compute valid indices for this lag
        if k >= 0:
            indices = range(n_points - k)
            expr = sum(h_var[i] * (1 - h_var[i + k]) for i in indices)
        else:
            indices = range(-k, n_points)
            expr = sum(h_var[i] * (1 - h_var[i + k]) for i in indices)
        constraints.append(expr * dx <= C)
    
    # Objective
    objective = Minimize(C)
    
    # Solve
    prob = Problem(objective, constraints)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    while time.time() - start_time < budget_s - 1:
        try:
            prob.solve(solver=MOSEK, verbose=False, time_limit=budget_s/2)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_solution = np.array(h_var.value).flatten()
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (target_sum / np.sum(h_solution))
                current_c5 = C.value
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h_solution.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
            else:
                break
        except:
            break
        
        # Perturb the solution
        h_perturbed = h_solution + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
        h = h_perturbed

    return (best_h.tolist(), float(best_c5), n_points)
