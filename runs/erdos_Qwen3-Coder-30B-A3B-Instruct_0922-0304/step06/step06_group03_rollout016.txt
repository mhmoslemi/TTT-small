import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    noise_scale = 0.02
    noise_decay = 0.95
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Identify high overlap regions
        corr = correlate(h, 1-h, mode="full")
        if len(corr) > 0:
            peak_indices = np.argsort(-corr)[:max(1, n_points // 10)]
            peak_positions = np.array(peak_indices) - (len(corr) // 2)
            # Focus perturbation on high overlap areas
            mask = np.zeros(n_points)
            for pos in peak_positions:
                # Convert to discrete index
                idx = pos + n_points // 2
                if 0 <= idx < n_points:
                    mask[idx] = 1.0
            mask = np.convolve(mask, np.ones(5), mode="same") / 5  # Smooth
        else:
            mask = np.ones(n_points)
            
        # Perturb with masking
        noise = np.random.normal(0, noise_scale, n_points)
        h = h + mask * noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = result.x
        except:
            pass
            
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

        # Adjust noise scale
        if iteration % 500 == 0 and noise_scale > 0.001:
            noise_scale *= noise_decay
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
