import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with best known construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    # Multi-start optimization with different initializations
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Try multiple initializations to avoid local minima
    initializations = []
    
    # Original construction
    initializations.append(base_h.copy())
    
    # Random perturbations
    for i in range(5):
        h = np.random.random(n_points) * 0.5
        h = normalize_h(h, target_sum)
        initializations.append(h.copy())
    
    # Step function variations
    for i in range(3):
        h = np.zeros(n_points)
        idx1 = int(n_points * (0.2 + 0.1 * i))
        idx2 = int(n_points * (0.8 - 0.1 * i))
        h[:idx1] = 1.0
        h[idx1:idx2] = 0.5
        h[idx2:] = 0.0
        h = normalize_h(h, target_sum)
        initializations.append(h.copy())
    
    # Gaussian-like distributions with different widths
    for i in range(2):
        sigma = 0.2 + 0.2 * i
        x = np.linspace(-1, 1, n_points)
        h = np.exp(-0.5 * (x / sigma)**2)
        h = normalize_h(h, target_sum)
        initializations.append(h.copy())
    
    # Optimization loop with multiple restarts
    for i, h_start in enumerate(initializations):
        h_current = h_start.copy()
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Try optimization with different strategies
        try:
            # Use smaller tolerance for better convergence
            result = minimize(objective, h_current, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h_opt.copy()
        except:
            pass  # Silently continue if optimization fails
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final verification and cleanup
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
