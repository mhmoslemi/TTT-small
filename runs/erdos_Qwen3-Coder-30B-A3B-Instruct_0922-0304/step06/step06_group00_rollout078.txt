import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h = h + np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    crossover_point = np.random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 100
    generations = 60
    mutation_rate_start = 0.2
    crossover_rate = 0.8
    elite_size = 5
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = mutate_individual(h.copy(), mutation_rate_start)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of population
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Track best individual in this generation
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population with elitism
        new_population = []
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring through selection, crossover, mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover_parents(parent1, parent2, crossover_rate)
            
            mutation_strength = mutation_rate_start * np.exp(-gen / 10.0)
            child1 = mutate_individual(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate_individual(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
