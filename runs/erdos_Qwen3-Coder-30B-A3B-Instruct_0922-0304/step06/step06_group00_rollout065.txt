import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_population(population, dx):
    """Evaluate the entire population and return fitness values."""
    fitness = np.array([compute_c5(individual, dx) for individual in population])
    return fitness

def tournament_selection(population, fitness, tournament_size=5):
    """Perform tournament selection."""
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin(fitness[indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, target_sum, dx):
    """Mutate an individual."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    mutated = np.clip(mutated, 0, 1)
    mutated = normalize_h(mutated, target_sum)
    return mutated

def local_optimization(h, target_sum, dx):
    """Perform local optimization on a single individual."""
    from scipy.optimize import minimize
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})
    
    if result.success:
        h_refined = result.x
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = normalize_h(h_refined, target_sum)
        return h_refined
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 100
    generations = 60
    mutation_rate_start = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            individual = np.array(initial_h_values)
            individual += np.random.normal(0, 0.01, n_points)
        else:
            individual = np.zeros(n_points)
            individual[:n_points//2] = 1
            individual += np.random.normal(0, 0.02, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for gen in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitness = evaluate_population(population, dx)
        
        # Track best solution
        gen_best_idx = np.argmin(fitness)
        if fitness[gen_best_idx] < best_c5:
            best_c5 = fitness[gen_best_idx]
            best_h = population[gen_best_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected = tournament_selection(population, fitness, tournament_size)
        
        # Create next generation
        new_population = []
        
        # Keep best individual
        new_population.append(population[gen_best_idx].copy())
        
        # Generate offspring
        for i in range(0, population_size - 1, 2):
            parent1 = selected[i]
            parent2 = selected[i+1]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutate
            child1 = mutate(child1, mutation_rate_start * (1 - gen / generations), target_sum, dx)
            child2 = mutate(child2, mutation_rate_start * (1 - gen / generations), target_sum, dx)
            
            # Local optimization
            child1 = local_optimization(child1, target_sum, dx)
            child2 = local_optimization(child2, target_sum, dx)
            
            new_population.extend([child1, child2])
            
        # Replace population
        population = new_population[:population_size]
    
    # Final check and return
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
