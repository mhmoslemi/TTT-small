import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 10000
    iteration = 0
    eta = 0.01
    mu = 0.9
    
    # Initialize momentum term
    velocity = np.zeros(n_points)
    
    def project_and_normalize(h):
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to meet constraint sum(h) = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current maximum lag
        g = np.zeros(n_points)
        k = max_k
        for i in range(n_points):
            if i + k < n_points:
                g[i] = (1 - h[i + k]) * dx
            else:
                g[i] = -h[i - (n_points - k)] * dx
        
        # Update with momentum and adaptive step size
        velocity = mu * velocity + g
        new_h = h - eta * velocity
        
        # Apply constraints
        new_h = project_and_normalize(new_h)
        
        h = new_h
        
        # Adapt step size and momentum
        if iteration % 1000 == 0:
            eta *= 0.5
            mu = max(0.5, mu * 0.95)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
