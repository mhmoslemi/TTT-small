import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, n_pop, target_sum):
    """Create an initial population for genetic algorithm."""
    population = []
    for _ in range(n_pop):
        # Use a variation of the initial solution if available
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            # Create a base function with structured pattern
            h = np.zeros(n_points)
            h[:n_points//2] = 1
            # Add some noise for diversity
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
        
        h = normalize_h(h, target_sum)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    # Single point crossover
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate, mutation_strength):
    """Mutate an individual."""
    if np.random.rand() > mutation_rate:
        return h.copy()
    
    noise = np.random.normal(0, mutation_strength, len(h))
    h_mut = h + noise
    return np.clip(h_mut, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Genetic algorithm parameters
    n_pop = 100
    generations = 600
    mutation_rate = 0.2
    mutation_strength = 0.1
    
    # Initialize population
    population = create_initial_population(n_points, n_pop, target_sum)
    
    # Evaluate initial population
    fitnesses = [compute_c5(h, dx) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new population
        new_population = []
        
        # Elitism - keep the best individual
        new_population.append(best_h.copy())
        
        # Generate offspring
        while len(new_population) < n_pop:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover and mutation
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate, mutation_strength)
            child2 = mutate(child2, mutation_rate, mutation_strength)
            
            # Normalize children
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        new_population = new_population[:n_pop]
        
        # Evaluate new population
        fitnesses = [compute_c5(h, dx) for h in new_population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = new_population[min_idx].copy()
            
        # Update population
        population = new_population
        
        # Adaptive mutation rate decrease
        mutation_rate *= 0.995
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
