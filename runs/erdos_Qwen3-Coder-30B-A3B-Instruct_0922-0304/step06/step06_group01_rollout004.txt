import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0

    # Normalize initial h to required sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize PSO particles
    particles = np.random.rand(n_particles, n_freq * 2) * 2 - 1  # [-1, 1]
    velocities = np.random.rand(n_particles, n_freq * 2) * 0.1
    personal_best = particles.copy()
    personal_best_fitness = np.full(n_particles, float('inf'))
    global_best_idx = 0
    global_best_fitness = float('inf')
    
    def h_from_fourier(coeffs):
        """Generate h from Fourier coefficients."""
        h_vals = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_freq):
            a = coeffs[i]
            b = coeffs[i + n_freq]
            h_vals += a * np.cos(i * np.pi * x) + b * np.sin(i * np.pi * x)
        return np.clip(h_vals, 0, 1)
        
    # PSO optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for p in range(n_particles):
            # Convert to h function
            h_particle = h_from_fourier(particles[p])
            h_particle = h_particle * (target_sum / np.sum(h_particle))
            
            # Evaluate fitness
            c5_particle = compute_c5(h_particle, dx)
            
            # Update personal best
            if c5_particle < personal_best_fitness[p]:
                personal_best_fitness[p] = c5_particle
                personal_best[p] = particles[p].copy()
            
            # Update global best
            if c5_particle < global_best_fitness:
                global_best_fitness = c5_particle
                global_best_idx = p
                
            # Update PSO velocity and position
            r1, r2 = np.random.rand(n_freq * 2), np.random.rand(n_freq * 2)
            velocities[p] = (w * velocities[p] + 
                            c1 * r1 * (personal_best[p] - particles[p]) + 
                            c2 * r2 * (personal_best[global_best_idx] - particles[p]))
            particles[p] += velocities[p]
            
        if global_best_fitness < best_c5:
            best_c5 = global_best_fitness
            # Recompute h from the global best coeffs
            best_h = h_from_fourier(personal_best[global_best_idx])
            best_h = best_h * (target_sum / np.sum(best_h))
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
        
    # Final local refinement
    def objective(h_vec):
        return compute_c5(h_vec, dx)
        
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (target_sum / np.sum(final_h))
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h

    return (best_h.tolist(), best_c5, n_points)
