import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        for _ in range(population_size):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    # GA parameters
    crossover_prob = 0.8
    mutation_prob = 0.2
    elitism_count = population_size // 10
    
    best_c5 = float('inf')
    best_individual = None
    
    gen = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_individual = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elitism_count):
            new_population.append(sorted_population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection (tournament)
            parent1_idx = sorted_indices[np.random.randint(0, population_size//2)]
            parent2_idx = sorted_indices[np.random.randint(0, population_size//2)]
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_prob:
                child1 += np.random.normal(0, 0.05, n_points)
                child1 = np.clip(child1, 0, 1)
                child1 = normalize_h(child1, target_sum)
            
            if np.random.rand() < mutation_prob:
                child2 += np.random.normal(0, 0.05, n_points)
                child2 = np.clip(child2, 0, 1)
                child2 = normalize_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:]
        gen += 1
        
        # Adjust hyperparameters based on performance
        if stagnation_count > 20:
            mutation_prob = min(0.3, mutation_prob * 1.1)
            if mutation_prob >= 0.25:
                population_size = min(40, population_size + 5)
        elif stagnation_count > 10:
            mutation_prob = max(0.05, mutation_prob * 0.9)
    
    if best_individual is None:
        # Fallback to initial if no improvement
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
