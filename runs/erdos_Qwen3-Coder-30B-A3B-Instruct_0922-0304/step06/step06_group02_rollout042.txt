import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Genetic Algorithm parameters
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            candidate = h + np.random.normal(0, 0.02, n_points)
        else:
            # Start with a simple pattern and add noise
            candidate = np.ones(n_points)
            candidate[n_points//2:] = 0
            candidate = candidate + np.random.normal(0, 0.02, n_points)
        candidate = np.clip(candidate, 0, 1)
        candidate = normalize_h(candidate, target_sum)
        population.append(candidate)
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
            
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]

        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection with tournament
            parent1 = sorted_population[np.random.randint(0, population_size // 2)]
            parent2 = sorted_population[np.random.randint(0, population_size // 2)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            else:
                child = parent1.copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.01)
                    
            # Enforce constraints
            child = np.clip(child, 0, 1)

                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]

                    return compute_c5(h, dx)
                
                # Refine with SLSQP
                result = minimize(objective, candidate, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                
                if result.success:
                    refined_candidate = result.x
                    refined_candidate = np.clip(refined_candidate, 0, 1)
                    refined_candidate = normalize_h(refined_candidate, target_sum)
                    new_population[i] = refined_candidate
                    
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
