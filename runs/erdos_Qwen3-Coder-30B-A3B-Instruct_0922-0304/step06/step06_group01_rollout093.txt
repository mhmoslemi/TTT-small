import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

        h = np.clip(h, 0, 1)

        tooth_width = 2.0 / n_teeth
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
        
        h = np.zeros(n_points)
        for i in range(n_teeth):
            start_pos = i * (tooth_width + spacing)
            end_pos = start_pos + tooth_width
            
            start_idx = int(start_pos / dx)
            end_idx = int(end_pos / dx)
            
            if start_idx < 0:
                start_idx = 0
            if end_idx >= n_points:
                end_idx = n_points - 1
                
            for j in range(start_idx, end_idx + 1):
                x = j * dx
                if x < start_pos:
                    h[j] = 0
                elif x > end_pos:
                    h[j] = 1
                else:
                    # Linear ramp
                    if x < start_pos + tooth_width/2:
                        h[j] = (x - start_pos) / (tooth_width/2)
                    else:
                        h[j] = 1 - (x - start_pos - tooth_width/2) / (tooth_width/2)
        
        h = np.clip(h, 0, 1)
    
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameter sweep
    n_sweeps = 500
    for sweep in range(n_sweeps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Vary parameters a bit
        n_teeth = 5 + np.random.randint(0, 6)
        tooth_width = 2.0 / n_teeth + 0.02 * (np.random.rand() - 0.5)

            end_idx = int(end_pos / dx)

                start_idx = 0
            if end_idx >= n_points:
                end_idx = n_points - 1
                
            for j in range(start_idx, end_idx + 1):

                if x < start_pos:
                    h_new[j] = 0
                elif x > end_pos:
                    h_new[j] = 1
                else:
                    # Linear ramp
                    if x < start_pos + tooth_width/2:

            return np.sum(h) - target_sum

            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}

            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            final_c5 = compute_c5(h_opt, dx)

                best_c5 = final_c5
                best_h = h_opt.copy()
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
