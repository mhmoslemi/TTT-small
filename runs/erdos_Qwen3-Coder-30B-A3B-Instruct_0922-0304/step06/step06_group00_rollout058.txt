import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    return h

def compute_gradient(h, dx, epsilon=1e-5):
    """Compute gradient of C5 with respect to h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_to_feasible(h_plus, np.sum(h))
        c5_plus = compute_c5(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= epsilon
        h_minus = project_to_feasible(h_minus, np.sum(h))
        c5_minus = compute_c5(h_minus, dx)
        
        grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
    
    return grad

def line_search(h, grad, dx, max_alpha=1.0, max_iterations=50):
    """Perform backtracking line search."""
    alpha = max_alpha
    c5_current = compute_c5(h, dx)
    for _ in range(max_iterations):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, np.sum(h))
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < c5_current:
            return alpha, h_new
        alpha *= 0.5
    return alpha, h

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Parameter for barrier method
    mu = 1.0
    decay_factor = 0.5
    update_interval = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute gradient of C5
        grad_c5 = compute_gradient(h, dx)
        
        # Compute barrier function gradient
        barrier_grad = np.zeros(n_points)
        for i in range(n_points):
            barrier_grad[i] = -1.0 / h[i] - 1.0 / (1.0 - h[i]) + 1.0 / (target_sum - np.sum(h))
        
        # Combined gradient
        combined_grad = grad_c5 + mu * barrier_grad
        
        # Line search
        alpha, h_new = line_search(h, combined_grad, dx)
        
        h = h_new
        h = project_to_feasible(h, target_sum)
        
        # Compute new C5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adapt mu
        if iteration % update_interval == 0 and iteration > 0:
            mu *= decay_factor

        # Early termination
