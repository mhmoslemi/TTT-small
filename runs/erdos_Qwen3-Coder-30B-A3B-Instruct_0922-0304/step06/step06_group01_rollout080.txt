import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h(x) in time domain."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 * np.ones(n_points)
    for k in range(1, n_freq + 1):
        h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    # Using DFT to solve the Fourier coefficients 
    # We compute the coefficients for k = 0, 1, ..., n_freq
    # which correspond to a_0, a_1, b_1, a_2, b_2, ..., a_n_freq, b_n_freq
    coeffs = np.zeros(2 * n_freq + 1)
    coeffs[0] = np.mean(h)
    for k in range(1, n_freq + 1):
        coeffs[2*k-1] = 2 * np.mean(h * np.cos(k * np.pi * np.linspace(0, 2, len(h))))
        coeffs[2*k] = 2 * np.mean(h * np.sin(k * np.pi * np.linspace(0, 2, len(h))))
    return coeffs

def constraint_sum(h, target_sum):
    return np.sum(h) - target_sum

def objective_fourier(coeffs, n_points, n_freq, target_sum, dx):
    h = fourier_to_h(coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of frequency components
    
    # Initial guess from provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Use a rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    # Convert initial h to coefficients for DE
    init_coeffs = h_to_fourier(h, n_freq)
    
    # Define bounds for coefficients
    bounds = [(0, 1) for _ in range(n_freq * 2 + 1)]
    
    # Objective for differential evolution
    def de_objective(coeffs):
        return objective_fourier(coeffs, n_points, n_freq, target_sum, dx)
    
    # Perform differential evolution
    de_result = differential_evolution(
        de_objective,
        bounds,
        seed=seed,
        maxiter=300,
        popsize=70,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.7,
        callback=lambda x, convergence: None
    )
    
    best_coeffs = de_result.x
    h = fourier_to_h(best_coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Refine with SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': lambda h: constraint_sum(h, target_sum)}
    
    def obj_slsqp(h):
        return compute_c5(h, dx)
    
    slsqp_result = minimize(
        obj_slsqp,
        h,
        method='SLSQP',
        bounds=bounds_slsqp,
        constraints=constraints,
        options={'maxiter': 100}
    )
    
    if slsqp_result.success:
        h = slsqp_result.x
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))

    final_c5 = compute_c5(h, dx)
    
    return (h.tolist(), final_c5, n_points)
