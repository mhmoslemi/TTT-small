import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate):
    """Mutate an individual with given rate."""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, 0.05, len(h))
    h = h + mask * noise
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2, crossover_rate):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a reasonable starting population
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm parameters
    population_size = 100
    max_generations = 60
    mutation_rate = 0.2
    crossover_rate = 0.8
    elite_size = 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = mutate_individual(h.copy(), 0.1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    # Evolution loop
    best_c5 = compute_c5(h, dx)
    best_individual = h.copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness of the population
        fitness = [compute_c5(ind, dx) for ind in population]
        
        # Keep best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep the best individuals
        elite_indices = np.argsort(fitness)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate_individual(child1, mutation_rate)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate_individual(child2, mutation_rate)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size if needed
        population = new_population[:population_size]
        
        # Update mutation rate (reduce over time)
        mutation_rate = max(0.01, mutation_rate * 0.98)
        
        generation += 1
    
    # Final refinement using gradient descent on best individual
    h_final = best_individual.copy()
    h_final = normalize_h(h_final, target_sum)
    current_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), current_c5, n_points)
