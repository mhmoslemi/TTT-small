import time
from scipy.signal import correlate

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute overlap integral using correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    max_iter = int(1e6)
    stagnation_count = 0
    stagnation_limit = 500
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and T > min_T and iter_count < max_iter:
        iter_count += 1

        # Perturb h: randomly select two indices and transfer mass between them
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i != j and can transfer
        if i == j:
            continue
            
        # Determine maximum amount to transfer
        delta = min(1 - h[i], h[j])
        if delta <= 0:
            continue
            
        # Apply change
        amount = np.random.uniform(0, delta)
        h[i] += amount
        h[j] -= amount
        
        # Ensure bounds and renormalize
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5
        new_c5 = compute_c5(h)
        
        # Accept with Metropolis probability if worse
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Accept worse solution with probability
            if np.random.rand() < np.exp(-(new_c5 - best_c5) / T):
                # Accept it
                pass
            else:
                # Reject, revert to previous state
                h = best_h.copy()
        
        # Update temperature
        T *= alpha
        
        # Stagnation detection
        if iter_count % stagnation_limit == 0:
            if stagnation_count > 3:
                T *= 1.2  # Increase temp to escape local minima
                stagnation_count = 0
            else:
                stagnation_count += 1
    
    # Final recompute for best
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
