import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral using correlation
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h from pre-imported values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        # Initialize with a step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Alternating optimization
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Optimize h given current overlap estimates
        # For this problem, we'll continue using gradient-based approach
        h_old = h.copy()
        
        # Add small random perturbation first
        perturbation = np.random.normal(0, 0.01, n_points)
        h = h + perturbation
        h = np.clip(h, 0, 1)
        
        # Normalize to target sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        
        # Local optimization using SLSQP
        def constraint_sum(h_vals):
            return np.sum(h_vals) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h_vals):
            return compute_c5(h_vals, dx)
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 50, 'ftol': 1e-6})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                if current_sum > 0:
                    h = h * (target_sum / current_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass  # Fall back to previous solution
        
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
