import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Initial heuristic: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 30000
    
    # Discrete levels for search
    levels = [0.0, 0.25, 0.5, 0.75, 1.0]
    
    # Use adaptive step size
    step_size = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Perturb and update
        perturbation = np.random.normal(0, step_size, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize sum
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Accept improvement or with probability based on step_size
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < c5 or np.random.rand() < 0.1:
            h = h_new
            c5 = new_c5
            
        # Occasionally try discrete optimization
        if iteration % 5 == 0:
            # Project to discrete levels
            h_proj = np.array([min(levels, key=lambda x: abs(x - val)) for val in h])
            h_proj = np.clip(h_proj, 0, 1)
            # Renormalize
            current_sum = np.sum(h_proj)
            if current_sum > 0:
                h_proj = h_proj * (target_sum / current_sum)
            proj_c5 = compute_c5(h_proj, dx)
            if proj_c5 < c5:
                h = h_proj
                c5 = proj_c5
                
        # Occasionally increase step size
        if iteration % 50 == 0 and step_size < 0.1:
            step_size *= 1.1
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
