import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    stagnation_count = 0
    mutation_rate = 0.2
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append((c5, h))
        
        # Sort by fitness (lower is better)
        fitness.sort(key=lambda x: x[0])
        best_fitness = fitness[0][0]
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_individual = fitness[0][1].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Select top 50%
        selected_count = population_size // 2
        selected = [h for _, h in fitness[:selected_count]]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        new_population.extend(selected[:population_size//10])
        
        # Crossover
        while len(new_population) < population_size:
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            parent1, parent2 = selected[idx1], selected[idx2]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            new_population.extend([child1, child2])

        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                h = new_population[i] + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
                
        population = new_population
        generation += 1
        
        # Adjust mutation rate if stagnating
        if stagnation_count > 20:
            mutation_rate = min(0.5, mutation_rate + 0.05)
            stagnation_count = 0
        elif stagnation_count > 10:
            mutation_rate = max(0.1, mutation_rate - 0.02)
    
    # Final evaluation of best individual
    if best_individual is not None:
        final_c5 = evaluate_individual(best_individual, dx)
    else:
        # Fallback to initial if no improvement found
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
        final_c5 = evaluate_individual(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
