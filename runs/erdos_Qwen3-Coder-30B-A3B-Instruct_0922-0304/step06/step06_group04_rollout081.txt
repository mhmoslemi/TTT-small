import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        # Add mutated versions
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        # Create random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            population.append(h.copy())
    
    # GA Parameters
    crossover_prob = 0.8
    mutation_prob = 0.2
    elite_size = population_size // 10
    stagnation_limit = 20
    stagnation_count = 0
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_scores[i] for i in sorted_indices]
        
        # Track best
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_individual = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Elitism
        new_population = sorted_population[:elite_size].copy()
        
        # Crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob:
                a = sorted_population[np.random.randint(0, population_size // 2)]
                b = sorted_population[np.random.randint(0, population_size // 2)]
                
                # Swap segments
                split_point = np.random.randint(1, n_points - 1)
                offspring1 = np.concatenate([a[:split_point], b[split_point:]])
                offspring2 = np.concatenate([b[:split_point], a[split_point:]])
                
                new_population.append(offspring1)
                if len(new_population) < population_size:
                    new_population.append(offspring2)
            else:
                # Copy from parent without crossover
                parent = sorted_population[np.random.randint(0, population_size // 2)]
                new_population.append(parent)
        
        # Mutation
        for i in range(elite_size, len(new_population)):
            if np.random.rand() < mutation_prob:
                h = new_population[i]
                h = h + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
        
        # Update population
        population = new_population
        generation += 1
        
        # Adjust parameters if stagnating
        if stagnation_count > stagnation_limit:
            if mutation_prob > 0.05:
                mutation_prob -= 0.02
            population_size = min(40, population_size + 5)
            stagnation_count = 0
    
    # Final best evaluation
    if best_individual is None:
        final_c5 = float('inf')
        final_individual = population[0]
    else:
        final_c5 = compute_c5(best_individual, dx)
        final_individual = best_individual
    
    return (final_individual.tolist(), final_c5, n_points)
