```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # For tracking noise amplitude decay
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to identify regions of high overlap
        corr = correlate(h, 1 - h, mode="full")
        # Get indices of peaks to focus perturbations
        peak_indices = np.argsort(corr)[-int(n_points*0.1):]  # Top 10%
        
        # Create a mask to focus on high overlap regions
        mask = np.zeros(n_points)
        for idx in peak_indices:
            # Add influence over a small region around the peak
            start = max(0, idx - 2)
            end = min(n_points, idx + 3)
            mask[start:end] = 1
            
        # Apply perturbations
        perturbation = np.random.normal(0, noise_amplitude, n_points)
        # Apply mask to perturbation
        perturbation = perturbation * mask
        
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Use SLSQP with constraints to refine
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        except:
            # Fallback if optimization fails
            pass
            
        # Occasionally do more aggressive search
        if iteration % 50 == 0 and noise_amplitude > 0.001:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Adjust noise amplitude
        if iteration % 500 == 0:
            noise_amplitude *= 0.95
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
