import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perturb h
        h_candidate = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure valid indices for perturbation
        if h[i] >= 1 or h[j] <= 0:
            continue
            
        delta = min(1 - h[i], h[j])
        delta *= np.random.random() * 0.1  # Random scaling
        
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Renormalize
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # New C5
        new_c5 = compute_c5(h_candidate)
        
        # Metropolis acceptance
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Cooling schedule
        if iteration % 100 == 0:
            T *= alpha
            
        # Handle stagnation
        if stagnation_count > max_stagnation:
            T *= 1.2  # Increase temperature to escape local minima
            stagnation_count = 0
            
        # Prevent T from becoming too small
        if T < min_T:
            break
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
