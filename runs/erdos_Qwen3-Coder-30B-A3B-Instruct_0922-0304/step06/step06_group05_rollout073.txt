import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    def line_search(h, g, max_iter=10):
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        current_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            new_h = h - eta * g
            new_h = project_h(new_h)
            new_c5 = compute_c5(new_h)
            
            if new_c5 <= current_c5 - alpha * eta * np.dot(g, g):
                return eta, new_h
            eta *= beta
            
        return eta, h

    def compute_gradient(h, k_star):
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                g[i] = -h[i - k_star] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()

    iteration = 0
    max_iter = 50000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap to find dominant k
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        current_c5 = overlap[max_k] * dx
        
        # Store best solution so far
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient
        g = compute_gradient(h, max_k)
        
        # Line search for optimal step size
        eta, new_h = line_search(h, g)
        
        # Update h
        h = new_h
        
        # Make sure projection maintains constraints
        h = project_h(h)
        
        # Perturb occasionally to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01 * np.exp(-iteration/1000), n_points)
            h = h + noise
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
