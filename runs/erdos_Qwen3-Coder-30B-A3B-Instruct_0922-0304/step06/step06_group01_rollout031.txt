import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x)."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs)) * np.pi * x[:, None]), axis=1
    )
    return np.clip(h, 0, 1)

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    x = np.linspace(0, 2, len(h))
    coeffs = [np.mean(h)]
    for k in range(1, n_freq):
        coeffs.append(2 * np.mean(h * np.cos(k * np.pi * x)) / 2)
    return np.array(coeffs)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    n_freq = 15
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = normalize_h(h_current, target_sum)
    else:
        # Simple pulse initialization
        h_current = np.zeros(n_points)
        h_current[n_points//4:3*n_points//4] = 1.0
        h_current = normalize_h(h_current, target_sum)
    
    # Define bounds for Fourier coefficients
    bounds = [(-1.0, 1.0)] * n_freq
    bounds[0] = (0.0, 1.0)  # a0 guess needs to be [0, 1] as h in [0,1]

    # Create objective function in Fourier space
    def objective_fourier(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    # Use differential evolution to search in Fourier space
    result = differential_evolution(
        objective_fourier,
        bounds,
        maxiter=300,
        popsize=70,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None
    )

    # Final tuning with local optimization
    best_coeffs = result.x
    best_h = fourier_to_h(best_coeffs, n_points)
    best_h = normalize_h(best_h, target_sum)
    
    # Local refine with SLSQP
    bounds_h = [(0, 1) for _ in range(n_points)]
    
    def objective_h(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    refine_result = minimize(
        objective_h,
        best_h.copy(),
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100}
    )
    
    if refine_result.success:
        final_h = np.clip(refine_result.x, 0, 1)
        final_h = normalize_h(final_h, target_sum)
    else:
        final_h = best_h
    
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
