import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        else:
            h_init = np.random.rand(n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(h_init.copy())
        p_best_vals.append(compute_c5(h_init, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    best_c5 = g_best_val
    best_h = g_best.copy()
    
    # PSO iterations
    for iter_num in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2, n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate particle
            c5 = compute_c5(particles[i], dx)
            if c5 < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = c5
                
                if c5 < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = c5
                    
                    if g_best_val <= 0.3807:
                        break
                        
        if g_best_val <= 0.3807:
            break
            
        # Apply frequency domain acceleration every 50 iterations
        if iter_num % 50 == 0 and iter_num > 0:
            # Sort particles by fitness
            sorted_indices = np.argsort(p_best_vals)
            worst_indices = sorted_indices[-n_particles//4:]  # Top 25% worst
            
            for idx in worst_indices:
                # Convert to frequency domain
                freq_rep = np.fft.fft(p_best[idx])
                # Apply high-pass filter
                cutoff = 0.3
                freq_rep[np.abs(np.fft.fftfreq(n_points)) < cutoff] = 0
                # Convert back
                filtered_h = np.real(np.fft.ifft(freq_rep))
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                particles[idx] = filtered_h
                p_best[idx] = filtered_h.copy()  # Update p_best
                p_best_vals[idx] = compute_c5(filtered_h, dx)
            
        # Gradient refinement every 100 iterations
        if iter_num % 100 == 0 and iter_num > 0:
            # Select top particles (10 best so far)
            top_indices = np.argsort(p_best_vals)[:10]
            for i in top_indices:
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, particles[i], method='SLSQP', 
                                bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined_h = result.x
                    refined_h = np.clip(refined_h, 0, 1)
                    refined_h = normalize_h(refined_h, target_sum)
                    new_c5 = compute_c5(refined_h, dx)
                    
                    if new_c5 < p_best_vals[i]:
                        particles[i] = refined_h
                        p_best[i] = refined_h.copy()
                        p_best_vals[i] = new_c5
                        
                        if new_c5 < g_best_val:
                            g_best = refined_h.copy()
                            g_best_val = new_c5
                
                if g_best_val <= 0.3807:
                    break
    
    # Final evaluation
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
