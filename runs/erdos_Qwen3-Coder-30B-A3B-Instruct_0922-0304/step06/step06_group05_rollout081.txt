import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback to a sinusoidal initialization
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / current_sum
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find the max k
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution found so far
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if we've achieved the target
        if best_c5 <= 0.3807:
            break
            
        # Compute the gradient analytically based on max_k
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            g[i] = (1 - h[j]) * dx
            
        # Apply momentum
        velocity = momentum * prev_grad + (1 - momentum) * g
        prev_grad = g.copy()
        
        # Update h
        new_h = h - step_size * velocity
        
        # Project to constraints
        h = project_h(new_h)
        
        # Adaptive step size and momentum
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum = max(0.5, momentum * 0.95)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
