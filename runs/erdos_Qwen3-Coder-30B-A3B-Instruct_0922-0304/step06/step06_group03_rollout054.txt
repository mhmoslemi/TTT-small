import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiple initializations
    initial_guesses = []
    
    # 1. Original guess
    initial_guesses.append(h.copy())
    
    # 2. Comb pattern
    h_comb = np.zeros(n_points)
    for i in range(0, n_points, 4):
        if i < n_points:
            h_comb[i] = 1.0
    h_comb = normalize_h(h_comb, target_sum)
    initial_guesses.append(h_comb)
    
    # 3. Random with sum constraint
    h_rand = np.random.rand(n_points)
    h_rand = normalize_h(h_rand, target_sum)
    initial_guesses.append(h_rand)
    
    # 4. Alternating pattern
    h_alt = np.zeros(n_points)
    for i in range(n_points):
        if i % 2 == 0:
            h_alt[i] = 0.8
        else:
            h_alt[i] = 0.2
    h_alt = normalize_h(h_alt, target_sum)
    initial_guesses.append(h_alt)
    
    # 5. Low-discrepancy initialization (simple approach)
    h_lowdisc = np.sin(np.arange(n_points) * np.pi / n_points)
    h_lowdisc = np.clip(h_lowdisc, 0, 1)
    h_lowdisc = normalize_h(h_lowdisc, target_sum)
    initial_guesses.append(h_lowdisc)
    
    # Optimization loop
    iteration = 0
    max_iter = 15000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try each initial guess
        for h_init in initial_guesses:
            h = h_init.copy()
            for _ in range(50):  # Inner optimization loop
                if time.time() - start_time > budget_s - 1:
                    break
                    
                # Evaluate current
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
                if best_c5 <= 0.3807:
                    break
                
                # Systematic perturbation
                perturb_std = max(0.005, 0.02 * (1 - iteration / max_iter))
                if iteration % 10 == 0:
                    h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
                    h = normalize_h(h, target_sum)
                else:
                    h = np.clip(h + np.random.normal(0, perturb_std/2, n_points), 0, 1)
                    h = normalize_h(h, target_sum)   
                    
                # Occasionally do more aggressive search
                if iteration % 50 == 0:
                    h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
                    h = normalize_h(h, target_sum)
                    
                iteration += 1
                
                if best_c5 <= 0.3807:
                    break

        # Random improvement moves
        h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
        h = normalize_h(h, target_sum)
        
        # Check again
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
