import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Track convergence
    last_improvement = 0
    patience = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        if best_c5 <= 0.3807:
            break
            
        # Adapt noise amplitude
        noise_amplitude = 0.02 * (0.95 ** (iteration // 500))
        
        # Compute cross-correlation to identify problematic regions
        corr = correlate(h, 1 - h, mode="full")
        # Identify indices where correlation is high (top 20%)
        peak_indices = np.argsort(corr)[-max(2, n_points//10):]
        
        # Apply perturbation to these regions
        perturbation = np.zeros(n_points)
        for idx in peak_indices:
            if idx < n_points:
                # Perturb multiple points around peak index
                for i in range(max(0, idx - 2), min(n_points, idx + 3)):
                    perturbation[i] += np.random.normal(0, noise_amplitude)
        
        # Apply perturbation and maintain constraints
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally do more aggressive search
        if iteration % 50 == 0:
            # Try more aggressive random perturbations
            h = np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1)
            h = normalize_h(h, target_sum)
        
        # Periodically reset to best solution if no improvement
        if iteration - last_improvement > patience:
            h = best_h.copy()
            last_improvement = iteration
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
