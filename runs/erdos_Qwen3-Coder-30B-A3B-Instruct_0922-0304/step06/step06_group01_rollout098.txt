import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    popsize = 2 * n_points
    lambda_ = popsize // 2
    mu = lambda_ // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # CMA-ES initialization
    mean = h.copy()
    sigma = 0.2
    C = np.eye(n_points)
    chi_n = np.sqrt(n_points)
    c1 = 2.0 / (n_points + 2.0)
    cmu = 2.0 * (mu - 2.0 + 1.0 / mu) / (n_points + (mu + 2.0) * (mu + 2.0))
    sigma_factor = 1.0 + 0.5 * np.sqrt(n_points)
    
    current_time = time.time()
    max_iter = 500
    
    for iteration in range(max_iter):
        if current_time - start_time > budget_s - 1:
            break
            
        # Sample new individuals
        populations = []
        for _ in range(popsize):
            x = np.random.multivariate_normal(mean, sigma**2 * C)
            x = np.clip(x, 0, 1)
            x = normalize_h(x, target_sum)
            populations.append(x)
        
        # Evaluate population
        fitnesses = [compute_c5(p, dx) for p in populations]
        
        # Sort by fitness
        idxs = np.argsort(fitnesses)
        best_idx = idxs[0]
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = populations[best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Update mean
        best_pop = [populations[i] for i in idxs[:mu]]
        mean = np.sum([weights[i] * best_pop[i] for i in range(mu)], axis=0)
        
        # Update covariance matrix
        C = (1 - c1 - cmu) * C
        for i in range(mu):
            z = best_pop[i] - mean
            C += cmu * np.outer(z, z)
        
        # Update step size
        sigma = sigma * np.exp((c1 / sigma_factor) * (np.linalg.norm(mean - h) / chi_n - 1))
        current_time = time.time()
    
    # Local polishing with SLSQP
    if current_time - start_time < budget_s - 5:
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum
        constraint = {'type': 'eq', 'fun': constraint_sum}
        result = minimize(lambda h: compute_c5(h, dx), best_h.copy(), 
                         method='SLSQP', bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h.copy()
                best_c5 = refined_c5
    
    return (best_h.tolist(), best_c5, n_points)
