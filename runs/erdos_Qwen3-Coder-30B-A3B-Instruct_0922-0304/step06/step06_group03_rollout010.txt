import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing and tooth height."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    h[tooth_positions] = tooth_height
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbations to a pattern."""
    perturbation = np.random.normal(0, sigma, len(h))
    return np.clip(h + perturbation, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a simple comb pattern
        h = create_comb_pattern(n_points, 3, 1.0)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute spacing options
    spacing_options = [2, 3, 4, 5, 6]
    
    # Multi-comb exploration with adaptive search
    iteration = 0
    max_iter = 50000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try several different combinations of combs
        if iteration % 10 == 0:
            # Randomly shift to different patterns
            new_h = h.copy()
            # Occasionally do aggressive changes
            if iteration % 50 == 0:
                # Create a new alternating pattern
                new_h = np.zeros(n_points)
                # Create a complex comb with two types of teeth
                for i in range(n_points):
                    if i % 4 < 2:  # alternate groups
                        new_h[i] = 1.0
                new_h = normalize_h(new_h, target_sum)
            else:
                # Small random noise
                noise = np.random.normal(0, 0.01, n_points)
                new_h = np.clip(h + noise, 0, 1)
                new_h = normalize_h(new_h, target_sum)
            
            h = new_h
            
        # Local optimization step
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Try optimization if we haven't optimized recently
        if iteration % 20 == 0:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = result.x
            else:
                # If optimization fails, keep h and continue
                h = h.copy()
        else:
            # Just make the normal evolution
            pass
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
