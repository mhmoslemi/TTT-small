import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with specified spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    h[tooth_positions] = tooth_height
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small Gaussian perturbations to the pattern."""
    perturbation = np.random.normal(0, sigma, len(h))
    return np.clip(h + perturbation, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Create comb patterns
    comb_spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    
    for spacing in comb_spacings:
        comb = create_comb_pattern(n_points, spacing)
        comb = normalize_h(comb, target_sum)
        comb_patterns.append(comb)
    
    # Optimization loop
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current best solution
        c5 = compute_c5(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try each comb pattern
        for comb in comb_patterns:
            # Create derivatives of this comb
            derivatives = [comb.copy()]
            for _ in range(4):
                perturbed = perturb_pattern(comb, sigma=0.02)
                derivatives.append(perturbed.copy())
            
            for deriv in derivatives:
                # Optimize with SLSQP
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    result = minimize(objective, deriv, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                    if result.success:
                        optimized = result.x
                        optimized = np.clip(optimized, 0, 1)
                        optimized = normalize_h(optimized, target_sum)
                        c5_opt = compute_c5(optimized, dx)
                        if c5_opt < best_c5:
                            best_c5 = c5_opt
                            best_h = optimized.copy()
                except:
                    pass
                
                if best_c5 <= 0.3807:
                    break
                    
        # Occasionally help escape local minima
        if iteration % 100 == 0:
            # Perturb the current best
            h = perturb_pattern(best_h, sigma=0.01)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            h = np.random.choice([best_h, h], p=[0.9, 0.1])
            
        iteration += 1
    
    # Final refinement using SLSQP on best solution
    if time.time() - start_time < budget_s - 5:
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = normalize_h(refined, target_sum)
                final_c5 = compute_c5(refined, dx)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = refined
        except:
            pass
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
