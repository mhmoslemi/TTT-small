import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    
    # Precompute all shift indices
    lags = np.arange(-n_points + 1, n_points)
    
    # Try to solve with SDP if time allows
    if budget_s > 300:
        try:
            # Create CVXPY variables
            h_var = cp.Variable(n_points, name="h")
            C5_var = cp.Variable(name="C5")
            
            # Constraints
            constraints = [
                cp.sum(h_var) == target_sum,
                h_var >= 0,
                h_var <= 1,
                C5_var >= 0
            ]
            
            # Add overlap constraints for all shifts
            for k in lags:
                # Build the overlap term for shift k
                # This is the sum over i of h[i] * (1 - h[(i-k) % n_points])
                if k >= 0:
                    h_shift = cp.hstack([h_var[k:], h_var[:k]])
                else:
                    h_shift = cp.hstack([h_var[-k:], h_var[:-k]])
                
                expr = cp.sum(cp.multiply(h_var, 1 - h_shift)) * dx
                constraints.append(expr <= C5_var)
            
            # Objective
            objective = cp.Minimize(C5_var)
            prob = cp.Problem(objective, constraints)
            
            # Attempt to solve
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                c5_opt = C5_var.value
                if h_opt is not None and c5_opt is not None:
                    h_opt = np.clip(h_opt, 0, 1)
                    current_sum = np.sum(h_opt)
                    if current_sum > 0:
                        h_opt = h_opt * (target_sum / current_sum)
                    best_h = h_opt
                    best_c5 = c5_opt
                
                # Early exit if good enough
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
        except Exception as e:
            pass  # Fall back to local search if SDP fails
        
        # Local optimization if we have time left
        if time.time() - start_time < budget_s - 5:
            iterations = 0
            max_iter = 200
            
            while time.time() - start_time < budget_s - 5 and iterations < max_iter:
                # Local optimization using SLSQP
                def objective(h):
                    return np.max(correlate(h, 1 - h, mode="full")) * dx
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                # Use scipy to optimize
                from scipy.optimize import minimize
                result = minimize(objective, h.copy(), method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 50, 'ftol': 1e-6})
                
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = h_refined * (target_sum / np.sum(h_refined))
                    h = h_refined
                    
                current_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
                
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                if best_c5 <= 0.3807:
                    break
                
                # Random mutation with decreasing strength
                mutation_strength = max(0.001, 0.05 * np.exp(-iterations / 200.0))
                h += np.random.normal(0, mutation_strength, n_points)
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
                
                iterations += 1
    
    # Final evaluation
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
