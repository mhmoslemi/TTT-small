import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, dx, eps=1e-5):
    """Compute the gradient of C5 with respect to h using finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = normalize_h(h_plus, len(h) / 2)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = np.clip(h_minus, 0, 1)
        h_minus = normalize_h(h_minus, len(h) / 2)
        
        grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * eps)
    
    return grad

def backtracking_line_search(h, grad, dx, alpha_max=1.0, rho=0.5, c=1e-4):
    """Perform backtracking line search to find suitable step size."""
    alpha = alpha_max
    current_c5 = compute_c5(h, dx)
    
    for _ in range(50):
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, len(h) / 2)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 <= current_c5 - c * alpha * np.dot(grad, grad):
            return alpha
        alpha *= rho
    
    return alpha

        # Update h
