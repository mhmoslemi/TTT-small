import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, std_dev):
    """Add normally distributed noise and project back to feasible region."""
    h = h + np.random.normal(0, std_dev, len(h))
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available, otherwise use structured pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a structured pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    best_iter = 0
    
    # Multiple initialization strategies to avoid local minima
    initial_guesses = []
    
    # Start with parent solution or default
    if 'initial_h_values' in globals():
        initial_guesses.append(np.array(initial_h_values))
    else:
        # Uniform distribution
        initial_guesses.append(np.ones(n_points))
        
    # Add comb pattern
    comb = np.zeros(n_points)
    stride = max(2, n_points // 4)
    for i in range(0, n_points, stride):
        comb[i] = 1.0
    initial_guesses.append(comb)
    
    # Add random with constraint
    random_h = np.random.rand(n_points)
    initial_guesses.append(normalize_h(random_h, target_sum))

    # Alternate optimization with different strategies
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Try different initializations if we haven't improved in a while
        if iteration % 500 == 0 and iteration > 0:
            # Try new initialization
            init_idx = iteration // 500 % len(initial_guesses)
            h = initial_guesses[init_idx].copy()
            h = normalize_h(h, target_sum)
            
        # Optimization step
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Try to improve with SLSQP
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, 
                          options={'maxiter': 50, 'ftol': 1e-8})
        
        if result.success:
            h = result.x
        else:
            # If optimization failed, try gradient descent
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Apply systematic perturbations
        std_dev = max(0.001, 0.02 * np.exp(-iteration / 1000))
        h = perturb_h(h, std_dev)
        h = normalize_h(h, target_sum)
        
        # Periodic large perturbation
        if iteration % 100 == 0:
            h = perturb_h(h, 0.05)
            h = normalize_h(h, target_sum)
            
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_iter = iteration
            
        iteration += 1
        
        # Early stopping if we've reached target
        if best_c5 <= 0.3807:
            break
            
        # Occasionally restart near best solution
        if iteration % 1000 == 0:
            h = best_h.copy()
            h = perturb_h(h, 0.02)
            h = normalize_h(h, target_sum)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
