import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initial bounds for binary search
    C_low = 0.0
    C_high = 1.0
    best_c5 = float('inf')
    best_h = h.copy()

    # Binary search setup
    tolerance = 1e-4
    max_iterations = 100
    iterations = 0

    while (C_high - C_low > tolerance) and (time.time() - start_time < budget_s - 1) and iterations < max_iterations:
        iterations += 1
        C_mid = (C_low + C_high) / 2.0
        
        # Convex feasibility formulation
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Check all lags k from -n+1 to n-1 excluding 0
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
                
            # Calculate overlap constraint
            lag_overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lag_overlap += h_vars[i] * (1 - h_vars[j])
                
            constraints.append(lag_overlap <= C_mid / dx)

        # Feasibility problem
        objective = cp.Minimize(0)  # Just checking feasibility
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers
        solver = None
        for s in [cp.MOSEK, cp.ECOS]:
            try:
                prob.solve(solver=s, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    solver = s
                    break
            except:
                continue
        
        # If infeasible, update lower bound
        if solver is None:
            C_low = C_mid
        else:
            # Feasible: update upper bound and save better solution
            C_high = C_mid
            # Retrieve optimal solution
            h_opt = h_vars.value
            if h_opt is not None:
                h_opt = np.clip(h_opt, 0, 1)
                # Renormalize to ensure sum = n_points/2
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
                c5_val = compute_c5(h_opt)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()

        # Early exit if improving target
        if best_c5 <= 0.3807:
            break

    # Final refinement with local search
    h = best_h.copy()
    max_iter = 20000
    iteration = 0
    step_size = 0.005
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            # Clamp and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h

    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
