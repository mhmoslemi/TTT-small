import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    step_size = 0.05
    
    sa_steps = 0
    max_sa_steps = 100000
    
    # Local refinement parameters
    local_refinement_interval = 10000
    max_local_iter = 200
    
    while time.time() - start_time < budget_s - 1 and sa_steps < max_sa_steps:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        
        # Perturb h
        h_new = h.copy()
        indices_to_change = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices_to_change:
            h_new[i] += np.random.normal(0, step_size)
        
        # Ensure h_new is within bounds
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize to maintain sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h = h_new
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        sa_steps += 1
        
        # Temperature cooling
        if sa_steps % 100 == 0:
            temp *= cooling_rate
            
        # Local refinement every 10k steps
        if sa_steps % local_refinement_interval == 0:
            # Use SLSQP for local refinement
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': max_local_iter})
            
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = normalize_h(h_refined, target_sum)
                h = h_refined.copy()
                
                # Update best if refined version is better
                refined_c5 = compute_c5(h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
