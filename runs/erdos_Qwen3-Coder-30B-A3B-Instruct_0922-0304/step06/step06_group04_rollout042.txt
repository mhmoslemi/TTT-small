import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient-based optimization with line search
    def loss_and_gradient(h):
        # Compute overlap
        overlap = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap)
        max_val = overlap[max_idx]
        
        # Compute gradient of the max term
        grad = np.zeros_like(h)
        # Gradient of correlation h * (1-h) at max index
        # Run forward convolution of (1-h) on h with full mode
        conv = correlate(h, 1 - h, mode="full")
        # The gradient is essentially the derivative of each term w.r.t. h
        # For simplicity using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            grad[i] = (overlap_plus[max_idx] - max_val) / eps
        
        return max_val * dx, grad
    
    # Simple line search backtracking
    def line_search(h, grad, alpha=0.1, beta=0.9):
        while True:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            new_val, _ = loss_and_gradient(h_new)
            if new_val <= loss_and_gradient(h)[0] or alpha < 1e-6:
                return h_new, alpha
            alpha *= beta
            
    # Optimization loop
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        h = best_individual.copy()
        
        # Random perturbation
        noise = np.random.normal(0, 0.02, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Gradient descent loop
        local_iter = 0
        while local_iter < 10 and time.time() - start_time < budget_s - 1:
            val, grad = loss_and_gradient(h)
            if np.linalg.norm(grad) < 1e-6:
                break
            h_new, alpha = line_search(h, grad)
            h = h_new
            local_iter += 1
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
