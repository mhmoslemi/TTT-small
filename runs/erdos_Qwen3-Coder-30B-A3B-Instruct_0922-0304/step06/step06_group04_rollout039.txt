import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
    
    # Normalize
    current_sum = np.sum(base_h)
    if current_sum > 0:
        base_h = base_h * (target_sum / current_sum)
        
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Create a better basis using a more structured approach
        h = base_h.copy()
        
        # Add structured variation
        h += 0.02 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        h = h * (target_sum / np.sum(h))
        
        # Direct optimization using Bilinear Programming
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
        
        # Compute overlap constraints and objective
        c5_expr = 0
        # Just check few key shifts to make it faster
        shifts = range(1, n_points//2)
        for k in shifts:
            expr = 0
            # This could be vectorized but keeping simple
            for i in range(n_points - k):
                expr += h_var[i] * (1 - h_var[i + k])
            expr *= dx
            if c5_expr == 0:
                c5_expr = expr
            else:
                c5_expr = cp.maximum(c5_expr, expr)
                
        # For full coverage, recompute manually
        objective = cp.Minimize(c5_expr)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                opt_h = h_var.value
                if opt_h is not None:
                    opt_h = np.clip(opt_h, 0, 1)
                    # Renormalize
                    opt_h = opt_h * (target_sum / np.sum(opt_h))
                    c5 = compute_c5(opt_h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = opt_h.copy()
        except:
            pass
        
        if best_c5 <= 0.3807:
            break
            
        iter_count += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
