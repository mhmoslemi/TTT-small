import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    pop_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h)
        for _ in range(pop_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)

    best_c5 = float('inf')
    best_individual = None
    
    gen = 0
    stagnation_count = 0
    mutation_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        elite_size = pop_size // 2
        selected = population[:elite_size]
        
        # Crossover
        new_pop = selected.copy()
        while len(new_pop) < pop_size:
            i, j = np.random.choice(elite_size, 2, replace=False)
            parent1, parent2 = selected[i], selected[j]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Apply to new pop
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        # Mutation
        for i in range(len(new_pop)):
            if np.random.rand() < mutation_rate:
                h = new_pop[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_pop[i] = h
                
        # Elitism: keep best
        new_pop[0] = selected[0]
        population = new_pop
        
        # Adjust mutation rate
        if stagnation_count > 20:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif stagnation_count > 10:
            mutation_rate = min(0.3, mutation_rate * 1.05)
            
        gen += 1
        
        # Early termination check
        if gen > 100:
            break
            
    if best_individual is None:
        h = population[0]
        c5 = evaluate_individual(h, dx)
        best_individual = h
        best_c5 = c5
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
