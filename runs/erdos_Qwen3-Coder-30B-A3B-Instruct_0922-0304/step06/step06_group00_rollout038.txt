import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    if diff > 0:
        # Need to increase values
        indices = np.argsort(h)
        for i in range(len(h)):
            if diff <= 0:
                break
            max_increase = 1 - h[indices[i]]
            increase = min(max_increase, diff)
            h[indices[i]] += increase
            diff -= increase
    else:
        # Need to decrease values
        indices = np.argsort(-h)
        for i in range(len(h)):
            if diff >= 0:
                break
            max_decrease = h[indices[i]]
            decrease = max(-max_decrease, diff)
            h[indices[i]] += decrease
            diff -= decrease
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Use a piecewise constant function
        h[:n_points//3] = 0.5
        h[n_points//3:2*n_points//3] = 1.0
        h[2*n_points//3:] = 0.0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    # Create initial particles
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle_h = np.array(initial_h_values) + np.random.normal(0, 0.01, n_points)
        else:
            particle_h = h + np.random.normal(0, 0.02, n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = normalize_h(particle_h, target_sum)
        particles.append(particle_h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        pbest.append(particle_h.copy())
        pbest_values.append(compute_c5(particle_h, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_values)
    global_best = pbest[global_best_idx].copy()
    global_best_value = pbest_values[global_best_idx]
    
    iterations = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * iterations / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce bounds and constraints
            particles[i] = project_h(particles[i], target_sum)
            
            # Evaluate particle
            current_value = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_value < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_value
        
        # Early termination
        if global_best_value <= 0.3807:
            break
            
        iterations += 1
    
    # Final check
    final_c5 = compute_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
