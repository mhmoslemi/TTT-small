import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for i in range(n_particles):
        # Initialize with random Fourier coefficients
        coeff = np.random.randn(n_freq * 2)
        particles.append(coeff)
        velocities.append(np.zeros_like(coeff))
        pbest.append(coeff.copy())
        pbest_c5.append(float('inf'))
    
    gbest = None
    gbest_c5 = float('inf')
    
    # Helper function to convert Fourier coefficients to h
    def coeffs_to_h(coeffs):
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_freq):
            h += coeffs[i] * np.cos(i * np.pi * x) + coeffs[i + n_freq] * np.sin(i * np.pi * x)
        return np.clip(h, 0, 1)
        
    eval_count = 0
    
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Convert to h
            h = coeffs_to_h(particles[i])
            h = normalize_h(h, target_sum)
            
            # Evaluate C5
            c5 = compute_c5(h, dx)
            eval_count += 1
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = particles[i].copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest = particles[i].copy()
                
            # Update global best for return
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            if gbest is not None:
                velocities[i] = (w * velocities[i] + 
                               c1 * r1 * (pbest[i] - particles[i]) + 
                               c2 * r2 * (gbest - particles[i]))
                particles[i] += velocities[i]
                
    # Post optimization with local refinement
    if gbest is not None:
        # Convert to best h from PSO
        h_final = coeffs_to_h(gbest)
        h_final = normalize_h(h_final, target_sum)
        
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h_final.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            
        c5_refined = compute_c5(h_final, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_final.copy()
    
    return (best_h.tolist(), best_c5, n_points)
