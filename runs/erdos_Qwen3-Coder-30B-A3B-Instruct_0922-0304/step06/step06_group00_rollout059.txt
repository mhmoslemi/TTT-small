import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a base function with structure
        h = np.ones(n_points) * 0.5
        h[:n_points//3] = 0.8
        h[n_points//3:2*n_points//3] = 0.2
        # Add slight randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iterations = 0
    max_iter = 3000
    
    # Gradient-based optimization with adaptive step
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Compute gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            grad[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * epsilon)
        
        # Update step
        step_size = 0.01 / (1.0 + 0.001 * iterations)  # Adaptive learning rate
        h_new = h - step_size * grad
        
        # Ensure bounds and constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        current_c5 = compute_c5(h_new, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            
            # Early termination if target achieved
            if best_c5 <= 0.3807:
                break
        
        h = h_new
        
        iterations += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
