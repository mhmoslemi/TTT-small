import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with specified spacing and tooth height."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    h[tooth_positions] = tooth_height
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbations to a pattern."""
    perturbation = np.random.normal(0, sigma, len(h))
    return np.clip(h + perturbation, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Load initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a basic comb pattern
        h = create_comb_pattern(n_points, 3, 1.0)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define comb spacings to try
    spacings = [2, 3, 4, 5, 6]
    
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Use comb patterns to generate candidates
        if iteration % 10 == 0:
            candidates = []
            for s in spacings:
                comb = create_comb_pattern(n_points, s)
                comb = normalize_h(comb, target_sum)
                candidates.append(comb)
                
                # Add perturbed versions
                for _ in range(2):
                    perturbed = perturb_pattern(comb, 0.02)
                    perturbed = normalize_h(perturbed, target_sum)
                    candidates.append(perturbed)
            
            # Optimize each candidate using SLSQP
            for candidate in candidates:
                def objective(h):
                    return compute_c5(h, dx)
                    
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, candidate, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    optimized = result.x
                else:
                    optimized = candidate
                    
                optimized = np.clip(optimized, 0, 1)
                optimized = normalize_h(optimized, target_sum)
                
                c5_opt = compute_c5(optimized, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = optimized.copy()
        
        # Try a small random perturbation
        h = h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        iteration += 1
    
    # Final optimization using SLSQP on best known solution
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    if result.success:
        best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
