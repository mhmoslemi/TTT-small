import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Parameters
    T = 1.0
    cooling_rate = 0.99
    min_temp = 1e-5
    stagnation_count = 0
    stagnation_limit = 500
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Make a random perturbation
        h_candidate = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure we don't go out of bounds
        if h_candidate[i] < 1 and h_candidate[j] > 0:
            delta = min(1 - h_candidate[i], h_candidate[j])
            delta *= np.random.rand()  # Random portion
            h_candidate[i] += delta
            h_candidate[j] -= delta
        
        # Renormalize
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Compute new cost
        c5_candidate = compute_c5(h_candidate)
        
        # Metropolis acceptance
        delta_c5 = c5_candidate - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Cooling schedule
        T *= cooling_rate
        
        # Restart if stuck
        if stagnation_count >= stagnation_limit:
            T *= 1.2
            stagnation_count = 0
            
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
