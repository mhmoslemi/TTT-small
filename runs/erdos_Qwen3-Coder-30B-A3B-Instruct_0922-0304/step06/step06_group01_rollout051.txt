import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Start with a simple rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to desired sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15
    max_iter = 500
    w, c1, c2 = 0.7, 1.5, 1.5
    
    # Initialize particles with random Fourier coefficients
    particles = []
    velocities = []
    pbest = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Random Fourier coefficients
        a = np.random.uniform(-1, 1, n_freq)
        b = np.random.uniform(-1, 1, n_freq)
        particles.append((a, b))
        velocities.append((np.zeros(n_freq), np.zeros(n_freq)))
        pbest.append((a.copy(), b.copy()))
        pbest_values.append(float('inf'))
    
    # Global best
    gbest = (np.zeros(n_freq), np.zeros(n_freq))
    gbest_value = float('inf')
    
    def reconstruct_h(a, b):
        x = np.linspace(0, 2, n_points)
        h = np.ones(n_points) * a[0]  # DC component
        for k in range(1, n_freq):
            h += a[k] * np.cos(k * np.pi * x) + b[k] * np.sin(k * np.pi * x)
        return np.clip(h, 0, 1)
    
    def eval_h(a, b):
        h = reconstruct_h(a, b)
        h = h * (target_sum / np.sum(h))  # normalize
        return compute_c5(h, dx)
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.8:
            break
            
        for i in range(n_particles):
            # Get current particle
            a, b = particles[i]
            
            # Evaluate current particle
            current_c5 = eval_h(a, b)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = (a.copy(), b.copy())
                pbest_values[i] = current_c5
            
            # Update global best
            if current_c5 < gbest_value:
                gbest = (a.copy(), b.copy())
                gbest_value = current_c5
                
            # Update best solution found
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = reconstruct_h(a, b)
                best_h = best_h * (target_sum / np.sum(best_h))
                
            # Early exit condition
            if best_c5 <= 0.3807:
                break
                
            # Update velocities
            r1, r2 = np.random.rand(2)
            va = w * velocities[i][0] + c1 * r1 * (pbest[i][0] - a) + c2 * r2 * (gbest[0] - a)
            vb = w * velocities[i][1] + c1 * r1 * (pbest[i][1] - b) + c2 * r2 * (gbest[1] - b)
            
            # Update positions
            new_a = a + va
            new_b = b + vb
            particles[i] = (new_a, new_b)
            velocities[i] = (va, vb)
            
        # Terminate if early condition met
        if best_c5 <= 0.3807:
            break
    
    # Refine the best solution with local optimization
    if time.time() - start_time < budget_s * 0.95:
        # Recreate better solution from gbest
        h_refined = reconstruct_h(gbest[0], gbest[1])
        h_refined = h_refined * (target_sum / np.sum(h_refined))
        
        # SLSQP local optimization with constraints
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_refined = np.clip(result.x, 0, 1)
            h_refined = h_refined * (target_sum / np.sum(h_refined))
            refined_c5 = compute_c5(h_refined, dx)
            if refined_c5 < best_c5:
                best_c5 = refined_c5
                best_h = h_refined
    
    # Final check of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
