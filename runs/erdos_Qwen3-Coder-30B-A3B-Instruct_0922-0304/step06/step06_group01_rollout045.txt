import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use CMA-ES approach
    iteration = 0
    max_iter = 500
    
    # CMA-ES parameters
    popsize = 2 * n_points
    lambda_ = popsize // 2
    sigma = 0.1
    
    # Initialize CMA-ES
    mean = h.copy()
    C = np.eye(n_points)
    z = np.zeros(n_points)
    chi = np.sqrt(n_points)
    weights = np.log(popsize/2 + 1) - np.log(np.arange(1, lambda_ + 1))
    weights = weights / np.sum(weights)
    mu = lambda_
    
    # Evolution paths
    pc = np.zeros(n_points)
    ps = np.zeros(n_points)
    c1 = 2 / (n_points + 2)
    cmu = 2 * (weights[:mu].sum() ** 2) / (n_points + 2)
    damps = 1 + 2 * max(0, np.sqrt((mu - 1) / (n_points + 1)) - 1) + 0.3
    cmean = 1
    
    while time.time() - start_time < budget_s - 2 and iteration < max_iter:
        # Sample population
        population = []
        for i in range(popsize):
            z = np.random.randn(n_points)
            x = mean + sigma * z @ np.linalg.cholesky(C)
            x = np.clip(x, 0, 1)
            x = normalize_h(x, target_sum)
            population.append(x)
        
        # Evaluate population
        fitness = [compute_c5(x, dx) for x in population]
        
        # Sort by fitness
        idx = np.argsort(fitness)
        best_idx = idx[0]
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Update evolution paths
        y = population[idx[0]] - mean
        ps = (1 - c1) * ps + np.sqrt(c1 * (2 - c1)) * (y / sigma)
        pc = (1 - cmu) * pc + np.sqrt(cmu * (2 - cmu)) * (y @ np.linalg.inv(C) / sigma)
        
        # Update covariance matrix
        C = (1 - c1 - cmu) * C + c1 * (pc @ pc.T) + cmu * np.sum([weights[i] * (population[idx[i]] - mean) @ (population[idx[i]] - mean).T for i in range(lambda_)], axis=0) / (sigma**2)
        
        # Update mean
        mean = mean + cmean * sigma * (population[idx[0]] - mean)
        
        # Update step size
        sigma = sigma * np.exp((np.linalg.norm(ps) / chi - 1) / (damps + 1))
        
        iteration += 1
    
    # Final polishing with SLSQP
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
