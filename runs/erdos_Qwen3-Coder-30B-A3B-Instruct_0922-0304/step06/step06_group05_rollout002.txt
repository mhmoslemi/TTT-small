import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h, n_points, target_sum):
        """Project h to ensure sum equals target_sum and values in [0,1]"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        # Scale to meet target sum
        if current_sum > 0:
            h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        return h

    def compute_gradient(h, k_star, n_points, dx):
        """Compute gradient analytically based on k_star"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at lag k_star, gradient contribution
            j = (i + k_star) % n_points
            g[i] += (1 - h[j]) * dx
            # Note: negative term would be from the conjugate, handled in correlation
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    step_decay = 0.5
    perturb_freq = 500
    perturb_std = 0.01
    perturb_decay = 0.9

    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

        # Compute gradient
        g = compute_gradient(h, max_k, n_points, dx)
        
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Project back to feasible set
        new_h = project_h(new_h, n_points, n_points / 2)
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % perturb_freq == 0:
            pert = np.random.normal(0, perturb_std, n_points)
            h = h + pert
            h = project_h(h, n_points, n_points / 2)
            perturb_std *= perturb_decay
            
        # Adaptive step size
        if iteration % 1000 == 0 and step_size > 1e-6:
            step_size *= step_decay

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
