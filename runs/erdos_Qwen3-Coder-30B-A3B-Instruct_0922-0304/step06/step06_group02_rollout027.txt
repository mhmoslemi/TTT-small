import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Alternate using different optimization strategies
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try multiple strategies
        strategy = iteration % 3
        
        if strategy == 0:  # L-BFGS-B with noise
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
                
            h_noisy = h + np.random.normal(0, 0.01, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            try:
                result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h_candidate = result.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = normalize_h(h_candidate, target_sum)
                    current_c5 = compute_c5(h_candidate, dx)
                    h = h_candidate.copy()
            except:
                pass
                
        elif strategy == 1:  # Population-based with adaptive noise
            # Make a small stochastic perturbation
            perturb_strength = 0.02 / (1 + iteration / 1000)
            noise = np.random.normal(0, perturb_strength, len(h))
            h = np.clip(h + noise, 0, 1)
            h = normalize_h(h, target_sum)
            
        else:  # Large perturbation if stuck
            if iteration % 100 == 0:
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
