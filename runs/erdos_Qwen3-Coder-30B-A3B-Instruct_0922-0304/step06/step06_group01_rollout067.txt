import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def build_comb_h(n_points, n_teeth, tooth_width, spacing, slope):
    """Build a comb-like h function with specified parameters."""
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    # Ensure tooth_width is not too large
    tooth_width = min(tooth_width, 2.0 / n_teeth - spacing * 0.1)
    
    # Calculate positions of teeth
    for i in range(n_teeth):
        tooth_start = i * (tooth_width + spacing)
        tooth_end = tooth_start + tooth_width
        
        # Determine indices for the tooth
        idx_start = int(tooth_start / dx)
        idx_end = int(tooth_end / dx) + 1
        idx_end = min(idx_end, n_points)
        
        # Handle transitions with triangular shape
        for j in range(idx_start, idx_end):
            x_pos = j * dx
            if x_pos < tooth_start:
                # Left transition
                if tooth_start - x_pos <= slope:
                    h[j] = max(0, 1 - (tooth_start - x_pos) / slope)
            elif x_pos > tooth_end:
                # Right transition
                if x_pos - tooth_end <= slope:
                    h[j] = max(0, 1 - (x_pos - tooth_end) / slope)
            else:
                # Inside the tooth
                h[j] = 1.0
    
    return h

def adjust_h_sum(h, target_sum, dx):
    """Adjust h to have the correct sum."""
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = adjust_h_sum(h, target_sum, dx)
    else:
        # Default construction: slightly different from the previous one
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = adjust_h_sum(h, target_sum, dx)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define parameter search space for comb structure
    n_teeth_vals = range(5, 12)
    tooth_width_vals = [2.0 / n for n in n_teeth_vals]
    
    # Parameters to search
    param_space = []
    for n_t in n_teeth_vals:
        tooth_w = 2.0 / n_t
        spacing = (2.0 - n_t * tooth_w) / (n_t + 1)
        if spacing > 0:
            for slope_factor in [0.25, 0.5, 1.0, 2.0]:
                slope = slope_factor * tooth_w / 2.0
                param_space.append((n_t, tooth_w, spacing, slope))
    
    # Evaluate top candidates
    n_eval = min(500, len(param_space))
    for i in range(n_eval):
        if time.time() - start_time > budget_s - 2:
            break
            
        n_t, tooth_w, spacing, slope = param_space[i]
        
        # Build comb structure
        h_test = build_comb_h(n_points, n_t, tooth_w, spacing, slope)
        h_test = adjust_h_sum(h_test, target_sum, dx)
        h_test = np.clip(h_test, 0, 1)
        
        c5_test = compute_c5(h_test, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test.copy()
        
        # Also try using SLSQP for optimization of parameters
        def objective(params):
            n_t, tooth_w, spacing, slope = params
            tooth_w = max(0.01, min(1.0, tooth_w))
            spacing = max(0.001, min(0.5, spacing))
            slope = max(0.001, min(0.5, slope))
            try:
                h_comb = build_comb_h(n_points, int(n_t), tooth_w, spacing, slope)
                h_comb = adjust_h_sum(h_comb, target_sum, dx)
                h_comb = np.clip(h_comb, 0, 1)
                return compute_c5(h_comb, dx)
            except:
                return float('inf')

        bounds = [(5, 10), (0.01, 0.5), (0.001, 0.5), (0.001, 0.5)]
        init_params = [n_t, tooth_w, spacing, slope]
        try:
            result = minimize(objective, init_params, bounds=bounds, method='L-BFGS-B')
            if result.success and result.fun < best_c5:
                n_t_opt, tooth_w_opt, spacing_opt, slope_opt = result.x
                h_opt = build_comb_h(n_points, int(n_t_opt), tooth_w_opt, spacing_opt, slope_opt)
                h_opt = adjust_h_sum(h_opt, target_sum, dx)
                h_opt = np.clip(h_opt, 0, 1)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass

    # Final local refinement with SLSQP
    if time.time() - start_time < budget_s - 2:
        # Refine using SLSQP on the best comb structure
        def ref_objective(h):
            h = np.clip(h, 0, 1)
            h = adjust_h_sum(h, target_sum, dx)
            return compute_c5(h, dx)
        
        # Constraint using a simple trick to reduce memory usage
        h_refine = best_h.copy()
        bounds = [(0, 1) for _ in range(n_points)]
        
        def sum_constraint(h):
            return np.sum(h) * dx - 1.0
        
        constraint = {'type': 'eq', 'fun': sum_constraint}
        
        try:
            result = minimize(ref_objective, h_refine, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 500})
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_h = adjust_h_sum(refined_h, target_sum, dx)
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
