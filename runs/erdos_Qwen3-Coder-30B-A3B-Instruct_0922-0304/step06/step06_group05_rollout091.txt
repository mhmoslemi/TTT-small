import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Add small random noise
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    T = 1.0  # Initial temperature
    alpha = 0.99  # Cooling rate
    min_T = 1e-5  # Minimum temperature
    stagnation_count = 0
    max_stagnation = 5000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Create candidate solution by perturbing h
        h_candidate = h.copy()
        
        # Choose two indices to adjust
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure we can transfer mass
        if h_candidate[i] < 1 and h_candidate[j] > 0:
            # Amount to transfer
            delta = min(1 - h_candidate[i], h_candidate[j])
            delta *= np.random.rand()  # Random fraction
            
            # Apply transfer
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            c5_candidate = compute_c5(h_candidate)
            
            # Metropolis acceptance
            delta_c5 = c5_candidate - best_c5
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                h = h_candidate.copy()
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        if T > min_T:
            T *= alpha
            
        # Escape stagnation
        if stagnation_count > max_stagnation:
            T *= 1.2
            stagnation_count = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
