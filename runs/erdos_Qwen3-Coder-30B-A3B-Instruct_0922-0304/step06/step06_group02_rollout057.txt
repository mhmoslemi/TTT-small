import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with local search
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use gradient-based local search with restarts
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
            
        # Try multiple restarts for better exploration
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        
        for _ in range(5):
            # Use a more sophisticated noise injection
            noise = np.random.normal(0, 0.03, len(h))
            h_noisy = h + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            # Use L-BFGS-B optimization to minimize C5
            try:
                result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, options={'maxiter': 200})

                    h_candidate = result.x

                    h_candidate = normalize_h(h_candidate, target_sum)
                    candidate_c5 = compute_c5(h_candidate, dx)
                    if candidate_c5 < best_candidate_c5:
                        best_candidate = h_candidate.copy()
                        best_candidate_c5 = candidate_c5
            except:
                continue
                
        # Accept improvement
        if best_candidate_c5 < current_c5:
            h = best_candidate.copy()
        
        # Periodic global perturbation to escape local minima
        if iteration % 10 == 0:
            # Apply random perturbation
            perturbation = np.random.normal(0, 0.02, len(h))
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
