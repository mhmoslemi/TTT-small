import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    # Create initial population with noise
    for i in range(population_size):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Genetic algorithm
    generation = 0
    stagnant = 0
    max_stagnant = 20
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            stagnant = 0
        else:
            stagnant += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected_size = population_size // 2
        selected = population[:selected_size]
        
        # Crossover
        new_population = selected.copy()
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                parent1 = selected[i]
                parent2 = selected[i+1]
                # Single point crossover
                crossover_point = np.random.randint(1, n_points-1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                new_population.extend([offspring1, offspring2])
            else:
                new_population.append(selected[i])
        
        # Mutation
        mutation_rate = 0.2
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                h = new_population[i]
                h = h + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
        
        # Elitism: keep best individual
        new_population[0] = best_individual.copy()
        
        # Maintain population size
        population = new_population[:population_size]
        
        # Adjust hyperparameters based on progress
        if stagnant > max_stagnant:
            if mutation_rate > 0.1:
                mutation_rate = 0.1
            if population_size < 40:
                population_size = 40
        elif stagnant == 0 and len(new_population) > 20:
            if population_size > 20:
                population_size = 20
        
        generation += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
