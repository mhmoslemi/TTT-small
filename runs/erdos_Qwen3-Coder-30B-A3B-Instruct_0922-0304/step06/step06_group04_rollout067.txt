import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Main optimization loop using gradient-like descent with projection
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Slightly perturb current best solution
        h = best_individual.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Simple local search by gradient descent with projections
        step_size = 0.01
        for _ in range(20):  # few gradient steps
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            base_c5 = compute_c5(h, dx)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = normalize_h(h_plus, target_sum)
                c5_plus = compute_c5(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
                
            # Update with gradient descent
            h = h - step_size * grad
            
            # Project onto feasible set: clamp and renormalize
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Early stop if improving
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
