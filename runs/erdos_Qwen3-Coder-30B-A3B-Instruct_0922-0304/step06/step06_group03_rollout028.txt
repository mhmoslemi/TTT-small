import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_h(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

    guesses.append(h_comb.copy())
    
    # 3. Random with normalization
    h_rand = np.random.rand(n_points)
    h_rand = normalize_h(h_rand, target_sum)

    h_alt = np.zeros(n_points)

    guesses.append(h_wave.copy())
