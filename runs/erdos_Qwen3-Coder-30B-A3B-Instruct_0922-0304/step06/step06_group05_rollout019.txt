import time
from scipy.signal import correlate

    start_time = time.time()

    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:

        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initial configuration
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    # Keep track of the best solution
    best_c5 = current_c5
    best_h = current_h.copy()
    
    # Main loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Make a small perturbation
        candidate_h = current_h.copy()
        
        # Select two indices to modify
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure i != j and we don't go outside bounds
        if i == j:
            continue
            
        # Compute delta
        delta = min(1 - candidate_h[i], candidate_h[j])
        if delta <= 1e-8:
            continue
            
        # Apply perturbation
        delta = np.random.uniform(0, delta)
        candidate_h[i] += delta 
        candidate_h[j] -= delta
        
        # Renormalize
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        candidate_h = np.clip(candidate_h, 0, 1)
        
        # Compute new C5
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject based on Metropolis criterion
        if candidate_c5 < current_c5:
            current_h = candidate_h
            current_c5 = candidate_c5
            stagnation_count = 0
        else:
            # Accept with probability exp(-Δ/T)
            delta_c5 = candidate_c5 - current_c5
            if np.random.rand() < np.exp(-delta_c5 / T):
                current_h = candidate_h
                current_c5 = candidate_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            stagnation_count = 0
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update temperature
        T *= alpha
        
        # Recovery from stagnation
        if stagnation_count > max_stagnation:
            T = min(T * 1.2, 1.0)
            stagnation_count = 0
        
        # Occasionally reinitialize if too slow
        if iter_count % 5000 == 0 and T < 0.01:
            # Restart with a random good solution
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            current_h = h.copy()
            current_c5 = compute_c5(current_h)
            T = 1.0
            stagnation_count = 0
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
