import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5
        h = h * (target_sum / np.sum(h))
    
    # Alternating optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Store current h and compute overlaps
        h_prev = h.copy()
        
        # Step 1: Optimize h while treating overlaps as fixed
        # Define constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Objective: minimize max overlap
        def objective(h):
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx
        
        # Try local optimization
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-6})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))
        except:
            pass
            
        # Step 2: Update overlaps
        c5_new = compute_c5(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Perturb slightly
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
