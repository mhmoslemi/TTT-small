import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, rate=0.7):
    """Crossover two parents to create offspring."""
    if np.random.rand() > rate:
        return parent1.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    return child

def mutate(h, rate=0.05):
    """Mutate the solution."""
    # Add small random perturbations
    noise = np.random.normal(0, 0.02, len(h))
    h_mut = h + noise
    return np.clip(h_mut, 0, 1)

def evaluate_population(population, dx):
    """Evaluate all candidates in population."""
    fitness = []
    for h in population:
        f = compute_c5(h, dx)
        fitness.append(f)
    return np.array(fitness)
