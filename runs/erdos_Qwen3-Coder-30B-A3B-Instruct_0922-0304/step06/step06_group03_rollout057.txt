import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO particles
    n_particles = 50
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            pH = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
            pH = np.clip(pH, 0, 1)
            pH = normalize_h(pH, target_sum)
        else:
            pH = h + np.random.normal(0, 0.05, n_points)
            pH = np.clip(pH, 0, 1)
            pH = normalize_h(pH, target_sum)
        particles.append(pH)
        velocities.append(np.zeros(n_points))
    
    # Track personal bests
    p_best = [p.copy() for p in particles]
    p_best_scores = [compute_c5(p, dx) for p in particles]
    
    # Track global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # PSO step
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            score = compute_c5(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                
                # Update global best
                if score < g_best_score:
                    g_best_score = score
                    g_best = particles[i].copy()
            
            # Early termination
            if g_best_score <= 0.3807:
                break
        
        # Periodic refinement with gradient-based method
        if iteration % 100 == 0 and iteration > 0:
            # Select top particles based on score
            sorted_indices = np.argsort(p_best_scores)
            top_particles = [p_best[i] for i in sorted_indices[:10]]
            
            for p in top_particles:
                # Apply SLSQP refinement
                def objective(h_p):
                    # Flatten the particle vector for optimization
                    return compute_c5(h_p, dx)
                    
                def constraint_sum(h_p):
                    return np.sum(h_p) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, p, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined_p = result.x
                    refined_p = np.clip(refined_p, 0, 1)
                    refined_p = normalize_h(refined_p, target_sum)
                    new_score = compute_c5(refined_p, dx)
                    
                    # Update if better
                    if new_score < g_best_score:
                        g_best_score = new_score
                        g_best = refined_p.copy()
                        
        # Periodic frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            # Apply high-pass filter in frequency domain
            for i in range(n_particles):
                # Transform to frequency domain
                fft_h = np.fft.fft(particles[i])
                freqs = np.fft.fftfreq(n_points, dx)
                
                # Apply high-pass filter (keeping high frequencies)
                cutoff_freq = 0.3
                high_pass_mask = np.abs(freqs) > cutoff_freq
                fft_h_filtered = fft_h * high_pass_mask
                
                # Transform back
                filtered_h = np.real(np.fft.ifft(fft_h_filtered))
                
                # Renormalize and clip
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                
                # Replace worst performers (bottom 25%)
                if np.random.rand() < 0.5:
                    worst_idx = np.argmax([compute_c5(p, dx) for p in particles])
                    particles[worst_idx] = filtered_h.copy()
                    # Update personal best if more recent is worse
                    score = compute_c5(filtered_h, dx)
                    if score < g_best_score:
                        g_best_score = score
                        g_best = filtered_h.copy()
                        
        iteration += 1
        
        if iteration % 500 == 0:
            print(f"Iteration {iteration}, Best C5: {g_best_score:.6f}")
    
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
