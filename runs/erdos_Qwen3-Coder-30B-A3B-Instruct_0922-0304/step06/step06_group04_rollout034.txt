import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    stagnation_count = 0
    max_stagnation = 20
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Check for stagnation
        if stagnation_count >= max_stagnation:
            # Increase population size if stagnation
            if len(population) < 40:
                population_size = 40
                # Fill with random individuals
                for _ in range(20):
                    h = np.random.rand(n_points)
                    h = normalize_h(h, target_sum)
                    population.append(h.copy())
            else:
                # Try aggressive mutation
                for i in range(len(population)):
                    if i < 5:  # Top 5 elite - keep unchanged
                        continue
                    h = population[i] + np.random.normal(0, 0.1, n_points)
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    population[i] = h
        
        # Selection - top 50%
        selected = population[:population_size//2]
        
        # Crossover
        new_population = selected.copy()
        while len(new_population) < population_size:
            i, j = np.random.choice(len(selected), 2, replace=False)
            parent1, parent2 = selected[i], selected[j]
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation
            if np.random.rand() < 0.2:
                child1 += np.random.normal(0, 0.05, n_points)
                child1 = np.clip(child1, 0, 1)
                child1 = normalize_h(child1, target_sum)
            
            if np.random.rand() < 0.2:
                child2 += np.random.normal(0, 0.05, n_points)
                child2 = np.clip(child2, 0, 1)
                child2 = normalize_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Elitism: keep best from previous pop
        if best_individual is not None:
            best_individual_copy = best_individual.copy()
            new_population[-1] = best_individual_copy
            
        population = new_population
        
    if best_individual is None:
        # Fall back to best from initial population
        fitness = [evaluate_individual(h, dx) for h in population]
        best_idx = np.argmin(fitness)
        best_individual = population[best_idx]
        best_c5 = fitness[best_idx]
    
    final_c5 = evaluate_individual(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
