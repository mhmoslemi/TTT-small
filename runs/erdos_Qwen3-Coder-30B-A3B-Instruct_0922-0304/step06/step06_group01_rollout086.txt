import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a more sophisticated starting point: step function with gradual transition
        h = np.zeros(n_points)
        mid = n_points // 2
        # Start with a rectangular step
        h[:mid] = 1.0
        # Add a smooth transition at the boundary to allow gradient-based optimization
        transition_width = 5
        for i in range(transition_width):
            if mid - transition_width + i >= 0:
                h[mid - transition_width + i] = (i / transition_width)
            if mid + i < n_points:
                h[mid + i] = 1.0 - (i / transition_width)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    
    # Local refinement step
    local_refine_step = 10000
    
    iteration = 0
    max_iter = int(budget_s * 100)  # Cap iterations based on time
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Perturb h: adjust ~10% of values
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.02)
            h_new[i] += perturbation
        
        # Clamp values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum to target_sum
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < math.exp(-delta / temp):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Cooling schedule
        if iteration % 100 == 0:
            temp *= cooling_rate
            if temp < min_temp:
                temp = min_temp
        
        # Local refinement every local_refine_step iterations
        if iteration % local_refine_step == 0 and iteration > 0:
            # Use SLSQP for local refinement
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
                
            def objective(h_vec):
                return compute_c5(h_vec, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h = result.x.copy()
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    current_c5 = compute_c5(h, dx)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
            except:
                pass  # If SLSQP fails, continue with SA
            
        iteration += 1
        
        # Early termination condition
        if best_c5 <= 0.3807:
            break
    
    # Final local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
        
    def objective(h_vec):
        return compute_c5(h_vec, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        if result.success:
            final_h = result.x.copy()
            final_h = np.clip(final_h, 0, 1)
            final_h = normalize_h(final_h, target_sum)
            final_c5 = compute_c5(final_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    except:
        pass  # If optimization fails, use best_h from the main loop
    
    return (best_h.tolist(), best_c5, n_points)
