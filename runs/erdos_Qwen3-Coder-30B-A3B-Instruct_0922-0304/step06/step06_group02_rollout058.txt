```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from cvxpy import *

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement
    iteration = 0
    max_iter = 30000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Step size for local search
        step_size = 0.01
        
        # Gradient-based local search (adaptive step)
        h_new = h.copy()
        h_new += np.random.normal(0, step_size, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5 * 0.99999:
            h = h_new.copy()
            
        # Occasionally perform global smoothing
        if iteration % 20 == 0:
            # Apply simple smoothing filter
            h_smooth = np.convolve(h, np.ones(5)/5, mode='same')
            h_smooth = np.clip(h_smooth, 0, 1)
            h_smooth = normalize_h(h_smooth, target_sum)
            smoothed_c5 = compute_c5(h_smooth, dx)
            if smoothed_c5 < current_c5:
                h = h_smooth.copy()
                
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
