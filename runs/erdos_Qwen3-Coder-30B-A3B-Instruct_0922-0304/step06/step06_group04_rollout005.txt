import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_population(population, dx):
    return [compute_c5(individual, dx) for individual in population]

def select_parents(population, fitnesses, num_parents):
    selected_indices = np.argsort(fitnesses)[:num_parents]
    return [population[i] for i in selected_indices]

def crossover(parent1, parent2, crossover_point):
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, sigma):
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, sigma, len(individual))
    individual = individual + mask * noise
    return np.clip(individual, 0, 1)

def genetic_algorithm(seed, budget_s, n_points=70):
    np.random.seed(seed)
    start_time = time.time()
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize population
    population_size = 20
    population = [base_h.copy()]
    for _ in range(population_size - 1):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = base_h.copy()
    best_c5 = compute_c5(best_h, dx)
    stagnation_count = 0
    max_stagnation = 20
    
    max_generations = 100
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        fitnesses = evaluate_population(population, dx)
        min_fitness = np.min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if best_c5 <= 0.3807:
            break
        
        if stagnation_count > max_stagnation:
            if population_size < 40:
                population_size = 40
                population = [best_h.copy()]
                for _ in range(population_size - 1):
                    h = best_h + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    population.append(h)
                stagnation_count = 0
            else:
                break
        
        # Selection
        num_parents = population_size // 2
        parents = select_parents(population, fitnesses, num_parents)
        
        # Crossover
        offspring = []
        num_crossover = len(parents) // 2
        for i in range(num_crossover):
            parent1 = parents[i]
            parent2 = parents[len(parents) - 1 - i]
            crossover_point = np.random.randint(1, n_points - 1)
            child1, child2 = crossover(parent1, parent2, crossover_point)
            offspring.extend([child1, child2])
        
        # Mutation
        mutation_rate = 0.2
        sigma = 0.05
        if stagnation_count > 10:
            mutation_rate = 0.1
        for i in range(len(offspring)):
            offspring[i] = mutate(offspring[i], mutation_rate, sigma)
            offspring[i] = np.clip(offspring[i], 0, 1)
            offspring[i] = normalize_h(offspring[i], target_sum)
        
        # Elitism
        elite = [best_h.copy()]
        # Next generation
        population = elite + offspring[:population_size - 1]
        
        generation += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    best_h, best_c5 = genetic_algorithm(seed, budget_s)
    return (best_h.tolist(), best_c5, 70)
