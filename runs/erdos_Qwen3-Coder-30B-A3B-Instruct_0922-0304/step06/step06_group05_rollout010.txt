import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Binary search on C
    C_low = 0.0
    C_high = 1.0
    best_C = float('inf')
    best_h = h.copy()
    
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        C = (C_low + C_high) / 2.0
        
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints for feasibility problem
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add overlap constraints
        # This checks all lags from -n+1 to n-1 except 0
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            # Compute overlap sum efficiently
            lag_sum = 0
            for i in range(n_points):
                idx = (i + k) % n_points
                lag_sum += h_vars[i] * (1 - h_vars[idx])
            constraints.append(lag_sum <= C / dx)
        
        # Formulate and solve problem
        objective = cp.Minimize(0)  # Any objective for feasibility
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                # feasible solution found
                if C < best_C:
                    best_C = C
                    best_h = h_vars.value.copy()
                C_high = C
            else:
                C_low = C
        except:
            C_low = C
            
        # Early exit if target achieved
        if best_C <= 0.3807:
            break
    
    # Refine solution with local optimization
    h = best_h.copy()
    h = np.clip(h, 0, 1)
    
    # Use gradient descent to minimize overlap
    max_iter = 10000
    step_size = 0.005
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient manually
        eps = 1e-4
        g = np.zeros(n_points)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[j] += eps
            h_minus[j] -= eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full").max()
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full").max()
            
            g[j] = (overlap_plus - overlap_minus) / (2 * eps) * dx

        # Update
        new_h = h - step_size * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        h = new_h
        
        # Occasionally perturb to escape local minima
        if i % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.001, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(h)
    
    return (h.tolist(), final_c5, n_points)
