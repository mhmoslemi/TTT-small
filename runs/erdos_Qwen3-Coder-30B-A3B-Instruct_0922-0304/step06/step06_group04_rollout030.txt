import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        
        # Add perturbed versions
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        # Generate random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_individual = None
    best_c5 = float('inf')
    stagnation_count = 0
    mutation_rate = 0.2
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = evaluate_individual(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_individual = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        elite_count = population_size // 2
        selected = sorted_population[:elite_count]
        
        # Crossover: pair and swap segments
        new_population = selected[:]
        
        while len(new_population) < population_size:
            i, j = np.random.choice(elite_count, 2, replace=False)
            parent1 = selected[i]
            parent2 = selected[j]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            # Normalize children
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Truncate to population size
        new_population = new_population[:population_size]
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h.copy()
        
        # Elitism: add the best from previous generation
        if best_individual is not None:
            if len(new_population) > 0:
                new_population[-1] = best_individual.copy()
        
        population = new_population[:]
        generation += 1
        
        # Adjust mutation rate if stagnating
        if stagnation_count > 20:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif stagnation_count > 10:
            mutation_rate = max(0.05, mutation_rate * 0.9)
    
    # Final evaluation
    if best_individual is None:
        # If no improvement, return the initial solution
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            h = normalize_h(h, target_sum)
        final_c5 = compute_c5(h, dx)
        best_individual = h
    else:
        final_c5 = evaluate_individual(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
