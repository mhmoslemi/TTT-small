import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple proportional adjustment
    if current_sum > target_sum:
        # Need to decrease values
        scale = target_sum / current_sum
        h = h * scale
    else:
        # Need to increase values by distributing deficit
        deficit = target_sum - current_sum
        # Distribute deficit as evenly as possible
        # Find indices with room to increase
        max_incr = 1 - h
        total_room = np.sum(max_incr)
        if total_room > 0:
            # Distribute deficit proportionally to remaining space
            h += deficit * max_incr / total_room
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create initial values
        h = np.random.uniform(0, 1, n_points)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            p = np.array(initial_h_values)
        else:
            p = np.random.uniform(0, 1, n_points)
        p = normalize_h(p, target_sum)
        particles.append(p.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_values = [compute_c5(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_values)
    gbest = pbest[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO iteration
    max_iter = 20000
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia weight
        w = w_start - (w_start - w_end) * iter_num / max_iter
        
        for i in range(n_particles):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (gbest - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Position update
            particles[i] += velocities[i]
            
            # Apply constraints
            particles[i] = project_to_constraints(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest[i] = particles[i].copy()
                pbest_values[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest = particles[i].copy()
                    gbest_value = current_c5
        
        # Early termination
        if gbest_value <= 0.3807:
            break
    
    # Final check
    final_c5 = compute_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
