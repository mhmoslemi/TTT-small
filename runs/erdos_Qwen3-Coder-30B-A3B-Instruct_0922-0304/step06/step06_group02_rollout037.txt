import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
    else:
        h_init = np.ones(n_points) * 0.5
    h_init = h_init * (target_sum / np.sum(h_init))
    
    # Variables
    h = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, sum(h) == target_sum, C >= 0]
    
    # Lag indices
    lag_range = range(-(n_points - 1), n_points)
    for k in lag_range:
        # Effective indices for the correlation at lag k
        if k >= 0:
            i_indices = range(n_points - k)
            j_indices = range(k, n_points)
        else:
            i_indices = range(-k, n_points)
            j_indices = range(n_points + k)
        # Build constraint
        expr = 0
        for i, j in zip(i_indices, j_indices):
            expr += h[i] * (1 - h[j])
        if len(i_indices) > 0:
            constraints.append(expr * dx <= C)
    
    # Objective
    objective = Minimize(C)
    
    # Solve
    prob = Problem(objective, constraints)
    
    # Attempt to solve with different solvers
    solver_options = {'max_iters': 5000}
    solver = None
    solver_list = ['MOSEK', 'GUROBI', 'SCIP', 'GLOP']
    for s in solver_list:
        try:
            prob.solve(solver=s, **solver_options)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                solver = s
                break
        except:
            continue
    
    if solver is None:
        # Fallback to ECOS
        try:
            prob.solve(solver='ECOS', **solver_options)
            solver = 'ECOS'
        except:
            # Last resort: just solve with default
            try:
                prob.solve()
                solver = 'DEFAULT'
            except:
                # Return initial values if all else fails
                h_final = h_init
                c5_val = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
                return (h_final.tolist(), c5_val, n_points)
    
    h_final = np.array(h.value).flatten()
    
    # Clip and renormalize
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    # Compute final C5
    c5_val = np.max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    
    return (h_final.tolist(), c5_val, n_points)
