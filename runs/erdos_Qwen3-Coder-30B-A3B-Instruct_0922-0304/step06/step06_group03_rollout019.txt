```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Perturbation control
    noise_amplitude = 0.02
    noise_decay = 0.95
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to guide perturbations
        corr = correlate(h, 1 - h, mode="full")
        # Get indices of top peaks (top 10%)
        peak_indices = np.argsort(corr)[-max(1, n_points//10):]
        
        # Create perturbation mask
        mask = np.zeros(n_points)
        for idx in peak_indices:
            # Find the actual index in the convolution
            if idx >= n_points - 1:
                # Tail of the correlation (positive shifts)
                shift = idx - (n_points - 1)
                for i in range(max(0, shift), min(n_points, shift + 5)):
                    if i < n_points:
                        mask[i] = 1
            else:
                # Beginning of correlation (negative shifts)
                shift = idx
                for i in range(max(0, shift - 4), min(n_points, shift + 5)):
                    if i < n_points:
                        mask[i] = 1
        
        # Apply perturbation
        perturbation = np.random.normal(0, noise_amplitude, n_points)
        h_pert = h + mask * perturbation
        h = np.clip(h_pert, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Optimize with SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
        except:
            pass
            
        # Occasionally do more aggressive search
        if iteration % 100 == 0:
            h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Decay noise amplitude
        if iteration % 500 == 0:
            noise_amplitude *= noise_decay
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
