import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def compute_h_from_comb(n_points, n_teeth, tooth_width, spacing, slope):
    """Construct h using comb structure."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Create teeth at regular intervals
    tooth_positions = []
    for i in range(n_teeth):
        start_pos = i * (tooth_width + spacing) + spacing
        end_pos = start_pos + tooth_width
        tooth_positions.append((start_pos, end_pos))
    
    # Distribute teeth evenly
    for i in range(n_teeth):
        start_pos, end_pos = tooth_positions[i]
        start_idx = int(start_pos / dx)
        end_idx = int(end_pos / dx)
        if start_idx >= n_points:
            break
        if end_idx > n_points:
            end_idx = n_points
            
        # Add tooth with smooth transitions
        for j in range(start_idx, end_idx):
            x = j * dx
            # Determine if we are in the tooth or edge transition
            tooth_center = start_pos + tooth_width / 2
            if x < start_pos:
                # Left edge transition
                dist = start_pos - x
                value = max(0, 1 - dist / slope)
            elif x > end_pos:
                # Right edge transition
                dist = x - end_pos
                value = max(0, 1 - dist / slope)
            else:
                # Inside tooth - full height
                value = 1.0
            h[j] += value
    
    # Normalize to target sum
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    return h
