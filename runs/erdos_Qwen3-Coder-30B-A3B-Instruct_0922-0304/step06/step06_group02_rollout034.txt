import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with simulated annealing inspired approach
    iteration = 0
    max_iter = 10000
    temp = 1.0
    min_temp = 1e-6
    alpha = 0.995
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap 
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Make a small random perturbation
        h_new = h.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_new[idx] += delta
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject with simulated annealing probability
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / temp):
            h = h_new.copy()
            
        # Cool down temperature
        temp = max(min_temp, temp * alpha)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
