import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
    
    # Normalize
    base_h = base_h * (target_sum / np.sum(base_h))
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Bilinear programming approach
    max_trials = 20  # Limit number of optimizations
    for trial in range(max_trials):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve bilinear program using cvxpy
        h_var = cp.Variable(n_points)
        z_vars = [cp.Variable() for _ in range(2*n_points - 1)]
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Objective and bilinear constraints
        objective = cp.Minimize(cp.max(z_vars))
        
        # Add bilinear constraints for each shift k
        for k in range(2*n_points - 1):
            # Determine valid indices for shift k
            if k < n_points:
                # Forward shift
                start_idx = 0
                end_idx = n_points - k
            else:
                # Backward shift
                start_idx = k - n_points + 1
                end_idx = n_points
                
            # z_k = sum over valid i of h[i]*(1 - h[i + k])
            # Using big-M relaxation
            # For numerical stability, make them tight constraints using relaxation
            # This is approximate: z_k >= sum(h[i]*(1-h[i+k])) i=0..end-start-1
            if end_idx > start_idx:
                if k < n_points:
                    # Forward shift z_k = h[i]*(1-h[i+k]) for i=0..end_idx-1
                    l = k
                    expr = cp.sum(cp.multiply(h_var[start_idx:end_idx], 
                                            1 - h_var[start_idx+l:end_idx+l]))
                else:
                    # Backward shift
                    l = k - n_points + 1
                    expr = cp.sum(cp.multiply(h_var[start_idx+l:end_idx], 
                                            1 - h_var[start_idx:end_idx-l]))
                constraints.append(z_vars[k] >= expr)
        
        # Add constraints in a single loop to make them tighter
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_result = h_var.value
                h_result = np.clip(h_result, 0, 1)
                # Re-normalize
                h_result = h_result * (target_sum / np.sum(h_result))
                c5 = compute_c5(h_result, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_result.copy()
        except Exception:
            pass
        
        # Perturb if we didn't improve (adaptive restart)
        if best_c5 > 0.3807:
            noise = np.random.normal(0, 0.01, n_points)
            base_h = base_h + noise
            base_h = np.clip(base_h, 0, 1)
            base_h = base_h * (target_sum / np.sum(base_h))
            best_h = base_h.copy()
        else:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
