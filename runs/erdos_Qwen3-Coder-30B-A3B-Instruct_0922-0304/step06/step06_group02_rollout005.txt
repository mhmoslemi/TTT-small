import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with the original solution and improve
    current_h = h.copy()
    
    # Iterative improvement with adaptive perturbation
    iteration = 0
    max_iter = 20000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(current_h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
          
        # Local optimization using SLSQP
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
            
        # Try several restarts with small random noise
        best_candidate = current_h.copy()
        best_candidate_c5 = current_c5
        for _ in range(5):
            noise = np.random.normal(0, 0.01, n_points)
            h_noisy = current_h + noise
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            result = minimize(objective, h_noisy, method='SLSQP', 
                              bounds=bounds, constraints=constraint, 
                              options={'maxiter': 50, 'ftol': 1e-6})
            
            if result.success:
                candidate = result.x
                candidate = np.clip(candidate, 0, 1)
                candidate = normalize_h(candidate, target_sum)
                candidate_c5 = compute_c5(candidate, dx)
                if candidate_c5 < best_candidate_c5:
                    best_candidate = candidate.copy()
                    best_candidate_c5 = candidate_c5
        
        # Use the best candidate found
        current_h = best_candidate.copy()
        
        # Periodic global perturbation
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            current_h = np.clip(current_h + noise, 0, 1)
            current_h = normalize_h(current_h, target_sum)
        
        # Add occasional larger perturbations
        if iteration % 200 == 0:
            for i in range(n_points):
                if np.random.random() < 0.05:
                    current_h[i] = np.random.random()
            current_h = normalize_h(current_h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
