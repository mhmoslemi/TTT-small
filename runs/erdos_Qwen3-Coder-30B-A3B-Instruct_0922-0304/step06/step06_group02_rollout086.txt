import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Construct the optimization model using CVXPY
    # Variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h_var) == target_sum)
    
    # Bounds
    constraints.append(h_var >= 0)
    constraints.append(h_var <= 1)
    
    # Overlap constraints
    # For each lag k from -(n_points-1) to n_points-1
    for k in range(-(n_points-1), n_points):
        # Compute valid indices for this lag
        if k >= 0:
            valid_indices = np.arange(0, n_points - k)
        else:
            valid_indices = np.arange(-k, n_points)
        
        if len(valid_indices) > 0:
            # Construct the overlap integral constraint
            expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[valid_indices + k])) * dx
            constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s - 1)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_opt = C.value
            
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
        else:
            # Fallback to gradient descent if QP failed
            h = h.copy()
            for _ in range(1000):
                if time.time() - start_time > budget_s - 1:
                    break
                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                
                # Simple gradient descent step with regularization
                # For this specific problem, we use a gradient approximation or heuristic
                h_old = h.copy()
                epsilon = 0.001
                h += epsilon * np.random.randn(n_points)  # Random perturbation
                h = np.clip(h, 0, 1)
                h = h * (target_sum / np.sum(h))  # Normalize
                
    except Exception:
        # Fallback to gradient descent
        h = h.copy()
        iteration = 0
        while time.time() - start_time < budget_s - 1 and iteration < 100000:
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Perturbation heuristic
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new
            iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
