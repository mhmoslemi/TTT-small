import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        
    # Normalize initial h
    current_sum = np.sum(base_h)
    base_h = base_h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Bilinear programming approach
    solver = cp.MOSEK
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 20:
        # Create variables
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        
        # Compute overlap terms with bilinear constraints
        overlaps = []
        for k in range(2 * n_points - 1):
            if k < n_points:
                # h[i] * (1 - h[i + k]) for i in valid range
                expr = 0
                for i in range(n_points - k):
                    expr += h[i] * (1 - h[i + k])
            else:
                # Reverse shift case
                expr = 0
                for i in range(k - n_points + 1, n_points):
                    expr += h[i] * (1 - h[i + k - n_points + 1])
            overlaps.append(expr)
        
        # Objective: minimize max of overlap terms scaled by dx
        max_overlap = cp.max(overlaps)
        objective = cp.Minimize(max_overlap * dx)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h.value
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (target_sum / np.sum(h_solution))
                c5 = compute_c5(h_solution, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_solution.copy()
        except Exception:
            pass
            
        # Perturbation fallback
        if iteration % 5 == 0 and iteration > 0:
            h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
            c5_new = compute_c5(h_perturbed, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_perturbed.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
