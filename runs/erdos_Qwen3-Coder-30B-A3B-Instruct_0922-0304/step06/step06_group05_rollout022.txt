import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h to feasible set: [0,1] and sum = n_points/2"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if np.abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h

    def compute_gradient(h, k_star):
        """Compute exact gradient for max overlap at lag k_star"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # For correlation term, we need to compute the derivative
            # Based on the derivative from each term in the correlation
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                # Wrap-around in correlation
                wrap_idx = i + k_star - n_points
                g[i] = -h[wrap_idx] * dx
        return g
    
    def line_search(h, g, max_iter=5):
        """Find optimal step size using backtracking line search"""
        eta = 0.01
        alpha = 0.1
        beta = 0.5
        current_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            new_h = h - eta * g
            new_h = project_h(new_h)
            new_c5 = compute_c5(new_h)
            
            # Armijo condition
            if new_c5 <= current_c5 - alpha * eta * np.dot(g, g):
                return eta, new_h
            eta *= beta
        
        return eta, h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 100000
    iteration = 0
    perturb_freq = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the maximum lag
        g = compute_gradient(h, max_k)
        
        # Perform line search
        eta, new_h = line_search(h, g)
        
        # Update h
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % perturb_freq == 0:
            # Add noise with decreasing variance
            pert = np.random.normal(0, 0.01 * np.exp(-iteration/1000.0), n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
