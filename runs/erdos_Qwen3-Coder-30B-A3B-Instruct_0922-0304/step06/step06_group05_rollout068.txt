import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb h
        h_candidate = h.copy()
        
        # Select two indices for adjustment
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure we can transfer mass
        if h[i] < 1 and h[j] > 0:
            delta = min(1 - h[i], h[j])
            delta = np.random.uniform(0, delta)
            
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize to maintain sum = n_points / 2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute C5 for both
            current_c5 = compute_c5(h)
            candidate_c5 = compute_c5(h_candidate)
            
            # Metropolis criterion
            if candidate_c5 < current_c5:
                h = h_candidate
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()
                    stagnation_count = 0
            else:
                # Accept with probability
                delta_c5 = candidate_c5 - current_c5
                if np.random.rand() < np.exp(-delta_c5 / T):
                    h = h_candidate
                    stagnation_count = 0
                else:
                    stagnation_count += 1

        # Adjust temperature
        if T > min_T:
            T *= alpha
            
        # Occasionally reset temperature to escape local minima
        if stagnation_count > max_stagnation:
            T = min(T * 1.2, 1.0)
            stagnation_count = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
