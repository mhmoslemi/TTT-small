import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stuck_limit = 500
    stuck_count = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb h
        h_candidate = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure we can move mass from j to i
        if h[j] > 0 and h[i] < 1:
            delta = min(1 - h[i], h[j])
            delta *= np.random.rand()  # Random proportion
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            h_candidate = np.clip(h_candidate, 0, 1)
        else:
            continue
            
        # Compute overlap for candidate
        c5_candidate = compute_c5(h_candidate)
        
        # Accept or reject
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
            stuck_count = 0
        else:
            # Metropolis criterion
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_candidate
                stuck_count = 0
            else:
                stuck_count += 1
                
        # Cooling schedule
        if iteration % 10 == 0:
            T *= alpha
            if T < min_T:
                T = min_T
                
        # Escape stagnation
        if stuck_count > stuck_limit:
            T *= 1.2
            stuck_count = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
