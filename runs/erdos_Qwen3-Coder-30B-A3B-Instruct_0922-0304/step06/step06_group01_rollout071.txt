import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try CMA-ES style optimization for better exploration
    max_iter = 500
    population_size = 2 * n_points
    # Initial parameter for CMA-ES approx
    sigma = 0.1
    weights = np.log(population_size / 2 + 1) - np.log(np.arange(1, population_size // 2 + 1))
    weights = weights / np.sum(weights)
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate population
        population = []
        for _ in range(population_size):
            # Sample from normal distribution
            perturbation = np.random.normal(0, sigma, n_points)
            h_candidate = h + perturbation
            # Clip to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            # Normalize
            h_candidate = normalize_h(h_candidate, target_sum)
            population.append(h_candidate)
            
        # Evaluate population
        fitness = []
        for h_candidate in population:
            c5 = compute_c5(h_candidate, dx)
            fitness.append(c5)
            
        # Selection
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size // 2]]
        selected_fitness = [fitness[i] for i in sorted_indices[:population_size // 2]]
        
        # Update step size and mean
        if len(selected) > 0:
            # Update mean
            h = np.average(selected, axis=0, weights=weights)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Update sigma
            if np.median(selected_fitness) < 0.3808:
                sigma = max(0.01, sigma * 0.98)
            
            # Local polishing
            def objective(h_local):
                return compute_c5(h_local, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h_local):
                return np.sum(h_local) - target_sum
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h.copy(), method='SLSQP', 
                              bounds=bounds, constraints=constraint, 
                              options={'maxiter': 50})
            
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize_h(h, target_sum)
                
            # Check for improvement
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
