import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate=0.05):
    """Mutate h with given rate."""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, 0.05, len(h))
    h_mut = h + mask * noise
    return np.clip(h_mut, 0, 1)

def crossover_h(h1, h2):
    """Crossover two h vectors."""
    crossover_point = np.random.randint(1, len(h1) - 1)
    h_child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    return h_child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm parameters
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_h(h.copy())
        h_mut = normalize_h(h_mut, target_sum)
        population.append(h_mut)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA loop
    generation = 0
    max_generations = 1000
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for ind in population:
            fit = compute_c5(ind, dx)
            fitness.append(fit)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_population = population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, population_size // 2)]
            parent2_idx = sorted_indices[np.random.randint(0, population_size // 2)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            if np.random.random() < crossover_rate:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
            
            child = mutate_h(child, mutation_rate)
            child = np.clip(child, 0, 1)
            child = normalize_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
        
        # Local refinement with SLSQP
        for i in range(min(elite_size, len(population))):
            def constraint_sum(h_vec):
                return np.sum(h_vec) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h_vec):
                return compute_c5(h_vec, dx)
            
            # Refine top candidates
            result = minimize(objective, population[i], method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                refined = np.clip(result.x, 0, 1)
                refined = normalize_h(refined, target_sum)
                population[i] = refined
        
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
