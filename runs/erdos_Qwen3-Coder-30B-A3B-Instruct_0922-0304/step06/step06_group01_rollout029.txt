import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a reasonable starting function
        h = np.zeros(n_points)
        h[n_points//4:3*n_points//4] = 1.0
        h = np.clip(h, 0, 1)

    # Normalize
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    n_freq = 15  # number of Fourier coefficients to optimize
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize PSO
    particles = np.random.rand(n_particles, n_freq * 2) * 2 - 1  # [-1, 1]
    velocities = np.random.rand(n_particles, n_freq * 2) * 0.1 - 0.05  # small initial velocities
    particle_best = particles.copy()
    particle_best_c5 = np.full(n_particles, float('inf'))
    global_best = particles[0].copy()
    global_best_c5 = float('inf')
    
    # Create initial h from first particle
    h_test = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        for j in range(n_freq):
            h_test[i] += particle_best[0, 2*j] * np.cos(j * np.pi * x) + \
                        particle_best[0, 2*j+1] * np.sin(j * np.pi * x)
    h_test = np.clip(h_test, 0, 1)
    h_test = h_test * (target_sum / np.sum(h_test))
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for p in range(n_particles):
            # Reconstruct h from Fourier coefficients
            h_test = np.zeros(n_points)
            for i in range(n_points):
                x = i * dx
                for j in range(n_freq):
                    h_test[i] += particles[p, 2*j] * np.cos(j * np.pi * x) + \
                                particles[p, 2*j+1] * np.sin(j * np.pi * x)
            
            # Ensure h stays within bounds and sum = target_sum
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (target_sum / np.sum(h_test))
            
            # Evaluate
            c5 = compute_c5(h_test, dx)
            
            if c5 < particle_best_c5[p]:
                particle_best[p] = particles[p].copy()
                particle_best_c5[p] = c5
                
            if c5 < global_best_c5:
                global_best = particles[p].copy()
                global_best_c5 = c5
                
            # Update velocity and position
            r1 = np.random.rand(n_freq * 2)
            r2 = np.random.rand(n_freq * 2)
            
            velocities[p] = (w * velocities[p] + 
                            c1 * r1 * (particle_best[p] - particles[p]) +
                            c2 * r2 * (global_best - particles[p]))
            
            particles[p] += velocities[p]
            
        iteration += 1
        
        # Early exit if solution is good
        if global_best_c5 <= 0.3807:
            break
    
    # Final refinement with local optimization
    h_final = np.zeros(n_points)
    for i in range(n_points):
        x = i * dx
        for j in range(n_freq):
            h_final[i] += global_best[2*j] * np.cos(j * np.pi * x) + \
                         global_best[2*j+1] * np.sin(j * np.pi * x)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))

    # Final local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h_final.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))

    final_c5 = compute_c5(h_final, dx)
    
    # Return final solution
    return (h_final.tolist(), final_c5, n_points)
