import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope):
    """Construct a comb-like h function."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Define tooth positions
    tooth_positions = []
    for i in range(n_teeth):
        start_pos = (i + 1) * spacing + i * tooth_width
        tooth_positions.append(start_pos)
    
    # Add teeth
    for i in range(n_teeth):
        start_idx = int(tooth_positions[i] / dx)
        end_idx = int((tooth_positions[i] + tooth_width) / dx)
        if i < n_teeth:
            # Create triangular transition
            for j in range(start_idx, end_idx):
                if j < n_points:
                    # Distance to tooth edge
                    dist_from_start = (j * dx - tooth_positions[i])
                    dist_from_end = (tooth_positions[i] + tooth_width) - (j * dx)
                    # Minimum distance to edges
                    dist = min(dist_from_start, dist_from_end)
                    if dist > 0:
                        h[j] = max(0, 1 - dist / slope)
    
    # Normalize sum to target
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default: simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Search comb parameters
    best_params = (7, 2./70, 2./71, 2./140)  # n_teeth, tooth_width, spacing, slope
    best_c5 = float('inf')
    
    # Grid search over comb parameters
    param_ranges = [
        range(5, 11),        # n_teeth
        [2./70 * (1 - 0.1 + 0.2 * i / 10) for i in range(11)], # tooth_width
        [2./71 * (1 - 0.1 + 0.2 * i / 10) for i in range(11)], # spacing
        [2./140 * (1 + 0.5 * i / 10) for i in range(11)] # slope
    ]
    
    # Start with default values
    grid_search_params = []
    for n_teeth in param_ranges[0]:
        for tooth_width in param_ranges[1]:
            for spacing in param_ranges[2]:
                for slope in param_ranges[3]:
                    if n_teeth * tooth_width + (n_teeth + 1) * spacing <= 2:
                        grid_search_params.append((n_teeth, tooth_width, spacing, slope))
    
    # Subset for faster search
    np.random.shuffle(grid_search_params)
    search_params = grid_search_params[:200]
    
    # Evaluate grid
    for params in search_params:
        if time.time() - start_time > budget_s - 1:
            break
        n_teeth, tooth_width, spacing, slope = params
        h_candidate = construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope)
        c5 = compute_c5(h_candidate, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()
            best_params = params
    
    # Refine best comb with SLSQP
    if time.time() - start_time < budget_s - 5:
        n_teeth, tooth_width, spacing, slope = best_params
        h_refined = construct_comb_h(n_points, n_teeth, tooth_width, spacing, slope)
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Run SLSQP optimization
        result = minimize(objective, h_refined.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
        
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            c5_final = compute_c5(h_final, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    
    # If no improvement, return the best from grid search
    return (best_h.tolist(), best_c5, n_points)
