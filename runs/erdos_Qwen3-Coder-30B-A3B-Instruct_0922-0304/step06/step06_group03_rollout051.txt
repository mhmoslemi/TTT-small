import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_comb_pattern(n_points):
    """Generate a comb pattern with high values."""
    h = np.zeros(n_points)
    for i in range(0, n_points, 4):
        if i < n_points:
            h[i] = 1.0
    return h

def generate_random_pattern(n_points, target_sum):
    """Generate a random pattern with specified sum."""
    h = np.random.random(n_points)
    h = normalize_h(h, target_sum)
    return h

def generate_alternating_pattern(n_points, target_sum):
    """Generate alternating high/low pattern."""
    h = np.ones(n_points)
    for i in range(1, n_points, 2):
        h[i] = 0.0
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
        initial_guesses.append(h0)
    else:
        # Default initialization
        h0 = np.ones(n_points)
        h0[n_points//2:] = 0
        h0 = np.clip(h0, 0, 1)
        h0 = normalize_h(h0, target_sum)
        initial_guesses.append(h0)
        
    # Generate diverse initial guesses
    initial_guesses.append(generate_comb_pattern(n_points))
    initial_guesses.append(generate_random_pattern(n_points, target_sum))
    initial_guesses.append(generate_alternating_pattern(n_points, target_sum))
    
    # Also try a refined version of the comb pattern
    comb_h = generate_comb_pattern(n_points)
    # Add some spread to avoid perfect comb
    for i in range(0, n_points, 3):
        if i < n_points:
            comb_h[i] = 0.8
    comb_h = normalize_h(comb_h, target_sum)
    initial_guesses.append(comb_h)
    
    best_c5 = float('inf')
    best_h = None
    current_h = None
    
    # Use maximally 20000 total iterations, and time budget
    iteration = 0
    max_iter = 20000
    
    for guess_idx, h in enumerate(initial_guesses):
        current_h = h.copy()
        last_improvement = 0
        
        # Use different perturbation schedules for different guesses
        use_coarse_perturb = (guess_idx % 3 == 1)  # Every 3rd guess uses more aggressive perturbation
        
        iter_counter = 0
        while time.time() - start_time < budget_s - 1 and iter_counter < 2000:
            # Compute current C5
            current_c5 = compute_c5(current_h, dx)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                last_improvement = iteration
            
            if best_c5 <= 0.3807:
                break
                
            # Determine perturbation size based on feedback
            base_perturbation = 0.02
            if use_coarse_perturb:
                base_perturbation *= 1.5
                
            if iteration - last_improvement > 1000:
                base_perturbation *= 2  # More aggressive if stuck
                
            # Check for convergence and reduce step size
            if iter_counter % 50 == 0 and iteration > 0:
                base_perturbation *= 0.95
            
            # SLSQP minimization with bound constraints
            def objective(h):
                return compute_c5(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, current_h, method='SLSQP', bounds=bounds, constraints=constraint, 
                                  options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
                if result.success:
                    current_h = result.x
                else:
                    # Use a simple perturbation if optimization fails
                    noise = np.random.normal(0, base_perturbation * 0.5, n_points)
                    current_h = current_h + noise
                    current_h = np.clip(current_h, 0, 1)
                    current_h = normalize_h(current_h, target_sum)
            except:
                # If something goes wrong, do a simple random perturbation
                noise = np.random.normal(0, base_perturbation * 0.5, n_points)
                current_h = current_h + noise
                current_h = np.clip(current_h, 0, 1)
                current_h = normalize_h(current_h, target_sum)
                
            # Apply deterministic perturbation periodically
            if iter_counter % 10 == 0:
                noise = np.random.normal(0, base_perturbation, n_points)
                current_h = current_h + noise
                current_h = np.clip(current_h, 0, 1)
                current_h = normalize_h(current_h, target_sum)
                
            iter_counter += 1
            iteration += 1
            
    # Final check with the best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to initial_h_values or the default
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx)
            return (h.tolist(), final_c5, n_points)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx) 
            return (h.tolist(), final_c5, n_points)
