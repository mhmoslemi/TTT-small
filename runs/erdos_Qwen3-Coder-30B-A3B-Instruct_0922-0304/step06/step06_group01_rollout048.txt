import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # CMA-ES parameters
    popsize = 2 * n_points
    lambda_ = popsize // 2
    sigma = 0.1
    weights = np.log(lambda_ + 1) - np.log(np.arange(1, lambda_ + 1))
    weights = weights / np.sum(weights)
    mueff = 1.0 / np.sum(weights**2)
    
    # Initialize
    mean = h.copy()
    C = np.eye(n_points)
    sigma = 0.1
    pc = np.zeros(n_points)
    ps = np.zeros(n_points)
    cc = 4.0 / (n_points + 4)
    cs = 4.0 / (n_points + 4)
    c1 = 2.0 / ((n_points + 1.3)**2)
    cmu = 2 * mueff / ((n_points + 2)**2)
    chi = np.sqrt(n_points) * (1 - 1.0/(4*n_points) + 1.0/(21*n_points**2))
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main CMA-ES loop
    iteration = 0
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Generate offspring
        offspring = []
        for _ in range(popsize):
            z = np.random.randn(n_points)
            x = mean + sigma * np.dot(C, z)
            # Clip and normalize
            x = np.clip(x, 0, 1)
            x = normalize_h(x, target_sum)
            offspring.append(x)
        
        # Evaluate offspring
        fitness = []
        for ind in offspring:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Sort by fitness
        indices = np.argsort(fitness)
        best_indices = indices[:lambda_]
        
        # Update mean
        h_new = np.zeros(n_points)
        for i in best_indices:
            h_new += weights[i] * offspring[i]
        mean = h_new
        
        # Update evolution paths
        ps = (1 - cs) * ps + np.sqrt(cs * (2 - cs) * mueff) * np.dot(C, np.random.randn(n_points))
        pc = (1 - cc) * pc + np.sqrt(cc * (2 - cc) * mueff) * np.dot(C, np.random.randn(n_points))
        
        # Update covariance matrix
        C = (1 - c1 - cmu) * C
        for i in best_indices:
            z = (offspring[i] - mean) / sigma
            C += c1 * np.outer(pc, pc) + cmu * weights[i] * np.outer(z, z)
        
        # Update step size
        sigma *= np.exp(cs / chi * (np.linalg.norm(ps) / chi - 1))
        
        # Best solution update
        current_best_idx = indices[0]
        current_best_c5 = fitness[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = offspring[current_best_idx].copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
