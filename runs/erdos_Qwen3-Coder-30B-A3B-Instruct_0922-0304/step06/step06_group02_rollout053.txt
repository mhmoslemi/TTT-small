import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Better initialization: smooth step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 10000
    iteration = 0
    patience = 0
    min_patience = 200
    c5_history = []
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if best_c5 <= 0.3807:
            break
            
        # Use a combination of global and local search
        # Global search by perturbing with noise
        noise_magnitude = max(0.02, 0.1 * (1 - (time.time() - start_time) / budget_s))
        h_perturbed = h + np.random.normal(0, noise_magnitude, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        # Local refinement using L-BFGS
        def objective(h_vec):
            return compute_c5(h_vec, dx)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = minimize(objective, h_perturbed, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 * 0.99999:
                h = h_candidate.copy()
        
        # Allow for periodic perturbations
        if iteration % 50 == 0:
            # Add small but strategic perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
        
        if patience > min_patience:
            # Reset with a different initialization if stuck
            h = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            patience = 0
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
