import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from scipy.spatial.distance import cdist

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES implementation
    popsize = 140  # 2 * n_points
    lambda_ = 70   # 1 * n_points
    mu = 35        # 0.5 * lambda_
    weights = np.log(lambda_ + 1) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    mean = h.copy()
    sigma = 0.2
    C = np.eye(n_points)
    diag_D = np.ones(n_points)
    hist_sigma = 1e-6
    c_sigma = 0.3
    d_sigma = 1.5
    c_c = 0.1
    c_1 = 0.2
    c_mu = 0.2
    chi_n = np.sqrt(n_points)
    
    ps = np.zeros(n_points)
    pc = np.zeros(n_points)
    
    # Main CMA-ES loop
    iteration = 0
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Generate offspring
        samples = np.random.multivariate_normal(mean, sigma**2 * C, popsize)
        offspring = []
        
        for i in range(popsize):
            h_k = np.clip(samples[i], 0, 1)
            h_k = normalize_h(h_k, target_sum)
            offspring.append(h_k)
        
        # Evaluate offspring
        fitness = []
        for i in range(popsize):
            c5 = compute_c5(offspring[i], dx)
            fitness.append(c5)
        
        # Rank offspring
        sorted_indices = np.argsort(fitness)
        selected = [offspring[i] for i in sorted_indices[:lambda_]]
        selected_fitness = [fitness[i] for i in sorted_indices[:lambda_]]
        
        # Update evolution paths
        y_w = np.dot(weights, np.array(selected) - mean)
        ps = (1 - c_sigma) * ps + (c_sigma / np.sqrt(1 - (1 - c_sigma)**(2 * iteration + 2))) * y_w / sigma
        
        # Update mean
        mean = mean + sigma * y_w
        
        # Update pc
        pc = (1 - c_c) * pc + c_c * y_w / sigma
        
        # Update covariance matrix
        if iteration > 0:
            # Update diagonal elements
            diag_D = (1 - c_1) * diag_D + c_1 * np.abs(pc)
            
            # Update C matrix
            A = np.zeros((n_points, n_points))
            for i in range(lambda_):
                v = selected[i] - mean
                A += weights[i] * np.outer(v, v)
            
            C = (1 - c_1 - c_mu) * C + c_1 * np.outer(pc, pc) + c_mu * A
        
        # Update sigma
        hist_sigma = (1 - c_sigma) * hist_sigma + c_sigma * np.linalg.norm(y_w) / chi_n
        sigma *= np.exp((c_sigma / d_sigma) * (hist_sigma / 0.001 - 1))
        
        # Update best solution
        min_fitness = min(selected_fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = selected[selected_fitness.index(min_fitness)].copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Local refinement using SLSQP
    def objective(h):
        return compute_c5(h, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
