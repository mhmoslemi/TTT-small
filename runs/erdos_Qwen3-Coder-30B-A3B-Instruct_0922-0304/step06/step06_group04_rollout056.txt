import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    # Track best solution
    best_c5 = float('inf')
    best_individual = None
    
    # Parameters
    max_generations = 100
    crossover_prob = 0.8
    mutation_prob = 0.2
    elitism_rate = 0.1
    
    # Genetic algorithm loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        selected_count = max(1, int(len(population) * 0.5))
        selected = population[:selected_count]
        
        # Elitism: carry over best individuals
        elites = selected[:max(1, int(elitism_rate * len(population)))]
        
        # Create new population
        new_population = elites.copy()
        
        # Crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                a, b = np.random.choice(selected, 2, replace=False)
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child1 = np.where(mask, a, b)
                child2 = np.where(mask, b, a)
                new_population.extend([child1, child2])
            else:
                # If no crossover, just copy selected
                parent = selected[np.random.randint(0, len(selected))]
                new_population.append(parent)
                
        # Limit to exact population size
        new_population = new_population[:population_size]
        
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_prob:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
        
        population = new_population
    
    if best_individual is None:
        # Fallback
        if 'initial_h_values' in globals():
            base_h = np.array(initial_h_values)
            base_h = np.clip(base_h, 0, 1)
            base_h = normalize_h(base_h, target_sum)
            best_individual = base_h
        else:
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
            best_individual = normalize_h(best_individual, target_sum)
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
