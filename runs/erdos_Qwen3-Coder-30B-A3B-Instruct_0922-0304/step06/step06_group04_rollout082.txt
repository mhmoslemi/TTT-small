import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    generation = 0
    stagnation_count = 0
    mutation_rate = 0.2
    max_stagnation = 20
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness_scores.append((c5, h))
        
        fitness_scores.sort(key=lambda x: x[0])
        current_best_c5 = fitness_scores[0][0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_individual = fitness_scores[0][1].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: keep top 50%
        selected = [h for _, h in fitness_scores[:population_size // 2]]
        
        # Create new population
        new_population = []
        new_population.extend(selected)
        
        # Elitism: always keep the best
        if best_individual is not None:
            new_population.append(best_individual.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if random.random() < 0.8 and len(selected) >= 2:
                parent1, parent2 = random.sample(selected, 2)
                crossover_point = random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if random.random() < mutation_rate:
                    child1 += np.random.normal(0, 0.05, n_points)
                    child1 = np.clip(child1, 0, 1)
                    child1 = normalize_h(child1, target_sum)
                if random.random() < mutation_rate:
                    child2 += np.random.normal(0, 0.05, n_points)
                    child2 = np.clip(child2, 0, 1)
                    child2 = normalize_h(child2, target_sum)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Mutation only
                parent = random.choice(selected)
                child = parent + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = normalize_h(child, target_sum)
                new_population.append(child)
        
        # Ensure population size
        population = new_population[:population_size]
        
        # Adjust mutation rate if stagnating
        if stagnation_count > max_stagnation:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            stagnation_count = 0
        else:
            mutation_rate = max(0.05, mutation_rate * 0.95)
        
        generation += 1
        
        # Increase population size if no improvement for a while
        if stagnation_count > max_stagnation and population_size < 40:
            population_size = 40
            while len(population) < population_size:
                h = np.random.rand(n_points)
                h = normalize_h(h, target_sum)
                population.append(h.copy())
    
    # Final evaluation
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        # Try a fallback to initial solution if no improvement
        if 'initial_h_values' in globals():
            base_h = np.array(initial_h_values)
            base_h = np.clip(base_h, 0, 1)
            base_h = normalize_h(base_h, target_sum)
            final_c5 = compute_c5(base_h, dx)
            best_individual = base_h
        else:
            final_c5 = float('inf')
    
    return (best_individual.tolist(), final_c5, n_points)
