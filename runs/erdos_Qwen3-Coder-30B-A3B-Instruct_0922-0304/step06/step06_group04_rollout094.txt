import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient-based optimization with line search
    max_iter = 1000
    step_size = 0.01
    beta = 0.9
    eps = 1e-8
    
    # Initialize with good starting point
    h = base_h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Calculate gradient approximation using finite differences
        h_plus = h.copy()
        h_minus = h.copy()
        eps_h = 1e-5
        
        # Compute current value
        current_c5 = compute_c5(h, dx)
        
        # Compute approximate gradient per component
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus[i] += eps_h
            h_minus[i] -= eps_h
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            h_minus = normalize_h(h_minus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps_h)
            
            # Reset for next iteration
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Line search
        alpha = step_size
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        
        # Backtracking line search
        while new_c5 >= current_c5 and alpha > 1e-8:
            alpha *= beta
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            new_c5 = compute_c5(h_new, dx)
        
        h = h_new
        current_c5 = new_c5
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
