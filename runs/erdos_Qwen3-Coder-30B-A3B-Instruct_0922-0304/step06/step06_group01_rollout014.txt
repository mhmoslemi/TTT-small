import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function with a gradual transition
        h = np.zeros(n_points)
        midpoint = n_points // 2
        # Gradual transition from 0 to 1
        transition_width = n_points // 10
        for i in range(max(0, midpoint - transition_width), min(n_points, midpoint + transition_width)):
            h[i] = max(0, min(1, (i - (midpoint - transition_width)) / (2 * transition_width)))
        # Make sure it sums to target_sum
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    
    # Local refinement parameters
    lsqp_iter = 200
    lsqp_freq = 10000
    
    sa_iter = 0
    max_sa_iter = 100000
    
    while time.time() - start_time < budget_s - 1 and sa_iter < max_sa_iter:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Create neighbor solution by perturbing 10% of the values
        h_new = h.copy()
        num_perturb = max(1, n_points // 10)
        perturb_indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for i in perturb_indices:
            # Add small random change
            h_new[i] += np.random.normal(0, 0.1)
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        # Normalize to sum to target
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h = h_new.copy()
            
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Cooling
        temp *= cooling_rate
        
        # Local refinement every 10000 steps
        if sa_iter % lsqp_freq == 0 and sa_iter > 0:
            # Optimizer refinement
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            def objective(h):
                return compute_c5(h, dx)
            
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': lsqp_iter})
            
            if result.success:
                h = result.x.copy()
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        sa_iter += 1
    
    # Final local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def objective(h):
        return compute_c5(h, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        best_h = result.x.copy()
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
