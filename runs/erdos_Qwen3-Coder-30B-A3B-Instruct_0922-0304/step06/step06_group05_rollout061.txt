import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Binary search for optimal C5
    c_low = 0.0
    c_high = 1.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Function to compute C5
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Convex optimization approach
    def solve_feasibility(h0, c_target):
        # Variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        # Create constraint for each lag k
        # Early exit if target already achieved
        if compute_c5(h0) <= c_target:
            return True, h0.copy()
        
        # For small problem size, we can add constraints explicitly
        # Approximate each overlap constraint
        overlap_max = 0
        for k in range(1, n_points):  # Only non-zero lags
            # Rearrange indices to avoid Python modulo
            sum_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                sum_expr += h[i] * (1 - h[j])
            constraints.append(sum_expr <= c_target / dx)
            overlap_max = max(overlap_max, sum_expr.value if sum_expr.value is not None else 0)
            
        # Try solving with ECOS or MOSEK (fallback to ECOS if MOSEK not available)
        try:
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return True, h.value
            else:
                return False, h0.copy()
        except:
            try:
                prob = cp.Problem(cp.Minimize(0), constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    return True, h.value
                else:
                    return False, h0.copy()
            except:
                return False, h0.copy()
    
    # Binary search for the minimal C that satisfies constraints
    iteration = 0
    max_iterations = 100
    
    while (c_high - c_low > 1e-4) and (time.time() - start_time < budget_s - 2) and iteration < max_iterations:
        iteration += 1
        c_mid = (c_low + c_high) / 2.0
        
        feasible, h_candidate = solve_feasibility(h, c_mid)
        
        if feasible:
            c_high = c_mid
            if compute_c5(h_candidate) < best_c5:
                best_c5 = compute_c5(h_candidate)
                best_h = h_candidate.copy()
        else:
            c_low = c_mid
            
        # Some regularization and local improvement
        if iteration % 20 == 0 and time.time() - start_time < budget_s - 2:
            h = best_h.copy()
            # Run simple gradient descent refinement
            for _ in range(50):
                overlap = correlate(h, 1 - h, mode="full")
                max_k = np.argmax(overlap)
                
                if overlap[max_k] * dx < c_low:
                    break
                    
                # Compute an approximate gradient
                eps = 1e-4
                g = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_minus = h.copy()
                    h_plus[i] += eps
                    h_minus[i] -= eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_minus = np.clip(h_minus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                    overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                    g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
                
                # Gradient descent step with line-search or fixed step
                h_copy = h.copy()
                h = h - 0.005 * g  # Fixed step size
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
                if compute_c5(h) < best_c5:
                    best_c5 = compute_c5(h)
                    best_h = h.copy()
                if best_c5 <= 0.3807:
                    break
                    
    # Final refinement
    for _ in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        overlap = correlate(best_h, 1 - best_h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        if max_overlap < best_c5:
            best_c5 = max_overlap
            
        # Approximate gradient
        eps = 1e-4
        g = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_minus = best_h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx
            
        # Gradient descent
        new_h = best_h - 0.002 * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        if compute_c5(new_h) < best_c5:
            best_h = new_h
            best_c5 = compute_c5(best_h)
        if best_c5 <= 0.3807:
            break
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
