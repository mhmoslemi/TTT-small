import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction or default
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    # Normalize
    current_sum = np.sum(h_base)
    if current_sum > 0:
        h_base = h_base * (target_sum / current_sum)
        
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Try multiple restarts
    max_restarts = 20
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Try solving with CVXPY bilinear programming approach
        try:
            # Create variables
            h = cp.Variable(n_points)
            z = cp.Variable(n_points * 2 - 1)  # For all shifts
            
            # Constraints
            constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
            
            # Objective (minimize max of overlap terms)
            objective = cp.Minimize(cp.max(z))
            
            # Build bilinear constraints manually
            # Using a simplified approach due to solver limitations
            # Direct bilinear constraints are difficult in standard convex solvers
            # So, we'll try alternating optimization with feasibility checks
            h_trial = h_base.copy()
            
            # Local search with perturbations
            for _ in range(50):  # iterations
                # Perturb
                noise = np.random.normal(0, 0.01, n_points)
                h_trial = h_trial + noise
                h_trial = np.clip(h_trial, 0, 1)
                
                # Normalize
                current_sum = np.sum(h_trial)
                if current_sum > 0:
                    h_trial = h_trial * (target_sum / current_sum)
                
                # Check viability
                c5_trial = compute_c5(h_trial, dx)
                if c5_trial < best_c5:
                    best_c5 = c5_trial
                    best_h = h_trial.copy()
                    
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    return (best_h.tolist(), best_c5, n_points)
                    
                # Break if spent too much time
                if time.time() - start_time > budget_s - 1:
                    break
                    
        except:
            pass
            
        # Random restart
        h_base = np.random.rand(n_points)
        h_base = np.clip(h_base, 0, 1)
        current_sum = np.sum(h_base)
        if current_sum > 0:
            h_base = h_base * (target_sum / current_sum)
        restart_count += 1
        
    return (best_h.tolist(), best_c5, n_points)
