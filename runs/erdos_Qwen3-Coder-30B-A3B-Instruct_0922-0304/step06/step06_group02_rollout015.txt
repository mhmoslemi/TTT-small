import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, rate=0.05, target_sum=35.0):
    """Apply mutation to h while maintaining constraints."""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < rate
    h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover_h(h1, h2, rate=0.7):
    """Perform crossover between two h vectors."""
    if np.random.random() > rate:
        return h1.copy()
    cut_point = np.random.randint(1, len(h1)-1)
    h_child = np.concatenate([h1[:cut_point], h2[cut_point:]])
    return h_child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 30
    generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_perturb = mutate_h(h, mutation_rate, target_sum)
        population.append(h_perturb)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitness.append(c5)
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            idx1 = np.random.randint(0, population_size // 2)
            idx2 = np.random.randint(0, population_size // 2)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            child = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            child = mutate_h(child, mutation_rate, target_sum)
            
            new_population.append(child)
            
        # Local SLSQP refinement for top performers
        for i in range(elite_size):
            def objective(h_candidate):
                return compute_c5(h_candidate, dx)
                
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h_candidate):
                return np.sum(h_candidate) - target_sum
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, new_population[i], method='SLSQP', 
                             bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined = np.clip(result.x, 0, 1)
                refined = normalize_h(refined, target_sum)
                new_population[i] = refined
        
        population = new_population
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
