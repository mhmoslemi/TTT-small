import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a smooth step function as initial guess
        h = np.zeros(n_points)
        # Create a smooth transition from 0 to 1
        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = 0.0
            elif i > mid + 5:
                h[i] = 1.0
            else:
                h[i] = (i - mid) / 5.0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with momentum and adaptive steps
    iteration = 0
    max_iter = 10000
    current_step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use a simple gradient-based update with momentum
        # Compute gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        h_base = h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step with momentum  
        momentum = 0.9
        if iteration == 0:
            velocity = np.zeros(n_points)
        else:
            velocity = momentum * velocity - current_step_size * grad
        
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally increase step size for faster convergence
        if iteration % 100 == 0 and current_step_size < 0.05:
            current_step_size *= 1.1
            
        iteration += 1
    
    # Refine the final solution with local search
    bounds = [(0, 1) for _ in range(n_points)]
    target_sum_fun = lambda h: np.sum(h) - target_sum
    
    def objective(h):
        return compute_c5(h, dx)
    
    constraint = {'type': 'eq', 'fun': target_sum_fun}
    
    # Try multiple random restarts to find better local minima
    for restart in range(5):
        h_noisy = h + np.random.normal(0, 0.02, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        try:
            result = minimize(objective, h_noisy, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 100})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()
        except:
            pass
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
