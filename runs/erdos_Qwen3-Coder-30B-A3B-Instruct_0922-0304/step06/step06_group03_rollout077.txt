import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_guess(n_points, method="comb"):
    """Create initial guess with different methods."""
    target_sum = n_points / 2
    if method == "comb":
        h = np.zeros(n_points)
        step = max(1, n_points // 5)
        h[::step] = 1.0
        h = normalize_h(h, target_sum)
    elif method == "random":
        h = np.random.random(n_points)
        h = normalize_h(h, target_sum)
    elif method == "alternating":
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = normalize_h(h, target_sum)
    elif method == "halton":
        # Simple halton-like pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = (i * 0.618033988749895) % 1
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h = normalize_h(h, target_sum)
    return np.clip(h, 0, 1)

def project_to_feasible(h, target_sum):
    """Project h to stay within feasible bounds and sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Get initial point if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multiple initial guesses
    initial_methods = ["comb", "random", "alternating", "halton"]
    
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try multiple strategies
        if iteration % 1000 == 0 and iteration > 0:
            # Start with new initial guess
            method = initial_methods[iteration // 1000 % len(initial_methods)]
            h = create_initial_guess(n_points, method)
            
        # Perturbation with adaptive schedule
        pert_std = max(0.001, 0.02 * (0.95 ** (iteration // 100)))
        if iteration % 10 == 0:
            perturbation = np.random.normal(0, pert_std, n_points)
            h = h + perturbation
            h = project_to_feasible(h, target_sum)
        else:
            # Random small perturbation
            h = h + np.random.normal(0, 0.005, n_points)
            h = project_to_feasible(h, target_sum)
            
        # Occasionally do larger perturbations
        if iteration % 50 == 0:
            h = h + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, target_sum)
            
        # Optimization based on gradient
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8})
            if result.success:
                h = result.x
                h = project_to_feasible(h, target_sum)
        except:
            pass  # Continue with perturbation if optimization fails
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
