import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize h to sum to target_sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different heuristics
    for heuristic_iter in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a fresh h for this heuristic try
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
        
        # Normalize
        h = h * (target_sum / np.sum(h))
        
        # Iterative improvement using CVXPY QP formulation
        max_iter = 500
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Define optimization variables
            h_var = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                cp.sum(h_var) == target_sum,
                h_var >= 0,
                h_var <= 1,
                C >= 0
            ]
            
            # Lag constraints: For each lag k, compute the overlap
            # The overlap integral for lag k is sum_i h[i] * (1 - h[i+k]) * dx
            for lag in range(-(n_points - 1), n_points):
                # Compute valid indices for lag
                valid_indices = []
                for i in range(n_points):
                    if 0 <= i + lag < n_points:
                        valid_indices.append(i)
                
                if valid_indices:
                    # Calculate the overlap term for this lag
                    overlap = 0
                    for i in valid_indices:
                        overlap += h_var[i] * (1 - h_var[i + lag])
                    
                    constraints.append(overlap * dx <= C)
            
            # Objective
            objective = cp.Minimize(C)
            
            # Create and solve the problem
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=2.0)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = h_var.value
                    if h_new is not None:
                        h_new = np.clip(h_new, 0, 1)
                        current_c5 = np.max(correlate(h_new, 1 - h_new, mode="full")) * dx
                        if current_c5 < best_c5:
                            best_c5 = current_c5
                            best_h = h_new.copy()
                            
                        # Use this as the starting point for next iteration
                        h = h_new
                    else:
                        # Fallback to a maximal descent step
                        if iter_count % 10 == 0:
                            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                            h = h * (target_sum / np.sum(h))
                else:
                    # Revert to a stochastic local search
                    if iter_count % 10 == 0:
                        h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                        h = h * (target_sum / np.sum(h))
                        
            except Exception:
                # Fallback to heuristic local search
                if iter_count % 10 == 0:
                    h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                    h = h * (target_sum / np.sum(h))
            
            # Early exit
            if best_c5 <= 0.3807:
                break
    
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
