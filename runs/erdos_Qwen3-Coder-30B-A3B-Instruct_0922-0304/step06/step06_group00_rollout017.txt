import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, strength=0.1):
    """Mutate an individual."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, strength)
    return mutated

def optimize_individual(h, target_sum, dx):
    """Refine an individual using gradient-based method."""
    # Simple gradient ascent step
    steps = 10
    lr = 0.01
    
    for _ in range(steps):
        # Approximate gradients using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update with gradient ascent
        h += lr * grad
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    return h

        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    # Initialize population with variations
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_fitness = float('inf')
    best_individual = h.copy()
    
    # Precompute full correlation lengths for efficiency

        # Evaluate fitness of entire population
        fitness = np.array([compute_c5(ind, dx) for ind in population])
        
        # Track best individual
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_fitness:

        # Early termination
        if best_fitness <= 0.3807:
