import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # SLSQP optimization with time tracking and early stopping
    max_iters = 10000
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Underlying heuristic: perturb and optimize locally
        # Sample nearby solutions for exploration
        if iteration % 5 == 0:
            # Add random noise and re-scale
            noise = np.random.normal(0, 0.02, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = normalize_h(h_perturbed, target_sum)
        else:
            h_perturbed = h.copy()
            
        # Local optimization using SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', 
                            bounds=bounds, constraints=constraints, options={'maxiter': 50})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                h = h_new.copy()
        except:
            pass
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

    # Final polishing
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
        
    def objective(h_vec):
        return compute_c5(h_vec, dx)
        
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h, method='SLSQP', 
                        bounds=bounds, constraints=constraints, options={'maxiter': 200})
        if result.success:
            final_h = result.x
            final_h = np.clip(final_h, 0, 1)
            final_h = normalize_h(final_h, target_sum)
            final_c5 = compute_c5(final_h, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h.copy()
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
