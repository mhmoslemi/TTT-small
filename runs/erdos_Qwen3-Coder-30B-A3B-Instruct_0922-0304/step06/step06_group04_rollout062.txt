import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    else:
        h_base = np.ones(n_points) * 0.5
        h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 20:
        # Solve bilinear program using CVXPY
        h = Variable(n_points)
        C5 = Variable()
        
        # Define constraints
        constraints = [0 <= h, h <= 1, sum(h) == target_sum, C5 >= 0]
        
        # Add bilinear constraints for each k
        for k in range(2 * n_points - 1):
            # Calculate the valid range for i for this shift k
            if k < n_points:
                valid_range = range(0, n_points - k)
            else:
                valid_range = range(k - n_points + 1, n_points)
            
            # Compute sum(h[i] * (1 - h[i+k])) for this shift
            sum_term = 0
            for i in valid_range:
                sum_term += h[i] * (1 - h[i + k])
            
            # Add constraint: sum_term <= C5 * dx
            if len(valid_range) > 0:
                constraints.append(sum_term <= C5 * dx)
        
        # Objective
        objective = Minimize(C5)
        
        # Create problem and solve
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=MOSEK, verbose=False, max_iters=1000)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_sol = np.array([float(h[i].value) for i in range(n_points)])
                c5 = compute_c5(h_sol, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
            else:
                # Fallback to direct optimization if the bilinear program fails
                h_sol = h_base.copy()
                noise = np.random.normal(0, 0.02, n_points)
                h_sol = h_sol + noise
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (target_sum / np.sum(h_sol))
                c5 = compute_c5(h_sol, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_sol.copy()
        except:
            # Fallback if MOSEK fails
            h_sol = h_base.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h_sol = h_sol + noise
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol * (target_sum / np.sum(h_sol))
            c5 = compute_c5(h_sol, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_sol.copy()
        
        # Perturb for next iteration
        perturbation = np.random.normal(0, 0.01, n_points)
        h_base = best_h + perturbation
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
        
        iteration += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
