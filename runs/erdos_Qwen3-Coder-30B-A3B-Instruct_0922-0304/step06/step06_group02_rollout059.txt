import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Gradient-based search with momentum
    velocity = np.zeros_like(h)
    momentum = 0.9
    learning_rate = 0.001
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Estimate gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with momentum
        velocity = momentum * velocity - learning_rate * grad
        h = h + velocity
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Periodic noise injection
        if iteration % 30 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    # Final correction
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
