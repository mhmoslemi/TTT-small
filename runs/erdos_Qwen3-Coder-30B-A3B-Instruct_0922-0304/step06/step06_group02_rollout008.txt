import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Iterative convex optimization approach
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 to set constraint level
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Create convex optimization problem
        h_var = Variable(n_points)
        C = Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == target_sum,
            C >= 0
        ]
        
        # Add overlap constraints using approximate quadratic relaxations
        # Build overlap matrix
        # We'll build a projected version that promotes smoothness
        h_orig = h.copy()
        
        # Simple local search with slight perturbations
        # Do local optimization around current point
        noise = np.random.normal(0, 0.02, n_points)
        h_trial = h_orig + noise
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (target_sum / np.sum(h_trial))
        
        trial_c5 = compute_c5(h_trial, dx)
        if trial_c5 < current_c5 * 0.999999:
            h = h_trial.copy()
            
        # Periodic global adjustment
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (target_sum / np.sum(h))
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
