import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        # Create random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    stagnation_counter = 0
    mutation_rate = 0.2
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Adjust mutation rate based on stagnation
        if stagnation_counter > 20:
            mutation_rate = 0.1
        else:
            mutation_rate = 0.2
            
        # Selection: keep top 50%
        selected_count = population_size // 2
        selected = population[:selected_count]
        new_population = selected[:]
        
        # Elitism: add best individual
        new_population.append(best_individual.copy())
        
        # Crossover: create offspring
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Mutate
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
        
        population = new_population
        generation += 1
    
    if best_individual is None:
        # Fallback to the best of initial
        fitness = [evaluate_individual(h, dx) for h in population]
        best_index = np.argmin(fitness)
        best_individual = population[best_index]
        best_c5 = fitness[best_index]
        
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
