import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction or a simple ramp
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Simple ramp construction
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO-inspired approach
    # Swarm of 25 particles
    n_particles = 25
    particles = []
    velocities = []
    best_positions = []
    best_scores = []
    
    # Initialize swarm with diverse samples
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values) + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.uniform(0, 1, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        best_positions.append(particle.copy())
        best_scores.append(float('inf'))
    
    # PSO parameters
    inertia = 0.7
    c1 = 1.5
    c2 = 1.5
    
    max_iter = 200
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate all particles
        scores = []
        for i, particle in enumerate(particles):
            score = compute_c5(particle, dx)
            scores.append(score)
            
            # Update personal best
            if score < best_scores[i]:
                best_scores[i] = score
                best_positions[i] = particle.copy()
            
            # Update global best
            if score < best_c5:
                best_c5 = score
                best_h = particle.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update positions
        global_best_idx = np.argmin(best_scores)
        global_best = best_positions[global_best_idx]
        
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (best_positions[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
        # Periodic perturbation of the best solution
        if iteration % 10 == 0:
            perturbation = np.random.normal(0, 0.03, n_points)
            perturbed_best = best_h + perturbation
            perturbed_best = np.clip(perturbed_best, 0, 1)
            perturbed_best = normalize_h(perturbed_best, target_sum)
            
            candidate_c5 = compute_c5(perturbed_best, dx)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = perturbed_best.copy()
        
        # Update inertia
        inertia = 0.7 + (0.4 - 0.7) * (iteration / max_iter)
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
