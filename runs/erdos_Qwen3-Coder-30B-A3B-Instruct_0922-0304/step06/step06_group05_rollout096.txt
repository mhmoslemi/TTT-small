import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with line search and perturbations
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    perturbation_freq = 500
    perturbation_std = 0.01
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def line_search(h, g, max_overlap, max_k):
        # Backtracking line search
        eta = 0.01
        alpha = 0.1
        beta = 0.5
        max_iter_ls = 10
        for _ in range(max_iter_ls):
            h_new = h - eta * g
            h_new = project_h(h_new)
            overlap_new = correlate(h_new, 1 - h_new, mode="full")
            new_max = overlap_new[max_k] * dx
            if new_max <= max_overlap - alpha * eta * np.dot(g, g):
                return eta, h_new
            eta *= beta
        return eta, h_new
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the max lag
        g = np.zeros(n_points)
        for i in range(n_points):
            # Gradient due to position i in h affecting lag k*
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            else:
                g[i] = -h[i - (n_points - max_k)] * dx

        # Line search to find optimal step size
        eta, h_new = line_search(h, g, max_overlap, max_k)
        
        # Update h
        h = h_new
        
        # Occasionally perturb to escape local minima
        if iteration % perturbation_freq == 0:
            pert = np.random.normal(0, perturbation_std, n_points)
            h = h + pert
            h = project_h(h)
            # Decay perturbation std
            perturbation_std *= 0.99

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
