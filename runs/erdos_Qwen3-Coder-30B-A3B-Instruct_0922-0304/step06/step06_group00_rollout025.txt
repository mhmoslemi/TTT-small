import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h_mut = h + np.random.normal(0, mutation_strength, len(h))
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover_parents(h1, h2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1) - 1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return h1_new, h2_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        # Create base pattern
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = normalize_h(h_init, target_sum)
    
    # GA parameters
    population_size = 100
    generations = 600
    mutation_strength_start = 0.1
    mutation_strength_end = 0.005
    
    # Initialize population
    population = [h_init.copy()]
    for i in range(population_size - 1):
        h = mutate_individual(h_init, 0.05)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = compute_c5(h_init, dx)
    best_h = h_init.copy()
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Select parents and create offspring
        new_population = []
        for _ in range(population_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Apply mutation
            mutation_strength = mutation_strength_start + (mutation_strength_end - mutation_strength_start) * (gen / generations)
            child1 = mutate_individual(child1, mutation_strength)
            child2 = mutate_individual(child2, mutation_strength)
            
            # Repair and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Keep best individuals for diversity
        elite_indices = np.argsort(fitnesses)[:population_size//10]
        for i in elite_indices:
            new_population[i] = population[i].copy()
            
        population = new_population
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
