import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more sophisticated local search routine
    iteration = 0
    max_iter = 5000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create a modified version with more sophisticated perturbations
        h_mod = h.copy()
        
        # Apply random local changes
        for _ in range(5):
            idx = np.random.randint(0, n_points)
            delta = np.random.normal(0, 0.03)
            h_mod[idx] += delta
        
        h_mod = np.clip(h_mod, 0, 1)
        h_mod = normalize_h(h_mod, target_sum)
        
        # Evaluate modified solution
        candidate_c5 = compute_c5(h_mod, dx)
        if candidate_c5 < current_c5:
            h = h_mod
        
        # Occasionally apply global mutations
        if iteration % 50 == 0:
            # Do a sweep of random step modifications
            h_sweep = h.copy()
            step_change = np.random.randint(0, n_points//4)
            for i in range(step_change):
                idx = np.random.randint(0, n_points-1)
                h_sweep[idx] = np.random.random()
            h_sweep = np.clip(h_sweep, 0, 1)
            h_sweep = normalize_h(h_sweep, target_sum)
            sweep_c5 = compute_c5(h_sweep, dx)
            if sweep_c5 < current_c5:
                h = h_sweep
                
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
