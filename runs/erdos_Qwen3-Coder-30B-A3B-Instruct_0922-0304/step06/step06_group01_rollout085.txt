import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    
    # Local optimization after SA
    local_iter = 0
    max_local_iter = 200
    
    # SA iterations
    sa_iter = 0
    max_sa_iter = 100000
    
    while time.time() - start_time < budget_s - 1 and sa_iter < max_sa_iter:
        # Create a neighbor solution by perturbing 10% of variables
        h_new = h.copy()
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            h_new[idx] = np.random.random()
        
        # Restore sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute change in C5
        c5_old = compute_c5(h, dx)
        c5_new = compute_c5(h_new, dx)
        delta = c5_new - c5_old
        
        # Accept or reject based on SA probability
        if delta < 0 or np.random.random() < math.exp(-delta / temp):
            h = h_new.copy()
        
        # Update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Cooling schedule
        if sa_iter % 100 == 0:
            temp = max(min_temp, temp * cooling_rate)
            
        # Local optimization occasionally
        if sa_iter > 0 and sa_iter % 10000 == 0:
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h):
                return np.sum(h) - target_sum
            constraint = {'type': 'eq', 'fun': constraint_sum}
            def objective(h):
                return compute_c5(h, dx)
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, 
                              constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize_h(h, target_sum)
            # Re-initialize temperature after local optimization
            temp = 0.1
            
        sa_iter += 1
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final local refinement
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h):
        return np.sum(h) - target_sum
    constraint = {'type': 'eq', 'fun': constraint_sum}
    def objective(h):
        return compute_c5(h, dx)
    result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, 
                      constraints=constraint, options={'maxiter': 100})
    if result.success:
        h = np.clip(result.x, 0, 1)
        h = normalize_h(h, target_sum)
        final_c5 = compute_c5(h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
