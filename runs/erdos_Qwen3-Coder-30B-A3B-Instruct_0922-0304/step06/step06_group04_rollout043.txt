import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent with line search
    max_iter = 5000
    step_size = 0.01
    beta = 0.9
    eps = 1e-3
    gamma = 0.9
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (finite differences)
        eps_h = 1e-4
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h, dx)
        
        for j in range(n_points):
            h_eps = h.copy()
            h_eps[j] += eps_h
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = normalize_h(h_eps, target_sum)
            c5_eps = compute_c5(h_eps, dx)
            grad[j] = (c5_eps - base_c5) / eps_h
        
        # Update with gradient descent
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Line search
        c5_new = compute_c5(h_new, dx)
        if c5_new < base_c5:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            step_size *= beta
            
        # Occasionally adjust step size
        if i % 100 == 0:
            step_size = max(step_size * gamma, 1e-6)
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
