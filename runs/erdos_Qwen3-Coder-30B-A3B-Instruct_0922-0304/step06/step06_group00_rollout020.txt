import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize to sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Try SDP approach for better solution
    try:
        # Create variables
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            c5_var >= 0
        ]
        
        # Overlap constraints
        full_length = 2 * n_points - 1
        lags = np.arange(-n_points + 1, n_points)
        
        # Precompute all shift indices
        shift_indices = []
        for k in lags:
            indices = np.arange(n_points)
            shifted_indices = (indices - k) % n_points
            shift_indices.append(shifted_indices)
        
        # Set up constraints for all shifts
        for k, indices in zip(lags, shift_indices):
            # For Lag k, compute sum h[i] * (1 - h[(i-k)%n])
            # This is equivalent to: sum(h[i] * (1 - h[shifted_indices[i]]))
            overlap_sum = cp.sum(cp.multiply(h_var, 1 - h_var[indices]))
            constraints.append(overlap_sum <= c5_var)
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s/2)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            if h_opt is not None:
                c5_opt = c5_var.value
                if not np.isnan(c5_opt):
                    h_opt = np.clip(h_opt, 0, 1)
                    # Re-normalize
                    h_opt_sum = np.sum(h_opt)
                    if h_opt_sum > 0:
                        h_opt = h_opt * (target_sum / h_opt_sum)
                    c5_opt_eval = evaluate_overlap(h_opt, dx)
                    if c5_opt_eval < best_c5:
                        best_h = h_opt
                        best_c5 = c5_opt_eval
    except:
        pass
    
    # Local refinement if needed
    if time.time() - start_time < budget_s - 5:
        # Use local gradient-based optimization for final refinement
        def objective(h_vector):
            return evaluate_overlap(h_vector, dx)
        
        from scipy.optimize import minimize
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}]
        
        result = minimize(objective, best_h, method='SLSQP', 
                         bounds=bounds, constraints=constraints, 
                         options={'maxiter': 200, 'ftol': 1e-8})
        
        if result.success:
            refined_h = np.clip(result.x, 0, 1)
            refined_sum = np.sum(refined_h)
            if refined_sum > 0:
                refined_h = refined_h * (target_sum / refined_sum)
            final_c5 = evaluate_overlap(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final validation and return
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h)
    if final_sum > 0:
        final_h = final_h * (target_sum / final_sum)
    final_c5 = evaluate_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
