import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x)."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs)) * np.pi * x[:, None]),
        axis=1
    )
    return h

def optimize_fourier(n_points=70, n_freq=15, budget_s=600):
    """Optimize h using Fourier representation with differential evolution."""
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize Fourier coefficients
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Compute Fourier coefficients from initial h
        coeffs_init = np.fft.rfft(h_init)
        coeffs_init = coeffs_init.real
    else:
        # Initialize with a rectangular pulse
        coeffs_init = np.zeros(n_freq + 1)
        coeffs_init[0] = 0.5  # Approximate DC component
    
    # Bounds for Fourier coefficients
    bounds = [(-1.0, 1.0) for _ in range(n_freq + 1)]
    bounds[0] = (0.0, 1.0)  # DC component should be positive
    
    def objective(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        return compute_c5(h, dx)
    
    # Perform differential evolution
    result = differential_evolution(
        objective, 
        bounds, 
        maxiter=300, 
        popsize=75,
        mutation=(0.5, 1.0),
        recombination=0.9,
        seed=42,
        disp=False
    )
    
    # Get optimized Fourier coefficients
    optimized_coeffs = result.x
    
    # Convert back to h(x)
    h_final = fourier_to_h(optimized_coeffs, n_points)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    # Local refinement
    bounds_local = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective_local(h):
        return compute_c5(h, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result_local = minimize(
        objective_local, 
        h_final.copy(), 
        method='SLSQP', 
        bounds=bounds_local, 
        constraints=constraint, 
        options={'maxiter': 100}
    )
    
    if result_local.success:
        h_final = result_local.x
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
    
    return h_final, compute_c5(h_final, dx)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    try:
        h, c5 = optimize_fourier(n_points, 15, budget_s)
    except:
        # Fallback to previous method if optimization fails
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
        
        # Ensure normalization
        h = h * (target_sum / np.sum(h))
        
        # Local optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 100})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            
        c5 = compute_c5(h, dx)
    
    return (h.tolist(), c5, n_points)
