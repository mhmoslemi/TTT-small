import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Binary search on the value of C5
    c_low = 0.0
    c_high = 1.0
    tol = 1e-4
    
    while time.time() - start_time < budget_s - 2 and (c_high - c_low) > tol:
        c_mid = (c_low + c_high) / 2.0
        
        # Create optimization problem
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            # Compute overlap for this lag k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= c_mid / dx)
        
        # Minimize a dummy objective (feasibility check)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Feasible
                c_high = c_mid
                best_h = h_var.value
                best_c5 = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

    # Final refinement with gradient descent if time allows
    if time.time() - start_time < budget_s - 2:
        # Convert best_h to a refinement
        h = best_h.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        max_iter = 20000
        step_size = 0.005
        iteration = 0
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            iteration += 1
            
            # Compute current overlap and max
            overlap = correlate(h, 1 - h, mode="full")
            max_k = np.argmax(overlap)
            max_overlap = overlap[max_k] * dx
            
            # Store the best solution
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
                
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Compute gradient via finite differences
            g = np.zeros(n_points)
            eps_val = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                # Clamp and renormalize
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                # Compute finite difference
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Update h with gradient descent
            new_h = h - step_size * g
            
            # Apply constraints
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            h = new_h
            
            # Occasionally perturb to escape local minima
            if iteration % 100 == 0 and np.random.rand() < 0.1:
                pert = np.random.normal(0, 0.005, n_points)
                h = h + pert
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
