import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more sophisticated optimization approach
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Generate perturbed solution using multiple strategies
        candidates = []
        
        # Strategy 1: Small random perturbations
        h1 = h + np.random.normal(0, 0.01, len(h))
        h1 = np.clip(h1, 0, 1)
        h1 = normalize_h(h1, target_sum)
        candidates.append(h1)
        
        # Strategy 2: Local smoothing
        h2 = np.convolve(h, np.ones(5)/5, mode='same')
        h2 = np.clip(h2, 0, 1)
        h2 = normalize_h(h2, target_sum)
        candidates.append(h2)
        
        # Strategy 3: Gradient-like perturbation using local differences
        diff = np.diff(h, append=h[-1])
        h3 = h - 0.01 * diff
        h3 = np.clip(h3, 0, 1)
        h3 = normalize_h(h3, target_sum)
        candidates.append(h3)
        
        # Evaluate all candidates
        for candidate in candidates:
            candidate_c5 = compute_c5(candidate, dx)
            if candidate_c5 < current_c5 * 0.99999:
                h = candidate.copy()
                break
        else:
            # If no improvement, try global adjustment with small noise
            h = np.clip(h + np.random.normal(0, 0.005, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Periodic global restart
        if iteration % 100 == 0:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
