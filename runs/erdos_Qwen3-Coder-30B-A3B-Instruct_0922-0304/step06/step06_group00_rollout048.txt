import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Try SDP approach first
    try:
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
        ]
        
        # Generate all shift constraints
        full_length = 2 * n_points - 1
        shift_indices = np.arange(-n_points + 1, n_points)
        
        # Precompute shift matrix for each lag
        for k in shift_indices:
            # Create shift matrix for lag k
            if k >= 0:
                shifted_indices = np.arange(k, n_points)
                original_indices = np.arange(0, n_points - k)
            else:
                shifted_indices = np.arange(0, n_points + k)
                original_indices = np.arange(-k, n_points)
            
            # Build the constraint
            # \sum_{i} h[i] * (1 - h[(i-k) mod n]) <= C5
            if len(shifted_indices) > 0:
                # This is the diagonal part of S_k * diag(h)
                # But in linalg terms it's easier to compute it as:
                # \sum_i h[i] * (1 - h[(i-k)%n_points])
                constraint_expr = 0
                for i in range(len(shifted_indices)):
                    constraint_expr += h_var[original_indices[i]] * (1 - h_var[shifted_indices[i]])
                constraints.append(constraint_expr <= C5_var)
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 10)
        
        if prob.status in ["optimal", "optimal_inaccurate"] and C5_var.value is not None:
            h = np.array(h_var.value).flatten()
            c5_bound = C5_var.value
            return (h.tolist(), c5_bound, n_points)
    except Exception as e:
        pass  # Fall back to heuristic method
    
    # Fallback to gradient-based optimization
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    iterations = 0
    max_iter = 2000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Optimization with SLSQP
        def objective(h_vec):
            return evaluate_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        from scipy.optimize import minimize
        
        result = minimize(objective, h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (target_sum / np.sum(h_refined))
            h = h_refined
            
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Perturb
        h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iterations += 1
    
    # Final refinement
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
