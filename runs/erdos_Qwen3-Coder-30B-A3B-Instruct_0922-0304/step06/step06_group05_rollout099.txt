import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def project(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def line_search(h, g, max_iter=10):
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        current_c5 = compute_c5(h)
        for _ in range(max_iter):
            new_h = h - eta * g
            new_h = project(new_h)
            new_c5 = compute_c5(new_h)
            if new_c5 <= current_c5 - alpha * eta * np.dot(g, g):
                return new_h, eta
            eta *= beta
        return h, eta
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 50000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find k*
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Exact gradient computation for max_k
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            else:
                g[i] = -h[i - (n_points - max_k)] * dx
        
        # Line search
        h, _ = line_search(h, g)
        
        # Project and renormalize
        h = project(h)
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            noise = np.random.normal(0, 0.01 * np.exp(-iteration/1000), n_points)
            h = h + noise
            h = project(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
