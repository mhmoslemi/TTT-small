import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h to feasible region: [0,1] and sum = n_points/2"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    def compute_exact_gradient(h, k_max):
        """Compute exact gradient for the maximum overlap lag k_max"""
        g = np.zeros(n_points)
        # Gradient computation for lag k_max
        for i in range(n_points):
            if i + k_max < n_points:
                g[i] = (1 - h[i + k_max]) * dx
            else:
                # wrap around
                g[i] = (1 - h[i + k_max - n_points]) * dx
        return g

    def line_search(h, g, max_iter=10):
        """Perform line search to find optimal step size"""
        eta = 0.01
        c5_prev = compute_c5(h)
        for _ in range(max_iter):
            h_new = h - eta * g
            h_new = project_h(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < c5_prev:
                return eta, h_new
            eta *= 0.5
        return eta, h

    best_c5 = float('inf')
    best_h = h.copy()
    iteration = 0
    perturbation_interval = 500
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current overlap
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient using the maximum lag
        g = compute_exact_gradient(h, k_max)
        
        # Line search for optimal step size
        eta, h_new = line_search(h, g)
        
        # Update step
        h = h_new
        
        # Perturb periodically to escape local minima
        if iteration % perturbation_interval == 0:
            sigma = 0.01 * np.exp(-iteration / 1000.0)
            pert = np.random.normal(0, sigma, n_points)
            h = h + pert
            h = project_h(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
