import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Formulate QP problem
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == target_sum]
    constraints += [0 <= h_var[i] for i in range(n_points)]
    constraints += [h_var[i] <= 1 for i in range(n_points)]
    
    # Build all overlap constraints
    for k in range(-(n_points - 1), n_points):
        # Compute valid indices for lag k
        valid_indices = []
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                valid_indices.append((i, j))
        if valid_indices:
            # Build constraint sum of h[i] * (1 - h[i+k]) <= C * dx
            expr = 0
            for i, j in valid_indices:
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= C * dx)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solver settings
    problem = cp.Problem(objective, constraints)
    
    # Solve
    solver = cp.MOSEK
    try:
        problem.solve(solver=solver, verbose=False, max_iters=5000, time_limit=budget_s - 1)
    except:
        # Fallback if MOSEK not available
        solver = cp.GLPK
        try:
            problem.solve(solver=solver, verbose=False, max_iters=1000)
        except:
            # If it still fails, use a simpler approach with local search
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            return (best_h.tolist(), best_c5, n_points)
        
    # Extract result
    if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h_var.value
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (target_sum / np.sum(h_opt))
        final_c5 = compute_c5(h_opt, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_opt.copy()
    else:
        # Failed to solve the QP; revert to previous method 
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
