import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def build_comb_h(n_points, n_teeth, tooth_width, spacing, slope):
    """Build a comb-shaped h function."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Ensure tooth width and spacing are consistent
    tooth_width = max(0.001, tooth_width)
    spacing = max(0.001, spacing)
    
    # Compute each tooth
    for i in range(n_teeth):
        tooth_start = i * (tooth_width + spacing)
        tooth_end = tooth_start + tooth_width
        
        # Convert to indices
        start_idx = int(tooth_start / dx)
        end_idx = int(tooth_end / dx)
        
        # Ensure we don't go out of bounds
        start_idx = max(0, min(start_idx, n_points - 1))
        end_idx = max(0, min(end_idx, n_points))
        
        if end_idx > start_idx:
            # Linear transition from 0 to 1 to 0
            mid = (tooth_start + tooth_end) / 2.0
            for j in range(start_idx, end_idx):
                x = j * dx
                if x < mid:
                    h[j] = max(0, 1 - 2 * (mid - x) / tooth_width)
                else:
                    h[j] = max(0, 1 - 2 * (x - mid) / tooth_width)
    
    return h

def optimize_comb_parameters(h, n_points, dx, target_sum, bounds):
    """Optimize comb parameters: n_teeth, tooth_width, spacing, slope"""
    def objective(params):
        n_teeth, tooth_width, spacing, slope = params
        h_new = build_comb_h(n_points, int(n_teeth), tooth_width, spacing, slope)
        h_new = normalize_h(h_new, target_sum)
        return compute_c5(h_new, dx)
    
    # Default guess
    initial_params = [7.0, 2.0/n_points*7, 0.02, 0.005]
    
    # Define bounds
    bnds = [(2.0, 15.0), (0.001, 0.5), (0.001, 0.5), (0.0001, 0.1)]
    
    result = minimize(objective, initial_params, bounds=bnds, method='L-BFGS-B', options={'maxiter': 100})
    
    return result.x

    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Main loop for optimization
    max_iter = 1000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Optimize teeth configuration
        new_h = build_comb_h(n_points, 7, 2.0/n_points*7, 0.02, 0.005)
        new_h = normalize_h(new_h, target_sum)

        def constraint_sum(h):
            return np.sum(h) - target_sum
