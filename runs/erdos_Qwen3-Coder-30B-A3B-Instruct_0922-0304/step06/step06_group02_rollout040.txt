import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a solution with constraint preservation."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_pop = [np.array(initial_h_values)]
    else:
        initial_pop = [np.ones(n_points)]
        initial_pop[0][n_points//2:] = 0
        initial_pop[0] = np.clip(initial_pop[0], 0, 1)
    
    population_size = 30
    population = []
    for i in range(population_size):
        if i < len(initial_pop):
            h = initial_pop[i].copy()
        else:
            h = initial_pop[0].copy()
            h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
        
    best_c5 = float('inf')
    best_h = None
    
    # Genetic algorithm loop
    generation = 0
    max_generations = 150
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            
        # Track best solution
        for i, c5 in enumerate(fitness):
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
                
        if best_c5 <= 0.3807:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Elitism: keep top 10%
        top_10_percent = selected[:population_size//10]
        for h in top_10_percent:
            new_population.append(h.copy())
            
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            child1, child2 = crossover(parent1, parent2)
            
            # Apply mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Apply constraints
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:population_size]
        
        # Local refinement for top 10% using SLSQP
        top_candidates = sorted_indices[:population_size//10]
        for i in top_candidates:
            h = population[i].copy()
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
                
            # Localize with SLSQP
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                refined_h = result.x.copy()
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = normalize_h(refined_h, target_sum)
                
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < fitness[i]:
                    population[i] = refined_h.copy()
        
        generation += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
