import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
from sklearn.decomposition import PCA

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a reasonable initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use CMA-ES inspired approach with evolution strategy
    max_iter = 500
    popsize = min(140, max(20, n_points))
    sigma = 0.1
    mu = popsize // 2
    weights = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Normalize weights to sum to 1
    weights = weights / np.sum(weights)
    
    # Initialize mean and covariance
    mean = h.copy()
    C = np.eye(n_points) * sigma**2
    diag_weights = weights.reshape(-1, 1)
    
    # Evolution loop
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Sample population
        population = []
        for _ in range(popsize):
            sample = np.random.multivariate_normal(mean, C)
            sample = np.clip(sample, 0, 1)
            sample = normalize_h(sample, target_sum)
            population.append(sample)
            
        # Evaluate population
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
            
        # Sort by fitness (lower is better)
        idxs = np.argsort(fitness)
        best_idx = idxs[0]
        
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Update mean and covariance
        best_individuals = [population[i] for i in idxs[:mu]]
        new_mean = np.sum([weights[i] * best_individuals[i] for i in range(mu)], axis=0)
        
        # Update covariance matrix
        if iteration > 0:
            diff = np.array(best_individuals) - mean
            C = 0.9 * C + 0.1 * np.cov(diff.T)
        
        mean = new_mean
        sigma = max(0.001, sigma * 0.999)  # decay sigma slowly
    
    # Final polishing with local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        
        if final_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = final_c5
            
    return (best_h.tolist(), best_c5, n_points)
