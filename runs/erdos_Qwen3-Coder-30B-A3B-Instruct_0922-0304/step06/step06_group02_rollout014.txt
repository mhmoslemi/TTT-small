import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def evaluate_solution(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define the QP problem
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_vars) == target_sum]
    constraints += [0 <= h_vars, h_vars <= 1]
    
    # Generate all lag constraints efficiently
    for k in range(-n_points + 1, n_points):
        # Compute valid indices for lag k
        if k >= 0:
            i_indices = np.arange(n_points - k)
            j_indices = np.arange(k, n_points)
        else:
            i_indices = np.arange(-k, n_points)
            j_indices = np.arange(n_points + k)
        
        # Compute Lagrangian constraints for this k
        lag_sum = cp.sum(cp.multiply(h_vars[i_indices], 1 - h_vars[j_indices])) * dx
        constraints.append(lag_sum <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Problem setup
    problem = cp.Problem(objective, constraints)
    
    # Solve repeatedly with different initializations
    max_iter = 5000
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Try to solve the QP
        try:
            h_init = h_vars.value
            if h_init is None:
                h_init = h.copy()
            else:
                h_init = np.clip(h_init, 0, 1)
            h_vars.value = h_init
            
            # Solve the problem
            problem.solve(solver=cp.GUROBI, max_iter=1000, time_limit=10.0, verbose=False)
            
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INaccurate]:
                current_c5 = C.value
                # Check if improvement
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    h_solution = h_vars.value
                    h_solution = np.clip(h_solution, 0, 1)
                    h_solution = h_solution * (target_sum / np.sum(h_solution))
                    best_h = h_solution.copy()
                    # Early exit if good solution
                    if best_c5 <= 0.3807:
                        break
                        
        except Exception:
            pass
        
        # Perturb h to escape local minima
        noise = np.random.normal(0, 0.01, n_points)
        h = np.clip(h + noise, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iter_count += 1
    
    final_c5 = evaluate_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
