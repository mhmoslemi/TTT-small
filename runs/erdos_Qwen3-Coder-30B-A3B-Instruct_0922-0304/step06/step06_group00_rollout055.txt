import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
        h = np.clip(h, 0, 1)
    return h

        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier parameter
    mu = 1.0
    max_iter = 10000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute full correlation
        overlap = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlap)
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            # Forward difference approximation
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, target_sum)
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            max_plus = np.max(overlap_plus)
            
            grad[i] = (max_plus - max_overlap) / eps
        
        # Barrier function gradient
        barrier_grad = np.zeros(n_points)
        for i in range(n_points):
            if h[i] > 0 and h[i] < 1:
                barrier_grad[i] += -1/h[i] - 1/(1 - h[i])
        
        # Total gradient
        total_grad = grad + mu * barrier_grad
        
        # Update step
        alpha = 1.0
        step_size = 0.01
        while alpha > 1e-8:
            h_new = h - alpha * total_grad
            h_new = project_to_feasible(h_new, target_sum)
            overlap_new = correlate(h_new, 1 - h_new, mode="full")
            max_new = np.max(overlap_new)
            
            if max_new < max_overlap:
                h = h_new
                break
            else:
                alpha *= 0.5
        
        # Adaptive barrier parameter
        if iter_count % 50 == 0 and iter_count > 0:
            mu *= 0.5
        
        # Update best if improved
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:

        if no_improvement_count > max_no_improvement:

            no_improvement_count = 0
            # Restart from best solution
            h = best_h.copy()

    # Final refinement
    current_c5 = compute_c5(best_h, dx)
