```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x) space."""
    x = np.linspace(0, 2, n_points)
    # DC component + cosine terms
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs)) * np.pi * x[:, None]),
        axis=1
    )
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Try to use the initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        best_h = h.copy()
    else:
        # Initialize with a rectangular pulse-type construction
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = normalize_h(h, target_sum)
        best_h = h.copy()
    
    best_c5 = compute_c5(best_h, dx)
    
    # Fourier representation: parameters are Fourier coefficients
    n_freq = 15
    # Bounds for coefficients
    bounds = [(-1.0, 1.0)] * (n_freq + 1)  # Includes DC term
    
    # Objective function in Fourier space
    def objective(coeffs):
        h_freq = fourier_to_h(coeffs, n_points)
        # Clip to [0,1] and normalize
        h_freq = np.clip(h_freq, 0, 1)
        h_freq = normalize_h(h_freq, target_sum)
        return compute_c5(h_freq, dx)
    
    # Differential Evolution (DE) approach
    de_options = {
        'maxiter': min(300, int((budget_s - 1) / 2)), 
        'popsize': 50,
        'tol': 1e-6,
        'mutation': (0.5, 1.0),
        'recombination': 0.9,
        'polish': False,
        'workers': 1,
        'updating': 'immediate',
        'disp': False
    }
    
    result = differential_evolution(objective, bounds, **de_options)
    
    # Refine with local optimization
    coeffs_best = result.x
    h_final = fourier_to_h(coeffs_best, n_points)
    h_final = np.clip(h_final, 0, 1)
    h_final = normalize_h(h_final, target_sum)
    
    # Final local refinement with SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective_slsqp(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result_slsqp = minimize(objective_slsqp, h_final, method='SLSQP', 
                            bounds=bounds_slsqp, constraints=constraint, 
                            options={'maxiter': 100})
    
    if result_slsqp.success:
        h_final = np.clip(result_slsqp.x, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    
    # If we improved, use the final result; otherwise, return initial for robustness
    if final_c5 < best_c5:
        best_h = h_final
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
```
