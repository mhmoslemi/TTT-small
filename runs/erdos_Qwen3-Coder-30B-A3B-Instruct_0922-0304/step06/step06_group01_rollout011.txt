import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    
    # SA loop
    iteration = 0
    max_iter = 100000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Perturb h
        h_new = h.copy()
        n_perturb = max(1, n_points // 10)  # Perturb ~10% of values
        
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] = np.random.uniform(0, 1)
        
        # Ensure sum constraint
        h_new = normalize_h(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        # Compute C5 values
        c5_old = compute_c5(h, dx)
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < c5_old or np.random.rand() < np.exp((c5_old - c5_new) / T):
            h = h_new.copy()
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
        iteration += 1
    
    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Perform local refinement
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
