import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project(h):
        """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n/2"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) < 1e-10:
            return h
        # Adjust values to meet the sum constraint
        diff = n_points/2 - current_sum
        if diff > 0:
            # Need to increase some values
            # Increase in the middle to avoid boundary issues
            increase_amount = diff / n_points
            h += increase_amount
        else:
            # Need to decrease some values
            decrease_amount = -diff / n_points
            h -= decrease_amount
        # Clamp again after adjustment
        h = np.clip(h, 0, 1)
        # Re-scale
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_gradient(h, k_opt):
        """Compute gradient with respect to h for a given k_opt"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # Find which indices are included in the correlation
            if i + k_opt < n_points:
                g[i] += (1 - h[i + k_opt]) * dx
            if i - k_opt >= 0:
                g[i] -= h[i - k_opt] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with local search
    max_iter = 100000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and find k that maximizes it
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically
        g = compute_gradient(h, max_k)

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply projection to maintain feasibility
        new_h = project(new_h)
        
        h = new_h
        
        # Adaptively reduce step size
        if iteration % 1000 == 0:
            step_size *= 0.5
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
