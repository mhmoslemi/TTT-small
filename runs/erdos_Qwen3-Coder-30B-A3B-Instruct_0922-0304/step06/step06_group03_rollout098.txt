import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def discretize_h(h, levels):
    """Discretize h to nearest allowed levels."""
    h_discrete = np.zeros_like(h)
    for i in range(len(h)):
        # Find closest level
        diffs = np.abs(h[i] - np.array(levels))
        h_discrete[i] = levels[np.argmin(diffs)]
    return h_discrete

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    levels = [0.0, 0.25, 0.5, 0.75, 1.0]
    
    # Initialize with a construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        # Discretize
        h = discretize_h(h, levels)
    else:
        # Create a simple construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
        h = discretize_h(h, levels)

    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try more aggressive local search
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Perturb more aggressively
        perturbation = np.random.normal(0, 0.01, n_points)
        h_new = h + perturbation
        h_new = np.clip(h_new, 0, 1)
        # Re-normalize
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        # Discretize
        h_new = discretize_h(h_new, levels)
        
        # Accept if better or sometimes randomly
        h_new_c5 = compute_c5(h_new, dx)
        if h_new_c5 < c5 or np.random.rand() < 0.05:
            h = h_new
            
        # Sometimes do large jumps
        if iteration % 100 == 0:
            small_h = np.random.choice(levels, size=n_points)
            small_h = np.clip(small_h, 0, 1)
            current_sum = np.sum(small_h)
            if current_sum > 0:
                small_h = small_h * (target_sum / current_sum)
            h = discretize_h(small_h, levels)
            
        iteration += 1
    
    # Final refinement using local search with higher precision
    h_best = best_h.copy()
    for _ in range(1000):
        # Generate small perturbations
        pert = np.random.normal(0, 0.005, n_points)
        test_h = h_best + pert
        test_h = np.clip(test_h, 0, 1)
        current_sum = np.sum(test_h)
        if current_sum > 0:
            test_h = test_h * (target_sum / current_sum)
        test_h = discretize_h(test_h, levels)
        
        test_c5 = compute_c5(test_h, dx)
        if test_c5 < best_c5:
            best_c5 = test_c5
            h_best = test_h.copy()
            
        if time.time() - start_time > budget_s - 1:
            break
            
    if compute_c5(h_best, dx) < best_c5:
        best_c5 = compute_c5(h_best, dx)
        best_h = h_best.copy()
        
    return (best_h.tolist(), best_c5, n_points)
