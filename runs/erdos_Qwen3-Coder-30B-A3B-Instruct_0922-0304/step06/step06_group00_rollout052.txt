import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitness, tournament_size=5):
    """Tournament selection for selecting parents."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_index])
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.2, noise_scale=0.05):
    """Gaussian mutation."""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, noise_scale, len(individual))
        individual += noise
        individual = np.clip(individual, 0, 1)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        # Create base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill population with random variations
    while len(population) < population_size:
        h = population[0].copy()
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Main GA loop
    generation = 0
    max_generations = 60  # Limit to avoid infinite loops
    best_c5 = float('inf')
    best_h = None
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Selection
        selected_parents = tournament_selection(population, fitness)
        
        # Create new population: keep best individual
        new_population = [best_h.copy()]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_parents, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            # Apply mutation
            child1 = mutate(child1, 0.2, 0.05)
            child2 = mutate(child2, 0.2, 0.05)
            
            # Ensure constraints
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        # If no improvement, fall back to the best from initial run or first individual
        final_c5 = compute_c5(population[0], dx)
        best_h = population[0].copy()
    
    return (best_h.tolist(), final_c5, n_points)
