import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to target sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use CVXPY to solve QP
    # Variables
    h_var = Variable(n_points)
    C = Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1]
    constraints.append(sum(h_var) == target_sum)
    
    # Compute all lags
    max_lag = n_points - 1
    for lag in range(-max_lag, max_lag + 1):
        # Find valid indices
        if lag >= 0:
            valid_indices = range(n_points - lag)
            left = h_var[valid_indices]
            right = 1 - h_var[valid_indices + lag]
        else:
            valid_indices = range(-lag, n_points)
            left = h_var[valid_indices]
            right = 1 - h_var[valid_indices - lag]
            
        # Compute overlap
        overlap = left @ right * dx
        constraints.append(overlap <= C)
    
    # Objective
    objective = Minimize(C)
    
    # Solve problem
    prob = Problem(objective, constraints)
    
    # Try to solve
    try:
        # Set solver and timeout
        prob.solve(solver=MOSEK, verbose=False, max_iters=5000, time_limit=budget_s - 1)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_solution = h_var.value
            c5_value = C.value
            if c5_value is not None and not np.isnan(c5_value):
                h_solution = np.clip(h_solution, 0, 1)
                h_sum = np.sum(h_solution)
                if h_sum > 0:
                    h_solution = h_solution * (target_sum / h_sum)
                best_h = h_solution
                best_c5 = c5_value
    except:
        pass
    
    # If no success, fallback to local search
    if best_c5 == float('inf'):
        # Fallback to iterative local search (simplified)
        from scipy.optimize import minimize
        
        def compute_c5(h, dx):
            # Manual correlation computation
            h = np.array(h)
            n = len(h)
            result = -np.inf
            for k in range(1-n, n):
                # Compute correlation at lag k
                if k >= 0:
                    corr = np.sum(h[:n-k] * (1 - h[k:])) * dx
                else:
                    corr = np.sum(h[-k:] * (1 - h[:-k])) * dx
                result = max(result, corr)
            return result
    
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Start from a better initial guess
        h_current = best_h.copy()
        
        # Local search
        max_iter = 5000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Add small noise
            h_noisy = h_current + np.random.normal(0, 0.01, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = h_noisy * (target_sum / np.sum(h_noisy))
            
            try:
                result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h_candidate = result.x
                    h_candidate = np.clip(h_candidate, 0, 1)
                    h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
                    c5_candidate = compute_c5(h_candidate, dx)
                    if c5_candidate < best_c5:
                        best_c5 = c5_candidate
                        best_h = h_candidate
                        if best_c5 <= 0.3807:
                            break
            except:
                pass
            
            # Update current solution
            h_current = best_h.copy()
            
            if i % 100 == 0:
                # Regularly check against time limit
                if time.time() - start_time > budget_s - 1:
                    break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
