import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Cutting-plane technique
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if iteration % 100 == 0:
            # Re-evaluate current best
            current_c5 = compute_c5(h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
        
        # Find the worst lag and its overlap
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps)
        worst_lag_idx = np.argmax(overlaps)
        # For full mode, we need to map back to actual lag indices
        # For size 2N-1 full overlap, the middle is lag=0
        lag = worst_lag_idx - (n_points - 1)
        
        # Create constraint for this lag only
        # we'll try some optimization
        def objective(h):
            return compute_c5(h, dx)
            
        # Use advanced optimization with constraints
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        # Add a constraint that forces max overlap at this lag to be minimal
        # We do this by using the worst lag as a guide and doing local perturbation
        h_noisy = h + np.random.normal(0, 0.02, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        # Use COBYLA or SLSQP as sometimes robust
        method = 'SLSQP'
        try:
            result = minimize(objective, h_noisy, method=method, bounds=bounds, constraints=cons,
                              options={'maxiter': 50})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5 or np.random.rand() < 0.1:
                    h = h_candidate.copy()
        except:
            # fallback to a simple local search
            h = np.clip(h + np.random.normal(0, 0.02, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
        
        if iteration % 30 == 0:
            # Periodic global search
            h = np.clip(h + 0.05 * (np.random.rand(n_points) - 0.5), 0, 1)
            h = normalize_h(h, target_sum)
        
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
