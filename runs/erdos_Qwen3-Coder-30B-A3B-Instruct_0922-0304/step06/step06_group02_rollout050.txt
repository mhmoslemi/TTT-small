import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize the initial h
    h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    max_iter = 300
    inertia_start = 0.8
    inertia_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        # Initialize particle with small random perturbations
        particle_h = h + np.random.normal(0, 0.02, n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = normalize_h(particle_h, target_sum)
        particles.append(particle_h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(particle_h.copy())
        pbest_vals.append(compute_c5(particle_h, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_vals)
    global_best = pbest[global_best_idx].copy()
    global_best_val = pbest_vals[global_best_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia * velocities[i] + 
                           c1 * r1 * (pbest[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Update particle
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate particle
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = current_c5
                
                # Update global best
                if current_c5 < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = current_c5
                    
        # Check for best solution found so far
        if global_best_val < best_c5:
            best_c5 = global_best_val
            best_h = global_best.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local refinement with L-BFGS-B
        if iteration % 5 == 0 and iteration > 0:
            # Refine global best using L-BFGS-B
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, global_best, method='L-BFGS-B', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 50})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_h = normalize_h(refined_h, target_sum)
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < global_best_val:
                    global_best = refined_h.copy()
                    global_best_val = refined_c5
                    
                    if refined_c5 < best_c5:
                        best_c5 = refined_c5
                        best_h = refined_h.copy()
            
            # Stop early if improvement is minimal
            if iteration > 10 and global_best_val > best_c5 * 0.999:
                break
    
    # Final local optimization
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h, method='L-BFGS-B', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 200})
    
    if result.success:
        final_h = np.clip(result.x, 0, 1)
        final_h = normalize_h(final_h, target_sum)
        final_c5 = compute_c5(final_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
