import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement_count = 0
    best_in_generation = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all individuals
        fitness = []
        for individual in population:
            c5 = evaluate_individual(individual, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            best_in_generation = best_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Crossover and mutation
        new_population = [population[0]]  # Elitism
        
        # Selection: top 50%
        selected = population[:population_size // 2]
        selected_fitness = fitness[:population_size // 2]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Crossover
            if np.random.rand() < 0.8 and len(selected) >= 2:
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                p1, p2 = selected[i1], selected[i2]
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
                new_population.extend([child1, child2])
            else:
                # Mutation
                parent = selected[np.random.randint(0, len(selected))]
                child = parent + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = normalize_h(child, target_sum)
                new_population.append(child)
        
        # Trim to exact size
        while len(new_population) > population_size:
            new_population.pop()
            
        # Update population
        population = new_population
        
        generation += 1
        if no_improvement_count > 20:
            # Increase population size if no improvement
            if population_size < 40:
                population_size = 40
                # Reinitialize with more diverse candidates
                population = [best_individual]
                for _ in range(population_size - 1):
                    noise = np.random.normal(0, 0.1, n_points)
                    h = best_individual + noise
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    population.append(h)
        
        # Check if we're stuck
        if generation > 100:
            if abs(best_in_generation - best_c5) < 1e-6:
                no_improvement_count += 1
            else:
                best_in_generation = best_c5
                no_improvement_count = 0
            if no_improvement_count > 10:
                break
                
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
