import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Convex optimization with projected gradient descent
    max_iter = 1000
    alpha = 0.01
    epsilon = 1e-3
    momentum = 0.9
    
    v = np.zeros_like(base_h)
    h = base_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient via finite differences
        c5_current = compute_c5(h, dx)
        grad = np.zeros_like(h)
        
        # Simple finite difference approximation of gradient
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            c5_plus = compute_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = normalize_h(h_minus, target_sum)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update with momentum
        v = momentum * v - alpha * grad
        h = h + v
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Evaluate
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
            
        if best_c5 <= 0.3807:
            break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
