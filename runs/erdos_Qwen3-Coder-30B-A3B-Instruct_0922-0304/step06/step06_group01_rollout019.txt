import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        # Default construction
        h = np.zeros(n_points)
        n_teeth = 7
        tooth_width = 2.0 / n_teeth
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
        # Create a comb-like structure
        for i in range(n_teeth):
            start_pos = i * (tooth_width + spacing)
            end_pos = start_pos + tooth_width
            # Convert to discrete indices
            start_idx = int(np.round(start_pos / dx))
            end_idx = int(np.round(end_pos / dx))
            if start_idx < 0:
                start_idx = 0
            if end_idx > n_points:
                end_idx = n_points
            if start_idx < end_idx:
                # Create a triangular transition
                mid_idx = (start_idx + end_idx) // 2
                for j in range(start_idx, end_idx):
                    if j < mid_idx:
                        h[j] = (j - start_idx) / (mid_idx - start_idx)
                    else:
                        h[j] = (end_idx - j) / (end_idx - mid_idx)
        # Normalize
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameter search space
    param_space = [
        {"n_teeth": 5, "tooth_width": 0.4, "spacing": 0.05, "slope": 0.1},
        {"n_teeth": 6, "tooth_width": 0.33, "spacing": 0.04, "slope": 0.08},
        {"n_teeth": 7, "tooth_width": 0.285, "spacing": 0.03, "slope": 0.07},
        {"n_teeth": 8, "tooth_width": 0.25, "spacing": 0.02, "slope": 0.06},
    ]

    # Sweep through comb parameters
    for params in param_space:
        if time.time() - start_time > budget_s - 5:
            break
            
        # Construct h from parameters
        h_cand = np.zeros(n_points)
        n_teeth = params["n_teeth"]
        tooth_width = params["tooth_width"]
        spacing = params["spacing"]
        slope = params["slope"]
        
        for i in range(n_teeth):
            start_pos = i * (tooth_width + spacing)
            end_pos = start_pos + tooth_width
            start_idx = max(0, int(np.round(start_pos / dx)))
            end_idx = min(n_points, int(np.round(end_pos / dx)))
            
            if start_idx < end_idx:
                mid_idx = (start_idx + end_idx) // 2
                for j in range(start_idx, end_idx):
                    if j < mid_idx:
                        h_cand[j] = max(0, 1.0 - (j - start_idx) / (slope * (mid_idx - start_idx)))
                    else:
                        h_cand[j] = max(0, 1.0 - (end_idx - j) / (slope * (end_idx - mid_idx)))
        
        # Normalize
        current_sum = np.sum(h_cand)
        if current_sum > 0:
            h_cand = h_cand * (target_sum / current_sum)
            
        # Local refinement with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h_cand.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            current_sum = np.sum(h_refined)
            if current_sum > 0:
                h_refined = h_refined * (target_sum / current_sum)
        else:
            h_refined = h_cand
            
        c5 = compute_c5(h_refined, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
            
        if best_c5 <= 0.3807:
            break

    # Final tuning
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        current_sum = np.sum(final_h)
        if current_sum > 0:
            final_h = final_h * (target_sum / current_sum)
    else:
        final_h = best_h.copy()
        
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
