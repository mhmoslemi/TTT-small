import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.ones(n_points) * 0.5  # midpoint initialization
    
    # Normalize
    base_h = base_h * (target_sum / np.sum(base_h))
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Alternating optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Optimize h while fixing overlaps
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Add noise to perturb
        noise = np.random.normal(0, 0.03, n_points)
        h = base_h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Optimize
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200, 'ftol': 1e-6})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (target_sum / np.sum(h))
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update base_h for next iteration
        base_h = best_individual.copy()
        
        iteration += 1
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
