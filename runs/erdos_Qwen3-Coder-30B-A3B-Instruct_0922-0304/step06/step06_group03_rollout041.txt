import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Perturbation parameters
    noise_amplitude = 0.02
    noise_decay = 0.95
    perturb_interval = 50
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Cross-correlation driven perturbation
        if iteration % perturb_interval == 0:
            # Compute cross-correlation to identify overlapping regions
            corr = correlate(h, 1 - h, mode="full")
            # Find strongest peaks (top 10%) 
            top_peaks = np.argsort(corr)[-max(1, n_points//10):]
            # Create mask to focus perturbation
            mask = np.zeros(n_points)
            for pk in top_peaks:
                # Add a window of influence around the peak
                start_idx = max(0, pk - n_points//8)
                end_idx = min(n_points, pk + n_points//8)
                mask[start_idx:end_idx] = 1
            
            # Apply perturbation with mask
            noise = np.random.normal(0, noise_amplitude, n_points)
            h = h + mask * noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Decrease noise amplitude over time
            noise_amplitude *= noise_decay
            
        # Local refinement using SLSQP
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = result.x
            else:
                # Fallback to simple perturbation if optimization fails
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        except:
            # Fallback to simple perturbation
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
