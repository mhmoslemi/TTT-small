import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    target_sum = n_points / 2

        h = np.clip(h, 0, 1)

        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization using CVXPY for QP formulation
    # Variables
    h_var = Variable(n_points)
    C = Variable(1)
    
    # Create constraints
    constraints = []
    
    # Sum constraint
    constraints.append(sum(h_var) == target_sum)
    
    # Bound constraints
    constraints.extend([h_var[i] >= 0 for i in range(n_points)])
    constraints.extend([h_var[i] <= 1 for i in range(n_points)])
    
    # Overlap constraints for all lags
    # For each lag k, compute sum h[i] * (1 - h[i+k]) <= C * dx
    # This is done by scanning over all valid indices and lags
    for k in range(-n_points + 1, n_points):
        # Find valid range for i such that i + k is valid
        if k >= 0:
            # i ranges from 0 to n_points - 1 - k
            valid_i = range(0, n_points - k)
            expr = sum(h_var[i] * (1 - h_var[i + k]) for i in valid_i)
        else:
            # i ranges from -k to n_points - 1 (but limited by valid indices)
            valid_i = range(-k, n_points)
            expr = sum(h_var[i] * (1 - h_var[i + k]) for i in valid_i)
        constraints.append(expr <= C * dx)
    
    # Objective
    objective = Minimize(C)
    
    # Problem formulation
    problem = Problem(objective, constraints)
    
    # Solve with solver
    solver_options = {'max_iters': 1000, 'verbose': False}
    result = problem.solve(solver=MOSEK, **solver_options)
    
    if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
        h_opt = h_var.value

            # Normalize to correct sum
            h_opt = h_opt * (target_sum / np.sum(h_opt))
            
            # Compute final C5
            # Evaluate overlap integral with all k
            overlap = np.correlate(h_opt, 1 - h_opt, mode='full')
            final_c5 = np.max(overlap) * dx
            
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_opt.copy()
                
    # If QP failed, fall back to gradient-based refinement with careful constraint enforcement
    if best_c5 == float('inf'):
        # Set up bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            # Do not use expensive correlation in gradient step; just use final check with true integral
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * dx
                
        # Use a basic local search with constraint enforcement
        iteration = 0
        max_iter = 10000
        h_current = h.copy()
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            # Compute new value for C5
            current_c5 = objective(h_current)
            
            # Update best
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_current.copy()

            h_new = h_current.copy()

            h_new += noise
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            # Check if we improved
            new_c5 = objective(h_new)
            if new_c5 <= current_c5:
                h_current = h_new.copy()
            
            iteration += 1

    # Final check with true objective
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), final_c5, n_points)
