import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            particle += np.random.normal(0, 0.05, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle += np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(n_particles)]
    
    # Personal bests
    p_best = [particle.copy() for particle in particles]
    p_best_c5 = [compute_c5(particle, dx) for particle in particles]
    
    # Global best
    g_best_idx = np.argmin(p_best_c5)
    g_best = p_best[g_best_idx].copy()
    g_best_c5 = p_best_c5[g_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update particles
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (g_best - particle))
            
            # Update position
            particle = particle + vel
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store updated particle
            particles[i] = particle
            velocities[i] = vel
            
            # Update personal best
            c5 = compute_c5(particle, dx)
            if c5 < p_best_c5[i]:
                p_best[i] = particle.copy()
                p_best_c5[i] = c5
                
                # Update global best
                if c5 < g_best_c5:
                    g_best = particle.copy()
                    g_best_c5 = c5
                    
                    if g_best_c5 <= 0.3807:
                        break
                        
        # Periodic refinement with gradient-based method
        if iteration % 100 == 0 and iteration > 0:
            # Select top 10 particles for refinement
            top_indices = np.argsort(p_best_c5)[:10]
            for idx in top_indices:
                # Use SLSQP for refinement
                bounds = [(0, 1) for _ in range(n_points)]
                def objective(h):
                    return compute_c5(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, p_best[idx], method='SLSQP',

                                options={'maxiter': 100})
                if result.success:
                    refined_h = result.x
                else:
                    refined_h = p_best[idx]
                    
                refined_h = np.clip(refined_h, 0, 1)
                refined_h = normalize_h(refined_h, target_sum)
                c5 = compute_c5(refined_h, dx)
                
                # Update if better
                if c5 < p_best_c5[idx]:
                    p_best[idx] = refined_h
                    p_best_c5[idx] = c5

                        g_best_c5 = c5

        if iteration % 50 == 0 and iteration > 0:
            # Apply high-pass filter to particles
            for i in range(n_particles):
                particle = particles[i]
                # FFT
                fft_h = np.fft.fft(particle)
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter
                cutoff = 0.3
                filter_mask = np.abs(freqs) > cutoff
                filtered_fft = fft_h * filter_mask
                # IFFT
                filtered_h = np.real(np.fft.ifft(filtered_fft))
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                
                # Replace worst particles
                if i >= n_particles - 25:
                    particles[i] = filtered_h
    
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
