import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Try SDP approach
    try:
        # Define variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            C5_var >= 0
        ]
        
        # Compute all shift constraints
        full_length = 2 * n_points - 1
        lags = np.arange(-n_points + 1, n_points)
        
        # Precompute shift matrices for every lag
        for k in lags:
            shift_idx = np.arange(n_points)
            if k >= 0:
                shifted_idx = (shift_idx - k) % n_points
            else:
                shifted_idx = (shift_idx + k) % n_points
                
            # Build objective for this shift
            # We compute: sum(h[i] * (1 - h[shifted_idx[i]])) * dx <= C5
            # For this, we need to model each cross term
            # The direct approach using semidefinite programming with quadratic constraints
            # is more complex than necessary for this task.
            
            # Just create the constraint directly
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_var[shifted_idx])) * dx
            constraints.append(constraint_expr <= C5_var)
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Form and solve the problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with MOSEK (assuming available)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 1)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_optimized = h_var.value
            c5_bound = C5_var.value
            
            # Validate solution
            if (np.all(h_optimized >= 0) and np.all(h_optimized <= 1) and
                abs(np.sum(h_optimized) - target_sum) < 1e-6):
                return (h_optimized.tolist(), float(c5_bound), n_points)
            else:
                print("SDP solution not feasible, falling back to local optimization")
        else:
            print("SDP not solved to optimality, falling back to local optimization")
        
    except Exception as e:
        print(f"SDP error: {e}")
    
    # Fall back to local optimization approach
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    iterations = 0
    max_iter = 2000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Simple gradient ascent step with projected gradient
        # Approximate gradient via finite differences
        epsilon = 1e-4
        grad = np.zeros(n_points)
        
        # For each dimension, compute finite difference
        current_val = compute_c5(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1, h[i] + epsilon)
            
            h_minus = h.copy()
            h_minus[i] = max(0, h[i] - epsilon)
            
            # Apply normalization for shifted values
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            val_plus = compute_c5(h_plus, dx)
            val_minus = compute_c5(h_minus, dx)
            
            grad[i] = (val_plus - val_minus) / (2 * epsilon)
        
        # Update with gradient step (reversed since we want to minimize)
        lr = 0.1
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))  # Re-normalize
        
        # If improvement, accept it
        current_c5 = compute_c5(h_new, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Reduce learning rate
            lr *= 0.98
        
        if best_c5 <= 0.3807:
            break
        
        iterations += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    if final_c5 <= 0.3807:
        return (best_h.tolist(), float(final_c5), n_points)
    else:
        return (best_h.tolist(), float(final_c5), n_points)
