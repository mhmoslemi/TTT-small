import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def perturb_h(h, target_sum, perturb_fraction=0.1):
    """Perturb h by adjusting a fraction of its values."""
    n_points = len(h)
    indices = np.random.choice(n_points, int(perturb_fraction * n_points), replace=False)
    h_new = h.copy()
    h_new[indices] = np.random.uniform(0, 1, len(indices))
    h_new = normalize_h(h_new, target_sum)
    return np.clip(h_new, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-8
    sa_steps = 0
    max_sa_steps = 100000
    
    # Local refinement parameters
    local_refine_interval = 10000
    local_refine_steps = 200
    
    while time.time() - start_time < budget_s - 1 and sa_steps < max_sa_steps:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Perturb h
        h_new = perturb_h(h, target_sum)
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject the new h
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Time-based early exit
        if best_c5 <= 0.3807:
            break
            
        # Annealing schedule
        sa_steps += 1
        if sa_steps % 100 == 0:
            T = max(min_T, T * cooling_rate)
            
        # Local refinement
        if sa_steps % local_refine_interval == 0 and sa_steps > 0:
            # Define bounds and constraints for SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize_h(h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
