import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_population(population, dx):
    """Evaluate all individuals in the population."""
    fitness = np.array([compute_c5(individual, dx) for individual in population])
    return fitness

def tournament_selection(population, fitness, tournament_size=5):
    """Perform tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size)
        tournament_fitness = fitness[tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform uniform crossover."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    mask = np.random.rand(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(individual, mutation_rate, noise_factor):
    """Apply Gaussian mutation."""
    mask = np.random.rand(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_factor, len(individual))
    individual = individual + mask * noise
    return np.clip(individual, 0, 1)

def local_optimize(individual, target_sum, dx, max_iter=50):
    """Apply local optimization using SLSQP."""
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(individual))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, individual, method='SLSQP', 
                      bounds=bounds, constraints=constraint, 
                      options={'maxiter': max_iter})
    
    if result.success:
        optimized = result.x
        optimized = np.clip(optimized, 0, 1)
        optimized = normalize_h(optimized, target_sum)
        return optimized
    else:
        return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 100
    mutation_rate_start = 0.2
    mutation_rate_decay = 0.95
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    generation = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate the population
        fitness = evaluate_population(population, dx)
        
        # Track best individual
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Selection
        selected = tournament_selection(population, fitness, tournament_size)
        
        # Create new population through crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Apply local optimization to offspring
            child1 = local_optimize(child1, target_sum, dx)
            child2 = local_optimize(child2, target_sum, dx)
            
            # Apply mutation
            mutation_rate = mutation_rate_start * (mutation_rate_decay ** generation)
            noise_factor = 0.05 * (mutation_rate_decay ** generation)
            child1 = mutate(child1, mutation_rate, noise_factor)
            child2 = mutate(child2, mutation_rate, noise_factor)
            
            new_population.append(child1)
            new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Final refinement to get best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
