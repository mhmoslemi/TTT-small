import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, offset=0):
    """Create a comb pattern with given spacing and offset."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Determine number of teeth based on spacing and offset
    tooth_positions = []
    pos = offset * dx
    while pos <= 2.0:
        idx = int(pos / dx)
        if idx < n_points:
            tooth_positions.append(idx)
        pos += spacing * dx
    
    # Set tooth heights to 1
    for idx in tooth_positions:
        if idx < n_points:
            h[idx] = 1.0
    
    # normalize
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbation."""
    noise = np.random.normal(0, sigma, len(h))
    h_perturbed = h + noise
    return np.clip(h_perturbed, 0, 1)

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70

    target_sum = n_points / 2

    if h is not None:

        h = normalize_h(h, target_sum)
    else:
        # Create initial comb pattern
        spacing = 3
        h = create_comb_pattern(n_points, spacing, 0)
    
    # Track best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Multi-resolution comb search
    spacing_options = [2, 3, 4, 5, 6]
    max_iterations = 20000
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Generate neighbor comb patterns
        new_h_candidates = []
        for spacing in spacing_options:
            new_h = create_comb_pattern(n_points, spacing, 0)
            new_h_candidates.append(new_h)
            
        # Add perturbations
        for _ in range(3):
            perturbed = perturb_pattern(h)
            new_h_candidates.append(perturbed)
            
        # Evaluate candidates
        for candidate in new_h_candidates:
            candidate = np.clip(candidate, 0, 1)
            candidate = normalize_h(candidate, target_sum)
            c5_candidate = compute_c5(candidate, dx)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = candidate.copy()
                h = candidate.copy()
                
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        if result.success:
            h = result.x
        else:
            # perturb if optimization fails
            h = perturb_pattern(h)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Periodically increase resolution for better precision
        if iteration % 100 == 0 and iteration > 0:
            n_points_high = 140
            dx_high = 2.0 / n_points_high
            h_high = np.interp(np.linspace(0, 2, n_points_high),

            h = h_high
            target_sum_high = n_points_high / 2

            n_points = n_points_high
            dx = dx_high
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
