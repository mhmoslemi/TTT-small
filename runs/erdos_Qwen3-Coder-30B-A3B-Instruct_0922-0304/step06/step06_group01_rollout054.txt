import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h function."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for k in range(n_freq):
        if k == 0:
            h += coeffs[k]
        else:
            h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of frequency components

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = project_and_normalize(h, target_sum)
    else:
        # Simple rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = project_and_normalize(h, target_sum)

    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles with Fourier coefficients
    particles = []
    velocities = []
    pbest_coeffs = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        coeffs = np.random.uniform(-1, 1, 2*n_freq)
        particles.append(coeffs)
        velocities.append(np.zeros(2*n_freq))
        pbest_coeffs.append(coeffs.copy())
        pbest_c5.append(float('inf'))

    # Global best
    gbest_coeffs = None
    gbest_c5 = float('inf')
    
    # PSO loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            # Convert coefficients to h
            h_candidate = fourier_to_h(particles[i], n_points, n_freq)
            h_candidate = project_and_normalize(h_candidate, target_sum)
            
            # Compute C5
            c5 = compute_c5(h_candidate, dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest_coeffs[i] = particles[i].copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_coeffs = particles[i].copy()
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest_coeffs[i] - particles[i]) + 
                             c2 * r2 * (gbest_coeffs - particles[i]))
            particles[i] += velocities[i]
            
    # Refine best solution with local optimization
    if gbest_coeffs is not None:
        refined_coeffs = gbest_coeffs.copy()
        
        def objective(coeffs):
            h_refined = fourier_to_h(coeffs, n_points, n_freq)
            h_refined = project_and_normalize(h_refined, target_sum)
            return compute_c5(h_refined, dx)
        
        # Bound the search space for the coefficients
        bounds = [(-2, 2) for _ in range(2*n_freq)]
        
        # Local optimization
        result = minimize(objective, refined_coeffs, method='L-BFGS-B', bounds=bounds, options={'maxiter': 100})
        if result.success:
            refined_coeffs = result.x
            
        # Final evaluation
        h_final = fourier_to_h(refined_coeffs, n_points, n_freq)
        h_final = project_and_normalize(h_final, target_sum)
        final_c5 = compute_c5(h_final, dx)
    else:
        # Fallback to original h
        h_final = h.copy()
        final_c5 = compute_c5(h_final, dx)

    return (h_final.tolist(), final_c5, n_points)
