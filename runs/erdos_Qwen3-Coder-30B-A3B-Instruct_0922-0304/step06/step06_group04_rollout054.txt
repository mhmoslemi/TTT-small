import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return compute_c5(h, dx)
    
    # Initialize with the parent solution if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create a simple starting point
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Multi-start optimization with different initializations
    num_starts = 20
    for i in range(num_starts):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random initialization with different distributions
        if i == 0:
            # Start with the base solution
            h = base_h.copy()
        else:
            # Random initialization
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            
        # Add some structure
        if i % 3 == 0:
            # Small perturbations
            noise = np.random.normal(0, 0.03, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        elif i % 3 == 1:
            # Gaussian-like shape
            x = np.linspace(0, 2, n_points)
            mean = 1.0
            std = 0.3
            gaussian = np.exp(-0.5 * ((x - mean) / std) ** 2)
            h = normalize_h(gaussian, target_sum)
        else:
            # Piece-wise constant with randomness
            h = np.zeros(n_points)
            segments = 4
            segment_size = n_points // segments
            for j in range(segments):
                start_idx = j * segment_size
                end_idx = (j + 1) * segment_size if j < segments - 1 else n_points
                h[start_idx:end_idx] = np.random.rand() * 2
            h = normalize_h(h, target_sum)
        
        # Local optimization using SLSQP
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 200})
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h_opt.copy()
        except:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # Final refinement of the best solution
    try:
        result = minimize(objective, best_individual, method='SLSQP', 
                         bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_final = result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            final_c5 = compute_c5(h_final, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_individual = h_final.copy()
    except:
        pass
                    
    return (best_individual.tolist(), best_c5, n_points)
