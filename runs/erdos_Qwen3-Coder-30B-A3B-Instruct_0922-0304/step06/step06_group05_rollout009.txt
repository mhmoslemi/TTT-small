import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: smooth step-like function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with enhanced strategy
    max_iter = 30000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find the maximizing lag
        overlap = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient w.r.t. the max lag
        g = np.zeros(n_points)
        if k_max < n_points:
            # Gradient contribution from the first part of the correlation
            for i in range(n_points):
                if i + k_max < n_points:
                    g[i] += (1 - h[i + k_max]) * dx
                else:
                    # Wrap around case, but in our computation it's wrapped anyway
                    g[i] += (1 - h[(i + k_max) % n_points]) * dx
        else:
            # In full mode, for the right half of the output
            for i in range(n_points):
                if i >= k_max - n_points + 1:
                    g[i] += -h[i - (k_max - n_points + 1)] * dx
                else:
                    pass  # Often 0 from the symmetric definition of convolution
        
        # Update with momentum
        velocity = momentum * prev_grad + g
        new_h = h - step_size * velocity
        prev_grad = g.copy()
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h

        # Adaptive step size and momentum decay
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum = max(0.5, momentum * 0.98)
        
        # Occasionally add noise to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.05:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
