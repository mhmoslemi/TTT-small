import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points):
    """Convert Fourier coefficients to h(x) using cosine basis."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 + np.sum(
        coeffs[1:] * np.cos(np.arange(1, len(coeffs))[:, None] * np.pi * x),
        axis=0
    )
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15

    # Initialize using the parent construction if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = normalize_h(h_initial, target_sum)
        # Convert to Fourier coefficients
        x = np.linspace(0, 2, n_points)
        coeffs_initial = np.fft.fft(h_initial)
        coeffs_initial = np.real(coeffs_initial[:n_freq+1])
    else:
        # Initialize with a simple step function
        h_initial = np.ones(n_points)
        h_initial[n_points//2:] = 0
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = normalize_h(h_initial, target_sum)
        coeffs_initial = np.fft.fft(h_initial)
        coeffs_initial = np.real(coeffs_initial[:n_freq+1])

    # Define the objective function for DE
    def objective(coeffs):
        h = fourier_to_h(coeffs, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    # Optimization bounds for coefficients
    bounds = [(-1.0, 1.0)] * (n_freq + 1)
    
    # Differential evolution for global search
    result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=70,
        seed=seed,
        disp=False,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.9,
    )
    
    # Best solution from DE
    best_coeffs = result.x
    h_best = fourier_to_h(best_coeffs, n_points)
    h_best = np.clip(h_best, 0, 1)
    h_best = normalize_h(h_best, target_sum)

    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds_h = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result_refine = minimize(
        objective,
        h_best.copy(),
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    
    if result_refine.success:
        h_final = result_refine.x
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    else:
        h_final = h_best

    final_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
