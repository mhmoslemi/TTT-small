import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initialization: step function
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Discretization levels: use more levels to allow finer exploration
    levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    iteration = 0
    max_iter = 30000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Perform a local search step by slightly modifying h values
        # Pick random indices to modify
        indices = np.random.choice(n_points, size=min(10, n_points), replace=False)
        
        # Modify selected indices with small random changes, clipping to [0,1]
        h_new = h.copy()
        for idx in indices:
            # Add small random noise
            change = np.random.normal(0, 0.03)
            h_new[idx] += change
            # Clip to valid range
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Re-normalize
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        h = h_new
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
