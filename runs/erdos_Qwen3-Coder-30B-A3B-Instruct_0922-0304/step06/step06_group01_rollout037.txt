import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-6
    sa_steps = 0
    max_sa_steps = 100000
    
    # Local refinement parameters
    local_refinement_freq = 10000
    local_refinement_iter = 200
    
    while time.time() - start_time < budget_s - 2 and sa_steps < max_sa_steps:
        # Compute current overlap
        current_c5 = compute_c5(h, dx)
        
        # Perturb h: adjust 10% of values
        indices_to_change = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new = h.copy()
        for i in indices_to_change:
            h_new[i] = np.random.random()
        
        # Ensure sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new overlap
        new_c5 = compute_c5(h_new, dx)
        
        # Accept or reject based on SA criteria
        delta = new_c5 - current_c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new.copy()
        
        # Update temperature
        T = max(T * cooling_rate, min_T)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
        
        sa_steps += 1
        
        # Local refinement every 10k iterations
        if sa_steps % local_refinement_freq == 0 and sa_steps > 0:
            # Apply SLSQP local optimization
            def objective(h):
                return compute_c5(h, dx)
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': local_refinement_iter})
            
            if result.success:
                h = result.x.copy()
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                
                # Update best solution after refinement
                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
