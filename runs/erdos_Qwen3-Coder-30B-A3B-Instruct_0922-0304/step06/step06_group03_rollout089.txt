import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    h[tooth_positions] = tooth_height
    return h

def apply_perturbation(h, sigma=0.02):
    """Apply small random perturbations to h."""
    pert = np.random.normal(0, sigma, len(h))
    h = h + pert
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try comb patterns with different spacings
    spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    
    for spacing in spacings:
        pattern = create_comb_pattern(n_points, spacing)
        pattern = normalize_h(pattern, target_sum)
        comb_patterns.append(pattern)
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate the current best solution
        c5 = compute_c5(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Select random comb pattern for optimization
        selected_pattern = comb_patterns[np.random.randint(len(comb_patterns))]
        h = selected_pattern.copy()

        # Small random perturbations to escape local optima
        h = apply_perturbation(h, sigma=0.02)
        h = normalize_h(h, target_sum)
        
        # Optimization step
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else:
                h = apply_perturbation(h, sigma=0.01)
                h = normalize_h(h, target_sum)
        except:
            h = apply_perturbation(h, sigma=0.01)
            h = normalize_h(h, target_sum)
            
        iteration += 1
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
