import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 20
    mutation_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_individual = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Adjust mutation rate
            mutation_rate = max(0.1, mutation_rate * 0.8)
            no_improvement_count = 0
        
        if best_c5 <= 0.3807:
            break
        
        # Select top half for breeding
        keep_count = population_size // 2
        selected = population[:keep_count]
        
        # Create new population
        new_population = selected.copy()
        
        # Elitism: keep best
        new_population.append(selected[0].copy())
        
        # Crossover
        while len(new_population) < population_size:
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            h1, h2 = selected[idx1], selected[idx2]
            
            # Single point crossover
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
            child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            
            new_population.extend([child1, child2])
        
        # Keep size
        new_population = new_population[:population_size]
        
        # Mutation
        num_mutations = int(mutation_rate * population_size)
        for _ in range(num_mutations):
            idx = np.random.randint(population_size)
            h = new_population[idx]
            noise = np.random.normal(0, 0.05, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            new_population[idx] = h
            
        population = new_population
        generation += 1
    
    if best_individual is None:
        # Fallback to initial
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
        else:
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
            best_individual = normalize_h(best_individual, target_sum)
    
    final_c5 = evaluate_individual(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
