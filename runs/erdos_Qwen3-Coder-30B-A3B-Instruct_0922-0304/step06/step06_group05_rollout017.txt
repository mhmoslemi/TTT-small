        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 500
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Normalize h to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    while time.time() - start_time < budget_s - 1:
        # Perturb h slightly
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure we're not changing the same element
        if i == j:
            j = (j + 1) % n_points
        
        # Ensure we can move the values
        if h[i] >= 1 or h[j] <= 0:
            continue
            
        # Compute change
        delta = min(1 - h[i], h[j])
        if delta == 0:
            continue
            
        # Apply perturbation
        new_h = h.copy()
        new_h[i] += delta
        new_h[j] -= delta
        
        # Re-normalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        new_c5 = compute_c5(new_h)
        
        # Metropolis criterion
        if new_c5 < best_c5:

            best_c5 = new_c5
            best_h = new_h.copy()

            if np.random.rand() < np.exp(-delta_c5 / T):
                h = new_h
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Cooling schedule
        T *= alpha
        
        # If stagnation detected, increase temperature to escape local minimum
        if stagnation_count > max_stagnation:
            T = min(T * 1.2, 1.0)
            stagnation_count = 0
