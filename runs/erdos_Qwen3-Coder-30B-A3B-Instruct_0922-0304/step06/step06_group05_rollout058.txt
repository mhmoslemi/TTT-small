import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h, n_points):
        """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones(n_points) * (n_points / 2) / n_points
        return h

    def compute_exact_gradient(h, k_star):
        """Compute exact gradient for the max overlap at lag k_star"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] * (1 - h[(i + k_star) % n_points])
            if i + k_star < n_points:
                g[i] += (1 - h[i + k_star]) * dx
            else:
                # Wrap around case
                g[i] += (1 - h[i + k_star - n_points]) * dx
            
            # Contribution from (1-h[i]) * h[(i - k_star) % n_points]  
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
            else:
                # Wrap around case
                g[i] -= h[i - k_star + n_points] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and find max k
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the max k
        g = compute_exact_gradient(h, max_k)
        
        # Update with momentum
        v = momentum * g_prev + g
        new_h = h - step_size * v
        
        # Update g_prev for next iteration
        g_prev = g.copy()
        
        # Apply constraints
        new_h = project_h(new_h, n_points)
        
        h = new_h
        
        # Adaptive step size
        if iteration % 1000 == 0 and step_size > 1e-6:
            step_size *= 0.5
            
        # Adaptive momentum
        if iteration % 2000 == 0 and momentum > 0.5:
            momentum *= 0.9
            
        # Occasionally perturb to escape local minima
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert
            h = project_h(h, n_points)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
