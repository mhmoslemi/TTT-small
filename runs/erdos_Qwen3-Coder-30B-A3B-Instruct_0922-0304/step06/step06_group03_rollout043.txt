import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the given parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Perturb the function
        if iteration % 50 == 0:
            # Re-evaluate overlap to identify regions that contribute most to C5
            overlap = correlate(h, 1 - h, mode="full")
            # Identify top 10% of peak locations
            top_indices = np.argsort(overlap)[-max(1, n_points//10):]
            mask = np.zeros(n_points)
            # Increase probability of perturbing regions where overlap peaks occur
            for i in top_indices:
                # Use full range of overlap indices to map to h indices
                idx = i - (n_points - 1)  # Shift to center at 0
                if 0 <= idx + n_points//2 < n_points:
                    mask[idx + n_points//2] = 1.0
                    
            # Apply perturbation
            noise = np.random.normal(0, noise_amplitude, n_points)
            h_perturbed = h + mask * noise
            h = np.clip(h_perturbed, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            # Small random perturbation
            noise = np.random.normal(0, noise_amplitude, n_points)
            h_perturbed = h + noise
            h = np.clip(h_perturbed, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Occasionally decrease noise amplitude
        if iteration % 500 == 0 and noise_amplitude > 0.001:
            noise_amplitude *= 0.95
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
