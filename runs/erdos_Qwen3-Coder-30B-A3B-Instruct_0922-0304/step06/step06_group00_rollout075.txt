import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h = h + np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Blend two parents to create offspring."""
    if np.random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    # Blend segments
    crossover_point = np.random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_indices = np.random.choice(len(population), tournament_size)
    best_index = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        # Create base function
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1
        initial_h = normalize_h(initial_h, target_sum)
    
    # Create initial population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = mutate_individual(initial_h, 0.05)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    mutation_strength = 0.2
    generation = 0
    max_generations = 60
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate all individuals in the population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population through selection, crossover and mutation
        new_population = []
        sorted_indices = np.argsort(fitnesses)
        
        # Elitism: keep top 10%
        num_elite = population_size // 10
        for i in range(num_elite):
            new_population.append(population[sorted_indices[i]])
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutate children
            child1 = mutate_individual(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            child2 = mutate_individual(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Decrease mutation strength
        mutation_strength *= 0.98
        
        generation += 1
    
    # Final evaluation of best individual
    best_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
