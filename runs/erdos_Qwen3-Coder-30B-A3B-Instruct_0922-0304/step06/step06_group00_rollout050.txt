import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual with Gaussian noise."""
    h += np.random.normal(0, mutation_strength, len(h))
    h = np.clip(h, 0, 1)
    return h

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return p1.copy(), p2.copy()
    
    crossover_point = np.random.randint(1, len(p1) - 1)
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    else:
        # Create base function with structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill rest of population with diverse samples
    for _ in range(population_size - 1):
        h = np.random.random(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    # GA parameters
    generations = 0
    max_generations = 600
    mutation_strength = 0.1
    mutation_decay_rate = 0.995
    
    while time.time() - start_time < budget_s - 1 and generations < max_generations:
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individual
        elite_idx = np.argmin(fitnesses)
        new_population.append(population[elite_idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate_individual(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Adapt mutation strength
        mutation_strength *= mutation_decay_rate
        
        generations += 1
    
    # Final check
    final_fitness = compute_c5(best_h, dx)
    return (best_h.tolist(), final_fitness, n_points)
