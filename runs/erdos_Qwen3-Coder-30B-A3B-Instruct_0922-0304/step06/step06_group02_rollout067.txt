import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate(h, rate=0.05):
    """Apply Gaussian mutation with constraint maintenance."""
    h_mut = h + np.random.normal(0, 0.02, len(h))
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover(h1, h2, rate=0.7):
    """Perform crossover between two solutions."""
    if np.random.random() > rate:
        return h1.copy(), h2.copy()
    crossover_point = np.random.randint(1, len(h1)-1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return h1_new, h2_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from global's initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback: simple step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_perturbed = mutate(h, mutation_rate)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        population.append(h_perturbed)
    
    generation = 0
    max_generations = 150
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate all candidates in population
        fitness_scores = []
        for h_candidate in population:
            c5 = compute_c5(h_candidate, dx)
            fitness_scores.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best if needed
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Select top candidates for reproduction
        selected_indices = sorted_indices[:population_size // 2]
        selected_pop = [population[i] for i in selected_indices]
        
        # Elitism: carry over top 10%
        new_population = [population[i].copy() for i in sorted_indices[:elite_size]]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1_idx, parent2_idx = np.random.choice(selected_indices, 2, replace=False)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate)
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate(child2, mutation_rate)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Keep only population_size number
        population = new_population[:population_size]
        
        # Local refinement with SLSQP for top elite
        for i in range(elite_size):
            h_elite = population[i].copy()
            
            # Define constraint and objective for SLSQP
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            # Add slight noise to avoid local minima issues
            h_noisy = h_elite + np.random.normal(0, 0.005, len(h_elite))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = normalize_h(refined, target_sum)
                c5_refined = compute_c5(refined, dx)
                if c5_refined < fitness_scores[i]:
                    population[i] = refined
            
        generation += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
