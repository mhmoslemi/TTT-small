import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Discretized values
    discrete_levels = np.array([0, 0.25, 0.5, 0.75, 1.0])
    
    # Initialize from existing construction or use a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Discretize to nearest allowed value
        h = discrete_levels[np.argmin(np.abs(h[:, None] - discrete_levels), axis=1)]
    else:
        # Simple step function initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Discretize to allowed levels
    h = discrete_levels[np.argmin(np.abs(h[:, None] - discrete_levels), axis=1)]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Heuristic search with random perturbations and local improvements
    attempts = 0
    max_attempts = 20000
    
    while time.time() - start_time < budget_s - 1 and attempts < max_attempts:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Local search: randomly modify some elements and re-discretize
        # Make changes to a random subset of elements
        indices_to_change = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        for idx in indices_to_change:
            # Randomly adjust the value and discretize again
            h[idx] += np.random.normal(0, 0.1)
            h[idx] = max(0, min(1, h[idx]))
            h[idx] = discrete_levels[np.argmin(np.abs(h[idx] - discrete_levels))]
            
        # Re-normalize the sum to meet constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Discretize again
        h = discrete_levels[np.argmin(np.abs(h[:, None] - discrete_levels), axis=1)]
        
        attempts += 1

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
