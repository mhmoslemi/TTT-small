import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate):
    """Mutate h with given rate."""
    mask = np.random.random(len(h)) < mutation_rate
    h[mask] += np.random.normal(0, 0.05, np.sum(mask))
    return np.clip(h, 0, 1)

def crossover_h(h1, h2, crossover_rate):
    """Crossover two h vectors."""
    mask = np.random.random(len(h1)) < crossover_rate
    h_child = np.where(mask, h1, h2)
    return h_child

def tournament_selection(population, fitness, tournament_size=5):
    """Select an individual via tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return population[best_idx]

def refine_h(h, target_sum, max_iter=50):
    """Refine h using projected gradient method."""
    h = normalize_h(h, target_sum)
    for _ in range(max_iter):
        # Simple gradient step in direction of steepest descent
        # This is simplified; a full projection method would be better
        grad = np.zeros_like(h)
        # Approximate gradient using finite differences
        eps = 1e-5
        base_c5 = compute_c5(h, 2.0/len(h))
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            plus_c5 = compute_c5(h_plus, 2.0/len(h))
            grad[i] = (plus_c5 - base_c5) / eps
        # Update with small step
        step_size = 0.01
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        if compute_c5(h_new, 2.0/len(h)) < compute_c5(h, 2.0/len(h)):
            h = h_new
        else:
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available

        h = np.clip(h, 0, 1)

    else:
        # Base construction
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # GA parameters
    population_size = 100
    mutation_rate_start = 0.2
    crossover_rate = 0.8
    max_generations = 1000
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_h(h.copy(), 0.1)
        h_mut = normalize_h(h_mut, target_sum)
        population.append(h_mut)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for ind in population:
            fit = compute_c5(ind, dx)
            fitness.append(fit)
        
        # Track best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Selection
        new_population = []
        for _ in range(population_size):
            parent1 = tournament_selection(population, fitness)
            parent2 = tournament_selection(population, fitness)
            
            # Crossover
            child = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            mutation_rate = mutation_rate_start * np.exp(-gen / 500.0)
            child = mutate_h(child, mutation_rate)
            
            # Normalize and refine
            child = np.clip(child, 0, 1)
            child = normalize_h(child, target_sum)
            child = refine_h(child, target_sum)
            
            new_population.append(child)
        
        population = new_population
    
    # Final refinement of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
