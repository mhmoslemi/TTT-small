import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct a comb-like h function with specified parameters."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 3.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        start_x = i * (tooth_width + spacing)
        end_x = start_x + tooth_width
        
        start_idx = int(start_x / dx)
        end_idx = int(end_x / dx)
        
        # Ensure indices are within bounds
        start_idx = max(0, start_idx)
        end_idx = min(n_points, end_idx)
        
        if start_idx >= n_points or end_idx <= 0:
            continue
        
        # Create triangular transition
        for j in range(start_idx, end_idx):
            x = j * dx
            if x < start_x or x > end_x:
                continue
            
            # Linear interpolation for teeth
            if x <= start_x + slope:
                val = 1.0 - (x - start_x) / slope
            elif x >= end_x - slope:
                val = 1.0 - (end_x - x) / slope
            else:
                val = 1.0
            
            h[j] += max(0, val)
    
    return h

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize based on the parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        n_teeth = 7
        h = construct_comb_h(n_points, n_teeth=n_teeth)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Tune comb parameters
    iterations = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Try different comb structures
        n_teeth = 5 + (iterations % 6)  # Cycle between 5 to 10 teeth
        tooth_width = 2.0 / n_teeth
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
        slope = tooth_width / 3.0
        
        h = construct_comb_h(n_points, n_teeth=n_teeth, 
                            tooth_width=tooth_width, 
                            spacing=spacing, 
                            slope=slope)
        h = normalize_h(h, target_sum)
        
        # Local refinement with SLSQP
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        def objective(h):
            return compute_c5(h, dx)
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                h = h_new.copy()
        except:
            pass  # Fall back to current solution if optimization fails
            
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final local refinement
    if best_c5 > 0.3807:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
        
        def objective(h):
            return compute_c5(h, dx)
        
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                final_c5 = compute_c5(h_new, dx)
                if final_c5 < best_c5:
                    best_h = h_new.copy()
                    best_c5 = final_c5
        except:
            pass  # Continue with previous best

    return (best_h.tolist(), best_c5, n_points)
