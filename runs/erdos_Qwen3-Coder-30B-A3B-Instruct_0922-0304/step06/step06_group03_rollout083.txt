import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            # Start nearby the initial solution
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.random.rand(n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle.copy())
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_c5.append(float('inf'))
    
    # Global best
    g_best = particles[0].copy()
    g_best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            c5 = compute_c5(particle, dx)
            
            if c5 < p_best_c5[i]:
                p_best[i] = particle.copy()
                p_best_c5[i] = c5
                
            if c5 < g_best_c5:
                g_best = particle.copy()
                g_best_c5 = c5
                
        # Early stopping
        if g_best_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (p_best[i] - particles[i]) +
                c2 * r2 * (g_best - particles[i])
            )
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
    # Final evaluation
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
