import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-8
    sa_steps = 100000
    perturb_frac = 0.1
    
    # Store original h for reseeding if needed
    original_h = h.copy()
    
    # SA loop
    for step in range(sa_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h - adjust 10% of the values
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(n_points * perturb_frac), replace=False)
        for i in indices:
            # Small random change
            delta = np.random.normal(0, 0.1)
            h_new[i] += delta
            
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Compute overlap difference
        c5_old = compute_c5(h, dx)
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject the new solution
        delta = c5_new - c5_old
        if delta < 0 or random.random() < np.exp(-delta / temp):
            h = h_new.copy()
            c5 = c5_new
        else:
            c5 = c5_old
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Cooling schedule
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, best_h.copy(), method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 200})
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_h = normalize_h(refined_h, target_sum)
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        except:
            pass
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
