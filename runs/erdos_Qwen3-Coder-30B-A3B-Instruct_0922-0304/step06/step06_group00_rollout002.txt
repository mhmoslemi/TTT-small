import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_initial_population(n_points, pop_size, target_sum, initial_h=None):
    """Create an initial population of feasible h functions."""
    population = []
    if initial_h is not None:
        h = np.array(initial_h)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Fill with random feasible solutions
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        # Clip and renormalize to ensure feasibility
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), k)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, sigma=0.05):
    """Mutate an individual."""
    if np.random.rand() > mutation_rate:
        return h
    
    noise = np.random.normal(0, sigma, len(h))
    h = h + noise
    return np.clip(h, 0, 1)

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    population = create_initial_population(n_points, pop_size, target_sum, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(population[current_best_idx])
        
        # Generate rest via selection, crossover, and mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            # Apply mutations
            child1 = mutate(child1, sigma=0.05 * np.exp(-generation/200))
            child2 = mutate(child2, sigma=0.05 * np.exp(-generation/200))
            
            # Ensure feasibility
            child1 = normalize_h(child1, target_sum)
            child1 = np.clip(child1, 0, 1)
            
            child2 = normalize_h(child2, target_sum)
            child2 = np.clip(child2, 0, 1)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
