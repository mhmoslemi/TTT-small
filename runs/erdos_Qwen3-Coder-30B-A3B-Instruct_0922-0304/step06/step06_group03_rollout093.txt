import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a piecewise constant function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Initialize noise amplitude
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to identify high overlap regions
        corr = correlate(h, 1 - h, mode="full")
        peak_indices = np.argsort(-np.abs(corr))[:n_points//4]  # Top 25% peaks
        
        # Create perturbation mask focusing on high overlap regions
        mask = np.zeros(n_points)
        for idx in peak_indices:
            # Convert full correlation index to center around 0
            if idx < len(corr) // 2:
                offset = idx - len(corr) // 2
            else:
                offset = idx - len(corr) // 2
            
            # Apply mask within a window around this offset
            for i in range(max(0, offset + n_points//2 - 3), min(n_points, offset + n_points//2 + 4)):
                if abs(i - (offset + n_points//2)) < 2:
                    mask[i] = 1.0
        
        # Apply perturbation
        noise = np.random.normal(0, noise_amplitude, n_points)
        h = h + mask * noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Do local optimization with SLSQP occasionally
        if iteration % 5 == 0:
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Local optimization
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        
        # Gradually decrease noise amplitude
        if iteration > 0 and iteration % 500 == 0:
            noise_amplitude *= 0.95
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
