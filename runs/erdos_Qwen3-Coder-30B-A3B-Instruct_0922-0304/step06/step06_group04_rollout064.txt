import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
    
    # Normalize to target sum
    base_h = base_h * (target_sum / np.sum(base_h))
    
    best_c5 = compute_c5(base_h, dx)
    best_individual = base_h.copy()
    
    max_iters = 50
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        # Try to improve using an alternating optimization approach
        h = best_individual.copy()
        improvement_found = False
        
        # Try different perturbations
        for _ in range(5):
            h_pert = h.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h_pert += noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (target_sum / np.sum(h_pert))
            
            # Use a simpler local optimization approach
            # Try random shuffling of segments
            segments = []
            segment_size = max(1, n_points // 10)
            for i in range(0, n_points, segment_size):
                segments.append(h_pert[i:i+segment_size])
            
            # Shuffle segments
            np.random.shuffle(segments)
            h_new = np.concatenate(segments)
            h_new = h_new * (target_sum / np.sum(h_new))
            
            c5 = compute_c5(h_new, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h_new.copy()
                improvement_found = True
                break
                
        if not improvement_found:
            # Try a more systematic approach - linear programming relaxation
            h = best_individual.copy()
            
            # Try to solve via relaxed bilinear program (simplified)
            # We'll use a simpler hill-climbing with mutations
            
            # Create perturbed version
            h_mutate = h.copy()
            # Randomly flip some small segments
            for _ in range(3):
                start_idx = np.random.randint(0, n_points - 5)
                end_idx = min(start_idx + 5, n_points)
                h_mutate[start_idx:end_idx] = 1 - h_mutate[start_idx:end_idx]
                
            h_mutate = np.clip(h_mutate, 0, 1)
            h_mutate = h_mutate * (target_sum / np.sum(h_mutate))
            
            c5 = compute_c5(h_mutate, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h_mutate.copy()
                improvement_found = True
                
        iter_count += 1
        
        if best_c5 <= 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
