import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient descent with smoothing and line search
    # Objective: minimize max_k correlate(h, 1-h, mode='full')[k] * dx
    
    def surrogate_loss(h, eps=1e-3):
        overlap = correlate(h, 1 - h, mode="full")
        # Use Huber loss to approximate max with differentiable function
        # For max approximation in smooth form: max(x) = lim_{p->inf} (1/p) * log(sum(exp(p*x)))
        # But since we are using autocorrelation, use direct max.
        # This is an approximation trick to make it differentiable.
        # For now, just use the max directly in the derivative approach below.
        return np.max(overlap) * dx
    
    # Nesterov-like momentum approach with backtracking line search
    momentum = 0.9
    m = np.zeros_like(base_h)
    alpha = 0.1
    beta = 0.9
    tolerance = 1e-6
    
    h = base_h.copy()
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient approximation using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Ensure they're still in bounds
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            h_minus = normalize_h(h_minus, target_sum)
            
            loss_plus = surrogate_loss(h_plus)
            loss_minus = surrogate_loss(h_minus)
            grad[i] = (loss_plus - loss_minus) / (2 * eps)
        
        # Accelerated gradient step (Nesterov momentum)
        m = momentum * m + grad
        h_new = h - alpha * m
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Backtracking line search
        c5_new = surrogate_loss(h_new)
        while c5_new > surrogate_loss(h) and alpha > 1e-10:
            alpha *= beta
            h_new = h - alpha * m
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            c5_new = surrogate_loss(h_new)
            
        h = h_new.copy()
        c5 = surrogate_loss(h)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Update step size
        if iteration % 10 == 0:
            alpha *= 0.99
            if alpha < 1e-8:
                alpha = 1e-8
    
    # Final check with the proper C5 computation
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
