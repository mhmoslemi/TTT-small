import numpy as np
from scipy.signal import correlate
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h_m = h + np.random.normal(0, mutation_strength, len(h))
    h_m = np.clip(h_m, 0, 1)
    return h_m

def crossover_parents(h1, h2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = random.randint(1, len(h1) - 1)
    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = random.sample(range(len(population)), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    else:
        # Create base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Fill up population with random variations
    for _ in range(population_size - 1):
        h = population[0].copy()
        h = mutate_individual(h, 0.05)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    # Evaluate initial population
    fitnesses = [compute_c5(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    generation = 0
    max_generations = 60  # Approximately 60 seconds per generation
    mutation_strength = 0.05
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Create new population
        new_population = []
        
        # Keep best individuals
        sorted_indices = np.argsort(fitnesses)
        new_population.extend([population[i].copy() for i in sorted_indices[:20]])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            child2 = mutate_individual(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        new_population = new_population[:population_size]
        
        # Evaluate new population
        fitnesses = [compute_c5(h, dx) for h in new_population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = new_population[min_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Update population
        population = new_population
        
        # Adapt mutation strength
        mutation_strength *= 0.98
        if mutation_strength < 0.001:
            mutation_strength = 0.001
            
        generation += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
