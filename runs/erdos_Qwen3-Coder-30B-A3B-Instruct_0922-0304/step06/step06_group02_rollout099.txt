import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use cutting-plane method with local refinement
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        worst_c5 = overlap[worst_lag_idx] * dx
        
        # Solve optimization problem with constraint for that lag only
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def constraint_worst_lag(h):
            # Compute overlap for the worst lag
            h_padded = np.pad(h, (0, n_points), mode='constant', constant_values=0)
            h_shifted = np.roll(h_padded, worst_lag_idx - n_points + 1)
            overlap_val = np.sum(h * (1 - h_shifted[:n_points]))
            return overlap_val * dx - worst_c5
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint1 = {'type': 'eq', 'fun': constraint_sum}
        constraint2 = {'type': 'ineq', 'fun': lambda x: -constraint_worst_lag(x)}
        
        def objective(h):
            return np.max(correlate(h, 1 - h, mode="full")) * dx
            
        # Try multiple random restarts
        best_candidate = h.copy()
        best_candidate_c5 = current_c5
        for _ in range(3):
            h_noisy = h + np.random.normal(0, 0.05, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            try:
                result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=[constraint1, constraint2], options={'maxiter': 100})
                if result.success:
                    candidate = result.x
                    candidate = np.clip(candidate, 0, 1)
                    candidate = normalize_h(candidate, target_sum)
                    candidate_c5 = compute_c5(candidate, dx)
                    if candidate_c5 < best_candidate_c5:
                        best_candidate = candidate.copy()
                        best_candidate_c5 = candidate_c5
            except:
                continue
                
        h = best_candidate.copy()
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
