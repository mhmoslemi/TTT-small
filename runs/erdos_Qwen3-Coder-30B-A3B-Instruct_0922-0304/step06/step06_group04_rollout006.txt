import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.random(n_points)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement_count = 0
    mutation_rate = 0.2
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Elitism: Keep the best individual
        if best_individual is not None:
            selected_population.append(best_individual)
            
        # Crossover
        new_population = selected_population[:]
        
        # Fill remaining slots with offspring
        while len(new_population) < population_size:
            if len(selected_population) >= 2:
                parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
                # Single-point crossover
                point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:point], parent2[point:]])
                child2 = np.concatenate([parent2[:point], parent1[point:]])
                new_population.extend([child1, child2])
            else:
                # If not enough candidates, just perturb
                ind = selected_population[0] if selected_population else population[0]
                child = ind + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = normalize_h(child, target_sum)
                new_population.append(child)
        
        # Control size
        if len(new_population) > population_size:
            new_population = new_population[:population_size]
            
        # Mutation
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                h = new_population[i] + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
                
        # Adjust mutation rate based on performance
        if no_improvement_count > 20:
            mutation_rate = min(0.5, mutation_rate * 1.1)
        elif no_improvement_count > 10:
            mutation_rate = max(0.05, mutation_rate * 0.9)
            
        population = new_population
        generation += 1
        
    # Re-evaluate final solution for consistency
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        final_c5 = float('inf')
        
    return (best_individual.tolist() if best_individual is not None else np.zeros(n_points).tolist(), 
            final_c5, n_points)
