import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller discretization for faster search
    n_points = 60
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Allowed discrete values
    allowed_values = [0.0, 0.25, 0.5, 0.75, 1.0]
    
    # Initialize with the known construction if available
    if 'initial_h_values' in globals():
        # Discretize the initial solution to allowed values
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Map to closest discrete allowed value
        h_discrete = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h])
        h = h_discrete
    else:
        # Default initialization
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1.0
        h = np.clip(h, 0, 1)
    
    # Normalize to exact sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    # Discretize
    h = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h])
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Perform local search with randomized perturbations
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Random small change to h
        idx = np.random.randint(0, n_points)
        old_val = h[idx]
        new_val = np.random.choice(allowed_values)
        h[idx] = new_val
        
        # Ensure sum constraint by adjusting the last element
        current_sum = np.sum(h)
        if current_sum != target_sum:
            diff = target_sum - current_sum
            h[-1] += diff
        
        # Keep all values within allowed range
        h = np.clip(h, 0, 1)
        h = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h])
        
        iteration += 1
        
        # Sometimes do a larger random perturbation
        if iteration % 50 == 0:
            idx_start = np.random.randint(0, n_points - 10)
            for i in range(10):
                if idx_start + i < n_points:
                    h[idx_start + i] = np.random.choice(allowed_values)
                    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
