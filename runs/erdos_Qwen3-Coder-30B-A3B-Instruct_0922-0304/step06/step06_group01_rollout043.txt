import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None, transition_shape='triangular'):
    """Construct a comb-like h function."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    tooth_positions = np.arange(n_teeth) * (tooth_width + spacing)
    
    for i in range(n_teeth):
        start_x = tooth_positions[i]
        end_x = start_x + tooth_width
        
        start_idx = int(start_x / dx)
        end_idx = int(end_x / dx)
        
        if start_idx >= n_points:
            break
        if end_idx >= n_points:
            end_idx = n_points - 1
            
        for j in range(start_idx, end_idx + 1):
            x = j * dx
            if x < start_x:
                # Left transition (if exists)
                if start_x - x < slope:
                    h[j] = max(0, 1 - (start_x - x) / slope)
            elif x > end_x:
                # Right transition (if exists)
                if x - end_x < slope:
                    h[j] = max(0, 1 - (x - end_x) / slope)
            else:
                # Inside the tooth
                h[j] = 1.0
    
    # Normalize
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        # Use the existing construction from the parent code
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a basic comb structure
        h = construct_comb_h(n_points, n_teeth=7, tooth_width=2.0/70, 
                            spacing=(2.0 - 7*(2.0/70))/(7+1), 
                            slope=(2.0/70)/2.0, 
                            transition_shape='triangular')

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Hyperparameters for comb scan
    n_teeth_list = [5, 6, 7, 8, 9]
    tooth_width_range = [0.01, 0.2]
    slope_range = [0.005, 0.1]
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try different comb configurations
        if iteration % 50 == 0:
            # Perform comb construction search
            for n_teeth in n_teeth_list:
                # Vary tooth width and slope
                w = np.linspace(tooth_width_range[0], tooth_width_range[1], 5)
                s = np.linspace(slope_range[0], slope_range[1], 5)
                for tooth_w in w:
                    for sl in s:
                        # Check constraints on spacing
                        spacing = (2.0 - n_teeth * tooth_w) / (n_teeth + 1)
                        if spacing > 0 and tooth_w > 0:
                            temp_h = construct_comb_h(n_points, n_teeth, tooth_w, spacing, sl)
                            c5 = compute_c5(temp_h, dx)
                            if c5 < best_c5:
                                best_c5 = c5
                                best_h = temp_h.copy()
        
        # One of two phases:
        # 1. Local search on current best
        # 2. Random improvement step for escape
        
        if iteration % 10 == 0:
            # Random tweak for exploration
            noise = np.random.normal(0, 0.005, n_points)
            h = best_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            h = best_h.copy()
        
        # Perform local optimization with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h_vals):
            return np.sum(h_vals) - target_sum
            
        def objective(h_vals):
            return compute_c5(h_vals, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h.copy(), method='SLSQP', 
                              bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                h = h_new.copy()
        except:
            pass
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
