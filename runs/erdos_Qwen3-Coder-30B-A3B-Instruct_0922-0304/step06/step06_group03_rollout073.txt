import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the known construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Discretize to allowed values {0, 0.25, 0.5, 0.75, 1}
    allowed_values = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    def discretize_h(h):
        # For each element, find the closest allowed value
        return np.array([allowed_values[np.argmin(np.abs(allowed_values - val))] for val in h])
    
    h = discretize_h(h)
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))  # Rescale to have correct sum
    
    # Initial evaluation
    best_c5 = compute_c5(h, dx)
    current_h = h.copy()
    
    # Discretization-based local search
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if iteration % 100 == 0:
            # Occasionally reset to a better known solution
            if 'initial_h_values' in globals():
                temp_h = np.array(initial_h_values)
                temp_h = np.clip(temp_h, 0, 1)
                temp_h = temp_h * (target_sum / np.sum(temp_h))
                temp_h = discretize_h(temp_h)
                c5 = compute_c5(temp_h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    current_h = temp_h.copy()
        
        # Perturb the solution
        pert_h = current_h.copy()
        if np.random.rand() < 0.3:
            # Change a few random entries
            change_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            for idx in change_indices:
                old_val = pert_h[idx]
                # Choose new value from allowed set
                new_val = allowed_values[np.random.randint(len(allowed_values))]
                pert_h[idx] = new_val
        else:
            # Small random changes
            noise = np.random.normal(0, 0.02, n_points)
            pert_h = pert_h + noise
            pert_h = np.clip(pert_h, 0, 1)
            
        # Discretize and rescale to maintain constraint
        pert_h = discretize_h(pert_h)
        pert_h = pert_h * (target_sum / np.sum(pert_h))
        
        # Evaluate
        new_c5 = compute_c5(pert_h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            current_h = pert_h.copy()
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final check of best solution
    final_c5 = compute_c5(current_h, dx)
    
    return (current_h.tolist(), final_c5, n_points)
