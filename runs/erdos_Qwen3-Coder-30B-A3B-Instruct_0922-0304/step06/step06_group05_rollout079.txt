import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clamp values to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Simple projection to keep sum equal to n_points/2
            adjustment = (n_points / 2 - current_sum) / n_points
            h = h + adjustment
            h = np.clip(h, 0, 1)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically
        g = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from lag k*
            if (i + max_k) < n_points:
                g[i] += (1 - h[(i + max_k) % n_points]) * dx
            else:
                g[i] -= h[i - max_k] * dx
                
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply projection to maintain constraints
        new_h = project_h(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_h(h)
            
        # Reduce step size periodically
        if iteration % 1000 == 0:
            step_size *= 0.5

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
