import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    # Normalize to target sum
    h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = compute_c5(h_base, dx)
    best_h = h_base.copy()
    
    # Use bilinear programming approach
    solver = cp.MOSEK
    
    for trial in range(20):  # Budget limited number of trials
        if time.time() - start_time > budget_s - 5:
            break
            
        # Define variables
        h = cp.Variable(n_points)
        z = cp.Variable(n_points + n_points - 1)  # Full correlation length
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Bilinear constraints: z_k = sum_{i} h[i] * (1 - h[i+k])
        # Corresponding to correlation computation
        for k in range(-n_points+1, n_points):
            idx_start = max(0, k)
            idx_end = min(n_points, n_points + k)
            if idx_start < idx_end:
                # For this shift k, compute the sum with indices
                if k >= 0:
                    expr = cp.sum(cp.multiply(h[idx_start:idx_end], 1 - h[:idx_end-idx_start]))
                else:
                    expr = cp.sum(cp.multiply(h[-k:], 1 - h[:len(h)+k]))
                # Align the index of z
                z_idx = k + n_points - 1
                constraints.append(z[z_idx] == expr)
        
        objective = cp.Minimize(cp.max(z) * dx)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                if h_sol is not None:
                    c5 = compute_c5(h_sol, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h_sol.copy()
                        
                    if best_c5 <= 0.3807:
                        break
                else:
                    # Fallback: try simplex/local optimization
                    pass
            else:
                # Restart with perturbed solution
                h_base = best_h + np.random.normal(0, 0.01, n_points)
                h_base = np.clip(h_base, 0, 1)
                h_base = h_base * (target_sum / np.sum(h_base))
        except Exception:
            # Fall back to perturbation
            pass
            
        # Perturb the best solution and repeat
        h_base = best_h + np.random.normal(0, 0.01, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
