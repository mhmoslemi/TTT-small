import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_transform(h, n_freq):
    """Compute Fourier coefficients."""
    n_points = len(h)
    coeffs = []
    for k in range(n_freq):
        real = np.sum(h * np.cos(k * np.pi * np.arange(n_points) / n_points))
        imag = np.sum(h * np.sin(k * np.pi * np.arange(n_points) / n_points))
        coeffs.append(real)
        coeffs.append(imag)
    return np.array(coeffs)

def inverse_fourier(coeffs, n_points, n_freq):
    """Reconstruct h from Fourier coefficients."""
    h = np.zeros(n_points)
    for i in range(n_points):
        x = i / n_points
        for k in range(n_freq):
            h[i] += coeffs[2*k] * np.cos(k * np.pi * x) + coeffs[2*k+1] * np.sin(k * np.pi * x)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    n_particles = 50
    for _ in range(n_particles):
        # Generate random Fourier coefficients
        coeffs = np.random.uniform(-1, 1, 2 * n_freq)
        particles.append(coeffs)
        velocities.append(np.random.uniform(-0.1, 0.1, 2 * n_freq))
        pbest.append(coeffs.copy())
        pbest_vals.append(float('inf'))
    
    gbest = particles[0].copy()
    gbest_val = float('inf')
    
    # PSO parameters
    w, c1, c2 = 0.7, 1.5, 1.5
    
    iteration = 0
    max_iter = 500
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(n_particles):
            coeffs = particles[i]
            # Reconstruct h
            h_candidate = inverse_fourier(coeffs, n_points, n_freq)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Compute C5
            c5 = compute_c5(h_candidate, dx)
            
            # Update particle best
            if c5 < pbest_vals[i]:
                pbest[i] = coeffs.copy()
                pbest_vals[i] = c5
            
            # Update global best
            if c5 < gbest_val:
                gbest = coeffs.copy()
                gbest_val = c5
        
        # Update particle velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            vel = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            particles[i] += vel
            velocities[i] = vel
            
        iteration += 1
        
        # Early exit
        if gbest_val <= 0.3807:
            break
    
    # Final optimization with SLSQP
    h_final = inverse_fourier(gbest, n_points, n_freq)
    h_final = np.clip(h_final, 0, 1)
    h_final = normalize_h(h_final, target_sum)
    
    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h_final.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
