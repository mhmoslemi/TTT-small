import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    try:

        h_var = cp.Variable(n_points)

        from scipy.optimize import minimize
        
        def objective(h):
            return compute_c5(h, dx)

                    c5_val = compute_c5(h, dx)
