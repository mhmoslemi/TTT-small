```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cma

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        h = base_h.copy()
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()

    # Use CMA-ES with adaptive steps
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        c5 = compute_c5(h, dx)
        return c5

    # CMA-ES parameters
    opts = cma.CMAOptions()
    opts.set('verbose', -1)
    opts.set('popsize', min(140, 4 * n_points))
    opts.set('tolfun', 1e-8)
    opts.set('tolx', 1e-8)

    # Initial mean is h, but it must be on [0, 1]
    init_h = np.clip(h, 0, 1)
    init_h = normalize_h(init_h, target_sum)
    
    # Optimize using CMA-ES
    es = cma.CMAEvolutionStrategy(init_h, 0.1, opts)
    cma_iter = 0
    max_cma_iter = 500
    
    while not es.stop() and time.time() - start_time < budget_s - 1 and cma_iter < max_cma_iter:
        # Generate a batch
        X = es.ask()
        fit_vals = []
        
        for x in X:
            h = np.clip(x, 0, 1)
            h = normalize_h(h, target_sum)
            fit_vals.append(objective(h))
            
        es.tell(X, fit_vals)
        cma_iter += 1
        
        # Get the current best from the strategy
        current_h = np.clip(es.best_x, 0, 1)
        current_h = normalize_h(current_h, target_sum)
        current_c5 = compute_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break

    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum

    def objective_slsqp(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective_slsqp, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < best_c5:
            best_h = refined_h.copy()
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
```
