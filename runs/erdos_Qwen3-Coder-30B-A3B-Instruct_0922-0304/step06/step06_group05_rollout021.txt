import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Initialize best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient with respect to the maximum lag
        g = np.zeros(n_points)
        max_k_idx = max_k - (n_points - 1)  # convert from full correlation indexing to shift indexing
        
        # The gradient is non-zero only for indices that are part of the maximum overlap term
        for i in range(n_points):
            # Contribution to gradient from term h[i] * (1 - h[i + max_k_idx]) 
            if (i + max_k_idx) < n_points and (i + max_k_idx) >= 0:
                g[i] = (1 - h[i + max_k_idx]) * dx
            # Contribution to gradient from term h[i] * (-h[i - max_k_idx])
            elif (i - max_k_idx) < n_points and (i - max_k_idx) >= 0:
                g[i] = -h[i - max_k_idx] * dx
            else:
                g[i] = 0
                
        # Perform backtracking line search
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        max_linesearch_iter = 10
        
        for _ in range(max_linesearch_iter):
            # Compute trial solution
            new_h = h - eta * g
            
            # Apply constraints
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Compute new objective value
            new_c5 = compute_c5(new_h)
            
            # Armijo condition check
            if new_c5 <= max_overlap - alpha * eta * np.dot(g, g):
                break
            else:
                eta *= beta
                
        # Update h 
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.3:
            # Add small random noise
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
