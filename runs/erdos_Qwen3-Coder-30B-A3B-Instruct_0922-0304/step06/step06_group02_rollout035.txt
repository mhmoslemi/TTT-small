import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))
    
    # Create variable for h and C
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h_var) == target_sum]
    constraints += [0 <= h_var[i] for i in range(n_points)]
    constraints += [h_var[i] <= 1 for i in range(n_points)]
    
    # Overlap constraints for all lags
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        # Compute indices for valid convolution
        if k >= 0:
            valid_indices = range(n_points - k)
            # Sum over valid indices where h[i] * (1 - h[i + k])
            expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[valid_indices + k])) * dx
        else:
            valid_indices = range(-k, n_points)
            expr = cp.sum(cp.multiply(h_var[valid_indices], 1 - h_var[valid_indices - k])) * dx
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # First solve with CVXPY
    try:
        prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s - 5)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = h_var.value
            c5_result = C.value
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = np.clip(h_result, 0, 1)
        else:
            # Use fallback if no solution from solver
            c5_result = np.inf
    except:
        c5_result = np.inf
    
    # Local refinement with scipy if needed
    if c5_result == np.inf or time.time() - start_time > budget_s - 10:
        return (best_h.tolist(), best_c5, n_points)
    
    # Early stopping if target is already achieved
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Use scipy for further refinement
    from scipy.optimize import minimize
    
    def objective(h):
        # Compute overlap
        overlap = np.correlate(h, 1-h, mode="full")
        return np.max(overlap) * dx
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Try local refinement
    iterations = 0
    max_iter = 5000
    while time.time() - start_time < budget_s - 5 and iterations < max_iter:
        # Use the previous best as initial guess
        h_noisy = best_h + np.random.normal(0, 0.005, n_points)
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = h_noisy * (target_sum / np.sum(h_noisy))
        
        result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (target_sum / np.sum(h_candidate))
            
            candidate_c5 = objective(h_candidate)
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = h_candidate.copy()
                
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final validation
    actual_c5 = np.max(np.correlate(best_h, 1-best_h, mode="full")) * dx
    return (best_h.tolist(), actual_c5, n_points)
