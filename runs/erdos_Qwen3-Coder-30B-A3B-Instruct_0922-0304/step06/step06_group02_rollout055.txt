import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use a more refined search approach
    iteration = 0
    max_iter = 20000
    adaptive_step = 0.02
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Try multiple strategies with random perturbations
        strategies = []
        for _ in range(5):
            h_candidate = h.copy()
            # Add noise
            noise = np.random.normal(0, adaptive_step, len(h))
            h_candidate += noise
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            strategies.append((h_candidate, compute_c5(h_candidate, dx)))
            
        # Also try deterministic perturbations
        h_perturbed = h.copy()
        h_perturbed = np.clip(h_perturbed + np.random.normal(0, 0.01, len(h)), 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        strategies.append((h_perturbed, compute_c5(h_perturbed, dx)))
        
        # Gradient-like update using local search in a small region
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Small perturbation around current solution
        h_noisy = h + np.random.normal(0, 0.005, len(h))
        h_noisy = np.clip(h_noisy, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            strategies.append((h_candidate, compute_c5(h_candidate, dx)))
        
        # Select best among all strategies
        best_strategy = min(strategies, key=lambda x: x[1])
        h = best_strategy[0]
        
        # Adaptive step adjustment
        if iteration % 50 == 0:
            if current_c5 < best_c5:
                adaptive_step *= 0.95
            else:
                adaptive_step *= 1.05
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
