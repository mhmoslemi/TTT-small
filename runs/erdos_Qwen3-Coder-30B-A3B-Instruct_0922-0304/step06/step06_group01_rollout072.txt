```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cma

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Construct rectangle pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()

    # Try CMA-ES approach
    # Define objective function for CMA-ES
    def objective_cma(h_vec):
        h = np.clip(h_vec, 0, 1)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    # Initial mean and sigma
    initial_h = h.copy()
    
    # Run CMA-ES
    es = cma.CMAEvolutionStrategy(initial_h, 0.2, {'popsize': 140, 'verbose': -1})
    iter_count = 0
    max_iter = 500

    while es.stop() is False and time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        solutions = es.ask()
        fitness_values = []
        for sol in solutions:
            try:
                h = np.clip(sol, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                fitness_values.append(c5)
            except Exception:
                fitness_values.append(float('inf'))
        
        es.tell(solutions, fitness_values)
        iter_count += 1
        
        # Check for improvement
        best_idx = np.argmin(fitness_values)
        current_best_c5 = fitness_values[best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = np.clip(solutions[best_idx], 0, 1)
            best_h = normalize_h(best_h, target_sum)

        # Early exit
        if best_c5 <= 0.3807:
            break
            
    # Final polishing with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            final_c5 = compute_c5(h_new, dx)
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_new.copy()
    except Exception:
        pass
        
    return (best_h.tolist(), best_c5, n_points)
```
