import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a smooth step function
        h = np.zeros(n_points)
        # Create a piecewise linear function
        mid = n_points // 2
        h[:mid] = np.linspace(0, 1, mid)
        h[mid:] = np.linspace(1, 0, n_points - mid)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use CMA-ES inspired approach to escape local minima
    num_generations = 500
    population_size = 2 * n_points  # 2 * n_points is good for CMA-ES
    sigma = 0.1  # Initial step size
    weights = np.log(population_size / 2 + 1) - np.log(np.arange(1, population_size + 1))
    weights = weights / np.sum(weights)  # Normalize
    mu = population_size // 2  # Number of best candidates
    weights_mu = weights[:mu]
    
    # Initialize covariance matrix
    C = np.eye(n_points)
    mean = h.copy()
    
    # Iteration counter
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < num_generations:
        # Sample new candidates from multivariate normal distribution
        candidates = []
        for _ in range(population_size):
            sampled = np.random.multivariate_normal(mean, sigma**2 * C)
            # Clip to [0,1] and normalize
            sampled = np.clip(sampled, 0, 1)
            sampled = normalize_h(sampled, target_sum)
            candidates.append(sampled)
        
        # Evaluate candidates
        c5_values = []
        for candidate in candidates:
            c5 = compute_c5(candidate, dx)
            c5_values.append(c5)
        
        # Sort candidates by C5 values (ascending order)
        sorted_indices = np.argsort(c5_values)
        top_candidates = [candidates[i] for i in sorted_indices[:mu]]
        top_c5 = [c5_values[i] for i in sorted_indices[:mu]]
        
        # Update mean
        new_mean = np.zeros(n_points)
        for i in range(mu):
            new_mean += weights_mu[i] * top_candidates[i]
        mean = new_mean
        
        # Update covariance matrix
        if iteration > 0:  # Skip the first update for stability
            # Centered differences
            diff = np.array(top_candidates) - mean
            # Update C using rank-one update formula
            C = (1 - 1/mu) * C + (1/mu) * np.outer(diff.T, diff)
            # Add small diagonal to maintain positive definiteness
            diag_weight = 0.001
            C += diag_weight * np.eye(n_points)
        
        # Update step size
        # For simplicity, use a fixed adaptation scheme
        if iteration > 0 and iteration % 10 == 0:
            sigma *= 1.05  # Adaptive increase
        
        # Keep track of best solution
        current_best_idx = np.argmin(c5_values)
        if c5_values[current_best_idx] < best_c5:
            best_c5 = c5_values[current_best_idx]
            best_h = candidates[current_best_idx].copy()
        
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final local refinement
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h
    
    return (best_h.tolist(), best_c5, n_points)
