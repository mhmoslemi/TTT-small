import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Save current state
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perturb h
        h_new = h.copy()
        
        # Select two random indices where we can adjust values
        i, j = np.random.randint(0, n_points, 2)
        
        # Make sure we can actually adjust values
        if h_new[i] < 1 and h_new[j] > 0:
            # Compute delta to move
            delta = min(1 - h_new[i], h_new[j])
            # Apply small random adjustment
            delta *= np.random.rand() * 0.2
            
            # Update values
            h_new[i] += delta
            h_new[j] -= delta
            
            # Renormalize to ensure the constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
        
        # Compute new overlap
        new_c5 = compute_c5(h_new)
        
        # Metropolis acceptance
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
        
        # Update temperature
        T *= alpha
        
        # Stagnation handling
        if delta_c5 >= 0:
            stagnation_count += 1
        else:
            stagnation_count = 0
            
        if stagnation_count > max_stagnation:
            T *= 1.2  # Increase temperature to escape local minimum
            stagnation_count = 0
        
        # Occasionally restart from best
        if iteration % 1000 == 0:
            h = best_h.copy()
            T = max(T * 0.9, 1e-5)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
