import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal pattern
        h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Ensure values are in [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimize with momentum and adaptive step size
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the max lag k
        # This is an analytical gradient for the specific correlation term
        g = np.zeros(n_points)
        for i in range(n_points):
            # Derivative of h[i] * (1 - h[(i + max_k) % n]) w.r.t. h[i]
            if (i + max_k) < n_points:
                g[i] = (1 - h[(i + max_k) % n]) * dx
            else:
                # Since the indexing wraps with modulo, we only need to care 
                #   about h[i - max_k] for i >= max_k so that i - max_k < n
                if i - max_k >= 0:
                    g[i] = -h[i - max_k] * dx
                else:
                    # Should not happen in this case, but to be safe
                    g[i] = 0
        
        # Update with momentum and step size
        new_h = h - step_size * (g + momentum * g_prev)
        g_prev = g.copy()
        
        # Project onto feasible set
        new_h = project_h(new_h)
        
        h = new_h
        
        # Adapt step size and momentum
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum *= 0.9
        if momentum < 0.5:
            momentum = 0.5
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.001, n_points)
            h = h + pert
            h = project_h(h)

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
