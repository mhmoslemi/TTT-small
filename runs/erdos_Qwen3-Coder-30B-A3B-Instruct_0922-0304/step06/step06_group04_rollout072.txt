import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Start with uniform distribution
        base_h = np.ones(n_points) * 0.5
    
    h = base_h.copy()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Alternating optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Step 1: Optimize h for fixed overlaps (use local optimization)
        # Direct approach using scipy.optimize
        def objective(h):
            return compute_c5(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Local search around current h
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 200, 'ftol': 1e-6})
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5_opt = compute_c5(h_opt, dx)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    h = h_opt.copy()
            else:
                h = normalize_h(h, target_sum)  # keep it valid
        except Exception:
            h = normalize_h(h, target_sum)  # fallback
            
        # Step 2: Adaptive perturbations to escape local minima
        if iteration % 10 == 0 and iteration > 0:
            h = best_h.copy()
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
