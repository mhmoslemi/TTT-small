import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, offset=0):
    """Create a comb pattern with given spacing and offset."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    tooth_width = dx
    for i in range(0, n_points, spacing):
        start_idx = i + offset
        if start_idx < n_points:
            end_idx = min(start_idx + 1, n_points)
            h[start_idx:end_idx] = 1.0
    return h

def optimize_pattern(h, target_sum, dx):
    """Optimize a pattern using SLSQP."""
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        if result.success:
            return result.x
    except:
        pass
    return h

    best_c5 = float('inf')
