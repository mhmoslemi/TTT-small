import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative cutting-plane approach
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        worst_c5 = overlap[worst_lag_idx] * dx
        
        # Optimization with constraint for worst lag
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def constraint_worst_lag(h):
            # Compute overlap for the worst lag
            # This is a simplified approach - discrete convolution handling
            # We compute the correlation for the specific lag and add it to the objective
            # The actual constraint in optimization would be more complex to handle
            return worst_c5 - overlap[worst_lag_idx] * dx
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint_sum_obj = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
            
        # Use a fixed local search with some mutation,
        # but only optimize for the worst lag constraint
        
        # Generate candidates with local perturbations
        candidates = [h.copy()]
        for _ in range(3):
            h_pert = h + np.random.normal(0, 0.02, len(h))
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = normalize_h(h_pert, target_sum)
            candidates.append(h_pert)
            
        # Use local optimization (SLSQP) on each candidate with constraint
        best_candidate = h
        best_candidate_c5 = current_c5
        
        for h_candidate in candidates:
            try:
                result = minimize(objective, h_candidate, method='SLSQP', 
                                bounds=bounds, constraints=constraint_sum_obj,
                                options={'maxiter': 50})
                if result.success:
                    h_test = result.x
                    h_test = np.clip(h_test, 0, 1)
                    h_test = normalize_h(h_test, target_sum)
                    test_c5 = compute_c5(h_test, dx)
                    if test_c5 < best_candidate_c5:
                        best_candidate_c5 = test_c5
                        best_candidate = h_test.copy()
            except:
                continue
                
        if best_candidate_c5 < current_c5 * 0.9999:
            h = best_candidate.copy()
        else:
            # Small random perturbations if no improvement
            h = h + np.random.normal(0, 0.01, len(h))
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
