import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Binary search on C5
    c_low = 0.0
    c_high = 1.0
    best_c5 = float('inf')
    best_h = h.copy()
    
    solver = 'MOSEK'  # Try MOSEK first for speed
    
    iter_count = 0
    while time.time() - start_time < budget_s - 2 and c_high - c_low > 1e-5:
        iter_count += 1
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find feasible solution for this c_mid
        try:
            # Build convex problem
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Add overlap constraints
            valid_k = list(range(1, n_points)) + list(range(-n_points+1, 0))
            
            for k in valid_k:
                if k > 0:
                    # lag k
                    expr = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, k)))
                else:
                    # lag k < 0
                    expr = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, k)))
                constraints.append(expr <= c_mid / dx)
            
            # Objective (we're just checking feasibility)
            prob = cp.Problem(cp.Minimize(0), constraints)
            
            prob.solve(solver=solver, verbose=False)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                new_h = h_vars.value
                # Renormalize
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = new_h.copy()
                if new_c5 <= target:
                    break                    
                c_high = c_mid
            else:
                c_low = c_mid
                
        except Exception as e:
            # fallback: increase lower bound
            c_low = c_mid

    # refine the best solution found so far with gradient descent if time allows
    h_refined = best_h.copy()
    best_refined_c5 = best_c5
    
    if time.time() - start_time < budget_s - 5:
        # local search to improve solution
        max_iter = 5000
        step_size = 0.005
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break

            # compute current overlap
            overlap = correlate(h_refined, 1 - h_refined, mode="full")
            max_k = np.argmax(overlap)
            max_overlap = overlap[max_k] * dx
            
            if max_overlap <= target:
                break

            # compute gradient using finite differences
            g = np.zeros(n_points)
            eps_val = 1e-5
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_minus = h_refined.copy()
                
                h_plus[j] += eps_val
                h_minus[j] -= eps_val
                
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[j] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # update
            new_h = h_refined - step_size * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            h_refined = new_h
            
        final_c5 = compute_c5(h_refined)
        if final_c5 < best_refined_c5:
            best_c5 = final_c5
            best_h = h_refined.copy()

    return (best_h.tolist(), best_c5, n_points)
