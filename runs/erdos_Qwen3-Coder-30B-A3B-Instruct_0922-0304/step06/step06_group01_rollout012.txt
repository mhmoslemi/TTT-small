import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a piecewise step function as a starting point
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    sa_steps = 100000
    
    # Local refinement
    local_refine_freq = 10000
    local_refine_iter = 200
    
    # Track SA steps for local refinement
    sa_step_count = 0
    
    for i in range(sa_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h: adjust 10% of values
        perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new = h.copy()
        for idx in perturb_indices:
            h_new[idx] = np.random.uniform(0, 1)
        
        # Ensure sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        # Compute current C5
        c5_current = compute_c5(h, dx)
        
        # Accept or reject based on SA
        if c5_new < c5_current or np.random.rand() < math.exp((c5_current - c5_new) / temp):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cooling schedule
        if i % 100 == 0:
            temp *= cooling_rate
            
        # Local refinement periodically
        if (i + 1) % local_refine_freq == 0 and i > 0:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': local_refine_iter})
            
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                h = h_new.copy()
                
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        final_h = result.x
        final_h = np.clip(final_h, 0, 1)
        final_h = normalize_h(final_h, target_sum)
        c5_bound = compute_c5(final_h, dx)
    else:
        final_h = best_h
        c5_bound = best_c5
    
    return (final_h.tolist(), c5_bound, n_points)
