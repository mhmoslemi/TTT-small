import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def to_fourier(h):
    """Convert h to Fourier coefficients."""
    return np.fft.fft(h)

def from_fourier(coeffs):
    """Convert Fourier coefficients to h."""
    return np.real(np.fft.ifft(coeffs))

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def clip_and_normalize(h, target_sum):
    """Clip h to [0, 1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of frequency components
    
    # Initialize with initial_h_values or a base pulse
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        # A simple rectangular pulse
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 1
        h_init = normalize_h(h_init, target_sum)
    
    # Fourier representation
    h_fourier = to_fourier(h_init)
    initial_coeffs = h_fourier[:n_freq+1].real.copy()
    
    def objective(coeffs):
        # Reconstruct h from Fourier coefficients
        # Set the full spectrum in a symmetric way (for real signal)
        new_coeffs = np.zeros(n_points, dtype=complex)
        new_coeffs[:n_freq+1] = coeffs + 1j * np.zeros(n_freq+1)
        # Use symmetry for real-valued output
        new_coeffs[n_points//2+1:] = np.conj(new_coeffs[1:n_points//2][::-1])
        
        h_reconstructed = from_fourier(new_coeffs)
        h = np.real(h_reconstructed)
        h = clip_and_normalize(h, target_sum)
        return compute_c5(h, dx)
    
    # Bounds for coefficients
    bounds = [(-1.0, 1.0) for _ in range(2*(n_freq+1))]
    
    # Differential Evolution
    de_result = differential_evolution(
        objective,
        bounds,
        maxiter=300,
        popsize=80,
        tol=1e-6,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        callback=lambda x, convergence: time.time() - start_time > budget_s - 1,
        disp=False
    )
    
    # Refine best solution with local optimization
    best_coeffs = de_result.x
    h_best = np.zeros(n_points)
    h_best[:n_freq+1] = best_coeffs[:n_freq+1] + 1j * best_coeffs[n_freq+1:2*(n_freq+1)]
    h_best[n_points//2+1:] = np.conj(h_best[1:n_points//2][::-1])
    h_best = np.real(np.fft.ifft(h_best))
    h_best = clip_and_normalize(h_best, target_sum)
    
    # Fine-tune with SLSQP
    def fine_objective(h):
        return compute_c5(h, dx)
    
    bounds_h = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        fine_objective,
        h_best,
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100}
    )
    
    if result.success:
        h_best = result.x.copy()
        h_best = clip_and_normalize(h_best, target_sum)
    
    final_c5 = compute_c5(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
