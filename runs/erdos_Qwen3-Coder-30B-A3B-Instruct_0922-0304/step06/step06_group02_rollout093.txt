import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate(h, rate=0.05):
    """Mutate h with given rate."""
    mask = np.random.random(len(h)) < rate
    noise = np.random.normal(0, 0.05, len(h))
    h = h + mask * noise
    return np.clip(h, 0, 1)

def crossover(p1, p2):
    """Perform crossover between two parents."""
    crossover_point = np.random.randint(1, len(p1) - 1)
    child = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    return child

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        initial_h = np.ones(n_points)
        initial_h[n_points//2:] = 0
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    
    # Create initial population
    population_size = 30
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = mutate(initial_h.copy(), 0.1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    generation = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 50%
        elite_size = population_size // 2
        selected = population[:elite_size]
        
        # Generate new offspring
        new_population = selected.copy()
        while len(new_population) < population_size:
            # Select two parents
            p1, p2 = np.random.choice(selected, size=2, replace=False)
            # Crossover
            child = crossover(p1, p2)
            # Mutate
            child = mutate(child, 0.05)
            # Normalize
            child = normalize_h(child, target_sum)
            new_population.append(child)
        
        # Local refinement with SLSQP for top 10%
        top_k = max(1, population_size // 10)
        for i in range(top_k):
            h = population[i]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = normalize_h(refined, target_sum)
                new_population[i] = refined
        
        population = new_population
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
