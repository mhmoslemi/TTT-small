import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate, target_sum):
    """Mutate h with constraint maintenance."""
    mask = np.random.random(len(h)) < mutation_rate
    h_new = h.copy()
    h_new[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h_new = np.clip(h_new, 0, 1)
    h_new = normalize_h(h_new, target_sum)
    return h_new

def crossover_h(h1, h2, crossover_rate):
    """Crossover two h vectors."""
    if np.random.random() > crossover_rate:
        return h1.copy()
    mask = np.random.random(len(h1)) < 0.5
    h_new = h1.copy()
    h_new[mask] = h2[mask]
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from `initial_h_values` if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm parameters
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    elite_size = population_size // 10
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.02, n_points)
        new_h = h + perturbation
        new_h = np.clip(new_h, 0, 1)
        new_h = normalize_h(new_h, target_sum)
        population.append(new_h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Evolution loop
    generation = 0
    max_generations = 1000
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitness = []
        for individual in population:
            fitness.append(compute_c5(individual, dx))
        
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Sort population
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Create new population
        new_population = sorted_population[:elite_size]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection using tournament
            parents = []
            for _ in range(2):
                tournament_size = 3
                tournament_indices = np.random.choice(len(sorted_population), tournament_size)
                tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
                winner_index = tournament_indices[np.argmin(tournament_fitness)]
                parents.append(sorted_population[winner_index])
            
            # Crossover
            child = crossover_h(parents[0], parents[1], crossover_rate)
            
            # Mutation
            child = mutate_h(child, mutation_rate, target_sum)
            
            new_population.append(child)
            
        population = new_population
        
        # Local SLSQP refinement on top 10%
        top_indices = sorted_indices[:elite_size]
        for idx in top_indices:
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            result = minimize(objective, population[idx], method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 100})
            
            if result.success:
                refined = np.clip(result.x, 0, 1)
                refined = normalize_h(refined, target_sum)
                population[idx] = refined
                
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
