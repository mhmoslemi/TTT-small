import time
from scipy.signal import correlate

    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000
    
    # Initialize h from initial_h_values if available

        h = np.clip(h, 0, 1)

        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    stagnation_limit = 500
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter and T > min_T:
        iteration += 1
        
        # Make a perturbation
        h_candidate = current_h.copy()
        
        # Select two indices to adjust
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure we can make a valid adjustment
        if h_candidate[i] < 1 and h_candidate[j] > 0:
            # Compute the maximum possible delta
            delta = min(1 - h_candidate[i], h_candidate[j])
            
            # Apply the change
            delta = np.random.rand() * delta  # Random fraction
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize to satisfy constraint sum(h) = n_points/2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            new_c5 = compute_c5(h_candidate)
            
            # Accept or reject the move
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):

                current_c5 = new_c5
                
                # Update best solution
                if new_c5 < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = new_c5
                    stagnation_count = 0  # Reset stagnation counter
                else:
                    stagnation_count += 1
            
            # Cooling schedule
            if iteration % 10 == 0:
                T *= alpha
                
            # Handle stagnation
            if stagnation_count > stagnation_limit:
                T *= 1.2  # Increase temperature to escape local minimum
                stagnation_count = 0
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final recomputation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
