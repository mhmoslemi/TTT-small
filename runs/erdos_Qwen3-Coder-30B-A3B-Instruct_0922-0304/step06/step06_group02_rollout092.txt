import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def genetic_algorithm_step(population, fitnesses, n_points, target_sum, dx, mutation_rate=0.05, crossover_rate=0.7):
    """Perform one generation of genetic algorithm."""
    # Sort by fitness
    sorted_indices = np.argsort(fitnesses)
    population = population[sorted_indices]
    fitnesses = fitnesses[sorted_indices]
    
    # Elitism: keep top 20%
    elite_size = max(1, len(population) // 5)
    elite = population[:elite_size]
    
    # Create offspring through crossover and mutation
    new_population = [elite[i % elite_size] for i in range(len(population))]
    
    # Crossover
    for i in range(elite_size, len(population)):
        if np.random.rand() < crossover_rate and i + 1 < len(population):
            parent1 = population[i]
            parent2 = population[i + 1]
            # Single point crossover
            crossover_point = np.random.randint(1, n_points)
            child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            new_population[i] = child
        else:
            new_population[i] = population[i].copy()
    
    # Mutation
    for i in range(len(new_population)):
        for j in range(n_points):
            if np.random.rand() < mutation_rate:
                new_population[i][j] = np.clip(new_population[i][j] + np.random.normal(0, 0.05), 0, 1)
        # Ensure sum constraint
        new_population[i] = normalize_h(new_population[i], target_sum)
    
    return np.array(new_population)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the provided construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Fallback initialization if no parent exists
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm parameters
    population_size = 30
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    # Create initial population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.02, n_points)
        individual = h + perturbation
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    population = np.array(population)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # GA loop
    generation = 0
    max_generations = 500
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate population
        fitnesses = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitnesses.append(c5)
        fitnesses = np.array(fitnesses)
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Local refinement with SLSQP for top candidates
        top_k = max(2, population_size // 5)
        for j in range(top_k):
            idx = np.argsort(fitnesses)[j]
            individual = population[idx].copy()
            
            # Local optimization using SLSQP
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
            
            def objective(x):
                return compute_c5(x, dx)
                
            # Small random perturbation
            noise = np.random.normal(0, 0.001, n_points)
            x0 = np.clip(individual + noise, 0, 1)
            x0 = normalize_h(x0, target_sum)
            
            try:
                result = minimize(objective, x0, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined = result.x
                    refined = np.clip(refined, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    refined_c5 = compute_c5(refined, dx)
                    if refined_c5 < fitnesses[idx]:
                        population[idx] = refined
            except:
                pass
                
        # Generate next generation
        population = genetic_algorithm_step(population, fitnesses, n_points, target_sum, dx, mutation_rate, crossover_rate)
        generation += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
