import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Genetic algorithm parameters
    population_size = 100
    generations = 60
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Create initial population
    population = []
    for _ in range(population_size):
        individual = h + np.random.normal(0, 0.05, n_points)
        individual = np.clip(individual, 0, 1)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness_scores = []
        for individual in population:
            c5_value = compute_c5(individual, dx)
            fitness_scores.append(c5_value)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = [population[0].copy()]  # Elitism
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            # Tournament selection
            idx1 = np.random.choice(tournament_size, replace=False)
            idx2 = np.random.choice(tournament_size, replace=False)
            parent1 = population[np.argmin([fitness_scores[i] for i in idx1])]
            parent2 = population[np.argmin([fitness_scores[i] for i in idx2])]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.02)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.02)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            # Local optimization with SLSQP
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            try:
                result1 = minimize(objective, child1, method='SLSQP', 
                                  bounds=bounds, constraints=constraint, 
                                  options={'maxiter': 100})
                if result1.success:
                    child1 = np.clip(result1.x, 0, 1)
                    child1 = normalize_h(child1, target_sum)
                    
                result2 = minimize(objective, child2, method='SLSQP', 
                                  bounds=bounds, constraints=constraint, 
                                  options={'maxiter': 100})
                if result2.success:
                    child2 = np.clip(result2.x, 0, 1)
                    child2 = normalize_h(child2, target_sum)
            except:
                pass
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Update population
        population = new_population
        mutation_rate *= 0.98  # Decrease mutation rate over generations
    
    # Final refinement to get best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
