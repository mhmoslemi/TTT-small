import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal function
        h = np.sin(np.linspace(0, 2*np.pi, n_points)) * 0.3 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Optimization parameters
    max_iter = 50000
    iteration = 0
    initial_eta = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags and find max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current max_k
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            else:
                g[i] = -h[i - (n_points - max_k)] * dx

        # Adaptive step size
        eta = initial_eta * (0.5 ** (iteration // 1000))
        
        # Update with momentum
        new_h = h - eta * (g + momentum * g_prev)
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Update momentum gradient
        g_prev = g
        
        h = new_h

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
