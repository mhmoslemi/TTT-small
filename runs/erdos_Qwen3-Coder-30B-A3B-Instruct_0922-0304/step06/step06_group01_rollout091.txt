import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct a comb-like h function."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 3.0

    h = np.zeros(n_points)
    dx = 2.0 / n_points
    tooth_height = 1.0  # Will normalize after construction
    
    for i in range(n_teeth):
        pos = i * (tooth_width + spacing)
        start_idx = int(pos / dx)
        end_idx = int((pos + tooth_width) / dx)
        
        if start_idx >= n_points:
            continue
        if end_idx > n_points:
            end_idx = n_points
            
        # Build transition edges for the tooth
        for j in range(start_idx, min(end_idx, n_points)):
            x = j * dx
            # Left edge
            if x >= pos and x <= pos + tooth_width / 2:
                dist = x - pos
                h[j] = max(0, 1 - dist / slope)
            # Right edge
            elif x > pos + tooth_width / 2 and x <= pos + tooth_width:
                dist = pos + tooth_width - x
                h[j] = max(0, 1 - dist / slope)
            # Tooth body
            elif x > pos and x <= pos + tooth_width:
                h[j] = 1.0
                
    # Normalize height so that sum(h) = n_points/2
    h = normalize_h(h, n_points / 2)
    
    return h

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    target_sum = n_points / 2
    dx = 2.0 / n_points
    
    # Initialize with potential parent solution or use comb design
    if 'initial_h_values' in globals():

    else:
        # Create base comb structure for exploration
        initial_h = create_comb_h(n_points, n_teeth=7)
        h = initial_h.copy()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimize to find the best comb parameters with SLSQP refinement
    best_params = [7, None, None, None]
    n_opt_loops = 0
    max_loops = 1000
    
    # Parameter sweep over combinations of n_teeth, tooth_width, spacing, slope
    sweeps = [
        (5, 9, 2),      # n_teeth range
        (10, 15, 2),    # tooth_width range, (step)
        (5, 10, 2),     # slope range
    ]
    
    while time.time() - start_time < budget_s - 1 and n_opt_loops < max_loops:
        # Construct a new comb function and evaluate
        n_teeth = best_params[0]
        if n_teeth < 3:
            n_teeth = 3
        elif n_teeth > 15:
            n_teeth = 15
            
        tooth_width = 2.0 / n_teeth if best_params[1] is None else best_params[1]
        slope = tooth_width / 3 if best_params[3] is None else best_params[3]
        
        # Randomize spacing a bit to escape local optima
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1) + (np.random.rand() - 0.5) * 0.05
        
        h = create_comb_h(n_points, n_teeth=n_teeth, tooth_width=tooth_width, spacing=spacing, slope=slope)
        
        # Apply local SLSQP optimization to refine h
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective(h):
            return compute_c5(h, dx)

            return np.sum(h) - target_sum
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        options = {'maxiter': 50, 'ftol': 1e-6}
        
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options=options)

    bounds = [(0, 1) for _ in range(n_points)]

        return compute_c5(h, dx)
    def constraint_final(h):
        return np.sum(h) - target_sum
    constraint = {'type': 'eq', 'fun': constraint_final}
    result = minimize(objective_final, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    if result.success:
        best_h = np.clip(result.x, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        final_c5 = compute_c5(best_h, dx)
        best_c5 = min(best_c5, final_c5)
    
    return (best_h.tolist(), best_c5, n_points)
