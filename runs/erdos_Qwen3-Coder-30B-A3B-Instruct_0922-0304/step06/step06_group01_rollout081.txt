import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h(x) in time domain."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 * np.ones(n_points)
    for k in range(1, n_freq + 1):
        h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    coeffs = np.zeros(2 * n_freq + 1)
    coeffs[0] = np.mean(h) * 2.0
    x = np.linspace(0, 2, len(h))
    for k in range(1, n_freq + 1):
        coeffs[2*k-1] = 2.0 * np.mean(h * np.cos(k * np.pi * x))
        coeffs[2*k] = 2.0 * np.mean(h * np.sin(k * np.pi * x))
    return coeffs

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of frequency components
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization: piecewise constant
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    # Convert to Fourier representation for optimization
    coeffs = h_to_fourier(h, n_freq)
    
    def objective(coeffs):
        h_temp = fourier_to_h(coeffs, n_points, n_freq)
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = normalize_h(h_temp, target_sum)
        return compute_c5(h_temp, dx)
    
    # Define bounds for Fourier coefficients
    bounds = [(-1.0, 1.0)] * (2 * n_freq + 1)
    
    # Set up differential evolution
    max_iter = int(budget_s / 2) if budget_s > 2 else 100
    de_result = differential_evolution(
        objective, 
        bounds,
        maxiter=max_iter,
        popsize=60,
        mutation=(0.5, 1.0),
        recombination=0.7,
        seed=seed,
        disp=False
    )
    
    # Extract best solution
    best_coeffs = de_result.x
    best_h = fourier_to_h(best_coeffs, n_points, n_freq)
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    
    # Refine with local optimization
    bounds_h = [(0, 1) for _ in range(n_points)]
    
    def local_objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    local_result = minimize(
        local_objective,
        best_h,
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100}
    )
    
    if local_result.success:
        best_h = local_result.x
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
