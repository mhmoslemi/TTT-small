import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization 
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Gradient-based refinement
    max_iter = 1000
    step_size = 0.01
    reg_weight = 1e-3
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        # For direct gradient, we approximate by finite differences
        eps = 1e-5
        gradient = np.zeros(n_points)
        
        base_c5 = compute_c5(h, dx)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = normalize_h(h_plus, target_sum)
            
            c5_plus = compute_c5(h_plus, dx)
            
            gradient[j] = (c5_plus - base_c5) / eps
        
        # Update step with regularization
        h_new = h - step_size * (gradient + reg_weight * (h - 0.5))
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
        
        if best_c5 <= 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
