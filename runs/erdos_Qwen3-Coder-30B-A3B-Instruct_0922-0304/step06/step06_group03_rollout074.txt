import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Discretize to allowed values
    allowed_values = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    h = allowed_values[np.round(h / 0.25).astype(int)]
    h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Refine discretization to match exact sum
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Adjust values to satisfy sum constraint
        diff = target_sum - current_sum
        if diff > 0:
            # Need to increase some values
            idxs = np.argsort(h)
            for i in range(int(np.ceil(abs(diff)))):
                if i < len(idxs):
                    h[idxs[i]] = min(1.0, h[idxs[i]] + 0.25)
        else:
            # Need to decrease some values
            idxs = np.argsort(-h)
            for i in range(int(np.ceil(abs(diff)))):
                if i < len(idxs):
                    h[idxs[i]] = max(0.0, h[idxs[i]] - 0.25)
    
    # Make sure it's exactly matching our allowed values
    h = np.round(h / 0.25) * 0.25
    h = np.clip(h, 0, 1)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

    iteration = 0

        current_h = h.copy()
        current_c5 = compute_c5(h, dx)

        # Perturbation strategy
        # Try a random small change
        change_idx = np.random.randint(0, n_points)
        old_value = h[change_idx]

        current_val = h[change_idx]
        possible_vals = allowed_values[abs(allowed_values - current_val) <= 0.25]
        if len(possible_vals) > 1:
            new_val = np.random.choice(possible_vals[possible_vals != current_val])
        else:
            new_val = current_val
            
        # Apply change
        h[change_idx] = new_val
        
        # Adjust the rest to keep sum constraint
        new_sum = np.sum(h)
        diff = target_sum - new_sum
        if abs(diff) > 1e-10 and n_points > 1:

                    h[adjust_indices[i]] = min(1.0, h[adjust_indices[i]] + step_size)
                elif diff < 0 and i < len(adjust_indices) and h[adjust_indices[i]] > 0.0:
                    h[adjust_indices[i]] = max(0.0, h[adjust_indices[i]] - step_size)
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
