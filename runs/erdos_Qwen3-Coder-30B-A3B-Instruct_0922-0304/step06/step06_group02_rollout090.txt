```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    """Crossover two parents to create offspring."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = np.random.randint(1, len(parent1))
    offspring1 = np.concatenate([parent1[:point], parent2[point:]])
    offspring2 = np.concatenate([parent2[:point], parent1[point:]])
    return offspring1, offspring2

def mutate(h, mutation_rate=0.05):
    """Mutate h with given rate."""
    h = h.copy()
    for i in range(len(h)):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from global initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # GA parameters
    population_size = 30
    generations = 1000
    mutation_rate = 0.05
    crossover_rate = 0.7
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        perturbation = np.random.normal(0, 0.02, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        population.append(h_perturbed)
    
    # GA iterations
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for candidate in population:
            c5 = compute_c5(candidate, dx)
            fitness.append(c5)
            
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection - top 50%
        selected_indices = np.argsort(fitness)[:population_size // 2]
        selected = [population[i] for i in selected_indices]
        
        # Create offspring
        offspring = []
        while len(offspring) < population_size - len(selected):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutate
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Ensure constraints
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            offspring.extend([child1, child2])
            
        # Keep elites
        elite_count = population_size // 10
        elite_indices = np.argsort(fitness)[:elite_count]
        elites = [population[i] for i in elite_indices]
        
        # Replace population
        population = elites + offspring[:population_size - elite_count]
        
        # SLSQP refinement on top candidates
        top_candidates = [population[i] for i in np.argsort(fitness)[:10]]
        for candidate in top_candidates:
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
                
            result = minimize(objective, candidate, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined = result.x
                refined = np.clip(refined, 0, 1)
                refined = normalize_h(refined, target_sum)
                refined_c5 = compute_c5(refined, dx)
                if refined_c5 < compute_c5(candidate, dx) * 0.9999:
                    # Replace worst in population with refined solution
                    worst_idx = np.argmax(fitness)
                    population[worst_idx] = refined.copy()
                    fitness[worst_idx] = refined_c5
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
