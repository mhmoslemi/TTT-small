import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Compute overlap integral for all shifts
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialization with initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Start with step function
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Alternating optimization between h and overlap variables
    iteration = 0
    max_iter_per_step = 200
    
    while time.time() - start_time < budget_s - 1:
        # Step 1: Optimize h while fixing overlaps
        h = best_h.copy()
        
        # Add small perturbation to escape local minima
        noise = np.random.normal(0, 0.03, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Use trust-constr for better convergence
        def objective(h):
            return compute_c5(h, dx)
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        try:
            result = minimize(objective, h, method='trust-constr', bounds=bounds, constraints=constraint, options={'maxiter': max_iter_per_step})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        except:
            pass    
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
