import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Multi-start with different initializations
    m = 15
    initializations = []
    
    # 1. Original
    initializations.append(base_h.copy())
    
    # 2. Random uniform
    for _ in range(m//3):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # 3. Gaussian-like
    for _ in range(m//3):
        h = np.random.normal(0.5, 0.2, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # 4. Step-like
    for _ in range(m//3):
        h = np.zeros(n_points)
        split = n_points // 3
        h[:split] = 1.0
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # Optimization loop
    iteration = 0
    for h_start in initializations:
        if time.time() - start_time > budget_s - 2:
            break
            
        h = h_start.copy()
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h, method='SLSQP', 
                            bounds=bounds, constraints=constraint, 
                            options={'maxiter': 200, 'ftol': 1e-8})
            
            if result.success:
                h_final = result.x
                h_final = np.clip(h_final, 0, 1)
                h_final = normalize_h(h_final, target_sum)
                c5 = compute_c5(h_final, dx)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h_final.copy()
                    
                if best_c5 <= 0.3807:
                    break
        except:
            pass
        
        iteration += 1
    
    # Final verification
    final_c5 = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
