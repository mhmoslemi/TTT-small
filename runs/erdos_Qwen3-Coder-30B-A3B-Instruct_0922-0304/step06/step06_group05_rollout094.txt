import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Store current state
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perturb h
        h_new = h.copy()
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Ensure we can adjust values
        if h_new[i] < 1 and h_new[j] > 0:
            delta = min(1 - h_new[i], h_new[j])
            delta *= np.random.random() * 0.1  # Small adjustment
            h_new[i] += delta
            h_new[j] -= delta
            
            # Renormalize to maintain sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_new = np.clip(h_new, 0, 1)
            
            # Compute new C5
            new_c5 = compute_c5(h_new)
            
            # Metropolis criterion
            delta_c5 = new_c5 - current_c5
            if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T):
                h = h_new
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        if stagnation_count > max_stagnation:
            T *= 1.2  # Increase temperature to escape local minima
            stagnation_count = 0
        else:
            T *= alpha
            
        if T < min_T:
            break
            
    # Final re-evaluation of best solution
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
