import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Convex optimization approach
    max_iter = 1000
    learning_rate = 0.01
    epsilon = 1e-3
    gradient_norm_threshold = 1e-6
    
    # Gradient-based optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        h = best_individual.copy() 
        # Compute gradient using finite differences
        # Use small perturbation to estimate gradient
        eps = 1e-4
        grad = np.zeros(n_points)
        base_loss = compute_c5(h, dx)
        
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = normalize_h(h_pert, target_sum)
            loss_plus = compute_c5(h_pert, dx)
            
            h_pert[i] -= 2*eps
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = normalize_h(h_pert, target_sum)
            loss_minus = compute_c5(h_pert, dx)
            
            grad[i] = (loss_plus - loss_minus) / (2 * eps)
            
        # Update with gradient descent
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = normalize_h(h_new, target_sum)
        
        # Compute new loss
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_individual = h_new.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 50 == 0:
            learning_rate *= 0.95
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
