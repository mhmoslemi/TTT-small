```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cma

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use smaller number of points for faster optimization
    n_points = 64
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple rectangular pulse
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()

    # CMA-ES optimization
    def objective(x):
        h_test = np.clip(x, 0, 1)
        h_test = normalize_h(h_test, target_sum)
        return compute_c5(h_test, dx)
    
    # Set up CMA-ES
    options = cma.CMAOptions()
    options.set('verbose', -1)  # Silent mode
    
    # Initialize with the starting point
    mean = h.copy()
    sigma = 0.2
    es = cma.CMAEvolutionStrategy(mean, sigma, options)
    
    max_iter = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate new candidate solutions
        population = es.ask()
        fitnesses = []
        
        for candidate in population:
            # Apply constraints
            h_candidate = np.clip(candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            fitness = compute_c5(h_candidate, dx)
            fitnesses.append(fitness)
        
        # Tell the optimizer
        es.tell(population, fitnesses)
        
        # Check for best solution so far
        best_idx = np.argmin(fitnesses)
        current_best = fitnesses[best_idx]
        
        if current_best < best_c5:
            best_c5 = current_best
            # Ensure we keep the best solution
            best_h = population[best_idx].copy()
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        
        # Early exit if we're close to target
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with SLSQP
    if time.time() - start_time < budget_s - 1:
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective_slsqp(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective_slsqp, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100, 'ftol': 1e-9})
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            final_c5 = compute_c5(refined_h, dx)
            if final_c5 < best_c5:
                best_h = refined_h
                best_c5 = final_c5
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    
    # Validate the final solution
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "h values out of bounds"
    assert abs(np.sum(best_h) - target_sum) < 1e-6, "Sum constraint violated"
    
    return (best_h.tolist(), final_c5, n_points)
```
