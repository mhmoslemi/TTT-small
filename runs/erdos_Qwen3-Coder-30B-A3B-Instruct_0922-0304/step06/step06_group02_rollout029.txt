import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    n_particles = 25
    n_iter = 100
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest = []
    pbest_c5 = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            ph = np.array(initial_h_values).copy()
            ph = np.clip(ph, 0, 1)
            ph = normalize_h(ph, target_sum)
        else:
            ph = np.ones(n_points)
            ph[n_points//2:] = 0
            ph = np.clip(ph, 0, 1)
            ph = normalize_h(ph, target_sum)
        
        particles.append(ph.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(ph.copy())
        pbest_c5.append(compute_c5(ph, dx))
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    # PSO iterations
    for iteration in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / n_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
            
            # Evaluate fitness
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = current_c5
        
        # Periodic local search
        if iteration % 10 == 0:
            for i in range(n_particles):
                # Local L-BFGS optimization
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                    
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                def objective(h):
                    return compute_c5(h, dx)
                
                result = minimize(objective, particles[i].copy(), method='L-BFGS-B', 
                                bounds=bounds, constraints=constraint, options={'maxiter': 50})
                
                if result.success:
                    refined_h = result.x
                    refined_h = np.clip(refined_h, 0, 1)
                    refined_h = normalize_h(refined_h, target_sum)
                    refined_c5 = compute_c5(refined_h, dx)
                    
                    if refined_c5 < pbest_c5[i]:
                        pbest[i] = refined_h.copy()
                        pbest_c5[i] = refined_c5
                        
                        if refined_c5 < gbest_c5:
                            gbest = refined_h.copy()
                            gbest_c5 = refined_c5
                
            # Update global best index
            gbest_idx = np.argmin(pbest_c5)
            gbest = pbest[gbest_idx].copy()
            gbest_c5 = pbest_c5[gbest_idx]
            
        # Periodic corrective sweep
        if iteration % 20 == 0:
            sweep_h = h.copy()
            for i in range(n_points):
                # Small perturbation in region where h is not too close to bounds
                if 0.1 < sweep_h[i] < 0.9:
                    perturbation = np.random.normal(0, 0.01)
                    sweep_h[i] += perturbation
            sweep_h = np.clip(sweep_h, 0, 1)
            sweep_h = normalize_h(sweep_h, target_sum)
            sweep_c5 = compute_c5(sweep_h, dx)
            
            if sweep_c5 < gbest_c5:
                gbest = sweep_h.copy()
                gbest_c5 = sweep_c5
    
    return (gbest.tolist(), gbest_c5, n_points)
