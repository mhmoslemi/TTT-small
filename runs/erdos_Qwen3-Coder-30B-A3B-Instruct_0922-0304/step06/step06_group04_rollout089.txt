import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        # Create initial solution
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    # normalize h_base to meet sum constraint
    h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Bilinear optimization approach
    solver_time = 0
    max_solver_time = 30  # seconds per solve
    
    while time.time() - start_time < budget_s - 1 and solver_time < max_solver_time:
        # Create optimization variables
        h = cp.Variable(n_points)
        
        # Add running sum constraint
        constraints = [cp.sum(h) == target_sum]
        
        # Add bounds
        constraints.extend([h[i] >= 0 for i in range(n_points)])
        constraints.extend([h[i] <= 1 for i in range(n_points)])
        
        # Additional constraints for overlap computation
        # Overlap version 1: Since we want to minimize max_k integral
        # We will track the maximum value over all lags
        overlaps = []
        for k in range(2 * n_points - 1):
            # Correlation over lag k
            # Window size: we compute up to min(n_points - k, n_points)
            if k >= n_points:
                continue
            else:
                # If k == 0: h * (1-h)
                # Otherwise: h[i] * (1 - h[i+k])
                corr = 0
                for i in range(n_points - k):
                    corr += h[i] * (1 - h[i + k])
                overlaps.append(corr * dx)
        
        # Objective function: minimize max of overlaps
        max_overlap = cp.Variable()
        for overlap in overlaps:
            constraints.append(overlap <= max_overlap)
        objective = cp.Minimize(max_overlap)
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            solver_time = time.time() - start_time
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_sol = h.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    # Normalize to target_sum
                    h_sol = h_sol * (target_sum / np.sum(h_sol))
                    c5_val = compute_c5(h_sol, dx)
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_sol.copy()
                        
        except Exception as e:
            # Fallback to original method
            print(f"Solver failed: {e}")
            break
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
    
    # If the bilinear approach didn't improve, fallback to local search
    if best_c5 >= 0.3807 and 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
    elif best_c5 >= 0.3807:
        h_base[:n_points//2] = 1.0
        h_base = h_base * (target_sum / np.sum(h_base))
    
    # Local optimization with perturbation
    n_local_iter = 0
    h_current = h_base.copy()
    
    while time.time() - start_time < budget_s - 1 and n_local_iter < 1000:
        # Start with current best
        h_current = best_h.copy()
        
        # Try local perturbations
        perturbation = np.random.normal(0, 0.01, n_points)
        h_perturbed = h_current + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))

        c5_pert = compute_c5(h_perturbed, dx)
        if c5_pert < best_c5:
            best_c5 = c5_pert
            best_h = h_perturbed.copy()
            
        if best_c5 <= 0.3807:
            break
            
        n_local_iter += 1
        
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
