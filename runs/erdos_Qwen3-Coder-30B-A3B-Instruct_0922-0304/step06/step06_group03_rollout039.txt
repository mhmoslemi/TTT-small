import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, offset=0):
    """Create a comb pattern with given spacing and offset."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Place teeth at regular intervals
    tooth_width = dx  # Tooth width is 1 grid point
    tooth_height = 1.0
    tooth_positions = np.arange(offset, n_points, spacing)
    
    # Add teeth (potential for symmetry considerations)
    for pos in tooth_positions:
        if 0 <= pos < n_points:
            h[pos] = tooth_height
    
    # Adjust teeth to maintain sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different comb patterns
    spacings = [2, 3, 4, 5, 6]
    comb_patterns = []
    for s in spacings:
        for offset in range(s):
            comb = create_comb_pattern(n_points, s, offset)
            comb_patterns.append(comb)
    
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create variants of current solution
        # Local search with comb patterns
        if iteration % 20 == 0:
            # Select a random comb pattern
            comb_idx = np.random.randint(0, len(comb_patterns))
            h = comb_patterns[comb_idx].copy()
        
        # Fine adjustment with SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP with random perturbations
        if iteration % 3 == 0:
            # Small random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        # Run optimization (may fail silently)
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else: 
                # Perturb
                h = h + np.random.normal(0, 0.01, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        except:
            # Fallback to simple perturbation
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
