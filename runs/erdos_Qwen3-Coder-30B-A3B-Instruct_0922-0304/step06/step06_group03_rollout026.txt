import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize to target sum"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate diverse initial guesses"""
    guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_and_normalize(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern
    h = np.zeros(n_points)
    step = max(1, n_points // 10)
    for i in range(0, n_points, step):
        h[i] = 1.0
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # Random with sum constraint
    h = np.random.rand(n_points)
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.ones(n_points)
    h[1::2] = 0.2
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    # Low-discrepancy (Halton-like) pattern
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = (i * 0.618033988749895) % 1.0
    h = project_and_normalize(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    max_iter = 15000
    perturb_std = 0.02
    
    # Run optimization attempts with different initializations
    for init_idx, h in enumerate(initial_guesses):
        h = h.copy()
        current_h = h.copy()
        current_c5 = compute_c5(current_h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Iterative local search with adaptive perturbations
        for i in range(max_iter // len(initial_guesses)):
            if time.time() - start_time >= budget_s - 1:
                break
                
            # SLSQP optimization
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, current_h, method='SLSQP', 
                                bounds=bounds, constraints=constraint, 
                                options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8})
                if result.success:
                    current_h = result.x
                else:
                    raise ValueError("Optimization failed")
            except:
                # If optimization fails, use perturbation
                pass
            
            # Apply perturbations
            noise = np.random.normal(0, perturb_std, n_points)
            current_h = current_h + noise
            current_h = project_and_normalize(current_h, target_sum)
            
            # Adaptive perturbation decay
            if i % 50 == 0 and perturb_std > 1e-4:
                perturb_std *= 0.95
            
            # Check if we improved
            current_c5 = compute_c5(current_h, dx)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
            # Early stopping
            if best_c5 <= 0.3807:
                break
                
            iteration += 1
    
    if best_h is None:
        # Fallback to the first initialization
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = project_and_normalize(best_h, target_sum)
        else:
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h = project_and_normalize(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
