import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal
        h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h onto feasible set: [0,1] and sum = n_points/2"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_exact_gradient(h, k_star):
        """Compute exact gradient for given k_star"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # Add contribution from h[i] to overlap at k_star
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                # wrap around
                g[i] = -h[i + k_star - n_points] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Algorithm parameters
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    # Main optimization loop
    max_iter = 10000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute all overlaps and find max k
        overlap = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(overlap)
        current_c5 = overlap[k_star] * dx

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for k_star
        g = compute_exact_gradient(h, k_star)
        
        # Apply momentum
        velocity = momentum * prev_grad + g
        prev_grad = g
        
        # Adaptive step size
        current_step = step_size * (0.99 ** (iteration // 1000))
        
        # Update h
        h = h - current_step * velocity
        
        # Project onto constraints
        h = project_h(h)
        
        # Decrease momentum over time
        momentum = max(0.5, momentum * 0.999)

    return (best_h.tolist(), best_c5, n_points)
