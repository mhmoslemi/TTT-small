import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to satisfy the sum constraint while maintaining bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = target_sum - current_sum
    # Distribute the difference proportionally
    if diff > 0:
        # Need to increase values
        # Find indices with potential to increase
        mask = h < 1
        if np.sum(mask) > 0:
            increase = np.minimum(diff, 1 - h[mask])  # Max amount to increase
            h[mask] += increase * diff / np.sum(increase)
    else:
        # Need to decrease values
        # Find indices with potential to decrease
        mask = h > 0
        if np.sum(mask) > 0:
            decrease = np.minimum(-diff, h[mask])  # Max amount to decrease
            h[mask] -= decrease * (-diff) / np.sum(decrease)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points) * 0.5
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize particles around base solution with small random perturbations
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = c2 = 1.5
    
    # Track personal bests
    personal_best_h = [p.copy() for p in particles]
    personal_best_c5 = [compute_c5(p, dx) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best_h = personal_best_h[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    max_iter = 5000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        w = w_start - (w_start - w_end) * iter_count / max_iter
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (personal_best_h[i] - h)
            social = c2 * r2 * (global_best_h - h)
            v = w * v + cognitive + social
            
            # Clamp velocity
            v = np.clip(v, -0.1, 0.1)
            
            # Update position
            h_new = h + v
            
            # Project back to feasible region
            h_new = project_to_feasible(h_new, target_sum)
            
            # Evaluate new position
            c5_new = compute_c5(h_new, dx)
            
            # Update personal best
            if c5_new < personal_best_c5[i]:
                personal_best_c5[i] = c5_new
                personal_best_h[i] = h_new.copy()
                
                # Update global best
                if c5_new < global_best_c5:
                    global_best_c5 = c5_new
                    global_best_h = h_new.copy()
            
            # Store updated values
            particles[i] = h_new
            velocities[i] = v
            
            # Early exit if target achieved
            if global_best_c5 <= 0.3807:
                break
    
        iter_count += 1
    
    # Final evaluation of global best
    final_c5 = compute_c5(global_best_h, dx)
    
    # Ensure final solution is valid
    global_best_h = project_to_feasible(global_best_h, target_sum)
    final_c5 = compute_c5(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
