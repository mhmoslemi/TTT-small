import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate):
    """Mutate h with given mutation rate."""
    mask = np.random.random(len(h)) < mutation_rate
    h[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h = np.clip(h, 0, 1)
    return h

def crossover_h(h1, h2, crossover_rate):
    """Crossover two h vectors."""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1))
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return h1_new, h2_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    # Initialize the population with variants of the base
    for _ in range(population_size):
        h_mutated = h + np.random.normal(0, 0.02, n_points)
        h_mutated = np.clip(h_mutated, 0, 1)
        h_mutated = normalize_h(h_mutated, target_sum)
        population.append(h_mutated.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    mutation_rate_start = 0.2
    mutation_rate_end = 0.01
    crossover_rate = 0.8
    
    iteration = 0
    max_iter = 600
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate fitness of each individual
        fitnesses = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = []
        
        # Keep the best individual
        best_idx = np.argmin(fitnesses)
        new_population.append(population[best_idx].copy())
        
        # Generate offspring through selection, crossover, and mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_h(parent1, parent2, crossover_rate)
            
            # Mutation
            mutation_rate = mutation_rate_start + (mutation_rate_end - mutation_rate_start) * (iteration / max_iter)
            child1 = mutate_h(child1, mutation_rate)
            child1 = normalize_h(child1, target_sum)
            
            child2 = mutate_h(child2, mutation_rate)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Truncate population if needed
        population = new_population[:population_size]
        iteration += 1
    
    # Final best solution
    if best_h is None:
        # Fallback to the best from last generation
        fitnesses = [compute_c5(h, dx) for h in population]
        best_idx = np.argmin(fitnesses)
        best_h = population[best_idx]
        best_c5 = fitnesses[best_idx]
    
    # Final clipping and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
