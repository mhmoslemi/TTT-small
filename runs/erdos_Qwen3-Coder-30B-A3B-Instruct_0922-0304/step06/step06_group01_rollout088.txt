import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        h = np.clip(base_h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    min_temp = 1e-6
    
    sa_steps = 100000
    local_refine_every = 10000
    perturb_fraction = 0.1
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < sa_steps:
        # Compute current C5
        current_c5 = compute_c5(h, dx)
        
        # Create perturbation
        h_new = h.copy()
        n_perturb = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        h_new[indices] = np.random.uniform(0, 1, n_perturb)
        
        # Ensure sum constraint
        h_new = normalize_h(h_new, target_sum)
        h_new = np.clip(h_new, 0, 1)
        
        new_c5 = compute_c5(h_new, dx)
        
        # Accept with probability based on temperature
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temp):
            h = h_new.copy()
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        # Adjust temperature
        temp *= cooling_rate
        if temp < min_temp:
            temp = min_temp
            
        # Local refinement
        if iteration % local_refine_every == 0 and iteration > 0:
            bounds = [(0, 1) for _ in range(n_points)]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            def objective(h):
                return compute_c5(h, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = normalize_h(h, target_sum)
                
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
