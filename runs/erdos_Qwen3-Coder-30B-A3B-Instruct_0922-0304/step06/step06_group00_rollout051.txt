import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate, noise_std):
    """Mutate an individual with Gaussian noise."""
    noise = np.random.normal(0, noise_std, len(h))
    h_mut = h + noise * mutation_rate
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover(parent1, parent2, crossover_rate):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 100
    generations = 600
    mutation_rate = 0.2
    crossover_rate = 0.8
    elitism_rate = 0.1
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        individual = mutate_individual(h, 0.2, 0.05)
        individual = normalize_h(individual, target_sum)
        population.append(individual)
    
    # Start evolution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    for generation in range(generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitnesses.append(c5)
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_count = int(elitism_rate * population_size)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_rate, 0.1)
            child2 = mutate_individual(child2, mutation_rate, 0.1)
            
            # Normalize to constraints
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Decay mutation rate
        mutation_rate *= 0.995
        
    return (best_h.tolist(), best_c5, n_points)
