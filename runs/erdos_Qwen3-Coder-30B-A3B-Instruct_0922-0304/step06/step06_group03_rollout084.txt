import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    p_best = []
    p_best_scores = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle = particle + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_scores.append(compute_c5(particle, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO iterations
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update each particle
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (g_best - particle))
            
            # Update position
            particle = particle + vel
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store updated particle
            particles[i] = particle
            
            # Evaluate fitness
            score = compute_c5(particle, dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best[i] = particle.copy()
                p_best_scores[i] = score
                
                # Update global best if needed
                if score < g_best_score:
                    g_best = particle.copy()
                    g_best_score = score
        
        # Frequency domain acceleration every 50 iterations
        if it % 50 == 0 and it > 0:
            for i in range(n_particles // 4):  # Update worst 25% particles
                particle_idx = np.argmax(p_best_scores)
                # High pass filter in frequency domain
                fft_h = np.fft.fft(p_best[particle_idx])
                freqs = np.fft.fftfreq(n_points, dx)
                # Create high-pass filter (cut off at 0.3)
                filter_mask = np.abs(freqs) > 0.3
                filtered_fft = fft_h * filter_mask
                # Inverse transform
                filtered_h = np.real(np.fft.ifft(filtered_fft))
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                
                # Replace the worst particle
                particles[particle_idx] = filtered_h
                p_best[particle_idx] = filtered_h
                p_best_scores[particle_idx] = compute_c5(filtered_h, dx)
                
                # Update global best if needed
                if p_best_scores[particle_idx] < g_best_score:
                    g_best = filtered_h.copy()
                    g_best_score = p_best_scores[particle_idx]
        
        # Gradient refinement every 100 iterations
        if it % 100 == 0 and it > 0:
            # Select top 10 particles
            top_indices = np.argsort(p_best_scores)[:10]
            for idx in top_indices:
                # Run local optimization
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, p_best[idx], method='SLSQP', 
                                    bounds=bounds, constraints=constraint, options={'maxiter': 100})
                
                if result.success:
                    new_h = result.x
                    new_h = np.clip(new_h, 0, 1)
                    new_h = normalize_h(new_h, target_sum)
                    score = compute_c5(new_h, dx)
                    # Update if better
                    if score < p_best_scores[idx]:
                        p_best[idx] = new_h
                        p_best_scores[idx] = score
                        
                        if score < g_best_score:
                            g_best = new_h.copy()
                            g_best_score = score
        
        # Early stopping condition
        if g_best_score <= 0.3807:
            break
    
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
