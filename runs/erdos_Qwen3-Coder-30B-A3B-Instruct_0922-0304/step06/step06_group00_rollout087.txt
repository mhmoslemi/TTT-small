import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while keeping in [0,1]."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust values to meet sum constraint
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
    # Simple adjustment approach
    if diff > 0:
        # Need to increase values, prioritize smaller ones
        sort_indices = np.argsort(h)
        for i in sort_indices:
            if diff <= 0:
                break
            increase = min(diff, 1 - h[i])
            h[i] += increase
            diff -= increase
    else:
        # Need to decrease values, prioritize larger ones
        sort_indices = np.argsort(h)[::-1]
        for i in sort_indices:
            if diff >= 0:
                break
            decrease = min(-diff, h[i])
            h[i] -= decrease
            diff += decrease
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
        h_base = normalize_h(h_base, target_sum)
    else:
        h_base = np.random.rand(n_points)
        h_base = normalize_h(h_base, target_sum)
    
    for _ in range(n_particles):
        h = h_base + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        
    # Initialize personal bests
    pbest = [particle.copy() for particle in particles]
    pbest_c5 = [compute_c5(particle, dx) for particle in particles]
    
    # Initialize global best
    gbidx = np.argmin(pbest_c5)
    gbest = pbest[gbidx].copy()
    gb_c5 = pbest_c5[gbidx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_iter = 10000
    velocity_clamp = 0.1
    
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * (iterations / max_iter)
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v + 
                 c1 * r1 * (pbest[i] - h) + 
                 c2 * r2 * (gbest - h))
            
            # Clamp velocity
            v = np.clip(v, -velocity_clamp, velocity_clamp)
            
            # Update position
            h = h + v
            
            # Project to feasible region
            h = project_to_constraints(h, target_sum)
            
            # Update particle
            particles[i] = h
            velocities[i] = v
            
            # Evaluate
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = h.copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gb_c5:
                    gb_c5 = c5
                    gbest = h.copy()
        
        # Early stop if target achieved
        if gb_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final evaluation of best solution
    final_c5 = compute_c5(gbest, dx)
    
    return (gbest.tolist(), final_c5, n_points)
