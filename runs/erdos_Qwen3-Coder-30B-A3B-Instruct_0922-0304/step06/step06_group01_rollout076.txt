import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h function."""
    x = np.linspace(0, 2, n_points)
    h = np.full(n_points, coeffs[0])  # DC component
    for k in range(1, min(n_freq, len(coeffs))):
        h += coeffs[k] * np.cos(k * np.pi * x)
        if k < len(coeffs) // 2:
            h += coeffs[k + len(coeffs) // 2] * np.sin(k * np.pi * x)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initialize from previous solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        # Simple initial guess
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = normalize_h(h_init, target_sum)

    # Convert initial h to Fourier representation
    f_init = np.fft.rfft(h_init)
    f_init = np.concatenate([f_init.real, f_init.imag])
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = np.random.uniform(-1, 1, (n_particles, len(f_init)))
    velocities = np.zeros_like(particles)
    pbest = particles.copy()
    pbest_c5 = np.full(n_particles, float('inf'))
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    # Initialize with the initial solution
    h_base = fourier_to_h(f_init, n_points, n_freq)
    h_base = np.clip(h_base, 0, 1)
    h_base = normalize_h(h_base, target_sum)
    c5_base = compute_c5(h_base, dx)
    if c5_base < gbest_c5:
        gbest_c5 = c5_base
        gbest_idx = -1
        
    # PSO loop
    iteration = 0
    while time.time() - start_time < budget_s - 5 and iteration < max_iter:
        for i in range(n_particles):
            # Convert particle to h
            h = fourier_to_h(particles[i], n_points, n_freq)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Compute C5
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = particles[i].copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (particles[gbest_idx] - particles[i]))
            particles[i] += velocities[i]
            
        iteration += 1
        
        # Regularization to encourage better solutions
        if iteration % 10 == 0 and gbest_c5 > 0.381:
            # Add slight perturbation to avoid stagnation
            for i in range(n_particles):
                if i != gbest_idx:
                    particles[i] += np.random.uniform(-0.01, 0.01, len(particles[i]))
                    
    # Refine best solution
    if gbest_idx >= 0:
        h_final = fourier_to_h(particles[gbest_idx], n_points, n_freq)
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    else:
        h_final = h_base
        
    # Final refinement with local optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
