import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to sum to n_points / 2
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with analytic gradient
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap and find maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Analytical gradient computation
        g = np.zeros(n_points)
        # This part requires computing the analytical gradient
        # For lag k, the gradient is non-zero only at positions where
        # 1 - h[i+k] or h[i-k] affects the overlap
        # We compute the gradient for the maximum overlap case
        k = max_k
        for i in range(n_points):
            if k <= i:
                g[i] = -(1 - h[i - k]) * dx
            else:
                # Wrap around
                wrap_i = (i + n_points - k) % n_points
                g[i] = -(1 - h[wrap_i]) * dx

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Project back to feasible set
        new_h = project_h(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            noise = np.random.normal(0, 0.005, n_points)
            h = h + noise
            h = project_h(h)
        
        # Decrease step size occasionally
        if iteration % 1000 == 0:
            step_size *= 0.5

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
