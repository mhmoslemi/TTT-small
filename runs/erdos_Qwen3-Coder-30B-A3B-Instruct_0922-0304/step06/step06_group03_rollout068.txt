import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a simple comb-like initial pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Track noise amplitude for perturbations
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to guide perturbations
        corr = correlate(h, 1 - h, mode="full")
        # Identify regions with high overlap (top 20% peaks)
        sorted_corr = np.sort(corr)[::-1]
        threshold = sorted_corr[int(0.8 * len(sorted_corr))]
        high_overlap_indices = np.where(corr >= threshold)[0]
        
        # Create perturbation mask based on these regions
        perturbation_mask = np.zeros(n_points)
        # Expand influence of peaks
        for idx in high_overlap_indices:
            # Map index to central region for perturbation
            if idx < n_points:
                left_idx = max(0, idx - 2)
                right_idx = min(n_points, idx + 3)
                perturbation_mask[left_idx:right_idx] = 1.0
                
        # Apply perturbations
        perturbation = np.random.normal(0, noise_amplitude, n_points)
        perturbation = perturbation_mask * perturbation
        
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization with constraints
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else:
                # Fall back to simple perturbation if optimization fails
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = normalize_h(h, target_sum)
        except:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Reducing noise amplitude over time
        if iteration % 100 == 0:
            noise_amplitude *= 0.95
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
