import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def barrier_grad(h, mu, target_sum):
    """Compute gradient of barrier function."""
    grad = np.zeros_like(h)
    # For h[i] > 0 and h[i] < 1: d/dh[i] log(h[i]) = 1/h[i], d/dh[i] log(1-h[i]) = -1/(1-h[i])
    grad += 1.0 / (h + 1e-12) - 1.0 / (1.0 - h + 1e-12)
    # For sum constraint
    current_sum = np.sum(h)
    if current_sum < target_sum:
        grad += 1.0 / (target_sum - current_sum + 1e-12)
    else:
        grad -= 1.0 / (current_sum - target_sum + 1e-12)
    grad *= mu
    return grad

def compute_c5_gradient(h, dx, eps=1e-5):
    """Estimate gradient of C5 with finite differences."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (np.sum(h) / np.sum(h_plus)) if np.sum(h_plus) > 0 else h_plus
        grad[i] = (compute_c5(h_plus, dx) - base_c5) / eps
    return grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
    else:
        # Create a base function
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = h * (target_sum / np.sum(h)) if np.sum(h) > 0 else h
        
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Parameters
    mu = 1.0
    max_iter = 2000
    iteration = 0
    
    # Adaptive learning rate
    alpha = 1e-2
    adaptation_freq = 100
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute gradient of C5
        grad_c5 = compute_c5_gradient(h, dx)
        
        # Compute gradient of barrier
        grad_barrier = barrier_grad(h, mu, target_sum)
        
        # Combined gradient
        grad_total = grad_c5 + grad_barrier
        
        # Update with gradient descent
        h_new = h - alpha * grad_total
        
        # Clip to feasible bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy the sum constraint
        h_new = h_new * (target_sum / np.sum(h_new)) if np.sum(h_new) > 0 else h_new
        
        # Try to accept this solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        # Adjustment of learning rate and barrier parameter
        if iteration % adaptation_freq == 0 and iteration > 0:
            if mu > 1e-4:
                mu *= 0.5
            if alpha > 1e-6:
                alpha *= 0.95
                
        h = h_new
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
