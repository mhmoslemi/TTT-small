import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Use a smooth approximation of the optimal solution
        h = np.zeros(n_points)
        # Place the mass more evenly
        half_points = n_points // 2
        h[:half_points] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to the required sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Compute initial C5
    best_c5 = np.max(correlate(h, 1 - h, mode="full")) * dx
    best_h = h.copy()
    
    # SDP approach to improve the solution
    # Define the SDP variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Objective: minimize C5
    objective = cp.Minimize(C5_var)
    
    # Constraints
    constraints = []
    
    # Integration constraint
    constraints.append(cp.sum(h_var) == target_sum)
    
    # Bounding constraints
    for i in range(n_points):
        constraints.append(h_var[i] >= 0)
        constraints.append(h_var[i] <= 1)
    
    # Overlap constraints for shifts
    # For all shifts k in [-n_points+1, n_points-1]
    full_length = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    
    # Create system matrix for the constraints
    shifts = []
    A_matrix = np.zeros((full_length, n_points))
    
    for k in range(-n_points + 1, n_points):
        shift = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k) % n_points
            shift[i] = 1 - h_var[j]
        shifts.append(shift)
        A_matrix[k + n_points - 1] = shift
    
    # Add overlap constraints
    for i in range(full_length):
        constraints.append(cp.sum(cp.multiply(A_matrix[i], h_var)) <= C5_var)
    
    # Formulate and solve the SDP
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - 5)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            c5_opt = C5_var.value
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
    except:
        pass
    
    # Refine with gradient descent if necessary
    if time.time() - start_time < budget_s - 5:
        from scipy.optimize import minimize
        
        def objective(h):
            return np.max(correlate(h, 1 - h, mode="full")) * dx
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use the current best solution as starting point for local refinement
        h_start = best_h.copy()
        result = minimize(objective, h_start, method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 200})
        
        if result.success:
            h_refined = result.x
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = normalize_h(h_refined, target_sum)
            current_c5 = np.max(correlate(h_refined, 1 - h_refined, mode="full")) * dx
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_refined
    
    return (best_h.tolist(), float(best_c5), n_points)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h
