import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal with some noise
        h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.full(n_points, 1.0 / n_points)
        return h

    def compute_exact_gradient(h, k_max):
        """Compute exact gradient for the max overlap at lag k_max"""
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] = (1 - h[i + k_max]) * dx
            else:
                # Wrap around
                grad[i] = (1 - h[i + k_max - n_points]) * dx
        return grad

    best_c5 = float('inf')
    best_h = h.copy()
    
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient
        grad = compute_exact_gradient(h, max_k)
        
        # Update with momentum and adaptive step size
        velocity = momentum * prev_grad + grad
        new_h = h - step_size * velocity
        
        # Apply constraints by projecting
        new_h = project_h(new_h)
        
        h = new_h
        prev_grad = grad
        
        # Adaptive step size decay
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum = max(0.5, momentum * 0.95)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
