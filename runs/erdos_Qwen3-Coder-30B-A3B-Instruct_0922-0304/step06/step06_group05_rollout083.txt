import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal with constraints
        h = 0.5 + 0.3 * np.sin(np.linspace(0, 4*np.pi, n_points))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    max_iter = 100000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and maximum lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient
        g = np.zeros(n_points)
        k = max_k
        
        for i in range(n_points):
            if i + k < n_points:
                g[i] = (1 - h[i + k]) * dx
            else:
                g[i] = -h[i - (n_points - k)] * dx

        # Update with momentum and adaptive step size
        new_h = h - step_size * (g + momentum * prev_grad)
        prev_grad = g.copy()
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        
        # Adaptive step size
        if iteration % 1000 == 0:
            step_size *= 0.5
            momentum = max(0.5, momentum * 0.95)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
