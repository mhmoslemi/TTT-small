import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraint(h, target_sum):
    """Project h to satisfy sum constraint while keeping bounds."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Simple projection: scale to target sum
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    # Ensure bounds
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_particles = 50
    
    # Initialize particles
    if 'initial_h_values' in globals():
        particles = np.array(initial_h_values)
        particles = np.clip(particles, 0, 1)
        particles = normalize_h(particles, target_sum)
        particles = np.tile(particles, (n_particles, 1))
    else:
        particles = np.random.rand(n_particles, n_points)
        for i in range(n_particles):
            particles[i] = normalize_h(particles[i], target_sum)
    
    # Initialize velocities
    velocities = np.random.rand(n_particles, n_points) * 0.05
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_c5 = np.array([compute_c5(p, dx) for p in particles])
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Apply bounds and normalization
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = project_to_constraint(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < personal_best_c5[i]:
                personal_best_c5[i] = current_c5
                personal_best[i] = particles[i].copy()
                
                # Update global best if needed
                if current_c5 < global_best_c5:
                    global_best_c5 = current_c5
                    global_best = particles[i].copy()
        
        iteration += 1
        
        if global_best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
