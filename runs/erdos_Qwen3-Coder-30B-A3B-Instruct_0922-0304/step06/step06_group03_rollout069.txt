```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a comb pattern
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 20000
    
    # Perturbation amplitude schedule
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute cross-correlation to identify high overlap regions
        corr = correlate(h, 1 - h, mode="full")
        # Find top peaks to focus perturbations
        sorted_indices = np.argsort(corr)[::-1]
        top_peaks = sorted_indices[:max(1, n_points // 10)]
        
        # Create mask for perturbation focusing on high overlap regions
        mask = np.zeros(n_points)
        for idx in top_peaks:
            # Adjust for valid indices in full correlation
            if idx < n_points:
                mask[idx] = 1
            elif idx >= n_points:
                # Handle circular correlation
                adjust_idx = idx - n_points
                if 0 <= adjust_idx < n_points:
                    mask[adjust_idx] = 1

        # Apply perturbation
        noise = np.random.normal(0, noise_amplitude, n_points)
        perturbed_h = h + mask * noise
        
        # Clip and normalize
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = normalize_h(perturbed_h, target_sum)
        
        # Local optimization
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP with constraints
        result = minimize(objective, perturbed_h, method='SLSQP', 
                         bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h = result.x
        else:
            # If optimization failed, use perturbed values
            h = perturbed_h
            
        # Occasionally do more aggressive search
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.04, n_points), 0, 1)
            h = normalize_h(h, target_sum)
            
        # Decay noise amplitude
        noise_amplitude = max(0.001, noise_amplitude * 0.95)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
