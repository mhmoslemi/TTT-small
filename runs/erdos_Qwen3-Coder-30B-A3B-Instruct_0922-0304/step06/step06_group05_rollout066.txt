import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    stagnation_count = 0
    last_best_c5 = best_c5
    
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Make a perturbation
        h_candidate = h.copy()
        
        # Select two indices to modify
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure we don't make invalid moves
        if h_candidate[i] < 1 and h_candidate[j] > 0:
            # Determine maximum delta
            delta = min(1 - h_candidate[i], h_candidate[j])
            
            # Apply perturbation
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize to sum to n_points/2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Evaluate new solution
            candidate_c5 = compute_c5(h_candidate)
            
            # Metropolis criterion
            delta_c5 = candidate_c5 - best_c5
            
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
                h = h_candidate
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h.copy()
                    last_best_c5 = best_c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
            # Adaptive temperature control
            if stagnation_count > 500:
                T *= 1.2
                stagnation_count = 0
                
        # Cool down
        T *= alpha
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
