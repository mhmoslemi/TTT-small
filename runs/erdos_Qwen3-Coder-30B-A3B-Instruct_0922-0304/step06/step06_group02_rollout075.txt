import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    n_points = 70  # Matches typical use
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)

    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO parameters
    pop_size = 25
    max_iter = 50
    inertia_start = 0.7
    inertia_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    # Initialize swarm
    swarm = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values) + np.random.normal(0, 0.01, n_points)
        else:
            particle = np.random.rand(n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        swarm.append((particle, particle.copy()))  # position, velocity
        
    pbest = [p[0].copy() for p in swarm]
    pbest_c5 = [compute_c5(p[0], dx) for p in swarm]
    
    # Global best
    best_idx = np.argmin(pbest_c5)
    gbest = pbest[best_idx].copy()
    gbest_c5 = pbest_c5[best_idx]
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update inertia
        inertia = inertia_start - (inertia_start - inertia_end) * (iteration / max_iter)
        
        # Update each particle
        for i, (pos, vel) in enumerate(swarm):
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * vel + 
                   c1 * r1 * (pbest[i] - pos) + 
                   c2 * r2 * (gbest - pos))
            
            # Limit velocity
            vel = np.clip(vel, -velocity_max, velocity_max)
            
            # Update position
            pos += vel
            pos = np.clip(pos, 0, 1)
            pos = normalize_h(pos, target_sum)
            
            # Evaluate
            current_c5 = compute_c5(pos, dx)
            
            # Update personal best
            if current_c5 < pbest_c5[i]:
                pbest[i] = pos.copy()
                pbest_c5[i] = current_c5
                
                # Update global best
                if current_c5 < gbest_c5:
                    gbest = pos.copy()
                    gbest_c5 = current_c5
                    # Early exit if target achieved
                    if gbest_c5 <= 0.3807:
                        break
                        
            # Store updated swarm
            swarm[i] = (pos, vel)
            
        iteration += 1
        
        # Occasionally improve global best with local search
        if iteration % 10 == 0:
            # Local search around gbest
            def constraint(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            cons = {'type': 'eq', 'fun': constraint}
            
            def obj(h):
                return compute_c5(h, dx)
                
            # Add small noise to gbest to escape local minima
            h_noisy = np.clip(gbest + np.random.normal(0, 0.02, n_points), 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            result = minimize(obj, h_noisy, method='L-BFGS-B', bounds=bounds, constraints=cons, options={'maxiter': 50})
            
            if result.success:
                new_h = result.x
                new_h = np.clip(new_h, 0, 1)
                new_h = normalize_h(new_h, target_sum)
                new_c5 = compute_c5(new_h, dx)
                
                if new_c5 < gbest_c5:
                    gbest = new_h.copy()
                    gbest_c5 = new_c5
                    
                    # Update pbest if needed
                    for j in range(len(pbest)):
                        if np.allclose(pbest[j], pbest[best_idx]):
                            pbest[j] = gbest.copy()
                            pbest_c5[j] = gbest_c5
                            break

    # Final refinement
    if best_c5 > gbest_c5:
        best_h = gbest.copy()
        best_c5 = gbest_c5
        
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
