import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def build_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Build a comb-shaped h function."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
        
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    
    for i in range(n_teeth):
        start_x = i * (tooth_width + spacing)
        end_x = start_x + tooth_width
        
        start_idx = int(start_x / dx)
        end_idx = int(end_x / dx)
        
        # Transition regions
        transition_width = slope
        left_transition_start = max(0, start_idx - int(transition_width/dx))
        right_transition_start = min(n_points, end_idx + int(transition_width/dx))
        
        for j in range(left_transition_start, right_transition_start):
            x = j * dx
            if start_x <= x <= end_x:
                h[j] = 1.0
            elif start_x - transition_width <= x <= start_x:
                h[j] = max(0, 1 - (x - (start_x - transition_width)) / transition_width)
            elif end_x <= x <= end_x + transition_width:
                h[j] = max(0, 1 - (x - end_x) / transition_width)
    
    return h

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Try to use initial
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Use synthetic comb construction
        h = build_comb_h(n_points, n_teeth=7, tooth_width=2.0/7, spacing=0.02, slope=0.1)
        
    # Optimization loop for comb parameters
    param_space = {
        'n_teeth': [5, 6, 7, 8, 9],
        'tooth_width': [2.0/10, 2.0/8, 2.0/7, 2.0/6, 2.0/5],
        'spacing': [0.02, 0.03, 0.04],  # Reduced spacing to create tighter teeth 
        'slope': [0.05, 0.1, 0.15, 0.2]
    }

    # Sample parameter combinations
    param_combinations = []
    for n_teeth in param_space['n_teeth']:
        for tooth_width in param_space['tooth_width']:
            for spacing in param_space['spacing']:
                for slope in param_space['slope']:
                    if n_teeth * (tooth_width + spacing) <= 2.0:  # Feasible
                        param_combinations.append((n_teeth, tooth_width, spacing, slope))
    
    # Evaluate each combination
    for i, (n_teeth, tooth_width, spacing, slope) in enumerate(param_combinations):
        if time.time() - start_time > budget_s - 5:
            break
            
        h_candidate = build_comb_h(n_points, n_teeth, tooth_width, spacing, slope)
        h_candidate = normalize_h(h_candidate, target_sum)
        c5 = compute_c5(h_candidate, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_candidate.copy()

    # If we didn't improve, try local refinement
    if best_h is None:
        best_h = h.copy()

    # Final SLSQP refinement
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return compute_c5(h, dx)
        
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 500, 'ftol': 1e-10})
    
    if result.success:
        best_h = np.clip(result.x, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        
    final_c5 = compute_c5(best_h, dx)
    
    # Early termination if we are good
    if final_c5 <= 0.3807:
        return (best_h.tolist(), final_c5, n_points)
        
    return (best_h.tolist(), final_c5, n_points)
