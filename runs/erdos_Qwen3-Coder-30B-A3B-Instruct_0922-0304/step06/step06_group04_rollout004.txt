import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution from parent if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    # Improved optimization with multi-start
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Multi-start optimization
    num_starts = 15
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < num_starts:
        # Generate diverse initializations
        if iteration == 0:
            # First start: use the base
            h = base_h.copy()
        elif iteration < 5:
            # Random with different distributions
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h, target_sum)
        elif iteration < 10:
            # Gaussian perturbation of best
            h = best_individual + np.random.normal(0, 0.03, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
        else:
            # Random step with more varied noise
            h = np.random.uniform(0, 1, n_points)
            h = normalize_h(h, target_sum)
            
        # Local optimization with bounds and constraint
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200, 'ftol': 1e-8})
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = normalize_h(h_opt, target_sum)
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h_opt.copy()
        except:
            pass  # Continue with next initialization on failure

        iteration += 1
        
        # Early exit if target is met
        if best_c5 <= 0.3807:
            break
        
        # Adaptive timeout check
        if time.time() - start_time > budget_s - 5:
            break

    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
