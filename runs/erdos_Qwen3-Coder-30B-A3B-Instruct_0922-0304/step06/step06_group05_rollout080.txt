import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal pattern
        h = np.sin(np.linspace(0, 2*np.pi, n_points)) * 0.3 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_exact_gradient(h, k_star):
        # Compute exact gradient for the current maximum lag k_star
        n = len(h)
        g = np.zeros(n)
        for i in range(n):
            if i + k_star < n:
                g[i] = (1 - h[i + k_star]) * dx
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 50000
    iteration = 0
    initial_eta = 0.01
    momentum = 0.9
    g_prev = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current max lag
        g = compute_exact_gradient(h, max_k)
        
        # Adaptive step size
        eta = initial_eta * (0.5 ** (iteration // 1000))
        
        # Update with momentum and adaptive step size
        v = momentum * g_prev + g
        new_h = h - eta * v
        
        # Update previous gradient
        g_prev = g
        
        # Apply constraints
        new_h = project_h(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.001, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
