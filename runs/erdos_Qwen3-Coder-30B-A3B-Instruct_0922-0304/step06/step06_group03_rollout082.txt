import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    for _ in range(n_particles):
        # Small perturbation around initial solution
        particle_h = h + np.random.normal(0, 0.02, n_points)
        particle_h = np.clip(particle_h, 0, 1)
        particle_h = normalize_h(particle_h, target_sum)
        particles.append(particle_h.copy())
        velocities.append(np.zeros(n_points))
        pbest_positions.append(particle_h.copy())
        pbest_values.append(compute_c5(particle_h, dx))
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[global_best_idx].copy()
    gbest_value = pbest_values[global_best_idx]
    
    iteration = 0
    max_iter = 3000
    
    time_budget = budget_s - 2  # Leave buffer for final checks
    
    while time.time() - start_time < time_budget and iteration < max_iter:
        # Evaluate all particles
        current_values = []
        for particle in particles:
            c5 = compute_c5(particle, dx)
            current_values.append(c5)
        
        # Update personal bests and global best
        for i in range(n_particles):
            if current_values[i] < pbest_values[i]:
                pbest_values[i] = current_values[i]
                pbest_positions[i] = particles[i].copy()
            
            if current_values[i] < gbest_value:
                gbest_value = current_values[i]
                gbest_position = particles[i].copy()
        
        # Early termination
        if gbest_value <= 0.3807:
            break
            
        # PSO velocity and position updates
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (pbest_positions[i] - particles[i]) + 
                           c2 * r2 * (gbest_position - particles[i]))
            particles[i] += velocities[i]
            
            # Enforce bounds
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = normalize_h(particles[i], target_sum)
        
        # Every 50 iterations: Apply frequency domain acceleration
        if iteration % 50 == 0:
            for i in range(n_particles // 2):  # Only update worst half
                particle_idx = np.argsort(current_values)[-i-1]
                # Fast Fourier Transform approach
                h_fft = np.fft.fft(particles[particle_idx])
                # Apply high-pass filter
                freq = np.fft.fftfreq(n_points, dx)
                # Suppress low frequencies
                filter_mask = np.abs(freq) > 0.3 * np.max(freq)
                h_fft_filtered = h_fft * filter_mask
                filtered_h = np.real(np.fft.ifft(h_fft_filtered))
                # Ensure positivity and correct sum
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                particles[particle_idx] = filtered_h
            
        # Every 100 iterations: Local optimization
        if iteration % 100 == 0 and iteration > 0:
            for i in range(min(10, n_particles)):
                # Select top particles for refinement
                top_idx = np.argsort(current_values)[:10]
                idx = top_idx[i] if i < len(top_idx) else np.argmin(current_values)
                particle_h = particles[idx].copy()
                
                # Gradient-based refinement (SLSQP)
                def objective(h):
                    return compute_c5(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    result = minimize(objective, particle_h, method='SLSQP', 
                                    bounds=bounds, constraints=constraint, options={'maxiter': 20})
                    if result.success:
                        refined_h = result.x
                        refined_h = np.clip(refined_h, 0, 1)
                        refined_h = normalize_h(refined_h, target_sum)
                        # Update particle if better
                        refined_c5 = compute_c5(refined_h, dx)
                        if refined_c5 < current_values[idx]:
                            particles[idx] = refined_h
                except:
                    pass  # Skip if optimization fails
        
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(gbest_position, dx)
    
    return (gbest_position.tolist(), final_c5, n_points)
