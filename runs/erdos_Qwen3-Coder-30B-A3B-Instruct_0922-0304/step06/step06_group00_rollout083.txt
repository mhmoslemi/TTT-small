import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def project_to_feasible(h, target_sum):
    """Project h to feasible region: [0,1] and sum = target_sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_to_feasible(h, target_sum)
    else:
        # Create a base function with more structure
        h = 0.5 * np.ones(n_points)
        # Add some randomness for diversity
        h += np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier method parameters
    mu = 1.0
    decay_factor = 0.5
    mu_min = 1e-8
    patience = 0
    max_patience = 100
    
    # Optimization loop
    iterations = 0
    while time.time() - start_time < budget_s - 1 and iterations < 5000:
        # Compute gradient of C5 using finite differences
        epsilon = 1e-5
        grad_c5 = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += epsilon
            h_minus[i] -= epsilon
            h_plus = project_to_feasible(h_plus, target_sum)
            h_minus = project_to_feasible(h_minus, target_sum)
            grad_c5[i] = (compute_c5(h_plus, dx) - compute_c5(h_minus, dx)) / (2 * epsilon)
        
        # Compute barrier gradient
        barrier_grad = np.zeros(n_points)
        for i in range(n_points):
            if h[i] > 0 and h[i] < 1:
                barrier_grad[i] -= 1/h[i] + 1/(1-h[i])
        
        # Combine gradients
        total_grad = grad_c5 + mu * barrier_grad
        
        # Line search
        alpha = 1.0
        max_backtrack = 50
        for _ in range(max_backtrack):
            h_new = h - alpha * total_grad
            h_new = project_to_feasible(h_new, target_sum)
            new_c5 = compute_c5(h_new, dx)
            
            if new_c5 < best_c5:
                h = h_new
                current_c5 = new_c5
                break
            else:
                alpha *= 0.5
        else:
            # If we failed to find a descent step, use the best step so far
            current_c5 = compute_c5(h, dx)
            
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Adapt mu
        if patience > 20:
            if mu > mu_min:
                mu *= decay_factor
            patience = 0
            
        # Early termination if target achieved
        if best_c5 <= 0.3807:
            break
            
        iterations += 1
    
    # Final refinement
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
