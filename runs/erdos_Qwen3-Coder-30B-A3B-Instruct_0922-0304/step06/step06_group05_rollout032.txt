import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_exact_gradient(h, k_star):
        # Compute gradient for specific lag k_star
        g = np.zeros(n_points)
        for i in range(n_points):
            # Gradient contribution from the term h[i] * (1 - h[(i+k_star) % n_points])
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            else:
                # Wrap around case
                g[i] = (1 - h[i + k_star - n_points]) * dx
                # The second term h[i+k_star-n_points] in the derivative of h[i]*(1-h[i+k_star])
                # But that would be in the "h" term, and we are doing derivatives wrt h[i]
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Adaptive gradient descent with momentum
    iteration = 0
    eta = 0.01  # initial step size
    mu = 0.9    # momentum
    g_prev = np.zeros(n_points) 
    
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(overlap)  # Index of maximum overlap
        max_overlap = overlap[k_star] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for current k_star
        g = compute_exact_gradient(h, k_star)
        
        # Update with momentum and adaptive step
        v = mu * g_prev + g
        new_h = h - eta * v
        g_prev = g
        
        # Apply constraints
        new_h = project_h(new_h)
        h = new_h
        
        # Adaptive step size adjustment
        if iteration % 1000 == 0:
            eta *= 0.5
            if eta < 1e-6:
                eta = 1e-6
                
        # Momentum decay to avoid perpetual oscillation
        if iteration % 1000 == 0:
            mu = max(0.5, mu * 0.95)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
