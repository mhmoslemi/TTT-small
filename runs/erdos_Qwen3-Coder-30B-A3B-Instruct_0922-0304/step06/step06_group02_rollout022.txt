import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with cutting plane approach
    iteration = 0
    max_iter = 5000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find the worst lag
        overlap = correlate(h, 1 - h, mode="full")
        max_lag_idx = np.argmax(overlap)
        max_overlap = overlap[max_lag_idx] * dx
        
        # Construct optimization to reduce the worst overlap
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return max_overlap  # This will be replaced by actual constraints
            
        # Set up the constraint for the worst lag
        # We want to enforce that overlap at max_lag_idx <= current_c5
        # So we minimize max_overlap with a constraint
        # This constraint is tricky to implement directly, so we guide with local search
        
        # Local perturbation technique with simulated annealing concept
        noise_factor = 0.05
        h_noise = h + np.random.normal(0, noise_factor, len(h))
        h_noise = np.clip(h_noise, 0, 1)
        h_noise = normalize_h(h_noise, target_sum)
        
        # Try to decrease overlap at the critical lag
        def objective(h):
            return compute_c5(h, dx)
            
        # Use SLSQP for local optimization
        result = minimize(objective, h_noise, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
        
        if result.success:
            h_candidate = result.x
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            
            # Apply the evolution within the constraint
            candidate_c5 = compute_c5(h_candidate, dx)
            if candidate_c5 < current_c5 * 0.999999:
                h = h_candidate.copy()
        
        # Occasionally make larger random changes
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.05, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
