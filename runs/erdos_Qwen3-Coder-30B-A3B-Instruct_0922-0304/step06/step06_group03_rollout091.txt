import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Memory for adaptive noise
    noise_amplitude = 0.02
    noise_decay = 0.95
    
    # Keep track of recent improvements
    recent_improvements = []
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            recent_improvements.append((c5, iteration))
            
        if best_c5 <= 0.3807:
            break
            
        # Adaptive noise
        if iteration % 50 == 0:
            noise_amplitude *= noise_decay
            
        # Cross-correlation driven perturbation
        if iteration % 20 == 0:
            # Compute correlation
            corr = correlate(h, 1 - h, mode="full")
            # Identify high overlap regions
            peaks = np.argsort(corr)[-10:]  # Top 10 peaks
            # Create a mask
            perturb_mask = np.zeros(n_points)
            for peak in peaks:
                # Apply broad influence
                for i in range(n_points):
                    dist = abs(peak - i)
                    if dist < n_points // 4:
                        perturb_mask[i] += np.exp(-dist / (n_points // 8))
            # Normalize
            if np.sum(perturb_mask) > 0:
                perturb_mask = perturb_mask / np.sum(perturb_mask)
            else:
                perturb_mask = np.ones(n_points) / n_points
        
        # Apply perturbation
        if iteration % 20 == 0:
            # Apply perturbation based on cross-correlation
            noise = np.random.normal(0, noise_amplitude, n_points)
            h = h + perturb_mask * noise
        else:
            # Standard random perturbation
            noise = np.random.normal(0, noise_amplitude, n_points)
            h = h + noise
            
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # Use SLSQP with constraints to refine
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h = result.x
            else:
                # Fallback to clipping and normalization if optimization fails
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
        except:
            # Exception handler
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
