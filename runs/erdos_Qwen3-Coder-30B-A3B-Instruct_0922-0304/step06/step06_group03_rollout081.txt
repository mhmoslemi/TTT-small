import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    w = 0.8
    c1 = c2 = 2.0
    max_iter = 3000
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.05, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle = particle + np.random.normal(0, 0.05, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_vals.append(compute_c5(particle, dx))
    
    # Global best tracking
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if g_best_val <= 0.3807:
            break
            
        # PSO iteration
        for i in range(n_particles):
            particle = particles[i]
            vel = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (w * vel + 
                   c1 * r1 * (p_best[i] - particle) + 
                   c2 * r2 * (g_best - particle))
            
            # Update position
            particle = particle + vel
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            # Store particle
            particles[i] = particle
            
            # Evaluate
            c5 = compute_c5(particle, dx)
            
            # Update personal best
            if c5 < p_best_vals[i]:
                p_best[i] = particle.copy()
                p_best_vals[i] = c5
                
                # Update global best
                if c5 < g_best_val:
                    g_best = particle.copy()
                    g_best_val = c5
        
        # Frequency domain acceleration every 50 iterations
        if iteration % 50 == 0 and iteration > 0:
            for i in range(n_particles // 2, n_particles):
                # FFT-based filtering
                h_fft = np.fft.fft(particles[i])
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter with cutoff at 0.3
                cutoff = 0.3
                mask = np.abs(freqs) > cutoff
                h_fft_filtered = h_fft * mask
                filtered = np.real(np.fft.ifft(h_fft_filtered))
                filtered = np.clip(filtered, 0, 1)
                particles[i] = normalize_h(filtered, target_sum)
        
        # Gradient refinement every 100 iterations
        if iteration % 100 == 0 and iteration > 0:
            # Select top 10 particles
            top_indices = np.argsort(p_best_vals)[:10]
            for idx in top_indices:
                particle = p_best[idx]
                
                # Run SLSQP optimization
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                try:
                    result = minimize(objective, particle, method='SLSQP', 
                                    bounds=bounds, constraints=constraint, 
                                    options={'maxiter': 100})
                    if result.success:
                        refined = result.x
                        refined = np.clip(refined, 0, 1)
                        refined = normalize_h(refined, target_sum)
                        new_c5 = compute_c5(refined, dx)
                        
                        # Update if better
                        if new_c5 < p_best_vals[idx]:
                            p_best[idx] = refined.copy()
                            p_best_vals[idx] = new_c5
                            
                            # Check for global improvement
                            if new_c5 < g_best_val:
                                g_best = refined.copy()
                                g_best_val = new_c5
                except:
                    pass  # Ignore optimization failures
        
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
