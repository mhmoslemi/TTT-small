```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = normalize_h(h, target_sum)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Define bounds and constraints for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # GA parameters
    population_size = 100
    generations = 600
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    for _ in range(population_size):
        ind = h + np.random.normal(0, 0.01, n_points)
        ind = np.clip(ind, 0, 1)
        ind = normalize_h(ind, target_sum)
        population.append(ind)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            c5 = compute_c5(individual, dx)
            fitness.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            best = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[best])
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i+1) % population_size]
            
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.02)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.02)
            
            # Clip and normalize
            child1 = np.clip(child1, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = np.clip(child2, 0, 1)
            child2 = normalize_h(child2, target_sum)
            
            # Local optimization with SLSQP
            def objective(h_vec):
                return compute_c5(h_vec, dx)
            
            # Optimization for child1
            constraint = {'type': 'eq', 'fun': constraint_sum}
            result = minimize(objective, child1, method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 50})
            if result.success:
                child1 = np.clip(result.x, 0, 1)
                child1 = normalize_h(child1, target_sum)
            
            # Optimization for child2
            result = minimize(objective, child2, method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 50})
            if result.success:
                child2 = np.clip(result.x, 0, 1)
                child2 = normalize_h(child2, target_sum)
            
            offspring.extend([child1, child2])
        
        # Replace worst individuals
        sorted_indices = np.argsort(fitness)
        for i in range(population_size):
            population[sorted_indices[-(i+1)]] = offspring[i]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Adjust mutation rate
        mutation_rate = max(0.01, mutation_rate * 0.995)
        
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final check with local optimization on the best solution
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(objective, best_h, method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 100})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
    else:
        final_c5 = compute_c5(best_h, dx)
        refined_h = best_h.copy()
    
    return (refined_h.tolist(), final_c5, n_points)
```
