import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create a base function
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Barrier method parameters
    mu = 1.0
    mu_decay = 0.5
    mu_min = 1e-6
    max_iter = 10000
    iter_count = 0
    
    # Gradient step size
    step_size = 0.01
    
    # Store recent improvement counts
    recent_improvements = 0
    improvement_window = 50
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Calculate gradient of C5 using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Compute barrier function gradient
        barrier_grad = np.zeros_like(h)
        for i in range(n_points):
            barrier_grad[i] = -1.0 / h[i] - 1.0 / (1.0 - h[i]) 
        barrier_grad += 1.0 / (target_sum - np.sum(h))
        
        # Combined gradient
        total_grad = grad + mu * barrier_grad
        
        # Update h
        h_new = h - step_size * total_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            recent_improvements += 1
        else:
            recent_improvements = max(0, recent_improvements - 1)
        
        # Update h
        h = h_new
        
        # Adjust parameters
        iter_count += 1
        if iter_count % 50 == 0:
            if recent_improvements == 0:
                mu = max(mu_min, mu * mu_decay)
                recent_improvements = 0
            else:
                recent_improvements = 0
                
        # Early termination
        if best_c5 <= 0.3807:
            break
    
    # Final refinement using local optimization
    h_refined = best_h.copy()
    
    # Try small perturbations
    for _ in range(200):
        if time.time() - start_time > budget_s - 2:
            break
        perturb = np.random.normal(0, 0.005, n_points)
        h_test = h_refined + perturb
        h_test = np.clip(h_test, 0, 1)
        h_test = h_test * (target_sum / np.sum(h_test))
        c5_test = compute_c5(h_test, dx)
        if c5_test < best_c5:
            best_c5 = c5_test
            h_refined = h_test.copy()
    
    return (h_refined.tolist(), best_c5, n_points)
