import numpy as np
from scipy.signal import correlate
import time
from itertools import product

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    # Discretize to allowed values: {0, 0.25, 0.5, 0.75, 1}
    allowed_values = np.array([0, 0.25, 0.5, 0.75, 1.0])
    
    def discretize(h):
        # Round to nearest allowed value
        return np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h])
    
    h = discretize(h)
    h = h * (target_sum / np.sum(h))  # Re-normalize
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 5000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break

        # Local search using gradient descent with discrete steps
        h_copy = h.copy()
        # Try small random changes
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        
        for i in indices:
            if np.random.random() < 0.7:  # 70% chance to modify
                old_val = h_copy[i]
                # Try slight modification
                neighbor_vals = []
                for val in allowed_values:
                    if abs(val - old_val) > 1e-8:  # Different value
                        neighbor_vals.append(val)
                if neighbor_vals:
                    new_val = np.random.choice(neighbor_vals)
                    h_copy[i] = new_val
                    # Ensure new h is valid
                    h_copy = discretize(h_copy)
                    new_sum = np.sum(h_copy)
                    if abs(new_sum - target_sum) < 1e-5:
                        h_copy = h_copy * (target_sum / new_sum)  # Re-normalize if needed
                    else:
                        h_copy = h_copy * (target_sum / new_sum)  # Ensure it's properly normalized
                else:
                    # If no neighbor, don't change but still normalize
                    pass
                # Normalize
                h_copy = np.clip(h_copy, 0, 1)
                h_copy = h_copy * (target_sum / np.sum(h_copy))
                h_copy = discretize(h_copy)  # Discretize again
                
        h = h_copy.copy()
        
        iteration += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
