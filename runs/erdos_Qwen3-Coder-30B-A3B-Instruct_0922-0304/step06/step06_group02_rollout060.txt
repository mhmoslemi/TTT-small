import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))
    
    # Define variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == target_sum]
    
    # Overlap constraints
    for k in range(-n_points + 1, n_points):
        if k >= 0:
            # Valid indices for h[i] * (1 - h[i + k])
            indices = np.arange(n_points - k)
            expr = cp.sum(cp.multiply(h_var[indices], 1 - h_var[indices + k])) * dx
        else:
            # Valid indices for h[i] * (1 - h[i + k])
            indices = np.arange(-k, n_points)
            expr = cp.sum(cp.multiply(h_var[indices], 1 - h_var[indices + k])) * dx
        constraints.append(expr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    
    # Solve the problem
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, max_iters=5000, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_result = np.array(h_var.value).flatten()
            c5_bound = C.value
        else:
            # Fallback to local search if QP fails
            h_result = h.copy()
            c5_bound = np.max(correlate(h_result, 1 - h_result, mode="full")) * dx
    except Exception:
        # Fallback to local search if QP fails
        h_result = h.copy()
        c5_bound = np.max(correlate(h_result, 1 - h_result, mode="full")) * dx
    
    # Local refinement
    best_h = h_result.copy()
    best_c5 = c5_bound
    
    def evaluate_c5(h_vec):
        return np.max(correlate(h_vec, 1 - h_vec, mode="full")) * dx
    
    def objective(h_vec):
        return evaluate_c5(h_vec)
    
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the solution
        h_pert = np.clip(h_result + np.random.normal(0, 0.01, n_points), 0, 1)
        h_pert = h_pert * (target_sum / np.sum(h_pert))
        
        # Evaluate
        c5_test = evaluate_c5(h_pert)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_pert.copy()
            h_result = h_pert.copy()
            
        # Keep trying
        if best_c5 <= 0.3807:
            break
            
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
