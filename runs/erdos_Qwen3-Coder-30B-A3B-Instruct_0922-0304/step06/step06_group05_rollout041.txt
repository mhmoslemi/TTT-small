import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    stagnation_count = 0
    max_stagnation = 1000
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Store current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Perturb h
        h_candidate = h.copy()
        
        # Select two indices for transfer
        i, j = np.random.choice(n_points, 2, replace=False)
        
        # Ensure we can transfer
        if h_candidate[i] < 1 and h_candidate[j] > 0:
            delta = min(1 - h_candidate[i], h_candidate[j])
            delta = np.random.uniform(0, delta)
            
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            new_c5 = compute_c5(h_candidate)
            
            # Metropolis criterion
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_candidate
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cool down
        T *= alpha
        
        # Occasionally restart if stuck
        if stagnation_count > max_stagnation:
            T = max(T * 1.2, 1e-5)
            stagnation_count = 0

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
