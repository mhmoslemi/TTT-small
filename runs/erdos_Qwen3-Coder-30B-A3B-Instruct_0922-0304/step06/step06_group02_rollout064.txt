import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Start with a simple step function
    h = np.ones(n_points)
    h[n_points//2:] = 0
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Convex optimization approach
    def solve_qp(h_initial, n_points, target_sum, dx):
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        constraints = [
            cp.sum(h) == target_sum,
            h >= 0,
            h <= 1,
        ]
        
        # Add overlap constraints for all lags
        for k in range(-n_points+1, n_points):
            # Compute valid indices for this lag
            if k >= 0:
                i_indices = np.arange(n_points - k)
                j_indices = np.arange(k, n_points)
            else:
                i_indices = np.arange(-k, n_points)
                j_indices = np.arange(n_points + k)
            
            if len(i_indices) > 0:
                overlap_expr = cp.sum(cp.multiply(h[i_indices], 1 - h[j_indices])) * dx
                constraints.append(overlap_expr <= C)
        
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.GUROBI, verbose=False, TimeLimit=budget_s/2)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, prob.value
        except:
            pass
        return h_initial, float('inf')
    
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        h_sol, c5 = solve_qp(h, n_points, target_sum, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_sol.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Perturb the solution slightly and restart optimization
        noise = np.random.normal(0, 0.01, n_points)
        h = np.clip(h_sol + noise, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        iteration += 1
    
    # Final validation
    final_c5 = max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    if final_c5 < best_c5:
        best_c5 = final_c5
        
    return (best_h.tolist(), best_c5, n_points)
