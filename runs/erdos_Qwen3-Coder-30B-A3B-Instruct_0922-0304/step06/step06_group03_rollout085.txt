import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, dx, target_sum):
    """Create a comb pattern with given spacing."""
    h = np.zeros(n_points)
    tooth_positions = np.arange(0, n_points, spacing)
    tooth_positions = tooth_positions[tooth_positions < n_points]
    h[tooth_positions] = 1.0
    h = normalize_h(h, target_sum)
    return h

def refine_pattern(h, dx, target_sum):
    """Refine a pattern using SLSQP."""
    bounds = [(0, 1) for _ in range(len(h))]
    
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
    if result.success:
        h = result.x
    else:
        # If optimization failed, clip and normalize
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbation."""
    h = h + np.random.normal(0, sigma, len(h))
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    if best_c5 <= 0.3807:
        return (best_h.tolist(), best_c5, n_points)
    
    # Multi-resolution comb pattern search
    spacings = [2, 3, 4, 5, 6]
    
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Try different comb patterns
        for spacing in spacings:
            pattern = create_comb_pattern(n_points, spacing, dx, target_sum)
            pattern = refine_pattern(pattern, dx, target_sum)
            c5 = compute_c5(pattern, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = pattern.copy()
                
        # Add more fine-grained comb patterns
        if iteration % 100 == 0:
            for spacing in [7, 8, 9]:
                pattern = create_comb_pattern(n_points, spacing, dx, target_sum)
                pattern = perturb_pattern(pattern, 0.02)
                pattern = refine_pattern(pattern, dx, target_sum)
                c5 = compute_c5(pattern, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = pattern.copy()
        
        # Random perturbation to escape local minima
        if iteration % 50 == 0:
            h = perturb_pattern(h, 0.01)
            h = normalize_h(h, target_sum)
            h = refine_pattern(h, dx, target_sum)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        iteration += 1
        
        # Early termination if we've found a good solution
        if best_c5 <= 0.3807:
            break
    
    # Final optimization on the best pattern
    if best_h is not None:
        best_h = refine_pattern(best_h, dx, target_sum)
        final_c5 = compute_c5(best_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
