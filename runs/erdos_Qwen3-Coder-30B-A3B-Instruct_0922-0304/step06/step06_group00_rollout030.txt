import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def barrier_gradient(h, target_sum, mu):
    """Compute gradient of the barrier function."""
    grad = np.zeros_like(h)
    # For h[i] > 0 and h[i] < 1
    grad += -1.0 / h + 1.0 / (1.0 - h)
    # For sum constraint
    grad_sum = np.sum(h) - target_sum
    grad += mu / (target_sum - grad_sum)
    return grad

def gradient_c5(h, dx, epsilon=1e-5):
    """Numerical gradient of C5 with respect to h."""
    n = len(h)
    grad = np.zeros(n)
    base_c5 = compute_c5(h, dx)
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (np.sum(h) / np.sum(h_plus)) if np.sum(h_plus) > 0 else h_plus
        grad[i] = (compute_c5(h_plus, dx) - base_c5) / epsilon
    return grad

def line_search(h, d, mu, target_sum, dx, max_iter=50):
    """Backtracking line search."""
    alpha = 1.0
    c1 = 1e-4
    rho = 0.5
    current_c5 = compute_c5(h, dx)
    
    for _ in range(max_iter):
        h_new = h + alpha * d
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new)) if np.sum(h_new) > 0 else h_new
        
        new_c5 = compute_c5(h_new, dx)
        if new_c5 <= current_c5 + c1 * alpha * np.dot(d, gradient_c5(h, dx)):
            return h_new, alpha
        alpha *= rho
    
    return h, alpha

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.random.rand(n_points) * 0.5 + 0.25
        h = h * (target_sum / np.sum(h))
        
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    mu = 1.0
    iteration = 0
    decay_interval = 50
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradients
        grad_c5 = gradient_c5(h, dx)
        grad_barrier = barrier_gradient(h, target_sum, mu)
        grad = grad_c5 + mu * grad_barrier
        
        # Line search direction
        d = -grad
        h, alpha = line_search(h, d, mu, target_sum, dx)
        
        # Evaluate
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
        if iteration % decay_interval == 0:
            mu = max(0.1, mu * 0.5)
            
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
