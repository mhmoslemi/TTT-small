import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, mutation_rate=0.05, target_sum=35):
    """Mutate h with constraint maintenance."""
    h_mut = h.copy()
    mask = np.random.random(len(h)) < mutation_rate
    h_mut[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover_h(h1, h2, crossover_rate=0.7):
    """Crossover two h vectors."""
    if np.random.random() > crossover_rate:
        return h1.copy(), h2.copy()
    
    crossover_point = np.random.randint(1, len(h1) - 1)
    h1_new = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    h2_new = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    
    return h1_new, h2_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from the parent construction or predefined pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # GA parameters
    population_size = 30
    elite_size = population_size // 10
    generations = 500
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_perturbed = h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        population.append(h_perturbed)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and generation < generations:
        # Evaluate all candidates
        fitness = []
        for i, h_candidate in enumerate(population):
            c5 = compute_c5(h_candidate, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Selection - top 50%
        selected = population[:population_size // 2]
        
        # Elitism - preserve best
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover_h(parent1, parent2)
            
            # Mutate
            child1 = mutate_h(child1, target_sum=target_sum)
            child2 = mutate_h(child2, target_sum=target_sum)
            
            new_population.extend([child1, child2])
            
        # Trim to population size
        population = new_population[:population_size]
        
        # Local refinement using SLSQP for top candidates
        for i in range(elite_size):
            h_refined = population[i]
            
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
                
            # Use SLSQP with local optimization
            result = minimize(objective, h_refined, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                population[i] = h_candidate
            
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
