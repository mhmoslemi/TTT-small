import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a simple initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Discretize the values to a coarse grid to explore local optima efficiently
    discrete_levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    iteration = 0
    max_iter = 30000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Stochastic local search
        h_new = h.copy()
        
        # Randomly modify a few points
        num_changes = np.random.randint(1, 5)
        indices = np.random.choice(n_points, size=num_changes, replace=False)
        
        for idx in indices:
            # Change to a discrete level
            old_val = h_new[idx]
            # Try to keep it near the current value but discrete
            if old_val == 0:
                h_new[idx] = np.random.choice(discrete_levels[1:])  # Avoid 0
            elif old_val == 1:
                h_new[idx] = np.random.choice(discrete_levels[:-1])  # Avoid 1
            else:
                # Choose a nearby discrete value
                nearest = discrete_levels[np.argmin(np.abs(discrete_levels - old_val))]
                # Add randomness
                move = np.random.choice([-0.25, 0.0, 0.25])
                new_val = nearest + move
                h_new[idx] = np.clip(new_val, 0, 1)
        
        # Ensure the sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        h = h_new
        iteration += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
