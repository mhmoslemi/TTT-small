import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    # Normalize
    h_base = h_base * (target_sum / np.sum(h_base))
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Bi-linear programming approach
    # Variables
    h = cp.Variable(n_points)
    z = cp.Variable(n_points * 2 - 1)  # For all shifts
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    
    # Objective
    obj = cp.Minimize(cp.max(z))
    
    # Problem
    prob = cp.Problem(obj, constraints)
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Create shifted version of h_base for initialization
        h_base = np.clip(h_base, 0, 1)
        h_base = h_base * (target_sum / np.sum(h_base))
        
        # Use heuristic to build starting variables
        h.value = h_base.copy()
        
        # Try to solve the bilinear program
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iter=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h.value
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    if best_c5 <= 0.3807:
                        break
        except:
            pass
        
        # Local search: perturb the solution
        h_perturbed = h_base + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (target_sum / np.sum(h_perturbed))
        c5 = compute_c5(h_perturbed, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
            if best_c5 <= 0.3807:
                break
        
        # Adaptive perturbation
        if iteration % 5 == 0:
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h_base = h_base + noise
            h_base = np.clip(h_base, 0, 1)
            h_base = h_base * (target_sum / np.sum(h_base))
        
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
