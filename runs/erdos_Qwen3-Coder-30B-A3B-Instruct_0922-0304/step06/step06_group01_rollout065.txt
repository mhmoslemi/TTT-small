import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct h as a comb function."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0
    
    h = np.zeros(n_points)
    dx = 2.0 / n_points
    tooth_height = 1.0
    
    for i in range(n_teeth):
        start_x = i * (tooth_width + spacing)
        end_x = start_x + tooth_width
        
        start_idx = int(start_x / dx)
        end_idx = int(end_x / dx)
        
        if start_idx >= n_points:
            break
            
        end_idx = min(end_idx, n_points)
        
        # Create linear transition
        for j in range(start_idx, end_idx):
            x_pos = j * dx
            if x_pos < start_x + slope:
                h[j] += tooth_height * (1 - (start_x + slope - x_pos) / slope)
            elif x_pos > end_x - slope:
                h[j] += tooth_height * (1 - (x_pos - (end_x - slope)) / slope)
            else:
                h[j] += tooth_height
                
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Construct initial comb structure
    if 'initial_h_values' in globals():

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = construct_comb_h(n_points, n_teeth=7)
        h = normalize_h(h, target_sum)
    
    # Parameter search bounds
    param_bounds = [
        (0.1, 0.5),     # tooth_width
        (0.001, 0.2),   # spacing
        (0.01, 0.5)     # slope
    ]
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative improvement using comb structure optimizations
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Optimize parameters
        best_params = [0.2857, 0.02, 0.14]
        best_local_c5 = float('inf')
        best_local_h = h.copy()
        
        # Grid search within bounds with local refinement
        search_points = 100
        for i in range(search_points):
            tooth_w = np.random.uniform(*param_bounds[0])
            spacing = np.random.uniform(*param_bounds[1])
            slope = np.random.uniform(*param_bounds[2])
            
            try:
                new_h = construct_comb_h(n_points, n_teeth=7, tooth_width=tooth_w, spacing=spacing, slope=slope)
                new_h = normalize_h(new_h, target_sum)
                
                c5 = compute_c5(new_h, dx)
                if c5 < best_local_c5:
                    best_local_c5 = c5
                    best_local_h = new_h.copy()
            except:
                continue
                
        h = best_local_h.copy()
        c5 = best_local_c5
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Early exit if threshold reached
        if best_c5 <= 0.3807:
            break
            
        # Local refinement with SLSQP to polish
        bounds = [(0, 1) for _ in range(n_points)]
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        def objective(h):
            return compute_c5(h, dx)
            
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h = h_new.copy()
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
