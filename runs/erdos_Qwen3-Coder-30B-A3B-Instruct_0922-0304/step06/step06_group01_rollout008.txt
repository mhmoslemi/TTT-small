import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coeffs, n_points, dx):
    """Convert Fourier coefficients to h(x) using cosine basis."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] * 0.5 * np.ones(n_points)
    for k in range(1, len(coeffs)):
        h += coeffs[k] * np.cos(k * np.pi * x)
    return np.clip(h, 0, 1)

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = normalize_h(h_init, target_sum)
    else:
        # Initialize with a simple rectangle
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0
        h_init = normalize_h(h_init, target_sum)
        
    # Fourier representation for optimization (using 15 coefficients)
    n_freq = 15
    bounds = [(-2, 2) for _ in range(n_freq + 1)]  # a0 to a_nfreq
    best_c5 = float('inf')
    best_h = h_init.copy()
    
    # Fourier coefficients for initial h
    h0_fft = np.fft.fft(h_init)
    initial_coeffs = [h0_fft[0].real] + [h0_fft[i].real for i in range(1, n_freq+1)]
    
    def objective_fourier(coeffs):
        h = fourier_to_h(coeffs, n_points, dx)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    # Differential Evolution for global search
    de_result = differential_evolution(
        objective_fourier,
        bounds,
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1),
        recombination=0.7,
        seed=seed,
        disp=False,
        callback=lambda x, convergence: None
    )
    
    if de_result.success:
        h_final = fourier_to_h(de_result.x, n_points, dx)
        h_final = normalize_h(h_final, target_sum)
        c5 = compute_c5(h_final, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
    
    # Local refinement with SLSQP
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds_h = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        lambda h: compute_c5(h, dx),
        best_h.copy(),
        method='SLSQP',
        bounds=bounds_h,
        constraints=constraint,
        options={'maxiter': 100}
    )
    
    if result.success:
        h_refined = np.clip(result.x, 0, 1)
        h_refined = normalize_h(h_refined, target_sum)
        c5_refined = compute_c5(h_refined, dx)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined.copy()
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
