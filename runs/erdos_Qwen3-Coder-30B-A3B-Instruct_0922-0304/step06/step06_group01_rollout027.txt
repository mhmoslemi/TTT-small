import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Simple initial construction
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Parameters for PSO
    n_particles = 50
    n_freq = 15  # number of Fourier components
    max_iter = 500
    
    # Initialize particles
    particles = []
    velocities = []
    
    for _ in range(n_particles):
        # Random Fourier coefficients
        a = np.random.randn(n_freq)
        b = np.random.randn(n_freq)
        particles.append(np.concatenate([a, b]))
        velocities.append(np.random.randn(2*n_freq) * 0.1)
    
    # Global best
    gbest = None
    gbest_c5 = float('inf')
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i, particle in enumerate(particles):
            # Compute h from Fourier coefficients
            h_temp = np.zeros(n_points)
            x_vals = np.linspace(0, 2, n_points)
            
            for k in range(n_freq):
                if k == 0:
                    h_temp += particle[k]  # DC component
                else:
                    h_temp += particle[k] * np.cos(k * np.pi * x_vals) + \
                              particle[k + n_freq] * np.sin(k * np.pi * x_vals)
            
            # Clip and normalize
            h_temp = np.clip(h_temp, 0, 1)
            current_sum = np.sum(h_temp)
            if current_sum > 0:
                h_temp = h_temp * (target_sum / current_sum)
            
            # Evaluate C5
            c5 = compute_c5(h_temp, dx)
            
            # Update personal best
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_temp.copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest = particle.copy()
            
            # PSO update
            if gbest is not None:
                r1, r2 = np.random.rand(2, 2*n_freq)
                velocities[i] = w * velocities[i] + \
                                c1 * r1 * (gbest - particle) + \
                                c2 * r2 * (particle - particle)
                particles[i] = particle + velocities[i]
                
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final refinement with SLSQP
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, best_h.copy(), method='SLSQP', 
                      bounds=bounds, constraints=constraint, 
                      options={'maxiter': 100})
    
    if result.success:
        refined_h = np.clip(result.x, 0, 1)
        current_sum = np.sum(refined_h)
        if current_sum > 0:
            refined_h = refined_h * (target_sum / current_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h.copy()
    
    # Check final bounds
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
