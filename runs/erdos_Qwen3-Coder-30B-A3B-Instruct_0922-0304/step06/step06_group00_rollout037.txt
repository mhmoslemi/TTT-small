```python
import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def enforce_constraints(h, target_sum):
    """Enforce bounds and sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    positions = []
    velocities = []
    
    # Use initial if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.random.uniform(0, 1, n_points)
        base_h = normalize_h(base_h, target_sum)
    
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = enforce_constraints(h, target_sum)
        positions.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_vel = 0.1
    
    # Initialize personal bests
    pbest_positions = [pos.copy() for pos in positions]
    pbest_values = [compute_c5(pos, dx) for pos in positions]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO loop
    iterations = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        w = w_start - (w_start - w_end) * (iterations / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (w * velocities[i] + 
                     c1 * r1 * (pbest_positions[i] - positions[i]) + 
                     c2 * r2 * (gbest_position - positions[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -max_vel, max_vel)
            velocities[i] = v_new
            
            # Update position
            positions[i] += velocities[i]
            
            # Enforce constraints
            positions[i] = enforce_constraints(positions[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(positions[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = positions[i].copy()
                
                # Check if new global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = positions[i].copy()
                    
                    # Early termination
                    if gbest_value <= 0.3807:
                        break
                        
        iterations += 1
    
    # Return best solution
    final_c5 = compute_c5(gbest_position, dx)
    return (gbest_position.tolist(), final_c5, n_points)
```
