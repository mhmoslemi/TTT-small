import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Discretize to allowed values
    allowed_values = [0, 0.25, 0.5, 0.75, 1.0]
    
    # Initialize with existing construction or simple step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        # Discretize to allowed values
        h_discrete = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h])
    else:
        # Create a simple step function with 0.5 amplitude
        h_discrete = np.ones(n_points)
        h_discrete[n_points//2:] = 0.5
        h_discrete = np.clip(h_discrete, 0, 1)
        # Discretize to allowed values
        h_discrete = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_discrete])
    
    # Normalize to target sum
    current_sum = np.sum(h_discrete)
    if current_sum > 0:
        h_discrete = h_discrete * (target_sum / current_sum)
    
    # Final discretization
    h_discrete = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_discrete])
    
    # Ensure sum constraint is satisfied by adjusting one element
    current_sum = np.sum(h_discrete)
    diff = target_sum - current_sum
    if abs(diff) > 1e-6:
        # Adjust last element to maintain sum constraint
        h_discrete[-1] += diff
        # Keep within bounds
        h_discrete[-1] = np.clip(h_discrete[-1], 0, 1)
        
        # Ensure final discretization
        h_discrete = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_discrete])
    
    best_c5 = float('inf')
    best_h = h_discrete.copy()
    
    # Optimization loop with gradient-free method
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(best_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = best_h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Try various random mutations with different intensities
        mutation_type = iteration % 5
        h_new = best_h.copy()
        
        if mutation_type == 0:
            # Small random changes
            perturb = np.random.normal(0, 0.01, n_points)
            h_new = h_new + perturb
            h_new = np.clip(h_new, 0, 1)
        elif mutation_type == 1:
            # Larger changes
            perturb = np.random.normal(0, 0.05, n_points)
            h_new = h_new + perturb
            h_new = np.clip(h_new, 0, 1)
        elif mutation_type == 2:
            # Discrete random changes
            for i in range(n_points):
                if np.random.random() < 0.2:
                    # Select a random discrete value
                    h_new[i] = np.random.choice(allowed_values)
        elif mutation_type == 3:
            # Single large jump
            idx = np.random.randint(0, n_points)
            h_new[idx] = np.random.choice(allowed_values)
        else:  # mutation_type == 4
            # Random distinct values
            indices = np.random.choice(n_points, size=min(5, n_points//2), replace=False)
            for i in indices:
                h_new[i] = np.random.choice(allowed_values)
        
        # Discretize and normalize
        h_new = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_new])
        
        # Normalize to target sum
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        # Ensure final discretization
        h_new = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_new])
        
        # Adjust to exactly meet sum
        current_sum = np.sum(h_new)
        diff = target_sum - current_sum
        # Distribute difference to last element
        if abs(diff) > 1e-6:
            h_new[-1] += diff
            h_new[-1] = np.clip(h_new[-1], 0, 1)
            h_new = np.array([allowed_values[np.argmin(np.abs(allowed_values - x))] for x in h_new])
        
        # Set new h to the best of current or mutated
        h_new_c5 = compute_c5(h_new, dx)
        if h_new_c5 < c5:
            best_h = h_new
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
