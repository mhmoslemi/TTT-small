import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Gradient descent with line search
    def surrogate_loss(h, eps=1e-3):
        # Use a soft maximum approximation for differentiability
        corr = correlate(h, 1 - h, mode="full")
        # For numerical stability, we approximate max with log-sum-exp
        max_val = np.max(corr)
        return max_val * dx
    
    def grad_surrogate_loss(h, eps=1e-3):
        # Compute a subgradient of surrogate loss
        corr = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        # Gradient of correlation term in direction of h
        # We approximate gradient by finite difference or due to convolution properties
        # By linearity, approximate with finite diff analysis or convolution gradients
        # Since the computation is complex, we'll use numerical gradient
        eps_h = 1e-4
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps_h
            h_minus = h.copy()
            h_minus[i] -= eps_h
            val_plus = surrogate_loss(h_plus)
            val_minus = surrogate_loss(h_minus)
            grad[i] = (val_plus - val_minus) / (2 * eps_h)
        return grad
    
    # Optimization loop
    iteration = 0
    max_iter = 10000
    step_size = 1e-2
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        h = best_individual.copy()
        
        # Adaptive perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Gradient descent step
        for _ in range(20):  # Inner iterations for better gradient descent
            grad = grad_surrogate_loss(h)
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = normalize_h(h_new, target_sum)
            h = h_new
        
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
