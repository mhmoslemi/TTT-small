import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Simple symmetric construction as fallback
        h = np.ones(n_points)
        mid = n_points // 2
        h[mid:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Iterative refinement with sequential constraint tightening
    iteration = 0
    max_iter = 10000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Find worst lag
        overlap_full = correlate(h, 1 - h, mode="full")
        max_idx = np.argmax(overlap_full)
        max_lag = max_idx - (n_points - 1)  # convert to actual lag
        
        # Construct constraint for worst lag
        def constraint_worst_lag(h):
            # Calculate the specific overlap contribution for this lag
            if max_lag >= 0:
                # h[i] * (1 - h[i + max_lag])
                overlap_sum = 0
                for i in range(n_points - max_lag):
                    overlap_sum += h[i] * (1 - h[i + max_lag])
                overlap_sum *= dx
            else:
                # h[i - max_lag] * (1 - h[i])
                overlap_sum = 0
                for i in range(-max_lag, n_points):
                    overlap_sum += h[i - (-max_lag)] * (1 - h[i])
                overlap_sum *= dx
            return overlap_sum - current_c5
        
        # Define optimization for this specific constraint
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'ineq', 'fun': lambda x: current_c5 - constraint_worst_lag(x)}
        
        # Objective is to minimize C₅ (which we'll reflect in constraint tightening)
        def objective(h):
            return compute_c5(h, dx)
        
        # Random perturbed initialization for the optimization
        h_trial = np.clip(h + np.random.normal(0, 0.02, n_points), 0, 1)
        h_trial = normalize_h(h_trial, target_sum)
        
        # Use SLSQP with constraint on current worst lag, reduce c5
        try:
            result = minimize(objective, h_trial, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success and not np.isnan(result.x).any():
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                # Check if it helped
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5:
                    h = h_candidate.copy()
        except:
            # fallback to simple perturbation if optimization fails
            pass
            
        # Periodic global changes to escape local minima
        if iteration % 50 == 0:
            # Perturb more aggressively
            h = np.clip(h + np.random.normal(0, 0.05, len(h)), 0, 1)
            h = normalize_h(h, target_sum)
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
