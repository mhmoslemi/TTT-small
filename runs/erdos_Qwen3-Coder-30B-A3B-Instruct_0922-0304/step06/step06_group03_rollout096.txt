import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Discretization levels
    levels = np.array([0.0, 0.25, 0.5, 0.75, 1.0])
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Discretize to nearest allowed value
        h_discrete = np.zeros(n_points)
        for i in range(n_points):
            distances = np.abs(h[i] - levels)
            h_discrete[i] = levels[np.argmin(distances)]
        h = h_discrete
    else:
        # Create a basic initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Discretize
        h_discrete = np.zeros(n_points)
        for i in range(n_points):
            distances = np.abs(h[i] - levels)
            h_discrete[i] = levels[np.argmin(distances)]
        h = h_discrete
        
    # Ensure sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    # Discretize again after scaling
    h_discrete = np.zeros(n_points)
    for i in range(n_points):
        distances = np.abs(h[i] - levels)
        h_discrete[i] = levels[np.argmin(distances)]
    h = h_discrete
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Make a small random change
        idx = np.random.randint(0, n_points)
        # Select a new level randomly (but avoid current)
        current_level = h[idx]
        available_levels = levels[levels != current_level]
        if len(available_levels) > 0:
            new_level = np.random.choice(available_levels)
            h[idx] = new_level
        else:
            h[idx] = np.random.choice(levels)
            
        # Ensure the constraint sum(h) = target_sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
            
        # Re-discretize after adjustment
        h_discrete = np.zeros(n_points)
        for i in range(n_points):
            distances = np.abs(h[i] - levels)
            h_discrete[i] = levels[np.argmin(distances)]
        h = h_discrete
        
        iteration += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
