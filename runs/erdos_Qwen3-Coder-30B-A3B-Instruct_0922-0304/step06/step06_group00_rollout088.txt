import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_feasible(h, target_sum):
    """Project h to feasible region with sum constraint."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Create base function with some structure
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1
        base_h += np.random.normal(0, 0.02, n_points)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    # Initialize particles around base
    for i in range(n_particles):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = project_to_feasible(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    v_max = 0.1
    
    # Initialize bests
    pbest_positions = [p.copy() for p in particles]
    pbest_values = [compute_c5(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update inertia weight
        w = w_start - (w_start - w_end) * (iteration / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.uniform(0, 1, 2)
            vel_update = (w * velocities[i] + 
                         c1 * r1 * (pbest_positions[i] - particles[i]) + 
                         c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            vel_update = np.clip(vel_update, -v_max, v_max)
            velocities[i] = vel_update
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], target_sum)
            
            # Evaluate and update personal best
            current_c5 = compute_c5(particles[i], dx)
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = particles[i].copy()
        
        # Early termination
        if gbest_value <= 0.3807:
            break
            
        iteration += 1
    
    return (gbest_position.tolist(), gbest_value, n_points)
