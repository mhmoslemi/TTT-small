import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with line search and perturbations
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Compute gradient exactly for max_k
        g = np.zeros(n_points)
        # For lag k, gradient is 1-h[i+k] if i+k < n, -h[i-k] if i-k >= 0
        # To ensure it works as circular ring:
        for i in range(n_points):
            # Gradient contribution from h[i] on the term (1-h[i+k])
            if (i + max_k) < n_points:
                g[i] += (1 - h[(i + max_k) % n_points]) * dx
            # Gradient contribution from h[i] on the term -h[i-k]
            if (i - max_k) >= 0:
                g[i] -= h[(i - max_k) % n_points] * dx

        # Line search
        eta = 0.01
        alpha = 0.1
        beta = 0.5
        max_line_search_iter = 10
        
        for _ in range(max_line_search_iter):
            new_h = h - eta * g
            # Projection onto feasible set
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            new_c5 = compute_c5(new_h)
            if new_c5 < max_overlap - alpha * eta * np.dot(g, g):
                break
            eta *= beta

        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            # Noise with decreasing variance
            noise_std = 0.01 * np.exp(-iteration / 1000.0)
            pert = np.random.normal(0, noise_std, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
