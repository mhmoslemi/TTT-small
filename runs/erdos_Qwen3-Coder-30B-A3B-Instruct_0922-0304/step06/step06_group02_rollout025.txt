import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Construct initial guess
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iteration = 0
    max_iter = 20000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Hybrid optimization: try both stochastic and deterministic methods
        # Try L-BFGS-B first
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        
        # Add small perturbation for diversity
        noise = np.random.normal(0, 0.02, n_points)
        h_noisy = np.clip(h + noise, 0, 1)
        h_noisy = normalize_h(h_noisy, target_sum)
        
        # L-BFGS-B optimization
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, 
                         options={'maxiter': 50, 'ftol': 1e-8})
        
        if result.success:
            candidate_h = result.x
            candidate_h = np.clip(candidate_h, 0, 1)
            candidate_h = normalize_h(candidate_h, target_sum)
            candidate_c5 = compute_c5(candidate_h, dx)
            if candidate_c5 < current_c5 * 0.99999:
                h = candidate_h.copy()
        
        # Periodic global search with structured perturbations
        if iteration % 30 == 0:
            # Create a new solution with structured changes
            h_new = h.copy()
            # Perturb a few random points
            for i in range(5):
                idx = np.random.randint(0, n_points)
                delta = np.random.normal(0, 0.075)
                h_new[idx] = min(1, max(0, h_new[idx] + delta))
            
            h_new = normalize_h(h_new, target_sum)
            
            # Evaluate new solution
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < current_c5:
                h = h_new.copy()
        
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
