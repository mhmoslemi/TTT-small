import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population.append(base_h.copy())
        for _ in range(population_size - 1):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h.copy())
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            h = np.clip(h, 0, 1)
            population.append(h.copy())
    
    # Track best solution
    best_c5 = float('inf')
    best_individual = None
    
    # Genetic algorithm parameters
    generations = 0
    stagnation_count = 0
    max_stagnation = 20
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected = population[:population_size//2]
        
        # Crossover
        offspring = []
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            
            # Single-point crossover
            crossover_point = np.random.randint(1, n_points-1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            offspring.extend([child1, child2])
            
        # Mutation
        mutated = []
        for h in offspring:
            if np.random.rand() < 0.2:  # 20% mutation rate
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
            mutated.append(h)
            
        # Elitism: Keep best from previous generation
        elite = [population[0]]
        
        # Update population
        population = elite + mutated[:population_size-1]
        
        generations += 1
        
        # Adapt hyperparameters
        if stagnation_count > max_stagnation:
            if population_size < 40:
                population_size = 40
            else:
                break
                
    # Final evaluation
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        # Resort to best from initial population
        final_fitness = [compute_c5(h, dx) for h in population]
        min_idx = np.argmin(final_fitness)
        best_individual = population[min_idx]
        final_c5 = final_fitness[min_idx]
        
    return (best_individual.tolist(), final_c5, n_points)
