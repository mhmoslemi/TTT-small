import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint using projection
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        diff = target_sum - current_sum
        # Distribute the difference
        if diff > 0:
            # Need to increase values
            h += diff / len(h)
        else:
            # Need to decrease values
            h += diff / len(h)
        # Re-project to ensure bounds
        h = np.clip(h, 0, 1)
        # Apply small adjustment to maintain sum constraint exactly
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Initialize with a step-like function and add some randomness
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    inertia_weight_start = 0.9
    inertia_weight_end = 0.4
    c1 = 1.5
    c2 = 1.5
    velocity_max = 0.1
    
    # Particles and velocities
    particles = np.array([h.copy() for _ in range(n_particles)])
    velocities = np.random.uniform(-0.05, 0.05, (n_particles, n_points))
    pbest_positions = particles.copy()
    pbest_values = np.array([compute_c5(p, dx) for p in particles])
    
    # Global best
    gbest_index = np.argmin(pbest_values)
    gbest_position = pbest_positions[gbest_index].copy()
    gbest_value = pbest_values[gbest_index]
    
    iterations = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iterations < max_iter:
        # Update inertia weight
        inertia_weight = inertia_weight_start - (inertia_weight_start - inertia_weight_end) * (iterations / max_iter)
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (inertia_weight * velocities[i] +
                           c1 * r1 * (pbest_positions[i] - particles[i]) +
                           c2 * r2 * (gbest_position - particles[i]))
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -velocity_max, velocity_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_to_constraints(particles[i], target_sum)
            
            # Evaluate
            current_c5 = compute_c5(particles[i], dx)
            
            # Update personal best
            if current_c5 < pbest_values[i]:
                pbest_values[i] = current_c5
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < gbest_value:
                    gbest_value = current_c5
                    gbest_position = particles[i].copy()
        
        # Early termination if target achieved
        if gbest_value <= 0.3807:
            break
            
        iterations += 1
    
    # Return best solution found
    final_c5 = compute_c5(gbest_position, dx)
    return (gbest_position.tolist(), final_c5, n_points)
