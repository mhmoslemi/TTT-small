import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters  
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback: smooth step-like function
        h = np.linspace(0, 1, n_points//2)
        h = np.concatenate([h, np.ones(n_points - n_points//2)])
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Use adaptive gradient descent with momentum
    max_iter = 100000
    iteration = 0
    
    # Adaptive step size
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store best solution 
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current max lag
        g = np.zeros(n_points)
        k = max_k - (n_points - 1)  # Convert full correlation index to lag
        
        for i in range(n_points):
            # Gradient wrt h[i] when considering overlap at lag k
            if i + k >= 0 and i + k < n_points:
                g[i] = (1 - h[i + k]) * dx
            if i - k >= 0 and i - k < n_points:
                g[i] -= h[i - k] * dx
                
        # Momentum update
        new_h = h - step_size * (g + momentum * prev_grad)
        
        # Project onto feasible set
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Update momentum and step size
        prev_grad = g.copy()
        if iteration % 1000 == 0:
            step_size *= 0.5
            if momentum > 0.5:
                momentum -= 0.05
                
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.05:
            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
