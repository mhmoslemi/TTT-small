import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 5000
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Save current state
        current_c5 = compute_c5(h)
        
        # Store best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Create candidate solution by perturbing h
        h_candidate = h.copy()
        
        # Select two indices for perturbation
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure valid indices (h[i] < 1 and h[j] > 0)
        if h[i] >= 1 or h[j] <= 0:
            # If not valid, get valid indices
            valid_i = np.where(h < 1)[0]
            valid_j = np.where(h > 0)[0]
            if len(valid_i) == 0 or len(valid_j) == 0:
                # Try to maintain current if invalid
                continue
            i = np.random.choice(valid_i)
            j = np.random.choice(valid_j)
            
        # Perform perturbation
        delta = min(1 - h[i], h[j])
        if delta > 0:
            h_candidate[i] += delta
            h_candidate[j] -= delta
            
            # Renormalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute new C5
            new_c5 = compute_c5(h_candidate)
            
            # Metropolis acceptance criterion
            if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_candidate.copy()
        
        # Cooling schedule
        if iteration % 100 == 0:
            T *= alpha
            if T < min_T:
                T = min_T
                
        # Reset temperature if stuck
        if stagnation_count > max_stagnation:
            T = min(T * 1.2, 1.0)
            stagnation_count = 0
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
