import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def construct_comb_h(n_points, n_teeth=7, tooth_width=None, spacing=None, slope=None):
    """Construct h using a comb structure."""
    if tooth_width is None:
        tooth_width = 2.0 / n_teeth
    if spacing is None:
        spacing = (2.0 - n_teeth * tooth_width) / (n_teeth + 1)
    if slope is None:
        slope = tooth_width / 2.0

    h = np.zeros(n_points)
    dx = 2.0 / n_points
    tooth_height = 1.0
    
    # Calculate positions
    tooth_positions = []
    for i in range(n_teeth):
        start_pos = i * (tooth_width + spacing)
        end_pos = start_pos + tooth_width
        tooth_positions.append((start_pos, end_pos))
    
    # Construct each tooth with smooth transitions
    for i, (start_pos, end_pos) in enumerate(tooth_positions):
        start_idx = int(start_pos / dx)
        end_idx = int(end_pos / dx)
        
        # Initialize with tooth height
        h[start_idx:end_idx] = tooth_height
        
        # Apply transition on left edge
        if start_idx > 0:
            left_boundary = start_idx
            left_transition_end = min(int((start_pos + slope) / dx), start_idx + 10)
            for j in range(left_boundary, left_transition_end):
                if j < n_points:
                    t = (j - left_boundary) / (left_transition_end - left_boundary)
                    h[j] *= (1 - t)
        
        # Apply transition on right edge
        if end_idx < n_points:
            right_boundary = end_idx
            right_transition_start = max(int((end_pos - slope) / dx), end_idx - 10)
            for j in range(right_transition_start, right_boundary):
                if j >= 0:
                    t = (right_boundary - j) / (right_boundary - right_transition_start)
                    h[j] *= (1 - t)  # Linear decrease

    # Ensure the h sum constraint
    target_sum = n_points / 2
    h = normalize_h(h, target_sum)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    best_c5 = float('inf')
    best_h = None
    
    # Try different comb configurations
    configs = [
        {'n_teeth': 5, 'tooth_width': 0.4, 'spacing': 0.05, 'slope': 0.1},
        {'n_teeth': 7, 'tooth_width': 0.2857, 'spacing': 0.04, 'slope': 0.07},
        {'n_teeth': 10, 'tooth_width': 0.2, 'spacing': 0.02, 'slope': 0.05},
    ]
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Exhaustive search over config space
    for config in configs:
        config_h = construct_comb_h(n_points, **config)
        c5 = compute_c5(config_h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = config_h.copy()
    
    if best_h is None:
        best_h = np.ones(n_points)
        best_h[n_points//2:] = 0
        best_h = np.clip(best_h, 0, 1)
        best_h = normalize_h(best_h, target_sum)
        best_c5 = compute_c5(best_h, dx)
    
    # Local refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, best_h.copy(), method='SLSQP', 
                          bounds=bounds, constraints=constraint, options={'maxiter': 500})
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = normalize_h(refined_h, target_sum)
            refined_c5 = compute_c5(refined_h, dx)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass

    # Final verification
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
