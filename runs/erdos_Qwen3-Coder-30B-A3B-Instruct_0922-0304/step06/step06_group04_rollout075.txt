import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize multiple candidates
    m = 15  # number of initializations
    candidates = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        candidates.append(base_h.copy())
    
    # Generate diverse initializations
    for i in range(m - len(candidates)):
        # Create a piecewise constant function with random steps
        h = np.zeros(n_points)
        num_steps = np.random.randint(2, 6)
        step_indices = np.sort(np.random.choice(n_points, num_steps, replace=False))
        step_indices = np.append(np.append(0, step_indices), n_points)
        
        for j in range(len(step_indices) - 1):
            start, end = step_indices[j], step_indices[j+1]
            h[start:end] = np.random.rand() * 2  # Random values in [0,2]
        
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        candidates.append(h.copy())
    
    # Additional pattern initialization
    h_pattern = np.zeros(n_points)
    h_pattern[:n_points//3] = 1.0
    h_pattern[n_points//3:2*n_points//3] = 0.5
    h_pattern[2*n_points//3:] = 0.0
    h_pattern = normalize_h(h_pattern, target_sum)
    candidates.append(h_pattern)
    
    best_c5 = float('inf')
    best_individual = None
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Pick a random candidate
        h = candidates[np.random.randint(len(candidates))].copy()
        
        # Add noise
        noise = np.random.normal(0, 0.03, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 150})
        
        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_individual = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    if best_individual is None:
        # Return initial solution
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
        else:
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
            best_individual = normalize_h(best_individual, target_sum)
        best_c5 = compute_c5(best_individual, dx)
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
