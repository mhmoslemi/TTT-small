import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Try to use convex optimization to improve
    # Use CVXPY with MOSEK if available
    solver = 'MOSEK'  # Can be changed to 'SCS' or 'ECOS' if MOSEK not available
    
    # Build the bilinear model
    def build_and_solve(h_init, solver_name):
        h = cp.Variable(n_points)
        z = cp.Variable(2 * n_points - 1)  # For all cross-correlations
        constraints = [h >= 0, h <= 1, cp.sum(h) == target_sum]
        
        # For each shift k, we compute sum(h[i] * (1 - h[i+k]))
        # This involves bilinear terms that we approximate using big-M method
        # Here we use a global min term
        max_z = cp.Variable(1)
        
        # Main objective: minimize max_z
        objective = cp.Minimize(max_z)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver_name, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h.value
                if h_sol is not None:
                    c5_val = compute_c5(h_sol, dx)
                    return h_sol, c5_val
        except Exception:
            pass
            
        return h_init.copy(), compute_c5(h_init, dx)
    
    # Main optimization loop
    keep_trying = True
    iteration = 0
    
    # First attempt with current best
    h_test = best_individual.copy()
    try:
        h_test, c5_test = build_and_solve(h_test, solver)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_individual = h_test.copy()
    except Exception:
        pass
    
    while time.time() - start_time < budget_s - 1 and keep_trying:
        # Try different approaches
        h_mutation = best_individual.copy()
        
        # Small noise mutation
        noise = np.random.normal(0, 0.01, n_points)
        h_mutation = h_mutation + noise
        h_mutation = np.clip(h_mutation, 0, 1)
        h_mutation = h_mutation * (target_sum / np.sum(h_mutation))
        
        # Try convex solve
        try:
            h_test, c5_test = build_and_solve(h_mutation, solver)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_individual = h_test.copy()
        except Exception:
            pass
        
        # Additional perturbations
        if iteration % 3 == 0:
            # Random spike perturbation
            sp_indices = np.random.choice(n_points, size=5, replace=False)
            for i in sp_indices:
                h_mutation[i] = min(1.0, h_mutation[i] + np.random.uniform(0.05, 0.15))
            h_mutation = np.clip(h_mutation, 0, 1)
            h_mutation = h_mutation * (target_sum / np.sum(h_mutation))
            
            try:
                h_test, c5_test = build_and_solve(h_mutation, solver)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_individual = h_test.copy()
            except Exception:
                pass
                
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            keep_trying = False
            
        iteration += 1
    
    # Final evaluation
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
