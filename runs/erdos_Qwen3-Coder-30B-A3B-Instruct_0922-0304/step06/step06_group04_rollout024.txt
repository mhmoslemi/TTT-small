import numpy as np
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    # Using correlate to compute overlaps efficiently
    overlap = np.correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Alternating optimization variables
    g = np.zeros(2 * n_points - 1)  # Overlap values for all shifts
    
    iteration = 0

        h_prev = h.copy()
        
        # Optimize h to minimize max(g[k])
        def optimize_h():

            def sum_constraint(h):
                return np.sum(h) - target_sum

            constraint = {'type': 'eq', 'fun': sum_constraint}
            
            # Objective is to minimize max(g[k])
            def objective(h):
                # Compute overlaps based on current h
                overlap = np.correlate(h, 1 - h, mode="full")
                return np.max(overlap) * dx
            
            # Optimize with SLSQP
            result = minimize(objective, h, method='SLSQP', 
                             bounds=bounds, constraints=constraint, 
                             options={'maxiter': 200, 'ftol': 1e-6, 'gtol': 1e-6})
            
            if result.success:
                return result.x
            else:
                # If optimization fails, return the previous h
                return h_prev
