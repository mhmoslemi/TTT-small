import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to h(x) in [0,2]."""
    x = np.linspace(0, 2, n_points)
    h = coeffs[0] / 2.0 * np.ones(n_points)
    for k in range(1, n_freq + 1):
        h += coeffs[2*k-1] * np.cos(k * np.pi * x) + coeffs[2*k] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h(x) to Fourier coefficients."""
    coeffs = np.zeros(2 * n_freq + 1)
    coeffs[0] = np.mean(h)
    x = np.linspace(0, 2, len(h))
    for k in range(1, n_freq + 1):
        coeffs[2*k-1] = 2 * np.dot(h, np.cos(k * np.pi * x)) / len(h)
        coeffs[2*k] = 2 * np.dot(h, np.sin(k * np.pi * x)) / len(h)
    return coeffs

def objective_fourier(coeffs, n_points, dx, n_freq):
    """Objective function for Fourier coefficients."""
    h = fourier_to_h(coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points / 2)
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15

    # Setup bounds for differential evolution
    bounds = [(-2, 2)] * (2 * n_freq + 1)
    
    best_c5 = float('inf')
    best_h = None

    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        h_current = np.clip(h_current, 0, 1)
        h_current = normalize_h(h_current, target_sum)
    else:
        # Default rectangular pulse
        h_current = np.zeros(n_points)
        h_current[:n_points//2] = 1
        h_current = normalize_h(h_current, target_sum)

    # Convert initial to Fourier
    initial_coeffs = h_to_fourier(h_current, n_freq)
    
    # Differential Evolution optimization
    de_result = differential_evolution(
        objective_fourier,
        bounds,
        args=(n_points, dx, n_freq),
        maxiter=300,
        popsize=75,
        mutation=(0.5, 1.0),
        recombination=0.8,
        seed=seed,
        tol=1e-6,
        callback=lambda x, convergence: None
    )

    if de_result.success:
        h_final = fourier_to_h(de_result.x, n_points, n_freq)
        h_final = np.clip(h_final, 0, 1)
        h_final = normalize_h(h_final, target_sum)

        # Fine-tune with SLSQP
        def fine_obj(h):
            h = np.clip(h, 0, 1)
            return compute_c5(h, dx)
        
        constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
        bounds_h = [(0, 1) for _ in range(n_points)]
        
        fine_result = minimize(fine_obj, h_final, method='SLSQP', bounds=bounds_h, constraints=constraint, options={'maxiter': 100})
        
        if fine_result.success:
            h_final = fine_result.x
            h_final = np.clip(h_final, 0, 1)
            h_final = normalize_h(h_final, target_sum)
            
        c5_final = compute_c5(h_final, dx)
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_final.copy()
    
    # Use the DE result if it's better
    if best_h is None:
        best_h = h_final
        best_c5 = c5_final
    
    # Ensure we return the best result found
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
