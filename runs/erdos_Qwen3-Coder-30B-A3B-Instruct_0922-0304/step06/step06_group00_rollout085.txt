import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint while maintaining bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    # Simple projection: redistribute the difference
    diff = target_sum - current_sum
    if abs(diff) < 1e-10:
        return h
        
    # Distribute the difference uniformly
    h += diff / len(h)
    h = np.clip(h, 0, 1)
    
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize swarm
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.random.rand(n_points)
        base_h = normalize_h(base_h, target_sum)
    
    for i in range(n_particles):
        h = base_h + np.random.normal(0, 0.02, n_points)
        h = project_to_constraints(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Initialize personal bests and global best
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [compute_c5(p, dx) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    w_start = 0.9
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    max_velocity = 0.1
    
    iteration = 0
    max_iter = 2000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        w = w_start - (w_start - w_end) * iteration / max_iter
        
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_best[i] - particle)
            social = c2 * r2 * (global_best - particle)
            velocity = w * velocity + cognitive + social
            
            # Clip velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update position
            particle += velocity
            
            # Apply constraints
            particle = project_to_constraints(particle, target_sum)
            
            # Evaluate
            score = compute_c5(particle, dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = particle.copy()
                
                # Update global best
                if score < global_best_score:
                    global_best_score = score
                    global_best = particle.copy()
                    
            # Store updated particle
            particles[i] = particle
            velocities[i] = velocity
            
        # Early termination
        if global_best_score <= 0.3807:
            break
            
        iteration += 1
    
    return (global_best.tolist(), global_best_score, n_points)
