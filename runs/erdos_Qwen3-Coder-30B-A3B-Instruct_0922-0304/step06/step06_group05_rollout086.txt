import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search for optimal C5
    c_low, c_high = 0.0, 1.0
    tolerance = 1e-4
    
    while time.time() - start_time < budget_s - 2 and c_high - c_low > tolerance:
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find feasible h for this c_mid using CVXPY
        h_var = cp.Variable(n_points)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints
        overlap_max = 0
        for k in range(1, n_points):
            # Compute convolution efficiently
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= c_mid / dx)
            overlap_max = max(overlap_max, overlap)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    c5_sol = compute_c5(h_sol)
                    if c5_sol < best_c5:
                        best_c5 = c5_sol
                        best_h = h_sol.copy()
                    if c5_sol <= 0.3807:
                        break
                    c_high = c_mid
                else:
                    c_low = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

    # Final refinement using gradient descent
    h = best_h.copy()
    max_iter = 10000
    step_size = 0.005
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            # Clamp and renormalize to sum to n_points/2
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
