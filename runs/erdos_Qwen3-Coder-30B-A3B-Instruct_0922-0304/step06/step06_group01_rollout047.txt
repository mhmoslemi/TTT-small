import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cma

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES optimization
    opts = cma.CMAOptions()
    opts.set('boundary_handling', 'bound')
    opts.set('CMA_stopsigma', 1e-10)
    opts.set('popsize', 140)
    opts.set('verbose', -9)
    
    def objective(h_pop):
        h = np.clip(h_pop, 0, 1)
        h = normalize_h(h, target_sum)
        return compute_c5(h, dx)
    
    # Initialize CMA-ES
    mean = h.copy()
    sigma = 0.2
    es = cma.CMAEvolutionStrategy(mean, sigma, opts)
    
    iteration = 0
    max_iter = 500
    
    while not es.stop() and time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Ask for new candidate solutions
        X = es.ask()
        fitness = []
        
        # Evaluate each candidate
        for h_candidate in X:
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = normalize_h(h_candidate, target_sum)
            fit = compute_c5(h_candidate, dx)
            fitness.append(fit)
        
        # Tell the optimizer the fitness values
        es.tell(X, fitness)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = X[best_idx].copy()
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            
        # Early exit condition
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final refinement with SLSQP
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    def objective_refine(h):
        return compute_c5(h, dx)
        
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective_refine, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        refined_c5 = compute_c5(refined_h, dx)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h.copy()
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
