import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search for the minimum C5
    c_low, c_high = 0.0, 1.0
    
    while c_high - c_low > 1e-4 and time.time() - start_time < budget_s - 1:
        c_mid = (c_low + c_high) / 2.0
        
        # Try to find a feasible h for this C
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # For all lags k
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            # Sum over all i of h[i] * (1 - h[(i+k) % n]) <= c_mid / dx
            # Let's compute it approximately by linearizing h[i] * h[(i+k) % n]
            # Approach: Use a simple constraint
            lag_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lag_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(lag_sum <= c_mid / dx)
        
        # Solve the optimization problem
        objective = cp.Minimize(cp.sum(h_vars))  # Dummy objective to satisfy cvxpy
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                # This should be a feasible solution
                h_trial = h_vars.value
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
                # Evaluate the actual C5 for this solution
                trial_c5 = compute_c5(h_trial)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = h_trial.copy()
                
                if trial_c5 <= 0.3807:
                    break
                c_high = c_mid
            else:
                c_low = c_mid
        except:
            c_low = c_mid

    # Refine the result using local search around the best feasible solution
    h = best_h.copy()
    max_iter = 10000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            # Clamp and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
