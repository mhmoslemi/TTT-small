import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

    else:

        h = np.zeros(n_points)
        half_point = n_points // 2

        # Discretize
        h = allowed_values[np.argmin(np.abs(allowed_values[:, None] - h), axis=0)]
