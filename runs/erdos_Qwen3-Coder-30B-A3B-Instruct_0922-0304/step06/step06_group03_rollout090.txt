```python
import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Initialize noise amplitude
    noise_amplitude = 0.02
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Compute correlation to find high overlap regions
        corr = correlate(h, 1 - h, mode="full")
        # Get top 10% of peaks to focus perturbations
        peak_indices = np.argsort(corr)[-max(1, n_points//10):]
        peak_indices = np.unique(peak_indices)
        
        # Create perturbation mask
        mask = np.zeros(n_points)
        # Focus on high overlap regions
        for idx in peak_indices:
            if idx < n_points:
                mask[idx] = 1.0
            elif idx >= n_points:
                # For full correlation, we might have indices beyond n_points
                mirror_idx = 2 * n_points - 2 - idx
                if 0 <= mirror_idx < n_points:
                    mask[mirror_idx] = 1.0
        
        # Perturb
        noise = np.random.normal(0, noise_amplitude, n_points)
        # Apply mask to perturbation
        perturbation = noise * mask
        h = h + perturbation
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Occasionally do more aggressive search or reset
        if iteration % 100 == 0 and noise_amplitude > 0.001:
            noise_amplitude *= 0.95
        elif iteration % 500 == 0:
            h = np.clip(h + np.random.normal(0, 0.03, n_points), 0, 1)
            h = normalize_h(h, target_sum)
        
        # Run optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
            
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 20})
        if result.success:
            h = result.x
        else:
            # Revert to last good if optimization fails
            h = best_h.copy() if iteration % 10 == 0 else h
            
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
```
