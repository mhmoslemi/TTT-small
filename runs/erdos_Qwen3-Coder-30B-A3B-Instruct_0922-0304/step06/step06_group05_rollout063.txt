import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target = 0.3807

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Default initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()

    # Binary search on the value of C5
    low, high = 0.0, 1.0
    tolerance = 1e-4
    solver = 'ECOS'  # fallback

    # Find initial feasible solution
    while high - low > tolerance and (time.time() - start_time) < budget_s - 5:
        mid = (low + high) / 2.0
        # Try to find a feasible solution for this C5
        try:
            # Create variables
            h_vars = cp.Variable(n_points)
            
            # Constraints
            constraints = [
                cp.sum(h_vars) == n_points / 2,
                h_vars >= 0,
                h_vars <= 1
            ]
            
            # Constraint for maximum overlap
            # We compute all lags (full cross-correlation)
            max_k = 0
            for k in range(1, n_points):  # exclude k=0
                lag_sum = 0
                for i in range(n_points):
                    j = (i + k) % n_points
                    lag_sum += h_vars[i] * (1 - h_vars[j])
                constraints.append(lag_sum <= mid / dx)
            
            # Solve the feasibility problem
            prob = cp.Problem(cp.Minimize(0), constraints)
            prob.solve(solver=solver, verbose=False)
            
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                c5_candidate = compute_c5(h_sol)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_sol.copy()
                high = mid
            else:
                low = mid
                
        except Exception as e:
            low = mid

    # Refinement using gradient-based approach from best solution
    if best_c5 >= target:
        h_refine = best_h.copy()
        max_iter = 20000
        iteration = 0
        step_size = 0.005
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            iteration += 1
            
            # Compute current overlap and max
            overlap = correlate(h_refine, 1 - h_refine, mode="full")
            max_overlap = overlap.max() * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h_refine.copy()
                
            if best_c5 <= target:
                break
            
            # Finite difference gradient estim.
            g = np.zeros(n_points)
            eps_val = 1e-4
            
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_minus = h_refine.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Update via gradient descent
            new_h = h_refine - step_size * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            h_refine = new_h

            if iteration % 1000 == 0:
                # Occasionally bounce out of local minima
                pert = np.random.normal(0, 0.002, n_points)
                h_refine = h_refine + pert
                h_refine = np.clip(h_refine, 0, 1)
                h_refine = h_refine * (n_points / 2) / np.sum(h_refine)

        final_c5 = compute_c5(best_h)
        if final_c5 < best_c5:
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
