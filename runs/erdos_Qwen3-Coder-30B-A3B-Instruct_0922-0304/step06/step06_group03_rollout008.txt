import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction or a simple step
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # PSO parameters
    n_particles = 50
    max_iter = 3000
    w = 0.8
    c1 = c2 = 2.0
    
    # Initialize particles and velocities
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = h + np.random.normal(0, 0.02, n_points)
        else:
            particle = np.ones(n_points)
            particle[n_points//2:] = 0
            particle = particle + np.random.normal(0, 0.02, n_points)
        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        particles.append(particle)
        velocities.append(np.zeros(n_points))
        p_best.append(particle.copy())
        p_best_values.append(compute_c5(particle, dx))
    
    # Global best
    g_best_idx = np.argmin(p_best_values)
    g_best = p_best[g_best_idx].copy()
    g_best_value = p_best_values[g_best_idx]
    
    best_h = g_best.copy()
    best_c5 = g_best_value
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute fitness
            fitness = compute_c5(particle, dx)
            
            # Update personal best
            if fitness < p_best_values[i]:
                p_best[i] = particle.copy()
                p_best_values[i] = fitness
                
                # Update global best
                if fitness < g_best_value:
                    g_best = particle.copy()
                    g_best_value = fitness
                    best_h = g_best.copy()
                    best_c5 = g_best_value
                    
                    if best_c5 <= 0.3807:
                        break
            
            # Update velocity and position
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (p_best[i] - particle) + 
                             c2 * r2 * (g_best - particle))
            particles[i] = particle + velocities[i]
            
            # Clip positions
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize sum
            particles[i] = normalize_h(particles[i], target_sum)
        
        # Periodic frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            # Reconstruct worst particles using filtered versions
            worst_indices = np.argsort(p_best_values)[-int(n_particles/4):]
            for idx in worst_indices:
                # Apply high-pass filter in frequency domain
                fft_h = np.fft.fft(p_best[idx])
                # Design high-pass filter
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter
                filter_mask = np.abs(freqs) > 0.3 * np.pi / dx
                filtered_fft = fft_h * filter_mask
                filtered_h = np.real(np.fft.ifft(filtered_fft))
                # Add some noise and normalize
                filtered_h = filtered_h + np.random.normal(0, 0.01, n_points)
                filtered_h = np.clip(filtered_h, 0, 1)
                filtered_h = normalize_h(filtered_h, target_sum)
                particles[idx] = filtered_h
        
        # Periodic gradient refinement
        if iteration % 100 == 0 and iteration > 0:
            top_indices = np.argsort(p_best_values)[:10]
            for idx in top_indices:
                current_h = p_best[idx]
                def objective(h):
                    return compute_c5(h, dx)
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                try:
                    result = minimize(objective, current_h, method='SLSQP', 
                                    bounds=bounds, constraints=constraint, options={'maxiter': 100})
                    if result.success:
                        refined_h = result.x
                        refined_h = np.clip(refined_h, 0, 1)
                        refined_h = normalize_h(refined_h, target_sum)
                        p_best[idx] = refined_h
                        p_best_values[idx] = compute_c5(refined_h, dx)
                except:
                    pass
        
        iteration += 1
        
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
