import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project(h):
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Adjust to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    def exact_gradient(h, k_star):
        # Build gradient analytically for the specific lag k_star
        n = len(h)
        g = np.zeros(n)
        for i in range(n):
            if i + k_star < n:
                g[i] = (1 - h[i + k_star]) * dx
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
        return g

    def line_search(h, g, max_iter=10):
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        
        # Compute current value
        current_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            h_new = h - eta * g
            h_new = project(h_new)
            new_c5 = compute_c5(h_new)
            
            # Accept if sufficient decrease
            if new_c5 <= current_c5 - alpha * eta * np.dot(g, g):
                return h_new, new_c5, eta
                
            # Reduce step size
            eta *= beta
            
        return h, current_c5, eta
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 30000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and identify the maximum
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the max overlap k
        g = exact_gradient(h, max_k)
        
        # Line search for optimal step size
        h_new, new_c5, eta = line_search(h, g)
        
        # Update h with the new values
        h = h_new
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            sigma = 0.01 * np.exp(-iteration / 1000.0)
            pert = np.random.normal(0, sigma, n_points)
            h = h + pert
            h = project(h)
    
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
