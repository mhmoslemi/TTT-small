import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute correlation
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing
    T = 1.0
    alpha = 0.99
    stagnation_count = 0
    max_stagnation = 1000
    
    current_c5 = compute_c5(h)
    
    while time.time() - start_time < budget_s - 1:
        # Perturb h
        i, j = np.random.randint(0, n_points, 2)
        while h[i] >= 1 or h[j] <= 0:
            i, j = np.random.randint(0, n_points, 2)
            
        delta = min(1 - h[i], h[j])
        delta *= np.random.rand() * 0.1  # Small random perturbation
        
        h_new = h.copy()
        h_new[i] += delta
        h_new[j] -= delta
        
        # Ensure bounds and constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = compute_c5(h_new)
        
        # Metropolis acceptance
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
            current_c5 = new_c5
            stagnation_count = 0
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
                # Early exit if target achieved
                if best_c5 <= 0.3807:
                    break
        else:
            stagnation_count += 1
            
        # Adjust temperature
        T *= alpha
        
        # Escape stagnation
        if stagnation_count > max_stagnation:
            T *= 1.2
            stagnation_count = 0
            
    # Final recompute
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
