import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution, minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create a basic step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()

    # Fourier domain optimization
    n_freq = 15  # Number of frequency components
    
    def h_from_fourier(coeffs):
        # coeffs: [a0, a1, a2, ..., an, b1, b2, ..., bn]
        a0 = coeffs[0]
        a_coeffs = coeffs[1:n_freq+1]
        b_coeffs = coeffs[n_freq+1:]
        
        x_vals = np.linspace(0, 2, n_points)
        h_vals = a0/2.0 + np.zeros_like(x_vals)
        
        for k in range(1, n_freq+1):
            h_vals += a_coeffs[k-1] * np.cos(k * np.pi * x_vals) + \
                      b_coeffs[k-1] * np.sin(k * np.pi * x_vals)
        
        return h_vals
    
    def objective_fourier(coeffs):
        h_vals = h_from_fourier(coeffs)
        # Clip and normalize
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        return compute_c5(h_vals, dx)
    
    # Bounds for Fourier coefficients
    bounds = [(0, 1)] + [(-1, 1)] * (2 * n_freq)  # a0, a1..an, b1..bn
    
    # Start with a good initial guess
    if 'initial_h_values' in globals():
        # Transform initial_h_values to Fourier coefficients
        init_h = np.array(initial_h_values)
        init_h = np.clip(init_h, 0, 1)
        init_h = init_h * (target_sum / np.sum(init_h))
        # Use DCT to estimate coefficients
        init_coeffs = [np.mean(init_h)]
        for k in range(1, n_freq+1):
            init_coeffs.append(np.mean(init_h * np.cos(k * np.pi * np.linspace(0, 2, n_points) / 2)))
            init_coeffs.append(np.mean(init_h * np.sin(k * np.pi * np.linspace(0, 2, n_points) / 2)))
        init_coeffs = np.array(init_coeffs)
        x0 = init_coeffs
    else:
        # Initialize with a simple pulse
        x0 = np.zeros(2*n_freq+1)
        x0[0] = 1.0  # DC component
    
    # Differential evolution for global search
    de_result = differential_evolution(
        objective_fourier, 
        bounds, 
        maxiter=300, 
        popsize=70, 
        tol=1e-6,
        seed=seed,
        disp=False
    )
    
    # Transform back to h
    h_final = h_from_fourier(de_result.x)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    # Fine-tune with local optimization
    def objective_local(h_vec):
        return compute_c5(h_vec, dx)
    
    # Create bounds for SLSQP
    bounds_slsqp = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(
        objective_local, 
        h_final.copy(), 
        method='SLSQP', 
        bounds=bounds_slsqp, 
        constraints=constraint, 
        options={'maxiter': 100}
    )
    
    if result.success:
        h_final = result.x
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (target_sum / np.sum(h_final))
    
    final_c5 = compute_c5(h_final, dx)
    return (h_final.tolist(), final_c5, n_points)
