import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate=0.05, target_sum=35.0):
    """Mutate an individual with constraint maintenance."""
    h_mut = h.copy()
    for i in range(len(h_mut)):
        if np.random.random() < mutation_rate:
            h_mut[i] += np.random.normal(0, 0.05)
    h_mut = np.clip(h_mut, 0, 1)
    h_mut = normalize_h(h_mut, target_sum)
    return h_mut

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_pop = [np.array(initial_h_values)]
    else:
        initial_pop = [np.ones(n_points)]
        initial_pop[0][n_points//2:] = 0
        initial_pop[0] = np.clip(initial_pop[0], 0, 1)
        
    # Generate initial population
    population = []
    for i in range(30):
        if i < len(initial_pop):
            h = initial_pop[i].copy()
        else:
            h = np.ones(n_points)
            h[n_points//2:] = 0
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            h += np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = None
    best_iter = 0
    
    iteration = 0
    max_iter = 100000
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate population
        fitness_list = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness_list.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_list)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness_list[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            best_iter = iteration
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Selection (top 50%)
        selected = sorted_population[:len(sorted_population)//2]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < 30:
            # Select two parents
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate_individual(child1, target_sum=target_sum)
            child2 = mutate_individual(child2, target_sum=target_sum)
            
            new_population.append(child1)
            if len(new_population) < 30:
                new_population.append(child2)
                
        # Local SLSQP refinement on top 10%
        top_candidates = new_population[:3]
        refined_candidates = []
        for candidate in top_candidates:
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            def objective(h):
                return compute_c5(h, dx)
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            try:
                result = minimize(objective, candidate, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 100})
                if result.success:
                    refined = np.clip(result.x, 0, 1)
                    refined = normalize_h(refined, target_sum)
                    refined_candidates.append(refined)
                else:
                    refined_candidates.append(candidate)
            except:
                refined_candidates.append(candidate)
                
        # Replace worst solutions with refined ones
        new_population = top_candidates + new_population[3:]
        new_population = new_population[:30]
        
        # Ensure population size
        if len(new_population) < 30:
            # Fill with new random individuals
            for _ in range(30 - len(new_population)):
                h = np.ones(n_points)
                h[n_points//2:] = 0
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                h += np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population.append(h)
                
        population = new_population.copy()
        iteration += 1
    
    if best_h is None:
        # Fallback to last evaluated
        fitness_list = [compute_c5(h, dx) for h in population]
        best_idx = np.argmin(fitness_list)
        best_h = population[best_idx]
        best_c5 = fitness_list[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
