```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate, mutation_strength):
    """Mutate an individual."""
    mask = np.random.rand(len(h)) < mutation_rate
    h += np.random.normal(0, mutation_strength, len(h)) * mask
    return np.clip(h, 0, 1)

def refine_individual(h, target_sum, dx):
    """Refine an individual using gradient descent."""
    def objective(h_vec):
        return evaluate_h(h_vec, dx)
    
    from scipy.optimize import minimize
    bounds = [(0, 1) for _ in range(len(h))]
    constraints = {'type': 'eq', 'fun': lambda x: np.sum(x) - target_sum}
    
    result = minimize(objective, h.copy(), method='SLSQP', 
                     bounds=bounds, constraints=constraints, 
                     options={'maxiter': 50})
    if result.success:
        h = result.x
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    population_size = 100
    max_generations = 60
    mutation_rate_start = 0.2
    mutation_rate_decay = 0.95
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
    
    # Add some diversity to the initial population
    for _ in range(population_size):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time >= budget_s - 1:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        sorted_indices = np.argsort(fitnesses)
        # Elitism: keep best 10%
        elite_count = population_size // 10
        for i in range(elite_count):
            new_population.append(population[sorted_indices[i]])
        
        # Generate offspring
        mutation_rate = mutation_rate_start * (mutation_rate_decay ** generation)
        for _ in range(population_size - elite_count):
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate, 0.05)
            child2 = mutate(child2, mutation_rate, 0.05)
            
            # Local refinement
            child1 = refine_individual(child1, target_sum, dx)
            child2 = refine_individual(child2, target_sum, dx)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final refinement of best solution
    if best_h is not None:
        best_h = refine_individual(best_h, target_sum, dx)
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    
    # Fallback solution
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    final_c5 = evaluate_h(h, dx)
    return (h.tolist(), final_c5, n_points)
```
