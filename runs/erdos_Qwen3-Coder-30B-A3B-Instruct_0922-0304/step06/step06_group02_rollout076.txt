import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Global search with alternating strategies
    iteration = 0
    max_iter = 5000
    ls_iter = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Use a hybrid approach: alternating between PSO-style update and local search
        if ls_iter % 3 == 0:
            # Local search using L-BFGS-B
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            # Slight perturbation to avoid local minima
            noise_magnitude = 0.03 * (1 - (time.time() - start_time) / budget_s)
            h_noisy = h + np.random.normal(0, noise_magnitude, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            # Minimize using L-BFGS-B
            result = minimize(objective, h_noisy, method='L-BFGS-B', bounds=bounds, constraints=constraint, options={'maxiter': 200})
            
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_h(h_candidate, target_sum)
                
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5 * 0.9999:  # Accept improvement
                    h = h_candidate.copy()
                    
        else:
            # Structured perturbation to escape local optima
            perturbation_strength = 0.02 * (1 - (time.time() - start_time) / budget_s)
            h += np.random.normal(0, perturbation_strength, len(h))
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
        ls_iter += 1
        iteration += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
