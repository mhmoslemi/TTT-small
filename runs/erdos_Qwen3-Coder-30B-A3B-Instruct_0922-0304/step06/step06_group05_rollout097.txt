import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        """Project h to feasible region: [0,1] and sum h = n/2"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def exact_gradient(h, k_star):
        """Compute exact gradient for the k_star lag"""
        g = np.zeros(n_points)
        for i in range(n_points):
            # For overlap at lag k_star, h[i] contributes to term h[(i+k_star)%n](1-h[i])
            # and h[(i-k_star)%n] contributes to term h[i](1-h[(i-k_star)%n])
            # But these are from different terms, so more care is needed
            # Let us recompute overlap gradient directly
            
            # The contribution is in the element (i+k_star)%n of the convolution h * (1-h)
            # the gradient is 
            if (i + k_star) < n_points:
                g[i] = (1 - h[(i + k_star) % n_points]) * dx
            else:
                # if i + k_star >= n_points, use the periodic boundary
                g[i] = 0
                
        return g
    
    def line_search(h, grad, max_iter=10):
        """Find optimal step size using backtracking line search"""
        alpha = 0.1
        beta = 0.5
        eta = 0.01
        current_c5 = compute_c5(h)
        
        for _ in range(max_iter):
            new_h = h - eta * grad
            new_h = project_h(new_h)
            new_c5 = compute_c5(new_h)
            
            # Armijo condition - ensure sufficient decrease
            if new_c5 <= current_c5 - alpha * eta * np.dot(grad, grad):
                return eta, new_h
            eta *= beta
            
        return eta, h
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop  
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute exact gradient for the current maximum lag  
        g = exact_gradient(h, max_k)
        
        # Line search for step size
        eta, new_h = line_search(h, g)
        
        # Update h
        h = new_h
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0:
            # Add noise with decaying variance
            sigma = 0.01 * np.exp(-iteration / 1000)
            noise = np.random.normal(0, sigma, n_points)
            h = h + noise
            h = project_h(h)
            
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
