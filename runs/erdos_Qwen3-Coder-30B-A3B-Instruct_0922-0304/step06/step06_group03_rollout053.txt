import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate multiple initial guesses."""
    guesses = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # Comb pattern
    h = np.zeros(n_points)
    step = max(1, n_points // 5)
    h[::step] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Random pattern
    h = np.random.rand(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Alternating pattern
    h = np.zeros(n_points)
    h[::2] = 0.8
    h[1::2] = 0.2
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # Low-discrepancy pattern (simple approach)
    h = np.array([i * 0.618033988749895 % 1 for i in range(n_points)])
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate multiple initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    # Optimization parameters
    max_iter = 15000
    perturb_freq = 10
    large_perturb_freq = 50
    perturb_std = 0.02
    perturb_decay = 0.95
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try each initial guess
        for i, h in enumerate(initial_guesses):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5
            c5 = compute_c5(h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Local optimization using SLSQP
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Use SLSQP with slightly perturbed h to escape local minima
            if iter_count % perturb_freq == 0:
                h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
            # Occasionally do more aggressive search
            if iter_count % large_perturb_freq == 0:
                h = np.clip(h + np.random.normal(0, perturb_std * 2, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
            # Run optimization step
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
                if result.success:
                    h = result.x
                else:
                    # If optimization failed, perturb more aggressively
                    h = np.clip(h + np.random.normal(0, perturb_std * 2, n_points), 0, 1)
                    h = normalize_h(h, target_sum)
            except:
                # Fallback to perturbation
                h = np.clip(h + np.random.normal(0, perturb_std, n_points), 0, 1)
                h = normalize_h(h, target_sum)
                
            # Update initial guess for next iteration
            initial_guesses[i] = h
            
        # Reduce perturbation over time
        if iter_count % 100 == 0:
            perturb_std *= perturb_decay
            
        if best_c5 <= 0.3807:
            break

    # Final validation
    if best_h is None:
        # Fallback to initial_h_values or basic construction
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
        else:
            best_h = np.ones(n_points)
            best_h[n_points//2:] = 0
            best_h = np.clip(best_h, 0, 1)
            best_h = normalize_h(best_h, target_sum)
            
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
