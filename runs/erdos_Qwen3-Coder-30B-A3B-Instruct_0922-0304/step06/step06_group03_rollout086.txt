import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with given spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Place teeth at regular intervals
    tooth_positions = np.arange(0, 2, spacing * dx)
    tooth_indices = np.floor(tooth_positions / dx).astype(int)
    tooth_indices = tooth_indices[(tooth_indices >= 0) & (tooth_indices < n_points)]
    
    h[tooth_indices] = tooth_height
    
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbations to pattern."""
    pert = np.random.normal(0, sigma, len(h))
    h_pert = h + pert
    return np.clip(h_pert, 0, 1)

def optimize_pattern(h, target_sum, dx):
    """Optimize a pattern using SLSQP with sum constraint."""
    def objective(h):
        return compute_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
        
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
    if result.success:
        return result.x
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Default initialization as in the previous method
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    spacing_list = [2, 3, 4, 5, 6]
    
    iteration = 0
    max_iter = 10000
    time_budget_left = budget_s - 1
    
    while time.time() - start_time < time_budget_left and iteration < max_iter:
        # Regular comb construction search
        for spacing in spacing_list:
            h_comb = create_comb_pattern(n_points, spacing)
            h_comb = normalize_h(h_comb, target_sum)
            
            # Try multiple perturbed versions
            for _ in range(5):
                h_pert = perturb_pattern(h_comb, 0.02)
                h_opt = optimize_pattern(h_pert, target_sum, dx)
                c5 = compute_c5(h_opt, dx)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    
                if best_c5 <= 0.3807:
                    break
            if best_c5 <= 0.3807:
                break
                
        if best_c5 <= 0.3807:
            break
            
        # Try a few random patterns to escape local optima
        for _ in range(3):
            h_rand = np.random.rand(n_points)
            h_rand = normalize_h(h_rand, target_sum)
            h_opt = optimize_pattern(h_rand, target_sum, dx)
            c5 = compute_c5(h_opt, dx)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
                
        if best_c5 <= 0.3807:
            break
            
        # Local refinement on best
        h = optimize_pattern(best_h, target_sum, dx)
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
    
    # Final refinement with higher precision
    best_h = optimize_pattern(best_h, target_sum, dx)
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
