import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        h_base = np.clip(h_base, 0, 1)
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1.0
    
    # Normalize to target sum
    current_sum = np.sum(h_base)
    if current_sum > 0:
        h_base = h_base * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    # Bilinear optimization approach
    solver = cp.MOSEK
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 20:
        h = h_base.copy()
        
        # Perturb slightly
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Set up bilinear program
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == target_sum
        ]
        
        # Compute overlap terms
        max_overlap = 0
        for k in range(2 * n_points - 1):
            if k < n_points:
                # h[i] * (1 - h[i + k]) for valid indices
                term = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            else:
                # Reverse shift case
                term = cp.sum(cp.multiply(h_var[k-n_points+1:], 1 - h_var[:n_points-(k-n_points+1)]))
            
            max_overlap = cp.maximum(max_overlap, term)
        
        # Objective
        objective = cp.Minimize(max_overlap * dx)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_opt = h_var.value
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (target_sum / np.sum(h_opt))
                c5 = compute_c5(h_opt, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
        except:
            pass
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
