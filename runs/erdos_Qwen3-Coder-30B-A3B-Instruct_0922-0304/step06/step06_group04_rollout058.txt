import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    stagnant = 0
    max_stagnant = 20
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Update best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnant = 0
        else:
            stagnant += 1
        
        # Stop if target reached
        if best_c5 <= 0.3807:
            break
            
        if stagnant > max_stagnant:
            break
            
        # Selection (top 50%)
        selected_indices = np.argsort(fitness)[:population_size // 2]
        selected = [population[i] for i in selected_indices]
        
        # Crossover
        offspring = []
        for _ in range(population_size - len(selected)):
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            offspring.extend([child1, child2])
        
        # Keep only population_size offspring
        offspring = offspring[:population_size - len(selected)]
        
        # Mutation
        mutation_rate = 0.2
        for i in range(len(offspring)):
            if np.random.random() < mutation_rate:
                offspring[i] += np.random.normal(0, 0.05, n_points)
                offspring[i] = np.clip(offspring[i], 0, 1)
                offspring[i] = normalize_h(offspring[i], target_sum)
        
        # Elitism: add best from previous generation
        offspring.append(best_individual.copy())
        
        # Update population
        population = selected + offspring[:population_size - len(selected)]
        
        generation += 1
        
        # Increase population size if stagnated
        if stagnant > 10 and population_size < 40:
            population_size = 40
    
    # Final validation
    if best_individual is None:
        if 'initial_h_values' in globals():
            best_individual = np.array(initial_h_values)
            best_individual = np.clip(best_individual, 0, 1)
            best_individual = normalize_h(best_individual, target_sum)
        else:
            best_individual = np.zeros(n_points)
            best_individual[:n_points//2] = 1.0
            best_individual = normalize_h(best_individual, target_sum)
    
    final_c5 = evaluate_individual(best_individual, dx)
    
    return (best_individual.tolist(), final_c5, n_points)
