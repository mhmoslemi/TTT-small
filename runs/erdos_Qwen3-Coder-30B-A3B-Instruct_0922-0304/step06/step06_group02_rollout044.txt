import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, crossover_rate=0.7):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05, noise_std=0.05):
    """Apply mutation to a candidate solution."""
    h_mut = h.copy()
    for i in range(len(h_mut)):
        if np.random.rand() < mutation_rate:
            h_mut[i] += np.random.normal(0, noise_std)
    return np.clip(h_mut, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1
        initial_h = normalize_h(initial_h, target_sum)
    
    population_size = 30
    population = []
    for _ in range(population_size):
        h = initial_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    generation = 0
    max_generations = 1000
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
        
        # Find best solution in this generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Early exit
        if best_c5 <= 0.3807:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        selected = [population[i] for i in sorted_indices[:population_size//2]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover and mutate
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            new_population.extend([child1, child2])
        
        # Keep only population_size
        population = new_population[:population_size]
        
        # Local refinement with SLSQP for top 10%
        top_k = max(1, population_size // 10)
        for i in sorted_indices[:top_k]:
            h = population[i]
            
            def obj_func(h_local):
                return compute_c5(h_local, dx)
            
            def constraint_sum(h_local):
                return np.sum(h_local) - target_sum
            
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Add small noise to avoid local minima
            h_noisy = h + np.random.normal(0, 0.01, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_h(h_noisy, target_sum)
            
            result = minimize(obj_func, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            
            if result.success:
                refined_h = np.clip(result.x, 0, 1)
                refined_h = normalize_h(refined_h, target_sum)
                refined_c5 = compute_c5(refined_h, dx)
                if refined_c5 < fitness[i]:
                    population[i] = refined_h
                    fitness[i] = refined_c5
        
        generation += 1
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
