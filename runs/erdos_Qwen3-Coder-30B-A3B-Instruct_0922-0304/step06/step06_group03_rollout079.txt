import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def generate_initial_guesses(n_points, target_sum):
    """Generate diverse initial guesses"""
    guesses = []
    
    # 1. Initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        guesses.append(h.copy())
    
    # 2. Comb pattern (alternating high/low)
    h = np.zeros(n_points)
    for i in range(0, n_points, 2):
        h[i] = 1.0
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 3. Random with sum constraint
    h = np.random.random(n_points)
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 4. Alternating pattern with some randomness
    h = np.ones(n_points)
    h[::2] = 0.5
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    # 5. Step pattern with randomization
    h = np.zeros(n_points)
    h[:n_points//3] = 1.5
    h[n_points//3:2*n_points//3] = 0.5
    h = normalize_h(h, target_sum)
    guesses.append(h.copy())
    
    return guesses

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Generate diverse initial guesses
    initial_guesses = generate_initial_guesses(n_points, target_sum)
    
    best_c5 = float('inf')
    best_h = None
    
    iteration = 0
    max_iter = 50000
    
    for init_guess in initial_guesses:
        h = init_guess.copy()
        current_h = h.copy()
        
        # Optimization loop
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            c5 = compute_c5(current_h, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = current_h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Optimization step
            def objective(h):
                return compute_c5(h, dx)
                
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            # Small random perturbation to escape local minima
            perturbation = np.random.normal(0, 0.02, n_points)
            new_h = current_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = normalize_h(new_h, target_sum)
            
            # Try to optimize from the perturbed state
            try:
                result = minimize(objective, new_h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 50, 'ftol': 1e-8})
                if result.success:
                    current_h = result.x
                else:
                    current_h = new_h
            except:
                current_h = new_h
                
            # Occasionally do more aggressive search
            if iteration % 50 == 0:
                perturbation = np.random.normal(0, 0.05, n_points)
                current_h = current_h + perturbation
                current_h = np.clip(current_h, 0, 1)
                current_h = normalize_h(current_h, target_sum)
                
            iteration += 1
    
    # Final validation of best solution
    if best_h is not None:
        final_c5 = compute_c5(best_h, dx)
    else:
        # Fallback to initial guess if nothing better found
        h = initial_guesses[0].copy()
        final_c5 = compute_c5(h, dx)
        best_h = h
    
    return (best_h.tolist(), final_c5, n_points)
