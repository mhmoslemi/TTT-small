import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

    start_time = time.time()

    dx = 2.0 / n_points
    target_sum = n_points / 2

    else:

        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # PSO-like approach with structured perturbations
    max_iter = 5000
    pop_size = 25
    max_pso_iter = 100
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
        else:

        particle = np.clip(particle, 0, 1)
        particle = normalize_h(particle, target_sum)
        population.append(particle)
    
    # PSO parameters
    w_start = 0.7
    w_end = 0.4
    c1 = 1.5
    c2 = 1.5
    
    # Global best
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    for iter_pso in range(max_pso_iter):
        # Check time
        if time.time() - start_time > budget_s - 1:

        # Evaluate population
        fitness = []

            c5 = compute_c5(particle, dx)
            fitness.append(c5)

                best_c5 = c5

            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
        
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Update velocities and positions
        for i, particle in enumerate(population):
            # Calculate w (inertia weight)
            w = w_start - (w_start - w_end) * iter_pso / max_pso_iter
            
            # Random components
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Best personal and global
            if i == gbest_idx:
                pbest = particle
            else:
                pbest = population[i]
                
            # Update velocity (simplified)
            v = w * (particle - pbest) + c1 * r1 * (pbest - particle) + c2 * r2 * (population[gbest_idx] - particle)
            
            # Update position
            particle = particle + v
            particle = np.clip(particle, 0, 1)
            particle = normalize_h(particle, target_sum)
            
            population[i] = particle

                    population[i] = np.clip(population[i] + np.random.normal(0, 0.02, n_points), 0, 1)
                    population[i] = normalize_h(population[i], target_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
