import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def fourier_to_h(coeffs, n_points, n_freq):
    """Convert Fourier coefficients to step function h."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for i in range(n_freq):
        k = i + 1
        h += coeffs[2*i] * np.cos(k * np.pi * x) + coeffs[2*i+1] * np.sin(k * np.pi * x)
    h += coeffs[-1]  # DC component
    return h

def h_to_fourier(h, n_freq):
    """Convert h to Fourier coefficients."""
    # Simple DCT-based approximation
    coeffs = []
    for i in range(n_freq):
        k = i + 1
        a = 2 * np.sum(h * np.cos(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        b = 2 * np.sum(h * np.sin(k * np.pi * np.linspace(0, 2, len(h)))) / len(h)
        coeffs.extend([a, b])
    dc = np.mean(h)
    coeffs.append(dc)
    return coeffs

def optimize_fourier_better(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15
    
    # Initialize from existing solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_coeffs = h_to_fourier(h, n_freq)
    else:
        # Simple initialization
        h = np.ones(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initial_coeffs = h_to_fourier(h, n_freq)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(swarm_size):
        coeffs = initial_coeffs.copy()
        # Add small noise
        coeffs += np.random.normal(0, 0.1, len(coeffs))
        particles.append(coeffs)
        velocities.append(np.zeros_like(coeffs))
        pbest.append(coeffs.copy())
        pbest_vals.append(float('inf'))
    
    gbest = initial_coeffs.copy()
    gbest_val = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            coeffs = particles[i]
            # Convert coeffs to h
            h = fourier_to_h(coeffs, n_points, n_freq)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Evaluate C5
            c5 = compute_c5(h, dx)
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = coeffs.copy()
                pbest_vals[i] = c5
                
            # Update global best
            if c5 < gbest_val:
                gbest = coeffs.copy()
                gbest_val = c5
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2), np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            particles[i] = particles[i] + velocities[i]
            
    # Final refinement with SLSQP
    best_coeffs = gbest
    h = fourier_to_h(best_coeffs, n_points, n_freq)
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    
    # SLSQP refinement
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
    
    if result.success:
        h = np.clip(result.x, 0, 1)
        h = normalize_h(h, target_sum)
        final_c5 = compute_c5(h, dx)
    else:
        final_c5 = compute_c5(h, dx)
        
    return h.tolist(), final_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    return optimize_fourier_better(seed, budget_s)
