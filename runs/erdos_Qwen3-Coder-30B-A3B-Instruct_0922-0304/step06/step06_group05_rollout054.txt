import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def project_h(h):
        """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to match target sum
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Simple projection: adjust entries away from boundaries
            diff = n_points/2 - current_sum
            if diff > 0:
                # Need to increase sum
                h = h + diff / n_points
                h = np.clip(h, 0, 1)
            else:
                # Need to decrease sum
                h = h + diff / n_points
                h = np.clip(h, 0, 1)
        return h
    
    def compute_gradient(h, k_star):
        """Compute gradient of overlap w.r.t. h for the max lag k_star"""
        n = len(h)
        g = np.zeros(n)
        # For lag k_star, the gradient at position i is:
        # if i + k_star < n: (1 - h[i + k_star]) * dx
        # if i - k_star >= 0: -h[i - k_star] * dx
        for i in range(n):
            if i + k_star < n:
                g[i] += (1 - h[i + k_star]) * dx
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
        return g
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with analytic gradients
    max_iter = 10000
    iteration = 0
    step_size = 0.01
    step_decay = 0.5
    step_decay_freq = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags, identify k_star
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient analytically for max lag
        g = compute_gradient(h, max_k)
        
        # Update h with gradient descent
        new_h = h - step_size * g
        new_h = project_h(new_h)
        h = new_h
        
        # Adjust step size periodically
        if iteration % step_decay_freq == 0:
            step_size *= step_decay
        
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
