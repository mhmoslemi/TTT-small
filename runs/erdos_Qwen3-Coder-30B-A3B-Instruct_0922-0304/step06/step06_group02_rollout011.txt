import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (target_sum / h_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Define CVXPy variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Build constraints
    constraints = [cp.sum(h_var) == target_sum]
    constraints += [0 <= h_var[i] for i in range(n_points)]
    constraints += [h_var[i] <= 1 for i in range(n_points)]
    
    # Compute all lags for constraint building
    max_lag = n_points - 1
    # We'll compute overlaps manually using convolution
    # Without circularity, we build "full" correlation by hand
    # But for each k in [-max_lag, max_lag], we will have valid overlaps
    
    # Make a big loop for all lags:
    # Compute sum_{i=0}^{n_points-1 - |k|} h[i](1 - h[i+k]) dx <= C
    
    # Short loop for constructing lag-specific constraints
    for k in range(-max_lag, max_lag + 1):
        # k is the shift in correlation
        # valid i such that both i and i+k are in [0, n_points-1]
        start_idx = max(0, -k)  # Ensures i >= 0
        end_idx = min(n_points, n_points - k)  # Ensures i+k < n_points

        term = 0
        for i in range(start_idx, end_idx):
            term += h_var[i] * (1 - h_var[i + k])
        constraints.append(term * dx <= C)
        
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    # Solver options
    solver = cp.GUROBI  # or use MOSEK or ECOS for open-source
    prob.solve(solver=solver, verbose=False, max_iter=10000, time_limit=budget_s - 1)
    
    # Extract result if successful
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = h_var.value
        c5_bound = C.value
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = np.sum(h_opt)
        if h_opt_sum > 0:
            h_opt = h_opt * (target_sum / h_opt_sum)
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h_opt.copy()

    return (best_h.tolist(), best_c5, n_points)
