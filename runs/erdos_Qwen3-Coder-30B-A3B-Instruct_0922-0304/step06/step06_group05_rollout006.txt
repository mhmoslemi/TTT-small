import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: smooth sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        return h

    def compute_gradient(h, k_star):
        # Compute exact gradient at lag k_star
        g = np.zeros(n_points)
        for i in range(n_points):
            if i + k_star < n_points:
                g[i] = (1 - h[i + k_star]) * dx
            if i - k_star >= 0:
                g[i] -= h[i - k_star] * dx
        return g

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimizer parameters
    eta = 0.01
    mu = 0.9
    g_prev = np.zeros(n_points)
    
    max_iter = 100000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap for all lags
        overlap = correlate(h, 1 - h, mode="full")
        k_star = np.argmax(overlap)
        max_overlap = overlap[k_star] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient
        g = compute_gradient(h, k_star)
        
        # Update with momentum
        v = mu * g_prev + g
        new_h = h - eta * v
        
        # Project onto feasible set
        new_h = project_h(new_h)
        
        # Update for next iteration
        h = new_h
        g_prev = g
        
        # Adaptive step size and momentum
        if iteration % 500 == 0:
            eta *= 0.9
            mu = max(0.5, mu * 0.99)
            
        # Occasionally perturb to escape local minima
        if iteration % 1000 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.01, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
