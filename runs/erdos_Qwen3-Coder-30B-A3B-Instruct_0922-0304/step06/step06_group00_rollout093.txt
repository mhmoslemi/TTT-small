import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate, mutation_strength):
    """Mutate the individual."""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, mutation_strength, len(h))
    h = h + mask * noise
    return np.clip(h, 0, 1)

def local_optimize(h, target_sum, dx):
    """Perform local optimization with constraints."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
        
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    result = minimize(objective, h, method='SLSQP', 
                     bounds=bounds, constraints=constraint, 
                     options={'maxiter': 50})
    
    if result.success:
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = normalize_h(h_opt, target_sum)
        return h_opt
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    else:
        h_base = np.zeros(n_points)
        h_base[:n_points//2] = 1
        h_base += np.random.normal(0, 0.02, n_points)
        h_base = np.clip(h_base, 0, 1)
        h_base = normalize_h(h_base, target_sum)
        population.append(h_base.copy())
    
    # Fill the rest randomly
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    best_c5 = float('inf')
    best_h = None
    
    # Genetic Algorithm parameters
    mutation_rate = 0.2
    mutation_decay = 0.99
    crossover_rate = 0.8
    max_gens = 60  # Each generation is ~10 seconds budget
    
    for gen in range(max_gens):
        if time.time() - start_time >= budget_s - 2:
            break
            
        # Evaluate fitness
        fitnesses = [compute_c5(h, dx) for h in population]
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
        
        if best_c5 <= 0.3807:
            break
            
        # Create new population
        new_population = []
        
        # Keep the best individual
        new_population.append(population[np.argmin(fitnesses)].copy())
        
        # Fill rest with offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            child1 = mutate(child1, mutation_rate, 0.02)
            child2 = mutate(child2, mutation_rate, 0.02)
            
            # Apply local optimization
            child1 = local_optimize(child1, target_sum, dx)
            child2 = local_optimize(child2, target_sum, dx)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:population_size]
        mutation_rate *= mutation_decay
    
    # Final local optimization of best solution
    if best_h is not None:
        best_h = local_optimize(best_h, target_sum, dx)
        final_c5 = compute_c5(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    
    return (population[0].tolist(), compute_c5(population[0], dx), n_points)
