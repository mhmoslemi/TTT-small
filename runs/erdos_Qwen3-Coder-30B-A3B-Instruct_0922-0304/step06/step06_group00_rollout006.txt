import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        # Create initial function with structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Adaptive learning rate and parameters
    lr = 0.01
    mu = 1.0
    decay_factor = 0.99
    min_lr = 0.0001
    
    iteration = 0
    patience = 0
    best_c5_since = best_c5
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient using finite differences
        epsilon = 1e-5
        grad = np.zeros_like(h)
        
        # For each point compute finite difference
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (target_sum / np.sum(h_minus))
            
            # Evaluate C5 for both
            c5_plus = compute_c5(h_plus, dx)
            c5_minus = compute_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Add barrier term gradient: -1/h[i] - 1/(1-h[i]) - 1/(sum-h.sum())
        barrier_grad = np.zeros_like(h)
        sum_h = np.sum(h)
        barrier_grad = -1.0 / h - 1.0 / (1.0 - h) - 1.0 / (target_sum - sum_h)
        grad += mu * barrier_grad
        
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (target_sum / np.sum(h_new))
        
        # Evaluate new point
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            patience = 0
            if best_c5 <= 0.3807:
                break
        else:
            patience += 1
            
        # Update parameters
        h = h_new
        lr = max(min_lr, lr * decay_factor)
        
        if patience > 10:
            lr = max(min_lr, lr * 0.5)
            patience = 0
            
        if iteration % 50 == 0:
            mu *= 0.5
            
        iteration += 1
        if iteration > 10000:
            break
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
