import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 1.0
    alpha = 0.99
    min_T = 1e-5
    stagnation_count = 0
    max_stagnation = 1000
    
    while time.time() - start_time < budget_s - 1:
        # Perturb h
        h_candidate = h.copy()
        i, j = np.random.randint(0, n_points, 2)
        
        # Ensure valid indices for perturbation
        if h[i] >= 1 or h[j] <= 0:
            continue
            
        delta = min(1 - h[i], h[j])
        delta *= np.random.rand()  # Random fraction
        
        h_candidate[i] += delta
        h_candidate[j] -= delta
        
        # Renormalize to satisfy sum(h) = n_points/2
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Ensure bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Compute C5 for both
        c5_current = compute_c5(h)
        c5_candidate = compute_c5(h_candidate)
        
        # Metropolis acceptance
        if c5_candidate < c5_current or np.random.rand() < np.exp(-(c5_candidate - c5_current) / T):
            h = h_candidate
            c5_current = c5_candidate
            
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Annealing schedule
        T = max(T * alpha, min_T)
        
        # Restart if stuck
        if stagnation_count > max_stagnation:
            T = min(T * 1.2, 1.0)
            stagnation_count = 0
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
