import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Renormalize to sum to n_points/2
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization with analytical gradients
    max_iter = 10000
    iteration = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap, find max lag
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Analytical gradient computation
        g = np.zeros(n_points)
        # Create a shifted version of h for the gradient computation
        for i in range(n_points):
            # Contribution from h[i] to overlap at lag max_k
            if i + max_k < n_points:
                g[i] = (1 - h[i + max_k]) * dx
            else:
                # This is modulo indexing, but using explicit indexing
                g[i] = -h[i - (n_points - max_k)] * dx
        
        # Update h with gradient descent
        new_h = h - step_size * g
        
        # Apply projection to constraints
        new_h = project_h(new_h)
        
        h = new_h
        
        # Adaptive step size decay
        if iteration % 1000 == 0:
            step_size *= 0.5
            
        # Occasionally perturb to escape local minima
        if iteration % 500 == 0 and np.random.rand() < 0.1:
            pert = np.random.normal(0, 0.005, n_points)
            h = h + pert
            h = project_h(h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
