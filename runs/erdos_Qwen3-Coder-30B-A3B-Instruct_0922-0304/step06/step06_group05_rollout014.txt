import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convert to CVXPY variables and constraints
    def solve_feasibility(h_init, C):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1
        ]
        
        # Add overlap constraints for all k except k=0
        # We'll discretize overlapping terms
        overlap_max = 0
        for k in range(1, n_points):
            # For lag k, sum of h[i] * (1 - h[(i+k)%n])
            # cvxpy doesn't support direct circular convolution
            # But we can approximate it with loop constraints
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h[i] * (1 - h[j])
            constraints.append(expr <= C / dx)
            overlap_max = max(overlap_max, expr)
        
        # Solve with ECOS as fallback
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
        except:
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
            except:
                return False, None
        
        if prob.status in ["optimal", "optimal_inaccurate"]:
            return True, h.value
        return False, None
    
    # Binary search for C5
    C_low = 0.0
    C_high = 1.0
    best_feasible_C = C_high
    best_feasible_h = h.copy()
    
    while time.time() - start_time < budget_s - 1 and C_high - C_low > 1e-4:
        C = (C_low + C_high) / 2
        
        feasible, h_sol = solve_feasibility(h, C)
        if feasible:
            C_high = C
            best_feasible_C = C
            best_feasible_h = h_sol.copy() if h_sol is not None else h.copy()
        else:
            C_low = C
                
        # Break early if target achieved
        if C_high <= 0.3807:
            break
    
    # Final check with better optimizer if necessary
    final_h = best_feasible_h.copy()
    final_c5 = 0.0
    
    # Local refinement using gradient method
    max_iter = 20000
    iteration = 0
    step_size = 0.005
    
    h = best_feasible_h.copy()
    local_c5 = compute_c5(h)
    
    # Simple gradient descent for final refinement
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # gradient via finite diffs 
        g = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps) * dx

        new_h = h - step_size * g
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        if compute_c5(new_h) < compute_c5(h):
            h = new_h
        
        # Track best
        current_c5 = compute_c5(h)
        if current_c5 < local_c5:
            local_c5 = current_c5
            final_h = h.copy()
            if local_c5 <= 0.3807:
                break

    final_c5 = compute_c5(final_h)
    return (final_h.tolist(), final_c5, n_points)
