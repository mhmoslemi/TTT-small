import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize particles
    n_particles = 50
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.ones(n_points)
        base_h[n_points//2:] = 0
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    
    for _ in range(n_particles):
        h = base_h + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # PSO parameters
    w = 0.8
    c1 = c2 = 2.0
    p_best = [p.copy() for p in particles]
    p_best_fitness = [compute_c5(p, dx) for p in particles]
    g_best_idx = np.argmin(p_best_fitness)
    g_best = p_best[g_best_idx].copy()
    g_best_fitness = p_best_fitness[g_best_idx]
    
    iteration = 0
    max_iter = 3000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * v + c1 * r1 * (p_best[i] - h) + c2 * r2 * (g_best - h)
            
            # Update position
            h = h + v
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            
            # Store particle
            particles[i] = h
            velocities[i] = v
            
            # Update personal best
            fitness = compute_c5(h, dx)
            if fitness < p_best_fitness[i]:
                p_best[i] = h.copy()
                p_best_fitness[i] = fitness
                
                # Update global best
                if fitness < g_best_fitness:
                    g_best = h.copy()
                    g_best_fitness = fitness
        
        # Every 50 iterations: apply frequency domain acceleration
        if iteration % 50 == 0 and iteration > 0:
            for i in range(n_particles):
                h = particles[i]
                # FFT-based filtering
                fft_h = np.fft.fft(h)
                freqs = np.fft.fftfreq(n_points, dx)
                # High-pass filter (suppress low frequencies)
                mask = np.abs(freqs) > 0.3 / (2 * dx)
                fft_h_filtered = fft_h * mask
                h_filtered = np.real(np.fft.ifft(fft_h_filtered))
                h_filtered = np.clip(h_filtered, 0, 1)
                h_filtered = normalize_h(h_filtered, target_sum)
                # Replace particle with filtered version (randomly)
                if np.random.rand() < 0.5:
                    particles[i] = h_filtered
            
        # Every 100 iterations: refine top particles with SLSQP
        if iteration % 100 == 0 and iteration > 0:
            # Get top 10 particles
            top_indices = np.argsort(p_best_fitness)[:10]
            for idx in top_indices:
                h = p_best[idx].copy()
                
                def objective(h):
                    return compute_c5(h, dx)
                
                def constraint_sum(h):
                    return np.sum(h) - target_sum
                
                bounds = [(0, 1) for _ in range(n_points)]
                constraint = {'type': 'eq', 'fun': constraint_sum}
                
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 100})
                if result.success:
                    h_refined = result.x
                    h_refined = np.clip(h_refined, 0, 1)
                    h_refined = normalize_h(h_refined, target_sum)
                    fitness = compute_c5(h_refined, dx)
                    if fitness < p_best_fitness[idx]:
                        p_best[idx] = h_refined
                        p_best_fitness[idx] = fitness
                        if fitness < g_best_fitness:
                            g_best = h_refined.copy()
                            g_best_fitness = fitness
        
        iteration += 1
        
        # Early termination
        if g_best_fitness <= 0.3807:
            break
    
    # Final check and return
    final_c5 = compute_c5(g_best, dx)
    return (g_best.tolist(), final_c5, n_points)
