import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import math

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a step function that starts at 0.5
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.99
    min_T = 1e-8
    sa_steps = 0
    max_sa_steps = 100000
    
    while time.time() - start_time < budget_s - 1 and sa_steps < max_sa_steps:
        # Make a perturbation to h
        h_new = h.copy()
        # Modify about 10% of the values
        indices_to_modify = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        
        for i in indices_to_modify:
            # Add small random change
            h_new[i] += np.random.normal(0, 0.02)
            
        # Ensure all values are within [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to maintain sum constraint
        h_new = normalize_h(h_new, target_sum)
        
        # Compute C5 for both current and new h
        c5_current = compute_c5(h, dx)
        c5_new = compute_c5(h_new, dx)
        
        # Accept or reject based on simulated annealing
        if c5_new < c5_current or np.random.rand() < math.exp((c5_current - c5_new) / T):
            h = h_new.copy()
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Cool down temperature
        T *= cooling_rate
        sa_steps += 1
        
        # Early exit if target reached
        if best_c5 <= 0.3807:
            break
            
    # Run local refinement with SLSQP
    def objective(h):
        return compute_c5(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    # Run SLSQP optimization
    result = minimize(objective, best_h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        refined_h = result.x
        refined_h = np.clip(refined_h, 0, 1)
        refined_h = normalize_h(refined_h, target_sum)
        final_c5 = compute_c5(refined_h, dx)
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = refined_h.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
