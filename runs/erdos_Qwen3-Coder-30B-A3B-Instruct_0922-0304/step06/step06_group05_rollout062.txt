import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search for tight bound
    c_low = 0.0
    c_high = 1.0
    tolerance = 1e-4
    
    while time.time() - start_time < budget_s - 1 and c_high - c_low > tolerance:
        c_mid = (c_low + c_high) / 2.0
        
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # For each lag k (nonzero), we add constraint:
        # sum h[i] * (1 - h[(i+k) % n]) <= c_mid / dx
        for k in range(1, n_points):
            # Compute the sum of h[i] * (1 - h[(i+k) % n])
            lag_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                lag_sum += h_vars[i] * (1 - h_vars[j])
            
            # Add constraint
            constraints.append(lag_sum <= c_mid / dx)
            
        # Objective - not actually used in feasibility check
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # Check if we've found a better solution
                # For now, just evaluate the solution found by solver
                h_solution = h_vars.value
                if h_solution is not None:
                    h_eval = np.clip(h_solution, 0, 1)
                    val = compute_c5(h_eval)
                    if val < best_c5:
                        best_c5 = val
                        best_h = h_eval.copy()
                    # If we found a feasible solution, try smaller C
                    c_high = c_mid
                else:
                    # If no feasible solution from solver, try larger C
                    c_low = c_mid
            else:
                # If optimization failed to find a solution, try larger C
                c_low = c_mid
        except:
            # If solver fails, use fallback strategy
            # Use a local search approach but still track best
            c_low = c_mid
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
    # Final refinement using gradient-based method if needed
    if time.time() - start_time < budget_s - 1:
        max_iter = 5000
        iteration = 0
        step_size = 0.005
        
        while time.time() - start_time < budget_s - 1 and iteration < max_iter:
            iteration += 1
            
            overlap = correlate(best_h, 1 - best_h, mode="full")
            max_k = np.argmax(overlap)
            max_overlap = overlap[max_k] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                
            # Compute gradient via finite differences
            g = np.zeros(n_points)
            eps_val = 1e-4
            
            for i in range(n_points):
                h_plus = best_h.copy()
                h_minus = best_h.copy()
                
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                
                # Clamp and renormalize
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                # Compute finite difference
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Update h with gradient descent
            new_h = best_h - step_size * g
            
            # Apply constraints
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            best_h = new_h
            
            # Occasionally perturb to escape local minima
            if iteration % 100 == 0 and np.random.rand() < 0.1:
                pert = np.random.normal(0, 0.005, n_points)
                best_h = best_h + pert
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (n_points / 2) / np.sum(best_h)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
