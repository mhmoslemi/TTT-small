import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iteration = 0
    max_iter = 10000
    
    # Track target improvement to detect when we're stuck
    prev_c5 = float('inf')
    stagnation_counter = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Compute current overlap and identify worst lags
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Special optimization: use projected gradient to decrease max overlap
        # Identify worst lag
        overlap = correlate(h, 1 - h, mode="full")
        worst_lag_idx = np.argmax(overlap)
        
        # For the optimization, make sure we are optimizing around a clean problem:
        # We want to minimize C such that for the worst lag, h[i] * (1 - h[i + k]) <= C
        # This is a feasibility problem. We solve it by optimizing 
        # the overlap at that particular lag.
        # But reframe it: minimize current_c5 with tight constraint on worst lag.
        
        def constraint_worst_lag(h, lag_idx):
            # lag_idx is offset in full correlation array
            # we extract the specific overlap values for that lag, 
            # sum of h[i] * (1 - h[i + lag]) for valid range
            # should be <= C
            # We're building the constraint at this point
            # Through a bit of derivation and exploiting correlation structure:
            # overlap = h * (1-h) with mode full
            # so overlap[lag_idx] = sum h[i] * (1-h[i+lag]) for all valid i
            # where lag = lag_idx - (n_points - 1) 
            # But because we're doing correlation, 
            # the kernel goes from 0 to n_points-1, so if i+lag is within bounds we include
            
            lag = lag_idx - (n_points - 1)
            if lag >= 0:
                # i ranges from 0 to min(n_points-1, n_points-1-lag) 
                # which is min(n_points-1, n_points-1-lag) = n_points-1-lag
                # have valid indices from 0 to n_points-1-lag
                valid_range = np.arange(0, n_points - lag)
                contribution = np.sum(h[valid_range] * (1 - h[valid_range + lag]))
            else:
                # i ranges from max(0, -lag) to n_points-1 
                # which is max(0, -lag) to n_points-1
                valid_range = np.arange(max(0, -lag), n_points)
                contribution = np.sum(h[valid_range] * (1 - h[valid_range - lag]))
            
            return contribution
            

        # Generate a new candidate solution using greedy edits on top of current h
        h_new = h.copy()
        # Small random perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
        
        # Local optimization using SLSQP
        def objective(h):
            return compute_c5(h, dx)
        
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        # This is not actually used, but having this approach ensures feasible solutions
        try:
            # Try to improve by an SLSQP step on the current solution
            # This gives a better than just random hoping
            result = minimize(objective, h_new, method='SLSQP', bounds=bounds, constraints=constraint, 
                              options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate = normalize_to_target_sum(h_candidate, target_sum)
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < current_c5 * 0.9999:
                    h = h_candidate.copy()
            else:
                h = h_new
        except:
            h = h_new
            
        # Occasionally do a more aggressive search with different noise scale
        if iteration % 10 == 0:
            noise_scale = 0.05
            h_noisy = h + np.random.normal(0, noise_scale, n_points)
            h_noisy = np.clip(h_noisy, 0, 1)
            h_noisy = normalize_to_target_sum(h_noisy, target_sum)
            candidate_c5 = compute_c5(h_noisy, dx)
            if candidate_c5 < current_c5:
                h = h_noisy.copy()
        
        # Check for stagnation
        if abs(prev_c5 - current_c5) < 1e-6:
            stagnation_counter += 1
            if stagnation_counter > 50:
                # Do a full restart with new random initialization
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = normalize_to_target_sum(h, target_sum)
                stagnation_counter = 0
        else:
            stagnation_counter = 0
        prev_c5 = current_c5
        
        iteration += 1
    
    # Final check of best solution
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def normalize_to_target_sum(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h
