import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def crossover(parent1, parent2, rate=0.8):
    """Blend two parents to create offspring."""
    if np.random.random() > rate:
        return parent1.copy(), parent2.copy()
    
    # Simple uniform crossover
    mask = np.random.random(len(parent1)) > 0.5
    child1 = np.where(mask, parent1, parent2)
    child2 = np.where(mask, parent2, parent1)
    return child1, child2

def mutate(h, strength):
    """Apply Gaussian noise mutation."""
    noise = np.random.normal(0, strength, len(h))
    h_mut = h + noise
    return np.clip(h_mut, 0, 1)

def selection_tournament(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create base function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize population with mutated versions of base
    for _ in range(population_size):
        h_mut = mutate(h, 0.05)
        h_mut = normalize_h(h_mut, target_sum)
        population.append(h_mut)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main GA loop
    generations = 0
    max_generations = 600
    mutation_strength = 0.1
    
    while time.time() - start_time < budget_s - 1 and generations < max_generations:
        # Evaluate fitness (C5 values) for entire population
        fitnesses = [compute_c5(ind, dx) for ind in population]
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        if best_c5 <= 0.3807:
            break
        
        # Create new population through selection, crossover, and mutation
        new_population = []
        
        # Keep best individual (elitism)
        new_population.append(population[min_idx])
        
        # Generate rest of population
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = selection_tournament(population, fitnesses)
            parent2 = selection_tournament(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_strength)
            child2 = mutate(child2, mutation_strength)
            
            # Repair constraints
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Reduce mutation strength over time
        mutation_strength *= 0.995
        
        generations += 1
    
    # Final normalization and evaluation of best solution
    final_h = normalize_h(best_h, target_sum)
    final_c5 = compute_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
