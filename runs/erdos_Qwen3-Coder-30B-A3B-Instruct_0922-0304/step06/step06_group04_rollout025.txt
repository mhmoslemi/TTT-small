import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize multiple candidates
    m = 15
    candidates = []
    
    # Use initial if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        candidates.append(base_h.copy())
    else:
        # Create some diverse starting points
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
        candidates.append(base_h.copy())
    
    # Add other random distributions
    for _ in range(m - 1):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        candidates.append(h.copy())

    best_c5 = float('inf')
    best_individual = candidates[0].copy()

    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Try optimization from each candidate
        for i in range(len(candidates)):
            h = candidates[i].copy()
            
            # Local optimization using SLSQP
            def constraint_sum(h):
                return np.sum(h) - target_sum
                
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': constraint_sum}
            
            def objective(h):
                return compute_c5(h, dx)
            
            try:
                result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                                constraints=constraint, options={'maxiter': 200})
                if result.success:
                    h = result.x
                    h = np.clip(h, 0, 1)
                    h = normalize_h(h, target_sum)
                    c5 = compute_c5(h, dx)
                    if c5 < best_c5:
                        best_c5 = c5
                        best_individual = h.copy()
            except:
                pass
            
            # Early exit if target achieved
            if best_c5 <= 0.3807:
                break
                
            # Generate a new candidate for next iteration
            if time.time() - start_time > budget_s - 5:
                break
                
            # Perturb current candidate for next round
            noise = np.random.normal(0, 0.03, n_points)
            h = candidates[i] + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            candidates[i] = h
            
        iteration += 1

    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
