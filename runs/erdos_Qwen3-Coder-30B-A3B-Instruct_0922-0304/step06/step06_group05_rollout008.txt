import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: sinusoidal pattern
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(np.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def project_h(h, target_sum):
        """Project h to satisfy sum constraint"""
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * target_sum / current_sum
        return h

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient-based optimization with momentum and adaptive step size
    max_iter = 50000
    iteration = 0
    step_size = 0.01
    momentum = 0.9
    prev_grad = np.zeros(n_points)
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
        
        # Compute exact gradient for the current maximum lag
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            if j >= i:
                g[i] = (1 - h[j]) * dx
            else:
                g[i] = -h[i - max_k] * dx
        
        # Update with momentum
        new_h = h - step_size * (g + momentum * prev_grad)
        prev_grad = g
        
        # Apply constraints
        new_h = project_h(new_h, n_points / 2)
        h = new_h
        
        # Adaptive step size and momentum decay
        if iteration % 1000 == 0 and step_size > 1e-6:
            step_size *= 0.5
            momentum *= 0.95
        
        # Occasionally perturb to escape local minima
        if iteration % 200 == 0 and np.random.rand() < 0.05:
            pert = np.random.normal(0, 0.002, n_points)
            h = h + pert
            h = project_h(h, n_points / 2)

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
