import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Multi-start optimization with diverse initializations
    m_starts = 15
    best_c5 = float('inf')
    best_individual = None
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    def objective(h):
        return compute_c5(h, dx)
    
    # Generate diverse initial candidates
    initial_candidates = []
    
    # Base from initial_h_values if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        initial_candidates.append(base_h.copy())
    
    # Add random initializations
    for _ in range(m_starts - len(initial_candidates)):
        # Start with a random step function
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        initial_candidates.append(h)
    
    # Add specially crafted candidates
    # One with a spike
    h_spike = np.zeros(n_points)
    h_spike[n_points//2] = 1.0
    h_spike = normalize_h(h_spike, target_sum)
    initial_candidates.append(h_spike)
    
    # Smoother distribution
    h_smooth = np.random.beta(2, 2, n_points)
    h_smooth = normalize_h(h_smooth, target_sum)
    initial_candidates.append(h_smooth)
    
    # Uniform initialization
    h_uniform = np.ones(n_points) * 0.5
    h_uniform = normalize_h(h_uniform, target_sum)
    initial_candidates.append(h_uniform)
    
    # Gaussian-like
    x = np.linspace(0, 2, n_points)
    mu, sigma = 1.0, 0.2
    h_gauss = np.exp(-0.5 * ((x - mu) / sigma) ** 2)
    h_gauss = normalize_h(h_gauss, target_sum)
    initial_candidates.append(h_gauss)
    
    # Run optimization for each candidate
    for i, h_init in enumerate(initial_candidates):
        h = h_init.copy()
        
        # Add noise and perturb
        noise = np.random.normal(0, 0.03, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        try:
            result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                              constraints=constraint, options={'maxiter': 200})
            
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h.copy()
        except:
            pass

        if time.time() - start_time > budget_s - 1:
            break
    
    # If no successful run, return the initial solution
    if best_individual is None:
        if 'initial_h_values' in globals():
            base_h = np.array(initial_h_values)
            base_h = np.clip(base_h, 0, 1)
            base_h = normalize_h(base_h, target_sum)
            best_individual = base_h
        
    # Final evaluation
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        # Fallback to uniform
        h_uniform = np.ones(n_points) * 0.5
        h_uniform = normalize_h(h_uniform, target_sum)
        final_c5 = compute_c5(h_uniform, dx)
        best_individual = h_uniform
    
    return (best_individual.tolist(), final_c5, n_points)
