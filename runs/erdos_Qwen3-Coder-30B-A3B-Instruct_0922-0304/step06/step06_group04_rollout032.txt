import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize population
    population_size = 20
    mutation_rate = 0.2
    crossover_prob = 0.8
    elite_size = population_size // 10
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        population = [base_h.copy()]
        for _ in range(population_size - 1):
            perturbed = base_h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = normalize_h(perturbed, target_sum)
            population.append(perturbed)
    else:
        population = []
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 20
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness of population
        fitness = [evaluate_individual(h, dx) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_individual = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 <= 0.3807:
            break
        
        # Selection: keep top half
        selected = population[:population_size // 2]
        selected_fitness = fitness[:population_size // 2]
        
        # Create new population
        new_population = selected.copy()
        
        # Elitism
        if elite_size > 0:
            new_population.extend(population[:elite_size])
        
        # Crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob:
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                p1, p2 = selected[i1], selected[i2]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
                child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just copy one individual
                i = np.random.randint(len(selected))
                new_population.append(selected[i].copy())
        
        # Mutate
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                child = new_population[i] + np.random.normal(0, 0.05, n_points)
                child = np.clip(child, 0, 1)
                child = normalize_h(child, target_sum)
                new_population[i] = child
        
        population = new_population
        generation += 1
        
        # Adjust mutation rate if no improvement for too long
        if no_improvement_count > max_no_improvement:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            no_improvement_count = 0
        
        # Increase population size if needed
        if generation % 20 == 0 and len(population) < 40:
            population_size = min(40, len(population) * 2)
            # Recompute to keep the same size
            if len(population) < population_size:
                # Add random individuals from best
                if len(population) < population_size:
                    for _ in range(population_size - len(population)):
                        h = np.random.rand(n_points)
                        h = normalize_h(h, target_sum)
                        population.append(h)
                else:
                    population = population[:population_size]
    
    # Final evaluation of best solution
    if best_individual is not None:
        final_c5 = compute_c5(best_individual, dx)
    else:
        # Fallback to initial if no solution found
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx)
            best_individual = h.copy()
        else:
            h = np.zeros(n_points)
            h[:n_points//2] = 1.0
            h = normalize_h(h, target_sum)
            final_c5 = compute_c5(h, dx)
            best_individual = h.copy()
    
    return (best_individual.tolist(), final_c5, n_points)
