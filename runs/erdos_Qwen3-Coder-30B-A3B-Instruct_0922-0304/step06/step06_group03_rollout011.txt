import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def create_comb_pattern(n_points, spacing, tooth_height=1.0):
    """Create a comb pattern with specified spacing."""
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Create tooth positions
    tooth_positions = np.arange(0, 2, spacing * dx)
    tooth_indices = np.floor(tooth_positions / dx).astype(int)
    tooth_indices = tooth_indices[tooth_indices < n_points]
    
    # Assign tooth heights
    for idx in tooth_indices:
        h[idx] = tooth_height
    
    return h

def perturb_pattern(h, sigma=0.02):
    """Add small random perturbations."""
    noise = np.random.normal(0, sigma, len(h))
    h_perturbed = h + noise
    return np.clip(h_perturbed, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from existing construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Start with a simple comb pattern
        h = create_comb_pattern(n_points, spacing=3)
        h = normalize_h(h, target_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Try different comb patterns
    spacings = [2, 3, 4, 5, 6]
    
    # Evaluate initial pattern
    c5 = compute_c5(h, dx)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
        
    # Multi-resolution search
    for i in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Select a random comb pattern
        spacing = np.random.choice(spacings)
        h_temp = create_comb_pattern(n_points, spacing)
        h_temp = normalize_h(h_temp, target_sum)
        
        # Perturb and optimize
        h_perturbed = perturb_pattern(h_temp, 0.02)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = normalize_h(h_perturbed, target_sum)
        
        def objective(h_vec):
            return compute_c5(h_vec, dx)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h_perturbed, method='SLSQP', 
                            bounds=bounds, constraints=constraint, options={'maxiter': 50})
            if result.success:
                h_new = result.x
                h_new = np.clip(h_new, 0, 1)
                h_new = normalize_h(h_new, target_sum)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            pass
        
        if best_c5 <= 0.3807:
            break
            
    # Now refine optimization at higher resolution
    if n_points == 70:
        n_points_high = 140
        dx_high = 2.0 / n_points_high
        target_sum_high = n_points_high / 2
        
        # Transfer best pattern to higher resolution
        h_high = np.interp(np.linspace(0, 2, n_points_high), 
                          np.linspace(0, 2, n_points), best_h)
        h_high = normalize_h(h_high, target_sum_high)
        
        # Final optimization at high resolution
        def objective_high(h_vec):
            return compute_c5(h_vec, dx_high)
            
        def constraint_sum_high(h_vec):
            return np.sum(h_vec) - target_sum_high
            
        bounds = [(0, 1) for _ in range(n_points_high)]
        constraint = {'type': 'eq', 'fun': constraint_sum_high}
        
        h_final = h_high.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            try:
                result = minimize(objective_high, h_final, method='SLSQP',
                                bounds=bounds, constraints=constraint, options={'maxiter': 10})
                if result.success:
                    h_final = result.x
                    h_final = np.clip(h_final, 0, 1)
                    h_final = normalize_h(h_final, target_sum_high)
                    c5_final = compute_c5(h_final, dx_high)
                    if c5_final < best_c5:
                        best_c5 = c5_final
                        best_h = h_final.copy()
                else:
                    # Perturb and try again
                    h_final = h_final + np.random.normal(0, 0.01, n_points_high)
                    h_final = np.clip(h_final, 0, 1)
                    h_final = normalize_h(h_final, target_sum_high)
            except:
                break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
