import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the same parameters as the parent
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        # Create an initial step function
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        h = np.clip(h, 0, 1)
    
    # Normalize to target sum
    h = h * (target_sum / np.sum(h))
    
    # Initialize best solution
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()
    
    # Try a more structured approach using SDP relaxation if possible
    try:
        # Create a simplified convex optimization approach
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Define the constraints
        constraints = [
            cp.sum(h_vars) == target_sum,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Add the overlap constraints for all possible shifts
        full_length = 2 * n_points - 1
        shift_indices = np.arange(-n_points + 1, n_points)
        
        for k in shift_indices:
            # Compute shifted indices
            shifted_indices = np.arange(n_points)
            if k > 0:
                shifted_indices = np.roll(shifted_indices, -k)
            elif k < 0:
                shifted_indices = np.roll(shifted_indices, -k)
                
            # The overlap is sum of h[i] * (1 - h[i-k])
            # We'll compute a linear approximation to avoid complex matrix multiplication
            if k >= 0:
                expr = cp.sum(cp.multiply(h_vars, 1 - h_vars[shifted_indices]))
            else:
                expr = cp.sum(cp.multiply(h_vars, 1 - h_vars[np.roll(np.arange(n_points), -k)]))
            
            constraints.append(expr <= C5)
        
        # Objective: minimize C5
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Solve the SDP
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=200)
        
        # Check if we got a solution in time
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_sol = h_vars.value
            if h_sol is not None:
                h = np.clip(h_sol, 0, 1)
                # Normalize to ensure sum constraint
                h = h * (target_sum / np.sum(h))
                current_c5 = compute_c5(h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
    except:
        # Fallback to the original optimization approach
        pass
    
    # If SDP failed or remaining time allows, apply local optimization
    if time.time() - start_time < budget_s - 1:
        iterations = 0
        max_iter = 2000
        while time.time() - start_time < budget_s - 1 and iterations < max_iter:
            # Simple local search
            updated = False
            for i in range(n_points):
                # Try small perturbations
                old_h_i = best_h[i]
                for delta in [-0.02, -0.01, 0.01, 0.02]:
                    test_h = best_h.copy()
                    test_h[i] += delta
                    test_h = np.clip(test_h, 0, 1)
                    # Normalize
                    test_h = test_h * (target_sum / np.sum(test_h))
                    test_c5 = compute_c5(test_h, dx)
                    if test_c5 < best_c5:
                        best_c5 = test_c5
                        best_h = test_h.copy()
                        updated = True
            
            # If no improvement, do a random small mutation
            if not updated:
                mutation_strength = max(0.001, 0.02 * np.exp(-iterations / 300.0))
                best_h += np.random.normal(0, mutation_strength, n_points)
                best_h = np.clip(best_h, 0, 1)
                best_h = best_h * (target_sum / np.sum(best_h))
                current_c5 = compute_c5(best_h, dx)
                if current_c5 < best_c5:
                    best_c5 = current_c5
            
            iterations += 1
            
            if best_c5 <= 0.3807:
                break
    
    return (best_h.tolist(), best_c5, n_points)
