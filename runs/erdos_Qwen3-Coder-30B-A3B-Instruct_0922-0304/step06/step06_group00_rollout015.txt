import numpy as np
from scipy.signal import correlate
import time
from scipy.optimize import minimize
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_rate, dx, target_sum):
    """Mutate an individual with given rate."""
    h_m = h.copy()
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h_m[i] += np.random.normal(0, 0.05)
    h_m = np.clip(h_m, 0, 1)
    h_m = normalize_h(h_m, target_sum)
    return h_m

def crossover(parent1, parent2, crossover_rate):
    """Perform crossover between two parents."""
    if random.random() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(tournament_size - 1):
        competitor_idx = random.randint(0, len(population) - 1)
        if fitnesses[competitor_idx] < fitnesses[selected_idx]:
            selected_idx = competitor_idx
    return population[selected_idx]

def local_optimize(h, target_sum, dx):
    """Locally optimize an individual using scipy minimize."""
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    bounds = [(0, 1) for _ in range(len(h))]
    constraint = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = minimize(objective, h.copy(), method='SLSQP', 
                         bounds=bounds, constraints=constraint, 
                         options={'maxiter': 50})
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = normalize_h(h_opt, target_sum)
            return h_opt
    except:
        pass
    return h.copy()

        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Genetic Algorithm parameters
    population_size = 100
    mutation_rate = 0.2
    crossover_rate = 0.8
    generations = 60
    
    # Initialize population
    population = [h.copy()]
    for _ in range(population_size - 1):
        mut_h = mutate_individual(h, mutation_rate, dx, target_sum)
        population.append(mut_h)
    
    best_c5 = compute_c5(h, dx)
    best_h = h.copy()

        current_best_idx = np.argmin(fitnesses)
