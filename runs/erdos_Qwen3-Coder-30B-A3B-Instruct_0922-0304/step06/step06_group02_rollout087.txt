import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (target_sum / current_sum)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Convex QP formulation
    # Variables: h vector, C (auxiliary variable)
    h_vars = cp.Variable(n_points)
    C = cp.Variable()
    
    # Sum constraint
    sum_constraint = cp.Equality(cp.sum(h_vars), target_sum)
    
    # Lag constraints: C >= sum h[i] * (1 - h[i + k]) * dx for all lags k
    lag_constraints = []
    
    for k in range(-n_points + 1, n_points):
        # Construct overlap sum for lag k
        indices = np.arange(n_points)
        if k >= 0:
            valid_indices = indices[:-k] if k < n_points else []
            sum_expr = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices + k]))
        else:
            valid_indices = indices[-k:] if -k < n_points else []
            sum_expr = cp.sum(cp.multiply(h_vars[valid_indices], 1 - h_vars[valid_indices - k]))
        
        lag_constraints.append(C >= sum_expr * dx)
    
    # Box constraints
    box_constraints = [h_vars >= 0, h_vars <= 1]
    
    # Solve with CVXPY
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, [sum_constraint] + lag_constraints + box_constraints)
    
    # Try to solve with multiple solvers in a loop
    solvers_to_try = ['ECOS', 'SCIP', 'GLPK']
    best_constraint_violation = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        try:
            prob.solve(solver=cp.SCS, max_iters=5000, verbose=False)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_solution = h_vars.value
                c5_sol = C.value
                if c5_sol < best_c5:
                    best_c5 = c5_sol
                    best_h = h_solution.copy()
                if best_c5 <= 0.3807:
                    break
            elif prob.status == 'infeasible':
                # Try to relax slightly
                try:
                    C.value = best_c5 * 1.01
                    prob.solve(solver=cp.SCS, max_iters=1000, verbose=False)
                    if prob.status == 'optimal' or prob.status == 'optimal_inaccurate':
                        h_solution = h_vars.value
                        c5_sol = C.value
                        if c5_sol < best_c5:
                            best_c5 = c5_sol
                            best_h = h_solution.copy()
                        if best_c5 <= 0.3807:
                            break
                except:
                    pass
        except:
            pass
        
        # If we don't have a solution, try a different approach
        try:
            # Simple local refinement with gradient-based method
            bounds = [(0, 1) for _ in range(n_points)]
            def objective(h):
                return compute_c5(h, dx)
            def constraint_sum(h):
                return np.sum(h) - target_sum
            
            # Perturb and optimize
            h_noisy = h + np.random.normal(0, 0.02, len(h))
            h_noisy = np.clip(h_noisy, 0, 1)
            current_sum = np.sum(h_noisy)
            h_noisy = h_noisy * (target_sum / current_sum)
            # Use scipy for local search if needed
            from scipy.optimize import minimize
            constraint = {'type': 'eq', 'fun': constraint_sum}
            result = minimize(objective, h_noisy, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 100})
            if result.success:
                h_candidate = result.x
                h_candidate = np.clip(h_candidate, 0, 1)
                h_candidate_sum = np.sum(h_candidate)
                h_candidate = h_candidate * (target_sum / h_candidate_sum)
                candidate_c5 = compute_c5(h_candidate, dx)
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = h_candidate.copy()
                if best_c5 <= 0.3807:
                    break
        except:
            pass
        
        # Slight perturbation to escape local optima
        h = np.clip(h + np.random.normal(0, 0.01, len(h)), 0, 1)
        current_sum = np.sum(h)
        h = h * (target_sum / current_sum)
    
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
