import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def evaluate_individual(h, dx):
    return compute_c5(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize population
    population_size = 20
    population = []
    
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
        for _ in range(population_size):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = normalize_h(h, target_sum)
            population.append(h)
    
    best_c5 = float('inf')
    best_individual = None
    
    # Tracking for termination
    patience = 0
    max_patience = 20
    
    generation = 0
    while time.time() - start_time < budget_s - 1:
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_individual(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Update best if improved
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_individual = sorted_population[0].copy()
            patience = 0
        else:
            patience += 1
            
        if best_c5 <= 0.3807:
            break
            
        # Selection: top 50%
        selected_size = population_size // 2
        selected_population = sorted_population[:selected_size]
        
        # Crossover: uniform segment swapping
        new_population = []
        for i in range(0, selected_size, 2):
            parent1 = selected_population[i]
            parent2 = selected_population[i+1] if i+1 < selected_size else selected_population[0]
            
            # Create offspring by randomly swapping parts
            crossover_point = np.random.randint(1, n_points - 1)
            child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        
            new_population.extend([child1, child2])
            
        # Keep best individual
        new_population.append(best_individual)
        
        # Mutate 20% of population
        mutation_rate = 0.2 if patience < 10 else 0.1
        for i in range(len(new_population)):
            if np.random.random() < mutation_rate:
                h = new_population[i]
                noise = np.random.normal(0, 0.05, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                new_population[i] = h
        
        # Elitism: keep best individual
        new_population[0] = best_individual
        
        # Maintain population size
        if len(new_population) > population_size:
            # Keep best individuals
            fitness_new = [evaluate_individual(h, dx) for h in new_population]
            sorted_indices = np.argsort(fitness_new)
            new_population = [new_population[i] for i in sorted_indices[:population_size]]
        
        population = new_population
        generation += 1
        
        if patience > max_patience:
            # Increase population size if no improvement
            if population_size < 40:
                population_size += 10
                # Add random individuals
                for _ in range(10):
                    h = np.random.rand(n_points)
                    h = normalize_h(h, target_sum)
                    population.append(h)
            else:
                break
    
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
