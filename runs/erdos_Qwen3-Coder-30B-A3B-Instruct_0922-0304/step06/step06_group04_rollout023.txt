import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with the parent construction
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        # Fallback initialization
        base_h = np.ones(n_points) * 0.5
        
    best_c5 = float('inf')
    best_h = base_h.copy()
    
    # Alternating optimization
    iteration = 0
    max_iter = 500
    perturb_mag = 0.05
    stagnation_counter = 0
    best_c5_history = []
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        h = best_h.copy()
        
        # Add perturbation
        noise = np.random.normal(0, perturb_mag, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        
        # Use a constrained optimization approach similar to previous but with 
        # improved numerical handling
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        # Use SLSQP with early stopping
        result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                          constraints=constraint, options={'maxiter': 100, 'ftol': 1e-6})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = normalize_h(h_opt, target_sum)
            
            c5_new = compute_c5(h_opt, dx)
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_opt.copy()
                stagnation_counter = 0
                best_c5_history.append(best_c5)
            else:
                stagnation_counter += 1
                
            # Reduce perturbation if stagnating
            if stagnation_counter > 10:
                perturb_mag = max(perturb_mag * 0.9, 0.005)
                
        else:
            # If optimization fails, try a simple perturbation
            h = best_h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = normalize_h(h, target_sum)
            c5_new = compute_c5(h, dx)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        # Check early termination
        if best_c5 <= 0.3807:
            break
            
        iteration += 1
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
