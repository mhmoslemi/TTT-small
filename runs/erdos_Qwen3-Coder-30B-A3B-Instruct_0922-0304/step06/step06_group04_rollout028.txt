import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        base_h = np.array(initial_h_values)
        base_h = np.clip(base_h, 0, 1)
        base_h = normalize_h(base_h, target_sum)
    else:
        base_h = np.zeros(n_points)
        base_h[:n_points//2] = 1.0
        base_h = normalize_h(base_h, target_sum)
    
    best_c5 = float('inf')
    best_individual = base_h.copy()
    
    # Multi-start optimization with different strategies
    m = 15  # Number of different initializations
    initializations = []
    
    # Strategy 1: Random noise around base solution
    for i in range(m//3):
        h = base_h + np.random.normal(0, 0.03, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # Strategy 2: Step functions
    for i in range(m//3):
        h = np.zeros(n_points)
        pos = np.random.randint(0, n_points//4)
        h[pos:pos+n_points//2] = 1.0
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # Strategy 3: Smoothed random
    for i in range(m//3):
        h = np.random.rand(n_points)
        # Apply Gaussian smoothing
        kernel = np.exp(-np.linspace(-2, 2, 21)**2 / 2)
        kernel = kernel / np.sum(kernel)
        h = np.convolve(h, kernel, mode='same')
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        initializations.append(h)
    
    # Optimization loop over initializations
    for i, h0 in enumerate(initializations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Local optimization
        def constraint_sum(h):
            return np.sum(h) - target_sum
            
        bounds = [(0, 1) for _ in range(n_points)]
        constraint = {'type': 'eq', 'fun': constraint_sum}
        
        def objective(h):
            return compute_c5(h, dx)
        
        try:
            result = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                            constraints=constraint, options={'maxiter': 200})
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = normalize_h(h, target_sum)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_individual = h.copy()
        except Exception:
            pass
        
        # Early exit
        if best_c5 <= 0.3807:
            break
    
    # Final validation
    final_c5 = compute_c5(best_individual, dx)
    return (best_individual.tolist(), final_c5, n_points)
