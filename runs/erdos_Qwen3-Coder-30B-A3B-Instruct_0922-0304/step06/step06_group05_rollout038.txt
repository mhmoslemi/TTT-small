import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    best_c5 = float('inf')
    best_h = h.copy()
    
    # Binary search + convex optimization approach
    C_low = 0.0
    C_high = 1.0
    tolerance = 1e-4
    
    while time.time() - start_time < budget_s - 5 and C_high - C_low > tolerance:
        C = (C_low + C_high) / 2.0
        
        # Try to find feasible h for this C
        h_vars = cp.Variable(n_points)
        constraints = [
            cp.sum(h_vars) == n_points / 2,
            h_vars >= 0,
            h_vars <= 1
        ]
        
        # Approximate all constraints using a multiple of the objective
        # Here we constrain the overlap integral to be bounded by C*dx
        # We directly construct an approximate constraint using the 
        # full convolution, but only a linearized problem.
        #
        # For verification, we use a simpler equality: 
        # This is a less accurate technique but serves as a feasibility test.
        
        # Try to solve for this C - use simple iterative descent
        h_trial = h.copy()
        step_size = 0.001
        max_iter = 1000
        converged = False
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Evaluate max overlap with current h_trial
            overlap = correlate(h_trial, 1 - h_trial, mode="full")
            max_overlap = overlap.max() * dx
            
            if max_overlap <= C:
                h = h_trial.copy()
                converged = True
                break
            
            # Simple gradient descent to reduce max overlap
            # Compute an approximate gradient direction
            g = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h_trial.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                g[i] = (overlap_plus.max() * dx - max_overlap) / eps
            
            # Gradient step
            new_h = h_trial - step_size * g
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            h_trial = new_h
            
        if converged:
            C_high = C
            if C < best_c5:
                best_c5 = C
                best_h = h.copy()
        else:
            C_low = C
            
        # Early exit if target is achieved
        if best_c5 <= 0.3807:
            break

    # Final refinement using gradient-based local search
    h = best_h.copy()
    iteration = 0
    max_iter_refinement = 20000
    
    while time.time() - start_time < budget_s - 2 and iteration < max_iter_refinement:
        iteration += 1
        
        # Compute current overlap and max
        overlap = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        # Store the best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Early exit if target achieved
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient via finite differences
        g = np.zeros(n_points)
        eps_val = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            
            h_plus[i] += eps_val
            h_minus[i] -= eps_val
            
            # Clamp and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            # Compute finite difference
            overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
            overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
            g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

        # Update h with gradient descent
        new_h = h - 0.01 * g
        
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h

    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
