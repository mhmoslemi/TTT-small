import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def fourier_to_h(coefficients, n_points, n_freq):
    """Convert Fourier coefficients to h values."""
    x = np.linspace(0, 2, n_points)
    h = np.zeros(n_points)
    for k in range(n_freq):
        h += coefficients[2*k] * np.cos(k * np.pi * x) + coefficients[2*k+1] * np.sin(k * np.pi * x)
    return h

def h_to_fourier(h, n_freq):
    """Convert h values to Fourier coefficients."""
    # We only need the first n_freq coefficients
    coeffs = []
    x = np.linspace(0, 2, len(h))
    for k in range(n_freq):
        if k == 0:
            # DC component
            coeffs.append(np.mean(h))
        else:
            # Cosine and sine components
            cos_integral = np.sum(h * np.cos(k * np.pi * x)) * 2 / len(h)  # Normalize by 2
            sin_integral = np.sum(h * np.sin(k * np.pi * x)) * 2 / len(h)
            coeffs.extend([cos_integral, sin_integral])
    return coeffs

def project_and_normalize(h, target_sum):
    """Project h to [0,1] and normalize to target sum."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    n_freq = 15  # Number of frequency components
    
    # Initial solution
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 1.0
        initial_h = project_and_normalize(initial_h, target_sum)

    # Convert initial solution to Fourier coefficients
    initial_coeffs = h_to_fourier(initial_h, n_freq)
    
    # PSO parameters
    population_size = 50
    max_iter = 500
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_scores = []
    
    for i in range(population_size):
        # Perturb initial coefficients
        coeffs = initial_coeffs.copy()
        noise = np.random.normal(0, 0.1, len(coeffs))
        coeffs += noise
        particles.append(coeffs)
        velocities.append(np.random.uniform(-0.1, 0.1, len(coeffs)))
        pbest.append(coeffs.copy())
        # Evaluate initial particle
        h = fourier_to_h(coeffs, n_points, n_freq)
        h = project_and_normalize(h, target_sum)
        score = compute_c5(h, dx)
        pbest_scores.append(score)
        
    # Global best
    gb_idx = np.argmin(pbest_scores)
    gbest = pbest[gb_idx].copy()
    gbest_score = pbest_scores[gb_idx]
    
    # PSO loop
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 1:
            break
            
        for i in range(population_size):
            # Update velocity
            r1, r2 = np.random.rand(len(velocities[i])), np.random.rand(len(velocities[i]))
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Evaluate
            h = fourier_to_h(particles[i], n_points, n_freq)
            h = project_and_normalize(h, target_sum)
            score = compute_c5(h, dx)
            
            # Update pbest
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
            
            # Update gbest
            if score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = score
                
        # Early stop
        if gbest_score <= 0.3807:
            break
            
    # Final refinement with local optimization
    h_final = fourier_to_h(gbest, n_points, n_freq)
    h_final = project_and_normalize(h_final, target_sum)
    
    # Local refinement
    def objective(h):
        return compute_c5(h, dx)
        
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    constraint = {'type': 'eq', 'fun': constraint_sum}
    result = minimize(objective, h_final, method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 200})
    
    if result.success:
        h_final = result.x
        h_final = project_and_normalize(h_final, target_sum)
    
    final_c5 = compute_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
