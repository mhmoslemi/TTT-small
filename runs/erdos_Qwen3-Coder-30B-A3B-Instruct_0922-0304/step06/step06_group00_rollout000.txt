import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def mutate_individual(h, mutation_strength):
    """Mutate an individual by adding Gaussian noise."""
    h_mut = h + np.random.normal(0, mutation_strength, len(h))
    h_mut = np.clip(h_mut, 0, 1)
    return h_mut

def crossover_parents(p1, p2, crossover_rate=0.8):
    """Crossover two parents to create offspring."""
    if np.random.rand() > crossover_rate:
        return p1.copy(), p2.copy()
    
    # Simple one-point crossover
    crossover_point = np.random.randint(1, len(p1))
    child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    else:
        # Create a base function with more structure
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        # Add some randomness for diversity
        h += np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
    
    # Initialize population
    population_size = 100
    population = [h.copy()]
    for _ in range(population_size - 1):
        h_mut = mutate_individual(h, 0.05)
        h_mut = normalize_h(h_mut, target_sum)
        population.append(h_mut)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = compute_c5(best_h, dx)
    
    # Genetic Algorithm parameters
    max_generations = 600
    mutation_strength = 0.2
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness of population
        fitnesses = []
        for ind in population:
            c5 = compute_c5(ind, dx)
            fitnesses.append(c5)
        
        # Update best solution
        min_fitness_index = np.argmin(fitnesses)
        if fitnesses[min_fitness_index] < best_c5:
            best_c5 = fitnesses[min_fitness_index]
            best_h = population[min_fitness_index].copy()
            
        if best_c5 <= 0.3807:
            break
            
        # Create new population through selection, crossover, and mutation
        new_population = []
        for _ in range(population_size // 2):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover_parents(parent1, parent2)
            
            # Mutation
            child1 = mutate_individual(child1, mutation_strength)
            child1 = normalize_h(child1, target_sum)
            child2 = mutate_individual(child2, mutation_strength)
            child2 = normalize_h(child2, target_sum)
            
            new_population.extend([child1, child2])
        
        # Replace population
        population = new_population
        
        # Decrease mutation strength
        mutation_strength = max(0.001, mutation_strength * 0.99)
    
    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
