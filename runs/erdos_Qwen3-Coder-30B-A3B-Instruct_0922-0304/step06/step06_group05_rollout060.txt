import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Fallback initialization: step function with some randomness
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Binary search for C5
    C_low = 0.0
    C_high = 1.0
    best_C5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h):
        overlap = correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    def solve_feasibility(h, C):
        """Check if there's a feasible h for C using convex optimization."""
        # Objective: minimize max overlap
        # Variables: h[i] 
        # Constraints: sum h == n_points/2, h in [0,1], all overlaps <= C/dx
        h_vars = cp.Variable(n_points)
        constraints = [cp.sum(h_vars) == n_points/2, h_vars >= 0, h_vars <= 1]
        
        # Constraint on overlaps: h[i](1-h[i+k]) <= C/dx
        for k in range(1, n_points):
            # Calculate overlap sum for lag k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= C / dx)
        
        # Minimize a dummy objective to check feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except:
            return False

    def refine_solution(h):
        """Refine solution with gradient descent."""
        best_h = h.copy()
        best_c5 = compute_c5(h)
        step_size = 0.01
        max_iter = 3000
        for it in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
            overlap = correlate(h, 1 - h, mode="full")
            max_k = np.argmax(overlap)
            max_overlap = overlap[max_k] * dx
            
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h.copy()
                
            if best_c5 <= 0.3807:
                break
                
            # Compute gradient with finite differences
            g = np.zeros(n_points)
            eps_val = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps_val
                h_minus[i] -= eps_val
                # Clamp and renormalize
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                # Compute finite difference
                overlap_plus = correlate(h_plus, 1 - h_plus, mode="full")
                overlap_minus = correlate(h_minus, 1 - h_minus, mode="full")
                g[i] = (overlap_plus.max() - overlap_minus.max()) / (2 * eps_val) * dx

            # Update h with gradient descent
            new_h = h - step_size * g
            # Apply constraints
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            h = new_h
            
        return best_h, best_c5

    # Binary search loop with feasibility check and refinement
    while C_high - C_low > 1e-4 and time.time() - start_time < budget_s - 5:
        C_mid = (C_low + C_high) / 2
        if solve_feasibility(h, C_mid):
            C_high = C_mid
        else:
            C_low = C_mid
    
    # Final refinement using gradient-based method
    final_h, final_c5 = refine_solution(best_h)
    
    # Ensure final fulfill the constraints
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
