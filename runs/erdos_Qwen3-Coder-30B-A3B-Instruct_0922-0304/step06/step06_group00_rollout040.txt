import numpy as np
from scipy.signal import correlate
import time

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def tournament_selection(population, fitness, k=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), size=k)
    best = np.argmin([fitness[i] for i in idxs])
    return population[idxs[best]].copy()

def crossover(parent1, parent2, rate=0.8):
    """Perform single-point crossover."""
    if np.random.rand() > rate:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    cx_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    
    return child1, child2

def mutate(individual, rate=0.2, strength=0.1):
    """Mutate an individual with Gaussian noise."""
    if np.random.rand() > rate:
        return individual.copy()
        
    noise = np.random.normal(0, strength, len(individual))
    individual += noise
    return np.clip(individual, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Population parameters
    pop_size = 100
    generations = 600  # Upper limit for time
    mutation_rate = 0.2
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population from initial_h_values if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = normalize_h(initial_h, target_sum)
    else:
        # Create a diverse base
        initial_h = np.zeros(n_points)
        initial_h[:n_points//3] = 1
        initial_h[n_points//3:2*n_points//3] = 0.5
        initial_h[2*n_points//3:] = 0.25
    
    population = [initial_h.copy()]
    for _ in range(pop_size - 1):
        # Add some diversity
        h = initial_h + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = normalize_h(h, target_sum)
        population.append(h.copy())
    
    best_fitness = float('inf')
    best_individual = None
    
    # Precompute all lags
    full_length = 2 * n_points - 1
    lags = np.arange(-n_points + 1, n_points)
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = compute_c5(h, dx)
            fitness.append(c5)
            if c5 < best_fitness:
                best_fitness = c5
                best_individual = h.copy()
        
        # Early termination if target achieved
        if best_fitness <= 0.3807:
            break
        
        # Selection, crossover, mutation
        new_population = []
        
        # Keep best 10% (elitism)
        elite_indices = np.argsort(fitness)[:pop_size//10]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate new population
        while len(new_population) < pop_size:
            # Tournament selection for parent 1
            parent1 = tournament_selection(population, fitness, tournament_size)
            parent2 = tournament_selection(population, fitness, tournament_size)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, crossover_rate)
            
            # Mutation  
            child1 = mutate(child1, mutation_rate, 0.02)
            child2 = mutate(child2, mutation_rate, 0.02)
            
            # Normalize
            child1 = normalize_h(child1, target_sum)
            child2 = normalize_h(child2, target_sum)
            
            # Local optimization (simple gradient descent)
            def obj(h):
                return compute_c5(h, dx)
            
            # Simple optimization using gradient
            for _ in range(3):
                # Simple gradient estimate (central difference)
                grad = np.zeros_like(child1)
                eps = 1e-5
                for i in range(len(child1)):
                    h_plus = child1.copy()
                    h_minus = child1.copy()
                    h_plus[i] += eps
                    h_minus[i] -= eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_minus = np.clip(h_minus, 0, 1)
                    h_plus = normalize_h(h_plus, target_sum)
                    h_minus = normalize_h(h_minus, target_sum)
                    grad[i] = (obj(h_plus) - obj(h_minus)) / (2 * eps)
                    
                # Update (gradient ascent for minimizing, so move against grad)
                child1 -= 0.01 * grad
                child1 = np.clip(child1, 0, 1)
                child1 = normalize_h(child1, target_sum)
            
            new_population.append(child1.copy())
            if len(new_population) < pop_size:
                new_population.append(child2.copy())
        
        population = new_population[:pop_size]
        
        # Decrease mutation rate
        mutation_rate *= 0.995
        
    # Final evaluation of best solution
    final_fitness = compute_c5(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
