import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import random

def compute_c5(h, dx):
    """Compute the C5 value for given h."""
    overlap = correlate(h, 1 - h, mode="full")
    return np.max(overlap) * dx

def evaluate_solution(h, dx):
    """Evaluate the solution and ensure feasibility."""
    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Initialize step function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points)
        h[n_points//2:] = 0
        h = h * (target_sum / np.sum(h))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # CMA-ES parameters
    popsize = 2 * n_points
    lambda_ = popsize // 2
    sigma = 0.1
    mu = popsize // 2
    weights = np.log(mu + 1) - np.log(np.arange(1, mu + 1))
    weights = weights / np.sum(weights)
    
    # Evolution paths
    pc = np.zeros(n_points)
    ps = np.zeros(n_points)
    
    # Covariance matrix
    C = np.eye(n_points)
    inv_C = np.eye(n_points)
    
    # Main CMA-ES loop
    for iteration in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Sample offspring
        offspring = []
        for _ in range(popsize):
            # Sample from multivariate normal using C
            z = np.random.randn(n_points)
            x = h + sigma * np.dot(C, z)
            # Clip and normalize
            x = evaluate_solution(x, dx)
            offspring.append(x)
            
        # Evaluate offspring
        offspring_scores = []
        for x in offspring:
            score = compute_c5(x, dx)
            offspring_scores.append(score)
            
        # Sort by fitness
        indices = np.argsort(offspring_scores)
        best_indices = indices[:lambda_]
        
        # Update distribution parameters
        # Best individuals
        best_individuals = [offspring[i] for i in best_indices]
        # New mean (weighted average)
        new_h = np.zeros(n_points)
        for i, idx in enumerate(best_indices):
            new_h += weights[i] * offspring[idx]
        h = evaluate_solution(new_h, dx)
        
        # Update paths
        y = h - best_h
        ps = (1 - 0.3) * ps + np.sqrt(0.3 * (2 - 0.3)) * np.dot(inv_C, y) / sigma
        
        # Update covariance matrix
        pc = (1 - 1/5) * pc + np.sqrt(1/5 * (2 - 1/5)) * y / sigma
        C = (1 - 1/5) * C + 1/5 * np.outer(pc, pc) + 0.001 * np.eye(n_points)
        
        # Adapt sigma
        sigma = sigma * np.exp(0.3 * (np.linalg.norm(ps) / np.sqrt(n_points) - 1))
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Local SLSQP refinement
        if iteration % 3 == 0:
            bounds = [(0, 1) for _ in range(n_points)]
            def constraint_sum(h_v):
                return np.sum(h_v) - target_sum
                
            def objective(h_v):
                return compute_c5(h_v, dx)
                
            constraint = {'type': 'eq', 'fun': constraint_sum}
            result = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=constraint, options={'maxiter': 50})
            
            if result.success:
                h = result.x
                h = evaluate_solution(h, dx)
                c5 = compute_c5(h, dx)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    
        if best_c5 <= 0.3807:
            break

    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
