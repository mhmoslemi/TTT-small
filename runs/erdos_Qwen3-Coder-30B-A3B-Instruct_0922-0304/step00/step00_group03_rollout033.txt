import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_h(h, n_points):
    # Ensure the sum is exactly n_points / 2 by flipping bits
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum == target_sum:
        return h.copy()
    
    diff = int(target_sum - current_sum)
    if diff > 0:  # Need to flip more zeros to ones
        indices = np.where(h == 0)[0]
        flip_indices = indices[:diff]
    else:  # Need to flip more ones to zeros
        indices = np.where(h == 1)[0]
        flip_indices = indices[:abs(diff)]
    
    new_h = h.copy()
    for idx in flip_indices:
        new_h[idx] = 1 - new_h[idx]
    return new_h

def tabu_search(initial_h, n_points, dx, budget_s):
    start_time = time.time()
    max_iter = 1000
    tabu_tenure = 100
    tabu_list = []
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Initial repair to satisfy constraint
    best_h = repair_h(best_h, n_points)
    current_h = best_h.copy()
    current_c5 = evaluate_h(current_h, dx)
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = max_iter // 4
    
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        iteration += 1
        
        # Generate neighbors by flipping 2-3 bits
        neighbors = []
        for _ in range(20):  # Generate 20 neighbor solutions
            h_copy = current_h.copy()
            # Pick 2 or 3 random indices to flip
            num_flips = np.random.choice([2, 3])
            indices = np.random.choice(n_points, num_flips, replace=False)
            for idx in indices:
                h_copy[idx] = 1 - h_copy[idx]
            
            # Repair to ensure correct sum
            h_copy = repair_h(h_copy, n_points)
            neighbors.append(h_copy)
        
        # Evaluate neighbors, excluding those in tabu list
        best_neighbor = None
        best_neighbor_c5 = float('inf')
        for neighbor in neighbors:
            neighbor_tuple = tuple(neighbor)
            if neighbor_tuple in tabu_list:
                continue
            c5 = evaluate_h(neighbor, dx)
            if c5 < best_neighbor_c5:
                best_neighbor = neighbor
                best_neighbor_c5 = c5
        
        if best_neighbor is None:
            # All neighbors are in tabu list; continue to next iteration
            continue
        
        # Update current solution
        current_h = best_neighbor
        current_c5 = best_neighbor_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Add previous solution to tabu list
        tabu_list.append(tuple(current_h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
        
        # Early termination if no improvement
        if no_improvement_count >= max_no_improvement:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with a simple construction
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    h = repair_h(h, n_points)  # Ensure correct sum
    # Run tabu search
    h_opt, c5_bound = tabu_search(h, n_points, dx, budget_s)
    # Final repair
    h_opt = repair_h(h_opt, n_points)
    
    return (h_opt.tolist(), c5_bound, n_points)
