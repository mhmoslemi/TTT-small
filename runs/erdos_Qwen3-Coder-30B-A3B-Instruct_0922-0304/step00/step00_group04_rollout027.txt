import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    # Gradient descent
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        # The correlation has length 2*n_points - 1
        # The lags go from -(n_points-1) to (n_points-1)
        # Correlation at index k corresponds to lag k
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        k = max_k - (n_points - 1)
        
        # Compute gradient w.r.t. h for this specific k
        # We compute d/dh[i] h[i] * (1 - h[i+k])
        # For i:
        #   if k >= 0: grad[i] = 1 - h[i+k] 
        #   if k < 0:  grad[i] = 1 - h[i+k] + 1 - h[i+k] = 2 - h[i+k] - h[i+k] + 1 = 2 - 2*h[i+k] (this is wrong)
        # Actually, we have to be more careful:
        # The overlap integral for a fixed lag k is integral h(x) * (1 - h(x+k)) dx
        # So we just take the derivative w.r.t. h[i] of one such integral = (1-h[i+k]) dx
        # But that's for the form where we treat h as a continuous variable
    
        # Compute gradient more rigorously
        grad = np.zeros_like(h)
        if k >= 0:
            # h[i] contributes to integral of h[i] * (1 - h[i+k]) -> grad[i] = 1 - h[i+k] 
            # So for i in range(n_points - k), include value of 1 - h[i+k] 
            grad[:-k] = 1.0 - h[k:]  # this creates problem when k > 0, need careful indexing
        else:
            # k < 0, so h[i+k] term comes from h[i+k] term, which for h[i] is 1 - h[i+k]
            # So for i in range(-k, n_points), grad[i] = 1 - h[i+k]
            grad[-k:] = 1.0 - h[:-k]  # this creates problem when k < 0
        
        # Fix indexing for k
        grad = np.zeros_like(h)
        if k >= 0:
            # For i = 0 to len-h-k, the contribution is 1 - h[i+k]
            # We assign to grad[i] = 1 - h[i+k]
            grad[:-k] = 1 - h[k:]
        else:
            # For i = -k to len-1, the contribution is 1 - h[i+k]
            # Assign to grad[i] = 1 - h[i+k] 
            # i+k ranges from k to k+len-1 = -1 to len-2, which is i from k to len-1
            # i from 0 to len+k-1, to i+k from k to len+k-1. So i goes from -k to len-k-1, 
            # which is where grad[i] = 1 - h[i+k] 
            # More simply: grad[i] = 1 - h[i+k] when i+k is valid index to h (i+k in [0, n_points))
            # So when i in [0, n_points+k-1), i+k in [k, n_points-1] 
            # So when i from 0 to n_points + k - 1, index i+k is valid and should be:
            # i: [0, n_points+k-1)
            # i+k: [k, n_points-1]
            # grad[i] = 1 - h[i+k] for i in range(n_points + k) 
            # But when i+k >= 0 and i+k < n_points, which is 0 <= i+k < n_points, which means
            # -k <= i < n_points - k
            if k < 0:
                # i: [0, n_points + k -1), i+k: [k, n_points-1]
                # So i goes from 0 to n_points+k-1.  
                # valid i: i in [0, n_points+k-1] 
                # Valid i: i from 0 to n_points+k-1  
                # i+k in [k, k+n_points+k-1-1] = [k, 2*n_points-2+k] 
                # But we want i+k in [0,n_points) , so we need to specify the overlap:  
                # i+k >= 0 -> i >= -k  
                # i+k < n_points -> i < n_points - k  
                # So i in [max(0, -k), min(n_points-1, n_points+k-2)] = [max(0, -k), n_points+k-2]
                # But we need strict inequality: i in [max(0, -k), n_points+k-1)
                # So we use i from max(0, -k) to min(n_points-1, n_points+k-2) which simplifies to 
                # [max(0, -k), n_points+k-2] (including n_points+k-2) 
                # But also note we're using indices [0, n_points) and slicing with [k:] or [:-k]
                j_start = max(0, -k)
                j_end = min(n_points, n_points + k)
                indices = np.arange(j_start, j_end)
                j_indices = indices + k
                # So, grad[indices] = 1 - h[j_indices]
                grad[indices] = 1 - h[j_indices]    
        # Now we evaluate performance assuming we just negated a positive gradient to decrease the max_corr.
        
        # Update step with fixed step size
        learning_rate = 0.01
        h_new = h - learning_rate * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        # Fix sum constraint by rescaling
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Evaluate new C5 bound
        new_C5 = compute_C5(h_new, dx)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            
        h = h_new
        
    return (best_h, best_C5, n_points)

def compute_C5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
