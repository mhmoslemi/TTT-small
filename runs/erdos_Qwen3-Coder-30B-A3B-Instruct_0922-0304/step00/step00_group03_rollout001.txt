import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple construction
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    
    max_iter = 50000
    learning_rate = 0.01
    tolerance = 1e-6
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral for all shifts
        # Using convolution to get all cross-correlations
        h_extended = np.concatenate([h, np.zeros(n_points)])
        one_minus_h_extended = np.concatenate([1 - h, np.zeros(n_points)])
        cross_corr = np.correlate(h_extended, one_minus_h_extended, mode='full')
        # Take only the valid overlaps (from shift -n+1 to n-1)
        overlaps = cross_corr[n_points-1:2*n_points-1] * dx
        c5 = np.max(overlaps)
        
        if np.isnan(c5) or np.isinf(c5):
            break
            
        if abs(c5 - last_c5) < tolerance and iteration > 10:
            break
            
        last_c5 = c5
        
        # Find the shift with max overlap
        max_shift_idx = np.argmax(overlaps)
        k = max_shift_idx - (n_points - 1)
        
        # Compute gradient of the overlap w.r.t. h
        grad = np.zeros(n_points)
        if k >= 0:
            grad[k:] += (1 - h[:-k]) * dx
            grad[:k] -= h[k:] * dx
        else:
            k = -k
            grad[:-k] += (1 - h[k:]) * dx
            grad[-k:] -= h[:-k] * dx
        
        # Update h in the direction of negative gradient
        h_new = h - learning_rate * grad
        
        # Project to [0, 1] interval
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain the sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-8:
            # Scale to maintain desired sum
            h_new = h_new * target_sum / current_sum
        
        h = h_new
    
    # Final evaluation
    h_extended = np.concatenate([h, np.zeros(n_points)])
    one_minus_h_extended = np.concatenate([1 - h, np.zeros(n_points)])
    cross_corr = np.correlate(h_extended, one_minus_h_extended, mode='full')
    overlaps = cross_corr[n_points-1:2*n_points-1] * dx
    final_c5 = np.max(overlaps)
    
    return (h.tolist(), float(final_c5), n_points)
