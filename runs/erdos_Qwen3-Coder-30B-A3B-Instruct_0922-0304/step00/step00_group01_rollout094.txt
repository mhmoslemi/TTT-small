import numpy as np
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum, Norm
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Ensure sum constraint
    initial_h = initial_h * target_sum / np.sum(initial_h)
    
    # Best solution tracking
    best_h = initial_h.copy()
    best_C5 = evaluate_C5(initial_h, n_points, dx)
    
    # Initialize search from multiple random starts plus initial_h
    initial_starts = [initial_h.copy()]
    for _ in range(4):
        perturbed = initial_h + np.random.normal(0, 0.005, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * target_sum / np.sum(perturbed)
        initial_starts.append(perturbed)
    
    best_h = best_h.copy()
    best_C5 = evaluate_C5(best_h, n_points, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        return h
    
    # Optimization
    max_iter = 1000
    iter_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1:
            break
        
        current_h = best_h.copy()
        # Compute overlaps
        overlaps = np.correlate(current_h, 1 - current_h, mode='full')[n_points-1:]
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient
        shifted = np.roll(1 - current_h, k_max)
        gradient = shifted - current_h * shifted
        
        # Normalize gradient
        norm_grad = np.linalg.norm(gradient)
        if norm_grad == 0:
            break
        normalized_grad = gradient / norm_grad
        
        # Line search
        alpha = 0.01
        rho = 0.5
        max_steps = 30
        for _ in range(max_steps):
            trial_h = current_h - alpha * normalized_grad
            trial_h = project_h(trial_h)
            trial_C5 = evaluate_C5(trial_h, n_points, dx)
            if trial_C5 < best_C5:
                best_h = trial_h.copy()
                best_C5 = trial_C5
                break
            alpha *= rho
        
        # Convex refinement every 50 steps
        if iter_count % 50 == 0:
            try:
                # Convex optimization on the max overlap term
                h = Variable(n_points)
                constraints = [h >= 0, h <= 1, sum(h) == target_sum]
                prob = Problem(Minimize(sum(h * np.roll(1 - h, k_max))), constraints)
                # Idealy solve with a convex solver, but using scipy as fallback
                def conv_obj(h_vals):
                    return np.sum(h_vals * np.roll(1 - h_vals, k_max))
                bounds = [(0, 1) for _ in range(n_points)]
                result = minimize(conv_obj, current_h, bounds=bounds, method='L-BFGS-B')
                if result.success:
                    refined_h = project_h(result.x)
                    refined_C5 = evaluate_C5(refined_h, n_points, dx)
                    if refined_C5 < best_C5:
                        best_h = refined_h
                        best_C5 = refined_C5
            except:
                pass
                
        iter_count += 1
        
        # Check improvement
        current_C5 = evaluate_C5(best_h, n_points, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, n, dx):
    h = np.array(h)
    overlaps = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlaps[n-1:])
    return max_overlap * dx
