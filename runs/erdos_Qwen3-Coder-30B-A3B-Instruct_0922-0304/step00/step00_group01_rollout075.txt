import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    # Take the maximum of the last n_points elements (positive shifts)
    max_overlap = np.max(overlap[n_points-1:])
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy the constraints: 0<=h<=1 and sum(h)=n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Match initial construction
    max_iter = 10000
    k_targets = 5
    step_size = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Precompute indices for shifts
    shift_indices = [i for i in range(1, n_points)]
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s and iter_count < max_iter and no_improvement_count < max_no_improvement:
        iter_count += 1
        
        # Compute overlaps for all shifts
        overlap = correlate(h, 1 - h, mode='full')
        # Extract positive shifts (last n_points elements)
        overlap_values = overlap[n_points-1:]
        # Find indices of top k largest overlaps
        sorted_indices = np.argsort(overlap_values)[::-1][:k_targets]
        
        # Create a copy to modify
        h_new = h.copy()
        
        # For each target peak, try to reduce it
        for k_idx in sorted_indices:
            k = shift_indices[k_idx]
            # Try to reduce h[i] where h[i] and h[i+k] are involved in high overlap
            # Find all pairs (i, i+k) contributing to this overlap
            pairs = []
            for i in range(n_points - k):
                pairs.append((i, i + k))
            
            # Simple strategy: reduce one element in each critical pair
            # Considering the specific nature of the overlap, 
            # we approach it by changes that should reduce the maximum overlap
            # This is an aggressive but simple approach:
            # Reduce elements that contribute most to high overlap
            for i in range(n_points - k):
                # Target a small region around this pair
                i_range = range(max(0, i-1), min(n_points, i+2))
                for j in i_range:
                    if np.random.rand() < 0.3:  # Stochastic step
                        if h_new[j] > 0.01:  # only decrease if not already small
                            h_new[j] -= step_size * 0.5
            
        # Project
        h_new = project_h(h_new, n_points)
        
        # Test new solution
        new_c5 = evaluate_erdos_solution(h_new, n_points)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Accept current solution if it's better (or stochastic acceptance if not)
        h = h_new
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
