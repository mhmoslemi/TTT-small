import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Defaults
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 5000
    alpha = 0.01
    beta = 0.9
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points) * 0.5 + 0.25
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        # We compute all correlations for all shifts
        corr_full = correlate(h, 1 - h, mode='full')
        # Shift correction to get actual indices
        n_corr = len(corr_full)
        shift_indices = np.arange(-(n_points - 1), n_points)
        corr_vals = corr_full
        max_corr_idx = np.argmax(corr_vals)
        max_corr_shift = shift_indices[max_corr_idx]
        max_corr_val = corr_vals[max_corr_idx]
        
        # Construct gradient
        grad = np.zeros_like(h)
        grad_corr = correlate(1 - h, np.ones_like(h), mode='full')
        # Only relevant for the specific largest overlap
        # But the optimization perspective is unstable due to non-convexity
        # Apply direct correction considering the max overlap
        # Specifically, we know that max overlap is formed by terms h[i] * (1 - h[i+k])
        # So gradient w.r.t h[i] is 1 - h[i+k] for all relevant k, where that term contributes
        
        # Recompute gradient based on the specific maximum overlap
        # For the case max_corr_shift=k, terms contributing to the gradient are:
        # For i=0 to (n_points - k - 1): h[i] * (1 - h[i+k]) contributes
        # So d/dh[i] = (1 - h[i+k]) & d/dh[i+k] = -h[i]
        # Gradient follows from influence of h[i] on max_k integral
        # Direct gradient computation
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + max_corr_shift >= 0 and i + max_corr_shift < n_points:
                grad[i] += 1 - h[i + max_corr_shift]
                if i < n_points - 1:
                    # Sum not directly needed, per gradient computation
                    pass  # already applied above
        
        # Apply momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - alpha * momentum
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Update h
        h = h_new
        
        # Update best solution if better
        if max_corr_val < best_c5:
            best_c5 = max_corr_val
            best_h = h.copy()
        
        # Early termination check
        if best_c5 <= target_bound:
            break
    
    # Final evaluation
    corr_full = correlate(best_h, 1 - best_h, mode='full')
    n_corr = len(corr_full)
    shift_indices = np.arange(-(n_points - 1), n_points)
    max_corr_val = np.max(corr_full)
    return (best_h.tolist(), max_corr_val, n_points)
