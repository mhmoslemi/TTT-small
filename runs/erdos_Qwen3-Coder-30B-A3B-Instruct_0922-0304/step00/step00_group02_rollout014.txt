import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        iter_count += 1
        current_time = time.time()
        
        # Compute overlaps for current h
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        
        # Objective: minimize max overlap
        max_var = cp.Variable()
        constraints.append(max_var >= overlaps)
        
        prob = cp.Problem(cp.Minimize(max_var), constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            relaxed_h = h_relaxed.value
        except:
            # Fallback to previous h if optimization fails
            relaxed_h = h.copy()
        
        # Project back to feasible binary set
        # Threshold
        proj_h = (relaxed_h > 0.5).astype(float)
        
        # Adjust to equal sum
        current_sum = np.sum(proj_h)
        target_sum = n_points / 2
        
        if current_sum != target_sum:
            # Use assignment method for accurate projection
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Need more 1s
                indices = np.argsort(-relaxed_h)  # descending
                for i in indices:
                    if diff <= 0:
                        break
                    if proj_h[i] == 0:
                        proj_h[i] = 1
                        diff -= 1
            elif diff < 0:
                # Need fewer 1s
                indices = np.argsort(relaxed_h)  # ascending
                for i in indices:
                    if diff >= 0:
                        break
                    if proj_h[i] == 1:
                        proj_h[i] = 0
                        diff += 1
                        
        h = proj_h
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h)
        
        if new_C5 < best_C5:
            best_h = h.copy()
            best_C5 = new_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Add small random perturbation to escape local minima
        if np.random.rand() < 0.1:
            perturb_indices = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            for i in perturb_indices:
                h[i] = 1 - h[i]
    
    return (best_h.tolist(), best_C5, n_points)
