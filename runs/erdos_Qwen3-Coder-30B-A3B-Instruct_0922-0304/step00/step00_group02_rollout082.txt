import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    # Compute C5 using correlation of h with (1-h)
    # We use full mode to get all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take maximum correlation (over all shifts) and scale
    c5 = np.max(corr) / len(h)
    return c5

        h[ones_indices] = 1.0
    
    # Ensure sum constraint
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    h = np.clip(h, 0, 1)
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

        max_corr_idx = np.argmax(corr_full)

        for i in range(n_points):
