import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute overlap integral
        corr = signal.correlate(h, 1 - h, mode='full')
        # Maximum over all shifts
        c5_val = np.max(corr) * dx
        return c5_val
    
    # Initial computation
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    delta = 0.1
    n_perturb = max(1, int(0.05 * n_points))
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Perturb some elements
        idxs = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Store old values and compute old energy
        old_h = h.copy()
        old_c5 = best_c5
        
        # Perturb chosen indices
        for i in idxs:
            # Random step
            step = np.random.uniform(-delta, delta)
            h[i] += step
            # Clip to [0, 1]
            h[i] = np.clip(h[i], 0, 1)
        
        # Renormalize to sum to n_points / 2
        h_sum = np.sum(h)
        if h_sum > 0:
            h *= (n_points / 2) / h_sum
        
        # Compute new c5
        new_c5 = compute_c5(h)
        
        # Accept or reject
        if new_c5 < old_c5 or np.random.rand() < np.exp(-(new_c5 - old_c5) / temp):
            best_c5 = new_c5
            best_h = h.copy()
        else:
            h = old_h
        
        # Cool down
        temp *= cooling_rate
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
