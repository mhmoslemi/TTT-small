import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    max_time = time.time() + budget_s
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create initial construction
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0  # First quarter ones
        h[n_points//4:3*n_points//4] = 0.0  # Middle half zeros  
        h[3*n_points//4:] = 1.0  # Last quarter ones
        # Adjust to make sum = n_points/2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        if diff != 0:
            # Distribute difference evenly
            h += diff / n_points
        
    # Make sure it's within bounds
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    h *= (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    def compute_c5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # The maximum correlation gives us the overlap
        c5_val = np.max(np.abs(corr)) * dx
        return c5_val
    
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T0 = 1.0
    T_min = 1e-4
    n_iterations = 10000
    alpha = np.exp(np.log(T_min / T0) / n_iterations) if n_iterations > 0 else 1
    T = T0
    delta_scale = 0.1
    accept_rate = 0.5
    accept_count = 0
    total_count = 0
    
    def perturb_h(h_vec, delta):
        # Create a copy to avoid changing the original
        new_h = h_vec.copy()
        n = len(new_h)
        
        # Try perturbation type 1: shift adjacent points
        if random.random() < 0.5 or n < 5:
            i1 = random.randint(0, n-1)
            i2 = (i1 + 1) % n
            # Limit change
            d = min(delta, 1.0 - new_h[i1], new_h[i2])
            if d > 1e-8:
                change = np.random.uniform(0, d)
                new_h[i1] += change
                new_h[i2] -= change
        else:
            # Perturbation type 2: random swap (lower probability)
            i1 = random.randint(0, n-1)
            i2 = random.randint(0, n-1)
            while i1 == i2:
                i2 = random.randint(0, n-1)
            
            # Compute max possible change
            max_change = min(new_h[i1], 1.0 - new_h[i2])
            if max_change > 1e-8:
                change = np.random.uniform(0, max_change)
                new_h[i1] -= change
                new_h[i2] += change
                
        # Ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Enforce sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust all values to match sum constraint
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # Distribute difference evenly
                new_h += diff / n_points

        return new_h
    
    # Main optimization loop
    iteration = 0
    while time.time() < max_time and iteration < n_iterations:
        # Perturb
        new_h = perturb_h(h, delta_scale)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h)
        
        # Metropolis acceptance criterion
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update metrics
        total_count += 1
        if delta_c5 <= 0:
            accept_count += 1
            
        # Adjust temperature
        T *= alpha
        
        # Adjust perturbation scale
        if total_count > 0 and accept_count / total_count < 0.3:
            delta_scale *= 0.9
        elif total_count > 0 and accept_count / total_count > 0.7:
            delta_scale *= 1.1
            
        iteration += 1
        
        if iteration % 1000 == 0:
            if time.time() > max_time - 5:  # rough time check
                break
                
    return (best_h.tolist(), float(best_c5), len(best_h))
