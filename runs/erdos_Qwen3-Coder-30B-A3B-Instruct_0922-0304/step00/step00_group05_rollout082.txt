import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Default construction: balance between 0s and 1s with intermediate values
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//4] = 1.0
        h[n_points//4:n_points//2] = 0.0
        
    # Normalize to sum = n_points / 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Calculate initial value
    dx = 2.0 / n_points
    C5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    best_C5 = C5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    n_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / n_iter)
    T = T_max
    delta = 0.1
    accept_count = 0
    target_accept_rate = 0.4
    max_accept_count = 100
    
    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try perturbing h
        h_new = h.copy()
        if np.random.random() < 0.5:
            # Swap two points
            idx1 = np.random.randint(n_points)
            idx2 = np.random.randint(n_points)
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        else:
            # Shift mass between adjacent points
            idx1 = np.random.randint(n_points-1)
            idx2 = idx1 + 1
            delta_change = np.random.uniform(-delta, delta)
            if h_new[idx1] + delta_change >= 0 and h_new[idx2] - delta_change >= 0 \
               and h_new[idx1] + delta_change <= 1 and h_new[idx2] - delta_change <= 1:
                h_new[idx1] += delta_change
                h_new[idx2] -= delta_change
        
        # Rescale to maintain constraint
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Compute new C5
        C5_new = np.max(correlate(h_new, 1 - h_new, mode='full')) * dx
        
        # Accept or reject
        delta_C = C5_new - C5
        if delta_C < 0 or np.random.random() < np.exp(-delta_C / T):
            h = h_new
            C5 = C5_new
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
            accept_count += 1
        else:
            accept_count = max(0, accept_count - 1)
            
        # Adaptive temperature and delta
        T *= alpha
        if accept_count >= max_accept_count:
            delta *= 1.1
            accept_count = 0
        elif accept_count <= 0 and delta > 1e-5:
            delta *= 0.9
            accept_count = 0
    
    return (best_h, best_C5, n_points)
