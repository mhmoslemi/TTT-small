import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Fallback: create a balanced step function
        n_points = 70
        h = np.ones(n_points // 2) * 0.5
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    T = T_max
    delta = 0.1
    accept_count = 0
    accept_rate = 0.5
    
    def rescale_h(h):
        h_sum = np.sum(h)
        target_sum = n_points / 2.0
        if h_sum > 0:
            h = h * target_sum / h_sum
        h = np.clip(h, 0, 1)
        return h
    
    def mutate_h(h, delta):
        h_new = h.copy()
        # Random swap between two points
        i, j = np.random.choice(n_points, 2, replace=False)
        diff = h_new[i] - h_new[j]
        change = np.random.rand() * delta
        change = min(change, abs(diff))
        if diff > 0:
            h_new[i] -= change
            h_new[j] += change
        else:
            h_new[i] += change
            h_new[j] -= change
        return rescale_h(h_new)
    
    def get_c5(h):
        # Compute correlation using scipy for fast evaluation
        corr = correlate(h, 1 - h, mode='full')
        # The maximum overlap is at the center of the correlation
        max_idx = len(corr) // 2
        return np.max(np.abs(corr)) * dx
    
    # Main simulated annealing loop
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        h_new = mutate_h(h, delta)
        c5_new = get_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        delta_c5 = c5_new - get_c5(h)
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            if delta_c5 < 0:
                accept_count += 1
        
        # Adapt temperature
        T *= alpha
        if T < T_min:
            T = T_min
            
        # Adapt delta based on acceptance rate
        if i > 0 and i % 100 == 0:
            accept_rate = accept_count / 100.0
            if accept_rate < 0.1:
                delta *= 0.9
            elif accept_rate > 0.3:
                delta *= 1.1
            accept_count = 0
    
    return (best_h, best_c5, n_points)
