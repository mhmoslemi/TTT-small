import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate C5 for given h_values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for all shifts
    overlaps = correlate(h, 1 - h, mode='full')
    # The overlap integral for shift k is just the k-th element of the correlation
    # But we need to be careful with indexing
    # For shift k, the overlap is sum over i of h[i] * (1 - h[(i+k)%n])
    # This corresponds to overlap[ n-1 + k ] for k from 0 to 2*n-2
    # But our convention maps it to overlap_k = overlaps[n-1+k]
    overlap_values = overlaps[n-1:]
    return np.max(overlap_values) * dx

def initial_h_values():
    return [0.5] * 45

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial h if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values())
    else:
        h = np.ones(45) * 0.5
    n = len(h)
    dx = 2.0 / n
    
    # Renormalize to integral = 1
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)  # Ensure exactly right sum
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    # Variables
    h_vars = cp.Variable(n)
    
    # Constraint to enforce integral = 1
    integral_cons = cp.Equality(cp.sum(h_vars), n / 2)
    
    # Bounds
    bounds = [h_vars >= 0, h_vars <= 1]
    
    # Start optimization loop
    iters = 0
    max_iters = 1000
    no_improvement = 0
    last_C5 = best_C5 + 1.0
    
    while time.time() - start_time < budget_s * 0.95 and iters < max_iters:
        iters += 1
        
        # Compute current overlaps using numpy
        overlaps = correlate(h, 1 - h, mode='full')
        overlap_values = overlaps[n-1:]
        k_max = np.argmax(overlap_values)
        max_overlap_val = overlap_values[k_max]
        
        # Create variable for C5
        C5_var = cp.Variable()
        
        # Create optimization problem
        prob = cp.Problem(
            cp.Minimize(C5_var),
            bounds + [integral_cons] + [cp.sum(cp.multiply(h_vars, 1 - cp.roll(h_vars, k_max))) <= C5_var]
        )
        
        # Try solving the problem
        try:
            result = prob.solve(solver=cp.ECOS, verbose=False)
        except:
            # Fallback or continue
            break
            
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_new = np.array(h_vars.value).reshape(-1)
            h_new = np.clip(h_new, 0, 1)
            # Renormalize exactly
            h_new = h_new * (n / 2) / np.sum(h_new)
            
            C5_new = evaluate_erdos_solution(h_new)
            
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
                no_improvement = 0
            else:
                no_improvement += 1
                
        # Check termination conditions
        if best_C5 < 0.3807:
            break
        if no_improvement > 50:
            break
            
        h = h_new.copy()

    return (list(best_h), best_C5, n)
