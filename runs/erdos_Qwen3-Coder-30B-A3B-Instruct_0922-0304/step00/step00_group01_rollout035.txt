def evaluate_erdos_solution(h_values):
    """Compute C5 bound for given h_values."""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all overlaps using correlation
    h = np.array(h_values)
    correlation = np.correlate(h, 1 - h, mode='full')
    # Extract max overlap for shifts 0 to n_points-1
    overlaps = correlation[n_points-1:] * dx
    return np.max(overlaps)

            h_new = h - alpha * gradient

            # Rescale to correct sum
