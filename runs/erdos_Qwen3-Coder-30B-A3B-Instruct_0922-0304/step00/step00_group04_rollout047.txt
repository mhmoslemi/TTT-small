import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """
    Evaluate the C_5 bound given a step function h.
    """
    # Compute correlation with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # Multiply by dx and return max
    return np.max(corr * dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create basic construction
        n_points = 42
        h = np.ones(n_points) * 0.5  # constant half
        h[0] = 0.0  # adjust first point
        h[1] = 0.0  # adjust second point
        h[-1] = 1.0  # adjust last point
        h[-2] = 1.0  # adjust second to last point
    
    # Ensure initial sum constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    assert abs(np.sum(h) - n_points / 2) < 1e-10, "Initial sum not correct"
    
    # Convex relaxation approach with CVXPY
    def conv_optimize_h(h_in, n_samples):
        # Relax the optimization to a convex problem
        h_relaxed = cp.Variable(n_samples)
        # Set bounds
        constraints = [h_relaxed >= 0, h_relaxed <= 1]
        # Sum constraint
        constraints.append(cp.sum(h_relaxed) == n_samples / 2)
        
        # Convex relaxation of overlapping correlations
        # We approximate the problem by modeling the constraints
        # Using the correlation estimation with relaxation
        objective = cp.Minimize(cp.sum(h_relaxed))  # dummy objective
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return np.array(h_relaxed.value)
        except:
            pass
        
        return h_in
    
    # Initial convex optimization
    h = conv_optimize_h(h, n_points)
    
    # Gradient descent refinement
    def gradient_descent(h_in, n_iter=100):
        h_current = np.copy(h_in)
        dx = 2.0 / len(h_current)
        
        for i in range(n_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute all overlaps
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            max_k = np.argmax(corr)
            max_val = corr[max_k]
            
            # Compute atomic gradient
            # finite differences for gradient estimation
            h_grad = np.zeros_like(h_current)
            eps = 1e-5
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                h_grad[j] = (corr_plus[max_k] - max_val) / eps

            # Update using gradient descent
            learning_rate = 0.1 if i < 50 else 0.01
            h_current = h_current - learning_rate * h_grad
            
            # Project back into [0,1]
            h_current = np.clip(h_current, 0, 1)
            
            # Rescale to meet sum constraint
            current_sum = np.sum(h_current)
            h_current = h_current * (n_points / 2) / current_sum
            
        return h_current
    
    # Perform gradient descent
    h = gradient_descent(h, 50)
    
    # Possibly increase resolution to 84 points for fine-tuning
    if n_points < 84 and time.time() - start_time < budget_s - 5:
        h_new = np.interp(np.linspace(0, 2, 84), np.linspace(0, 2, n_points), h)
        h = h_new
        n_points = 84
        dx = 2.0 / 84
        h = conv_optimize_h(h, 84)
        h = gradient_descent(h, 100)
    
    # Final validation
    if time.time() - start_time < budget_s - 1:
        # Compute final C_5
        c5_bound = evaluate_erdos_solution(h, dx)
    else:
        c5_bound = evaluate_erdos_solution(h, dx)
    
    return (h.tolist(), float(c5_bound), n_points)
