import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h)
    except:
        # Fallback: symmetric construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add some variation
        h[::2] = 0.6
        h[1::2] = 0.4
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper to compute C5
    def compute_c5(h_vals):
        h_arr = np.array(h_vals)
        overlaps = np.correlate(h_arr, 1 - h_arr, mode='full')
        return np.max(overlaps) * (2.0 / n_points)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Time budget check
    def check_time():
        return time.time() - start_time < budget_s * 0.95
    
    # Gradient-based update for one shift
    def optimize_shift(h_vals, k_max):
        h_arr = np.array(h_vals)
        n = len(h_arr)
        
        # Function to compute overlap for shift k_max
        def overlap_fn(h_vec):
            h_shifted = np.roll(h_vec, k_max)
            return np.dot(h_vec, 1 - h_shifted)
        
        # Objective to minimize overlap(k_max)
        def obj(h_vec):
            return -overlap_fn(h_vec)  # Negative because we minimize
        
        # Constraints
        cons = [
            {'type': 'eq', 'fun': lambda h_vec: np.sum(h_vec) - n / 2},
            {'type': 'ineq', 'fun': lambda h_vec: h_vec},
            {'type': 'ineq', 'fun': lambda h_vec: 1 - h_vec}
        ]
        
        # Bounds
        bounds = [(0, 1)] * n
        
        # Optimize
        res = minimize(obj, h_arr.copy(), method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100})
        if res.success:
            return np.clip(res.x, 0, 1)
        return h_arr
    
    iter_count = 0
    max_iter = 1000

    while check_time() and iter_count < max_iter:
        iter_count += 1
        # Compute all overlaps
        h_arr = np.array(h)
        overlaps = np.correlate(h_arr, 1 - h_arr, mode='full')
        dx = 2.0 / n_points
        overlaps *= dx
        k_max = np.argmax(overlaps)
        new_c5 = overlaps[k_max]
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break
        
        # Update with optimization for max overlap shift
        h_new = optimize_shift(h, k_max)
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
