import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a symmetric step function
        n_points = 45
        h = np.ones(n_points) * 0.5
        # Make it a step function
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Ensure sum constraint
        h *= n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    
    # Parameters for optimization
    max_iter = 100000
    delta_base = 0.1
    delta_decay = 0.9
    delta_min = 1e-6
    iter_decay = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Adjust delta
        delta = max(delta_base * (delta_decay ** (iteration // iter_decay)), delta_min)
        
        # Select two indices for modification
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Make sure i and j are different
        if i == j:
            j = (j + 1) % n_points
            
        # Try perturbing
        h_new = h.copy()
        delta_i = delta * (0.5 - np.random.random())
        delta_j = -delta_i
        
        h_new[i] += delta_i
        h_new[j] += delta_j
        
        # Ensure bounds
        h_new[i] = np.clip(h_new[i], 0.0, 1.0)
        h_new[j] = np.clip(h_new[j], 0.0, 1.0)
        
        # Correct for sum constraint (maintain sum == n_points/2)
        correction = (n_points/2 - np.sum(h_new)) / 2
        if abs(correction) > 1e-10:
            h_new[i] += correction
            h_new[j] += correction
            # Clamp again
            h_new[i] = np.clip(h_new[i], 0.0, 1.0)
            h_new[j] = np.clip(h_new[j], 0.0, 1.0)
            
        # Compute new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        elif np.random.random() < 0.01:  # Small probability to accept worse
            h = h_new
            
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    #   C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # which corresponds to max of correlation between h and (1-h)
    n = len(h)
    
    # Create 1 - h
    h_complement = 1 - h
    
    # Compute full correlation
    correlation = correlate(h, h_complement, mode='full')
    
    # The overlap integral for k in [-n+1, n-1]
    # Each element of correlation corresponds to a k value
    # k ranges from -n+1 to n-1, so index k+n-1 in correlation corresponds to k
    max_correlation = np.max(correlation)
    
    # Account for dx
    # Correlation is a sum of products, need to multiply by dx
    # The indexing places k=0 at index n-1
    # So k is in range [-n+1, n-1] = indices [0, 2*n-2]
    return max_correlation * dx
