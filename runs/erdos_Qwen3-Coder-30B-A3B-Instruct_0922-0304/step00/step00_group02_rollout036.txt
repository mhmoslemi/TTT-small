import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                break
                
        # Compute overlaps for current h
        overlaps = correlate(h, 1 - h, mode="full") * dx
        
        # Formulate and solve convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        
        # Objective: min max_overlaps
        max_overlap = cp.Variable()
        constraints.append(max_overlap >= overlaps[i] for i in range(len(overlaps)))
        
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                relaxed_solution = h_relaxed.value
            else:
                # fallback to projection
                relaxed_solution = h
        except:
            # fallback to projection
            relaxed_solution = h
            
        # Project back to feasible binary set
        # Keep top half of values as 1s, rest as 0s
        sorted_indices = np.argsort(relaxed_solution)[::-1]
        new_h = np.zeros(n_points)
        # Set top n_points//2 elements to 1
        new_h[sorted_indices[:n_points//2]] = 1.0
        
        # Adjust if the sum is not exactly n_points/2
        current_sum = np.sum(new_h)
        if current_sum < n_points / 2:
            # Add more 1s
            zeros_indices = np.where(new_h == 0)[0]
            for i in range(int(n_points/2 - current_sum)):
                if i < len(zeros_indices):
                    new_h[zeros_indices[i]] = 1.0
        elif current_sum > n_points / 2:
            # Remove some 1s
            ones_indices = np.where(new_h == 1)[0]
            for i in range(int(current_sum - n_points / 2)):
                if i < len(ones_indices):
                    new_h[ones_indices[i]] = 0.0
        
        h = new_h.copy()
        
        # Perturb a bit to explore
        perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for idx in perturb_indices:
            if np.random.rand() < 0.5:
                h[idx] = 1 - h[idx]  # flip
                
        # Ensure sum constraint
        while abs(np.sum(h) - n_points / 2) > 1e-8:
            # Bring sum to target exactly
            current_sum = np.sum(h)
            if current_sum > n_points / 2:
                # Find largest 1 and turn into 0
                idx = np.argmax(h)
                if h[idx] == 1:
                    h[idx] = 0
            else:
                # Find smallest 0 and turn into 1
                idx = np.argmin(h)
                if h[idx] == 0:
                    h[idx] = 1

    return (best_h, best_C5, n_points)
