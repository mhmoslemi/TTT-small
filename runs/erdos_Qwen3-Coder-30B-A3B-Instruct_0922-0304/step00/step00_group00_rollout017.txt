import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    n = len(h)
    overlaps = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlaps) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = 1000
    patience = 50
    patience_counter = 0
    last_c5 = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        overlap_val = overlaps[k_max]
        
        # Compute gradient for shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
            
        # Update h using gradient descent
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if improvement is made
        current_c5 = evaluate_erdos_solution(h_new, dx)
        if current_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                alpha *= 0.5
                patience_counter = 0
                
        h = h_new
        
    # Final convex refinement if needed
    if time.time() - start_time < budget_s * 0.95:
        # Look for worst shift again and try convex refinement
        overlaps = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(overlaps)
        w = Variable(n_points)
        obj = cp.Minimize(cp.sum(w))
        constraints = [0 <= w[i] for i in range(n_points)] + [w[i] <= 1 for i in range(n_points)]
        # Add constraint for the worst shift
        total = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            total += best_h[i] * (1 - best_h[j])
        # Ensure no worse overlap than current
        constraints.append(total <= best_c5 / dx)
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == "optimal":
                refined_h = np.array(w.value).clip(0, 1)
                refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
                final_c5 = evaluate_erdos_solution(refined_h, dx)
                if final_c5 < best_c5:
                    best_h = refined_h
                    best_c5 = final_c5
        except Exception:
            pass
            
    # Final validation check
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
