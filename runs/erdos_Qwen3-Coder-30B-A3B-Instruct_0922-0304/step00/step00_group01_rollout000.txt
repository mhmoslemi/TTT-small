import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 47
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration - last_improvement < 100:
        iteration += 1
        
        # Compute overlap
        overlap = correlate(h, 1-h, mode="full")
        overlap = overlap[n_points-1:] * dx
        
        # Identify top 5 peaks
        peaks = np.argsort(overlap)[::-1][:5]
        
        # Update h to reduce these peaks
        h_new = h.copy()
        update_step = 0.05
        
        for k in peaks:
            if k >= n_points:
                continue
            for i in range(n_points):
                if i + k < n_points:
                    # Reduce h[i] to decrease overlap
                    if h_new[i] > 0.01:
                        h_new[i] = max(0, h_new[i] - update_step)
                        h_new[i+k] = max(0, h_new[i+k] - update_step)
        
        # Project h back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iteration
        
        h = h_new.copy()
        
        if best_c5 < target_c5:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the overlap integral C5
    overlap = correlate(h, 1-h, mode="full")
   overlap = overlap[len(h)-1:] * dx
    return np.max(overlap)
