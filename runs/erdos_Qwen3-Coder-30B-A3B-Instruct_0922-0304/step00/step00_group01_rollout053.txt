import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values, n_points):
    """Evaluate the C5 bound for given h_values"""
    dx = 2.0 / n_points
    # Compute overlap integral for all lag values
    overlap = signal.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Take only the positive lags (the meaningful ones for C5)
    overlap_positive = overlap[n_points-1:]
    c5 = np.max(overlap_positive) * dx
    return c5
