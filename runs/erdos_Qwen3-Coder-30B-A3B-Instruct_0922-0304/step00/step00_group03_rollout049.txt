import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values or create from scratch
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        
    # Ensure sum(h) = n_points / 2
    h = h * (len(h) / 2) / np.sum(h)
    
    # Parameters
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 1000
    tol = 1e-4
    learning_rate = 1e-2
    entropy_reg = 1e-4
    rho = 1e-3
    beta_decay = 0.9
    beta = entropy_reg
    
    # ADMM variables
    lambda_k = np.zeros(2*n_points - 1)  # overlap constraints
    mu = 0.0  # integral constraint
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Compute initial overlaps
    h_shifted = 1 - h
    overlaps = correlate(h, h_shifted, mode='full') * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h using convex optimization
        # min_h (h * (1-h))[k] * dx + sum_k(lambda_k * (C5 - (h * (1-h))[k])) + mu * (sum(h) - 1) + (rho/2) * ||h||^2
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.max(cp.multiply(lambda_k, overlaps)) +
            mu * (cp.sum(h_var) - 1) +
            (rho/2) * cp.sum_squares(h_var)
        )
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == 1.0 / dx  # normalized constraint
        ]
        
        # Solve convex subproblem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            new_h = h_var.value
            if new_h is not None:
                h = new_h
        except:
            pass
            
        # Recompute overlaps
        h_shifted = 1 - h
        overlaps = correlate(h, h_shifted, mode='full') * dx
        c5 = np.max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Update dual variables
        lambda_k += learning_rate * (c5 - overlaps)
        
        # Apply entropy regularization
        lambda_k = lambda_k + beta * np.log(1 + lambda_k)
        lambda_k = np.clip(lambda_k, -100, 100)
        
        mu += learning_rate * (np.sum(h) - 1)
        
        # Adapt entropy regularization if needed
        if iteration % 100 == 0 and iteration > 0:
            beta *= beta_decay
            
    # Final evaluation
    h_shifted = 1 - best_h
    overlaps = correlate(best_h, h_shifted, mode='full') * (2.0 / len(best_h))
    c5_bound = np.max(overlaps)
    
    return (best_h.tolist(), c5_bound, len(best_h))
