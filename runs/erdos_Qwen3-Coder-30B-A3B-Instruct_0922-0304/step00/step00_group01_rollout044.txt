import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h_vals):
    """Evaluate the C5 bound for given h values"""
    n = len(h_vals)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = np.correlate(h_vals, 1 - h_vals, mode="full")
    # Take only relevant overlaps (the ones that make sense)
    overlaps = overlaps[n-1:]
    # Scale by dx
    overlaps *= dx
    return np.max(overlaps)

def project_to_feasible(h, n_points):
    """Ensure h is within [0,1] and sum equals n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    max_iter = 1000
    refine_freq = 50
    min_improvement = 1e-6
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5)
    
    # Make sum correct
    h_current = project_to_feasible(h_current, n_points)
    
    best_c5 = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    
    # Optimize loop
    for it in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = np.correlate(h_current, 1 - h_current, mode="full")
        overlaps = overlaps[n-1:] * dx
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max]
        
        # If improved, save best
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            
        # Compute gradient
        # We want to minimize sum over all k of int h(x)(1 - h(x+k)) dx
        # Which is sum over i of h[i] * (1 - h[(i-k)%n]) for a fixed k
        # The gradient w.r.t. h[i] is (1 - h[(i-k)%n]) - h[i] * (1 - h[(i-k)%n])
        # But that's wrong. Let's recompute properly.
        shifted_h = np.roll(1 - h_current, k_max)
        gradient = shifted_h - h_current * shifted_h
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-10:
            continue
            
        # Backtracking line search
        alpha = 0.01
        rho = 0.5
        max_backtrack = 30
        
        for _ in range(max_backtrack):
            h_new = h_current - alpha * gradient / grad_norm
            h_new = project_to_feasible(h_new, n_points)
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < c5_current - min_improvement:
                h_current = h_new
                break
            else:
                alpha *= rho
                
        # Convex refinement step
        if (it + 1) % refine_freq == 0:
            # Minimize overlap for the current worst shift
            # Create CVXPY variables
            h = cp.Variable(n_points)
            # Objective: minimize sum h[i] * (1 - h[(i-k_max) % n_points])
            objective = cp.Minimize(
                cp.sum(cp.multiply(h, 1 - cp.hstack([
                    h[k_max:] if k_max > 0 else h,
                    h[:k_max] if k_max > 0 else []
                ])))
            )
            
            # Constraints
            constraints = [
                h >= 0,
                h <= 1,
                cp.sum(h) == n_points / 2
            ]
            
            # Solve
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-4)
                if prob.status == cp.OPTIMAL:
                    h_opt = h.value
                    h_current = project_to_feasible(h_opt, n_points)
            except:
                pass
                
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
