def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # compute all overlaps for shifts k from -n+1 to n-1
    h_extended = np.concatenate([h, h])
    overlaps = []
    for k in range(-n+1, n):
        shift = k if k >= 0 else n + k
        overlap = np.sum(h * h_extended[shift:shift+n]) * dx
        overlaps.append(overlap)
    return max(overlaps)
