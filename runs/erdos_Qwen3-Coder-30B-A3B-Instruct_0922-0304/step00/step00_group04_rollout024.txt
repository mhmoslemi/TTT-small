def evaluate_erdos_solution(h):
    # Compute the maximum overlap integral
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode='full')

    # The lags go from -n+1 to n-1
    # We are interested only in the valid lags 0 to 2*n-2
    # But empirically, the maximum value in the middle segment
    max_val = np.max(corr)
    return max_val * dx
