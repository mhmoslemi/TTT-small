import numpy as np
import scipy.optimize as opt
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 47
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)  # Normalize to sum = n_points/2
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    # Track best solution
    best_h, best_c5 = h.copy(), float('inf')
    
    # Helper to evaluate C5
    def evaluate_C5(h_vec):
        # Pad h for correlation
        h_padded = np.concatenate([h_vec, h_vec])
        # Compute all overlaps
        overlaps = np.correlate(h_padded, 1 - h_vec, mode='full')
        max_overlap = np.max(overlaps[n_points-1:])
        return max_overlap * dx
    
    # Line search
    def line_search(h_vec, grad_vec, max_iter=30, rho=0.5, alpha_init=0.01):
        alpha = alpha_init
        grad_norm = np.linalg.norm(grad_vec)
        if grad_norm == 0:
            return h_vec.copy()
        normalized_grad = grad_vec / grad_norm
        
        for _ in range(max_iter):
            new_h = h_vec - alpha * normalized_grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            if evaluate_C5(new_h) < evaluate_C5(h_vec):
                return new_h
            alpha *= rho
        return h_vec.copy()
    
    # Convex refinement
    def convex_refinement(h_vec, k_max):
        h = Variable(n_points)
        constraints = [0 <= h[i] <= 1 for i in range(n_points)]
        constraints.append(sum(h) == n_points / 2)
        
        # Objective: minimize the k_max overlap
        shifted_h = [h_vec[(i - k_max) % n_points] for i in range(n_points)]
        objective = sum(h[i] * (1 - shifted_h[i]) for i in range(n_points))
        
        prob = Problem(Minimize(objective), constraints)
        try:
            prob.solve(solver=SCS, verbose=False)
            if prob.status == 'optimal':
                return np.array([h[i].value for i in range(n_points)])
        except:
            pass
        return h_vec.copy()
    
    # Main optimization loop
    start_time = time.time()
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps and find worst shift
        h_padded = np.concatenate([h, h])
        overlaps = np.correlate(h_padded, 1 - h, mode='full')
        k_max = np.argmax(overlaps[n_points-1:])  
        current_c5 = overlaps[k_max] * dx
        
        # Update best if improved
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Compute gradient
        shifted_h = np.roll(1 - h, k_max)
        gradient = shifted_h - h * shifted_h
        
        # Apply line search
        h = line_search(h, gradient)
        
        # Periodic convex refinement: every 50 iterations
        if iteration % 50 == 0:
            h = convex_refinement(h, k_max)
        
        # Ensure constraints after update
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        iteration += 1
    
    # Final check for best
    final_c5 = evaluate_C5(best_h)
    return (best_h.tolist(), final_c5, n_points)
