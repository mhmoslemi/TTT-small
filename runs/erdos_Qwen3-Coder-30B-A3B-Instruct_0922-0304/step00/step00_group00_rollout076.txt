from scipy import signal
import time

    np.random.seed(seed)

    dx = 2.0 / n_points
    
    # Normalize h to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Set initial bounds
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = int(1e6)
    patience = 50
    no_improve_count = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        # Use full convolution for overlap computation
        h_reflected = np.concatenate([h, np.zeros(n_points - 1)])
        h_shifted = np.concatenate([np.zeros(n_points - 1), h])
        full_conv = signal.convolve(h_reflected, 1 - h_shifted, mode='full')
        overlaps = full_conv[n_points - 1:] * dx  # Only positive shifts
        
        # Find the shift that gives maximum overlap
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient for this k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            if j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h_new = h - alpha * grad
        
        # Ensure h remains in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to sum to n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        h = h_new
        old_max_overlap = max_overlap
        
        # Recalculate max overlap after update
        # Use numpy correlation for efficiency
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early stopping if no improvement for patience steps
        if no_improve_count >= patience:
            break
            
        # Adapt learning rate
        if current_c5 >= old_max_overlap:
            alpha *= 0.9995
        else:
            alpha = min(alpha * 1.01, 0.5)
    
    # Final check with proper evaluator
    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_final)
    
    return (best_h.tolist(), final_c5, len(best_h))
