import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    # compute C5 = max_k integral of h(x)(1 - h(x+k)) dx
    conv = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_feasible(h, dx, n_points):
    # ensure sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    # ensure bounds [0, 1]
    h = np.clip(h, 0, 1)
    return h

    swarm_size = 100
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_iter = 10000
    
    # initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    gbest = None
    gbest_val = float('inf')
    
    for i in range(swarm_size):
        if i == 0 and 'initial_h_values' in globals():
            h = h0.copy()
        else:
            h = np.random.rand(n_points)
        h = project_feasible(h, dx, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        val = evaluate_h(h, dx)
        pbest.append(h.copy())
        pbest_vals.append(val)
        if val < gbest_val:
            gbest = h.copy()
            gbest_val = val
    
    # main PSO loop
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        for i in range(swarm_size):
            # update velocity
            r1, r2 = np.random.rand(2, n_points)
            v_new = (inertia * velocities[i] + 
                     cognitive * r1 * (pbest[i] - particles[i]) + 
                     social * r2 * (gbest - particles[i]))
            
            # update position
            new_pos = particles[i] + v_new
            
            # project to feasible set
            new_pos = project_feasible(new_pos, dx, n_points)
            
            # evaluate
            new_val = evaluate_h(new_pos, dx)
            
            # update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = new_pos.copy()
                pbest_vals[i] = new_val
                
                if new_val < gbest_val:
                    gbest = new_pos.copy()
                    gbest_val = new_val
                    last_improvement = iter_count
        
        # update velocities
        for i in range(swarm_size):
            velocities[i] = v_new
    
    # finalize result
    result_h = project_feasible(gbest, dx, n_points)
    final_c5 = evaluate_h(result_h, dx)
    
    return (result_h.tolist(), final_c5, len(result_h))
