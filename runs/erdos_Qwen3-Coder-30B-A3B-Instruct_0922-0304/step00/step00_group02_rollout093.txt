import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Adjust sum to match constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode='full')
        # The maximum is at the center
        max_overlap = np.max(overlaps[len(h)-1:])
        return max_overlap * dx
    
    def mutate(h, rate=0.01):
        h = h.copy()
        indices = np.random.choice(len(h), size=int(len(h) * rate), replace=False)
        for i in indices:
            h[i] += np.random.normal(0, 0.1)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to maintain sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    # GA parameters
    population_size = 100
    generations = 1000
    elite_size = population_size // 2
    mutation_rate = 0.01
    
    # Initialize population
    population = [h0.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
        population.append(h)
        
    best_h = h0.copy()
    best_c5 = evaluate(h0)
    
    # Evolutionary loop
    gen = 0
    no_improvement = 0
    while time.time() - start_time < budget_s - 1 and gen < generations:
        gen += 1
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate(h)
            fitness.append(1.0 / (c5 + 1e-12))  # Avoid division by zero
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        current_best_c5 = evaluate(population[0])
        if current_best_c5 < best_c5:
            best_h = population[0].copy()
            best_c5 = current_best_c5
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > 100:
            break
            
        # Selection - keep top half
        selected = population[:elite_size]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover + Mutation
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover (single point)
            point = np.random.randint(1, n_points)
            child1 = np.concatenate([parent1[:point], parent2[point:]])
            child2 = np.concatenate([parent2[:point], parent1[point:]])
            
            # Mutate
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
        
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (list(best_h), final_c5, n_points)
