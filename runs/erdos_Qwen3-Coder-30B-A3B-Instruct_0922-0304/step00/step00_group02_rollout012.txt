import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure exact sum constraint
    current_sum = np.sum(h)
    if current_sum != target_sum:
        # Adjust to meet sum constraint
        diff = target_sum - current_sum
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(diff)):
                h[indices[i]] = 1.0
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(-diff)):
                h[indices[i]] = 0.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    def evaluate_erdos_solution(h_values):
        # Compute overlap integral
        overlaps = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    def project_to_binary(h_relaxed, target_sum):
        # Threshold at 0.5
        h_binary = (h_relaxed > 0.5).astype(float)
        current_sum = np.sum(h_binary)
        
        if abs(current_sum - target_sum) < 1e-6:
            return h_binary
            
        # If sum too high, reduce some 1s to 0s
        if current_sum > target_sum:
            indices = np.argsort(h_relaxed)[::-1]  # Sort by descending values
            for i in indices:
                if current_sum <= target_sum:
                    break
                if h_binary[i] == 1:
                    h_binary[i] = 0
                    current_sum -= 1
        # If sum too low, increase some 0s to 1s
        elif current_sum < target_sum:
            indices = np.argsort(h_relaxed)
            for i in indices:
                if current_sum >= target_sum:
                    break
                if h_binary[i] == 0:
                    h_binary[i] = 1
                    current_sum += 1
        return h_binary
    
    iter_count = 0
    iter_since_improve = 0
    max_iters = 500
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iters:
        iter_count += 1
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iter_since_improve = 0
        else:
            iter_since_improve += 1
            # Early stopping condition
            if iter_since_improve > 50:
                break
                
        # Solve convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == target_sum]
        
        # Approximate current overlap gradient for better initial point
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        
        try:
            # Formulate objective (approximate max of overlaps)
            obj = cp.Minimize(1e10 * max_overlap)  # Dummy objective - use initial point
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False)
        except Exception:
            # Fall back to simple convex relaxation with soft constraint
            try:
                h_relaxed.value = np.array(h)
            except Exception:
                pass
        
        # Update h with projected relaxed solution
        if h_relaxed.value is not None:
            new_h = h_relaxed.value
            h = project_to_binary(new_h, target_sum)
        else:
            h = project_to_binary(h, target_sum)
            
        # Do some noise perturbation (5% of elements)
        noise_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for idx in noise_indices:
            # Flip a bit or slightly adjust value (0.05 range)
            if h[idx] == 0:
                h[idx] = min(1.0, h[idx] + 0.1)
            else:
                h[idx] = max(0.0, h[idx] - 0.1)
        
        # Project after noise to maintain sum and bounds
        h = project_to_binary(h, target_sum)
    
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), best_c5, n_points)
