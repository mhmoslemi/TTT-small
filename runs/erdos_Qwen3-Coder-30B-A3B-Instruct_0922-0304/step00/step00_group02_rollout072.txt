import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Start with initial values or create binary starting vector
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.concatenate([
        np.ones(n_points // 2), np.zeros(n_points - n_points // 2)
    ]).astype(float)
    
    # Normalize to sum = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_no_improvement = 100
    no_improvement_count = 0
    iteration = 0
    perturbation_freq = 500
    
    def evaluate(h_vec):
        # Use convolution to compute overlaps for all shifts
        h_correlate = correlate(h_vec, 1 - h_vec, mode="full")
        # Extract the overlapping parts (excluding the zero-padding)
        overlap_vec = h_correlate[len(h_vec)-1:]  # the actual overlaps (k=0 to 2*dx)
        # Scale by dx to match integral definition
        overlap_vec *= dx
        return np.max(overlap_vec), overlap_vec
    
    def project_h(h_vec):
        # Clip to [0, 1], then scale so sum = n_points / 2
        h_clipped = np.clip(h_vec, 0, 1)
        if np.sum(h_clipped) > 0:
            scale_factor = (n_points / 2) / np.sum(h_clipped)
            h_projected = h_clipped * scale_factor
        else:
            h_projected = np.full_like(h_clipped, 1.0 / n_points)
        return h_projected
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute current overlap
        c5, overlaps = evaluate(h)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if c5 <= target_c5:
            break
        
        # Stop if no improvement for too long
        if no_improvement_count >= max_no_improvement:
            break
            
        # Perturbation to escape local minima
        if iteration % perturbation_freq == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = project_h(h)
            continue
            
        # Compute gradient
        k_max = np.argmax(overlaps)
        # Gradient computation approximation using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        h_plus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            c5_plus, _ = evaluate(h_plus)
            grad[i] = (c5_plus - c5) / eps
            h_plus[i] = h[i]  # back to original
            
        # Update h using gradient
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        
        # Test if new value improves
        new_c5, _ = evaluate(h_new)
        if new_c5 < c5:
            h = h_new.copy()
        else:
            alpha *= gamma
            if alpha < alpha_min:
                break
    
    # Final evaluation of best solution
    best_c5, _ = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
