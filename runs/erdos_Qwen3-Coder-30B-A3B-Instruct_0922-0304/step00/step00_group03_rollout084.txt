```python
import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def is_valid_h(h):
    """Check if h satisfies constraints."""
    return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - 1.0) < 1e-10

def generate_neighbor(h, n_points):
    """Generate a neighbor by flipping 2 bits."""
    h_new = h.copy()
    # Pick 2 random indices to flip
    idxs = np.random.choice(n_points, size=2, replace=False)
    h_new[idxs] = 1 - h_new[idxs]
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available, else create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create an initial binary vector with 1s and 0s
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)

    # Ensure sum(h) = 1.0
    h = h * (n_points / 2) / np.sum(h)
    dx = 2.0 / n_points

    # Tabu search setup
    tabu_list = []
    tabu_tenure = 100
    max_iter = 1000
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate neighbors and evaluate them
        neighbors = []
        for _ in range(50):  # Generate 50 neighbors
            neighbor = generate_neighbor(h, n_points)
            if is_valid_h(neighbor):
                c5 = evaluate_c5(neighbor, dx)
                neighbors.append((neighbor, c5))
                
        if not neighbors:
            continue
            
        # Sort neighbors by C5
        neighbors.sort(key=lambda x: x[1])
        best_neighbor, best_neighbor_c5 = neighbors[0]
        
        # Accept best non-tabu neighbor or accept if tabu but better than best
        is_tabu = any(np.array_equal(h, tabu_h) for tabu_h in tabu_list)
        
        if not is_tabu or best_neighbor_c5 < best_c5:
            h = best_neighbor.copy()
            if best_neighbor_c5 < best_c5:
                best_h = best_neighbor.copy()
                best_c5 = best_neighbor_c5
                
            # Add current h to tabu list
            tabu_list.append(h.copy())
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
