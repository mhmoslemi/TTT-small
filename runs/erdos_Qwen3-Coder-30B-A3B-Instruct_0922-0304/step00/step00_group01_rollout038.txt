import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute overlaps for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from 0 to n_points-1)
    overlap_vals = overlap[n_points-1:] * dx
    return np.max(overlap_vals)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    alpha_init = 0.01
    rho = 0.5
    eta = 0.5
    max_backtrack = 30
    patience = 100
    best_c5 = float('inf')
    best_h = None

        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points, dtype=float) * 0.5
    
    # Ensure correct integral
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            if best_c5 < target_bound:
                break
        
        # Find k_max that gives max overlap
        overlap = correlate(h, 1 - h, mode='full')
        overlap_vals = overlap[n_points-1:] * dx
        k_max = np.argmax(overlap_vals)
        max_overlap = overlap_vals[k_max]
        
        # Compute gradient with respect to h for this k_max
        gradient = np.zeros(n_points)
        h_shifted = np.concatenate([h, np.zeros(k_max)])
        # Gradient term from h[i] * (1 - h[i + k_max])
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] += -(1 - h[i + k_max])
            # Also add term for h[i] in the product h[i] * (1 - h[i - k_max])
            if i >= k_max:
                gradient[i] += h[i - k_max]
        
        # Gradient just takes into account the term that depends on h[i]
        # Actually let's compute it properly using the full derivative
        # dC/dh[i] = (1 - h_clipped[i + k_max]) - (h[i] * (1 - h_clipped[i - k_max]))
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # First term: 1 - h[i + k_max] 
            if i + k_max < n_points:
                gradient[i] += (1 - h[i + k_max])
            else:
                gradient[i] += 0
            # Second term: - h[i] * (1 - h[i - k_max])
            if i >= k_max:
                gradient[i] += -h[i] * (1 - h[i - k_max])
            else:
                gradient[i] += -h[i]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-12:
            gradient = gradient / grad_norm
        else:
            # No meaningful direction, break
            break

        # Line search
        alpha = alpha_init
        for _ in range(max_backtrack):
            # Try new h
            h_new = h - alpha * gradient
            # Clip h_new to [0, 1] and rescale
            h_new = np.clip(h_new, 0, 1)
            current_sum = np.sum(h_new)
            if abs(current_sum - n_points / 2) > 1e-6:
                h_new = h_new * (n_points / 2) / current_sum
            # Check if it improves
            new_c5 = evaluate_erdos_solution(h_new, n_points)
            if new_c5 < current_c5 - 1e-12:
                h = h_new
                break
            alpha *= rho
        else:
            # If line search fails, stop
            break
    
    if best_h is None:
        best_h = h.copy()
        best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), best_c5, n_points)
