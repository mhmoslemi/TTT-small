```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    # compute all correlations
    correlations = np.correlate(h_values, 1 - np.array(h_values), mode="full")
    c5 = max(correlations) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 42
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * n_points / 2
    else:
        # fallback initialization
        h_start = np.ones(n_points) * 0.5
        h_start[0] = 0.6
        h_start[1] = 0.4
    
    def optimize_convex(h0):
        h = Variable(n_points)
        # constrain the sum of h to be n_points/2
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
        # relaxation for optimization
        obj = cp.Minimize(cp.max(cp.vstack([cp.sum(cp.multiply(h, cp.roll(1-h, k)))*2/n_points for k in range(n_points)])))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            return np.array(h.value).flatten()
        except:
            return h0
    
    def refine_with_gradient(h0):
        dx = 2.0 / n_points
        h = h0.copy()
        iterations = 100
        eta = 0.1
        for i in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # compute correlations
            corr = np.correlate(h, 1 - h, mode="full")
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)
            # compute gradient w.r.t. the maxcorrelation term
            # gradient of h[l] in correlation with (1-h) at lag k
            gradient = np.zeros_like(h)
            # derivative for l in [0, n_points-1]
            for l in range(n_points):
                # corr_k with k=0..n_points-1 (full size)
                # corr has size 2*n_points-1
                c_idx = l + k + n_points - 1
                if 0 <= c_idx < len(corr):
                    gradient[l] += (1 - h[l]) if (c_idx == l + k + n_points - 1) else 0
            # take gradient step
            h -= eta * gradient
            # project h to [0,1]
            h = np.clip(h, 0, 1)
            # rescale to maintain integral constraint
            h *= (n_points / 2) / np.sum(h)
        return h

    # convex optimization to get good starting point
    if time.time() - start_time < budget_s * 0.4:
        h_convex = optimize_convex(h_start)
        h_refined = refine_with_gradient(h_convex)
        current_c5 = evaluate_erdos_solution(h_refined, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
        # upsample to 84 points
        n_points = 84
        h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, 42), h_refined)
        current_c5 = evaluate_erdos_solution(h_upsampled, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_upsampled.copy()
        # use 84 pts for final refinement
        h_refined = refine_with_gradient(h_upsampled)
        current_c5 = evaluate_erdos_solution(h_refined, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    else:
        h_refined = refine_with_gradient(h_start)
        current_c5 = evaluate_erdos_solution(h_refined, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
    
    return (best_h, best_c5, n_points)
```
