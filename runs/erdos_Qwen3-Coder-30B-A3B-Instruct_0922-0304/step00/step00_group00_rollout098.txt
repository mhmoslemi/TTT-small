import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h)
    except NameError:
        n_points = 45

    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Ensure constraint sum(h) * dx = 1 (i.e., sum(h) == n_points / 2)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h *= (target_sum / current_sum)
    
    def evaluate(h_vec):
        # Compute overlap integral C5 = max_k int h(x)(1 - h(x+k)) dx
        # Use correlate with mode="full" to compute all shifts
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Take max over all shifts, scaled by dx
        c5 = np.max(corr) * dx
        return c5
    
    def perturb(h_vec, delta=0.1, frac_perturb=0.05):
        # Make a copy to avoid modifying the input
        new_h = h_vec.copy()
        n = len(new_h)
        num_perturb = max(1, int(n * frac_perturb))
        indices = np.random.choice(n, size=num_perturb, replace=False)
        
        # Perturb selected indices
        for i in indices:
            # Add random perturbation
            new_h[i] += np.random.uniform(-delta, delta)
            # Ensure it's within [0, 1]
            new_h[i] = max(0.0, min(1.0, new_h[i]))
        
        # Renormalize to ensure sum(h) == target_sum
        current_sum = np.sum(new_h)
        scale = target_sum / current_sum
        new_h *= scale
        
        return new_h
    
    # Start with current solution
    current_h = h.copy()
    current_c5 = evaluate(current_h)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    delta = 0.1
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # At certain intervals, adjust delta based on progress
        if iteration % 10000 == 0 and iteration > 0:
            # Slow down delta for fine-tuning
            delta *= 0.98
        
        # Make new candidate
        new_h = perturb(current_h, delta=delta)
        new_c5 = evaluate(new_h)
        
        # Accept or reject based on Metropolis criterion
        delta_e = new_c5 - current_c5
        
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / temp):
            current_h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Cool down
        temp *= cooling_rate
        
        iteration += 1
        
        # Early exit if good enough
        if best_c5 <= target_bound:
            break
    
    return (best_h, best_c5, n_points)
