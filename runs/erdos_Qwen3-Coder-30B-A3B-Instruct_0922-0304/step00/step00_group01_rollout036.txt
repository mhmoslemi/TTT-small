import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_steps = 30
    alpha_init = 0.01
    rho = 0.5
    eta = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        cross_corr = correlate(h, 1 - h, mode='full')
        overlaps = cross_corr[n_points-1:] * dx
        
        # Find k_max that gives maximum overlap
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max]
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        if c5_current < target_bound:
            break
            
        # Compute gradient for k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference
            if i + k_max < n_points:
                gradient[i] += (1 - h[i + k_max])
            # Backward difference (adjust for symmetric nature)
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
            
        gradient = gradient / grad_norm
        
        # Line search
        alpha = alpha_init
        improved = False
        
        for _ in range(line_search_steps):
            h_new = h - alpha * gradient
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum to n_points/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new overlap
            cross_corr_new = correlate(h_new, 1 - h_new, mode='full')
            overlaps_new = cross_corr_new[n_points-1:] * dx
            c5_new = overlaps_new[k_max]
            
            if c5_new < c5_current:
                h = h_new
                improved = True
                break
            alpha *= rho
            
        if not improved:
            break
    
    # Ensure final solution is feasible
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final evaluation
    cross_corr_final = correlate(h_final, 1 - h_final, mode='full')
    overlaps_final = cross_corr_final[n_points-1:]
    c5_bound = np.max(overlaps_final) * dx
    
    return (h_final.tolist(), c5_bound, n_points)
