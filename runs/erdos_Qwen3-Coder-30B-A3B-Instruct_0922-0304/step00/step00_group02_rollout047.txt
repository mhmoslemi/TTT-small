    max_iter = 20000
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 100
    best_c5 = float('inf')

    start_time = time.time()
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute all overlaps
        h_rev = 1 - h
        overlaps = signal.correlate(h, h_rev, mode='full')
        overlaps = overlaps[len(h_rev)-1:]  # Take the valid part
        overlaps = overlaps * (2.0 / n_points)
        
        # Find max overlap index
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Early stopping
        if best_c5 <= 0.3807:
            break

        grad = np.zeros_like(h)
        for i in range(n_points):

            grad[i] += (1 - h[j])
        
        # Update step
        h_new = h - alpha * grad

        h_temp = h_new
