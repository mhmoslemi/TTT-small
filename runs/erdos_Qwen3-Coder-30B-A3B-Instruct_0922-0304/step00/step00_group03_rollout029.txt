    best_h = h0.copy()
    best_C5 = evaluate_C5(h0, dx)

    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:

        overlaps = np.correlate(best_h, 1 - best_h, mode='full')
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max] * dx
        
        # Gradient: each h[i] contributes to overlap with h[i+k]
        # For the maximum overlap term
        grad = np.zeros(n_points)
        # For index k_max (in full correlation space), the actual shift
        shift = k_max - (n_points - 1)
        for i in range(n_points):
            j = i - shift
            if 0 <= j < n_points:
                grad[i] += (1 - best_h[j]) * dx
                
        # Step size for gradient descent
        step_size = 0.001
        new_h = best_h - step_size * grad
        
        # Project onto constraints
        # 1. Clip to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # 2. Adjust to satisfy integral constraint
        integral = np.sum(new_h) * dx
        new_h = new_h * (1.0 / integral)
        
        # Project back onto [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Check C5 and update if improving
        new_C5 = evaluate_C5(new_h, dx)
        if new_C5 < best_C5 - 1e-10:
            best_h = new_h.copy()
            best_C5 = new_C5

            current_C5 = evaluate_C5(best_h, dx)
            if abs(current_C5 - best_C5) > 1e-8:
                best_C5 = current_C5

    return (best_h, best_C5, n_points)

def evaluate_C5(h, dx):
    # Compute all overlaps
    overlaps = np.correlate(h, 1-h, mode='full')
    return np.max(overlaps) * dx
