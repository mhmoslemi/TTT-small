import cvxpy as cp

import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 45
    dx = 2.0 / n_points
    target_bound = 0.3807

    else:
        h = np.ones(n_points) * 0.5

    h = h * (n_points / 2) / np.sum(h)

    iteration = 0
    max_iter = 50000
    no_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate current C5 - max overlap
        conv = np.correlate(h, 1-h, mode='full') * dx
        current_c5 = np.max(conv)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if best_c5 <= target_bound:
            break
            
        if no_improvement > 50:
            # Break if no improvement for 50 iterations
            break
            
        # Find the most constraining shift
        max_idx = np.argmax(conv)
        shift = max_idx - (n_points - 1)
        
        # Use cvxpy to optimize with McCormick relaxation
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Build McCormick relaxation for the specific shift
        # Add linear constraints to approximate h[i] * (1 - h[i+shift])
        for i in range(n_points):
            j = (i + shift) % n_points
            # McCormick relaxation for product h[i] * (1 - h[j])
            # We use: z >= h[i] + h[j] - 1, z >= 0, z <= h[i], z <= h[j]
            # Here z represents h[i] * (1-h[j])
            z = cp.Variable()
            constraints.extend([
                z >= h_vars[i] + h_vars[j] - 1,
                z >= 0,
                z <= h_vars[i],
                z <= 1 - h_vars[j]
            ])

            # so simplify approach to direct numerical optimization when needed

        # Instead we use numerical optimization:
        # Create a bounded optimization that uses the same constraint framework
        
        # Alternative: Just use simple projected gradient descent-like steps
        # from current h
        h_new = h.copy()
        
        # Random perturbation step
        perturbs = np.random.normal(0, 0.02, n_points)
        h_new += perturbs
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize to integral = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this improved
        conv_new = np.correlate(h_new, 1-h_new, mode='full') * dx
        c5_new = np.max(conv_new)
        
        if c5_new < current_c5:
            h = h_new
        else:
            # Heuristic: try reducing overlap from the worst shift
            # Single adjustment step
            if shift >= 0:
                k = shift
            else:
                k = n_points + shift
                
            if k >= n_points:
                k = k % n_points
            
            # Take a small step towards reducing that overlap
            # More effectively we compute derivative and do small steps
            
            # Analytical gradient: approximate
            epsilon = 1e-3
            h_plus = h.copy()
            h_minus = h.copy()
            
            # Perturb at position i to reduce (h[i] * (1 - h[i+k]))
            if k <= n_points // 2:
                h_plus[(k+1) % n_points] += epsilon
            else:
                h_plus[(k+1) % n_points] -= epsilon
                
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            conv_p = np.correlate(h_plus, 1-h_plus, mode='full') * dx
            c5_p = np.max(conv_p)
            
            if c5_p < current_c5:
                h = h_plus
            else:
                # Sometimes no progress and we can't use more complex methods

    # Final evaluation

    final_c5 = np.max(final_conv)
    
    return (best_h.tolist(), final_c5, n_points)
