def evaluate_c5(h):
    """Evaluate C5 for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps (full correlation)
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum and convert to C5
    max_corr = np.max(corr)
    return max_corr * dx
