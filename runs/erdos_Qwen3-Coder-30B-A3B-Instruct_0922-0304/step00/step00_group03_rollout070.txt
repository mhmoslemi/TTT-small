import cvxpy as cp
from scipy.signal import correlate
import time

        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize dual variables
    lambda_k = np.zeros(2*n_points - 1)
    mu = 0.0
    
    # Optimization loop
    start_time = time.time()

        if time.time() - start_time > budget_s - 1:
            break

        h_var = cp.Variable(n_points)

        h_conv = np.convolve(h, np.ones(1) - h, mode='full')

        obj = cp.Minimize(
            cp.max(cp.Variable(2*n_points - 1) - lambda_k @ cp.Variable(2*n_points - 1) + mu * (cp.sum(h_var) - 1))

        )

        constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == n_points / 2]

            h_new = h.copy()

        # Entropy regularization for lambda
        lambda_k = lambda_k + beta * np.log(1 + lambda_k)
        
        # Clip lambda for stability
        lambda_k = np.clip(lambda_k, -10, 10)
        
        # Update beta (entropy regularization)
        if iteration % 100 == 0:
            beta *= beta_decay
