import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_bound = 0.3807
    initial_T = 0.1
    cooling_rate = 0.995
    perturb_fraction = 0.05
    max_iter = int(1e6)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Renormalize h to have sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute overlap integral
    def compute_C5(h):
        # Compute correlation of h with (1-h) using full mode
        corr = signal.correlate(h, 1 - h, mode='full')
        # Take maximum over all shifts
        max_corr = np.max(corr)
        # Scale by dx
        return max_corr * dx
    
    # Initial computation
    C5_current = compute_C5(h)
    C5_best = C5_current
    h_best = h.copy()
    
    # Annealing
    T = initial_T
    start_time = time.time()
    
    # Number of points to perturb at each step
    perturb_count = max(1, int(n_points * perturb_fraction))
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Make a copy of current h
        h_new = h.copy()
        
        # Select random indices to perturb
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        
        # Apply small random perturbations
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            # Clamp to [0, 1]
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Renormalize to ensure sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        C5_new = compute_C5(h_new)
        
        # Accept or reject based on Metropolis criterion
        delta = C5_new - C5_current
        if delta < 0 or np.exp(-delta / T) > np.random.rand():
            h = h_new
            C5_current = C5_new
            
            if C5_current < C5_best:
                C5_best = C5_current
                h_best = h.copy()
                
                # Early termination if we reach the target bound
                if C5_best <= target_bound:
                    break
        
        # Cool down temperature
        T *= cooling_rate
        
        # Occasionally check for early termination
        if i % 1000 == 0:
            if C5_best <= target_bound:
                break
    
    return (h_best.tolist(), float(C5_best), n_points)
