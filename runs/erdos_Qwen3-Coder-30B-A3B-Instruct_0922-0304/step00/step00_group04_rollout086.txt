def evaluate_erdos_solution(h_values, dx):
    """Evaluate the overlap integral C_5 for given h_values."""

    corr = correlate(h, 1 - h, mode='full')
    # The valid correlations are from index n-1 to 2*n-2 (corresponding to lags -n+1 to n-1)
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx
