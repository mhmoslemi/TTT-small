import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction or create a new one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        # Initialize with 1s in random positions to satisfy integral constraint
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones_indices] = 1.0
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count - last_improvement < 100:
        iter_count += 1
        
        # Compute overlaps for current h
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            # Compute overlap for lag k
            lagged = np.roll(h, k)
            overlaps[k] = np.sum(h * (1 - lagged)) * dx
        
        max_overlap = np.max(overlaps)
        
        # Try to improve by changing individual elements
        h_new = h.copy()
        # Randomly sample indices to change
        candidates = np.random.permutation(n_points)
        
        # Adjust positions
        count = 0
        for i in candidates:
            if count > n_points // 4:  # Limit updates per iteration
                break
            h_trial = h_new.copy()
            h_trial[i] = 1 - h_trial[i]  # Flip the bit
            
            # Ensure constraint sum(h) = n_points/2
            current_sum = np.sum(h_new)
            if h_new[i] == 1 and current_sum > n_points / 2:
                # Need to flip another element to maintain sum
                flip_candidate = np.where(h_new == 1)[0]
                if len(flip_candidate) > 1:
                    # Find an element that is 1 (not the one we are flipping)
                    flip_to_zero = np.random.choice([x for x in flip_candidate if x != i])
                    h_trial[flip_to_zero] = 0
            elif h_new[i] == 0 and current_sum < n_points / 2:
                flip_candidate = np.where(h_new == 0)[0]
                if len(flip_candidate) > 1:
                    flip_to_one = np.random.choice([x for x in flip_candidate if x != i])
                    h_trial[flip_to_one] = 1
            
            # Re-evaluate overlap for new h
            new_overlaps = np.zeros(n_points)
            for k in range(n_points):
                lagged = np.roll(h_trial, k)
                new_overlaps[k] = np.sum(h_trial * (1 - lagged)) * dx
            new_max = np.max(new_overlaps)
            
            if new_max < max_overlap:
                h_new = h_trial
                max_overlap = new_max
                last_improvement = iter_count
                count += 1
        
        # Finalize
        h = h_new
        
        # Re-normalize to satisfy integral constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-6:
            # Compute scale factor to adjust to constraint
            scale = (n_points / 2) / h_sum
            h = np.clip(h * scale, 0, 1)
        
        # Update best solution if improved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    n = len(h)
    result = 0.0
    for k in range(n):
        lagged = np.roll(h, k)
        overlap = np.sum(h * (1 - lagged)) * dx
        result = max(result, overlap)
    return result
