import numpy as np
import time
from scipy import correlate

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for a given step function h"""
    dx = 2.0 / n_points
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode="full")
    # The maximum overlap
    c5 = np.max(overlaps) * dx
    return c5

def ensure_feasible(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        # Scale to target sum
        h = h * target_sum / current_sum
    return h

def mutate(h, n_points, rate=0.01):
    """Apply mutation to a candidate solution"""
    h = h.copy()
    # Randomly select indices to mutate
    n_mutate = int(rate * n_points)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    # Mutate each selected index
    for i in indices:
        # Random change in [0, 1]
        h[i] += np.random.normal(0, 0.1)
    # Ensure feasibility
    h = ensure_feasible(h, n_points)
    return h

def crossover(parent1, parent2, n_points):
    """Perform crossover between two parents"""
    # Random crossover point
    crossover_point = np.random.randint(1, n_points-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    # Ensure both children are feasible
    child1 = ensure_feasible(child1, n_points)
    child2 = ensure_feasible(child2, n_points)
    return child1, child2
