from scipy import signal
import time

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlap integrals C_m
    conv = signal.correlate(h, 1 - h, mode='full')
    # Adjust for proper indexing

    max_val = np.max(conv)
    return max_val * dx

    # Gradient descent loop

        # Compute overlap integrals at all lags
        conv = signal.correlate(h, 1 - h, mode='full')
        m_vals = np.arange(-n_points + 1, n_points)
        # Find maximum value
        max_val = np.max(conv)
        # Find all indices where maximum occurs
        max_indices = np.where(np.abs(conv - max_val) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        # For each maximum lag, accumulate gradient contribution
        for i in max_indices:
            # Lag corresponding to this index
            lag = m_vals[i]
            # If lag is positive, we shift h by i to match the indices
            if lag >= 0:
                # h[i] contributes to the integral corresponding to `lag`
                # We take derivative of h[i] * (1 - h[i + lag])
                if i + lag < n_points:
                    gradient[i] += 1 - h[i + lag]
            else:
                # For negative lag, h[i] contributes to lag-neg
                if i - lag >= 0:
                    gradient[i] += 1 - h[i - lag]

        # Update step
        h_new = h - learning_rate * gradient
        
        # Project to bounds
        h_new = np.clip(h_new, 0, 1)
        # Scale to required integral
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = iteration
            
        h = h_new.copy()
        
        # Reduce learning rate periodically
        if iteration > 0 and iteration % 100 == 0 and iteration > last_improvement:
            learning_rate *= 0.1
            
    return (list(best_h), best_c5, n_points)
