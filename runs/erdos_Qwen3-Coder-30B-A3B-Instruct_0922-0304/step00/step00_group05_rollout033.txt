import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    T_max = 1.0
    T_min = 1e-4
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    delta_init = 0.1
    min_sum = n_points / 2.0
    max_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points // 4) * 0.5
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0.0)
        h = np.pad(h, (n_points//4, 0), 'constant', constant_values=1.0)
        h = h[:n_points]
        
    # Normalize to correct sum
    h_sum = np.sum(h)
    h = h * min_sum / h_sum
    
    # Validate constraints
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - min_sum) < 1e-10
    
    # Initialize variables
    T = T_max
    delta = delta_init
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    start_time = time.time()
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Acceptance rate adaptation
        if i % 100 == 0 and i > 0:
            delta = max(0.001, delta * 0.95) if i % 500 == 0 else delta
        
        h_new = h.copy()
        
        # Perturb h
        if np.random.random() < 0.5:
            # Swap two points
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            val1, val2 = h_new[i1], h_new[i2]
            h_new[i1], h_new[i2] = val2, val1
        else:
            # Adjust adjacent points
            i1 = np.random.randint(n_points)
            i2 = (i1 + 1) % n_points
            delta_h = np.random.uniform(-delta, delta)
            h_new[i1] += delta_h
            h_new[i2] -= delta_h
            
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to fix sum constraint
        h_sum = np.sum(h_new)
        h_new = h_new * min_sum / h_sum
        
        # Compute new C5
        new_C5 = evaluate_C5(h_new, dx)
        
        # Accept or reject
        delta_C5 = new_C5 - best_C5
        if delta_C5 < 0 or np.random.random() < np.exp(-delta_C5 / T):
            h = h_new
            if new_C5 < best_C5:
                best_h = h_new.copy()
                best_C5 = new_C5
        
        # Update temperature
        T *= alpha
        
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    """Evaluate C5 using convolution."""
    h = np.array(h)
    # Compute correlation of h and (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum over all lags, scaled by dx
    max_val = np.max(np.abs(corr)) * dx
    return max_val
