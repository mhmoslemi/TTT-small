import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        # Initialize with approximately half 1s
        ones_indices = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones_indices] = 1.0
    
    # Ensure correct integral
    if np.sum(h) != n_points / 2:
        # Adjust to maintain sum = n_points / 2 by scaling
        old_sum = np.sum(h)
        if old_sum > 0:
            h = h * (n_points / 2) / old_sum

    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Alternating optimization with binary refinement
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
        
        # Compute the forbidden regions
        S = n_points
        forbidden = np.zeros(S)
        for i in range(S):
            for k in range(S):
                # Compute overlap contribution
                j = (i - k) % S
                if h[j] > 0.5:  # Count if neighbor has a 1 at this position in shifted version
                    forbidden[i] += 1
        
        # Create new binary version: Set h[i] = 1 if it reduces maximum overlap
        new_h = h.copy()
        # The updates will be greedy: minimal overlap strategy tends to be best for this problem
        # Start with all 0s, and greedily add 1s where beneficial
        # To make it fast: most greedy approach—sort by forbidden count, use the lowest forbidden ones
        available_indices = list(range(S))
        target_ones = S // 2
        current_ones = np.sum(new_h > 0.5)
        
        # Update until we get to desired number of 1s
        if current_ones < target_ones:
            # Compute which indices to set to 1 (greedy tries to minimize current max overlap)
            # For speed, we'll use a direct improvement check:
            for i in range(target_ones - current_ones):
                if time.time() - start_time > budget_s - 1.0:
                    break
                min_max_shifted = float('inf')
                best_idx = -1
                # Try setting each unassigned 0 to 1
                unassigned = [idx for idx in available_indices if new_h[idx] <= 0.5]
                # Evaluate: check if setting it to 1 reduces the maximum overlap
                # This is a rough method to prioritize reducing overlaps
                for idx in unassigned:
                    # Temporarily set to 1
                    temp_h = new_h.copy()
                    temp_h[idx] = 1.0
                    c5 = evaluate_c5(temp_h)
                    if c5 < min_max_shifted:
                        min_max_shifted = c5
                        best_idx = idx
                if best_idx != -1:
                    new_h[best_idx] = 1.0
                    available_indices.remove(best_idx)
                    
        elif current_ones > target_ones:
            remaining = current_ones - target_ones
            # Remove 1s
            ones_indices = [i for i in range(S) if new_h[i] > 0.5]
            ones_indices.sort(key=lambda i: forbidden[i])
            for i in range(remaining):
                if i >= len(ones_indices):
                    break
                idx = ones_indices[i]
                new_h[idx] = 0.0

        # Ensure correct sum
        actual_sum = np.sum(new_h)
        if actual_sum > 0:
            new_h = new_h * (S / 2) / actual_sum

        # Check if there's an improvement
        c5 = evaluate_c5(new_h)
        if c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = c5

        # Final clipping to [0,1], just enforcing the constraint
        best_h = np.clip(best_h, 0, 1)
        if evaluate_c5(best_h) < best_c5:
            best_c5 = evaluate_c5(best_h)

        # Early stopping if target reached
        if best_c5 < 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
