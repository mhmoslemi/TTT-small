import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        current_sum = np.sum(h_init)
        if abs(current_sum - target_sum) > 1e-6:
            # Repair the sum by adjusting values
            diff = target_sum - current_sum
            idxs = np.argsort(h_init)
            if diff > 0:
                for i in range(int(diff)):
                    h_init[idxs[i]] = min(1.0, h_init[idxs[i]] + 1.0)
            else:
                for i in range(int(-diff)):
                    h_init[idxs[-(i+1)]] = max(0.0, h_init[idxs[-(i+1)]] - 1.0)
    else:
        # Initialize random solution with proper sum
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) * target_sum
    
    # Prepare the correlation computation
    def compute_c5(h):
        h = np.array(h)
        # Compute the overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        # The number of valid shifts is 2*n_points - 1
        shifts = np.arange(-n_points + 1, n_points)
        # Take only the relevant parts, which are the shifts from -1 to 1
        relevant_indices = np.arange(n_points - 1, 2 * n_points - 1)
        c5_vals = corr[relevant_indices] * dx
        return np.max(c5_vals)
    
    # Reduce precision for computation and avoid numerical issues
    tolerance = 1e-4
    t_low = 0.0
    t_high = 0.5
    best_h = h_init.copy()
    best_c5 = compute_c5(h_init)
    
    # Begin binary search
    while time() - start_time < budget_s - 5 and t_high - t_low > tolerance:
        t_mid = (t_low + t_high) / 2.0
        # Solve convex feasibility problem
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # Add overlap constraints
        for k in range(1, n_points):
            # Left shift by k
            # Constraints for h[i] * (1 - h[i+k]) 
            w = cp.Variable(n_points - k)
            for i in range(n_points - k):
                constraints.append(w[i] <= h[i])
                constraints.append(w[i] <= h[i + k])
            # Actual sum of products
            prod_sum = cp.sum(w)
            # h[i] - h[i]*h[i+k]
            h_sum = cp.sum(h[:n_points-k])
            constraints.append(h_sum - prod_sum <= t_mid)
            
            # Right shift by k
            w2 = cp.Variable(n_points - k)
            for i in range(n_points - k):
                constraints.append(w2[i] <= h[i])
                constraints.append(w2[i] <= h[i - k])
            prod_sum2 = cp.sum(w2)
            h_sum2 = cp.sum(h[k:])
            constraints.append(h_sum2 - prod_sum2 <= t_mid)
        
        # Also for k = 0 shift (the original case)
        # We compute a smoothed version
        w0 = cp.Variable(n_points)
        for i in range(n_points):
            constraints.append(w0[i] <= h[i])
            constraints.append(w0[i] <= 1 - h[i])
        constraints.append(cp.sum(h) - cp.sum(w0) <= t_mid)
        
        # Try the feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            feasible = prob.status in ['optimal', 'optimal_inaccurate']
        except:
            feasible = False
            
        if feasible:
            t_high = t_mid
            # Recovery steps to build a feasible integer solution
            if prob.status == 'optimal':
                h_vals = h.value
                # Round values to 0 or 1, adjust accordingly
                h_rounded = np.round(h_vals).astype(float)
                diff = target_sum - np.sum(h_rounded)
                if abs(diff) > 1e-4:
                    # Flip some entries to match exactly
                    # Find indices to increase
                    to_increase = np.where(h_rounded == 0)[0]
                    to_decrease = np.where(h_rounded == 1)[0]
                    if diff > 0:
                        # Increase some zero values
                        if len(to_increase) >= int(diff):
                            for i in range(int(diff)):
                                idx = to_increase[i]
                                h_rounded[idx] = 1.0
                    else:
                        # Decrease some one values
                        if len(to_decrease) >= int(-diff):
                            for i in range(int(-diff)):
                                idx = to_decrease[i]
                                h_rounded[idx] = 0.0
                best_h = h_rounded
                best_c5 = compute_c5(best_h)
        else:
            t_low = t_mid
            
        if time() - start_time > budget_s - 10:
            break
    
    # Final evaluation with best solution
    try:
        final_c5 = compute_c5(best_h)
    except:
        final_h = h_init
        final_c5 = compute_c5(final_h)
    else:
        final_h = best_h
        final_c5 = best_c5
        
    return (final_h.tolist(), final_c5, n_points)
