import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create random feasible h
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with line search
    max_iter = 10000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break

        # Compute correlations and find max
        corr = correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        C5 = max_corr * dx
        
        if C5 < best_c5:
            best_c5 = C5
            best_h = h.copy()
        
        # Compute gradient w.r.t. max lag
        k = max_k - (n_points - 1)
        grad = np.zeros(n_points)
        if k >= 0:
            grad[k:] += (1 - h[:-k]) * dx
            grad[:-k] -= h[k:] * dx
        else:
            k = -k
            grad[k:] -= h[:-k] * dx
            grad[:-k] += (1 - h[k:]) * dx
        
        # Line search step size
        alpha = 0.1
        while alpha > 1e-10:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Enforce sum constraint
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            new_c5 = compute_c5(h_new, dx)
            if new_c5 < C5:
                h = h_new
                break
            alpha *= 0.5
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
