import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C5 for given step function h."""
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Default construction if none provided
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure constraint ∫h dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000
    patience_counter = 0
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all lags where overlap equals max_overlap
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient for these lags
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # Overlap at lag k is sum over i of h[i] * (1 - h[i+k])
                # Gradient contribution = -h[i+k] for i where i+k < n_points
                valid = np.arange(0, n_points - lag)
                gradient[valid] += (1 - h[valid + lag]) * dx
        
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy the constraint ∫ h dx = 1
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adaptive learning rate
        if patience_counter >= gamma_patience:
            learning_rate *= gamma_decrease_factor
            patience_counter = 0
    
    return (best_h, best_c5, n_points)
