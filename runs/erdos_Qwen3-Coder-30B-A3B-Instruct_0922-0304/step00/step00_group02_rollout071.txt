import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a binary vector
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:23] = 1.0  # First 23 elements are 1, rest 0 (to get sum = 22.5)
        np.random.shuffle(h)
        # Renormalize to exactly sum to 22.5
        h = h * (22.5 / np.sum(h))
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    iter_since_improvement = 0
    max_no_improvement = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        correlate_result = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlate_result * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Gradient computation: for max_k, we need d/dh_i of the overlap term
        # This is a simplified gradient: direct sensitivity to each h[i]
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_k) % len(h)
            grad[i] = (1 - h[j]) - (1 - h[i])  # Approximate gradient
        # simplified version: actually, it's more complex; we'll do a better one
        # Use finite differences for numerical gradient estimation
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps
            overlap_plus = np.max(signal.correlate(h_plus, 1 - h_plus, mode='full') * dx)
            overlap_minus = np.max(signal.correlate(h_minus, 1 - h_minus, mode='full') * dx)
            grad[i] = (overlap_plus - overlap_minus) / (2.0 * eps)
        
        # Update h
        h_temp = h - alpha * grad
        h_temp = np.clip(h_temp, 0, 1)
        scale = 22.5 / np.sum(h_temp)
        h_new = h_temp * scale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (22.5 / np.sum(h_new))  # Final renormalize
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept or reject update
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            iter_since_improvement = 0
            best_h = h.copy()
        else:
            iter_since_improvement += 1
            if iter_since_improvement > max_no_improvement:
                if alpha > alpha_min:
                    alpha *= gamma
                iter_since_improvement = 0
            h = h_new
        
        # Periodic perturbation to escape local minima
        if iteration % 1000 == 0 and iteration > 0:
            perturb_idx = np.random.choice(len(h), size=int(0.05 * len(h)), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h = np.clip(h, 0, 1)
            h = h * (22.5 / np.sum(h))

    return (best_h.tolist(), best_c5, len(best_h))

def evaluate_c5(h, dx):
    """Evaluate C5 given h and dx."""
    correlate_result = signal.correlate(h, 1 - h, mode='full')
    return np.max(correlate_result) * dx
