import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_erdos_solution(h):
    # Evaluate C5 using the given method
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    # Initialize
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a binary distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create initial binary vector with 1s and 0s
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Ensure sum constraint
    dx = 2.0 / len(h)
    target_sum = len(h) / 2.0
    current_sum = np.sum(h)
    if current_sum != target_sum:
        # Adjust to make sum correct
        diff = target_sum - current_sum
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(diff)):
                h[indices[i]] = 1.0
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(-diff)):
                h[indices[i]] = 0.0
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Iteration with time budget
    iterations = 0
    patience = 0
    max_patience = 50
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        iterations += 1
        # Compute current overlaps
        current_overlaps = np.correlate(h, 1-h, mode="full") * dx
        
        # Formulate and solve convex optimization
        n_points = len(h)
        h_relaxed = cp.Variable(n_points)
        
        # Objective: minimize max overlap
        max_overlap = cp.Variable()
        
        # Constraints
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == target_sum
        ]
        
        # Add constraint for max overlap
        for k in range(len(current_overlaps)):
            constraints.append(h_relaxed @ np.roll(1-h, k) >= current_overlaps[k])
        
        # So the max overlap is minimized
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
            else:
                # fallback
                h_new = h.copy()
        except:
            # fallback to original if optimization fails
            h_new = h.copy()
        
        # Project back to proper binary-like values
        # threshold and adjust to get sum = target_sum
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        adjusted_h = h_new.copy()
        
        diff = target_sum - current_sum
        if abs(diff) > 1e-6:
            # Adjust: turn some 0s to 1s or 1s to 0s
            if diff > 0:
                # Need to add more 1s
                indices = np.argsort(adjusted_h)
                flip_indices = indices[:int(diff)]
                adjusted_h[flip_indices] = 1.0
            else:
                # Need to subtract some 1s
                indices = np.argsort(adjusted_h)[::-1]
                flip_indices = indices[:int(-diff)]
                adjusted_h[flip_indices] = 0.0
        
        # Evaluate and update best
        candidate_h = adjusted_h
        candidate_C5 = evaluate_erdos_solution(candidate_h)
        
        if candidate_C5 < best_C5 - 1e-10:
            best_C5 = candidate_C5
            best_h = candidate_h.copy()
            patience = 0
        else:
            patience += 1
        
        # Slight perturbation to escape local minima
        perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for i in perturb_indices:
            if np.random.rand() < 0.5:
                adjusted_h[i] = 1.0
            else:
                adjusted_h[i] = 0.0
        
        # Project again
        current_sum = np.sum(adjusted_h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-6:
            if diff > 0:
                indices = np.argsort(adjusted_h)
                flip_indices = indices[:int(diff)]
                adjusted_h[flip_indices] = 1.0
            else:
                indices = np.argsort(adjusted_h)[::-1]
                flip_indices = indices[:int(-diff)]
                adjusted_h[flip_indices] = 0.0
                
        h = adjusted_h    
    
    # Final check and return
    final_C5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_C5, len(best_h))
