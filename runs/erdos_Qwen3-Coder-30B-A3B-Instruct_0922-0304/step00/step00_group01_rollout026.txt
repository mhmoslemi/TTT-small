import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create initial guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
    
    # Initialize dx and target sum
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Precompute overlap for initial solution
    def compute_c5(h_vec):
        # Only compute the maximum overlap part
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take the latter half of correlation (the overlaps)
        overlaps = corr[len(corr)//2:]
        return np.max(overlaps) * dx

    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Main optimization loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iterations += 1
        
        # Compute overlap at all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        overlaps = corr[len(corr)//2:]  # Only take the interesting overlaps
        
        # Get indices of largest overlaps
        peak_indices = np.argsort(overlaps)[::-1]
        
        # Update h based on top overlaps
        step_size = 0.05
        h_new = h.copy()
        
        # Update for top 5 peaks
        for i in range(min(5, len(peak_indices))):
            k = peak_indices[i]
            # For overlap at shift k, we want to reduce h[i] where h[i+k] is large 
            # and increase where h[i+k] is small
            for idx in range(n_points):
                if idx + k < n_points:
                    # Simple gradient-like update
                    if h[idx] > 0.01 and h[idx + k] > 0.01:
                        # Tune h[idx] based on how much overlap we have
                        h_new[idx] = np.clip(h_new[idx] - step_size * 0.1, 0, 1)
                        h_new[idx + k] = np.clip(h_new[idx + k] - step_size * 0.1, 0, 1)
                    elif h[idx] < 0.99 and h[idx + k] < 0.99:
                        # Boost those that will reduce overlap
                        h_new[idx] = np.clip(h_new[idx] + step_size * 0.1, 0, 1)
                        h_new[idx + k] = np.clip(h_new[idx + k] + step_size * 0.1, 0, 1)
                else:
                    break
        
        # Project back to feasible set
        sum_h = np.sum(h_new)
        h_new = h_new * target_sum / sum_h
        h_new = np.clip(h_new, 0, 1)
        
        # Do the final adjustment so that sum equals exactly target_sum
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate
        new_c5 = compute_c5(h_new)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping condition
        if current_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
