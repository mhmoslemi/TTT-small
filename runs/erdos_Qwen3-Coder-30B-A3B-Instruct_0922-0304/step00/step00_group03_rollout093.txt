import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to meet constraint
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Rescale again to meet sum constraint
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Hyperparameters
    lr = 0.01
    momentum = 0.9
    decay_every = 100
    max_iter = 10000
    target_bound = 0.3807
    
    # Buffers
    v = np.zeros_like(h)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        # The full correlation has 2*n - 1 elements
        # shift k ranges from -(n-1) to (n-1)
        # We only care about positive shifts: 0 to n-1 (just because it makes indexing easier)
        # Actually, need overlap for all k. The full gives all k from -(n-1) to n-1.
        # We take the max over all k.
        max_overlap = np.max(corr)
        
        if max_overlap <= target_bound:
            break
            
        # Compute gradient
        # The derivative of overlap at shift k with respect to h[i] is:
        # (1 - h[i+k]) if i+k <= n-1, 0 otherwise
        # For simplicity, compute cross correlation of h with 1-h, then interpret
        # get indices of maximal overlap
        max_idx = np.argmax(corr)
        # This tells us the shift k that gives the max
        # k from - (n-1) to (n-1) corresponds to indices [0, 2*n-2]
        shift_k = max_idx - (n_points - 1)
        
        # Gradient calculation based on impact of h[i] on overlap at shift k
        # At shift k, gradient for h[i] = (1 - h[i+k]) if i+k < n and i+k >= 0
        # so we need to compute (1 - h[i + shift_k]) for valid indices
        grad = np.zeros(n_points)
        if shift_k >= 0:
            grad[shift_k:] = 1 - h[:-shift_k]
        else:
            grad[:shift_k] = 1 - h[-shift_k:]
        
        # Update with momentum
        v = momentum * v - lr * grad
        h = h + v
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        
        # Maintain sum = n/2
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
    
    # Final evaluation
    corr = signal.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h.tolist(), final_c5, n_points)
