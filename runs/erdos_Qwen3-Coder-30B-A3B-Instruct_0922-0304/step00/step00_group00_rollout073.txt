import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Helper function to compute C5
    def compute_c5(h_vals):
        # Use scipy's correlate for full mode
        conv = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # Select only the relevant part of the correlation
        # This corresponds to k from 0 to 2 with 2/n spacing
        # But we take the max over k, which is the maximum of the convolution
        # Filter out the first (n_points-1) elements which are for negative k
        # and last (n_points-1) elements which are for positive k, keeping only k=0 to n_points-1
        # Actually, we just want the max of all these
        return np.max(conv) * dx
    
    # Compute initial C5
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05
    max_perturb = max(1, int(perturb_frac * n_points))
    delta = 0.1
    max_iter = int(1e6)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        # Make a copy of current state
        h_new = h.copy()
        
        # Decide how many points to change
        n_change = np.random.randint(1, max_perturb + 1)
        indices = np.random.choice(n_points, n_change, replace=False)
        
        # Perturb selected indices
        for i in indices:
            # Random change within bounds
            delta_h = np.random.uniform(-delta, delta)
            h_new[i] += delta_h
            # Clamp to [0,1]
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to ensure sum(h) = n_points/2
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Metropolis acceptance criterion
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Reduce temperature
        T *= cooling_rate
        
        # Early termination if good enough
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
