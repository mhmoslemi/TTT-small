import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback construction: uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Constraint: sum(h) * dx = 1 => sum(h) = n_points / 2
    def adjust_sum(h):
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            diff = (target_sum - current_sum) / n_points
            h += diff
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        return h
    
    # Evaluate C5
    def evaluate_C5(h):
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_rev = h[::-1]
        corr = correlate(h_padded, h_rev, mode='full')
        # Take the maximum overlap (excluding trivial shifts)
        max_overlap = 0
        for k in range(1, n_points):
            overlap = corr[n_points - 1 + k] * dx
            max_overlap = max(max_overlap, overlap)
        return max_overlap

    # Gradient descent
    start_time = time.time()
    max_iter = 10000
    lr = 0.01
    min_improvement = 1e-6
    best_C5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current C5
        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Compute correlation for all shifts
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_rev = h[::-1]
        corr = correlate(h_padded, h_rev, mode='full')
        overlaps = np.array([corr[n_points - 1 + k] * dx for k in range(1, n_points)])
        
        active_k = np.argmax(overlaps)
        shift = active_k + 1
        
        # Gradient for shift k (approximate by finite difference)
        # We want to minimize sum(h[i] * (1 - h[i + shift]))
        # Gradient w.r.t. h[i] is (1 - h[i + shift]) - h[i + shift] if i + shift < n_points
        # But handle indices carefully to avoid out-of-bounds
        
        # Manual gradient computation
        grad = np.zeros_like(h)
        for i in range(n_points):
            next_idx = i + shift
            if next_idx < n_points:
                grad[i] = (1 - h[next_idx]) - h[next_idx]
        
        # Gradient descent step
        h = h - lr * grad
        h = adjust_sum(h)
        
        # Early termination
        if iteration > 10 and abs(current_C5 - evaluate_C5(h)) < min_improvement:
            break
    
    # Final adjust and evaluate
    final_C5 = evaluate_C5(best_h)
    
    return (list(best_h), final_C5, n_points)
