    # Sort of relaxation approach with binary search
    while t_high - t_low > tol and (time() - start_time) < budget_s * 0.95:
        t_mid = (t_low + t_high) / 2.0
        
        # Convex formulation
        h = cp.Variable(n_points)
        w = cp.Variable((n_points, n_points))
        
        # Constraints
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == target_sum,
        ]
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            # Shifted correlation term
            if k >= 0:
                corr = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            else:
                corr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
            
            # Add to constraints
            constraints.append(corr <= t_mid)
        
        # Objective: minimize t (not used in feasibility, just ensuring constraints)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)

                h_val = h.value

                h_rounded = np.round(h_val)

                    # Increase some 0's to 1's
                    indices = np.argsort(h_val)

                    # Decrease some 1's to 0's
                    indices = np.argsort(h_val)

                        if indices[i] < n_points and h_rounded[indices[i]] == 1.0:
                            h_rounded[indices[i]] = 0.0

                h_rounded = np.clip(h_rounded, 0, 1)
