    start_time = time.time()

        # Compute overlaps for all shifts
        conv = signal.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        overlaps = conv * dx
        
        # Find the shift that gives max overlap
        max_idx = np.argmax(overlaps)
        k_max = k_indices[max_idx]
        
        # Compute gradient for k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            cooldown = 0

            cooldown += 1
            
        # Adaptive learning rate

            alpha *= 0.95
            cooldown = 0

        if best_c5 <= 0.3807:

        if cooldown > max_cooldown:
