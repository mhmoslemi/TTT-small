import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 5000
    lr = 0.01
    decay_every = 100
    momentum_beta = 0.9
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure integral constraint
    h_sum = np.sum(h)
    if h_sum != n_points / 2:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization
    start_time = time.time()
    v = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals for all shifts
        h_rev = np.flip(h)
        overlaps = np.correlate(h, h_rev, mode='full') * dx
        max_overlap = np.max(overlaps)
        
        # Gradient w.r.t. max_overlap shift
        shift_idx = np.argmax(overlaps)
        max_shift = shift_idx - (n_points - 1)
        
        # Compute per-point gradient
        grad = np.zeros_like(h)
        for k in range(n_points):
            if 0 <= k + max_shift < n_points:
                grad[k] += 1 - h[k + max_shift]
            if 0 <= k - max_shift < n_points:
                grad[k] += 1 - h[k - max_shift]
        
        # Update with momentum
        v = momentum_beta * v - lr * grad
        h_new = h + v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_sum_new = np.sum(h_new)
        if h_sum_new > 0:
            h_new = h_new * (n_points / 2) / h_sum_new
        
        h = h_new
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if max_overlap <= target_bound:
            break
            
        if i % decay_every == 0 and i > 0:
            lr *= 0.9
    
    # Final evaluation
    h_rev = np.flip(best_h)
    overlaps = np.correlate(best_h, h_rev, mode='full') * dx
    final_c5 = np.max(overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
