import numpy as np
import time
from scipy.fftpack import fft, ifft
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Fourier domain optimization
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute FFT
        H = fft(h)
        
        # Compute the spectrum |H|^2
        spectrum = np.abs(H)**2
        
        # Flatten the spectrum (avoid zero division)
        eps = 1e-10
        weight = 1.0 / (spectrum + eps)
        weight = weight / np.sum(weight) * len(weight)
        
        # Update H to flatten spectrum (not exactly flattening but it helps)
        H_new = H * np.sqrt(weight)
        
        # Inverse FFT and project back to feasible region
        h_new = np.real(ifft(H_new))
        h_new = np.clip(h_new, 0, 1)
        
        # Make sure the integral constraint is maintained
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Check if this improves the bound
        c5_new = evaluate_erdos_solution(h_new, dx)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        else:
            h = h_new.copy()
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h_values, dx):
    """Evaluate the C5 constant for a given h function."""
    # The overlap integral is the maximum of the convolution of h and (1-h)
    # We calculate the full convolution and find its maximum value
    n = len(h_values)
    h = np.array(h_values)
    conv = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value of the convolution (but it's a full convolution)
    # The relevant indices are from n-1 to 2*n-2 (inclusive)
    max_idx = np.argmax(conv[n-1:2*n-1])
    c5 = conv[n-1+max_idx] * dx
    return c5
