def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps
    overlap = np.correlate(h_values, 1 - h_values, mode="full")
    # Take maximum over all shifts
    c5 = np.max(overlap) * dx
    return c5
