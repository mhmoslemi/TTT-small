import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple initial guess
        n = 70
        h = np.ones(n) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Warm-up to make sure we have an up-to-date reference
    start_time = time.time()
    
    # Optimize using gradient-based method
    # Use a simple gradient descent approach with projection onto constraints
    
    max_iter = 5000
    tol = 1e-6
    step_size = 0.01
    
    def compute_C5(h):
        # Compute all overlaps using full cross-correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Extract the relevant part (overlap values)
        n_corr = len(corr)
        # The overlap is computed over shifts from -n+1 to n-1
        # We are interested in positive shifts
        offset = n_corr // 2
        overlaps = corr[offset:]  # shifts from 0 to n-1
        # Add a small penalty to ensure c5 is the maximum
        return np.max(overlaps) * dx

    def compute_grad(h, k_shift):
        # Compute gradient of overlap with shift k_shift
        # Boundary handling: wrap around
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_shift) % n_points
            grad[i] = 1 - 2 * h[j]
        return grad

    def project(h):
        # Project h to satisfy constraints: sum(h) = target_sum, h in [0, 1]
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        else:
            h = np.ones_like(h) * (target_sum / n_points)
        return h

    # Optimization loop
    current_time = time.time()
    iteration = 0
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    while current_time - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        # Evaluate current C5 and find maximum overlap shift
        # Compute all overlaps to find worst case
        corr = np.correlate(h, 1 - h, mode='full')
        n_corr = len(corr)
        offset = n_corr // 2
        overlaps = corr[offset:]  # 0 to n-1 shifts
        k_max = np.argmax(overlaps)
        
        # Compute gradient
        grad = compute_grad(h, k_max)
        # Update: gradient descent step
        h_new = h - step_size * grad
        # Project to feasible set
        h_new = project(h_new)
        
        # Check if improvement
        c5_new = compute_C5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new.copy()
        current_time = time.time()
        
    # Final projection to enforce the integral constraint
    h_final = project(best_h)
    c5_final = compute_C5(h_final)
    
    return (h_final.tolist(), c5_final, n_points)
