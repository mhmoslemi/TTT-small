import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum as cp_sum, abs as cp_abs
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vec):
        # Compute overlap integral
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        overlap = corr[len(h_vec)-1:]  # Only the valid overlaps
        return np.max(overlap) * dx
    
    def project(h_vec):
        # Clip and rescale
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if np.abs(current_sum) > 1e-10:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def line_search(h_vec, grad, k_max):
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            return h_vec, 0.0
        grad = grad / grad_norm
        
        # Backtracking line search
        alpha = 0.01
        rho = 0.5
        max_steps = 30
        step_size = 0.0
        
        for _ in range(max_steps):
            new_h = h_vec - alpha * grad
            new_h = project(new_h)
            new_c5 = compute_c5(new_h)
            
            if new_c5 < best_c5:
                step_size = alpha
                break
            
            alpha *= rho
        return project(h_vec - step_size * grad), step_size

    def convex_refinement(h_vec, k_max):
        # Solve convex subproblem with CVXPY
        h_vars = Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp_sum(h_vars) == target_sum]
        
        # Compute full correlation for this k_max
        shifted = np.roll(1 - h_vec, k_max)
        
        # Minimize the overlap with this shift
        objective = cp_sum(cp.multiply(h_vars, shifted))
        prob = Problem(Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.GLPK)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = np.array(h_vars.value).flatten()
                return h_new
        except:
            pass
        return h_vec
    
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        overlap = corr[n_points-1:]
        k_max = np.argmax(overlap)
        c5_current = overlap[k_max] * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement = iteration
        
        # Compute gradient
        shifted = np.roll(1 - h, k_max)
        gradient = shifted - h * shifted
        
        # Line search
        h_new, _ = line_search(h, gradient, k_max)
        
        # Apply update
        h = h_new
        
        # Every 50 iterations, do convex refinement
        if iteration % 50 == 0 and iteration > 0:
            h = convex_refinement(h, k_max)
            h = project(h)
            
        iteration += 1
        
        # Early stopping
        if iteration - last_improvement > 500:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
