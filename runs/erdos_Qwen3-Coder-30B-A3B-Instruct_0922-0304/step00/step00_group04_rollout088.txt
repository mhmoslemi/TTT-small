import numpy as np
from cvxpy import *
import time
from math import floor, ceil

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 42
    dx = 2.0 / n_points
    lag_max = n_points - 1
    
    # Start with initial solution if available
    try:
        h_start = initial_h_values
        if len(h_start) != n_points:
            raise ValueError()
    except:
        h_start = np.random.rand(n_points)
        h_start = h_start / np.sum(h_start) * (n_points / 2)
    
    # Create the optimization problem
    h = Variable(n_points, name='h')
    C5 = Variable(name='C5')
    
    # Constraints
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Add lag constraints
    for k in range(-lag_max, lag_max + 1):
        if k >= 0:
            corr = sum(h[i] * (1 - h[i + k]) for i in range(n_points - k))
        else:
            corr = sum(h[i + (-k)] * (1 - h[i]) for i in range(n_points + k))
        constraints.append(corr <= C5)
    
    # Objective
    objective = Minimize(C5)
    
    # Create problem
    prob = Problem(objective, constraints)
    
    # Set solver parameters for better performance
    solver_args = {
        'solver': 'GLPK_MI',  # Mixed integer solver
        'epsg': 1e-6,
        'max_iters': 10000
    }
    
    # Start timing
    start_time = time.time()
    
    try:
        # Solve the problem
        prob.solve(**solver_args)
        
        # Extract solution
        h_opt = h.value
        c5_opt = C5.value
        
        # Post-processing: convert to binary
        h_binary = np.zeros(n_points)
        # Sort indices by h values in descending order
        order = np.argsort(h_opt)[::-1]
        # Fill with 1s first
        for i in range(n_points):
            if i < n_points // 2:
                h_binary[order[i]] = 1
            else:
                h_binary[order[i]] = 0
                
        # Ensure sum constraint
        ones_count = np.sum(h_binary)
        if ones_count != n_points // 2:
            diff = int(n_points // 2 - ones_count)
            # Adjust to meet constraint
            if diff < 0:
                # Need to flip some 1s to 0s
                for i in range(n_points):
                    if h_binary[i] == 1 and diff < 0:
                        h_binary[i] = 0
                        diff += 1
            else:
                # Need to flip some 0s to 1s
                for i in range(n_points):
                    if h_binary[i] == 0 and diff > 0:
                        h_binary[i] = 1
                        diff -= 1
        
        # Re-evaluate the final cost
        h_final = h_binary
        h_final = h_final.astype(float)
        
        # Use alternative method for better numerical stability
        h_final = np.clip(h_final, 0, 1)
        h_final = np.round(h_final)
        if np.sum(h_final) != n_points / 2:
            # adjust to meet exact constraint
            indices_one = np.where(h_final == 1)[0]
            if len(indices_one) > n_points / 2:
                # too many 1s, set some to 0
                for i in range(len(indices_one) - int(n_points / 2)):
                    h_final[indices_one[i]] = 0
            elif len(indices_one) < n_points / 2:
                # too few 1s, set some to 1
                indices_zero = np.where(h_final == 0)[0]
                for i in range(int(n_points / 2) - len(indices_one)):
                    h_final[indices_zero[i]] = 1
    except:
        h_final = h_start
        h_final = np.clip(h_final, 0, 1)
        
    # Final evaluation of C5
    try:
        # Compute maximum overlap using the evaluate function
        c5_bound_computed = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    except:
        # Fallback to estimate if correlation fails
        c5_bound_computed = 0.5  # Placeholder
        
    if np.isnan(c5_bound_computed) or c5_bound_computed < 0:
        c5_bound_computed = 0.5
        
    return (h_final, float(c5_bound_computed), n_points)
