    # Gradient descent parameters

    rho = 0.5
    eta = 0.5
    max_iter = 1000
    patience = 100
    
    iter_count = 0
    no_improvement_count = 0

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlaps
        full_correlation = np.correlate(h, 1 - h, mode="full")
        overlaps = full_correlation[n_points-1:] * dx
        
        # Find k_max that maximizes overlap
        k_max = np.argmax(overlaps)
        C5_current = overlaps[k_max]
        
        # Early stopping
        if C5_current <= target_bound:
            break
            
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference
            if i + k_max < n_points:
                gradient[i] += (1 - h[i + k_max])
            # Backward difference
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
                
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm > 1e-12:
            gradient /= grad_norm
        else:
            break

        # Line search
        alpha = alpha_initial
        descent = False
        for _ in range(30):  # Max backtrack steps
            h_new = h - alpha * gradient
            # Clip to feasible region
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check if this improves objective
            new_overlaps = np.correlate(h_new, 1 - h_new, mode="full")[n_points-1:] * dx
            new_C5 = new_overlaps[k_max]
            
            if new_C5 < C5_current:
                h = h_new
                descent = True
                break
            else:
                alpha *= rho
                
        if not descent:
            break
            
        iter_count += 1
        
    # Final check with best solution
    final_C5 = evaluate_erdos_solution(best_h) if np.sum(np.abs(best_h - h)) > 1e-10 else evaluate_erdos_solution(h)
