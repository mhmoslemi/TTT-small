import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Use the available initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from the provided construction or create random feasible
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Adjust to correct sum with small perturbation
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            diff = target_sum - current_sum
            h[0] += diff
    else:
        # Random feasible start
        h = np.random.rand(n_points)
        # Normalize to correct sum
        h *= target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    lr = 0.1
    max_iter = 10000
    patience = 50
    patience_counter = 0
    last_loss = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Compute overlap integrals
        corr = correlate(h, 1 - h, mode='full')
        # The paper defines lag k from -n+1 to n-1, so we need to extract the valid range
        # The middle is where k=0, and we work with positive lags and negative lags
        n_corr = len(corr)
        middle_idx = n_corr // 2
        c5_vals = np.zeros(n_corr)
        for k in range(n_corr):
            c5_vals[k] = corr[k] * dx
        
        # Find maximum
        max_idx = np.argmax(c5_vals)
        max_c5 = c5_vals[max_idx]
        max_lag = max_idx - middle_idx
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            
        # Compute gradient for this lag only
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # h[i] * (1 - h[i + max_lag])
            for i in range(len(h) - max_lag):
                grad[i] += (1 - h[i + max_lag]) * dx
            for i in range(max_lag, len(h)):
                grad[i] -= h[i - max_lag] * dx
        else:
            # max_lag < 0
            for i in range(len(h) + max_lag):
                grad[i] -= h[i - max_lag] * dx
            for i in range(-max_lag, len(h)):
                grad[i] += (1 - h[i + max_lag]) * dx

        # Update with learning rate
        h_new = h - lr * grad
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        h = h_new
        
        # Check convergence
        if abs(max_c5 - last_loss) < 1e-8:
            patience_counter += 1
        else:
            patience_counter = 0
        last_loss = max_c5
        
        if patience_counter > patience:
            break
        
    return (best_h.tolist(), best_c5, n_points)
