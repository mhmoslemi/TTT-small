import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple construction
        n_points = 45
        h = np.full(n_points, 0.5)
        h[10:35] = 1.0  # Step function
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum to n_points/2
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_c5(h_vec):
        # Compute cross-correlation with complement
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Extract the valid maximum (k from -n+1 to n-1)
        # The valid range is from n-1 to 2*n-2, so we take indices from n-1 to 2*n-2
        c5_val = np.max(corr[n_points-1:2*n_points-1]) * dx
        return c5_val
    
    def project_h(h_vec):
        """Project h to feasible region: [0,1] and sum = n_points/2"""
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization parameters
    max_iter = 100000
    delta = 0.1
    decay_factor = 0.9
    decay_interval = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        h_try = h.copy()
        
        # Select two random indices
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Perturbation
        perturbation = delta * np.random.choice([-1, 1])
        
        # Ensure we don't go out of bounds
        if h_try[idx1] + perturbation >= 0 and h_try[idx1] + perturbation <= 1:
            h_try[idx1] += perturbation
            h_try[idx2] -= perturbation
            # Project to feasible region
            h_try = project_h(h_try)
            
            new_c5 = compute_c5(h_try)
            if new_c5 < best_c5:
                h = h_try.copy()
                best_c5 = new_c5
                best_h = h.copy()
        
        # Decrease delta occasionally
        if (i + 1) % decay_interval == 0:
            delta *= decay_factor
            
    return (best_h.tolist(), best_c5, n_points)
