import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize with a simple construction
        h = np.zeros(n_points)
        # Place some mass in the middle
        mid = n_points // 2
        h[mid - 5:mid + 5] = 1.0
        # Normalize to correct sum
        h *= target_sum / np.sum(h)
    
    # Optimization variables
    temp_max = 1.0
    temp_min = 1e-4
    iter_max = 10000
    alpha = np.exp(np.log(temp_min / temp_max) / iter_max)
    
    # Adaptive perturbation scale
    perturb_scale = 0.1
    accept_count = 0
    accept_threshold = 0.1
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Timing
    start_time = time.time()
    
    for iteration in range(iter_max):
        if time.time() - start_time > budget_s:
            break
            
        # Make a small perturbation
        h_new = h.copy()
        perturb_type = random.choice(['swap', 'shift'])
        
        if perturb_type == 'swap':
            # Pick two random indices and swap values
            i, j = random.sample(range(n_points), 2)
            h_new[i], h_new[j] = h_new[j], h_new[i]
        else:
            # Adjacent shift
            i = random.randint(0, n_points - 2)
            # Move some small amount from i to i+1
            delta = random.uniform(-perturb_scale, perturb_scale)
            if 0 <= h_new[i] - delta <= 1 and 0 <= h_new[i+1] + delta <= 1:
                h_new[i] -= delta
                h_new[i+1] += delta
        
        # Clamp to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new *= target_sum / current_sum
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        delta_c5 = c5_new - best_c5
        temp = temp_max * (alpha ** iteration)
        
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / temp):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        # Adaptive perturbation scale
        accept_count += (delta_c5 < 0) or (random.random() < np.exp(-delta_c5 / temp))
        if iteration > 0 and iteration % 100 == 0:
            accept_rate = float(accept_count) / 100.0
            if accept_rate < accept_threshold:
                perturb_scale *= 0.95
            elif accept_rate > 0.2:
                perturb_scale *= 1.05
            accept_count = 0
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Mirror the function for convolution
    h_mirrored = np.flip(1 - h)
    # Cross-correlation with full mode
    corr = signal.correlate(h, h_mirrored, mode="full")
    # The maximum value in the cross-correlation times dx
    max_corr = np.max(corr)
    return max_corr * dx
