from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

    def compute_C5(h_vals):
        # Compute overlap integral for all shifts
        h_shifted = np.concatenate([h_vals, np.zeros(n_points)])
        overlaps = []
        for k in range(n_points):
            shifted = h_shifted[k:k+n_points]
            overlap = np.sum(h_vals * (1 - shifted)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    def project_feasible(h_vals):
        # Ensure h_vals is in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h_vals) * dx
        desired_sum = 1.0
        adjustment = (desired_sum - current_sum) / n_points
        h_vals += adjustment
        h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    def solve_subproblem(h_vals, max_k):
        # Locate the worst shift k
        h_shifted = np.concatenate([h_vals, np.zeros(n_points)])
        overlap_vals = []
        for k in range(n_points):
            shifted = h_shifted[k:k+n_points]
            overlap = np.sum(h_vals * (1 - shifted)) * dx
            overlap_vals.append(overlap)
        
        if max_k is None:
            max_k = np.argmax(overlap_vals)
        
        # Construct the objective: minimize overlap at max_k
        # This is a linear objective
        def obj(h_new):
            shifted = np.concatenate([h_new, np.zeros(n_points)])[max_k:max_k+n_points]
            return np.sum(h_new * (1 - shifted)) * dx
        
        # Optimization with bounds and sum constraint
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Additional constraint: sum(h) = n_points / 2
        def sum_constraint(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': sum_constraint}

        try:

            res = minimize(obj, h_vals, method='SLSQP', bounds=bounds, constraints=cons,

                new_h = res.x
                new_h = project_feasible(new_h)
                return new_h

            pass
        
        # Fallback to simple projection if optimization fails
        return project_feasible(h_vals)
