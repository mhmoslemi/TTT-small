def evaluate_erdos_solution(h_values):
    """Evaluate the Erdős constant C₅ for given step function h."""

    # Compute all correlations
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Extract the relevant part
    max_corr = np.max(corr)
    return max_corr * dx
