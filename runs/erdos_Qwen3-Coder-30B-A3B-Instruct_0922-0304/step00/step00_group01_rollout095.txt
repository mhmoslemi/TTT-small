import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    # Adjust to satisfy sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Initialization
    lambda_lag = 0.0
    alpha = 0.01
    max_iter = 1000
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    
    # For tracking and early termination
    last_improvement = 0
    patience = 100
    min_alpha = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:  # Leave 1s for cleanup
            break
            
        # Compute all overlaps
        overlap_values = []
        for k in range(n_points):
            # Compute correlation h * (1 - h_k)
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1.0 - shifted_h))
            overlap_values.append(overlap)
        
        # Find maximum overlap
        max_k = np.argmax(overlap_values)
        max_overlap = overlap_values[max_k]
        
        # Check for improvement
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iteration
        
        # Check termination criteria
        if max_overlap < 0.3807:
            break
        
        if iteration - last_improvement > patience:
            break
            
        if alpha < min_alpha:
            break
        
        # Compute subgradient for max_k
        g = np.zeros(n_points)
        shifted_h = np.roll(h, max_k)
        for i in range(n_points):
            g[i] = (1.0 - shifted_h[i]) - h[i] + lambda_lag
        
        # Update h
        h = h - alpha * g
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust to maintain sum constraint
        scale = (n_points / 2.0) / np.sum(h)
        h = h * scale
        
        # Update Lagrange multiplier
        lambda_lag = lambda_lag + alpha * (np.sum(h) - n_points / 2.0)
        
        # Adaptive step-size adjustment
        if iteration > 0 and max_overlap >= overlap_values[np.argmax(overlap_values)]:
            alpha *= 0.5
        alpha = max(alpha, min_alpha)
    
    # Final evaluation
    final_overlap_values = []
    for k in range(n_points):
        shifted_h = np.roll(best_h, k)
        overlap = np.sum(best_h * (1.0 - shifted_h))
        final_overlap_values.append(overlap)
    final_c5 = max(final_overlap_values) * 2.0 / n_points
    
    return (best_h.tolist(), final_c5, n_points)
