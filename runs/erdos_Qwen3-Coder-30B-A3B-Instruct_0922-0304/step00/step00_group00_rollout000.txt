import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure h sums to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 50000
    alpha = 0.1
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all shifts
        conv = signal.correlate(h, 1 - h, mode='full')
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = conv * dx
        
        # Find the shift k that maximizes overlap
        k_max = shifts[np.argmax(overlaps)]
        
        # Compute gradient w.r.t. this k
        grad = np.zeros(n_points)
        
        # For the maximum overlap shift k_max,
        # update gradient: each h[i] contributes 1-h[i+k] and h[i+k] contributes -h[i]
        for i_idx in range(n_points):
            j_idx = i_idx + k_max
            if 0 <= j_idx < n_points:
                grad[i_idx] += (1 - h[j_idx]) * dx
                grad[j_idx] -= h[i_idx] * dx
        
        # Update h
        h = h - alpha * grad
        
        # Ensure bounds [0,1]
        h = np.clip(h, 0, 1)
        
        # Renormalize h to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute new C5
        conv_new = signal.correlate(h, 1 - h, mode='full')
        shifts_new = np.arange(-n_points + 1, n_points)
        overlaps_new = conv_new * dx
        new_c5 = np.max(overlaps_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = i
            
        if i - last_improvement > 50:
            # Try a different learning rate
            alpha *= 0.95
            last_improvement = i
            
        if best_c5 < 0.3807:
            break
    
    # Final validation of the best solution
    conv_final = signal.correlate(best_h, 1 - best_h, mode='full')
    shifts_final = np.arange(-n_points + 1, n_points)
    overlaps_final = conv_final * dx
    final_c5 = np.max(overlaps_final)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
