import time
from scipy.signal import correlate

    max_iter = 10000
    alpha = 0.01
    min_alpha = 1e-5
    patience = 100
    target_c5 = 0.3807

    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Lagrange multiplier
    lambda_ = 0.0
    
    # Time tracking
    start_time = time.time()
    
    # For detecting stagnation
    last_improvement = 0
    prev_max_overlap = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all shifts
        # Using correlate for efficiency
        h_padded = np.concatenate([h, h])  # double length for circular correlation
        overlaps = np.zeros(n_points)
        
        for k in range(n_points):
            # Compute correlation for shift k
            # h[i] * (1 - h[(i-k) % n])
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h))
            overlaps[k] = overlap
        
        # Find maximum overlap
        max_overlap_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_overlap_idx]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= target_c5:
            break
            
        # Compute subgradient
        shifted_h = np.roll(h, max_overlap_idx)
        g = (1 - shifted_h) - h + lambda_
        
        # Update h
        h = h - alpha * g
        
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Update Lagrange multiplier
        lambda_ = lambda_ + alpha * (h_sum - n_points / 2)
        
        # Adapt step size
        if max_overlap >= prev_max_overlap:
            alpha *= 0.5
            if alpha < min_alpha:
                alpha = min_alpha
        else:
            alpha = min(alpha * 1.05, 0.1)
            
        prev_max_overlap = max_overlap
        last_improvement += 1
        
        if last_improvement > patience:
            break
    
    # Compute final C5 using evaluation function
    h_final = best_h
    # Trim to n_points if needed
    h_final = np.array(h_final[:n_points])
    # Ensure sum constraint
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    c5_bound = max(np.correlate(h_final, 1-h_final, mode="full")) * 2.0 / n_points
    
    # Ensure final constraints
    assert np.all(h_final >= 0) and np.all(h_final <= 1)
    assert abs(np.sum(h_final) - n_points / 2) < 1e-6
    
    return (h_final.tolist(), c5_bound, n_points)
