import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define constants
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with alternating 0.5s
        h = np.ones(n_points) * 0.5
        h[::2] = 0.0
    
    # Normalize to integral constraint
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Clip values and rescale if needed
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Gradient Descent parameters
    learning_rate = 0.01
    momentum = 0.9
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute correlations for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        # Shifted indices correspond to [0, 2*n_points - 2]
        # Extract the valid overlaps, indexed by k = 0, ..., n_points-1
        corr_values = []
        for k in range(n_points):
            # The overlap for shift k is in corr[n_points-1+k]
            # Positive shift: h[i] * (1 - h[i+k]) 
            # Negative shift: h[i+|k|] * (1 - h[i])
            # Full correlation includes all possible shifts
            corr_shift = corr[n_points-1+k]
            corr_values.append(corr_shift * dx)
        
        # Find maximum overlap index
        max_corr_index = np.argmax(corr_values)
        max_corr_value = corr_values[max_corr_index]
        
        if max_corr_value < best_c5:
            best_c5 = max_corr_value
            best_h = h.copy()
        
        if max_corr_value <= target_bound:
            break
            
        # Compute gradient w.r.t max overlap
        grad = np.zeros_like(h)
        # The maximum overlap comes from shift k = max_corr_index
        k = max_corr_index
        # For positive shifts, h[i] * (1 - h[i + k])  
        # So gradient w.r.t h[i] = 1 - h[i + k] for i = 0, ..., n_points - k - 1
        if k < n_points:
            for i in range(n_points - k):
                grad[i] += 1 - h[i + k]
        # For negative shifts, h[i - k] * (1 - h[i]) for i = k, ..., n_points - 1
        # So gradient w.r.t h[i] = -h[i - k] for i = k, ..., n_points - 1
        if k > 0:
            for i in range(k, n_points):
                grad[i] -= h[i - k]
                
        # Update with momentum (gradient descent with momentum)
        v = momentum * v - learning_rate * grad
        h = h + v
        
        # Projection onto feasibility set [0,1]
        h = np.clip(h, 0, 1)
        
        # Enforce integral constraint: sum(h) * dx = 1
        h_sum = np.sum(h) * dx
        h *= (1.0 / h_sum)  # Rescale to target integral
        h = np.clip(h, 0, 1)
        
        # Final rescale to exact integral
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        iter_count += 1
        
        # Decay learning rate
        if iter_count % 100 == 0:
            learning_rate *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)
