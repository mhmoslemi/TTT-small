import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Repair sum constraint if needed
        current_sum = np.sum(h_init)
        if abs(current_sum - target_sum) > 1e-6:
            # Flip some entries to adjust sum
            diff = int(target_sum - current_sum)
            if diff > 0:
                # Need to increase sum by diff
                indices = np.argsort(h_init)
                flip_indices = indices[:diff]
                h_init[flip_indices] = 1.0
            elif diff < 0:
                # Need to decrease sum by -diff
                indices = np.argsort(1 - h_init)
                flip_indices = indices[:(-diff)]
                h_init[flip_indices] = 0.0
    else:
        # Initialize simple construction
        h_init = np.ones(n_points) * 0.5
        # Adjust to sum constraint
        h_init[:int(target_sum)] = 1.0
        h_init[int(target_sum):] = 0.0
    
    # Binary search setup
    t_low = 0.0
    t_high = 0.5
    tolerance = 1e-4
    start_time = time()

    def solve_feasibility(t_val):
        h = cp.Variable(n_points)
        # Binary constraints with relaxed bounds
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == target_sum
        ]
        
        # For all possible shifts k from -n_points+1 to n_points-1
        max_shift = n_points - 1
        for k in range(-max_shift, max_shift + 1):
            if k >= 0:
                # Indices for h[i] and h[i+k]
                valid_indices = range(n_points - k)
                h1 = h[valid_indices]
                h2 = h[valid_indices + k]
                # Approximate h*x constraints by enforcing h1 and h2 are bounded by h
                w = cp.Variable(len(valid_indices))
                # w[i] = h[i] * h[i + k] (approximate)
                constraints.extend([
                    w <= h1,
                    w <= h2,
                    cp.sum(h1) - cp.sum(w) <= t_val
                ])
            else:
                # k < 0
                valid_indices = range(-k, n_points)
                h1 = h[valid_indices]
                h2 = h[valid_indices - k]
                w = cp.Variable(len(valid_indices))
                constraints.extend([
                    w <= h1,
                    w <= h2,
                    cp.sum(h1) - cp.sum(w) <= t_val
                ])
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
            return prob.status == cp.OPTIMAL or prob.status == cp.INFEASIBLE or prob.status == cp.UNBOUNDED
        except:
            return False
            
    # Binary search
    while time() - start_time < budget_s * 0.95 and abs(t_high - t_low) > tolerance:
        t_mid = (t_low + t_high) / 2
        if solve_feasibility(t_mid):
            t_high = t_mid
        else:
            t_low = t_mid
            
        if time() - start_time > budget_s * 0.9:
            break
            
    # Final optimization using rounded solution from binary search
    def evaluate_solution(h_vec):
        h_vec = np.array(h_vec)
        # Compute correlations with 1-h
        conv_result = np.correlate(h_vec, 1-h_vec, mode='full')
        # Extract valid correlations
        valid_peak = max(conv_result)
        return valid_peak * dx

    # Use projected gradient or local search
    h_final = h_init.copy()
    # Use a simple heuristic to improve initial guess
    for _ in range(50):
        if time() - start_time > budget_s * 0.95:
            break
        # Evaluate
        c5 = evaluate_solution(h_final)
        if c5 <= t_high:
            break
            
        # Small adjustment to improve
        # Quick fix: adjust some extreme values
        for i in range(n_points):
            if h_final[i] < 1e-2 and h_final[i] > 0:
                h_final[i] = 0.0
            elif h_final[i] > 1 - 1e-2 and h_final[i] < 1:
                h_final[i] = 1.0
        
        # Re-adjust sum constraint
        current_sum = np.sum(h_final)
        diff = current_sum - target_sum
        if abs(diff) > 1e-4:
            # Find indices to flip
            if diff > 0:
                # Need to decrease sum
                flip_indices = np.argsort(h_final)[-int(diff):]
                h_final[flip_indices] = 0.0
            else:
                # Need to increase sum
                flip_indices = np.argsort(1-h_final)[-int(-diff):]
                h_final[flip_indices] = 1.0
                
    # Refine final one more time
    c5_final = evaluate_solution(h_final)
    
    # Possibly round to binary values if needed
    h_result = np.round(h_final).clip(0, 1)
    # Repair sum
    current_sum = np.sum(h_result)
    diff = current_sum - target_sum
    if abs(diff) > 1e-6:
        indices = np.argsort(1 - h_result)
        if diff > 0:
            flip_indices = indices[:int(diff)]
            h_result[flip_indices] = 0.0
        else:
            flip_indices = indices[:int(-diff)]
            h_result[flip_indices] = 1.0
    
    return (h_result.tolist(), c5_final, n_points)
