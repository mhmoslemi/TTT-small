import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Gradient descent
    alpha = 0.01
    beta = 0.9  # momentum
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        # Compute all correlations
        h_padded = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h = 1 - h
        one_minus_h_padded = np.pad(one_minus_h, (0, n_points-1), mode='constant', constant_values=0)
        
        # Cross-correlation
        corr = correlate(h_padded, one_minus_h_padded, mode='full')
        corr_vals = corr[n_points-1:]  # Remove padding
        c5_val = np.max(corr_vals) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        if c5_val <= target_bound:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            if k < n_points:
                grad += np.convolve(h, one_minus_h, mode='valid')[k] * np.ones(n_points-k) * dx
            else:
                # This part needs careful handling of the correlation
                pass
        
        # Simplified gradient approximation
        # Use the element-wise gradient based on positions of maximum correlation
        max_k = np.argmax(corr_vals)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = i + max_k
            if j < n_points:
                grad[i] += (1 - h[j]) * dx
            j = i - max_k
            if j >= 0:
                grad[i] += (1 - h[j]) * dx
        
        # Momentum update
        v = beta * v + (1 - beta) * grad
        h_new = h - alpha * v
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to enforce constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        iter_count += 1
        if iter_count % 100 == 0:
            alpha *= 0.9  # Decay learning rate
            
    return (best_h.tolist(), best_c5, n_points)
