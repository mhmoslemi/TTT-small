import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # Only consider the relevant part (k from 0 to n-1)
    c5 = np.max(overlaps[n-1:]) * dx
    return c5

def ensure_constraints(h):
    """Ensure h is feasible: 0 <= h <= 1 and sum(h) = n/2."""
    n = len(h)
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to exactly n/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

def mutate(h, mutation_rate=0.01):
    """Apply mutation to a candidate h."""
    n = len(h)
    # Randomly select indices to mutate
    num_mutations = int(n * mutation_rate)
    indices = np.random.choice(n, size=num_mutations, replace=False)
    # Add small random perturbations
    for idx in indices:
        h[idx] += np.random.normal(0, 0.05)
    # Ensure constraints
    return ensure_constraints(h)

def crossover(h1, h2):
    """Perform crossover between two candidates."""
    n = len(h1)
    # Select a random crossover point
    cross_point = np.random.randint(1, n)
    # Create children
    child1 = np.concatenate([h1[:cross_point], h2[cross_point:]])
    child2 = np.concatenate([h2[:cross_point], h1[cross_point:]])
    # Ensure constraints
    child1 = ensure_constraints(child1)
    child2 = ensure_constraints(child2)
    return child1, child2

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = ensure_constraints(h0)
        population.append(h0)
    else:
        # Use a simple construction
        h0 = np.full(n_points, 0.5)
        h0 = ensure_constraints(h0)
        population.append(h0)
    
    # Fill the rest with random feasible solutions
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = ensure_constraints(h)
        population.append(h)
    
    # Genetic algorithm
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s and no_improvement_count < max_no_improvement:
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness_scores.append(1.0 / (c5 + 1e-10))  # Avoid division by zero
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for stagnation
        if no_improvement_count == 0:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Sort population by fitness (descending)
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        
        # Select top 50% for breeding
        selected_size = population_size // 2
        selected = population[:selected_size]
        
        # Generate new population
        new_population = selected.copy()
        
        # Create offspring through crossover and mutation
        for i in range(0, len(selected), 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % len(selected)]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        # Trim to population size
        population = new_population[:population_size]
        
        no_improvement_count += 1
    
    if best_h is None:
        # Fallback
        best_h = population[0]
        best_c5 = evaluate_c5(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
