import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Variable, Minimize, Problem, sum_entries, multiply, sqrt, norm
import cvxpy as cp

def evaluate_erdos_solution(h_vals, dx):
    n = len(h_vals)
    h = np.array(h_vals)
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise create uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    alpha = 0.1
    max_iter = 5000
    stagnation_count = 0
    last_best = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Update current evaluation
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > 50:
            alpha *= 0.5
            stagnation_count = 0
            
        # Compute overlaps for all shifts
        h_vec = np.array(h)
        overlaps = np.correlate(h_vec, 1 - h_vec, mode="full")
        k_max = np.argmax(overlaps) - (n_points - 1)
        
        # Compute gradient for the worst-case shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Gradient step
        h = h - alpha * grad
        
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Convex refinement step every 5 iterations
        if iteration % 5 == 4:
            # Formulate and solve QP for refining at worst shift k_max
            h_new = h.copy()
            try:
                # Calculate x = h_{k_max} using corresp. indices
                x = np.zeros(n_points)
                for i in range(n_points):
                    j = (i + k_max) % n_points
                    x[i] = h[j]
                
                # Define variables for QP
                w = Variable(n_points)
                
                # Objective function: minimize sum of w_i
                objective = Minimize(sum_entries(w))
                
                # Constraints
                constraints = [
                    w >= 0,
                    w <= 1,
                    sum_entries(w) == n_points / 2,
                    multiply(h, 1 - x) >= w  # element-wise
                ]
                
                # Solve problem
                prob = Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_new = np.array(w.value).flatten()
                    h_new = np.clip(h_new, 0, 1)
                    
                h = h_new
                h = h / np.sum(h) * (n_points / 2)
                
            except:
                pass  # if QP fails, continue with gradient descent

    # Final validation check
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
