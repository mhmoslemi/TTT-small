import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlaps = []
    for k in range(-n_points + 1, n_points):
        shift = np.roll(h, k)
        overlap = np.sum(h * (1 - shift)) * dx
        overlaps.append(overlap)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()

    def optimize_step():
        nonlocal h, best_h, best_c5
        if time.time() - start_time > budget_s * 0.9:
            return
        # Find worst shift
        overlaps = []
        for k in range(-n_points + 1, n_points):
            shift = np.roll(h, k)
            overlap = np.sum(h * (1 - shift)) * dx
            overlaps.append(overlap)
        max_k = np.argmax(overlaps) - (n_points - 1)
        max_overlap = max(overlaps)
        # Gradient descent step
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        # Adaptive step size
        alpha = 0.1
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        # QP refinement
        if time.time() - start_time > budget_s * 0.9:
            return
        h_refined = h_new.copy()
        try:
            # Initialize QP variables
            x = Variable(n_points)
            # Objective: minimize sum of x[i]
            objective = Minimize(cp.sum(x))
            # Constraints
            constraints = []
            for i in range(n_points):
                j = (i + max_k) % n_points
                constraints.append(x[i] >= h[i] * (1 - h[j]) * dx)
            constraints.append(cp.sum(x) == n_points / 2)
            constraints.append(x >= 0)
            constraints.append(x <= 1)
            # Solve QP
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                x_vals = x.value
                if x_vals is not None:
                    h_refined = np.clip(x_vals, 0, 1)
            else:
                h_refined = h_new
        except Exception:
            h_refined = h_new
        # Refine and validate
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        new_c5 = evaluate_erdos_solution(h_refined, n_points)
        if new_c5 < best_c5:
            best_h = h_refined.copy()
            best_c5 = new_c5
    
    # Iterate
    iteration = 0
    while time.time() - start_time < budget_s * 0.9:
        optimize_step()
        iteration += 1
        if iteration % 20 == 0:
            break

    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
