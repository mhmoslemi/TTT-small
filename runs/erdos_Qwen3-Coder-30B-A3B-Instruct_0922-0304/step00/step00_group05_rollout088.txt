import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C₅ for given step function h."""
    # Compute overlap integrals for all lags
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlaps)

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()

    # Parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000
    last_improvement = 0
    
    start_time = time.time()
    
    # Optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break

        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        
        # Find max overlap and corresponding lags
        max_overlap = np.max(overlaps)
        max_lags = np.where(np.abs(overlaps - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # Overlap[i] = h[i] * (1 - h[i + lag]) * dx
                # Gradient w.r.t. h[i] is (1 - h[i + lag]) * dx when lag > 0
                # But should include lag-specific gradient
                for i in range(n_points - lag):
                    gradient[i] += (1 - h[i + lag]) * dx
        
        # Normalize gradient by number of max lags
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to correct integral
        h *= (n_points / 2) / np.sum(h)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iteration
            
        # Adapt learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
    
    return (best_h.tolist(), best_c5, n_points)
