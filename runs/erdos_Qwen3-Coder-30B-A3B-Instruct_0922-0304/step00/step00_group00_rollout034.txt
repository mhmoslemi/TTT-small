import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    overlap = correlate(h, 1 - h, mode='full')
    # Take the maximum over all shifts
    max_overlap = np.max(overlap)
    return max_overlap * dx
