import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    
    # Normalize initial h
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Precompute all possible shifts
    shifts = list(range(1, n))
    
    # Track iterations without improvement
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < 0.95 * budget_s:
        # Compute overlaps for all shifts
        overlaps = np.zeros(len(shifts))
        for i, k in enumerate(shifts):
            overlap = np.sum(h * np.roll(1 - h, k))
            overlaps[i] = overlap
            
        # Find shift with maximum overlap
        k_max_idx = np.argmax(overlaps)
        k_max = shifts[k_max_idx]
        max_overlap = overlaps[k_max_idx]
        
        # If we haven't improved for too long, break
        if no_improvement_count > max_no_improvement:
            break
            
        # Define the quadratic constraint for the max overlap shift
        # The constraint is sum_i h[i] * (1 - h[(i + k_max) % n]) <= C5
        # This is equivalent to sum_i h[i] - sum_i h[i] * h[(i + k_max) % n] <= C5
        
        # Solve optimization problem
        h_var = cp.Variable(n)
        C5_var = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(C5_var)
        
        # Constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n / 2,
            cp.sum(h_var * (1 - cp.roll(h_var, k_max))) <= C5_var
        ]
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n / 2) / np.sum(new_h)
                
                # Compute new C5
                new_c5 = evaluate_erdos_solution(new_h)
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                h = new_h.copy()
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1
        
        # Early termination check
        if best_c5 < 0.3807:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_erdos_solution(h):
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    
    # Compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode='full')
    
    # Get the maximum from the full correlation
    # The valid indices in the full correlation array
    max_idx = len(corr) // 2
    max_corr = corr[max_idx]
    
    # For each valid shift from the leftmost connection
    min_corr = 0
    for i in range(1, n):
        corr_val = corr[max_idx + i]
        if corr_val > min_corr:
            min_corr = corr_val
    
    return min_corr * dx
