import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 50000
    alpha = 0.1
    patience = 50
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Ensure initial h is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        # Use full convolution to get all overlaps
        correlation = signal.correlate(h, 1 - h, mode='full')
        # The overlap for shift k is at index k + n - 1
        overlaps = correlation * dx
        # We want max over all valid shifts
        # shifts go from -(n-1) to n-1, but we only want k in [0, n-1] 
        # since the problem is about positive shifts (we have h(x) and h(x+k))
        # Actually let's reconsider the indexing
        # correlation size is 2*n - 1
        # Normalized: correlation[i] corresponds to shift i - (n-1) = i - n + 1
        # So for positive shifts: k = [0, n-1] corresponds to indices [n-1, 2*n-2]
        shifts = np.arange(-n_points + 1, n_points)
        support = np.arange(n_points)
        overlap_vals = []
        for k in range(-n_points + 1, n_points):
            # Find the index in correlations for shift k
            index = k + n_points - 1
            if 0 <= index < len(correlation):
                overlap_vals.append(overlaps[index])
            else:
                overlap_vals.append(0)
        overlap_vals = np.array(overlap_vals)
        
        k_max = np.argmax(overlap_vals)
        max_overlap = overlap_vals[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if best_c5 < target_c5 or patience_counter >= patience:
            break
            
        # Compute gradient w.r.t h for the maximum overlap shift
        grad = np.zeros(n_points)
        k = k_max - (n_points - 1)  # the actual shift
        
        # Gradient calculation:
        # For the case where we do h[i] * (1 - h[i+k]), 
        # we're interested in partial derivatives:
        # d/dh[i] = (1 - h[i+k]) if it's valid,  
        # d/dh[i+k] = -h[i] if it's valid.
        
        for i in range(n_points):
            # Add the gradient from the direct term h[i] * (1 - h[i+k])
            if i + k >= 0 and i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
                grad[i + k] -= h[i] * dx

        # Update h
        h -= alpha * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        # Re-normalize to ensure sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    if best_c5 < float('inf'):
        h = best_h
        # Recompute C5 to ensure accuracy
        correlation = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlation * dx
        final_c5 = np.max(overlaps)
    else:
        # Final evaluation of the final solution
        correlation = signal.correlate(h, 1 - h, mode='full')
        overlaps = correlation * dx
        final_c5 = np.max(overlaps)
    
    return (h.tolist(), final_c5, n_points)
