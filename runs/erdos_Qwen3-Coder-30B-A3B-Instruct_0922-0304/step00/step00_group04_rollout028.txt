import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values or create a random feasible h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure h is in [0,1] and sum equals n_points/2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_c5(h, dx):
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Gradient-based optimization
    max_iter = 1000
    lr = 0.01
    
    for i in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        k = max_idx - (n_points - 1)
        
        # Compute gradient contribution for this specific lag k
        # This is a simplified gradient approximation
        grad = np.zeros_like(h)
        # Forward difference approximation
        if k >= 0:
            grad[k:] += (1 - h[:-k]) / dx
            grad[:-k] += -h[k:] / dx
        else:
            grad[:k] += (1 - h[-k:]) / dx
            grad[-k:] += -h[:k] / dx
        
        # Update h using gradient descent
        h = h - lr * grad
        h = project_h(h)
        
        # Compute new C5 value
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final projection to be sure
    best_h = project_h(best_h)
    best_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, len(best_h))
