import numpy as np
from scipy import optimize
import time


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def eval_C5(h_vec):
        # Compute all overlaps
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = corr[n_points-1:] * dx
        return np.max(overlaps)
    
    def project_h(h_vec):
        # Ensure h is in [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to meet sum constraint
        current_sum = np.sum(h_vec)
        if abs(current_sum) > 1e-10:
            h_vec = h_vec * (target_sum / current_sum)
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient of C(k_max) w.r.t. h
        grad = np.zeros_like(h_vec)
        # For each position, calculate the gradient
        for i in range(len(h_vec)):
            # Contribution from the first term (1 - h[i+k_max])
            if i + k_max < len(h_vec):
                grad[i] += -1.0
            # Contribution from h[i] * (1 - h[i-k_max])
            if i - k_max >= 0:
                grad[i] += h_vec[i - k_max]
        return grad
    
    def line_search(h_vec, grad, k_max, alpha_init=0.01, rho=0.5, c=0.5):
        # Backtracking line search
        alpha = alpha_init
        current_C5 = eval_C5(h_vec)
        for _ in range(30):  # max backtracking steps
            h_new = h_vec - alpha * grad
            h_new = project_h(h_new)
            new_C5 = eval_C5(h_new)
            # Armijo condition (simplified)
            if new_C5 <= current_C5 - c * alpha * np.dot(grad, grad):
                return h_new, alpha
            alpha *= rho
        return h_vec, alpha_init * 0.1  # fallback
    
    # Main optimization loop
    best_C5 = eval_C5(h)
    best_h = h.copy()
    
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1.0 and no_improvement_count < max_no_improvement:
        iteration += 1
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        current_C5 = overlaps[k_max]
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Early stop if target achieved
        if best_C5 < 0.3807:
            break
        
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Apply line search
        h_new, alpha_used = line_search(h, grad, k_max)
        
        # Break if no progress
        if np.allclose(h_new, h, atol=1e-6):
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            h = h_new
        
        # Ensure feasibility
        h = project_h(h)
    
    # Final check of best solution
    final_C5 = eval_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
