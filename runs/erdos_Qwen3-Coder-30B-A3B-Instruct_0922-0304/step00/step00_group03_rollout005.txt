import numpy as np
import time
from collections import deque

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    
    def evaluate(h):
        # This is the fast correlation evaluation engine
        h_rev = np.flip(h)
        corr = np.convolve(h, h_rev, mode='full')
        # Compute 1-h values only once efficiently
        one_minus_h = 1 - h
        corr_full = np.convolve(h, one_minus_h, mode='full')
        max_overlap = np.max(corr_full) * dx
        return max_overlap
    
    def repair_h(h, n_points):
        """Make sure sum is exactly n_points/2 and values are 0 or 1"""
        # Ensure binary
        h = np.round(h).astype(int)
        # Adjust sum to be exactly n_points//2
        current_sum = np.sum(h)
        target_sum = n_points // 2
        diff = target_sum - current_sum
        
        if diff > 0:
            # Need to add more 1s
            indices = np.where(h == 0)[0]
            if len(indices) >= diff:
                h[indices[:diff]] = 1
        elif diff < 0:
            # Need to remove 1s
            indices = np.where(h == 1)[0]
            if len(indices) >= -diff:
                h[indices[:(-diff)]] = 0
                
        return h
    
    def generate_neighbor(h):
        # Flip 2 or 3 bits
        new_h = h.copy()
        n_flip = np.random.choice([2, 3])
        positions = np.random.choice(n_points, size=n_flip, replace=False)
        for pos in positions:
            new_h[pos] = 1 - new_h[pos]  # Flip bit
        # Repair to ensure correct sum
        new_h = repair_h(new_h, n_points)
        return new_h
    
    # Tabu search parameters
    tabu_list = deque(maxlen=100)
    max_iter = 10000
    early_terminate_iter = max_iter // 4
    
    current_h = h.copy()
    current_val = evaluate(current_h)
    best_h = current_h.copy()
    best_val = current_val
    no_improvement_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        neighbor = generate_neighbor(current_h)
        
        # Check if neighbor is in tabu list
        neighbor_tuple = tuple(neighbor)
        if neighbor_tuple in tabu_list:
            continue
        
        neighbor_val = evaluate(neighbor)
        
        # Update tabu list
        tabu_list.append(tuple(current_h))
        
        if neighbor_val < current_val:
            current_h = neighbor.copy()
            current_val = neighbor_val
            if neighbor_val < best_val:
                best_h = neighbor.copy()
                best_val = neighbor_val
                no_improvement_counter = 0
            else:
                no_improvement_counter += 1
        else:
            no_improvement_counter += 1
            
        # Early termination if no improvement in last portion of iterations
        if no_improvement_counter >= early_terminate_iter:
            break
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
