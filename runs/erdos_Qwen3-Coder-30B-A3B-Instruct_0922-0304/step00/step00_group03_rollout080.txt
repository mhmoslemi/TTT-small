import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the C5 bound for given h"""
    # Compute overlap integral for all k values
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx for continuous approximation
    max_corr = np.max(corr) * dx
    return max_corr

def ensure_sum_constraint(h, target_sum):
    """Ensure h has exactly target_sum ones, maintaining feasibility"""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    
    diff = int(target_sum - current_sum)
    
    if diff > 0:  # Need to set more zeros to ones
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            for i in range(diff):
                h[zero_indices[i]] = 1
    elif diff < 0:  # Need to set more ones to zeros
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= abs(diff):
            for i in range(abs(diff)):
                h[one_indices[i]] = 0
    
    return h

def generate_neighbor(h, target_sum, n_points):
    """Generate a neighboring solution by flipping 2 bits"""
    h_new = h.copy()
    
    # Select 2 random indices to flip
    indices = np.random.choice(n_points, 2, replace=False)
    
    # Ensure we maintain exact sum constraint
    if h_new[indices[0]] == h_new[indices[1]]:
        # If same values, flip one to the other value
        if h_new[indices[0]] == 1:
            h_new[indices[0]] = 0
            h_new[indices[1]] = 0
        else:
            h_new[indices[0]] = 1
            h_new[indices[1]] = 1
    else:
        # Simple flip
        h_new[indices[0]] = 1 - h_new[indices[0]]
        h_new[indices[1]] = 1 - h_new[indices[1]]
    
    # Ensure sum constraint
    h_new = ensure_sum_constraint(h_new, target_sum)
    return h_new
