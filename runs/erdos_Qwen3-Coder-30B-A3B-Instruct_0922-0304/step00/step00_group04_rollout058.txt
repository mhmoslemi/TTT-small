import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlations for all lags
    correlations = np.correlate(h, 1 - h, mode="full")
    # Return the maximum correlation scaled by dx
    return np.max(correlations) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Simple uniform distribution with slight perturbation
        h = np.full(n_points, 0.5)
        
    # Initial evaluation
    current_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization with gradient descent (trust-constr)
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Ensure integral constraint
        current_sum = np.sum(h_vec) * dx
        penalty = 100 * (current_sum - 0.5)**2
        return evaluate_erdos_solution(h_vec) + penalty
    
    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 0.5
        
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Gradient-based optimization
    iteration = 0
    
    while time.time() - start_time < 0.8 * budget_s:
        # Random restarts to escape local optima
        if iteration % 10 == 0:
            h_restart = np.random.rand(n_points)
            h_restart = h_restart / np.sum(h_restart) * 0.5 * n_points
            h_restart = np.clip(h_restart, 0, 1)
            h = h_restart.copy()
        
        # Use scipy's trust-constr for constrained optimization
        res = minimize(objective, h, method='trust-constr', constraints=cons,
                       bounds=[(0, 1)] * n_points,
                       options={'maxiter': 20, 'disp': False})
        
        if not res.success:
            break
            
        h_new = np.clip(res.x, 0, 1)
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            
        h = h_new.copy()
        iteration += 1
    
    # Final refinement with gradient-free method (Nelder-Mead)
    if time.time() - start_time < 0.9 * budget_s:
        res = minimize(objective, best_h, method='Nelder-Mead',
                       options={'maxiter': 50, 'disp': False})
        if res.success:
            h_final = np.clip(res.x, 0, 1)
            c5_final = evaluate_erdos_solution(h_final)
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
                
    # Ensure feasibility of final solution
    if np.abs(np.sum(best_h) * dx - 0.5) > 1e-6:
        # Adjust to meet constraint
        scale_factor = 0.5 / (np.sum(best_h) * dx)
        best_h = np.clip(best_h * scale_factor, 0, 1)
    
    return (best_h.tolist(), best_c5, n_points)
