import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # If no initial construction, create a simple one
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    delta = 0.1
    last_delta_update = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Update delta periodically
        if iteration - last_delta_update > 100:
            delta *= 0.9
            last_delta_update = iteration
        
        # Select two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        if i == j:
            continue
        
        # Try perturbation
        h_new = h.copy()
        delta_h = delta * (np.random.rand() * 2 - 1)  # random sign and magnitude
        
        # Adjust both values
        h_new[i] += delta_h
        h_new[j] -= delta_h
        
        # Ensure bounds
        if h_new[i] < 0 or h_new[i] > 1 or h_new[j] < 0 or h_new[j] > 1:
            continue
            
        # Adjust to maintain sum constraint
        h_new[j] += (np.sum(h) - np.sum(h_new)) / (n_points - 1)
        
        # Ensure bounds again
        if h_new[j] < 0 or h_new[j] > 1:
            continue
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Occasionally accept worse solutions to escape local optima
            if np.random.rand() < 0.01:
                h = h_new.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    # Compute all correlation values using numpy's correlate
    # This computes all values of sum_i h[i] * (1 - h[i+k])
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value over all shifts is what we want
    return np.max(corr) * dx
