import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 1000000
    initial_T = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05
    delta = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values[:n_points])
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        h_extended = np.concatenate([h, h])
        conv = correlate(h_extended, 1 - h, mode='full')[:n_points]
        return np.max(conv) * dx
    
    def normalize(h):
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return np.clip(h, 0, 1)
    
    def perturb(h, T):
        h_new = h.copy()
        n_perturb = max(1, int(n_points * perturb_frac))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        for i in indices:
            delta_h = np.random.uniform(-delta, delta) * T
            h_new[i] += delta_h
        return normalize(h_new)
    
    # Annealing loop
    T = initial_T
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s:
        h_new = perturb(h, T)
        c5_new = compute_c5(h_new)
        
        delta_c5 = c5_new - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = c5_new
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
        T *= cooling_rate
        iteration += 1
        
        if best_c5 < target_c5:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
