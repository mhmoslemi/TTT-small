import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialization
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    alpha = 0.1
    max_iter = 100000
    patience = 50
    patience_counter = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all k
        overlaps = np.correlate(h, 1 - h, mode="full")
        # Only consider valid overlaps (k from -n+1 to n-1)
        k_indices = np.arange(-n_points + 1, n_points)
        # Extract only the relevant portion
        overlaps = overlaps[n_points - 1:2 * n_points - 1]
        
        # Find worst k
        max_k_idx = np.argmax(overlaps)
        max_k = k_indices[max_k_idx]
        
        # Compute gradient w.r.t. max_k
        grad = np.zeros_like(h)
        for i_idx in range(n_points):
            j_idx = i_idx + max_k
            if 0 <= j_idx < n_points:
                grad[i_idx] += (1 - h[j_idx]) * dx
                grad[j_idx] -= h[i_idx] * dx
                
        # Update h
        h = h - alpha * grad
        
        # Maintain constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if patience_counter >= patience:
            break
            
        # Adaptive step size
        if i % 100 == 0:
            alpha *= 0.995
    
    # Final check
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode="full")) * dx
    return (best_h.tolist(), float(final_c5), len(best_h))
