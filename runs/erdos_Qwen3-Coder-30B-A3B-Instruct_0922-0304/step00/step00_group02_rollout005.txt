import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 45
        h = np.zeros(n_points)
        mid = n_points // 2
        h[:mid] = 1
        np.random.shuffle(h)
    
    # Normalize to sum to 1
    h = h / np.sum(h) * (n_points / 2)
    
    def compute_C5(hv):
        # Compute correlation
        corr = np.correlate(hv, 1 - hv, mode="full")
        # Convert back to original form of C5 (stored in max correlation)
        max_corr = np.max(corr)
        return max_corr / (n_points / 2.0)
    
    def harmonic_ratio(hv):
        # Try to compute objective function directly
        h_vals = hv.copy()
        if np.sum(h_vals) > n_points / 2:
            # Scale down
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        elif np.sum(h_vals) < n_points / 2:
            # Scale up
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_vals = np.clip(h_vals, 0, 1)
        return compute_C5(h_vals)
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    no_improvement_iter = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Create a candidate through local search
        h_new = h.copy()
        indices = np.random.permutation(n_points)
        
        for i in indices:
            # Try flipping bit (in binary mode)
            current_val = h_new[i]
            proposed_val = 1 - current_val
            h_new[i] = proposed_val
            
            # Enforce sum
            if np.sum(h_new) > n_points / 2:
                # Scale down
                scale = (n_points / 2) / np.sum(h_new)
                h_new = h_new * scale
            elif np.sum(h_new) < n_points / 2:
                # Scale up
                scale = (n_points / 2) / np.sum(h_new)
                h_new = h_new * scale
            h_new = np.clip(h_new, 0, 1)
            
            # Compute new C5
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                h = h_new.copy()
                best_C5 = new_C5
                no_improvement_iter = 0
            else:
                h_new[i] = current_val  # revert
                no_improvement_iter += 1
                
        if no_improvement_iter >= max_no_improvement:
            break
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    final_C5 = compute_C5(best_h)
    final_h = best_h.copy()
    
    return (final_h.tolist(), final_C5, len(final_h))
