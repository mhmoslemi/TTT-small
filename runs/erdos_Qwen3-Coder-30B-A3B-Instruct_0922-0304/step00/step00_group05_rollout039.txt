import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
    except NameError:
        # Fallback: create a simple piecewise constant function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = int(budget_s * 1000)
    
    last_improvement = 0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all indices where max overlap occurs
        max_lags = np.where(overlap == max_overlap)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for m in max_lags:
            if m < n_points:
                # For lag m, contribution to gradient from h[i] is (1 - h[i+m])
                # Only valid positions
                valid_indices = np.arange(0, n_points - m)
                gradient[valid_indices] += (1 - h[valid_indices + m])
            else:
                # For negative lag
                valid_indices = np.arange(m - n_points + 1, n_points)
                gradient[valid_indices] += (1 - h[valid_indices + m - n_points + 1])
        
        # Normalize gradient
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Gradient step
        h -= learning_rate * gradient
        
        # Projection: clip to [0,1] then rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h *= (n_points / 2) / h_sum
            
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Evaluate overlap integral
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)
