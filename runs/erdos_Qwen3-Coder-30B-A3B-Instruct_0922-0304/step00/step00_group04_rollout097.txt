import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default initialization: piecewise constant
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.4
        h0[1] = 0.6
        h0[2] = 0.55
        h0[3] = 0.45
        # Ensure integral = 1
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize parameters
    dx = 2.0 / n_points
    current_h = h0.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_erdos_solution(current_h, n_points, dx)
    
    # Convex relaxation helper
    def solve_convex_relaxation(h_init, n_points):
        h = Variable(n_points)
        # Constraints
        constraints = [h >= 0, h <= 1, sum(h) == n_points / 2]
        
        # Relaxation: approximate overlap integral using quadratic constraints
        # Convex relaxation of h[i] * h[j] is not trivial
        # Instead, just solve using standard convex constraints
        prob = Problem(Minimize(0), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            return np.array(h.value).reshape(-1)
        except:
            return h_init
    
    # Gradient helper for one lag
    def compute_gradient(h, k):
        # Compute correlation with 1-h for lag k
        l = len(h)
        dx = 2.0 / l
        corr = np.correlate(h, 1 - h, mode='full')
        # Focus on lag k
        grad = np.zeros_like(h)
        # Gradients w.r.t. h[i] for correlation [i+k]
        for i in range(l):
            if i + k < l:
                grad[i] += (1 - h[i+k]) * dx
            if i - k >= 0:
                grad[i] += h[i-k] * dx
        return grad

    # Gradient descent stage
    def gradient_descent(current_h, max_iter=100):
        h = current_h.copy()
        dx = 2.0 / len(h)
        for it in range(max_iter):
            # Compute overlaps for all lags
            corr = np.correlate(h, 1 - h, mode='full')
            max_k = np.argmax(corr)
            # Compute gradient
            g = compute_gradient(h, max_k)
            # Adaptive learning rate
            lr = 0.05 * (0.9 ** it)
            h_new = h - lr * g
            # Project back to feasible region
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            if time.time() - start_time > budget_s * 0.95:
                break
        return h
    
    # Main loop
    iter_count = 0
    while time.time() - start_time < budget_s * 0.99:
        # Recompute current c5
        current_c5 = evaluate_erdos_solution(current_h, n_points, dx)
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

        # Convex relaxation step
        h_convex = solve_convex_relaxation(current_h, n_points)
        # Refine with gradient descent
        h_refined = gradient_descent(h_convex, max_iter=20)
        current_h = h_refined

        # Optionally upsampling
        if iter_count < 5 and n_points < 84:
            # Upsample to 84 points
            prev_h = current_h.copy()
            new_h = np.interp(np.linspace(0, 2, 84), np.linspace(0, 2, n_points),  prev_h)
            new_h = new_h * 42 / np.sum(new_h)
            n_points = 84
            dx = 2.0 / n_points
            current_h = new_h

        iter_count += 1
        if time.time() - start_time > budget_s * 0.95:
            break

    # Final verification
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    return (best_h.tolist(), final_c5, n_points)

# Not used here due to the task constraints and local import
def evaluate_erdos_solution(h, n_points, dx):
    # Do direct correlation
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
