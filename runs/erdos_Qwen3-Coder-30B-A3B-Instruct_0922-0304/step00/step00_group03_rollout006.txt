import numpy as np
from scipy.signal import correlate
import time

def evaluate(h_values, dx):
    # Compute overlap integral C5
    h = np.array(h_values)
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Discretization parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.round(h).astype(int)
    else:
        # Create a sample binary vector with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
    
    # Ensure we have exactly n_points/2 ones
    current_sum = np.sum(h)
    needed = n_points // 2 - current_sum
    if needed > 0:
        # Find zeros and set some to 1
        zero_indices = np.where(h == 0)[0]
        for i in range(needed):
            if i < len(zero_indices):
                h[zero_indices[i]] = 1
    elif needed < 0:
        # Find ones and set some to 0
        one_indices = np.where(h == 1)[0]
        for i in range(-needed):
            if i < len(one_indices):
                h[one_indices[i]] = 0
    
    # Initialize tabu search
    max_iter = 10000
    tabu_tenure = 100
    tabu_list = []
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    last_improvement = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate neighbors by flipping 2-3 bits
        neighbor_h = h.copy()
        
        # Choose number of flips (2 or 3)
        n_flips = np.random.choice([2, 3])
        
        # Get indices of 0s and 1s
        zeros = np.where(neighbor_h == 0)[0]
        ones = np.where(neighbor_h == 1)[0]
        
        if len(zeros) < 1 or len(ones) < 1:
            continue
            
        # Flip bits
        flip_indices = []
        if n_flips == 2:
            # Flip one '1' to '0' and one '0' to '1'
            i1 = np.random.choice(ones)
            i2 = np.random.choice(zeros)
            flip_indices = [i1, i2]
            neighbor_h[i1] = 0
            neighbor_h[i2] = 1
        elif n_flips == 3:
            # Flip two '1's to '0's and one '0' to '1'
            i1 = np.random.choice(ones)
            i2 = np.random.choice(ones)
            while i2 == i1:
                i2 = np.random.choice(ones)
            i3 = np.random.choice(zeros)
            flip_indices = [i1, i2, i3]
            neighbor_h[i1] = 0
            neighbor_h[i2] = 0
            neighbor_h[i3] = 1
        
        # Check if neighbor is in tabu list
        if tuple(neighbor_h) in tabu_list:
            continue
            
        # Evaluate neighbor
        neighbor_c5 = evaluate(neighbor_h, dx)
        
        # Add last h to tabu
        if len(tabu_list) >= tabu_tenure:
            tabu_list.pop(0)
        tabu_list.append(tuple(h))
        
        # Update best solution if better
        if neighbor_c5 < best_c5:
            best_h = neighbor_h.copy()
            best_c5 = neighbor_c5
            last_improvement = iteration
                
        # Accept neighbor
        h = neighbor_h.copy()
        
        # Early termination
        if iteration - last_improvement > max_iter // 4:
            break
    
    # Final check
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
