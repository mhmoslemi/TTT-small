import numpy as np
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:int(target_sum)] = 1.0
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if current_sum != target_sum:
        # Adjust to ensure exact sum
        indices = np.argsort(h)
        diff = int(target_sum - current_sum)
        if diff > 0:
            for i in range(diff):
                h[indices[i]] = 1.0
        else:
            for i in range(-diff):
                h[indices[-(i+1)]] = 0.0

    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    
    def evaluate_C5(h_vec):
        overlaps = correlate(h_vec, 1 - h_vec, mode="full") * dx
        return np.max(overlaps)
    
    def project_to_binary(h_relaxed, target_sum=target_sum):
        # Threshold at 0.5
        h_binary = (h_relaxed > 0.5).astype(float)
        current_sum = np.sum(h_binary)
        diff = int(target_sum - current_sum)
        
        if diff == 0:
            return h_binary
        elif diff > 0:
            # Need to set more 0s to 1s
            indices = np.argsort(h_relaxed)
            for i in range(diff):
                h_binary[indices[i]] = 1.0
        else:
            # Need to set more 1s to 0s
            indices = np.argsort(h_relaxed)
            for i in range(-diff):
                h_binary[indices[-(i+1)]] = 0.0
        return h_binary
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if np.abs(current_sum - target_sum) > 1e-8:
            # Scale to target sum
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    iterations = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        if no_improvement_count > max_no_improvement:
            break
        
        # Evaluate current solution
        current_C5 = evaluate_C5(best_h)
        
        # Step 1: Relax to convex problem with current h
        try:
            # Create a maximization problem
            # We want to minimize max_k integral
            overlaps = correlate(best_h, 1 - best_h, mode="full") * dx
            max_overlap = np.max(overlaps)
            
            # Create a relaxed version
            h_relaxed = cp.Variable(n_points, name="h_relaxed")
            
            # Objective: minimize max overlap over all shifts
            objective = cp.Minimize(max_overlap)
            
            # Set cvxpy constraints
            constraints = [
                h_relaxed >= 0,
                h_relaxed <= 1,
                cp.sum(h_relaxed) == target_sum
            ]
            
            # Create and solve problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_relaxed.value
                # Project back to binary feasible set
                h_binary = project_to_binary(new_h)
                h_binary = project_to_feasible(h_binary)
                
                # Evaluate new solution
                new_C5 = evaluate_C5(h_binary)
                
                # Accept improvement
                if new_C5 < best_C5:
                    best_h = h_binary.copy()
                    best_C5 = new_C5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
            else:
                # Fall back to using current h
                no_improvement_count += 1
                
        except Exception as e:
            no_improvement_count += 1
            
        # Add small random perturbations to escape local minima
        if np.random.rand() < 0.2 and no_improvement_count > 0:
            # Perturb 5% of values
            num_perturb = max(1, int(0.05 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                if np.random.rand() < 0.5:
                    best_h[i] = min(1.0, best_h[i] + 0.1)
                else:
                    best_h[i] = max(0.0, best_h[i] - 0.1)
            # Project to feasible set
            best_h = project_to_feasible(best_h)
            new_C5 = evaluate_C5(best_h)
            if new_C5 < best_C5:
                best_C5 = new_C5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h_vec):
    # Recreate behavior of the evaluation in the harness
    h = np.array(h_vec)
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    return np.max(overlaps)
