import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum value and scale by dx
    return np.max(corr) * dx

def initialize_h(n_points):
    """Initialize h from initial_h_values or create random binary vector."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = (h > 0.5).astype(int)
    else:
        # Generate random binary vector with exactly n_points/2 ones
        h = np.zeros(n_points, dtype=int)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    return h

def repair_sum(h, target_sum):
    """Ensure h has exactly target_sum ones."""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h
    if current_sum > target_sum:
        # Need to flip some 1s to 0s
        indices = np.where(h == 1)[0]
        flip_indices = np.random.choice(indices, current_sum - target_sum, replace=False)
        h[flip_indices] = 0
    else:
        # Need to flip some 0s to 1s
        indices = np.where(h == 0)[0]
        flip_indices = np.random.choice(indices, target_sum - current_sum, replace=False)
        h[flip_indices] = 1
    return h

def generate_neighbors(h, n_points, target_sum, tabu_list):
    """Generate neighboring solutions by flipping 2-3 bits."""
    neighbors = []
    for _ in range(100):  # Generate 100 neighbors
        # Copy current solution
        new_h = h.copy()
        # Determine how many bits to flip
        n_flip = np.random.choice([2, 3])
        # Pick random indices to flip
        flip_indices = np.random.choice(n_points, n_flip, replace=False)
        # Flip bits
        for i in flip_indices:
            new_h[i] = 1 - new_h[i]
        # Repair sum to ensure it's correct
        new_h = repair_sum(new_h, target_sum)
        # Check if this neighbor is in tabu list
        h_tuple = tuple(new_h)
        if h_tuple not in tabu_list:
            neighbors.append(new_h)
    return neighbors

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Define parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points // 2  # Exactly half ones
    
    # Initialize start
    h = initialize_h(n_points)
    h = repair_sum(h, target_sum)
    
    # Initialize with current best
    best_h = h.copy()
    best_c5 = evaluate(h, dx)
    
    # Tabu search parameters
    tabu_tenure = 100
    tabu_list = set()
    
    max_iter = 1000
    start_time = time.time()
    
    # Iteration counter
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s:
        # Generate neighbors
        neighbors = generate_neighbors(h, n_points, target_sum, tabu_list)
        
        if not neighbors:
            # No valid neighbors, break out
            break
        
        # Evaluate all neighbors
        neighbor_scores = []
        for neighbor in neighbors:
            score = evaluate(neighbor, dx)
            neighbor_scores.append((neighbor, score))
        
        # Find the best neighbor
        best_neighbor, best_score = min(neighbor_scores, key=lambda x: x[1])
        
        # Update best solution
        if best_score < best_c5:
            best_h = best_neighbor.copy()
            best_c5 = best_score
        
        # Add previous solution to tabu list
        tabu_list.add(tuple(h))
        if len(tabu_list) > tabu_tenure:
            # Remove oldest
            tabu_list.pop()
        
        # Move to best neighbor
        h = best_neighbor
        iteration += 1
    
    # Ensure final solution is correct
    h = repair_sum(h, target_sum)
    final_c5 = evaluate(h, dx)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h, best_c5, n_points)
