import numpy as np
from scipy.fft import fft, ifft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available, else create a reasonable one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a reasonable initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial function evaluation
    def evaluate(h):
        # Use the provided evaluation method
        h_padded = np.concatenate([h, np.zeros(n_points-1)])
        h_rev = np.flip(h)
        corr = np.correlate(h_padded, h_rev, mode='full')
        max_corr = np.max(corr[:n_points])
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    # Optimization loop
    max_iter = 2000
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Create a new candidate using perturbation
        h_new = h.copy()
        
        # Randomly pick indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        for i in indices:
            # Add small random change within bounds
            delta = np.random.normal(0, 0.05)
            h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        
        # Re-normalize to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate the new function
        c5_new = evaluate(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
            
        h = h_new.copy()
        iter_count += 1
        
        # Stop early if no significant improvement for a while
        if iter_count - last_improvement > 50:
            break
    
    return (best_h.tolist(), best_c5, n_points)
