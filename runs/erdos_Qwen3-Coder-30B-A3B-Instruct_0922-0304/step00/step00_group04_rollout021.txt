import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5
    
    # Add some randomness to initial vector
    h += np.random.normal(0, 0.01, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h_vals):
        h_arr = np.array(h_vals)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_and_rescale(h_vals):
        # Clip to [0, 1]
        h = np.clip(h_vals, 0, 1)
        # Rescale to sum to n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    def gradient_descent_step(h_vals, rate=0.1):
        h = np.array(h_vals)
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        # Compute finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_and_rescale(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_and_rescale(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (corr_plus[max_k] - corr_minus[max_k]) / (2 * eps)
        
        new_h = h - rate * grad
        return project_and_rescale(new_h)
    
    def convex_relaxation(h_vals):
        h = np.array(h_vals)
        n = len(h)
        # Create relaxed variables
        y = cp.Variable((n, n), symmetric=True)
        # Objective: minimize max correlation
        obj = cp.Minimize(cp.max(cp.multiply(y, np.ones((n, n)))))
        
        # Constraints
        constraints = []
        for i in range(n):
            for j in range(n):
                if i >= j:
                    if i == j:
                        constraints.append(y[i, j] >= h[i] * h[j])
                    else:
                        constraints.append(y[i, j] >= h[i] * h[j])
                        constraints.append(y[i, j] >= h[j] * h[i])
                else:
                    constraints.append(y[i, j] == y[j, i])
                    constraints.append(y[i, j] >= h[i] * h[j])
        
        # Sum constraint
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Bound constraints
        for i in range(n):
            constraints.append(h[i] >= 0)
            constraints.append(h[i] <= 1)
            
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            return h.value
        except:
            return h_vals
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    # Stage 1: Convex relaxation to get an initial solution
    if time.time() - start_time < budget_s * 0.5:
        try:
            h_new = convex_relaxation(h)
            h_new = project_and_rescale(h_new)
            C5_new = evaluate_C5(h_new)
            if C5_new < best_C5:
                best_h = h_new.copy()
                best_C5 = C5_new
        except:
            pass

    # Stage 2: Gradient descent refinement
    current_h = best_h.copy()
    lr = 0.1
    iter_count = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        current_h = gradient_descent_step(current_h, rate=lr)
        C5 = evaluate_C5(current_h)
        if C5 < best_C5:
            best_h = current_h.copy()
            best_C5 = C5
        iter_count += 1
        if iter_count % 100 == 0:
            lr *= 0.5  # Decrease learning rate

    # Stage 3: Try higher resolution if time permits
    if time.time() - start_time < budget_s * 0.8:
        # Upsample to 84 points
        h_upsampled = np.interp(np.linspace(0, 2, 84), np.linspace(0, 2, n_points), best_h)
        # Convex relaxation again
        try:
            h_upsampled = convex_relaxation(h_upsampled)
            h_upsampled = project_and_rescale(h_upsampled)
            C5_upsampled = evaluate_C5(h_upsampled)
            if C5_upsampled < best_C5:
                best_h = h_upsampled.copy()
                best_C5 = C5_upsampled
        except:
            pass
        
        # Gradient descent at higher resolution
        current_h = best_h.copy()
        lr = 0.1
        iter_count = 0
        max_iter = 100
        
        while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
            current_h = gradient_descent_step(current_h, rate=lr)
            C5 = evaluate_C5(current_h)
            if C5 < best_C5:
                best_h = current_h.copy()
                best_C5 = C5
            iter_count += 1
            if iter_count % 100 == 0:
                lr *= 0.5  # Decrease learning rate
    
    # Final validation
    final_h = project_and_rescale(best_h)
    final_C5 = evaluate_C5(final_h)
    
    return (final_h.tolist(), final_C5, len(final_h))
