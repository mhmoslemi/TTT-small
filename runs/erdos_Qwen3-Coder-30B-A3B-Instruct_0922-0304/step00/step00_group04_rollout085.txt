import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    lag_max = n_points - 1
    
    # Get initial construction if available
    try:
        h0 = initial_h_values
        h0 = np.array(h0)
    except:
        h0 = np.ones(n_points) * 0.5
    
    # Optimization variables
    h = Variable(n_points)
    C5 = Variable()
    
    # Constraints for step function properties
    constraints = [
        h >= 0,
        h <= 1,
        sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Get complete correlation constraints
    # We'll generate the correlation constraint for all possible lags
    for k in range(-lag_max, lag_max + 1):
        if k >= 0:
            # For positive lag, correlate h with 1-h
            corr_len = n_points - k
            if corr_len > 0:
                # Compute correlation sum and add constraint
                corr_sum = sum(h[i] * (1 - h[i + k]) for i in range(corr_len))
                constraints.append(corr_sum <= C5)
        else:
            # For negative lag, the syntax is reversed but result is same
            corr_len = n_points + k
            if corr_len > 0:
                corr_sum = sum(h[i - k] * (1 - h[i]) for i in range(corr_len))
                constraints.append(corr_sum <= C5)
    
    # Objective to minimize
    objective = Minimize(C5)
    
    # Create problem
    prob = Problem(objective, constraints)
    
    start_time = time.time()
    
    try:
        # Use a simple heuristic as fallback
        h_opt = h0.copy()
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
        best_h = h_opt
        best_c5 = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
        
        # Run optimization for remaining time
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 0.1:
            try:
                # Solve with GLPK if available
                prob.solve(solver=GLPK_MI, verbose=False, max_iters=5000)
                if prob.status in [OPTIMAL, OPTIMIZER_ERR]:
                    h_opt = np.array(h.value).flatten()
                    h_opt = np.clip(h_opt, 0, 1)
                    h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
                    c5_bound = np.max(np.correlate(h_opt, 1 - h_opt, mode='full')) * dx
                    if c5_bound < best_c5:
                        best_h = h_opt
                        best_c5 = c5_bound
            except:
                pass
    except:
        pass
    
    # Final validation to ensure constraints
    h_final = np.clip(best_h, 0, 1)
    h_final = h_final / np.sum(h_final) * (n_points / 2)
    c5_final = np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx

    return (h_final.tolist(), c5_final, n_points)
