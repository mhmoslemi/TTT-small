import time

def evaluate_h(h, n_points):
    """Evaluate C5 for given h values."""
    dx = 2.0 / n_points

    corr = correlate(h, 1 - h, mode='full')
    # It's a full correlation, we want the maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx
