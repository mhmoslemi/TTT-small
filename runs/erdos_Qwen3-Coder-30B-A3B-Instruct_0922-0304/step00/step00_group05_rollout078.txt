    h = h * (n_points / 2) / np.sum(h)

    gamma = 0.01
    gamma_decay = 0.1
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate_c5(h_val):
        # Compute the convolution
        conv = np.correlate(h_val, 1 - h_val, mode='full')
        # Extract the relevant part(s) - the maximum overlap
        # Since we're looking at [0,2] interval with step function, 
        # this is a modified version of the overlap integral
        # For full convolution with length 2n-1, we just need to take the maximum
        # in this particular case, considering possible values of k in [-1,1]
        # effectively it's C_k = \int h(x)(1-h(x+k))dx
        # where k varies from -1 to 1 on 2 unit interval.
        # We will discretize properly for our step function
        conv_shifted = np.concatenate([conv[len(conv)//2:], conv[:len(conv)//2]])
        c5 = np.max(conv_shifted) * dx
        return c5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                gamma *= gamma_decay
                patience_counter = 0
        
        # Compute gradients using finite differences (simplest approach)
        # Compute all overlap terms and gradient for each point
        # We will use symbolic gradients closely aligned with derivative intuition
        grad = np.zeros_like(h)
        
        # Compute L = max_k C_k = \int h(x)(1-h(x+k))dx
        # For a small change in h[i], we have effect on C_k for all lags k
        conv = np.correlate(h, 1-h, mode='full')
        # Extract maximum overlap term index (simplified assumption)
        max_idx = np.argmax(conv)
        max_c = conv[max_idx]
        # Take full convolution, effectively no derivatives (projected gradient)
        for idx in range(len(h)):
            # Compute contribution to gradient from all convolution terms
            # Simplified gradient computation for each point
            # G[i] proportional to sum over all phases of (1 - h[i + s]) * delta(s)
            delta_c = np.ones(2*n_points - 1) * 0.0  # will mask properly

            # the derivative of (h[i] * (1-h[i+k]))  with respect to h[i] is (1 - h[i+k])
            contribs = []
            for k in range(-n_points+1, n_points, 1):
                if 0 <= idx + k < n_points:
                    contribs.append(1 - h[idx + k])
            if len(contribs) > 0:
                deriv = np.mean(contribs)
                grad[idx] = deriv
                
        # Gradient ascent step (we're minimizing, so negative)
        h = h - gamma * grad
        
        # Project h back to valid range [0,1]
        h = np.clip(h, 0, 1)
        
        # Project sum to constraint \int_0^2 h(x) dx = 1 (i.e. sum(h)*dx = 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Return best solution found
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
