import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize based on the provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 1000
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Precompute all possible shifts
    shift_range = range(1, n_points)

    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts (precompute for efficiency)
        overlaps = []
        for k in shift_range:
            # This computes for shift k: sum h[i] * (1 - h[(i+k) % n])
            corr = np.correlate(h, np.flip(1 - h), mode='full')
            overlap = corr[n_points - 1 + k] 
            overlaps.append((k, overlap))
            
        # Find the maximum overlap
        k_max, overlap_max = max(overlaps, key=lambda x: x[1])
        
        # Solve convex optimization problem
        # Decision variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            C5_var >= 0
        ]
        
        # The quadratic constraint for the worst case
        # Sum h[i] * (1 - h[(i + k_max) % n]) <= C5
        # This is equivalent to: sum h[i] - sum h[i] * h[(i + k_max) % n] <= C5
        # Which we implement using elementwise products
        if k_max == 0:
            product_terms = h_var * h_var
        else:
            shifted_h = np.concatenate([h_var[k_max:], h_var[:k_max]])
            product_terms = h_var * shifted_h
            
        constraints.append(cp.sum(h_var) - cp.sum(product_terms) <= C5_var)
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Solve with solver that supports quadratic constraints
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                new_c5 = C5_var.value
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                # Update variables
                h = h_new
                
                if best_c5 < 0.3807:
                    break
                    
                if no_improvement_count > max_no_improvement:
                    break
                    
        except Exception as e:
            # Fallback to standard update if optimization fails
            pass
    
    # Final evaluation of best solution
    final_overlaps = []
    for k in shift_range:
        corr = np.correlate(best_h, np.flip(1 - best_h), mode='full')
        overlap = corr[n_points - 1 + k] 
        final_overlaps.append(overlap)
        
    final_c5 = max(final_overlaps)
    
    return (best_h.tolist(), float(final_c5), n_points)
