import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # The full correlation has length 2*n_points - 1
        # The offsets are from -(n_points-1) to (n_points-1)
        # We want to ignore the "wraparound" correlations that are not legitimate
        # But in our case, stored correlations are from -n_points+1 to n_points-1
        # We care only about lags 1 to n_points-1 for k>0, and -n_points+1 to -1 for k<0
        # The calculation gives us only positive lags meaning it's from 0 to 2*n_points-2
        # The index i=0 corresponds to k=-(n_points-1)
        # so the true lag k for the cross-correlation index i is i - (n_points - 1)
        # k = -(n_points - 1) to (n_points - 1) 
        # For valid lags that are not wrapping around the array, we want:
        # 1 to n_points-1 (positive lags)  
        # -n_points+1 to -1 (negative lags)
        # The full correlation length is 2*n_points - 1
        # Cross-correlations at positions 0 to n_points-2 are negative lags
        # Cross-correlations at positions n_points-1 to 2*n_points-2 are positive lags
        
        # For computational simplification:
        # We can extract the lag 1 to n_points-1 ones (positive lags follows indices n_points-1 to 2*n_points-2)
        # And the negative lags are from indices 0 to n_points-2
        # But we're calculating integral over overlaps so we handle negatives Might be better 
        # to directly use numpy.convolve and correct indices
        # correct way:
        # corr[k] = sum h[i] * (1 - h[i+k])   for valid range
        # Equivalently, correlate(h, 1-h) in numpy returns the correlations for all lags
        
        # So s = n_points-1, 0 to s = s (k=-n+1 to 0)
        # and s+1 to 2s = 2*s (k=1 to n-1)
        # k=-n+1 to n-1 where sol index = k + n-1 
        k_offsets = np.arange(0, 2*n_points - 1) - (n_points - 1)
        lags = k_offsets
        valid_lags = lags # may want to filter if needed to avoid wraparound bias
        
        # But need to be more careful, with 2* length, index i corresponds to k = i - (n_points - 1)
        # The overlap integrand for k lag is corr[i] = dx * sum_{j} h[j] * (1 - h[j+k]) for valid j's
        c5_values = corr * dx 
        c5 = np.max(c5_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # While we cannot do analytic gradients, 
        # we can compute steepest descent direction relative to the current max lag
        
        # Find which lag achieves the maximum
        max_idx = np.argmax(c5_values)
        # Compute true lag k from index
        k = max_idx - (n_points - 1)
        
        # Gradient w.r.t h for the specific lag k
        # d/dh_i ( sum_j h_j (1 - h_{j+k}) ) = (1 - h_{i+k}) if k>=0, 1 - h_{i-k} otherwise
        # But we scale by dx and the full displacement may differ
        
        # But since we are considering overlaps, we only care about how current h changes affect one of those overlaps
        # So we compute the gradient for the constraint lag k
        
        # Gradient for numerator (relevant for the specific k, which is a linear combination of h_shifted by k and others)
        # Gradient is proportional to (1-h_{i+k}) in k-th direction
        # And -h_{i-k} in -k direction
        
        # simplified by quantifying dependence of each term on h_i:
        # For lag k (k > 0), we compute: 
        # d/dh_i [ \sum_j h_j (1 - h_{j+k}) ] = (1 - h_{i+k}) for i in valid range
        # But our function already includes scaling dx, which we are using
        
        # So let’s face it: we go with gradient descent based on the maximum xcorr segment at current lag k,
        # Using finite differences or finite approximation of gradient direction
        # Create a simple finite-gradient update for h
        
        # Simple upate, (f'(x_i) ~ f(x_i + dx) - f(x_i-dx) ) / (2*dx)
        # We know x = h and y = corr[i] = \sum_j h_j * (1 - h_{j + i - n+1}) dx
        # we're looking to reduce the max overlap
        # Thus we're looking for which h_i to increase/decrease to lower it
        
        # So between i = 0 to n_points - 1
        # Gradient is such that the k-lag overlap contribution comes from:
        # \sum_{i} h[i] * (1 - h[i+k])  where each term diff by h[i] is (1 - h[i+k])
        # But since the length can vary, better use a direct computation.
        
        # label from 0: to n_points-1,  (correlation index from 0 to 2n_points-2)
        # index i corresponds to lag = (i - (n_points - 1)) = i - n + 1 
        # For positive lag, that means index > n_points -1, and lag >= 1
        # lag = k = i - (n_points - 1) => i = k + n - 1, j in [0,n-1], j+lags  = i must be valid
        # for j+lags = i and 0 <= i <= n-1
        # => j = i - lag, need 0 <= i-lg <= n-1 => lg <= i <= n-1+lg
        # so i = d for valid, d >= lg and d <= n-1+lg
        # 

        # More careful way based on standard convolution gradient:
        # In convolution corr of h, 1-h:
        # corr[k] = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k]) where we let h[i+k] = 0 when i+k >= n_points or < 0
        # Gradient w.r.t h[i]: d/dh[i] corr[l] = (1 - h[i-l]) if l >= 0, 0 otherwise
        
        # So at k-th lag:
        #  dcoh[i] / dh[i] = (1 - h[i-k]) for i+k < n == i < n-k, 0 otherwise  
        # for k>0, we look at the gradient: (1 - h[i-k]) which is from h[i-k] terms starting at i>=k
        
        # Instead of building a full Jacobian:
        # We construct the functional direction to go
        
        # Forward mode gradient in direction of max lag k
        grad = np.zeros(n_points)
        if k >= 0:  # so the shifted version is h[i+k], valid when i+k < n_points, so i < n-k
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])  # Remember: this is dx-multiplied
        else:  # k < 0. So h[i-k] = h[i+k] but valid when i+k >= 0, i >= -k
            for i in range(-k, n_points):
                grad[i] += -h[i + k]  # note the sign of gradient for coeff of h[i+k] -> -h[i+k] in product of h[i]*(-h[i+k]) 
                
        # normalize or stabilize for descent step size
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 1e-12:
            grad = grad / grad_norm
            
        # update step:
        # We seek to reduce C5, which we sample as the maximum of correlations
        # More direct way:
        # if gradient descent on maximum, sample the vector whose direction reduces it most
        
        # Gradient descent update
        step_size = 0.01  # fixed step
        h = h - step_size * grad
        
        # Project h back into feasible region: [0,1] and sum = n_points/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
    return (best_h.tolist(), float(best_c5), int(n_points))
