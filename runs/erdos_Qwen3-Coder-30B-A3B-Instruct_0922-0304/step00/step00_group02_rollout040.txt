import numpy as np
from scipy import correlate
import time
import random

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # Take the maximum and scale by dx
    c5 = np.max(overlaps) * dx
    return c5

def validate_h(h):
    """Ensure h meets constraints: 0 <= h <= 1 and sum = 1."""
    n = len(h)
    dx = 2.0 / n
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    # Check sum constraint (sum*h*dx = 1)
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        return False
    return True

def initialize_population(n_points, pop_size, initial_h_values):
    """Initialize population with diverse candidates."""
    population = []
    # Add the initial construction
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = np.array(initial_h_values)
        if validate_h(h):
            population.append(h.copy())
        else:
            # try to fix it
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            if validate_h(h):
                population.append(h.copy())
    
    # Fill the rest with random feasible solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        # Rescale to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clip
        h = np.clip(h, 0, 1)
        if validate_h(h):
            population.append(h.copy())
    return population

def selection(population, fitnesses):
    """Select top 50% by fitness."""
    sorted_indices = np.argsort(fitnesses)
    selected = [population[i] for i in sorted_indices[:len(population)//2]]
    return selected

def crossover(p1, p2):
    """Create two children by segment swapping."""
    n = len(p1)
    # Random crossover point
    cx_point = random.randint(1, n - 1)
    c1 = np.concatenate([p1[:cx_point], p2[cx_point:]])
    c2 = np.concatenate([p2[:cx_point], p1[cx_point:]])
    # Ensure sum constraint
    c1_sum = np.sum(c1)
    c2_sum = np.sum(c2)
    if abs(c1_sum - n/2) > 1e-10:
        c1 = c1 * (n/2) / c1_sum
    if abs(c2_sum - n/2) > 1e-10:
        c2 = c2 * (n/2) / c2_sum
    return c1, c2

def mutate(h, mutation_rate):
    """Mutate h with given rate."""
    n = len(h)
    for i in range(n):
        if random.random() < mutation_rate:
            # Perturb
            delta = random.uniform(-0.1, 0.1)
            h[i] += delta
            # Clamp
            h[i] = max(0, min(1, h[i]))
    # Ensure sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n/2) > 1e-10:
        h = h * (n/2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    pop_size = 100
    mutation_rate = 0.01
    max_generations = 1000
    max_no_improve = 100
    
    # Initialize
    start_time = time.time()
    
    # Check for initial construction
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = None
    else:
        initial_h = None
    
    population = initialize_population(n_points, pop_size, initial_h)
    best_found = None
    best_c5 = float('inf')
    no_improve_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (higher is better)
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(1.0 / (c5 + 1e-15))  # Avoid division by zero
            
        # Find best in current population
        current_best_idx = np.argmax(fitnesses)
        current_best_c5 = 1.0 / fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_found = population[current_best_idx]
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= max_no_improve:
            break
            
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if len(selected) < 2:
                # Just reseed
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                if validate_h(h):
                    new_population.append(h)
            else:
                # Pick two parents
                p1 = random.choice(selected)
                p2 = random.choice(selected)
                c1, c2 = crossover(p1, p2)
                # Mutate
                c1 = mutate(c1.copy(), mutation_rate)
                c2 = mutate(c2.copy(), mutation_rate)
                # Validate
                if validate_h(c1):
                    new_population.append(c1.copy())
                if validate_h(c2) and len(new_population) < pop_size:
                    new_population.append(c2.copy())
        
        population = new_population
    
    # Final validation and evaluation
    if best_found is not None:
        final_c5 = evaluate_h(best_found)
        return (best_found.tolist(), final_c5, n_points)
    else:
        # Return initial solution if we could not find anything better
        if initial_h_values is not None and len(initial_h_values) == n_points:
            h = np.array(initial_h_values)
            # Normalize if needed
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            if validate_h(h):
                c5 = evaluate_h(h)
                return (h.tolist(), c5, n_points)
        # Generate a default if nothing works
        h_default = np.ones(n_points) * 0.5
        h_default = h_default * (n_points / 2) / np.sum(h_default)
        c5_default = evaluate_h(h_default)
        return (h_default.tolist(), c5_default, n_points)
