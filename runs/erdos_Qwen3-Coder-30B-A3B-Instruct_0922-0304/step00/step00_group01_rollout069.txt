import numpy as np
import cvxpy as cp
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Returns the C5 bound for given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    corr = signal.correlate(h, 1 - h, mode='full')
    overlaps = corr[n-1:] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 47
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum = n/2
    h *= n_points / (2 * np.sum(h))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent with convex refinement
    max_iter = 10000
    conv_refine_freq = 50
    alpha = 0.01
    rho = 0.5
    max_backtrack = 30
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        corr = signal.correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:]  # Only shifts k in [0, n-1] 
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max] * 2.0 / n_points
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Compute gradient for worst case shift k_max
        shifted_h = np.roll(1 - h, k_max)
        gradient = shifted_h - h * shifted_h
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm == 0:
            continue
        gradient = gradient / grad_norm
        
        # Backtracking line search
        alpha_search = alpha
        for _ in range(max_backtrack):
            h_new = h - alpha_search * gradient
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum = n/2
            h_new *= (n_points / 2) / np.sum(h_new)
            
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept if improvement
            if c5_new <= c5_current:
                h = h_new
                break
            else:
                alpha_search *= rho
                
        # Convex refinement
        if iteration % conv_refine_freq == 0:
            # Optimize for k_max specifically with convex programming
            h_cvx = cp.Variable(n_points)
            constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
            
            # Objective: minimize overlap at k_max
            shifted_h_cvx = cp.Variable(n_points)
            for i in range(n_points):
                shifted_h_cvx[i] = h_cvx[(i + k_max) % n_points]
            objective = cp.sum(cp.multiply(h_cvx, 1 - shifted_h_cvx)) 
            
            prob = cp.Problem(cp.Minimize(objective), constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_new = h_cvx.value
                    h_new = np.clip(h_new, 0, 1)
                    h_new *= (n_points / 2) / np.sum(h_new)
                    c5_new = evaluate_erdos_solution(h_new)
                    if c5_new <= best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                    h = h_new
            except:
                pass
    
    return (best_h.tolist(), float(best_c5), int(n_points))
