import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a basic construction: step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Normalize to have sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    gamma = 0.01
    eps = 1e-6
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute the indices for convolution
    lag_indices = []
    for m in range(2*n_points - 1):
        lag_indices.append(m - (n_points - 1))
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        # Using numpy correlate for full convolution
        conv = np.correlate(h, 1 - h, mode='full')
        # Extract only the relevant part
        C_vals = []
        for i in range(len(conv)):
            offset = i - (n_points - 1)
            # Only consider valid overlaps (position shift within [0,2]-domain)
            if offset >= 0 and offset < n_points:
                C_vals.append(conv[i] * dx)
        
        C_vals = np.array(C_vals)
        max_C = np.max(C_vals)
        
        if max_C < best_c5:
            best_c5 = max_C
            best_h = h.copy()
        
        # Compute gradient for max C_m
        max_indices = np.where(np.abs(C_vals - max_C) < eps)[0]
        gradient = np.zeros(n_points)
        for idx in max_indices:
            lag = idx - (n_points - 1)
            # Convert lag to positive relative index
            if lag < 0:
                # Backward convolution
                start_idx = -lag
                end_idx = n_points
                for i in range(start_idx, end_idx):
                    gradient[i] += 1 - h[i + lag]
            else:
                # Forward convolution
                start_idx = 0
                end_idx = n_points - lag
                for i in range(start_idx, end_idx):
                    gradient[i] += 1 - h[i + lag]
        
        # Update h and project
        h = h - gamma * gradient
        h = np.clip(h, 0, 1)
        # Normalize sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            gamma *= 0.9
    
    return (best_h.tolist(), best_c5, n_points)
