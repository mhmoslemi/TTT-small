import numpy as np
from scipy.signal import correlate
from time import time

def evaluate_erdos_solution(h, dx):
    # Evaluate the maximum overlap integral
    overlaps = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600):

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()

    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    iteration = 0
    max_iterations = budget_s * 1000
    no_improvement_count = 0
    
    start_time = time()
    
    while iteration < max_iterations and (time() - start_time) < budget_s:
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        
        # Get all indices that achieve the max overlap
        max_lags = np.where(overlaps == max_overlap)[0]
        
        # Compute gradient for max_lags
        gradient = np.zeros(n_points)
        for lag in max_lags:
            # For each lag, the gradient contribution from h[i](1-h[i+lag])
            # The terms that contribute to overlap at lag are h[i] * (1 - h[i + lag])
            # So gradient[i] += (1 - h[i + lag]) for valid i
            if lag < n_points:
                for i in range(n_points - lag):
                    gradient[i] += (1 - h[i + lag])
            else:
                # lag >= n_points, so we're shifted backwards
                for i in range(lag - n_points + 1, n_points):
                    gradient[i] += (1 - h[i - (lag - n_points + 1)])
        
        # Average over number of contributing lags
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h using gradient descent
        h -= learning_rate * gradient
        
        # Project h to feasibility: [0,1] and correct sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution if better
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adapt learning rate
        if no_improvement_count >= gamma_patience:
            learning_rate *= gamma_decrease_factor
            no_improvement_count = 0
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
