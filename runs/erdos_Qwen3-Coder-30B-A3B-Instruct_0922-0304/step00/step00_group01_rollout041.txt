import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47
    max_iter = 10000
    refine_freq = 50
    alpha0 = 0.01
    rho = 0.5
    max_backtrack = 30
    
    # Initialize
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5)
    
    # Normalize to sum to n/2
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps
        h_rev = 1 - h_current
        overlaps = np.correlate(h_current, h_rev, mode='full')
        overlap_values = overlaps[n_points-1:]
        k_max = np.argmax(overlap_values)
        c5 = overlap_values[k_max] * 2.0 / n_points
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_current.copy()
            
        # Convert to maximization problem
        def objective(h):
            return -np.sum(h * np.roll(h_rev, k_max))
            
        # Gradient computation
        shifted_h = np.roll(h_rev, k_max)
        grad = shifted_h - h_current * shifted_h
        
        # Line search
        alpha = alpha0
        step = grad / (np.linalg.norm(grad) + 1e-12)
        for _ in range(max_backtrack):
            h_new = h_current - alpha * step
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_new = -objective(h_new)
            if c5_new <= c5 or alpha < 1e-10:
                h_current = h_new
                break
            alpha *= rho
            
        # Convex refinement every refine_freq iterations
        if iter_num % refine_freq == 0 and iter_num > 0:
            # Define convex optimization problem for worst overlap
            h_vars = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(cp.multiply(h_vars, np.roll(h_rev, k_max))))
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n_points / 2
            ]
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.GLPK, quiet=True)
                if prob.status == cp.OPTIMAL:
                    h_current = np.array(h_vars.value).flatten()
            except:
                pass  # Fall back to previous step if optimization fails
                
    # Final evaluation
    h_rev = 1 - best_h
    overlaps = np.correlate(best_h, h_rev, mode='full')
    overlap_values = overlaps[n_points-1:]
    final_c5 = np.max(overlap_values) * 2.0 / n_points
    
    return (list(best_h), float(final_c5), n_points)
