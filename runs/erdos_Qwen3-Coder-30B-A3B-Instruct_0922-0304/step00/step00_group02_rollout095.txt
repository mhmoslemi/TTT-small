import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    patience = 100
    perturbation_freq = 500
    perturb_ratio = 0.05
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_C5(h, n_points, dx)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        active_k = np.argmax(overlaps)
        max_overlap = overlaps[active_k]
        
        # Update with gradient step
        # Gradient of C5 w.r.t. h for active k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + active_k) % n_points
            grad[i] = (1 - h[j])  # Gradient contribution from active shift
        
        # Update h without constraints
        h_new = h - alpha * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to meet sum constraint
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new *= scale_factor
        
        # Evaluate new C5
        new_C5 = evaluate_C5(h_new, n_points, dx)
        
        # Accept if better
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            alpha *= gamma
            if alpha < alpha_min:
                break
        
        # Perturb occasionally
        if iter_count % perturbation_freq == 0 and iter_count > 0:
            indices = np.random.choice(n_points, int(n_points * perturb_ratio), replace=False)
            h[indices] += np.random.normal(0, 0.05, len(indices))
            h = np.clip(h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h *= scale_factor
        
        iter_count += 1
        if best_C5 < target_bound:
            break
    
    return (best_h, best_C5, n_points)

def evaluate_C5(h, n_points, dx):
    # Compute maximum overlap
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)
