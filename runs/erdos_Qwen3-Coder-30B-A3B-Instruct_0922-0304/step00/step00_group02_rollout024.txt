import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values or create a simple binary construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_no_improvement = 100
    no_improvement_count = 0
    iter_count = 0
    
    def evaluate_c5(h_values, dx):
        # Compute all overlaps efficiently
        h = h_values
        one_minus_h = 1 - h
        # np.correlate with mode='full' gives all lagged overlaps
        full_correlation = correlate(h, one_minus_h, mode='full')
        # The full correlation length is 2*n - 1 (both directions)
        # We want the maximum among the truncated overlaps
        # Each valid overlap i is from i-th lag for lags from 0 to 2*n-2 (inclusive)
        # But for each k, k must be within [0, n-1]
        # Lets compute all valid overlaps using the definition
        max_overlap = 0
        for k in range(n_points):
            overlap_sum = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h[i] * (1 - h[j])
            overlap_value = overlap_sum * dx
            max_overlap = max(max_overlap, overlap_value)
        return max_overlap
    
    def project_h(h_values, target_sum, dx):
        # Clip to [0, 1]
        h_clipped = np.clip(h_values, 0, 1)
        # Scale to match target sum
        current_sum = np.sum(h_clipped)
        if abs(current_sum) > 1e-10:
            scale_factor = target_sum / current_sum
            h_projected = h_clipped * scale_factor
        else:
            # If the sum is nearly 0, set to uniform
            h_projected = np.full_like(h_clipped, target_sum / len(h_clipped))
        return h_projected
    
    # Gradient-based optimization
    while time.time() - start_time < budget_s:
        # Compute overlaps
        overlaps = []
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        
        # Find maximum overlap index
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # If we have a new best, store it
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Compute gradient for the active shift
        # Here we're seeking the gradient of max overlap w.r.t h
        # Distance from each index to the max overlap set
        # For k-th shift, the gradient of overlap w.r.t h[i] is:
        # (1 - h[(i+k) % n])  (But we do it in reverse for modeling)
        
        # Simplified approach: use finite difference or analytical gradients 
        # What we want is:
        #      d/dh[i] sum_j h[j] * (1 - h[(j+k) % n]) = (1 - h[(i+k) % n])
        # So gradient for the kth shift is:
        # grad = [1 - h[(i+k) % n]] for i in range(n)
        
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + max_idx) % n_points
            grad[i] = 1 - h[j] 
                
        # Update h
        h_new = h - alpha * grad
        
        # Project back
        h = project_h(h_new, target_sum, dx)
        
        # Adaptive step size
        if no_improvement_count > 0 and no_improvement_count % 5 == 0:
            alpha *= gamma
        
        # Perturbation every few hundred iterations
        if iter_count % 500 == 0 and iter_count > 0:
            perturb = 0.05 * np.random.random(len(h)) - 0.025
            h = h + perturb
            h = project_h(h, target_sum, dx)
        
        iter_count += 1
        
        # Break if not improving
        if no_improvement_count > max_no_improvement:
            break
        
        # Break early if target reached
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
