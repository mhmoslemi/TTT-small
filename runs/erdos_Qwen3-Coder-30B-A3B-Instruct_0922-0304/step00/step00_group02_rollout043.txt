import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 45
    population_size = 100
    generations = 1000
    mutation_rate = 0.01
    elite_size = population_size // 2
    dx = 2.0 / n_points
    
    # Initialize population
    def create_individual():
        # Start with initial construction
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
        else:
            h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        return np.clip(h, 0, 1)
    
    def evaluate(individual):
        # Compute overlaps using correlate
        h = individual
        # Compute all overlaps for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        # We just care about the maximum overlap
        max_overlap = np.max(corr)
        c5 = max_overlap * dx
        return c5
    
    def repair(individual):
        # Ensure sum = n_points/2 and bounds
        h = individual.copy()
        current_sum = np.sum(h)
        target_sum = n_points / 2
        
        if current_sum != target_sum:
            scale = target_sum / current_sum
            h = h * scale
            
        h = np.clip(h, 0, 1)
        return h
    
    # Check bounds and sum
    def is_feasible(individual):
        h = individual
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - n_points/2) < 1e-8)
    
    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    for i in range(population_size):
        population[i] = repair(population[i])
    
    best_individual = None
    best_c5 = float('inf')
    generation_without_improvement = 0
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Evaluate population
        fitnesses = []
        for i, individual in enumerate(population):
            c5 = evaluate(individual)
            fitnesses.append(1.0 / (c5 + 1e-12))  # Avoid division by zero
            
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        current_best = population[0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_individual = current_best.copy()
            generation_without_improvement = 0
        else:
            generation_without_improvement += 1
            
        if generation_without_improvement > 100:
            break
            
        # Selection (top 50%)
        selected = population[:elite_size]
        
        # Crossover and mutation
        new_population = selected[:]
        
        # Generate offspring
        for _ in range(population_size - elite_size):
            # Tournament selection
            parent1 = selected[random.randint(0, elite_size-1)]
            parent2 = selected[random.randint(0, elite_size-1)]
            
            # Crossover: single point split
            point = random.randint(1, n_points-1)
            child = np.concatenate([parent1[:point], parent2[point:]])
            
            # Mutation: 1% of elements
            for i in range(n_points):
                if random.random() < mutation_rate:
                    child[i] += np.random.normal(0, 0.05)
            
            # Repair
            child = repair(child)
            
            new_population.append(child)
            
        population = new_population
    
    # Final evaluation
    if best_individual is not None:
        final_c5 = evaluate(best_individual)
        return (best_individual, final_c5, n_points)
    else:
        # Fallback
        test_individual = create_individual()
        test_individual = repair(test_individual)
        final_c5 = evaluate(test_individual)
        return (test_individual, final_c5, n_points)
