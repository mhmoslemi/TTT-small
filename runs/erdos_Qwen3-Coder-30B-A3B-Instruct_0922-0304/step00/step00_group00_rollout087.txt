import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h: use initial_h_values if available, otherwise start with uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Ensure sum equals n_points/2
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    max_iter = 500
    iter_no_improve = 0
    c5_threshold = 0.3807
    
    # Precompute dx
    dx = 2.0 / n_points
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Find the worst overlap (k that maximizes the integral)
        full_correlation = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(full_correlation)
        max_c5 = full_correlation[k_max] * dx
        
        if max_c5 < c5_threshold:
            break
            
        # Update best solution if improved
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            if iter_no_improve > 50:
                break

        # Formulate and solve convex problem
        # Variable vector h
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable()

        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            c5_var >= 0
        ]

        # Add specific constraint for the worst case shift
        # For shift k_max, we compute: sum h[i] * (1 - h[(i+k_max)%n])
        # This is equivalent to sum h[i] - h[i] * h[(i+k_max)%n]
        # So it's sum h[i] - sum h[i] * h[(i+k_max)%n] <= c5
        # Or equivalently: sum h[i] - sum h[i] * h[(i+k_max)%n] <= c5
        # i ranges from 0 to n-1.
        # Confine this to a named variable so we can build the proper expression
        # Let's calculate this as a quadratic constraint:
        dot_product = 0
        for i in range(n_points):
            shift_idx = (i + k_max) % n_points
            dot_product += h_var[i] * h_var[shift_idx]
        
        # Since the integral is sum h[i] * (1 - h[(i+k)%n]) * dx = (sum h[i] - dot_product) * dx
        # We want that to be <= c5 * dx, so sum h[i] - dot_product <= c5
        constraints.append(cp.sum(h_var) - dot_product <= c5_var)

        # Minimize c5
        objective = cp.Minimize(c5_var)
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.ECOS, max_iters=100, verbose=False)
            if prob.status == cp.OPTIMAL:
                h = h_var.value
                # Ensure bounds still hold
                h = np.clip(h, 0.0, 1.0)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass # Use previous h

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h):
    # Compute overlap integral using numpy's correlate function
    n = len(h)
    dx = 2.0 / n
    # Normalize h
    h = np.array(h)
    # Compute full correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # Find maximum
    max_overlap = np.max(correlation)
    return max_overlap * dx
