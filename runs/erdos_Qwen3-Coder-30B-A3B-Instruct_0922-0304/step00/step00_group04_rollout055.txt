import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap for the given step function h"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Fallback to a reasonable initialization
        n_points = 42
        h0 = np.ones(n_points) * 0.5
    
    # Discretization params
    dx = 2.0 / n_points
    
    # Function to optimize
    def objective(h):
        return evaluate_overlap(h, dx)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def constraint_bounds(h):
        return np.min(h), np.max(h) - 1.0
    
    # CVXPY approach for convex relaxation:
    try:
        # Define optimization variables
        h_vars = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2.0
        ]
        
        # Using a more direct approach for now: use scipy minimize
        # to try to locally optimize from initial values
        
        # Gradient-based optimization using scipy
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2.0}
        ]
        
        # Initial guess
        h_init = h0.copy()
        h_init = np.clip(h_init, 0, 1)
        
        # Optimize using L-BFGS-B approximation
        result = minimize(
            objective,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 10000, 'ftol': 1e-9, 'gtol': 1e-7}
        )
        
        if result.success and time.time() - start_time < budget_s * 0.9:
            h_result = result.x
        else:
            # Fallback to initial if optimization fails
            h_result = h0
        
        # Final evaluation
        c5_bound = objective(h_result)
        
        return (h_result, c5_bound, n_points)
        
    except Exception as e:
        # Fallback to just returning initial construction
        try:
            c5_bound = evaluate_overlap(h0, dx)
            return (h0, c5_bound, n_points)
        except:
            # One final fallback
            h_fallback = np.ones(n_points) * 0.5
            c5_fallback = evaluate_overlap(h_fallback, dx)
            return (h_fallback, c5_fallback, n_points)
    
    # If we got here, return best result
    return (h_result, c5_bound, n_points)
