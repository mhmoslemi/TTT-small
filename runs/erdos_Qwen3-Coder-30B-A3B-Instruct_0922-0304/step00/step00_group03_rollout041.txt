import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    dx = 2.0 / n_points

    # Adjust h to satisfy integral constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)

    # Initialize parameters
    alpha = 0.01
    momentum = 0.9
    vel = np.zeros_like(h)
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    max_iter = 5000
    restarts = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Check if time has expired
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate current overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        corr = corr * dx
        c5_current = np.max(corr)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Stop early if target achieved
        if best_c5 <= 0.3807:
            break

        # Identify max overlap index and shift
        max_idx = np.argmax(corr)
        k_max = max_idx - (len(h) - 1)
        ndx = np.arange(len(h))
        
        # Compute gradient with respect to h for that max shift
        # Create gradient for current correlation
        grad = np.zeros_like(h)
        k = k_max
        if k < 0:
            # h[i] appears in overlap with (1-h[i-k]) 
            # -> d/dh[i] -> - (1-h[i-k]) if i-k >= 0
            grad[:-k] -= (1 - h[-k:])
        elif k == 0:
            # Appears in overlap with (1-h[i])
            # -> d/dh[i] -> - (1-h[i]) = -1 + h[i]
            grad += h - 1
        else:
            # k > 0: h[i] appears with (1-h[i+k]) 
            # -> d/dh[i] -> - (1-h[i+k]) if i+k < len(h)
            grad[k:] -= (1 - h[:-k])

        # Update with gradient descent with momentum
        vel = momentum * vel + alpha * grad
        h_new = h - vel
        
        # Project: clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Check for stagnation or restart
        if np.allclose(h, h_new, atol=1e-6) and iter_count > 100:
            # Restart with slight random perturbation
            h = h * 0.999 + 0.001 * np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            vel.fill(0)
            restarts += 1
            if restarts > 10:
                break
        else:
            h = h_new
            
        iter_count += 1

    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    corr = corr * dx
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), float(final_c5), n_points)
