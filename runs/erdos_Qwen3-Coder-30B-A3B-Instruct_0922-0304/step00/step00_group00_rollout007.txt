import numpy as np
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all correlations for shifts k in -n+1 to n-1
    correlations = np.correlate(h, 1 - h, mode="full")
    # Only consider shifts 0 to n-1 (non-negative shifts)
    c5 = np.max(correlations[:n_points]) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        n_points = 45
        h0 = np.full(n_points, 0.5)
    
    # Ensure the integral constraint ∫₀² h(x) dx = 1
    dx = 2.0 / n_points
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    def obj_func(h):
        # Compute the maximum overlap for given h
        correlations = np.correlate(h, 1 - h, mode="full")
        return np.max(correlations[:n_points]) * dx
    
    # Use a simple gradient-based optimization approach
    # We produce a sequence of h_vals, each corresponding to a lower C5 value
    
    best_h = h0.copy()
    best_c5 = obj_func(best_h)
    best_iteration = 0
    
    # Try various methods to improve solution
    # Method 1: Gradient-based optimization with box constraints
    method1_iters = 0
    h_current = best_h.copy()
    
    # Define bounds for h (0 to 1)
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simple iterative descent
    for _ in range(5000):  # limit iterations
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate current function
        c5_current = obj_func(h_current)
        
        # Use finite differences to estimate gradient
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_minus = h_current.copy()
            h_minus[i] -= eps
            grad[i] = (obj_func(h_plus) - obj_func(h_minus)) / (2*eps)
        
        # Project gradient to satisfy constraint ∫ h dx = 1
        # Adjust gradient to maintain the integral constraint
        avg_grad = np.mean(grad)
        grad -= avg_grad
        
        # Perform step
        step_size = 0.01
        h_new = h_current - step_size * grad
        
        # Apply bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto integral constraint sum(h) = n/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        if obj_func(h_new) < obj_func(h_current):
            h_current = h_new
        else:
            # Reduce step size to avoid large step
            step_size *= 0.5
            
        # Update best if better
        c5_current = obj_func(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            best_iteration = method1_iters
            
        method1_iters += 1
    
    # Additional method: calling evaluate_erdos_solution on our best
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    return (best_h.tolist(), final_c5, n_points)
