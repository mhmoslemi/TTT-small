import numpy as np
from scipy import signal
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlations for all lags
    correlations = signal.correlate(h, 1 - h, mode="full")
    # Extract the valid lags (center to center)
    center = n_points - 1
    valid_correlations = correlations[center:2*center]
    # Maximum is the C5 bound
    return np.max(valid_correlations) * dx
