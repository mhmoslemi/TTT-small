import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import Variable, Minimize, Problem, sum_entries, multiply, norm
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup parameters
    n_points = 47
    dx = 2.0 / n_points
    max_iter = 1000
    refinement_freq = 50
    start_time = time.time()
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
        
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = compute_C5(h, n_points, dx)
    
    # Gradient descent with line search
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = np.correlate(h, 1 - h, mode='full')[n_points-1:]
        k_max = np.argmax(overlap)
        current_C5 = overlap[k_max] * dx
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Compute gradient for k_max shift
        shifted_h = np.roll(1 - h, k_max)
        gradient = shifted_h - h * shifted_h
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-10:
            continue
            
        gradient = gradient / grad_norm
        
        # Backtracking line search
        alpha = 0.01
        rho = 0.5
        max_steps = 30
        
        for _ in range(max_steps):
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            new_C5 = compute_C5(h_new, n_points, dx)
            if new_C5 < current_C5:
                h = h_new
                break
            alpha *= rho
    
        # Convex refinement
        if iteration % refinement_freq == 0:
            # Solve convex relaxation for the worst case
            h_cvx = solve_convex_refinement(h, k_max, n_points)
            if h_cvx is not None:
                h = h_cvx

    return (best_h, best_C5, n_points)

def compute_C5(h, n_points, dx):
    overlap = np.correlate(h, 1 - h, mode='full')[n_points-1:]
    return np.max(overlap) * dx

def solve_convex_refinement(h, k_max, n_points):
    # Reconstruct problem with fixed shift k_max
    h_vars = Variable(n_points)
    
    # Objective: minimize overlap for shift k_max
    sum_overlaps = 0
    for i in range(n_points):
        sum_overlaps += h_vars[i] * (1 - h_vars[(i - k_max) % n_points])
    
    # Constraints
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        sum_entries(h_vars) == n_points / 2
    ]
    
    # Solve
    try:
        prob = Problem(Minimize(sum_overlaps), constraints)
        prob.solve(solver=cp.GLPK, verbose=False)
        
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_new = np.array([h_vars[i].value for i in range(n_points)])
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return h_new
    except:
        pass
    
    return None
