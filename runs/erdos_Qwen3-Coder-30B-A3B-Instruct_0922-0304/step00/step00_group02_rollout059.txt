from scipy.signal import correlate

def evaluate_c5(h):
    # h is assumed to be normalized (sum(h) * dx == 1)
    dx = 2.0 / len(h)
    # compute all overlaps with 1-h
    corr = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
