def run(seed=42, budget_s=600, **kwargs):

def compute_all_overlaps(h, n_points, dx):
    """Compute the array of overlaps for all shifts."""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the overlap values
    overlaps = np.zeros(n_points)
    for k in range(n_points):
        overlaps[k] = corr[n_points - 1 + k] * dx
    return overlaps

def evaluate_c5(h, n_points, dx):
    """Evaluate the C5 value."""
    return np.max(np.correlate(h, 1 - h, mode='full') * dx)
