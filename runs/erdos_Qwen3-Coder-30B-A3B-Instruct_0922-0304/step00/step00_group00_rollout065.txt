def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h array"""
    n = len(h)
    dx = 2.0 / n
    overlaps = []
    for k in range(-n+1, n):
        overlap = np.sum(h * np.roll(1-h, k)) * dx
        overlaps.append(overlap)
    return max(overlaps)
