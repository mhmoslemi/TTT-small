import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 45  # Use the same size as initial_h_values
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = int(1e6)  # Safety limit
    cooling_rate = 0.995
    initial_T = 0.1
    delta = 0.1
    perturb_frac = 0.05  # 5% of points to perturb
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute overlap integral
    def compute_c5(h_vec):
        # Reflect h to compute all correlations
        h_reflected = np.concatenate([h_vec, np.zeros(n_points)])
        corr = correlate(h_reflected, 1 - h_vec, mode='full')
        max_corr = np.max(corr[n_points-1:])  # Take valid overlapping parts
        return max_corr * dx
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    T = initial_T
    
    # Main optimization loop
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb a fraction of indices
        num_perturb = max(1, int(perturb_frac * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Make a copy and perturb
        new_h = h.copy()
        for idx in indices:
            # Random walk step
            new_h[idx] += np.random.uniform(-delta, delta)
            # Clip to [0, 1]
            new_h[idx] = np.clip(new_h[idx], 0, 1)
        
        # Renormalize
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new energy
        new_c5 = compute_c5(new_h)
        
        # Metropolis accept/reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = new_h
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
                # Early stopping if target reached
                if best_c5 < target_bound:
                    break
        
        # Cool down
        T *= cooling_rate
    
    return (best_h.tolist(), best_c5, n_points)
