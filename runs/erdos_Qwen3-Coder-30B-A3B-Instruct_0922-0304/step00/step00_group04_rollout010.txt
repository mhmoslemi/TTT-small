import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h, dx=2.0/42):
    """Evaluate C5 for given h values."""
    if len(h) < 2:
        return 0.0
    # Compute full correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all lags (excluding zero lag)
    max_val = np.max(corr[len(h)-1:])  # Only positive lags
    return max_val * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize with the given construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Normalize to satisfy constraint
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    else:
        # Simple start
        h = np.ones(n_points) * 0.5
    
    start_time = time.time()
    
    # Gradient-based optimization approach:
    def objective(h):
        # Form the full correlation as a numpy convolution
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        # only consider positive lags (from 0 to n_points-1) 
        # and excluding zero (which would be the entire sum)
        vals = corr[len(h)-1:] # as we compute over range (0, n_points-1)
        return np.max(vals) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2  # must equal to n_points/2
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimize for several restarts with different initializations
    best_h = h.copy()
    best_c5 = objective(best_h)
    restarts = 0
    
    while time.time() - start_time < budget_s * 0.8:  # Reserve some time at end
        # Random restarts
        if restarts % 10 == 0:
            # Introduce some randomness
            new_h = np.random.uniform(0, 1, n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_h = np.clip(new_h, 0, 1)
        else:
            # Incrementally tweak current best
            new_h = best_h.copy()
            noise = np.random.normal(0, 0.02, n_points)
            new_h += noise
            new_h = np.clip(new_h, 0, 1)
            # Re-normalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        result = optimize.minimize(
            objective,
            new_h,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 500, 'ftol': 1e-6},
            callback=lambda x: None  # No callback during actual runtime
        )
        
        restarts += 1
        
        if result.success:
            test_h = np.clip(result.x, 0, 1)
            test_h = test_h * (n_points / 2) / np.sum(test_h)
            current_c5 = objective(test_h)
            if current_c5 < best_c5:
                best_h = test_h.copy()
                best_c5 = current_c5
        
        if time.time() - start_time > budget_s * 0.8:
            break
    
    # Final check and validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_erdos_solution(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
