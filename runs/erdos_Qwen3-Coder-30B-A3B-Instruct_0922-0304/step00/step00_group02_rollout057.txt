import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Random binary initialization with exactly n_points/2 ones
        h = np.zeros(n_points)
        ones = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones] = 1.0
    
    # Iteration loop
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points, dx)
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count - last_improvement < 100:
        iter_count += 1
        
        # Compute overlap for all lags
        overlaps = compute_all_overlaps(h, n_points)
        max_overlaps = np.max(overlaps, axis=1)
        
        # Update each position to minimize maximum overlap
        new_h = h.copy()
        # For each index, calculate if flipping improves the maximum overlap
        indices = list(range(n_points))
        np.random.shuffle(indices)
        
        num_ones = int(np.sum(h))
        target_ones = n_points // 2
        
        for i in indices:
            # Try flipping bit at i
            new_h[i] = 1 - h[i]
            new_overlaps = compute_all_overlaps(new_h, n_points)
            new_max_overlaps = np.max(new_overlaps, axis=1)
            
            old_overlap = max_overlaps[i]
            new_overlap = new_max_overlaps[i]
            
            # If this improves, keep the change
            if new_overlap < old_overlap or (new_overlap == old_overlap and 
                                           (h[i] == 1 and new_h[i] == 0) or 
                                           (h[i] == 0 and new_h[i] == 1)):
                h[i] = new_h[i]
            
            # Adjust for total number of ones
            if np.sum(h) > target_ones:
                # Find zeros to flip to one
                zeros_idx = np.where(h == 0)[0]
                if len(zeros_idx) > 0:
                    flip_idx = np.random.choice(zeros_idx)
                    h[flip_idx] = 1
            elif np.sum(h) < target_ones:
                # Find ones to flip to zero
                ones_idx = np.where(h == 1)[0]
                if len(ones_idx) > 0:
                    flip_idx = np.random.choice(ones_idx)
                    h[flip_idx] = 0
                    
            # Keep sum exactly at target
            current_ones = int(np.sum(h))
            if current_ones > target_ones:
                # Flip excess ones to zeros
                excess = current_ones - target_ones
                ones_idx = np.where(h == 1)[0]
                flip_indices = np.random.choice(ones_idx, size=excess, replace=False)
                h[flip_indices] = 0
            elif current_ones < target_ones:
                # Flip needed ones to ones
                deficit = target_ones - current_ones
                zeros_idx = np.where(h == 0)[0]
                flip_indices = np.random.choice(zeros_idx, size=deficit, replace=False)
                h[flip_indices] = 1
        
        # Revalidate that constraint is satisfied
        # Clip values to prevent numerical issues
        h = np.clip(h, 0, 1)
        
        # Maintain integral constraint
        current_sum = np.sum(h) * dx
        if abs(current_sum - 1.0) > 1e-10:
            h = h * (1.0 / current_sum)
        
        # Compute new C5
        new_c5 = compute_c5(h, n_points, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iter_count
        
        # Termination condition
        if best_c5 <= target_c5:
            break
    
    # Final computation to ensure correctness
    final_c5 = compute_c5(best_h, n_points, dx)
    return (best_h.tolist(), final_c5, n_points)

def compute_all_overlaps(h, n_points):
    """Compute all overlaps for each lag."""
    # The overlap with a shifted version by k is:
    # sum(h[i] * (1 - h[(i+k)%n_points]) for i in range(n_points))
    # This can be computed using correlation
    h_full = np.concatenate([h, h])
    ones = np.ones(n_points)
    overlaps = np.zeros((n_points, n_points))
    
    for k in range(n_points):
        shifted = np.roll(h_full, k)
        overlaps[:, k] = np.sum(h_full[:n_points] * (ones - shifted[:n_points]))
    
    return overlaps

def compute_c5(h, n_points, dx):
    """Compute the C5 bound."""
    # This is a full convolution with (1-h) then take max
    h_full = np.concatenate([h, h])
    h_complement = 1 - h_full[:n_points]
    correlations = signal.correlate(h_full[:n_points], h_complement, mode='full')
    # Take maximum over all lags
    max_correlation = np.max(correlations)
    return max_correlation * dx
