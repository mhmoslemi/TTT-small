import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate C5 for a given h and dx"""
    # Compute the overlap integral for all shifts
    overlaps = correlate(h, 1 - h, mode="full")
    # The maximum overlap is what we want to minimize
    return np.max(overlaps) * dx

def is_feasible(h):
    """Check if h is a feasible solution"""
    return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - len(h) / 2.0) < 1e-10

def normalize(h):
    """Normalize h to ensure sum(h) = N/2"""
    N = len(h)
    target_sum = N / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Adjust all values proportionally
    factor = target_sum / current_sum
    h = h * factor
    return h

def mutate(h, rate=0.01):
    """Mutate h by slightly adjusting values"""
    h = h.copy()
    N = len(h)
    num_mutations = int(N * rate)
    indices = np.random.choice(N, num_mutations, replace=False)
    for i in indices:
        # Add a small random perturbation
        delta = np.random.normal(0, 0.05)
        h[i] += delta
        # Clamp to [0,1]
        h[i] = np.clip(h[i], 0, 1)
    # Normalize to maintain sum constraint
    return normalize(h)

def crossover(p1, p2):
    """Create offspring from parents via segment swapping"""
    N = len(p1)
    # Choose a random split point
    split = np.random.randint(1, N)
    child1 = np.concatenate([p1[:split], p2[split:]])
    child2 = np.concatenate([p2[:split], p1[split:]])
    # Normalize children
    child1 = normalize(child1)
    child2 = normalize(child2)
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    N = 45
    pop_size = 100
    generations = 1000
    mutation_rate = 0.01
    
    # Initialize the population
    population = []
    
    # Add initial solution if it exists
    if 'initial_h_values' in globals():

        h0 = normalize(h0)
        population.append(h0)
    else:
        # Create a random initial solution
        h0 = np.random.rand(N)
        h0 = normalize(h0)
        population.append(h0)
    
    # Fill the rest of the population with random feasible solutions
    while len(population) < pop_size:
        h = np.random.rand(N)
        h = normalize(h)
        population.append(h)
    
    dx = 2.0 / N
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    for generation in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate(h, dx)
            fitnesses.append(1.0 / (c5 + 1e-12))  # Avoid division by zero
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection: take top 50%
        sorted_indices = np.argsort(fitnesses)[::-1]
        selected_indices = sorted_indices[:pop_size//2]
        selected = [population[i] for i in selected_indices]
        
        # Create new population
        new_population = []
        new_population.extend(selected)  # Keep the best
        
        # Generate offspring
        while len(new_population) < pop_size:
            if len(selected) >= 2:
                # Crossover
                i1, i2 = np.random.choice(len(selected), 2, replace=False)
                child1, child2 = crossover(selected[i1], selected[i2])
                
                # Mutate children
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                
                new_population.extend([child1, child2])
            else:
                # If only one individual, mutate it and add a random one
                new_h = mutate(selected[0], mutation_rate)
                random_h = np.random.rand(N)
                random_h = normalize(random_h)
                new_population.extend([new_h, random_h])
        
        population = new_population[:pop_size]  # Trim if needed
        
        # Increment improvement counter
        if generation > 0 and abs(best_c5 - (best_c5 if 'prev_best' in locals() else float('inf'))) < 1e-6:
            no_improvement_count += 1
        else:
            no_improvement_count = 0

    if best_h is None:
        # Fallback to a simple solution
        h = np.ones(N) * 0.5
        best_c5 = evaluate(h, dx)
        best_h = h
    
    # Final check of the best solution
    best_c5 = evaluate(best_h, dx)
    
    # Return the solution
    return (best_h.tolist(), best_c5, N)
