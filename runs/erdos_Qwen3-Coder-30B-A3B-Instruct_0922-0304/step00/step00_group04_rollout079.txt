import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Compute C5 for given h"""
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, dx, n_points):
    """Project h to feasible region"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to match sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Scale h to match target sum = 1.0
        scale = 1.0 / current_sum
        h = h * scale
    return h
