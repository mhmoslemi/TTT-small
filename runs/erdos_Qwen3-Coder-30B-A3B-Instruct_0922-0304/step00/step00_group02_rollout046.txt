import time
from scipy import signal

    alpha = 0.1
    alpha_min = 1e-6

    max_no_improve = 100
    no_improve_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vec):
        # Compute overlap integral
        overlap = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Rescale to match integral definition
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_proj = np.clip(h_vec, 0, 1)
        # Scale to sum = n_points/2
        scale = (n_points / 2) / np.sum(h_proj)
        h_proj = h_proj * scale
        return h_proj
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        
        # Compute current C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > max_no_improve or c5 < 0.3807:
            break
            
        # Compute overlaps for all shifts
        overlap_full = signal.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(overlap_full)

        eps = 1e-4
        grad = np.zeros(n_points)

            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h
        h_new = h - alpha * grad
        h_new = project_h(h_new)

    return (best_h, best_c5, n_points)
