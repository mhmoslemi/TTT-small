import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 10000
    patience = 500
    
    best_h = h.copy()
    best_C5 = float('inf')
    last_improvement = 0
    
    lambda_k = np.zeros(2 * n_points - 1)
    mu = 0.0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h using convex relaxation (quadratic regularization)
        h_var = cp.Variable(n_points)
        # Lagrangian with regularization
        overlaps = []
        for k in range(2*n_points - 1):
            # skip calculation for simplicity since we compute it below after solve
            pass
            
        # Create overlap matrix once
        h_full = np.pad(h, (0, n_points-1), 'constant')
        one_minus_h_full = np.pad(1-h, (0, n_points-1), 'constant')
        tmp_corr = correlate(h_full, one_minus_h_full, mode='full')
        overlap_values = tmp_corr[:2*n_points-1]
        
        # Use approximate gradient step via CVXPY
        obj = cp.Minimize(
            cp.sum(cp.multiply(lambda_k, overlap_values)) + 
            mu * (cp.sum(h_var) - n_points/2) +
            rho/2 * cp.sum_squares(h_var - h)
        )
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points/2
        ]
        
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
        except:
            pass  # fall back to previous h
            
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_new = h_var.value
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            h = h_new
        else:
            # early exit with old h
            h = np.clip(h, 0, 1)
            h = h * (n_points/2) / np.sum(h)
        
        # compute current overlaps
        h_full = np.pad(h, (0, n_points-1), 'constant')
        one_minus_h_full = np.pad(1-h, (0, n_points-1), 'constant')
        overlap_values = correlate(h_full, one_minus_h_full, mode='full')[:2*n_points-1]
        overlap_values *= dx
        
        C5 = np.max(overlap_values)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improvement = iter_count
            
        if C5 < 0.3807:
            break
            
        if iter_count - last_improvement > patience:
            break
            
        # Update dual variables
        for k in range(len(overlap_values)):
            lambda_k[k] += alpha * max(0, C5 - overlap_values[k]) 
        mu += alpha * (2 * np.sum(h) - n_points)
        
        # Entropy regularization for lambda
        lambda_k += beta * np.log(1 + np.abs(lambda_k))
        beta *= 0.999 if iter_count % 100 == 0 else 1.0

    h_opt = best_h.copy()
    c5_final = best_C5
    
    return (h_opt.tolist(), c5_final, len(h_opt))
