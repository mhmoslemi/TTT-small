import numpy as np

import time

def run(seed=42, budget_s=600, **kwargs):

    h = h0.copy()
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    # Gradient descent for optimization
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate current C5 and find the maximum overlap
        cross_corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(cross_corr)
        max_overlap = cross_corr[k_max] * dx
        C5 = max_overlap
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Compute gradient for the active constraint
        # Derivative of h[i](1 - h[i+k]) w.r.t. h[i] = 1 - h[i+k]
        # We compute active constraint gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            k = i - k_max + n_points - 1
            if 0 <= k < n_points:
                grad[i] = 1 - h[k]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad /= grad_norm
            
        # Step size
        step_size = 0.01
        
        # Update h in direction of minus gradient
        h_new = h - step_size * grad
        
        # Clip h to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust h to maintain sum constraint
        sum_h = np.sum(h_new)
        h_new = h_new * (n_points / 2) / sum_h
        
        # Update h
        h = h_new
        
        # Early exit if no significant improvement
        if iteration % 100 == 0:
            if abs(best_C5 - evaluate_C5(h, dx)) < 1e-6:
                break
    
    return (best_h, best_C5, n_points)

def evaluate_C5(h, dx):
    cross_corr = np.correlate(h, 1 - h, mode='full')
    return np.max(cross_corr) * dx
