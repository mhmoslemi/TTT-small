import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 47
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Use or initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points, dtype=float) * 0.5
    
    # Normalize h to satisfy the constraint ∫h = 1 (i.e., sum(h) * dx = 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s:
        # Compute current C5
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr[n_points-1:]) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if c5 <= target_c5:
            break

        if no_improvement_count > patience:
            # Attempt to escape local minimum
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0

        # Get peaks and their indices
        corr = signal.correlate(h, 1 - h, mode='full')
        overlap_values = corr[n_points-1:]
        peak_indices = np.argsort(overlap_values)[::-1][:5]  # Top 5 peaks

        # Update h to reduce peak values
        update = np.zeros(n_points)
        for idx in peak_indices:
            if idx > 0:  # Avoid boundary effects
                # This is a simplified model; increase h at lower shadow positions
                update[idx] -= 0.01
                update[(idx + 1) % n_points] += 0.01

        # Apply updates with small step size
        h_new = h + update * 0.02
        h_new = np.clip(h_new, 0, 1)
        # Project back to required sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

        iter_count += 1

        # Occasionally print progress
        if iter_count % 1000 == 0:
            print(f"Iteration {iter_count}, C5 = {best_c5:.6f}")

    if best_h is None:
        best_h = h.copy()

    # Final validation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr[n_points-1:]) * dx
    return (best_h.tolist(), final_c5, n_points)
