def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    def compute_overlap(h, k):
        n = len(h)
        shifted = np.roll(h, k)
        return np.sum(h * (1 - shifted))
    
    def compute_all_overlaps(h):
        n = len(h)
        overlaps = []
        for k in range(-n+1, n):
            overlaps.append(compute_overlap(h, k))
        return np.array(overlaps)
    
    def get_max_overlap(h):
        overlaps = compute_all_overlaps(h)
        return np.max(overlaps) * (2.0 / n_points)
    
    # Gradient descent + QP refinement loop
    max_iter = 10000
    iter_count = 0
    alpha = 0.1
    time_limit = start_time + budget_s * 0.9
    
    while time.time() < time_limit and iter_count < max_iter:
        iter_count += 1
        
        # 1. Compute overlaps for all shifts and find max
        overlaps = compute_all_overlaps(h)
        k_max = np.argmax(overlaps) - (len(h) - 1)
        max_overlap = overlaps[np.argmax(overlaps)]
        
        # 2. Gradient descent step
        grad = np.zeros_like(h)
        n = len(h)
        for i in range(n):
            j = (i + k_max) % n
            grad[i] += (1 - h[j])
            grad[j] -= h[i]
        
        # Step size and update
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement
        old_c5 = get_max_overlap(h)
        new_c5 = get_max_overlap(h_new)
        
        if new_c5 < old_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            alpha *= 0.9  # Reduce step size
            
        # Every 5 steps also do a refinement step
        if iter_count % 5 == 0:
            # Simple constraint-based refinement using QP (approximate)
            try:
                # Create a simple QP to refine around k_max
                # Linear constraint: sum(h) = n_points/2
                # Bound constraints: 0 <= h[i] <= 1
                # Try to reduce overlap for k_max
                h_prev = h.copy()
                
                # Try some modification
                for i in range(10):

                    test_h = h_prev + 0.01 * grad

                    if get_max_overlap(test_h) < get_max_overlap(h):
                        h = test_h.copy()
                        break
                        
            except:
                pass  # Skip if anything goes wrong
                
    # Final validation and return
    final_c5 = evaluate_erdos_solution(best_h)
