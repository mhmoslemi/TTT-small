import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h."""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    return np.max(overlaps)

def project_binary(h_relaxed, target_sum):
    """Project relaxed solution to binary with exact sum constraint."""
    # Sort indices by relaxed values (descending)
    idxs = np.argsort(h_relaxed)[::-1]
    
    # Threshold at 0.5
    h_binary = (h_relaxed > 0.5).astype(float)
    current_sum = np.sum(h_binary)
    
    # Adjust to match target_sum
    if current_sum > target_sum:
        # Need to turn off some 1's to reduce sum
        diff = int(current_sum - target_sum)
        for i in range(diff):
            # Turn off the smallest values that are currently 1
            for j in range(len(idxs)):
                if h_binary[idxs[j]] == 1:
                    h_binary[idxs[j]] = 0
                    break
    elif current_sum < target_sum:
        # Need to turn on some 0's to increase sum
        diff = int(target_sum - current_sum)
        for i in range(diff):
            # Turn on the largest values that are currently 0
            for j in range(len(idxs)):
                if h_binary[idxs[j]] == 0:
                    h_binary[idxs[j]] = 1
                    break
    
    return h_binary
