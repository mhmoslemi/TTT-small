import numpy as np
import cvxpy as cp
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Configuration
    n_points = 70
    dx = 2.0 / n_points
    target = 0.3807
    max_iter = 10000
    tol = 1e-4
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a simple initial guess
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Dual variables
    lambda_k = np.zeros(2 * n_points - 1)
    mu = 0.0
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    beta_decay = 0.9
    
    # Track best solution
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex subproblem for h
        h_var = cp.Variable(n_points)
        # Objective: minimize C5 + rho/2 * ||h - h_current||^2 + mu * (sum(h) - 1)
        # But we want to minimize max_k (h * (1-h))[k] * dx
        # So minimize: max_k (h * (1-h))[k] * dx  + rho/2 * ||h - h_current||^2
        
        # Compute overlaps first
        overlaps = signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(overlaps)
        
        # Create objective that is a convex approximation of the max constraint
        # formulation: minimize max_k over_k * dx + rho/2 * ||h - h_prev||^2
        # However, since max_k is non-smooth, we will use an iterative outer
        # approximation that keeps improving the max overlap.
        
        # We rewrite problem as constrained convex optimization
        # min_h max_k [ (h ⋆ (1−h))[k] * dx ]
        # subject to: sum(h) * dx = 1, 0 <= h_i <= 1
        
        # For computational efficiency here, we use CVXPY for solving
        # the convex subproblem given the current criterion
        
        objective = cp.Minimize(cp.sum_squares(h_var - h) * 0.5 * rho + max_overlap * dx)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            h_new = h_var.value
        except:
            h_new = h.copy()
            
        # Update h
        h = h_new
        
        # Compute overlaps
        overlaps = signal.correlate(h, 1 - h, mode='full')
        max_overlaps = np.max(overlaps)
        C5 = max_overlaps * dx
        
        # Update dual variables
        lambda_k += alpha * (max_overlaps - np.max(overlaps))
        mu += alpha * (np.sum(h) - n_points / 2)
        
        # Entropy regularization on lambda
        lambda_k += beta * np.log(1 + lambda_k)
        lambda_k = np.maximum(lambda_k, 0)
        
        # Track best solution
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
        
        # Stop if target is met
        if best_C5 < target:
            break
            
        # Decay entropy coefficient
        if iteration % 100 == 0:
            beta *= beta_decay
            
    # Final check and return
    final_h = best_h.copy()
    final_overlaps = signal.correlate(final_h, 1 - final_h, mode='full')
    final_C5 = np.max(final_overlaps) * dx
    
    return (final_h.tolist(), float(final_C5), n_points)
