        # Compute the full convolution (overlap integral)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # We only care about the overlapping parts
        max_corr = np.max(corr[len(corr)//2:]) * dx
        return max_corr
        
    def project_and_rescale(h_vals):
        # Project h to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to maintain sum = n_points / 2
        current_sum = np.sum(h_vals)
        target_sum = n_points / 2
        if current_sum > 0:
            h_vals = h_vals * target_sum / current_sum
        return h_vals

        # We use a combination of convex relaxation and gradient descent
        # First, let's make a convex approximation to get an initial solution
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        obj = cp.Minimize(cp.max(cp.conv(h, 1-h, method='direct')))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return np.array(h.value)
        except:
            return h_init
        return h_init

    # Convex optimization stage
    h = conv_optimize_stage(h, n_points, dx)
    h = project_and_rescale(h)

    # Gradient descent stage
    def objective_grad(params):
        h_local = params.copy()
        h_local = project_and_rescale(h_local)
        C5_val = evaluate_C5(h_local)
        return C5_val, h_local

    def gradient_estimate(h_vals, k):

        h_plus[k] = h_plus[k] + epsilon

        h_minus[k] = h_minus[k] - epsilon
        h_minus = project_and_rescale(h_minus)
        grad = (evaluate_C5(h_plus) - evaluate_C5(h_minus)) / (2 * epsilon)
        return grad
