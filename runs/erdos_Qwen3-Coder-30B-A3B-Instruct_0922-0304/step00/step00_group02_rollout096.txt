import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
        
    # Ensure constraint ∫h(x)dx = 1, i.e., sum(h)*dx = 1
    dx = 2.0 / n_points
    h = h * (1.0 / (np.sum(h) * dx))  # This will make sum(h) * dx = 1
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points, dx)
    
    # Parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 0
    max_patience = 100
    
    def project(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Scale to make sum(h)*dx = 1
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (1.0 / (h_sum * dx))
        return h
    
    def evaluate_c5(h, n, dx):
        # Compute all overlaps using correlation
        c = correlate(h, 1 - h, mode='full')
        # Only consider the relevant overlaps (shifts 0 to n-1)
        overlaps = c[n-1:] * dx
        return np.max(overlaps)
    
    # Gradient-based optimization loop
    i = 0
    while time.time() - start_time < budget_s - 0.1:
        i += 1
        # Evaluate current C5
        current_c5 = evaluate_c5(h, n_points, dx)
        
        # Compute overlaps
        c = correlate(h, 1 - h, mode='full')
        overlaps = c[n_points-1:] * dx
        k_active = np.argmax(overlaps)
        
        # Compute gradient w.r.t. active overlap
        grad = np.zeros(n_points)
        for i_idx in range(n_points):
            j_idx = (i_idx + k_active) % n_points
            grad[i_idx] = (1 - h[j_idx])
        
        # Gradient descent step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        h_new = project(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, n_points, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            patience = 0  # Reset patience
        else:
            patience += 1
            alpha *= gamma  # Decrease step size
            
        # Early termination
        if best_c5 < 0.3807:
            break
            
        # Check if we should perturb or terminate
        if patience > max_patience or alpha < alpha_min:
            if np.random.rand() < 0.1:  # 10% chance to perturb
                # Small random perturbation
                perturb = np.random.normal(0, 0.05, n_points)
                h = h + perturb
                h = project(h)
                patience = 0  # Reset patience
            
        # Occasionally reduce alpha
        if i % 100 == 0 and alpha >= alpha_min:
            alpha *= gamma
            
    return best_h.tolist(), best_c5, n_points
