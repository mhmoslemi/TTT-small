import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create uniform initial solution
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure constraint sum h * dx = 1 (i.e., sum(h) = n_points / 2)
    h_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(h_sum - target_sum) > 1e-10:
        h *= target_sum / h_sum
    
    dx = 2.0 / n_points
    
    def compute_c5(h_vec):
        # Compute overlap integral with mode='full'
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # The full correlation has length 2*n_points - 1
        # We take max over relevant lags (0 to n_points-1)
        # For lag k=0, we have h[0]*(1-h[0]) + h[1]*(1-h[1]) + ...
        # For lag k=1, we have h[0]*(1-h[1]) + h[1]*(1-h[2]) + ...
        # so idx = n_points - 1 + k with k = 0..n_points-1
        max_k = n_points - 1
        result = np.max(corr[n_points - 1 : n_points - 1 + max_k]) * dx
        return result
    
    # Initial evaluation
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturb_fraction = 0.05  # 5% of indices perturbed
    delta = 0.1  # initial perturbation size
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        if iteration % 1000 == 0:
            elapsed = time.time() - start_time
            if elapsed > 0.95 * budget_s:
                break
        
        # Select indices to perturb
        n_perturb = max(1, int(n_points * perturb_fraction))
        indices_to_perturb = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Make a copy of the current solution and perturb it
        h_new = h.copy()
        for idx in indices_to_perturb:
            perturbation = np.random.uniform(-delta, delta)
            h_new[idx] += perturbation
            # Clamp to [0, 1]
            h_new[idx] = np.clip(h_new[idx], 0, 1)
        
        # Renormalize to maintain constraint
        h_new_sum = np.sum(h_new)
        h_new *= target_sum / h_new_sum
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on Metropolis criterion
        dE = new_c5 - current_c5
        
        if dE < 0 or np.random.rand() < np.exp(-dE / T):
            h = h_new
            current_c5 = new_c5
            
            # Update best solution
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Cool temperature
        T *= cooling_rate
        
        # Reduce step size as we cool
        delta = 0.1 * (0.99 ** iteration)
        
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
