def run(seed=42, budget_s=600, **kwargs):

    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    alpha = 0.1
    last_improvement = 0
    iteration = 0
    start_time = time()
    
    while time() - start_time < budget_s - 0.1 and iteration - last_improvement < 50:
        # Compute overlaps for all shifts
        h_extended = np.concatenate([h, h])
        overlaps = []
        for k in range(-n_points+1, n_points):
            # Compute overlap for shift k
            if k >= 0:
                overlap = np.sum(h[:n_points-k] * (1 - h_extended[k:k+n_points-k]))
            else:
                overlap = np.sum(h[-k:] * (1 - h_extended[:n_points+k]))
            overlaps.append(overlap * dx)
        
        # Find worst shift
        k_max = np.argmax(overlaps)
        k = k_max - n_points + 1
        
        # Compute gradient
        grad = np.zeros(n_points)
        h_extended = np.concatenate([h, h])
        if k >= 0:
            for i in range(n_points - k):
                grad[i] += (1 - h_extended[i + k]) * dx
                grad[i + k] -= h[i] * dx
        else:
            for i in range(-k, n_points):
                grad[i] += (1 - h_extended[i + k]) * dx
                grad[i + k] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, n_points)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iteration
        
        iteration += 1
        # Adaptive step size
        if iteration % 10 == 0 and iteration > 0:
            alpha *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
