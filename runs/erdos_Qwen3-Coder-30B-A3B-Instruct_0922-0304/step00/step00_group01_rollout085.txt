from scipy import optimize
import time

    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_max_iter = 30
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5

    else:

    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts [0, n_points)
        correlation = np.correlate(h, 1 - h, mode='full')
        overlaps = correlation[n_points-1:] * dx
        
        # Find shift with maximum overlap
        k_max = np.argmax(overlaps)
        current_c5 = overlaps[k_max]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 < target_bound:
            break
            
        # Compute gradient for shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward difference term
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            # Backward difference term
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]
                
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-12:
            continue
            
        grad = grad / grad_norm
        
        # Line search
        alpha = alpha_initial
        for _ in range(line_search_max_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to unit integral
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new overlap
            correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
            overlaps_new = correlation_new[n_points-1:] * dx
            new_c5 = overlaps_new[k_max]
            
            if new_c5 < current_c5 or alpha < 1e-10:
                h = h_new
                break
            alpha *= rho

    # Final evaluation
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    overlaps_final = correlation_final[n_points-1:] * dx
    final_c5 = np.max(overlaps_final)
    
    return (best_h.tolist(), final_c5, n_points)
