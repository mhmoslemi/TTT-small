import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient descent parameters
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    max_iter = 1000
    patience = 100
    
    iter_since_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode='full')[n_points-1:]
        k_max = np.argmax(overlap)
        c5_current = overlap[k_max] * dx
        
        if c5_current < target_bound:
            break
            
        # Compute gradient for k_max
        gradient = np.zeros(n_points)
        h_clipped = np.concatenate([h, [0] * (n_points - 1)])
        h_clipped = np.concatenate([[0] * (n_points - 1), h_clipped])
        
        # Gradient computation
        for i in range(n_points):
            if i + k_max < n_points:
                gradient[i] = 1 - h_clipped[i + k_max]
            else:
                gradient[i] = 1
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
            
        gradient /= grad_norm
        
        # Line search
        alpha = alpha_initial
        valid_step = False
        
        for _ in range(30):
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to maintain sum = n_points / 2
            current_sum = np.sum(h_new)
            if abs(current_sum) > 1e-12:
                h_new *= (n_points / 2) / current_sum
                
            # Check improvement
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < c5_current:
                h = h_new
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    iter_since_improvement = 0
                else:
                    iter_since_improvement += 1
                valid_step = True
                break
            alpha *= rho
        
        if not valid_step:
            break
            
        if iter_since_improvement > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 for a given h."""
    overlap = correlate(h, 1 - h, mode='full')
    c5 = np.max(overlap[len(h)-1:]) * dx
    return c5
