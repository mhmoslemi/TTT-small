from scipy.optimize import minimize
import time

    start_time = time.time()

    target_sum = n_points / 2.0
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)

    current_sum = np.sum(initial_h)
    h = initial_h * (target_sum / current_sum)
    
    def evaluate_c5(h_vals):
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    def project_and_rescale(h_vals):
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to maintain constraint sum(h) = target_sum
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * (target_sum / current_sum)
        return h_vals
    
    # Gradient descent refinement
    def loss_func(h_vals):
        return evaluate_c5(h_vals)
    
    def gradient(h_vals, k):
        h_vals = np.array(h_vals)
        # Compute gradient through finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            h_plus = project_and_rescale(h_plus)
            h_minus = project_and_rescale(h_minus)
            grad[i] = (evaluate_c5(h_plus) - evaluate_c5(h_minus)) / (2 * eps)
        return grad
    
    # Iterative refine
    current_h = h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    lr = 0.1
    
    # Main optimization loop
    while time.time() - start_time < budget_s * 0.9:
        # Find dominant k
        corr = np.correlate(current_h, 1 - current_h, mode="full")

        # Gradient descent along this direction
        g = gradient(current_h, k_dominant)
        current_h = current_h - lr * g

        new_c5 = evaluate_c5(current_h)

            best_c5 = new_c5
        # Adapt learning rate

        corr = np.correlate(best_h, 1 - best_h, mode="full")
        best_c5 = np.max(corr) * dx
