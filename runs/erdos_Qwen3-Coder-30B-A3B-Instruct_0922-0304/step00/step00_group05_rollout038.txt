import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all lags
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:
        h = initial_h_values.copy()
    except NameError:
        # Create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization settings
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    last_improvement = 0
    
    max_iterations = budget_s * 1000  # Roughly 1000 iterations per second
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Check time budget
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all indices where the maximum overlap occurs
        max_lags = np.where(overlap == max_overlap)[0]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for lag in max_lags:
            # Handle convolution bounds
            if lag < n_points:
                valid = np.arange(0, n_points - lag)
                grad[valid] += (1 - h[valid + lag])
            else:
                valid = np.arange(lag - n_points + 1, n_points)
                grad[valid] += (1 - h[valid - (lag - n_points + 1)])
        
        grad /= len(max_lags)
        
        # Update h by gradient descent
        h -= learning_rate * grad
        
        # Project h to feasible region
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale so that sum(h)*dx = 1
        current_sum = np.sum(h) * dx
        h = h * (n_points / 2) / current_sum
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adapt learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
    
    return (best_h, best_c5, n_points)
