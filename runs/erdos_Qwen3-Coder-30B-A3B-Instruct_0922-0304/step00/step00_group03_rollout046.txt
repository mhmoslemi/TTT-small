import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)])
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 5000
    target_c5 = 0.3807
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Dual variables
    lambda_ = np.zeros(2 * n_points - 1)
    mu = 0.0
    
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode="full")
        return np.max(corr) * dx
    
    def update_h(h_vals, lambda_vals, mu_vals):
        # Formulate convex optimization problem
        h_var = cp.Variable(n_points)
        obj = cp.Minimize(
            cp.max(cp.multiply(lambda_vals, 
                              cp.conv(1 - h_vals, h_var, method='direct')[n_points-1:2*n_points-1])) +  
            mu_vals * (cp.sum(h_var) - 0.5 * n_points) 
            + 0.5 * rho * cp.sum_squares(h_var - h_vals)
        )
        constraints = [0 <= h_var, h_var <= 1]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
            return h_var.value
        except:
            return h_vals
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Update h using convex optimization
        h_new = update_h(h, lambda_, mu)
        
        # Compute overlaps
        overlaps = correlate(h_new, 1 - h_new, mode="full")
        c5_val = np.max(overlaps) * dx
        
        # Update dual variables
        max_over = np.max(overlaps)
        lambda_ += alpha * (np.maximum(0, max_over - c5_val + 1e-9) - lambda_)
        lambda_ = np.clip(lambda_, 0, 1e6)
        mu += alpha * (np.sum(h_new) - n_points / 2)
        
        # Entropy regularization
        lambda_ += beta * np.log(1 + lambda_)
        beta *= 0.9999 if iter_count % 100 == 0 else 1.0
        
        h = h_new
        
        # Check for better solution
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            last_improvement = iter_count
            
        if c5_val < target_c5:
            break
            
        if iter_count - last_improvement > 500:
            break
            
    return (best_h.tolist(), best_c5, n_points)
