import numpy as np
import cvxpy as cp
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for full correlation
    corr = correlate(h, 1 - h, mode='full')
    # Only take the relevant part (shifts 0 to n-1) and multiply by dx
    c5_val = np.max(corr[n-1:]) * dx
    return c5_val

            constraints = [
