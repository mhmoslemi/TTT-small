import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    
    # Use initial_h_values if available, otherwise create a binary one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Compute current overlap
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps)
        
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2.0
        ]
        
        # Objective: minimize max overlap
        max_overlap_var = cp.Variable()
        constraints.append(max_overlap_var >= overlaps)
        
        prob = cp.Problem(cp.Minimize(max_overlap_var), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                new_h = h_relaxed.value
                # Project back to binary
                threshold = 0.5
                
                # First attempt: threshold
                binary_h = (new_h > threshold).astype(float)
                
                # Adjust to fit sum constraint
                current_sum = np.sum(binary_h)
                target_sum = n_points / 2
                
                if current_sum != target_sum:
                    # Find indices to flip
                    diff = int(target_sum - current_sum)
                    if diff > 0:
                        # Flip lowest values from 0 to 1
                        flip_indices = np.argsort(new_h)[:diff]
                        binary_h[flip_indices] = 1.0
                    elif diff < 0:
                        # Flip highest values from 1 to 0
                        flip_indices = np.argsort(new_h)[diff:]
                        binary_h[flip_indices] = 0.0
                
                # Evaluate new solution
                C5_new = evaluate_erdos_solution(binary_h)
                
                if C5_new < best_C5:
                    best_h = binary_h.copy()
                    best_C5 = C5_new
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                
                h = binary_h.copy()
                
                # Perturb slightly
                perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for i in perturb_indices:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                
                # Renormalize
                h_sum = np.sum(h)
                h *= (n_points / 2) / h_sum
                
            else:
                # Fallback: perturb slightly if infeasible
                no_improvement_count += 1
                perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                for i in perturb_indices:
                    h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
                # Renormalize
                h_sum = np.sum(h)
                h *= (n_points / 2) / h_sum
                
        except Exception:
            # Fallback: perturb slightly
            no_improvement_count += 1
            perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
            for i in perturb_indices:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.05), 0, 1)
            # Renormalize
            h_sum = np.sum(h)
            h *= (n_points / 2) / h_sum
        
        iteration += 1
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(final_h) * dx
    if abs(final_sum - 1.0) > 1e-6:
        # Adjust to force exact integral
        final_h = final_h * (1.0 / final_sum)
    final_C5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
