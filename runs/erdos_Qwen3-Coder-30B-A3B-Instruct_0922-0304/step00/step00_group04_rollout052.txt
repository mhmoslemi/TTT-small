import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, else create random feasible h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    lr = 0.01
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        
        # Find maximum correlation value and its lag
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        k = max_idx - (n_points - 1)  # lag
        
        # Compute gradient direction for the dominant k
        grad = np.zeros(n_points)
        for i in range(n_points):
            if 0 <= i + k < n_points:
                grad[i] = -(1 - h[i + k])
            if 0 <= i - k < n_points:
                grad[i] += h[i - k]
        
        # Update h with gradient descent
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Check if this improves the objective
        new_c5 = compute_c5(h_new, n_points)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
        
        h = h_new
        
        # Adaptive learning rate
        if iter_count % 100 == 0:
            lr *= 0.95  # Exponential decay
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
