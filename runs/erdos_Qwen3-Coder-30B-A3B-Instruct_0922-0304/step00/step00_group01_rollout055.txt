import numpy as np

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the part that corresponds to valid shifts (0 to 2)
    # The full correlation has length 2*n - 1
    # We want the center part of length n, from shift -n+1 to n-1
    # So that's indices from n-1 to 2*n-2
    corr_centered = corr[n-1:2*n-1]
    # The shift k = 0, 1, ..., n-1 corresponds to the indices in the above
    # We want the maximum overlap, which requires scaling by dx
    max_overlap = np.max(corr_centered) * dx
    # But as the problem is stated, it's actually "max_k H(x)(1 - H(x+k)) dx"
    # So the correct case is the actual overlap integral
    return max_overlap

def project_to_feasible(h, n_points):
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Equalize the sum to n_points/2
    total = np.sum(h)
    target_sum = n_points / 2
    if abs(total - target_sum) < 1e-10:
        return h
    # Linear adjustment: shift the values
    # Add the difference to the middle elements
    diff = target_sum - total
    if abs(diff) > 1e-10:
        # Distribute the difference as evenly as possible
        # Simple method: adjust low values up and high values down
        h = h + diff / n_points
        h = np.clip(h, 0, 1)
    return h
