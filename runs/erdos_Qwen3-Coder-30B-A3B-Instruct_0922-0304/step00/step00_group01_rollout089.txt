import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 47
    dx = 2.0 / n_points
    h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    max_iter = 1000
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    patience = 100
    
    iter_count = 0
    no_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Compute all overlaps
        overlap_full = correlate(h, 1 - h, mode='full')
        overlaps = overlap_full[n_points - 1:] * dx
        k_max = np.argmax(overlaps)
        C5_current = overlaps[k_max]
        
        if C5_current < best_c5:
            best_c5 = C5_current
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if C5_current < 0.3807:
            break
            
        if no_improvement > patience:
            break
            
        # Compute gradient w.r.t. shift k_max
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # h[i] * (1 - h[i + k_max]) if i + k_max < n_points else 0
            # (1 - h[i + k_max]) if i + k_max < n_points else 0
            if i + k_max < n_points:
                # term 1: (1 - h[i + k_max])
                gradient[i] += 1 - h[i + k_max]
                # term 2: -h[i] * (1 - h[i + k_max]) -> -h[i] + h[i] * h[i + k_max]
                gradient[i] -= h[i] * (1 - h[i + k_max])
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
        gradient /= grad_norm
        
        # Line search
        alpha = alpha_initial
        for _ in range(30):
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            overlap_new = correlate(h_new, 1 - h_new, mode='full')[n_points - 1:] * dx
            C5_new = overlap_new[k_max]
            
            if C5_new < C5_current:
                h = h_new
                break
            else:
                alpha *= rho
                
        if time.time() - start_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    overlap_full = correlate(h, 1 - h, mode='full')
    overlaps = overlap_full[n_points - 1:] * dx
    return np.max(overlaps)
