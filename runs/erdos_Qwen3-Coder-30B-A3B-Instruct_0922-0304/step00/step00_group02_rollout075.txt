import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except:
        # Create default initial construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Validate bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Timing
    start_time = time.time()
    
    # Search parameters
    delta = 0.1
    decay_factor = 0.9
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        # Select two random indices
        i, j = np.random.randint(0, n_points, 2)
        
        # Adjust step size
        if iter_count % 100 == 0 and delta > 1e-6:
            delta *= decay_factor
            
        # Try small perturbation
        perturb = np.random.uniform(-delta, delta)
        h_new = h.copy()
        h_new[i] += perturb
        h_new[j] -= perturb
        
        # Enforce bounds
        if h_new[i] < 0:
            h_new[j] += h_new[i]
            h_new[i] = 0
        elif h_new[i] > 1:
            h_new[j] -= (h_new[i] - 1)
            h_new[i] = 1
        
        if h_new[j] < 0:
            h_new[i] += h_new[j]
            h_new[j] = 0
        elif h_new[j] > 1:
            h_new[i] -= (h_new[j] - 1)
            h_new[j] = 1
            
        # Adjust to satisfy sum constraint
        diff = np.sum(h_new) - (n_points / 2)
        if abs(diff) > 1e-10:
            # Add the difference to the last element
            h_new[-1] -= diff
            
        # Rebound
        h_new = np.clip(h_new, 0, 1)
            
        # Evaluate new configuration
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept if improvement
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
            
        iter_count += 1
        
        # Occasionally recompute full C5 to ensure precision
        if iter_count % 1000 == 0:
            c5_cached = evaluate_c5(best_h, dx)
            if abs(c5_cached - best_c5) > 1e-6:
                best_c5 = c5_cached
                
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Correlate h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum of the correlation values across all shifts
    max_correlation = np.max(corr)
    return max_correlation * dx
