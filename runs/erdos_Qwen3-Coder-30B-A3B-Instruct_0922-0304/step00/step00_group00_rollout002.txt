from scipy import signal
import time

    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Normalize to sum to n_points/2 exactly
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimize
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute overlap integral for all shifts k (correlation-like)
        # Compute full convolution of h and (1 - h) to get overlap for all k
        h_rev = 1 - h[::-1]
        full_corr = signal.correlate(h, h_rev, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        overlaps = full_corr * dx
        max_k_idx = np.argmax(overlaps)
        max_k = k_indices[max_k_idx]
        current_C5 = overlaps[max_k_idx]
        
        # Early termination if we reached target
        if current_C5 <= 0.3807:
            break
            
        # Compute gradient for this k
        grad = np.zeros(n_points)
        if max_k >= 0:
            # h[i] * (1 - h[i + max_k]) contributions
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k]) * dx
            # (1 - h[i]) * h[i - max_k] contributions
            for i in range(max_k, n_points):
                grad[i] -= h[i - max_k] * dx
        else:
            # h[i] * (1 - h[i + max_k]) contributions
            for i in range(n_points + max_k):
                grad[i] += (1 - h[i + max_k]) * dx
            # (1 - h[i]) * h[i - max_k] contributions
            for i in range(-max_k):
                grad[i] -= h[i - max_k] * dx

        # Update h using gradient descent
        alpha = 0.1 * np.exp(-iteration / 1000)
        h = h - alpha * grad
        
        # Ensure bounds and renormalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Track improvement
        if iteration % 50 == 0:
            if current_C5 <= 0.3807:
                break
            last_improvement = iteration
        iteration += 1
        
        # Avoid stagnation
        if iteration - last_improvement > 500 and current_C5 <= 0.38085:
            break
    
    # Final evaluation
    h_rev = 1 - h[::-1]
    full_corr = signal.correlate(h, h_rev, mode='full')
    k_indices = np.arange(-n_points + 1, n_points)

    final_C5 = np.max(overlaps)
