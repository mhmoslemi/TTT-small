import numpy as np
from scipy import correlate
import time

def evaluate_solution(h, dx):
    """Evaluate the C5 bound for given h values"""
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def ensure_feasible(h, n_points):
    """Ensure h has exactly n_points/2 ones and values in {0,1}"""
    # Clamp to {0,1}
    h = np.clip(h, 0, 1)
    # Ensure exactly n_points/2 ones
    target_ones = n_points // 2
    current_ones = int(np.sum(h))
    if current_ones > target_ones:
        # Need to turn some 1s into 0s
        indices = np.where(h == 1)[0]
        to_change = current_ones - target_ones
        for i in range(to_change):
            h[indices[i]] = 0
    elif current_ones < target_ones:
        # Need to turn some 0s into 1s
        indices = np.where(h == 0)[0]
        to_change = target_ones - current_ones
        for i in range(to_change):
            h[indices[i]] = 1
    return h

def get_neighborhood(h, n_points):
    """Generate neighboring solutions by flipping 2-3 bits while maintaining sum(h)=n_points/2"""
    neighbors = []
    target_ones = n_points // 2
    ones_indices = np.where(h == 1)[0]
    zeros_indices = np.where(h == 0)[0]
    
    # Try flipping 2 bits: 1 zero to 1, 1 one to 0
    if len(ones_indices) > 0 and len(zeros_indices) > 0:
        for i in range(min(10, len(ones_indices))):
            for j in range(min(10, len(zeros_indices))):
                new_h = h.copy()
                new_h[ones_indices[i]] = 0
                new_h[zeros_indices[j]] = 1
                neighbors.append(new_h)

    # Try flipping 3 bits: 2 zeros to 1s, 1 one to 0
    if len(ones_indices) > 0 and len(zeros_indices) >= 2:
        for i in range(min(5, len(ones_indices))):
            for j in range(min(5, len(zeros_indices))):
                for k in range(j+1, min(5, len(zeros_indices))):
                    new_h = h.copy()
                    new_h[ones_indices[i]] = 0
                    new_h[zeros_indices[j]] = 1
                    new_h[zeros_indices[k]] = 1
                    neighbors.append(new_h)

    return neighbors

def run(seed=42, budget_s=600, **kwargs):
    """Run tabu search to minimize C5 bound"""
    np.random.seed(seed)
    
    # Use the available initial construction if present
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        h = ensure_feasible(h, n_points)
    else:
        # Create a reasonable starting point
        n_points = 70
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = ensure_feasible(h, n_points)

    # Tabu search parameters
    tabu_list = []
    tabu_tenure = 100
    max_iter = 1000
    best_h = h.copy()
    best_c5 = evaluate_solution(h, dx)
    last_improvement = 0
    start_time = time.time()

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Generate neighbors
        neighbors = get_neighborhood(h, n_points)
        
        # Evaluate neighbors
        neighbor_evals = []
        for neighbor in neighbors:
            if tuple(neighbor) not in tabu_list:
                c5 = evaluate_solution(neighbor, dx)
                neighbor_evals.append((neighbor, c5))

        # Select best neighbor
        if not neighbor_evals:
            continue

        best_neighbor, best_neighbor_c5 = min(neighbor_evals, key=lambda x: x[1])

        # Update tabu list
        tabu_list.append(tuple(h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)

        # Update solution
        if best_neighbor_c5 < best_c5:
            h = best_neighbor.copy()
            best_c5 = best_neighbor_c5
            last_improvement = iteration
            best_h = h.copy()
        else:
            # Move to best neighbor anyway for exploration
            h = best_neighbor.copy()

        # Early stopping condition
        if iteration - last_improvement > max_iter // 4:
            break

    # Ensure final solution is valid
    final_h = ensure_feasible(best_h, n_points)
    final_c5 = evaluate_solution(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
