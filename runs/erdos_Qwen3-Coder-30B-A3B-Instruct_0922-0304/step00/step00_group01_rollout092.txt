import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute overlaps for all shifts
    correlations = np.correlate(h_values, 1 - h_values, mode='full')
    # Only consider valid overlaps (shifts 0 to n-1)
    overlaps = correlations[n-1:]
    # Compute the maximum overlap
    c5 = np.max(overlaps) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create a reasonable starting point
    n_points = 47
    try:
        h0 = initial_h_values.copy()
        h0 = h0 / np.sum(h0) * (n_points / 2)
    except:
        h0 = np.ones(n_points) * 0.5
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Initialize variables
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    dx = 2.0 / n_points
    
    # Helper to rescale h to satisfy sum constraint
    def rescale(h):
        return h * (n_points / 2) / np.sum(h)
    
    # Gradient descent with convex refinement
    max_iter = 2000
    iterations_without_improvement = 0
    max_no_improvement = 100
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        correlations = np.correlate(best_h, 1 - best_h, mode='full')
        overlaps = correlations[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient for k_max shift
        shifted = np.roll(1 - best_h, k_max)
        grad = shifted - best_h * shifted
        
        # Normalize gradient
        norm_grad = np.linalg.norm(grad)
        if norm_grad == 0:
            break
        grad = grad / norm_grad
        
        # Line search to find suitable step size
        alpha = 0.01
        rho = 0.5
        max_backtrack = 30
        for _ in range(max_backtrack):
            h_new = best_h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = rescale(h_new)
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < best_c5 - 1e-8:
                break
            alpha *= rho
        
        best_h = h_new
        new_c5 = evaluate_erdos_solution(best_h)
        
        if new_c5 < best_c5 - 1e-8:
            best_c5 = new_c5
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
        
        # Every 50 iterations, do convex refinement
        if iter_num % 50 == 0 and iter_num > 0:
            # Solve convex optimization problem for the current worst case
            k_max = np.argmax(overlaps)
            x = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(cp.multiply(x, 1 - cp.roll(x, k_max))))
            constraints = [
                cp.sum(x) == n_points / 2,
                x >= 0,
                x <= 1
            ]
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
                if prob.status == cp.OPTIMAL:
                    conv_h = x.value
                    conv_h = np.clip(conv_h, 0, 1)
                    conv_h = rescale(conv_h)
                    conv_c5 = evaluate_erdos_solution(conv_h)
                    if conv_c5 < best_c5:
                        best_h = conv_h
                        best_c5 = conv_c5
            except:
                pass
                
        if iterations_without_improvement > max_no_improvement:
            break
    
    # Return final result
    return (list(best_h), best_c5, n_points)
