import numpy as np

import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    correlations = np.correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlations)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):

        h_new = h.copy()
        # Use a simple gradient descent approximation for solving

        # Instead, solve the optimization with an L2 regularization:
        # min_h L(h, lambda, mu) + rho/2 * ||h||_2^2

        # Simplified formulation

        # Minimize max correlation * dx subject to h in [0,1], sum(h) = 1
        try:
            obj = Minimize(0.0)
            constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points/2]

            # and compute an empirical approximation to the max overlap
            obj_val = 0.0
            for i in range(len(current_overlap)):
                obj_val = max(obj_val, current_overlap[i])
            obj = Minimize(obj_val)
            
            # Use a small convex optimization approach to get updated h
            
            # For simplicity, we'll use L-BFGS optimization
            def objective(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode="full")
                return np.max(corr) * dx
            
            def jac(h_vec):
                corr = np.correlate(h_vec, 1 - h_vec, mode="full")
                k_max = np.argmax(corr)
                # Gradient of the max overlap with respect to h
                # We'll use central difference approximation
                eps = 1e-6
                grad = np.zeros_like(h_vec)
                for i in range(len(h_vec)):
                    h_plus = h_vec.copy()
