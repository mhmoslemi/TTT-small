import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    # Evaluate C5 using the provided evaluation method
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        n_points = len(h_initial)
    else:
        # Fallback initialization
        n_points = 42
        h_initial = np.ones(n_points) * 0.5  # Start with uniform
        h_initial[0] = 1.0  # Force some variation
        h_initial[1] = 0.0
        h_initial[2] = 1.0
    
    dx = 2.0 / n_points
    h_solution = h_initial.copy()
    c5_bound = evaluate_erdos_solution(h_solution, dx)
    
    # Try to solve with MILP
    # Setup for CVXPY MILP
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Constraints
    constraints = []
    
    # Sum constraint
    constraints.append(cp.sum(h) == n_points / 2)
    
    # Bounded constraints
    constraints.append(h >= 0)
    constraints.append(h <= 1)
    
    # Max overlap constraints for all lags
    max_lag = n_points - 1
    for k in range(-max_lag, max_lag + 1):
        corr_sum = 0
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i in [0, n_points - k - 1]
            for i in range(n_points - k):
                corr_sum += h[i] * (1 - h[i + k])
        else:
            # h[i - k] * (1 - h[i]) for i in [0, n_points + k - 1]
            for i in range(n_points + k):
                corr_sum += h[i - k] * (1 - h[i])
        constraints.append(corr_sum <= C5)
    
    # Formulate and solve
    prob = cp.Problem(objective, constraints)
    
    # Try to solve with different solvers
    solver_list = [cp.GLPK, cp.CBC, cp.ECOS, cp.SCS]
    for solver in solver_list:
        if time.time() - start_time > budget_s * 0.9:
            break
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_solution_opt = h.value
                c5_opt = C5.value
                c5_opt = max(0.0, c5_opt)  # Ensure non-negative
                # Check feasibility
                if (np.abs(np.sum(h_solution_opt) - n_points / 2) < 1e-6 and
                    np.all(h_solution_opt >= 0) and
                    np.all(h_solution_opt <= 1)):
                    h_solution = h_solution_opt
                    c5_bound = c5_opt
                    break
        except Exception:
            continue
    
    # Final validation using convolution
    c5_bound = evaluate_erdos_solution(h_solution, dx)
    
    # Ensure final h is binary if not already
    h_final = h_solution.copy()
    if np.any(h_final < 0.5):
        # Simple rounding to nearest binary values with correct sum
        order = np.argsort(h_final)
        s = np.sum(h_final)
        target = n_points / 2
        delta = s - target
        
        if delta > 0:
            # Need to set some to 0
            for i in range(int(delta)):
                h_final[order[i]] = 0.0
        else:
            # Need to set some to 1
            for i in range(int(-delta)):
                h_final[order[-(i+1)]] = 1.0
                
        # Adjust any floating point error
        diff = np.sum(h_final) - target
        if abs(diff) > 1e-6:
            # Find indices where value can be adjusted
            indices = np.where((h_final > 0) & (h_final < 1))[0]
            if len(indices) > 0:
                idx = indices[0]
                h_final[idx] = np.clip(h_final[idx] - diff, 0, 1)
    
    c5_final = evaluate_erdos_solution(h_final, dx)
    
    return (h_final.tolist(), c5_final, n_points)
