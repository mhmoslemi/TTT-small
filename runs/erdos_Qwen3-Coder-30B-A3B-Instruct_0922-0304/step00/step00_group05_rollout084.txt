import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialization
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[n_points//4:2*n_points//4] = 0.0
        h[2*n_points//4:3*n_points//4] = 1.0
        # Adjust to meet constraint sum(h) = n_points/2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        diff = target_sum - current_sum
        if abs(diff) > 1e-8:
            # Spread the difference
            for i in range(n_points):
                if diff > 0 and h[i] < 1.0:
                    add = min(diff, 1.0 - h[i])
                    h[i] += add
                    diff -= add
                elif diff < 0 and h[i] > 0.0:
                    sub = min(-diff, h[i])
                    h[i] -= sub
                    diff += sub
                if abs(diff) < 1e-8:
                    break
    
    # Simulated Annealing
    T_max = 1.0
    T_min = 1e-4
    n_iterations = 10000
    cooling_rate = np.exp(np.log(T_min / T_max) / n_iterations)
    T = T_max
    delta = 0.1
    
    best_h = h.copy()
    current_c5 = max(correlate(h, 1 - h, mode="full")) * dx
    best_c5 = current_c5
    
    start_time = time.time()
    for i in range(n_iterations):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Perturbation
        h_new = h.copy()
        i1, i2 = np.random.randint(0, n_points, 2)
        
        # Choose type of perturbation
        if np.random.rand() < 0.5:  # Swap
            val1, val2 = h_new[i1], h_new[i2]
            h_new[i1] = val2
            h_new[i2] = val1
        else:  # Shift mass between adjacent points
            i1 = np.random.randint(0, n_points-1)
            i2 = i1 + 1
            shift_amount = np.random.rand() * delta
            shift_amount = min(shift_amount, h_new[i1], 1.0 - h_new[i2])
            h_new[i1] -= shift_amount
            h_new[i2] += shift_amount
        
        # Apply bounds to boundary points
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraint
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-8:
            # Adjust at random points to preserve sum
            scale_factor = target_sum / current_sum
            for j in range(n_points//10):  # Slightly adjust some points
                idx = np.random.randint(0, n_points)
                h_new[idx] *= scale_factor
                h_new[idx] = np.clip(h_new[idx], 0, 1)
            
            # Ensuring constraint:
            # First try simple normalization over the entire array
            new_sum = np.sum(h_new)
            h_new *= (target_sum / new_sum)
            h_new = np.clip(h_new, 0, 1)
            # Re-adjust to exact target
            final_sum = np.sum(h_new)
            diff = target_sum - final_sum
            if abs(diff) > 1e-8:
                # Distribute the difference among all indices
                h_new += diff / n_points
                h_new = np.clip(h_new, 0, 1)
        
        # Compute new C5
        new_c5 = max(correlate(h_new, 1 - h_new, mode="full")) * dx
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Update temperature and perturbation size
        T *= cooling_rate
        if i % 100 == 0 and i > 0:
            # Adjust delta based on recent acceptance rate
            last_100 = [1 if np.random.rand() < np.exp(-delta_c5 / T) else 0 for _ in range(100)]
            accepted = sum(last_100)
            if accepted < 30:
                delta *= 0.95
            elif accepted > 70:
                delta *= 1.05
                
    return (list(best_h), best_c5, n_points)
