import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure normalization
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    last_improvement = 0
    alpha = 0.1
    
    while time.time() - start_time < budget_s * 0.9:
        iteration += 1
        
        # Estimate overlaps for all shifts
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            shift = np.roll(h, k)
            overlap = np.sum(h * (1 - shift)) * dx
            overlaps[k + n_points - 1] = overlap
        
        # Find worst shift
        k_max = np.argmax(overlaps) - (n_points - 1)
        
        # Gradient step
        grad = np.zeros(n_points)
        if k_max >= 0:
            for i in range(n_points):
                j = (i + k_max) % n_points
                grad[i] += (1 - h[j]) * dx
                if j < n_points:
                    grad[j] -= h[i] * dx
        else:
            for i in range(n_points):
                j = (i + k_max) % n_points
                grad[i] += (1 - h[j]) * dx
                if j >= 0:
                    grad[j] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Convex refinement
        if iteration % 5 == 0:
            k_worst = np.argmax(overlaps) - (n_points - 1)
            # Formulate QP for center of worst case shift
            w = cp.Variable(n_points)
            obj = cp.Minimize(cp.sum(w))
            constraints = [
                cp.sum(w) == 1,
                w >= 0,
                w <= 1,
            ]
            # Add constraint for the worst shift
            shift_worst = np.roll(h, k_worst)
            expr = cp.sum(cp.multiply(h, 1 - shift_worst))
            constraints.append(expr <= 0.5)  # Conservative bound
            
            prob = cp.Problem(obj, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    new_h = np.array(w.value)
                    h = new_h
                    h = h * (n_points / 2) / np.sum(h)
            except:
                pass
        
        # Evaluate and update best
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Adaptive learning rate
        if iteration - last_improvement > 50 and alpha > 0.01:
            alpha *= 0.7
        elif iteration - last_improvement < 5 and alpha < 0.5:
            alpha *= 1.1
            
        # Early termination if no improvement
        if iteration - last_improvement > 100:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    """Helper function to evaluate the current solution"""
    n = len(h)
    overlaps = []
    for k in range(-n + 1, n):
        shift = np.roll(h, k)
        overlap = np.sum(h * (1 - shift)) * dx
        overlaps.append(overlap)
    return max(overlaps)
