import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    except NameError:
        # Default initial construction
        n = 45
        h = np.full(n, 0.5)
    
    # Ensure the constraint sum(h) * dx = 1 => sum(h) = n/2
    dx = 2.0 / n
    h = h * (n / 2) / np.sum(h)
    
    # Clip to [0,1] and renormalize
    h = np.clip(h, 0, 1)
    h = h * (n / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = int(budget_s * 1000)  # ~1000 iterations per second
    patience = 50
    patience_counter = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts k
        # We need to compute full convolution for all valid shifts
        h_extended = np.concatenate([h, h])
        overlaps = []
        for k in range(-n+1, n):
            # Valid overlap is from k to n-1+k (with wrap-around)
            # Actually compute: h[i] * (1 - h[i+k]) 
            # Using scipy correlate:
            # correlate(h, 1-h, mode="full") gives all shifts
            # But we need only the ones where the correlation makes sense
            # We'll compute overlaps manually for safety
            if k >= 0:
                # h[0:n-k] * (1 - h[k:n])
                overlap = np.sum(h[:n-k] * (1 - h[k:]))
            else:
                # h[-k:n] * (1 - h[0:n+k])  
                overlap = np.sum(h[-k:] * (1 - h[:n+k]))
            overlaps.append(overlap)
        
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute gradient for k_max shift
        grad = np.zeros(n)
        if k_max >= 0:
            # For k_max >= 0, h[i] * (1 - h[i + k_max])  
            # Gradient w.r.t h[i]: (1 - h[i + k_max])
            # Gradient w.r.t h[i + k_max]: -h[i]
            # So gradient for i is (1 - h[i + k_max]) 
            # and for i + k_max is -h[i]
            for i in range(n - k_max):
                grad[i] += (1 - h[i + k_max]) * dx
            for i in range(n - k_max):
                grad[i + k_max] -= h[i] * dx
        else:
            # k_max < 0
            k_abs = -k_max
            for i in range(n - k_abs):
                grad[i] += (1 - h[i + k_abs]) * dx
            for i in range(n - k_abs):
                grad[i + k_abs] -= h[i] * dx
        # This computation is wrong because of misinterpretation of indices.
        # The right way with correlation:
        # But safe path is direct loop if it's too complex:
        # Simple and reliable:
        # Compute overlapping integrals for shift k_max and its gradient for h[i]
        # Adapted from the pattern:
        grad = np.zeros(n)
        
        # Do a full correlation
        one_minus_h = 1 - h
        corr = correlate(h, one_minus_h, mode='full')
        # The index for shift k is k + n-1 in full correlation
        # So for shift 0, index = n-1, for shift 1 index = n, etc.
        # For shift -1, index = n-2
        # For shift k, index = k + (n-1)
        
        # Correct way:
        k_max = np.argmax([np.sum(h[i] * (1 - h[(i+k)%n]) ) for k in range(-n+1, n)])
        max_corr = max([np.sum(h[i] * (1 - h[(i+k)%n]) ) for k in range(-n+1, n)])
        
        # Recompute more precisely:
        corr = correlate(h, one_minus_h, mode='full')
        shift_offsets = np.arange(-n+1, n)  # shifts
        k_max = shift_offsets[np.argmax(corr)]  # Find shift with max overlap
        # Now compute gradient for that k
        
        # Gradient for shift k: see how changing h[i] affects overlap at shift k
        # For overlap = sum over i of h[i] * (1 - h[i+k])
        # Gradient wrt h[i] = (1 - h[i+k])  
        # Although h[i+k] could be a circular index too.
        
        # Useful for direct computation, but will be accurate enough to use;
        # Do it manually for grayscale vector operations:
        grad = np.zeros(n)
        for i in range(n):
            j = (i + k_max) % n
            grad[i] += (1 - h[j]) * dx
        for i in range(n):
            j = (i - k_max) % n
            grad[j] -= h[i] * dx
        
        # Update h: h = h - alpha * grad
        h = h - alpha * grad
        
        # Ensure feasibility (0 <= h[i] <= 1 and sum = n/2)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n / 2) / h_sum

        # Evaluate current solution
        current_C5 = evaluate_erdos_solution(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            alpha *= 0.99  # gradually reduce learning rate
            patience_counter = 0

    # Ensure final return is feasible
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    best_h = best_h * (n / 2) / best_h_sum
    
    return best_h.tolist(), best_C5, n
