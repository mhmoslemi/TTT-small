import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Tabu search parameters
    tabu_tenure = 100
    tabu_list = []
    max_iter = 1000
    no_improve_iter = 0
    max_no_improve = max_iter // 4
    
    def evaluate_c5(h_vec, dx):
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        return np.max(corr) * dx
    
    def get_neighbors(h_vec):
        neighbors = []
        # Try flipping 2 or 3 bits while maintaining sum = n_points/2
        ones_indices = np.where(h_vec == 1)[0]
        zeros_indices = np.where(h_vec == 0)[0]
        
        # Generate neighbors by flipping 2 bits
        for i in range(min(10, len(ones_indices))):
            for j in range(i+1, min(10, len(ones_indices))):
                for k in range(min(10, len(zeros_indices))):
                    if len(neighbors) >= 50:
                        break
                    h_new = h_vec.copy()
                    h_new[ones_indices[i]] = 0
                    h_new[ones_indices[j]] = 0
                    h_new[zeros_indices[k]] = 1
                    if np.sum(h_new) == n_points/2:
                        neighbors.append(h_new)
        
        # Generate neighbors by flipping 3 bits
        for i in range(min(5, len(ones_indices))):
            for j in range(i+1, min(5, len(ones_indices))):
                for k in range(j+1, min(5, len(ones_indices))):
                    for l in range(min(5, len(zeros_indices))):
                        if len(neighbors) >= 50:
                            break
                        h_new = h_vec.copy()
                        h_new[ones_indices[i]] = 0
                        h_new[ones_indices[j]] = 0
                        h_new[ones_indices[k]] = 0
                        h_new[zeros_indices[l]] = 1
                        if np.sum(h_new) == n_points/2:
                            neighbors.append(h_new)
        
        return neighbors
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get neighbors
        neighbors = get_neighbors(h)
        
        # Remove tabu solutions
        filtered_neighbors = []
        for neighbor in neighbors:
            if not any(np.array_equal(neighbor, tabu) for tabu in tabu_list):
                filtered_neighbors.append(neighbor)
        
        if not filtered_neighbors:
            continue
            
        # Evaluate neighbors
        c5_values = [evaluate_c5(neighbor, dx) for neighbor in filtered_neighbors]
        
        # Find best neighbor
        best_neighbor_idx = np.argmin(c5_values)
        best_neighbor = filtered_neighbors[best_neighbor_idx]
        best_neighbor_c5 = c5_values[best_neighbor_idx]
        
        # Update if better
        if best_neighbor_c5 < best_c5:
            best_h = best_neighbor.copy()
            best_c5 = best_neighbor_c5
            no_improve_iter = 0
        else:
            no_improve_iter += 1
            
        # Update tabu list
        tabu_list.append(h)
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Update current solution
        h = best_neighbor.copy()
        
        # Early stop if no improvement for a while
        if no_improve_iter > max_no_improve:
            break

    return (best_h.tolist(), best_c5, n_points)
