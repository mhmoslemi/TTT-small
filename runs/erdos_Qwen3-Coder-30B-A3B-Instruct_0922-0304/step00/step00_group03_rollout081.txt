import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    # Compute correlation of h with (1-h) 
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value multiplied by dx
    return np.max(corr) * dx

def get_h_from_binary(h_binary, n_points):
    """Convert binary representation to h vector"""
    h = h_binary.astype(float)
    # Ensure that sum is exactly n_points/2 by adjusting
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    
    if current_sum != target_sum:
        # Simple correction by flipping bits
        diff = int(target_sum - current_sum)
        if diff > 0:
            # Flip some 0s to 1s
            zero_indices = np.where(h == 0)[0]
            if len(zero_indices) >= diff:
                for i in range(diff):
                    h[zero_indices[i]] = 1.0
        elif diff < 0:
            # Flip some 1s to 0s
            one_indices = np.where(h == 1)[0]
            if len(one_indices) >= -diff:
                for i in range(-diff):
                    h[one_indices[i]] = 0.0
    
    return h

def tabu_search_step(h_binary, n_points, dx, tabu_list, tabu_tenure, best_h, best_c5):
    """Perform one tabu search step"""
    current_h = get_h_from_binary(h_binary, n_points)
    current_c5 = evaluate_h(current_h, dx)
    
    # If current solution is better than best, update best
    if current_c5 < best_c5:
        best_h = current_h.copy()
        best_c5 = current_c5
    
    # Generate neighbors by flipping 2-3 bits
    neighbors = []
    for _ in range(20):  # Generate 20 neighbors
        # Pick 2 or 3 random indices to flip
        num_flips = random.choice([2, 3])
        indices = random.sample(range(n_points), num_flips)
        
        # Create neighbor by flipping the selected bits
        neighbor_binary = h_binary.copy()
        for i in indices:
            neighbor_binary[i] = 1 - neighbor_binary[i]
            
        # Ensure it's valid
        neighbor_h = get_h_from_binary(neighbor_binary, n_points)
        if abs(np.sum(neighbor_h) - n_points / 2.0) < 0.1:
            neighbors.append((neighbor_binary, evaluate_h(neighbor_h, dx)))
    
    # Return best neighbor that's not tabu
    best_neighbor = None
    best_neighbor_c5 = float('inf')
    
    for neighbor_binary, neighbor_c5 in neighbors:
        neighbor_tuple = tuple(neighbor_binary.tolist())
        if neighbor_tuple not in tabu_list:
            if neighbor_c5 < best_neighbor_c5:
                best_neighbor = neighbor_binary
                best_neighbor_c5 = neighbor_c5
    
    return best_neighbor, best_neighbor_c5, best_h, best_c5
