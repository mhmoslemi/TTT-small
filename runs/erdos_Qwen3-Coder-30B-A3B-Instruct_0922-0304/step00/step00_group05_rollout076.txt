import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: construct a simple initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        h[4] = 0.2
    
    # Scale to correct integral
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 5000
    gamma = 0.01
    decay_factor = 0.1
    patience = 100
    best_loss = float('inf')
    best_h = h.copy()
    
    def compute_C5(h_vals):
        # Calculate the full convolution
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The convolution result has length 2*n_points - 1
        # The maximum overlap is the maximum value among relevant elements
        # that correspond to different lags (k) from the integral
        max_val = max(conv)
        return max_val * dx
    
    def project_h(h_vals):
        # Clip to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Scale to correct integral
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def compute_gradient(h_vals, loss_lags):
        # Forward pass to get L
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Rotate and align correctly based on indexing
        # Full convolution indices go from -(n-1) to (n-1)
        # For overlap integral, we consider lags from -n+1 to n-1
        # We only care about positive lags (k) from 0 to n-1
        # That means we select from conv with indices from n-1 to 2*n-2
        # Actually, follow the actual formula for derivative
   
        # Derivative of integral w.r.t. h[i] = sum_{l where i+l is in range} (1-h[i+l])
        # So for each lag l, add (1-h[i+l]) if this is a valid index
        gradients = np.zeros_like(h_vals)
        # indices of the full cross-correlation where lags are from negative to positive
        # The full convolution indices correspond to lag from -(n-1) to (n-1)
        # Given that the correlation between h and 1-h gives 2*n -1 values,
        # index i in correlation corresponds to lag i-(n-1)
        # So we look where correlation[i] = L
        for i in range(len(h_vals)):
            for l in loss_lags:
                if i+l >= 0 and i+l < len(h_vals):
                    gradients[i] += 1 - h_vals[i+l]
        return gradients
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current
        loss = compute_C5(h)
        
        if loss < best_loss - 1e-6:
            best_loss = loss
            best_h = h.copy()
            last_improvement = iter_count
            
        # Compute gradients
        # Compute all correlations to identify maximums
        conv = np.correlate(h, 1 - h, mode="full")
        # Get indices for lags in [0, 2*n-2] that map to valid integral indices
        # We care about overlap integral with lag k = 0 to n-1, so from 1st half 
        # of full convolution of positive lags
        lags_with_max = []
        max_val = max(conv)
        for i in range(len(conv)):
            if abs(conv[i] - max_val) < 1e-6:
                lag = i - (n_points - 1)
                if lag >= 0:
                    lags_with_max.append(lag)
        
        # Compute gradient for each point based on loss lags
        grad = compute_gradient(h, lags_with_max)
        
        # Update
        h = h - gamma * grad
        
        # Project
        h = project_h(h)
        
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count - last_improvement > patience:
            gamma *= decay_factor
            last_improvement = iter_count
    
    # Final evaluation
    final_loss = compute_C5(best_h)
    
    return (best_h.tolist(), final_loss, n_points)
