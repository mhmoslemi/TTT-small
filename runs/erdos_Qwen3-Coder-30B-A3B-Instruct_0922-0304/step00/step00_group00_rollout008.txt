def compute_c5(h):
    # Recreate the overlap computation logic
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    # Full correlations: 
    # For each shift k, compute sum h[i] (1 - h[i+k mod n])
    max_overlap = 0.0
    for k in range(n):
        overlap = 0.0
        for i in range(n):
            overlap += h[i] * (1 - h[(i+k)%n])
        if overlap > max_overlap:
            max_overlap = overlap
    return max_overlap * dx
