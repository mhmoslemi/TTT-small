import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h values"""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_overlap = np.max(corr) * dx
    return max_overlap

def is_valid_solution(h, n_points):
    """Check if solution is valid (binary, correct sum)"""
    return (np.all((h == 0) | (h == 1)) and 
            abs(np.sum(h) - n_points / 2) < 1e-10)

def repair_solution(h, n_points):
    """Ensure solution has correct number of ones"""
    desired_sum = n_points // 2
    current_sum = int(np.sum(h))
    
    if current_sum == desired_sum:
        return h.copy()
    
    # Convert to binary if needed
    h_binary = (h > 0.5).astype(int)
    
    # Adjust to get exactly desired_sum ones
    diff = current_sum - desired_sum
    
    if diff > 0:
        # Need to remove some ones
        zero_indices = np.where(h_binary == 1)[0]
        remove_indices = zero_indices[:diff]
        h_binary[remove_indices] = 0
    elif diff < 0:
        # Need to add some ones
        one_indices = np.where(h_binary == 0)[0]
        add_indices = one_indices[:abs(diff)]
        h_binary[add_indices] = 1
    
    return h_binary

def generate_neighbor(h, n_points):
    """Generate a neighbor by flipping 2-3 bits"""
    # Make copy of current solution
    new_h = h.copy()
    n_flips = np.random.randint(2, 4)  # 2 or 3 flips
    
    # Get indices of 0s and 1s
    zeros = np.where(new_h == 0)[0]
    ones = np.where(new_h == 1)[0]
    
    if len(zeros) == 0 or len(ones) == 0:
        # If all same, just flip a few anyway
        indices = np.random.choice(n_points, size=n_flips, replace=False)
        new_h[indices] = 1 - new_h[indices]
        return new_h
    
    # Flip some 0s to 1s and some 1s to 0s
    flip_zeros = np.random.choice(zeros, size=min(n_flips//2, len(zeros)), replace=False)
    flip_ones = np.random.choice(ones, size=n_flips - len(flip_zeros), replace=False)
    
    new_h[flip_zeros] = 1
    new_h[flip_ones] = 0
    
    return new_h

def tabu_search_h(n_points, max_iterations, tabu_tenure, start_h):
    """Tabu search for finding optimal h"""
    dx = 2.0 / n_points
    current_h = repair_solution(start_h, n_points)
    best_h = current_h.copy()
    best_value = evaluate_h(current_h, dx)
    
    tabu_list = []
    tabu_tenure = min(tabu_tenure, max_iterations // 4)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > 590:  # Allow some buffer
            break
            
        # Generate neighbors
        neighbors = []
        for _ in range(50):  # Generate 50 neighbors
            neighbor = generate_neighbor(current_h, n_points)
            if is_valid_solution(neighbor, n_points):
                neighbors.append(neighbor)
        
        # Evaluate neighbors
        if not neighbors:
            continue
            
        best_neighbor = None
        best_neighbor_value = float('inf')
        
        for neighbor in neighbors:
            neighbor_tuple = tuple(neighbor)
            if neighbor_tuple in tabu_list:
                continue
                
            value = evaluate_h(neighbor, dx)
            if value < best_neighbor_value:
                best_neighbor = neighbor
                best_neighbor_value = value
        
        if best_neighbor is None:
            continue
            
        current_h = best_neighbor
        current_value = best_neighbor_value
        
        # Update best if improved
        if current_value < best_value:
            best_h = current_h.copy()
            best_value = current_value
            
        # Update tabu list
        tabu_list.append(tuple(current_h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Occasionally stop early if no improvement
        if iteration > max_iterations // 4 and \
           best_value < 0.3809: 
            # Early exit if we're close to target
            break
            
    return best_h, best_value
