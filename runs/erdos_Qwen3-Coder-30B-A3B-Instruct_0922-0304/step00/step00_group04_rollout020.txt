import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 bound for given h_values"""
    n = len(h_values)
    dx = 2.0 / n
    # Compute full correlation
    corr = np.correlate(h_values, 1 - h_values, mode='full')
    # The maximum value is what we want for C5
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Fallback: construct simple piecewise constant
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        # Adjust to satisfy sum constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h = h_init.copy().astype(float)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Convex relaxation helper
    def solve_convex_relaxation(h_in, n_points):
        # Create variables
        h_vars = cp.Variable(n_points)
        # Objective
        obj = cp.Minimize(cp.sum_squares(h_vars - h_in))
        # Constraints
        constraints = [0 <= h_vars, h_vars <= 1, 
                       cp.sum(h_vars) == n_points / 2]
        # Problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_vars.value
                return h_new
        except:
            return h_in
        return h_in
    
    # Run optimization
    time_left_s = budget_s - (time.time() - start_time)
    if time_left_s <= 10:
        return (best_h, best_c5, n_points)
    
    # Stage 1: Convex relaxation
    time_left_s = budget_s - (time.time() - start_time)
    if time_left_s > 20:
        h = solve_convex_relaxation(h, n_points)
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Stage 2: Gradient-based refinement with fixed step
    def compute_gradient(h, k):
        # Compute gradient for lag k
        nh = len(h)
        dx = 2.0 / nh
        # Shifted h at lag k
        h_shifted = np.concatenate([h[k:], h[:k]])
        # Gradient is -(1 - 2*h) for correlation
        return -(1 - 2*h) * dx
    
    def step_descent(h_in, n_iter=200, eta=0.02):
        h = h_in.copy()
        for i in range(n_iter):
            # Compute all correlation values
            corr = np.correlate(h, 1 - h, mode='full')
            # Identify dominant lag
            max_pos = np.argmax(corr)
            # If max_pos past center, adjust
            lag = max_pos - (len(h) - 1)
            # Compute gradient for this lag
            grad = compute_gradient(h, lag % len(h))
            # Update h
            h = h - eta * grad
            # Project
            h = np.clip(h, 0, 1)
            # Rescale
            h = h * (n_points / 2) / np.sum(h)
            if i % 50 == 0:
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
            if time.time() - start_time > budget_s - 5:
                return h
        return h
    
    # Refine
    time_left_s = budget_s - (time.time() - start_time)
    if time_left_s > 100:
        h = step_descent(h, n_iter=200, eta=0.02)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Stage 3: Upsample and refine
    if time.time() - start_time < budget_s - 100:
        # Create new resolution 2x
        n_points_up = 2 * n_points
        h_up = np.interp(np.linspace(0, 2, n_points_up), 
                         np.linspace(0, 2, n_points), 
                         h)
        # Now refine
        h_up = solve_convex_relaxation(h_up, n_points_up)
        h_up = step_descent(h_up, n_iter=100, eta=0.01)
        c5_up = evaluate_erdos_solution(h_up)
        if c5_up < best_c5:
            best_c5 = c5_up
            best_h = h_up.copy()
            n_points = n_points_up
    
    return (best_h, best_c5, n_points)
