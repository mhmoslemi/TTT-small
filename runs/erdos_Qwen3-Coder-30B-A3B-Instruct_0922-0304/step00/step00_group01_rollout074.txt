import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 47
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.01
    min_alpha = 1e-5
    patience = 100
    patience_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        # Using scipy correlate to compute overlaps efficiently
        overlap = np.zeros(n_points)
        for k in range(n_points):
            # Compute correlation h[i] * (1 - h[(i-k) % n])
            shifted = np.roll(h, k)
            overlap[k] = np.sum(h * (1 - shifted))
        
        # Find the worst case shift k_max that maximizes overlap
        k_max = np.argmax(overlap)
        max_overlap = overlap[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Early stopping condition
        if best_c5 <= 0.3807:
            break
            
        if patience_count >= patience:
            break
            
        # Compute gradient: for k_max
        g = np.zeros(n_points)
        shifted = np.roll(h, k_max)
        for i in range(n_points):
            g[i] = (1 - shifted[i]) - h[i]
            
        # Update h
        h = h - alpha * g
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint: sum = n/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if iteration % 10 == 0 and iteration > 0:
            # If we're not improving much, reduce step size
            if patience_count > 50:
                alpha = max(alpha * 0.5, min_alpha)
    
    # Final evaluation
    overlap = np.zeros(n_points)
    for k in range(n_points):
        shifted = np.roll(best_h, k)
        overlap[k] = np.sum(best_h * (1 - shifted))
    
    final_c5 = np.max(overlap)
    
    return (best_h, final_c5, n_points)
