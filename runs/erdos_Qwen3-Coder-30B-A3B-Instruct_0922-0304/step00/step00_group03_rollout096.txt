import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points // 2)
        h = np.concatenate([h, np.zeros(n_points - len(h))])
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 1000
    patience = 500
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Dual variables
    lambda_k = np.zeros(2 * n_points - 1)
    mu = 0
    
    # Precompute 1-h for efficiency
    one_minus_h = 1 - h
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h by solving: min_h L + rho/2 ||h||_2^2
        # L = sum_k lambda_k * (h ⋆ (1-h))[k] + mu * (sum(h) - n_points/2)
        # We approximate this via cvxpy convex subproblem
        
        # Compute current overlaps
        overlaps = correlate(h, one_minus_h, mode='full')
        current_c5 = np.max(overlaps) * dx
        
        # If we found a better C5, store it
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 < target_c5:
            break
            
        # Solve convex subproblem for h
        h_var = cp.Variable(n_points)
        
        # Objective with regularization and constraints
        obj = cp.Minimize(
            cp.sum(cp.multiply(lambda_k, overlaps)) * dx + 
            mu * (cp.sum(h_var) - n_points / 2) + 
            rho / 2 * cp.sum_squares(h_var)
        )
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, max_iters=1000, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_ERR]:
                h_new = h_var.value
                h = h_new
            else:
                # Fallback if optimization fails
                h = h * 0.99 + 0.01 * np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback approach
            h = h * 0.99 + 0.01 * np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Update dual variables
        new_overlaps = correlate(h, one_minus_h, mode='full')
        lambda_k += alpha * np.maximum(0, new_overlaps * dx + 1e-6 - current_c5)
        mu += alpha * (np.sum(h) - n_points / 2)
        
        # Entropy regularization for lambda
        lambda_k = lambda_k + beta * np.log(1 + lambda_k)
        lambda_k = np.clip(lambda_k, -1e6, 1e6)
        
        # Heavy clipping to maintain feasibility
        lambda_k = np.clip(lambda_k, 0, 1e6)
        mu = np.clip(mu, -1e6, 1e6)
        
        # Check for convergence
        if iteration > 0 and iteration % 100 == 0:
            beta *= 0.9
            alpha *= 0.95
            
    # Verify final solution
    final_overlaps = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlaps) * dx
    
    return (best_h.tolist(), final_c5, n_points)
