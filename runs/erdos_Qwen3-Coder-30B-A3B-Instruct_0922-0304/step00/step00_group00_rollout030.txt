import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h = h / np.sum(h) * (n_points / 2)
    
    best_c5 = np.inf
    best_h = h.copy()
    
    # Helper function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        overlaps = np.correlate(h, 1 - h, mode='full')
        return np.max(overlaps) * dx
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1 and iter_count - last_improvement < 50:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
            
        # Compute current overlaps to find most constraining
        h = np.array(h)
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max] * dx

        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if max_overlap <= 0.3807:
            break
            
        # Build constraint matrix
        # Use a linear approximation with McCormick relaxations
        # For a product h[i] * (1 - h[j]), we add constraints based on:
        # Let z[i,j] = h[i] * (1 - h[j]) 
        # Then: z <= h[i], z <= 1 - h[j], z >= h[i] + h[j] - 1, z >= 0
        
        # Simple approach: increase the problematic overlap by adding constraints
        # This is a greedy approach to guide the solver
        
        # Reinitialize with more focused optimization
        # Use scipy minimize with bounds and constraints to refine
        bounds = [(0, 1) for _ in range(n_points)]
        def objective(h_vals):
            return compute_c5(h_vals)
            
        def constraint_func(h_vals):
            return np.sum(h_vals) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_func}
        
        try:
            h_new = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'ftol': 1e-8})
            if h_new.success:
                h = h_new.x
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
            else:
                # fallback
                h = h / np.sum(h) * (n_points / 2)
                h = np.clip(h, 0, 1)
        except:
            h = h / np.sum(h) * (n_points / 2)
            h = np.clip(h, 0, 1)
            
        iter_count += 1
        last_improvement = iter_count if max_overlap < best_c5 else last_improvement
        
        if iter_count % 10 == 0:
            #do periodic full evaluation
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
    
    best_h = np.clip(best_h, 0, 1)
    final_sum = np.sum(best_h)
    if final_sum > 0:
        best_h = best_h * (n_points / 2) / final_sum
        
    return (best_h.tolist(), best_c5, n_points)
