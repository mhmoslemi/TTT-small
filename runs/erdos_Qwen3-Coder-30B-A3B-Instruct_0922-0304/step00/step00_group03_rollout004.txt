import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: Create a basic construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure constraint is met
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Gradient-based optimization
    max_iter = 10000
    step_size = 1e-4
    tol = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and identify k that produces maximum overlap
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(cross_corr)
        max_overlap = cross_corr[k_max] * dx
        
        # Update best if found better
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Gradient computation for the k_max case
        # Gradient of S_k(h) = sum h[i] * (1 - h[i + k]) wrt h[i]
        # This can be expressed as: g[i] = (1 - h[i + k]) - h[i + k] * delta_{i, i+k} 
        # But it's easier to compute directly by finite differences or Lagrange multipliers
        
        # Simple gradient descent approach with projection to constraints
        grad = np.zeros_like(h)
        # Compute gradient for the active constraint
        for j in range(n_points):
            k_idx = k_max - n_points + 1 + j
            if 0 <= k_idx < n_points:
                grad[j] = (1 - h[k_idx]) - h[k_idx]
        
        # Update h in the direction of negative gradient
        h_new = h - step_size * grad
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project to feasible sum
        current_sum = np.sum(h_new)
        h_new = h_new * target_sum / current_sum
        
        # Check stopping criterion
        if np.allclose(h, h_new, atol=tol):
            break
            
        h = h_new
    
    # Final verification and return
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), int(n_points))

def compute_c5(h, dx):
    # Compute cross correlation  
    corr = signal.correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx
