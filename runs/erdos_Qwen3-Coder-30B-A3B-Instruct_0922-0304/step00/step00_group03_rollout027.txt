import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction if no initial_h_values
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure initial integral constraint
    h *= (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization
    max_iter = 5000
    step_size = 0.01
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and identify worst shift
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5_vals = cross_corr * dx
        max_k = np.argmax(c5_vals)
        max_val = c5_vals[max_k]
        
        if max_val < best_c5:
            best_h = h.copy()
            best_c5 = max_val
            
        if max_val < 0.3807:
            break
            
        # Compute gradient of max overlap w.r.t. h
        # For shift k, gradient = 1 - 2*h[i] at positions where the overlap is maximized
        # We do this by considering the derivative of the convolution w.r.t. h_i
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Local effect of changing h[i]
            # Consider all terms involving h[i] in the overlap sum
            for j in range(n_points):
                k_idx = j - i + n_points - 1  # shift index
                if 0 <= k_idx < len(cross_corr):
                    # For h[i] in the overlap at shift k_idx,
                    # we have term h[i] * (1 - h[j]) in total overlap
                    if j >= i:
                        grad[i] -= dx  # simplified gradient approximation
                    else:
                        grad[i] += dx
        
        # Normalize gradient
        grad = grad / (np.linalg.norm(grad) + 1e-12)
        
        # Update h
        h_new = h - step_size * grad
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Check descent
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < max_val:
            h = h_new
        else:
            step_size *= 0.95  # Reduce step size if no progress
            
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    # Compute full convolution
    corr = np.correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx
