def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    C5_vals = []
    for k in range(n):
        shift = np.roll(h, k)
        overlap = np.sum(h * (1 - shift)) * dx
        C5_vals.append(overlap)
    return max(C5_vals)
