import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    dx = 2.0 / n_points
    current_sum = np.sum(h0) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Adjust to make sum = 1
        required_scale = 1.0 / current_sum
        h0 = h0 * required_scale
    
    # Normalize to valid range
    h0 = np.clip(h0, 0, 1)
    h0 = h0 / np.sum(h0) * n_points / 2
    
    def evaluate_h(h_vec):
        # Compute the overlap integral
        h = np.array(h_vec)
        h = np.clip(h, 0, 1)
        # Evaluate all overlaps via full convolution
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)
    
    # Binary search setup
    t_low, t_high = 0.0, 0.5
    tolerance = 1e-4
    best_h = h0.copy()
    best_c5 = evaluate_h(h0)
    
    while t_high - t_low > tolerance and (time.time() - start_time) < budget_s - 1:
        t_mid = (t_low + t_high) / 2.0
        
        # Convex feasibility formulation
        h = cp.Variable(n_points)
        # Product constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0,
        ]
        
        # Add overlap constraints for each k
        for k in range(1 - n_points, n_points):
            # compute indices for shifting
            if k >= 0:
                h1 = h[:-k] if k > 0 else h[:]
                h2 = h[k:] if k > 0 else h[:]
            else:
                h1 = h[-k:] if k < 0 else h[:]
                h2 = h[:k] if k < 0 else h[:]
                
            # We'll rewrite: sum(h[i] * (1 - h[i+k])) <= t_mid
            # s[i] = h[i] * (1 - h[i+k]) is linearized via big-M constraints
            # Use binary variables to model the relationship under certain conditions
                
            # Simpler linearization:
            # constraint: sum(h[i] - h[i]*h[i+k]) <= t_mid * dx * n_points
            # which becomes: sum(h[i]) - sum(h[i]*h[i+k]) <= t_mid * dx * n_points
            
            # Reformulate using auxiliary products w[i][k] = h[i]*h[i+k]
            # This is still not convex, so we relax with bigM polytope.
            # Actually, for a shuffle, use direct linear inequalities
            # We directly add an individual constraint:
            # sum(h[i] - h[i]*h[i+k]) <= t_mid
            # but that's not valid
            
            # Approximate with bilinear:
            # Use a quadratic or bilinear relaxation (LP relaxation of a product uses big-M)
            # Can we actually write an LP? Not exactly.
            # Let's go frequency domain-simplify or brute-force constraint solution
            # simple iteration over all shift indices k and create constraints
            # For all positive k:
            # We want sum_{i} h[i] * (1 - h[i+k]) <= t_mid
            if k >= 0:
                s = 0
                for i in range(n_points - k):
                    s += h[i] * (1 - h[i + k])
                constraints.append(s <= t_mid)
            else:
                # k is negative
                s = 0
                for i in range(-k, n_points):
                    s += h[i] * (1 - h[i + k])
                constraints.append(s <= t_mid)
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    c5_sol = evaluate_h(h_sol)
                    if c5_sol <= t_mid + 1e-6:
                        t_high = t_mid
                        best_h = h_sol.copy()
                        best_c5 = c5_sol
                    else:
                        t_low = t_mid
                else:
                    t_low = t_mid
            else:
                t_low = t_mid
        except Exception:
            # Optimization failed, don't change bounds
            t_low = t_mid
    
    # Final evaluation with best solution
    if (time.time() - start_time) < budget_s - 1:
        final_c5 = evaluate_h(best_h)
        best_h_binary = np.round(best_h).astype(float)
        if abs(np.sum(best_h_binary) - n_points / 2) > 1e-6:
            # Repair sum constraint by adjusting
            current_sum = np.sum(best_h_binary)
            to_change = int(abs(current_sum - n_points / 2))
            if to_change > 0:
                indices = np.argsort(np.abs(best_h_binary - 0.5))[:to_change]
                for idx in indices:
                    if current_sum > n_points / 2:
                        best_h_binary[idx] = 0
                    else:
                        best_h_binary[idx] = 1
                    current_sum = np.sum(best_h_binary)
                    if abs(current_sum - n_points / 2) <= 1e-6:
                        break
        final_c5 = evaluate_h(best_h_binary)
        return (best_h_binary.tolist(), final_c5, n_points)
    else:
        # Return last best result if no time
        return (best_h.tolist(), best_c5, n_points)
