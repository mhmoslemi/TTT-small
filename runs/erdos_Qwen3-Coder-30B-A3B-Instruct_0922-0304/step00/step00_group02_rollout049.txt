import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback initialization
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Function to evaluate C5
    def evaluate_c5(h_vals):
        # Correlate h with (1-h)
        full_correlation = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # Compute maximum overlap
        max_overlap = np.max(full_correlation) * dx
        return max_overlap
    
    # Project to feasible set
    def project_h(h_vals):
        # Clip to [0,1]
        h_clipped = np.clip(h_vals, 0, 1)
        # Scale to sum to target_sum
        current_sum = np.sum(h_clipped)
        if current_sum > 0:
            h_projected = h_clipped * (target_sum / current_sum)
        else:
            h_projected = np.ones_like(h_clipped) * (target_sum / n_points)
        return h_projected
    
    # Gradient computation via finite differences
    def compute_gradient(h_vals, active_k):
        # Compute overlaps with mode full
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full') * dx
        # Active k is the index that gives maximum overlap
        # For gradient: d/dh[i] of sum h[j] * (1 - h[(j+k) % n]) 
        # This corresponds to (1 - h[(i+k) % n]) - for active k
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            j = (i + active_k) % n_points
            grad[i] = (1.0 - h_vals[j])
        
        return grad
    
    # Gradient descent with projection
    max_iter = 10000
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    improvement_count = 0
    max_no_improvement = 100
    perturbation_freq = 500
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    start_time = time.time()
    
    current_h = h.copy()
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate current function value
        current_c5 = evaluate_c5(current_h)
        
        # Compute all overlaps for current h to find max
        full_corr = signal.correlate(current_h, 1 - current_h, mode='full') * dx
        active_k = np.argmax(full_corr)
        max_overlap = full_corr[active_k]
        
        # Compute gradient
        gradient = compute_gradient(current_h, active_k)
        
        # Update step
        h_temp = current_h - alpha * gradient
        
        # Project to feasible set
        h_projected = project_h(h_temp)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_projected)
        
        # Accept or reject update
        if new_c5 < current_c5:
            current_h = h_projected
            improvement_count = 0
            if new_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = new_c5
        else:
            improvement_count += 1
            alpha *= gamma
            if alpha < alpha_min:
                alpha = alpha_min
                
        # Perturbation to escape local minima
        if iter_num % perturbation_freq == 0 and iter_num > 0:
            # Add small random perturbations (5% of values)
            n_perturb = max(1, n_points // 20)
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            perturb_amount = np.random.normal(0, alpha * 0.1, size=n_perturb)
            current_h[indices] += perturb_amount
            current_h = project_h(current_h)
        
        if improvement_count > max_no_improvement:
            break
            
        if best_c5 < 0.3807:
            break
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
