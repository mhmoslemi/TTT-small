import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h *= (n_points / 2) / np.sum(h)
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Helper to compute C5
    def compute_c5(h_vec):
        # Shifted correlation
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Extract the relevant part (max over all shifts)
        c5_val = np.max(corr) * dx
        return c5_val
    
    # Initial computation
    current_c5 = compute_c5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05  # 5% of n_points
    delta_max = 0.1
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Adaptive delta
        delta = delta_max * (0.9 ** (iteration // 100))
        
        # Perturb a subset of points
        n_perturb = max(1, int(n_points * perturb_frac))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        h_new = h.copy()
        for i in indices:
            # Random perturbation
            perturb = np.random.uniform(-delta, delta)
            h_new[i] += perturb
            
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
