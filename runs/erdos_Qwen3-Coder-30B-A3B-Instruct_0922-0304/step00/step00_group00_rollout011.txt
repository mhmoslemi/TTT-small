import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts 0 to n-1
    overlaps = correlate(h, 1 - h, mode='full')
    # Return maximum overlap
    return np.max(overlaps) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    
    # Normalize to sum to n/2
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Cut-plane method
    max_iter = 300
    no_improvement_count = 0
    best_c5_history = []
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Find the shift with maximum overlap
        overlaps = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Add constraint for this shift
        # We're working with full correlation
        # So for shift k_max, the constraint is:
        # sum h[i] * (1 - h[(i + k_max) % n]) <= C5
        
        # Build the quadratic constraint
        A = np.zeros((n, n))
        for i in range(n):
            j = (i + k_max) % n
            A[i, j] = h[i]
        
        # Formulate the optimization problem
        h_var = cp.Variable(n)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n / 2,
            cp.sum(cp.multiply(h_var, 1 - h_var[(np.arange(n) + k_max) % n])) <= C5_var
        ]
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Solve
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n / 2) / np.sum(h_new)  # Renormalize
                    
                    # Evaluate new solution
                    c5_val = evaluate_erdos_solution(h_new)
                    best_c5_history.append(c5_val)
                    
                    if c5_val < best_c5:
                        best_c5 = c5_val
                        best_h = h_new.copy()
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                    
                    # Check stopping criteria
                    if best_c5 < 0.3807:
                        break
                        
                    if no_improvement_count > 50:
                        break
                        
                    h = h_new
                    
        except Exception as e:
            pass
    
    return (best_h.tolist(), best_c5, n)
