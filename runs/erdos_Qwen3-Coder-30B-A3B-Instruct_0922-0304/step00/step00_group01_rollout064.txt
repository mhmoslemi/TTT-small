import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    # Compute overlap integral for the given h_values
    h = np.array(h_values)
    dx = 2.0 / n_points
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (k from 0 to n_points-1)
    overlaps = conv[n_points-1:] * dx
    return np.max(overlaps)

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient of overlap with respect to h for shift k_max"""
    grad = np.zeros(n_points)
    for i in range(n_points):
        # Gradient contribution from h[i]
        if i + k_max < n_points:
            grad[i] += (1 - h[i + k_max])  # Influence of h[i+k_max]
        if i - k_max >= 0:
            grad[i] -= h[i - k_max] * (1 - h[i])  # Influence of h[i-k_max]
    return grad

def line_search(h, gradient, k_max, n_points, dx):
    """Backtracking line search for optimal step size"""
    alpha = 0.01
    rho = 0.5
    c1 = 1e-4
    current_overlap = np.max(correlate(h, 1 - h, mode='full')[n_points-1:]) * dx
    
    for _ in range(30):
        h_new = h - alpha * gradient
        h_new = project_h(h_new, n_points)
        new_overlap = np.max(correlate(h_new, 1 - h_new, mode='full')[n_points-1:]) * dx
        if new_overlap <= current_overlap - c1 * alpha * np.dot(gradient, gradient):
            return h_new, alpha
        alpha *= rho
    return h, 0.0

        conv = correlate(h, 1 - h, mode='full')
        overlaps = conv[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max]
        
        if c5_current < best_c5:

        gradient = compute_gradient(h, k_max, n_points)
