import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute correlation for all lags and return max
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    start_time = time.time()

    # Try SDP relaxation first
    try:
        # Create variables for the relaxation
        y = cp.Variable((n_points, n_points), symmetric=True)
        h = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.diag(y) == h,  # Link diagonal to h
            cp.sum(h) == n_points / 2,  # Sum constraint
            y >> 0,  # PSD constraint
            C5 >= 0  # C5 non-negative
        ]
        
        # Correlation constraints
        for k in range(2*n_points - 1):
            i_range = range(max(0, k - (n_points - 1)), min(n_points, k + 1))
            j_range = range(max(0, (n_points - 1) - k), min(n_points, (n_points - 1) + 1))
            
            # Reconstruct the correlation term as sum(h[i]) - sum(y[i, i+k])
            corr_term = cp.sum(h[i_range]) - cp.sum(y[i_range, j_range])
            constraints.append(corr_term <= C5)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Try to solve with MOSEK or ECOS
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s * 0.8)
        except:
            prob.solve(solver=cp.ECOS, verbose=False)
            
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_val = h.value
            if h_val is not None:
                # Ensure boundary constraints
                h_val = np.clip(h_val, 0, 1)
                h_val = h_val / np.sum(h_val) * (n_points / 2)
                c5_bound = evaluate_c5(h_val, dx)
                return (h_val, c5_bound, n_points)
    except Exception:
        pass

    # Fallback: gradient-based optimization
    # Start with initial values
    if 'initial_h_values' in globals():
        h0 = np.copy(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Ensure it sums to n_points/2
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    def objective(h):
        return evaluate_c5(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    constraint1 = {'type': 'eq', 'fun': constraint_sum}
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize
    res = minimize(objective, h0, method='SLSQP', bounds=bounds, 
                   constraints=constraint1, options={'maxiter': 1000},
                   callback=lambda x: None if time.time() - start_time > budget_s - 5 
                   else None)
    
    # Return best solution
    h_final = np.clip(res.x, 0, 1)
    c5_final = evaluate_c5(h_final, dx)
    
    return (h_final, c5_final, n_points)
