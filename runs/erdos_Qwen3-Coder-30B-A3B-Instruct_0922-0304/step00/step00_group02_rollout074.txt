import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 45
        h = np.zeros(n)
        h[:n//2] = 1.0
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project to correct sum
    h = h * (target_sum / np.sum(h))
    
    # Gradient descent parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_iter = 10000
    patience = 100
    no_improve_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_values):
        # Compute overlaps
        conv = signal.correlate(h_values, 1 - h_values, mode='full') * dx
        return np.max(conv)
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_clipped = np.clip(h_vec, 0, 1)
        # Scale to target sum
        scale = target_sum / np.sum(h_clipped)
        return h_clipped * scale
    
    # Iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current overlaps and C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > patience or c5 < 0.3807:
            break
            
        # Compute gradient with respect to active shift
        conv = signal.correlate(h, 1 - h, mode='full') * dx
        active_k = np.argmax(conv)
        
        # Compute analytical gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + active_k) % n_points
            grad[i] = (1 - h[j]) - h[i]  # Approximate gradient
        # More accurate finite difference approach:
        eps = 1e-6
        grad_approx = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad_approx[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update
        h_temp = h - alpha * grad_approx
        h = project_h(h_temp)
        
        # Adaptive step size
        if np.allclose(h_temp, h, atol=1e-6):
            alpha *= gamma
            if alpha < alpha_min:
                break
        else:
            if alpha < 0.1:
                alpha = min(alpha * 1.05, 0.1)
    
    # Final projection and validation
    final_h = project_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), int(n_points))
