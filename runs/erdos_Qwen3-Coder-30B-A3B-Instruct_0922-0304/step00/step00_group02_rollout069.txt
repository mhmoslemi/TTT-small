import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlaps using correlation
    overlaps = signal.correlate(h, 1 - h, mode='full')
    # Extract the valid range (excluding padding)
    center = len(overlaps) // 2
    valid_overlaps = overlaps[center - n + 1:center + n - 1]
    c5 = np.max(valid_overlaps) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies constraints"""
    n = len(h)
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n/2) < 1e-10)

def repair_feasibility(h):
    """Repair infeasible h by clipping and rescaling"""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n/2) > 1e-10:
        # Adjust by scaling
        if current_sum > 0:
            h = h * (n/2) / current_sum
        else:
            h = np.full_like(h, 1.0/n)
    return np.clip(h, 0, 1)

def mutate_individual(h, mutation_rate=0.01):
    """Mutate an individual with given rate"""
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    h_new = h.copy()
    h_new[mask] += np.random.normal(0, 0.05, np.sum(mask))
    h_new = repair_feasibility(h_new)
    return h_new

def crossover(parent1, parent2):
    """Perform crossover between two parents"""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    # Repair for feasibility
    child1 = repair_feasibility(child1)
    child2 = repair_feasibility(child2)
    return child1, child2

        h = np.random.uniform(0, 1, n)
        h = repair_feasibility(h)

        if time.time() - start_time > budget_s - 1:
            break

        fitnesses = []

            c5 = evaluate_h(h)
            fitnesses.append(1.0 / (c5 + 1e-12))  # Avoid divide by zero
        
        # Sort population by fitness
        idxs = np.argsort(fitnesses)[::-1]
        best_idx = idxs[0]
        
        # Update best if needed
        if fitnesses[best_idx] > 1.0 / best_c5 if best_c5 < float('inf') else True:
            best_h = population[best_idx].copy()
            best_c5 = evaluate_h(best_h)
        
        # Selection: top 50%
        selected = [population[i] for i in idxs[:elite_size]]
        
        # Repopulate
        new_population = selected[:]
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            if np.random.random() < 0.5:
                child1 = mutate_individual(child1, mutation_rate)
            if np.random.random() < 0.5:
                child2 = mutate_individual(child2, mutation_rate)
                
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:pop_size]
    
    # Final evaluation
    if best_h is None:
        # Re-evaluate from final population if needed
        c5s = [evaluate_h(h) for h in population]
        best_idx = np.argmin(c5s)
        best_h = population[best_idx]
        best_c5 = c5s[best_idx]
    
    return best_h, best_c5
