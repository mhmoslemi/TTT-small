import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    overlaps = []
    for k in range(-n+1, n):
        shift = np.roll(h_values, k)
        overlap = np.sum(h_values * (1 - shift)) * dx
        overlaps.append(overlap)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():

        n = len(h)
    else:

    dx = 2.0 / n
    # Renormalize to sum to n/2
    h *= (n/2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    alpha = 0.1
    max_iter = 5000
    iter_since_improvement = 0
    improvement_threshold = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Compute all overlaps to find max
        overlaps = []
        for k in range(-n+1, n):
            shift = np.roll(h, k)
            overlap = np.sum(h * (1 - shift)) * dx
            overlaps.append(overlap)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Compute gradient for the worst case k
        grad = np.zeros(n)
        k = max_k - (n-1)  # convert index to actual shift
        for i in range(n):
            j = (i + k) % n
            grad[i] += (1 - h[j]) * dx
            if j >= 0:
                grad[j] -= h[i] * dx

        # Gradient step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new *= (n/2) / np.sum(h_new)
        h = h_new

        # Convex refinement (QP)
        if iteration % 5 == 0:
            k_max = max_k - (n-1)
            
            # QP for k_max
            w = cp.Variable(n)
            objective = cp.Minimize(cp.sum(w))
            
            constraints = [w[i] >= h[i] * (1 - h[(i + k_max) % n]) * dx for i in range(n)]
            constraints.append(cp.sum(w) <= max_overlap)
            constraints.append(cp.sum(w) >= 0)
            constraints.append(w[i] >= 0 for i in range(n))
            
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    w_vals = np.array(w.value)
                    h = np.clip(w_vals, 0, 1)
                    # Renormalize
                    h *= (n/2) / np.sum(h)
            except:
                pass

        # Check improvement
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Adaptive learning rate
        if iter_since_improvement > 10:
            alpha *= 0.9
            iter_since_improvement = 0

    return (best_h.tolist(), float(best_c5), n)
