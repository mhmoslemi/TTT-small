import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C_5 for given step function h"""
    # Compute all overlaps using correlation
    overlaps = signal.correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlaps)

    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000
    
    # Track improvements for learning rate adaptation
    last_improvement = 0
    current_c5 = best_c5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        # Check time budget
        if time.time() - start_time > budget_s:

        # Compute all overlaps
        overlaps = signal.correlate(h, 1 - h, mode="full") * dx
        
        # Find the maximum overlap value and matching lags
        max_overlap = np.max(overlaps)
        
        # Get indices where maximum occurs
        max_lags = np.where(np.abs(overlaps - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for lag_idx in max_lags:
            # Convert lag index to actual lag offset
            lag = lag_idx - (n_points - 1)
            # Compute contribution to gradient with proper indexing
            if lag >= 0:
                # h[i] contributes to overlap[i+lag] where i=0,1,...,n_points-1-lag
                valid_indices = np.arange(n_points - lag)
                gradient[valid_indices] += (1 - h[valid_indices + lag]) * dx
            else:
                # h[i+(-lag)] contributes to overlap[i] where i=0,1,...,n_points+lag-1
                valid_indices = np.arange(-lag, n_points)
                gradient[valid_indices] += (1 - h[valid_indices + lag]) * dx
                
        # Average over number of maximum overlaps
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        
        # Project onto feasible set [0,1] and enforce sum constraint
        h = np.clip(h, 0, 1)

        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if new_c5 < current_c5:
            best_h = h.copy()
            best_c5 = new_c5
            current_c5 = new_c5
            last_improvement = iteration
        else:
            # Adaptive learning rate
            if iteration - last_improvement > gamma_patience:
                learning_rate *= gamma_decrease_factor
                last_improvement = iteration
