import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize h to satisfy integral constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    best_h = h.copy()
    best_C5 = float('inf')
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        # Compute overlap integrals for all shifts
        # We can compute them efficiently with correlate
        corr_full = correlate(h, 1 - h, mode='full')
        # The valid overlaps are the middle part (conveniently, length = 2*n_points - 1)
        # But we want overlap for shifts k in [-n_points+1, n_points-1]
        # so a shift of k corresponds to corr_full[n_points-1+k]
        overlaps = np.array([corr_full[n_points-1+k] * dx for k in range(-n_points+1, n_points)])
        max_overlap = np.max(overlaps)
        max_idx = np.argmax(overlaps)
        shift_k = max_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h vector for max overlap term
        # For shift k, gradient w.r.t. h[i] is (1 - h[i+k])
        # (where we know that square in correlate gives contribution)
        # But we only compute full cross correlation and then extract specific terms
        grad = np.zeros_like(h)
        for i in range(n_points):
            if 0 <= i + shift_k < n_points:
                grad[i] = 1 - h[i + shift_k]
        
        # Update with momentum
        velocity = momentum * velocity + learning_rate * grad
        h_new = h - velocity
        
        # Project onto feasible region
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to satisfy integral constraint
        h_sum_new = np.sum(h_new)
        if h_sum_new > 0:
            h_new = h_new * (n_points / 2) / h_sum_new
        
        h = h_new
        
        # Update best solution if we improved
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
        
        # Decrease learning rate every 100 iterations
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
        
        # Early stopping
        if best_C5 < target_bound:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_C5, n_points)
