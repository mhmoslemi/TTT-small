def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral for the given step function h."""
    n = len(h)
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # Extract the overlap values for all shifts k from 0 to n-1
    # The shift k=0 maps to index n-1, k=1 to n-2, ..., k=n-1 to 0
    # We want the overlaps for k = 0, 1, ..., n-1
    overlaps = corr[n-1:n-1+n]  # This is equivalent to corr[n-1:]
    c5 = np.max(overlaps) * dx
    return c5
