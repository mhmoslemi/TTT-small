import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdos constant C5 for the given step function h."""
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the initial construction if available
    try:
        h = initial_h_values.copy()
    except:
        # Fallback: create a simple symmetric step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = int(budget_s * 1000)
    
    # Project to feasible region
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    last_improvement = 0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlap at all lags
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all lags where maximum overlap occurs
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # h[i] * (1 - h[i + lag]) for valid indices
                valid_indices = np.arange(0, n_points - lag)
                gradient[valid_indices] += (1 - h[valid_indices + lag]) * dx
        
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        h = project_h(h)
        
        # Evaluate
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adapt learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
            
    return (best_h, best_c5, n_points)
