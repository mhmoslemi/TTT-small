import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial values or create a symmetric starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create symmetric starting point
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize initial h to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h):
        """Computes the maximum overlap integral"""
        h = np.array(h)
        # Compute full correlation integrals
        corr = np.correlate(h, 1 - h, mode="full")
        # Select the valid part
        c5_values = corr[len(h)-1:] * dx
        return np.max(c5_values)
    
    def project_feasible(h):
        """Project h to satisfy constraints"""
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h) * dx
        if abs(current_sum - 1.0) > 1e-10:
            h = h * (1.0 / current_sum)
        return h
    
    def optimize_single_shift(h, k_max):
        """Optimize for a fixed shift k_max"""
        # Create a convex optimization to minimize C(k_max)
        # Variables: h[i] for i=0..n-1 (subject to bounds)
        # Objective: minimize sum(h[i] * (1 - h[(i-k_max)%n]))
        # Constraints: 0 <= h[i] <= 1, sum(h) = n/2
                
        def obj(h_vec):
            # Use convolution for speed
            h_vec = np.array(h_vec)
            shift_h = np.roll(h_vec, k_max)
            return np.sum(h_vec * (1 - shift_h)) * dx
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': constraint_sum}]
        
        # Try to minimize the objective
        res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=constraints, 
                       options={'maxiter': 50, 'ftol': 1e-8})
        if res.success:
            return res.x
        return h
    
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    max_iter = 100
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        current_c5 = compute_C5(best_h)
        if current_c5 <= 0.3807:
            break
            
        # Compute overlap values
        h_current = best_h
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        c5_values = corr[len(h_current)-1:] * dx
        k_max = np.argmax(c5_values)
        
        # Optimize for that shift
        optimized_h = optimize_single_shift(best_h, k_max)
        optimized_h = project_feasible(optimized_h)
        new_c5 = compute_C5(optimized_h)
        
        if new_c5 < best_c5:
            best_h = optimized_h
            best_c5 = new_c5
    
    # Final evaluation
    best_c5 = compute_C5(best_h)
    
    return (list(best_h), best_c5, n_points)
