import cvxpy as cp
from time import time

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k = -n+1 to n-1
    overlaps = np.correlate(h, 1 - h, mode='full')
    # Only consider valid overlaps (k=0 to n-1)
    max_overlap = max(overlaps[n-1:]) * dx
    return max_overlap
