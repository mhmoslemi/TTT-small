import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * (n_points / (2 * np.sum(h)))  # Normalize to sum = n_points/2
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters
    lr = 0.01
    eps = 1e-4
    max_iter = int(budget_s * 10)  # Rough estimate
    patience = 100
    patience_counter = 0
    last_best = best_c5
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals using convolution
        # For each k, we need to compute integral of h(x) * (1 - h(x+k))
        # This is equivalent to conv(h, 1-h, mode="full") with appropriate indices
        
        # Implementation: compute full correlation with 1-h as kernel
        h_rev = np.flip(1 - h)
        correlations = signal.convolve(h, h_rev, mode='full') * dx
        
        # Extract relevant parts 
        # For k = 0, ..., n_points-1, we want to look at correlation[k] = integral of h(x)(1-h(x+k))
        # But for x in [0,2], the shift means that only valid ranges matter

        # Actually model this correctly: when convolving h with 1-h,
        # element j of result corresponds to integral over [-1,1] of h(t)(1-h(t + j-(n-1)))
        # Let's use equivalent computation
        max_corr = np.max(correlations)
        
        # Find where maximum is achieved
        indices = np.where(np.abs(correlations - max_corr) < 1e-10)[0]
        
        # Gradient computation based on which indices contribute to max
        grad = np.zeros(n_points)
        for idx in indices:
            # idx is the shift from the convolution
            # We are effectively looking at h[i] * (1 - h[i+shift])
            # where shift = idx - (n_points-1)
            shift = idx - (n_points - 1)
            for j in range(n_points):
                if 0 <= j + shift < n_points:
                    grad[j] += (1 - h[j + shift])
        
        # Update step
        h_new = h - lr * grad
        
        # Project to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / (2 * np.sum(h_new)))  # rescale
        
        h = h_new
        
        # Check improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5 - eps:
            best_h = h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                lr *= 0.9
                patience_counter = 0

    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_values, dx):
    # Compute C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    h = np.array(h_values)
    n = len(h)
    h_rev = np.flip(1 - h)
    correlations = signal.convolve(h, h_rev, mode='full') * dx
    # The maximum correlation gives the desired C5
    return np.max(correlations)
