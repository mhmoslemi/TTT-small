import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure h sums to 1 (after normalization)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    start_time = time.time()
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Solvers - try to use ECOS for speed
    solver = cp.ECOS
    
    # Optimization parameters
    max_iter = 1000
    patience = 50
    improvement_count = 0
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > 0.95 * budget_s:
            break
            
        # Compute all overlaps
        # Note: We compute correlation with reversed sequence (1-h)
        # Using numpy correlate with full mode gives us all shifts
        h_rev = 1 - h
        overlaps = np.correlate(h, h_rev, mode='full')
        k_shifts = np.arange(-len(h)+1, len(h))
        max_idx = np.argmax(overlaps)
        max_k = k_shifts[max_idx]
        max_overlap = overlaps[max_idx]
        
        # Update best solution
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            improvement_count = 0
        else:
            improvement_count += 1
            
        if best_C5 < 0.3807:
            break
        if improvement_count > patience:
            break
            
        # Formulate the convex optimization problem
        # We add constraint: h[k] * (1 - h[(k + max_k) % n]) <= C5 for all k
        # But actually we just add the most violating constraint to tighten it
        if max_k < 0:
            shifted_indices = np.arange(len(h) + max_k)
            left_indices = shifted_indices
            right_indices = np.arange(-max_k, len(h))
        else:
            shifted_indices = np.arange(max_k, len(h))
            left_indices = np.arange(len(h) - max_k)
            right_indices = shifted_indices
            
        # Build quadratic constraint
        # sum_{k in shifted_indices} h[k] * (1 - h[(k + max_k) % n]) <= C5
        # This is equivalent to:
        # sum_{k} h[k] * (1 - h[(k + max_k) % n]) <= C5
        # Let's do it directly with quadratic cost function in cvxpy
        
        # Define cvxpy variables
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Initialize h_var with current h
        h_var.value = h
        
        # Objective: minimize C5
        objective = cp.Minimize(C5_var)
        
        # Constraints
        constraints = []
        
        # Sum constraint
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Bound constraints
        constraints.append(h_var >= 0)
        constraints.append(h_var <= 1)
        
        # The problematic constraint: sum_k h[k] * (1 - h[(k + max_k) % n]) <= C5
        # This is a quadratic constraint
        # Compute it manually for precision
        if max_k > 0:
            constraint_expr = 0.0
            for i in range(len(h)):
                j = (i + max_k) % len(h)
                constraint_expr += h_var[i] * (1 - h_var[j])
        else:
            constraint_expr = 0.0
            for i in range(len(h)):
                j = (i + max_k) % len(h)
                constraint_expr += h_var[i] * (1 - h_var[j])
                
        # Inequality constraint
        constraints.append(constraint_expr <= C5_var)
        
        # Try solving
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=solver, verbose=False, max_iter=500)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                # Normalize to sum to n/2
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                # Clamp
                h_new = np.clip(h_new, 0, 1)
                h = h_new
        except:
            # If optimization fails, continue with current h
            pass
            
    # Final evaluation and rounding to 6 decimal places
    dx = 2.0 / n_points
    overlaps = np.correlate(best_h, 1 - best_h, mode='full')
    final_C5 = np.max(overlaps) * dx
    final_C5 = round(final_C5, 6)
    
    return (best_h.tolist(), final_C5, n_points)
