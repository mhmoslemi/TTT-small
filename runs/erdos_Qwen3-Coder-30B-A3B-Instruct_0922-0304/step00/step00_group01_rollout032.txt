def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute all overlaps using convolution
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the overlapping parts within the range [0, 2]
    # The full correlation is of length 2*n-1. We map the shifts to the interval [0, 2]
    # The maximum overlap corresponds to indices where shift k = 0 to 2
    # In general, the effective indices for k from 0 to n-1 to 2k from 0 to 2(n-1)
    # So the valid range for k that maps to [0,2] is actually at most n-1
    max_c5 = max(corr[n-1:2*n-1]) * dx
    return max_c5
