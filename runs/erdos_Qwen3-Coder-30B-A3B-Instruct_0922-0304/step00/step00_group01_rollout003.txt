import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode="full")
    # Get the part corresponding to positive shifts only
    n = len(h)
    overlap_vals = overlap[n-1:]
    # Scale by dx and return max
    return np.max(overlap_vals) * dx

def project_h(h, target_sum):
    """Project h to satisfy sum(h) = target_sum while maintaining bounds."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Rescale to hit target sum
    scale_factor = target_sum / current_sum
    h = h * scale_factor
    # Re-clip (may slightly exceed bounds but not much)
    h = np.clip(h, 0, 1)
    return h

    peak_indices = np.argsort(overlap_vals)[::-1][:num_peaks]
    
    # Create a new h by reducing peaks
    h_new = h.copy()

        for i in range(n - k):

    # Ensure feasibility
    h_new = project_h(h_new, target_sum)
    return h_new
