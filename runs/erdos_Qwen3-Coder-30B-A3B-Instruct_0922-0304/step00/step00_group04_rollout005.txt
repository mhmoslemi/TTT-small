```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx=2.0/42):
    """Compute C5 = max_k integral of h(x)(1-h(x+k))dx"""
    corr = np.correlate(h, 1-h, mode="full")
    return np.max(corr) * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Try convex relaxation first
    try:
        # Try to solve using SDP relaxation approach
        h = Variable(n_points)
        y = Variable((n_points, n_points), symmetric=True)
        C5 = Variable()
        
        # Constraints for PSD relaxation
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2,
            y == h @ h.T,
        ]
        
        # Add correlation constraints
        for k in range(n_points):
            # Compute beginning of full correlation (k=0 to n_points-1) 
            corr_k = sum(h[i] - y[i, (i+k) % n_points] for i in range(n_points))
            constraints.append(corr_k <= C5 / dx)
            
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        # Try different solvers
        solver_choice = [ECOS, MOSEK, CVXOPT]
        for solver in solver_choice:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    h_vals = h.value
                    if h_vals is not None:
                        # Ensure feasibility
                        h_vals = np.clip(h_vals, 0, 1)
                        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
                        c5_bound = evaluate_erdos_solution(h_vals, dx)
                        return (h_vals, c5_bound, n_points)
            except:
                continue
                
    except:
        pass
    
    # Fallback: gradient-based optimization
    h0 = initial_h_values.copy() if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Use scipy minimize with trust-constr for constraints
    def objective(h):
        return evaluate_erdos_solution(h, dx)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1-h])  # h[i] >= 0, h[i] <= 1
    
    # Create bounds factor
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    # Optimization loop with cutoff time
    max_iter = 1000
    best_h = h0.copy()
    best_c5 = objective(h0)
    
    for i in range(max_iter):
        current_time = time.time()
        if current_time - start_time > 0.9 * budget_s:
            break
            
        # Randomize starting point for variety
        h_start = best_h + np.random.normal(0, 0.05, n_points)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
        
        # Optimization with constraints
        result = minimize(
            objective,
            h_start,
            method='trust-constr',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'disp': False}
        )
        
        if result.success and not np.isnan(result.fun):
            c5_val = result.fun
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = result.x
        
        # Add some random search too
        if np.random.random() < 0.1:
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            c5_random = objective(h_random)
            if c5_random < best_c5:
                best_c5 = c5_random
                best_h = h_random
    
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)
```
