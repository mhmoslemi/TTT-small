import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    n = len(h)
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n = len(h)
    else:
        n = 45
        h = np.ones(n) * 0.5
    dx = 2.0 / n
    
    # Normalize h to sum to n/2
    h = h * (n / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Parameters for optimization
    max_iter = 5000
    alpha = 0.1
    patience = 50
    patience_count = 0
    last_improvement = best_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Identify worst shift
        overlaps = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps) - (n - 1)
        
        # Compute gradient for k_max
        grad = np.zeros(n)
        for i in range(n):
            j = (i + k_max) % n
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Update h using gradient descent
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new = h_new * (n / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = best_c5
            patience_count = 0
        else:
            patience_count += 1
            if patience_count > patience:
                alpha *= 0.5
                patience_count = 0
                
        h = h_new
    
    return (list(best_h), best_c5, n)
