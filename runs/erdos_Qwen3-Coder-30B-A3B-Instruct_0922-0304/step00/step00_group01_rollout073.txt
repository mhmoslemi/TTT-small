import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for different shifts
    overlaps = np.correlate(h, 1 - h, mode="full")
    # Take only the relevant part (maximum overlap)
    max_overlap = np.max(overlaps[n-1:])
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Step size and parameters
    n = 47
    max_iters = 10000
    alpha = 0.01
    min_alpha = 1e-5
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n, 0.5)
    
    # Normalize to ensure sum(h) = n/2
    h = h * (n / 2) / np.sum(h)
    
    # Lagrange multiplier
    lambda_ = 0.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.correlate(h, 1 - h, mode="full")
        # The maximum overlap is at the center (index n-1 to 2*n-2), but we take max of shifted overlaps
        shift_overlaps = overlaps[n-1:]
        k_max = np.argmax(shift_overlaps)
        max_overlap = shift_overlaps[k_max]
                
        # Check if we've improved
        current_c5 = max_overlap * (2.0 / n)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 < 0.3807:
            break
            
        if no_improvement_count >= 100:
            alpha *= 0.5
            if alpha < min_alpha:
                break
                
        # Compute subgradient for the worst-case overlap
        g = np.zeros_like(h)
        for i in range(n):
            j = (i - k_max) % n
            g[i] = (1 - h[j]) - h[i] + lambda_
        
        # Update h
        h = h - alpha * g
        # Project h to [0, 1]
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h = h * (n / 2) / np.sum(h)
        
        # Update Lagrange multiplier
        lambda_ = lambda_ + alpha * (np.sum(h) - n/2)
    
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n)
