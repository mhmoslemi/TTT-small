import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.random.rand(42)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial sum matches constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        # Find max correlation and corresponding lag
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        k = max_corr_idx - (len(h) - 1)  # convert index to lag
        
        # Compute gradient for this lag
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Gradient contribution to h[i] from correlation at lag k
            if 0 <= i + k < len(h):
                grad[i] += (1 - h[i + k]) * dx
        
        # Apply gradient descent with small step size
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to feasible region
        # 1. Ensure bounds
        h_new = np.clip(h_new, 0, 1)
        
        # 2. Ensure sum constraint
        current_sum = np.sum(h_new)
        h_new = h_new * target_sum / current_sum
        
        # Check improvement
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
        else:
            # Slight perturbation to escape local minima
            h = h + 0.001 * (np.random.rand(n_points) - 0.5)
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
