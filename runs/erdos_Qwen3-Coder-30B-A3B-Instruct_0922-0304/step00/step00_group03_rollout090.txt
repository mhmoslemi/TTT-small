import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Scale to make sum = n_points/2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Initialize optimizer state
    alpha = 0.01
    beta = 0.9  # momentum
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute full correlation for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Adjust indexing to match shift k=-n+1 to k=n-1
        shift_indices = np.arange(-n_points + 1, n_points)  # k values from -n+1 to n-1
        corr_values = corr  # All values from full correlation
        
        # Find max overlap and corresponding shift
        max_idx = np.argmax(corr_values)
        max_shift = shift_indices[max_idx]
        max_overlap = corr_values[max_idx] * dx
        
        # Update best solution
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            
        # If we've reached target, break early
        if max_overlap <= 0.3807:
            break

        # Gradient computation
        grad = np.zeros_like(h)
        if max_shift >= 0:
            # Positive shift: h[i] contributes to overlap with h[i + shift]
            for i in range(n_points - max_shift):
                grad[i] += (1 - h[i + max_shift])
        else:
            # Negative shift: h[i - shift] contributes to overlap with h[i]
            for i in range(-max_shift, n_points):
                grad[i] += (1 - h[i + max_shift])
        
        # Add momentum
        velocity = beta * velocity + (1 - beta) * grad
        h_new = h - alpha * velocity
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet sum constraint
        h_sum_new = np.sum(h_new)
        h = h_new * (n_points / 2) / h_sum_new
        
        # Occasionally decrease learning rate
        if iteration % 100 == 0:
            alpha *= 0.9
            
    return (list(best_h), best_C5, n_points)
