import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given h."""
    dx = 2.0 / len(h)
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode="full")
    # Take max and multiply by dx
    c5 = np.max(overlaps) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies all constraints."""
    return (np.all(h >= 0) and np.all(h <= 1) and
            abs(np.sum(h) - len(h)/2) < 1e-10)

def repair_h(h):
    """Repair h to be feasible by clipping and rescaling."""
    h = np.clip(h, 0, 1)
    # Rescale to make the sum correct
    current_sum = np.sum(h)
    target_sum = len(h) / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h):
    """Initialize a population with one copy of initial_h and random h."""
    population = []
    # Add the initial_h as the first individual
    h0 = repair_h(initial_h.copy())
    if not is_feasible(h0):
        raise ValueError("Initial h is infeasible")
    population.append(h0)
    
    # Add random individuals
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = repair_h(h)
        population.append(h)
        
    return population

def select_parents(population, fitnesses):
    """Select parents using tournament selection."""
    selected = []
    n_parents = len(population) // 2
    for _ in range(n_parents):
        # Tournament size = 3
        indices = np.random.choice(len(population), size=3, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in indices]
        winner_idx = indices[np.argmax(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n_points = len(parent1)
    # One-point crossover
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Repair to make feasible
    child1 = repair_h(child1)
    child2 = repair_h(child2)
    
    return child1, child2

def mutate(h, mutation_rate=0.01):
    """Apply mutation to h."""
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, size=num_mutations, replace=False)
    
    # Add Gaussian noise
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    
    # Repair to make feasible
    h = repair_h(h)
    return h

def genetic_algorithm(n_points, pop_size, initial_h, max_time):
    """Run genetic algorithm to optimize h."""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    # Fitness function: lower c5 is better
    fitnesses = [1.0 / (evaluate_h(h) + 1e-15) for h in population]
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < max_time:
        # Evaluate current fitness
        fitnesses = [1.0 / (evaluate_h(h) + 1e-15) for h in population]
        
        # Track best individual
        current_best_idx = np.argmax(fitnesses)
        current_best_c5 = evaluate_h(population[current_best_idx])
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            break  # No improvement in 100 generations
            
        # Select parents
        parents = select_parents(population, fitnesses)
        
        # Create offspring
        offspring = []
        num_pairs = len(parents) // 2
        for i in range(num_pairs):
            p1 = parents[2*i]
            p2 = parents[2*i + 1]
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            offspring.extend([c1, c2])
            
        # Replace worst individuals with offspring
        # Sort by fitness and replace worst
        sort_indices = np.argsort(fitnesses)
        for i in range(len(offspring)):
            worst_idx = sort_indices[i]
            population[worst_idx] = offspring[i]
            
        generation += 1
        
    return best_h, best_c5
