import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute the overlap integral for all valid shifts
    overlap = correlate(h, 1 - h, mode="full")
    # Take only the relevant part of the overlap (shifts from 0 to 2)
    max_overlap = np.max(overlap[n_points-1:])
    return max_overlap * dx

def project_h(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to target sum
        h = h * target_sum / current_sum
    return h
