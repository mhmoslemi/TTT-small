import numpy as np
import time
from scipy import correlate

def evaluate_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute overlap integral
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum overlap (from the last n_points elements)
    overlap = corr[n_points - 1:]
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, n_points):
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Scale to maintain integral = 1
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    best_h = h.copy()
    best_c5 = evaluate_solution(h, n_points)
    
    # Optimization parameters
    max_iter = 10000
    step_size = 0.05
    n_peaks = 5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        c5 = evaluate_solution(h, n_points)
        if c5 < 0.3807:
            break
            
        # Compute correlation to identify peaks
        corr = correlate(h, 1 - h, mode="full")
        overlap = corr[n_points - 1:]
        
        # Find peaks (largest overlaps)
        peak_indices = np.argsort(overlap)[::-1][:n_peaks]
        
        # Make a copy of h for the update
        h_new = h.copy()
        
        # For each peak, try to reduce it in a targeted way
        for k in peak_indices:
            # Reduce h[i] where 1-h[i+k] is high contribution to overlap
            # and we can increase others to compensate
            for i in range(n_points - k):
                if h[i] > 0 and h[i + k] < 1:
                    # Try to decrease h[i] and increase h[i+k]
                    if h[i] > 0.01 and h[i+k] < 0.99:
                        h_new[i] -= step_size * 0.1
                        h_new[i + k] += step_size * 0.1
                    
        # Project h_new to ensure constraints
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        proposed_c5 = evaluate_solution(h_new, n_points)
        
        # Accept if better or occasionally accept worse for escape
        if proposed_c5 < c5 or np.random.rand() < 0.05:
            h = h_new.copy()
            if proposed_c5 < best_c5:
                best_c5 = proposed_c5
                best_h = h.copy()
                
    return (best_h.tolist(), best_c5, n_points)
