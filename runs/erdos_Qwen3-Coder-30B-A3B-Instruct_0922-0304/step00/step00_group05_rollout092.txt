import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    # Compute overlap for all lags
    overlap = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx and return max
    return np.max(overlap) * dx

def project_h(h, dx):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)*dx=1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (2.0 / (dx * h_sum))
    else:
        h = np.full_like(h, 1.0 / len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set parameters
    start_time = time.time()
    if 'initial_h_values' in globals():

        n_points = len(h_start)
    else:
        n_points = 70
        h_start = np.random.rand(n_points)
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm_size = 100
    particles = []
    velocities = []
    
    # Initialize particles
    particles.append(h_start.copy())
    velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    for _ in range(swarm_size - 1):
        particle = np.random.rand(n_points)
        particle = project_h(particle, dx)
        particles.append(particle)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize personal bests
    pbest = [particles[i].copy() for i in range(swarm_size)]
    pbest_scores = [evaluate_h(p, dx) for p in particles]
    
    # Initialize global best
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Main loop
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (pbest[i] - particles[i]) +
                            c2 * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_h(particles[i], dx)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = particles[i].copy()
                gbest_score = score
                
        iteration += 1
    
    return (gbest.tolist(), gbest_score, n_points)
