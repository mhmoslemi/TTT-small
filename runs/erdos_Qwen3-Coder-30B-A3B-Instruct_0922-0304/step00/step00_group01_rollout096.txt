import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h values."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts from -n+1 to n-1)
    mid = len(overlaps) // 2
    max_overlap = np.max(overlaps[mid - n + 1:mid + n - 1]) * dx
    return max_overlap
