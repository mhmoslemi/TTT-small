from scipy import fft
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        h[0] = 0.4
        h[-1] = 0.6
    
    # Normalize to sum to 1
    h = h * 2.0 / h.sum()
    dx = 2.0 / n_points
    
    def evaluate_c5(h_vec):
        # Compute the overlap integral
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    def project_feasible(h_vec):
        # Project onto feasible set: 0 <= h <= 1 and sum(h) = 1
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / h_vec.sum()
        return h_vec
    
    def optimize_step(h_vec):
        # Fast approximation of step function optimization in frequency domain
        # using a simple iterative scheme with frequency domain projection
        h_vec = project_feasible(h_vec.copy())
        
        # FFT shift for efficient handling
        H = fft.fft(h_vec)
        
        # Compute magnitude squared
        S = np.abs(H)**2
        
        # Simple projection: flatten spectrum and keep DC component
        # Place more weight in the low frequencies to make h more regular
        # Weaker projection to allow changes
        # Update in frequency domain based on spectral flatness
        new_H = H * np.sqrt(1.0/S)  # This creates an inverse spectrum projection
        # Simple projection: avoid DC distortion
        new_H[0] = H[0]
        # Reconstruct in time domain
        new_h = np.real(fft.ifft(new_H))
        
        # Project back onto feasible set
        new_h = project_feasible(new_h)
        
        return new_h
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        # This is not a full gradient descent but a heuristic inspired by frequency domain ideas
        # We iteratively update in frequency domain for better smoothness and structural properties
        h_new = optimize_step(h)
        c5_new = evaluate_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        iteration += 1
        
        # Early exit if improvement is insufficient
        if iteration % 50 == 0:
            if evaluate_c5(h) > best_c5 * 0.9999:
                break
    
    return (best_h.tolist(), best_c5, n_points)
