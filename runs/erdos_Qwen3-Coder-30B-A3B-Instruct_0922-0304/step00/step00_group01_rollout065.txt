import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except:
        n_points = 47
        h_init = np.ones(n_points) * 0.5
    
    h_init = np.array(h_init)
    h_init = h_init * (n_points / 2) / np.sum(h_init)  # ensure sum = n_points/2
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, n_points)
    
    # Perturbations for multi-start
    perturbations = [h_init.copy()]
    for _ in range(4):
        pert = h_init + np.random.normal(0, 0.01, n_points)
        pert = np.clip(pert, 0, 1)
        pert = pert * (n_points / 2) / np.sum(pert)
        perturbations.append(pert)
    
    for h0 in perturbations:
        h = h0.copy()
        h_prev = h.copy()
        
        iter_count = 0
        last_improvement = 0
        
        while time.time() - start_time < budget_s - 1:
            iter_count += 1
            
            # Compute current C5 and worst overlap
            overlap = correlate(h, 1-h, mode='full')
            overlap = overlap[len(h)-1:]  # Only consider non-negative shifts
            k_max = np.argmax(overlap)
            c5 = overlap[k_max] / n_points * 2
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iter_count
            
            # Gradient for the worst overlap
            shifted_h = np.roll(1-h, k_max)
            grad = shifted_h - h * shifted_h
            
            # Line search
            alpha = 0.01
            max_steps = 30
            for _ in range(max_steps):
                h_new = h - alpha * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)  # rescale
                
                # Check if it improves
                new_overlap = correlate(h_new, 1-h_new, mode='full')
                new_overlap = new_overlap[len(h_new)-1:]
                new_c5 = new_overlap[k_max] / n_points * 2
                
                if new_c5 < c5:
                    h = h_new
                    break
                alpha *= 0.5
            
            # Convex refinement every 50 iterations
            if iter_count % 50 == 0:
                # Convex program to reduce the maximum overlap for k_max
                h_var = cp.Variable(n_points)
                constraints = [
                    h_var >= 0,
                    h_var <= 1,
                    cp.sum(h_var) == n_points/2
                ]
                
                # Objective: minimize overlap at k_max
                objective = cp.Minimize(
                    cp.sum(cp.multiply(h_var, np.roll(1-h, k_max)))
                )
                
                prob = cp.Problem(objective, constraints)
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)  # Fast solver
                    h_cvx = h_var.value
                    if h_cvx is not None:
                        h = np.clip(h_cvx, 0, 1)
                        h = h * (n_points / 2) / np.sum(h)
                except:
                    pass  # Skip if infeasible

            # Termination check
            if iter_count - last_improvement > 200 and iter_count > 500:
                break
            if time.time() - start_time > budget_s - 0.1:
                break
                
    return (best_h, best_c5, n_points)

def evaluate_h(h, n_points):
    # Compute maximum overlap
    overlap = correlate(h, 1-h, mode='full')
    overlap = overlap[n_points-1:]  # Only consider non-negative shifts
    return np.max(overlap) / n_points * 2
