import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05
    delta = 0.1
    
    # Track best solution
    best_h = h.copy()
    current_energy = evaluate(h, dx)
    best_energy = current_energy
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Perturb a fraction of indices
        n_perturb = max(1, int(n_points * perturb_frac))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Make perturbations
        h_new = h.copy()
        for i in indices:
            perturbation = np.random.uniform(-delta, delta)
            h_new[i] += perturbation
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to satisfy integral constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new energy
        new_energy = evaluate(h_new, dx)
        
        # Accept or reject
        delta_e = new_energy - current_energy
        if delta_e < 0 or np.exp(-delta_e / T) > np.random.rand():
            h = h_new
            current_energy = new_energy
            
            if current_energy < best_energy:
                best_energy = current_energy
                best_h = h.copy()
        
        # Cool temperature
        T *= cooling_rate
        
        # Early stopping
        if best_energy < 0.3807:
            break
    
    return (best_h.tolist(), best_energy, n_points)

def evaluate(h, dx):
    # Compute overlap integral for all k
    conv = correlate(h, 1-h, mode='full')
    # Keep only valid correlations (not overlapping)
    n = len(h)
    result = np.max(conv[n-1:]) * dx
    return result
