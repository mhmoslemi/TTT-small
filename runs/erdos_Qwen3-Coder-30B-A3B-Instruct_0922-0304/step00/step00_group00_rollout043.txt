    def compute_c5(h_values):
        n = len(h_values)
        dx = 2.0 / n
        # For each shift k, compute overlap integral: int h(x)(1 - h(x+k)) dx
        overlaps = []
        # Using discrete correlation
        h_ext = np.concatenate([h_values, h_values])
        for k in range(1-n, n):
            # Shift indices based on k
            if k >= 0:
                # h[0] -> h[k], ..., h[n-1-k] -> h[n-1]
                shift_indices = np.arange(k, n)
            else:
                # h[0] -> h[n+k], ..., h[n+k-1] -> h[n-1], 
                # h[n+k] -> h[0], ..., h[n-1] -> h[n-1-k]
                shift_indices = np.arange(0, n + k)
            # Compare with extended array
            if k >= 0:
                vals = np.take(h_ext, shift_indices, mode='wrap')
            else:
                vals = np.take(h_ext, np.arange(n+k, n), mode='wrap')
            overlap = np.sum(h_values[:len(vals)] * (1 - vals)) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    # Gradient descent helper for k-th shift
    def gradient_step(h_val, k_shift):
        n = len(h_val)
        grad = np.zeros_like(h_val)
        # Compute for shift k
        # d/dh[i] of sum h[i](1-h[(i+k)%n])
        # = 1 - h[(i+k)%n]  - h[i] * d/dh[i](h[(i+k)%n]) (but for term in sum, differentiate h[i])
        # = 1 - h[(i+k)%n]  (i.e., gradient for h[i] contribution to element i from overlap at k)
        for i in range(n):
            j = (i + k_shift) % n
            grad[i] += (1 - h_val[j])
        return grad
    
    # Iteration loop
    iter_count = 0
    while iter_count < max_iters and (time.time() - start_time) < budget_s - 0.1:

        dx = 2.0 / n

            overlaps.append(overlap)
        k_max = np.argmax(overlaps)
        k_max_shift = k_max - (n - 1)
        
        # Gradient descent step
        alpha = 0.05
        grad = gradient_step(h, k_max_shift)
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        h = h_new
        
        # Convex QP refinement (simplified version)

        # In case of complicated constraints, just use clipping and normalizing
        # However, refine using constraint-solving
        # Simple quadratic program approach for this step:
        # Minimize sum of (h_new[i] - h[i])^2, subject to:
        # 1. 0 <= h[i] <= 1
        # 2. sum h = n/2
        # This is a quadratic programming problem in cvxpy.
        # But can be greatly simplified by just making sure norms are good.
        # We'll do a small number of QP steps:
        
        # Convex relaxation approach (simplified)
        # We make adaptive lossestimate update by approx minimization of worst case
        
        # Project h onto feasible set
        if np.sum(h) > n_points / 2:
            h = h * (n_points / 2.0) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        iter_count += 1
    
    # Final check and return
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
