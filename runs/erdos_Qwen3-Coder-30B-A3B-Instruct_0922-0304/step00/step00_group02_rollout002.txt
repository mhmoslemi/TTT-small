import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
    except:
        # Fallback: create a simple construction
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Initial setup
    start_time = time.time()
    delta = 0.1
    iter_count = 0
    
    while time.time() - start_time < budget_s:
        iter_count += 1
        
        # Adjust delta over time
        if iter_count % 100 == 0:
            delta *= 0.9
            
        # Select two random indices
        i, j = np.random.randint(0, n_points, 2)
        
        # Skip if same index
        if i == j:
            continue
            
        # Try small perturbation
        h_new = h.copy()
        perturbation = np.random.uniform(-delta, delta)
        
        # Ensure we don't go out of bounds
        if h_new[i] + perturbation > 1 or h_new[j] - perturbation < 0:
            continue
            
        h_new[i] += perturbation
        h_new[j] -= perturbation
        
        # Maintain sum constraint
        # Since we're only changing two points, we adjust the last one to maintain sum
        # But here we do a direct sum check
        if abs(np.sum(h_new) * dx - 1.0) > 1e-10:
            continue
            
        # Evaluate C5
        new_c5 = evaluate_c5(h_new, dx)
        
        # Accept if improvement
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            best_h = h.copy()
        
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute correlation for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum
    max_corr = np.max(corr)
    # Scale by dx
    return max_corr * dx
