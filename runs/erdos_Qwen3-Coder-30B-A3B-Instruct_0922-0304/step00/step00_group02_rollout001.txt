import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial construction
        n_points = 45
        h = np.full(n_points, 0.5)
        h[0] = 0.0
        h[1] = 1.0
        h[2] = 0.0
        h[3] = 1.0
        h[-1] = 0.0
        h[-2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    def compute_c5(h_vals):
        # Compute the overlap integral
        conv = correlate(h_vals, 1 - h_vals, mode="full")
        # We only care about the valid range of k
        # The max correlation corresponds to max_k of the integral
        max_corr = np.max(conv)
        return max_corr * dx

    def feasibility_check(h_vals):
        # Check sum constraint
        if abs(np.sum(h_vals) - target_sum) > 1e-6:
            return False
        # Check bounds
        if np.any(h_vals < 0) or np.any(h_vals > 1):
            return False
        return True

    def project_onto_feasible(h_vals):
        # Project onto feasible region while maintaining sum
        h_copy = np.copy(h_vals)
        current_sum = np.sum(h_copy)
        diff = current_sum - target_sum
        
        # Simple projection to ensure sum = target_sum
        if abs(diff) > 1e-6:
            h_copy -= diff / n_points
            
        # Clamp to [0,1]
        np.clip(h_copy, 0, 1, out=h_copy)
        
        return h_copy

    best_h = np.copy(h)
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    # Initial adjustments
    h = project_onto_feasible(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_h = np.copy(h)
        best_c5 = c5

    # Main optimization loop
    iters = 0
    max_iters = 50000
    delta = 0.1
    delta_decay = 0.9999
    iter_delta_update = 1000

    while time.time() - start_time < budget_s and iters < max_iters:
        iters += 1
        
        # Slightly decrease delta over time
        if iters % iter_delta_update == 0:
            delta *= delta_decay
            
        # Pick two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Clone current h
        h_new = np.copy(h)
        
        # Apply perturbation
        if delta > 1e-10:
            perturbation = np.random.uniform(-delta, delta)
            if i != j:
                h_new[i] += perturbation
                h_new[j] -= perturbation
            else:
                h_new[i] += perturbation
            
            # Project onto feasible set
            h_new = project_onto_feasible(h_new)
            
            # Compute new C5
            new_c5 = compute_c5(h_new)
            
            # Accept change if better
            if new_c5 < best_c5:
                h = h_new
                best_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = np.copy(h)
            else:
                # Continue with old h
                pass
        else:
            # End early if delta is too small
            break

    # Final validation
    final_h = project_onto_feasible(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), float(final_c5), int(n_points))
