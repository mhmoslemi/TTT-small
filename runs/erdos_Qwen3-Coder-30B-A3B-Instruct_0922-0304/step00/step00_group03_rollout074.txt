import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Default construction: step function
        n_points = 70
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
    
    # Normalize h0 to sum to n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create optimization variables
    h = cp.Variable(n_points)
    
    # Initialize
    h_value = h0.copy()
    lambda_k = np.zeros(2 * n_points - 1)  # overlap indices
    mu = 0.0
    rho = 1e-3  # penalty parameter
    
    # ADMM iterations
    iter_count = 0
    max_iter = 5000
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h0.copy()
    
    def compute_c5(h_vals):
        # Compute overlaps: correlation of h with (1-h)
        h_conv_1mh = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Convolution gives indices from -(n-1) to (n-1)
        c5_val = np.max(h_conv_1mh)
        return c5_val
    
    def compute_overlap_indices(n_points):
        # indices of overlap convolution
        return np.arange(-n_points + 1, n_points)
    
    max_k = 2 * n_points - 1
    k_indices = compute_overlap_indices(n_points)
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Solve convex subproblem: minimize quadratic + overlap term
        # ||h - h_old||^2 + rho/2 * ||h||^2 + sum(lambda_k * (h * (1-h))_k)
        # For efficiency, approximate the correlation computation
        
        # Build lagrangian
        # For this we'll solve the subproblem using CVXPY
        # L = 
        # <h, (h_old - rho*h0) > + (rho/2) * ||h||^2 + 
        # <lambda, (h ⋆ (1-h))> 
        # We'll do this with a parameterized solve over overlap values to derive gradient
        
        # Define a new problem, keeping previous indicator of current overlaps
        prob = cp.Problem(
            cp.Minimize(
                cp.sum_squares(h - h_value) + 
                rho * cp.sum_squares(h) + 
                cp.sum(lambda_k * (cp.conv(1-h, h)[n_points-1:2*n_points-2]))
            ),
            [0 <= h, h <= 1, cp.sum(h) * 2.0 / n_points == 1.0]
        )
        
        try:
            # Deactivate warm start since not supported by all solvers 
            prob.solve(solver=cp.CVXOPT, verbose=False)
            h_new = h.value
        except:
            # Fallback to current solution
            h_new = h_value.copy()
        
        if h_new is None:
            h_new = h_value.copy()
        
        # Update dual variables
        # Evaluate overlaps using new h
        h_new_overlap = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(h_new_overlap)
        # Update lambda with projected gradient ascent
        lambda_update = np.zeros_like(lambda_k)
        for i in range(len(lambda_k)):
            # Correlation term: 
            # overlaps[i] = np.sum(h_new[j] * (1 - h_new[i+j]) for j in range(n_points) if 0 <= i+j < n_points)
            overlap_i = h_new_overlap[i]  
            lambda_update[i] = lambda_k[i] + 1e-2 * max(0, c5_new + 1e-8 - overlap_i)
        
        # Update mu
        mu += 1e-2 * (np.sum(h_new) * 2.0 / n_points - 1.0)
        
        lambda_k = lambda_update
        h_value = h_new
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = iter_count
        
        if c5_new < 0.3807:
            break
            
        if iter_count - last_improvement > 500:
            break
        
        if iter_count % 100 == 0 and time.time() - start_time > 0.9 * budget_s:
            break
    
    # Final adjust to meet constraint exactly
    # Recompute the exact overlap with best_h
    final_overlap = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_overlap)
    
    # Re-scale to mean = 0.5
    best_h = best_h * np.sum(best_h) / (n_points / 2)
    
    return (best_h.tolist(), final_c5, n_points)
