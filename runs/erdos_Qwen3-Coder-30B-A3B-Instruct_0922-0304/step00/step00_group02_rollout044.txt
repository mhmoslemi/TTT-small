import numpy as np
import time
from scipy import correlate

def evaluate_h(h, n_points):
    dx = 2.0 / n_points
    overlaps = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(overlaps)
    c5 = max_overlap * dx
    return c5

def is_feasible(h, n_points):
    return np.all(h >= 0) and np.all(h <= 1) and abs(np.sum(h) - n_points / 2) < 1e-10

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        diff = n_points / 2 - current_sum
        h += diff / n_points
        h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.01):
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    if np.any(mask):
        h[mask] += np.random.normal(0, 0.05, np.sum(mask))
        h = repair_h(h, len(h))
    return h

def ga_optimize(initial_h, n_points, budget_s):
    population_size = 100
    max_generations = 1000
    mutation_rate = 0.01
    elite_size = population_size // 2
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h, n_points)
    start_time = time.time()
    no_improvement_count = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all candidates
        fitness_scores = []

            c5 = evaluate_h(h, n_points)
            fitness_scores.append(1 / (c5 + 1e-10))
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]

        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            
        # Termination condition
        if gen > 10 and abs(best_c5 - evaluate_h(population[0], n_points)) < 1e-6:
            no_improvement_count += 1
            if no_improvement_count >= 100:
                break
        else:
            no_improvement_count = 0
            
        # Selection
        selected = population[:elite_size]
        
        # Crossover and mutation
        new_population = selected.copy()
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Repair if needed
            child1 = repair_h(child1, n_points)
            child2 = repair_h(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population
    
    return best_h, best_c5
