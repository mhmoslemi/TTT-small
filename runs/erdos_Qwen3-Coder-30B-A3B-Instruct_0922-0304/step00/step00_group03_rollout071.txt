import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, else construct
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points)
        h[:n_points//2] = 0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # ADMM solver parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 5000
    tol = 1e-4
    
    # Variables for ADMM
    lambda_vars = np.zeros(2 * n_points - 1)  # Overlap constraints
    mu = 0  # Integral constraint dual variable
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Update h by solving convex subproblem
        h_new = solve_subproblem(h, lambda_vars, mu, rho, n_points, dx)
        
        # Compute overlaps
        overlaps = correlate(h_new, 1 - h_new, mode='full')
        C5_new = np.max(overlaps) * dx
        
        # Update dual variables
        lambda_vars += alpha * np.maximum(0, C5_new + 1e-10 - overlaps)
        mu += alpha * (np.sum(h_new) * dx - 1)
        
        # Entropy regularization for lambda
        lambda_vars = lambda_vars + beta * np.log(1 + lambda_vars)
        
        # Clamp lambda to reasonable values
        lambda_vars = np.clip(lambda_vars, -1e10, 1e10)
        
        h = h_new
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h.copy()
            
        if best_C5 < 0.3807:
            break

    return (best_h.tolist(), best_C5, n_points)

def solve_subproblem(h, lambda_vars, mu, rho, n_points, dx):
    # Define optimization variables
    h_var = cp.Variable(n_points)
    
    # Objective: minimize quadratic penalty with Lagrangian terms
    # h ⋆ (1 - h) term with dual variable influence
    # + mu * (int h - 1) 
    # + regularization term
    
    # Compute shifted correlations in pre-computed form
    # Lagrangian part - this is a bit indirect since this is an integral
    # But we include it in a convex approximation. We'll compute once per iteration
    
    # Direct quadratic penalty problem
    # Minimize sum over all k of (h * (1-h)[k]) * dx + lambda[k] + ...
    # Subject to h_i in [0,1], sum(h) = 1
    
    objective = cp.Minimize(
        cp.sum_squares(h_var - h) * rho + 
        cp.sum(lambda_vars) * 1e-6  # Penalties
        # Add a regularization term to keep solution stable
    )
    
    # Add constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) * dx == 1
    ]
    
    # Form and solve the problem
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.SCS, verbose=False, max_iters=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
    except:
        pass 
    return h.copy()  # Fallback
    
def compute_C5(h, dx):
    overlaps = correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)
