import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//4] = 1.0
        h[n_points//4:2*n_points//4] = 0.0
        h[2*n_points//4:3*n_points//4] = 1.0
        h[3*n_points//4:] = 0.0
        # Normalize to sum = n_points/2 
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum

    dx = 2.0 / n_points
    start_time = time.time()
    
    # Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = int(1e4)
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    
    # Initial C5
    current_c5 = np.max(signal.correlate(h, 1 - h, mode="full")) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Initial adaptive perturbation
    delta_h = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Perturb h
        new_h = h.copy()
        
        # Random swap between two points
        if np.random.random() < 0.5:
            i, j = np.random.choice(n_points, 2, replace=False)
            # Swap and adjust to maintain sum
            val_i, val_j = new_h[i], new_h[j]
            delta = val_i - val_j
            new_h[i] = val_j
            new_h[j] = val_i
            
            # Adjust to maintain sum constraint
            if delta != 0:
                # Randomly select other indices for adjustment
                adjust_idx = np.random.choice(n_points)
                new_h[adjust_idx] = np.clip(new_h[adjust_idx] - delta, 0, 1)
                
        else: # Adjacent imbalance shift
            i = np.random.randint(n_points - 1)
            delta = np.random.uniform(-delta_h, delta_h)
            # Shift mass with bounded constraint
            new_h[i] = np.clip(new_h[i] + delta, 0, 1)
            new_h[i+1] = np.clip(new_h[i+1] - delta, 0, 1)
            
        # Rescale to satisfy constraint sum(h) = n_points/2
        h_sum = np.sum(new_h)
        new_h *= (n_points / 2) / h_sum
        
        # Evaluate new C5
        new_c5 = np.max(signal.correlate(new_h, 1 - new_h, mode="full")) * dx
        
        # Accept or reject based on simulated annealing
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.random() < np.exp(-delta_c5 / T_max):
            h = new_h
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Update temperature
        T_max *= alpha
        
        # Adapt delta_h based on acceptance rate if needed
        if iteration % 100 == 0: 
            delta_h = max(0.001, delta_h * 0.99)

    return (best_h.tolist(), best_c5, n_points)
