import numpy as np
from scipy import correlate
import time
import random

def evaluate_h(h, dx):
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_sum(h, target_sum):
    # Ensures exactly target_sum ones in h
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h.copy()
    
    # Find indices of 0s and 1s
    zeros = np.where(h == 0)[0]
    ones = np.where(h == 1)[0]
    
    if current_sum < target_sum:
        # Need to flip more 0s to 1s
        needed = int(target_sum - current_sum)
        flip_indices = zeros[:needed]
        h[flip_indices] = 1
    else:
        # Need to flip some 1s to 0s
        needed = int(current_sum - target_sum)
        flip_indices = ones[:needed]
        h[flip_indices] = 0
    
    return h.copy()

def generate_neighbor(h, n_points, target_sum):
    # Generate a neighbor by flipping 2 bits
    h_new = h.copy()
    indices = np.random.choice(n_points, 2, replace=False)
    h_new[indices[0]] = 1 - h_new[indices[0]]
    h_new[indices[1]] = 1 - h_new[indices[1]]
    
    # Repair the sum to be exactly target_sum
    h_new = repair_sum(h_new, target_sum)
    return h_new

def tabu_search(initial_h, n_points, target_sum, max_iter, tabu_tenure, dx):
    # Initialize
    current_h = initial_h.copy()
    best_h = current_h.copy()
    best_value = evaluate_h(current_h, dx)
    
    tabu_list = []
    tabu_tenure_count = 0
    
    for i in range(max_iter):
        # Generate neighbors
        neighbors = []
        for _ in range(10):  # Generate 10 neighbors
            neighbor = generate_neighbor(current_h, n_points, target_sum)
            if tuple(neighbor) not in tabu_list:
                neighbors.append(neighbor)
        
        if not neighbors:
            continue
            
        # Evaluate neighbors
        neighbor_values = []
        for neighbor in neighbors:
            val = evaluate_h(neighbor, dx)
            neighbor_values.append(val)
        
        # Select best neighbor
        best_neighbor_idx = np.argmin(neighbor_values)
        best_neighbor = neighbors[best_neighbor_idx]
        neighbor_value = neighbor_values[best_neighbor_idx]
        
        # Update current solution
        current_h = best_neighbor.copy()
        
        # Update best if improved
        if neighbor_value < best_value:
            best_h = current_h.copy()
            best_value = neighbor_value
        
        # Manage tabu list
        tabu_list.append(tuple(current_h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
    
    return best_h, best_value

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)

    start_time = time.time()
    
    # Set problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points // 2
    
    # Precondition initial_h_values if possible, otherwise setup
    try:
        h = initial_h_values.copy()
        h = repair_sum(h, target_sum)
    except NameError:
        # Create initial binary vector
        h = np.zeros(n_points, dtype=int)
        h[:target_sum] = 1
        np.random.shuffle(h)
    
    # Run tabu search
    # Estimate iterations within budget
    max_iter = 1000  # Safe estimate
    tabu_tenure = 100
    
    # Execute tabu search
    best_h, best_value = tabu_search(h, n_points, target_sum, max_iter, tabu_tenure, dx)
    
    # Final validation
    best_h = repair_sum(best_h, target_sum)
    final_c5 = evaluate_h(best_h, dx)
    
    # Final check if we've exceeded time
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:  # Leave 1 sec for return

    return (best_h, final_c5, n_points)
