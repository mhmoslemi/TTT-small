import numpy as np
from scipy import optimize
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Fall back to a simple construction
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[1] = 0.0
        h0[2] = 1.0
        h0[3] = 1.0
        h0[4] = 0.0
        h0[5] = 0.0
        h0[6] = 1.0
        h0[7] = 1.0
        
        # Normalize to sum to n_points/2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Helper: evaluate C5
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        # compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # take only the valid overlap values (not including the full convolution)
        mid = len(h) - 1
        valid_corr = corr[mid : mid + len(h)]
        c5 = np.max(valid_corr) * dx
        return c5
    
    # Helper: project h to [0,1] and maintain sum
    def project_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Gradient descent refinement of h0
    h = h0.copy()
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.05
    decay = 0.98
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # compute current C5 and dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        mid = len(h) - 1
        valid_corr = corr[mid : mid + len(h)]
        k_max = np.argmax(valid_corr)
        c5_cur = valid_corr[k_max]
        
        # compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        h_plus = h.copy()
        for j in range(len(h)):
            h_plus[j] = min(1, h[j] + eps)
            corr_plus = np.correlate(h_plus, 1 - h, mode='full')
            valid_corr_plus = corr_plus[mid : mid + len(h)]
            grad[j] = (valid_corr_plus[k_max] - c5_cur) / eps
            h_plus[j] = h[j]
            
        # update h
        h = h - learning_rate * grad
        h = project_h(h)
        
        # decay learning rate
        if i % 20 == 0:
            learning_rate *= decay
            
    # evaluate final solution
    best_c5 = evaluate_c5(h)
    
    # Optionally increase resolution to 84 for more precision if time allows
    if time.time() - start_time < budget_s - 5 and n_points == 42:
        n_points_new = 84
        h_new = np.zeros(n_points_new)
        h_new[::2] = h
        h_new[1::2] = (h[:-1] + h[1:]) / 2
        h = h_new
        n_points = n_points_new
        
        # again refine the finer grid
        max_iter = 500
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            corr = np.correlate(h, 1 - h, mode='full')
            mid = len(h) - 1
            valid_corr = corr[mid : mid + len(h)]
            k_max = np.argmax(valid_corr)
            c5_cur = valid_corr[k_max]
            
            eps = 1e-6
            grad = np.zeros_like(h)
            h_plus = h.copy()
            for j in range(len(h)):
                h_plus[j] = min(1, h[j] + eps)
                corr_plus = np.correlate(h_plus, 1 - h, mode='full')
                valid_corr_plus = corr_plus[mid : mid + len(h)]
                grad[j] = (valid_corr_plus[k_max] - c5_cur) / eps
                h_plus[j] = h[j]
                
            h = h - learning_rate * grad
            h = project_h(h)
            
            if i % 20 == 0:
                learning_rate *= decay
                
        best_c5 = evaluate_c5(h)
        dx = 2.0 / n_points
        
    return (h, best_c5, n_points)
