import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a given h."""
    # Compute correlation with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Normalize by dx
    corr *= dx
    # Return the maximum value (which should be the maximum overlap)
    return np.max(corr)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Given problem parameters
    n_points = 42
    dx = 2.0 / n_points

        h0 = np.ones(n_points) * 0.5

    h0 = np.array(h0)
    
    # Normalize to sum to n_points / 2
    h0_sum = np.sum(h0)
    if h0_sum > 0:
        h0 *= (n_points / 2) / h0_sum
    
    # Define optimization problem using scipy
    def objective(h):
        return evaluate_overlap(h, dx)
        
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
        
    def constraint_bounds(h):
        # Ensure h[i] is in [0,1]
        return np.concatenate([
            h,
            1 - h
        ])
    
    # Set up bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    cons = [
        {'type': 'eq', 'fun': constraint_sum},
    ]
    
    # Optimization parameters
    best_h = h0.copy()
    best_value = evaluate_overlap(h0, dx)
    start_time = time.time()
    remaining_time = budget_s
    
    # Try gradient-based optimization
    try:
        result = minimize(
            objective, 
            h0, 
            method='SLSQP', 
            bounds=bounds, 
            constraints=cons,
            options={'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6},
            tol=1e-6
        )
        if result.success:
            h_opt = result.x
            opt_value = evaluate_overlap(h_opt, dx)
            if opt_value < best_value:

    def local_optimize(h_start):

            improvement = False
            # Try small perturbations
            for i in range(n_points):
                # Try to increment
                h_try = h.copy()
                h_try[i] = min(1.0, h[i] + 0.01)
                if np.sum(h_try) > n_points / 2:
                    h_try = h_try * (n_points / 2) / np.sum(h_try)
                v_try = evaluate_overlap(h_try, dx)
                if v_try < current_value:
                    h = h_try
                    current_value = v_try
                    improvement = True

                h_try = h.copy()
                h_try[i] = max(0.0, h[i] - 0.01)

                if v_try < current_value:

                    current_value = v_try
                    improvement = True
            # Project back to feasible sum
            h_sum = np.sum(h)
            if h_sum > 0:
                h *= (n_points / 2) / h_sum
        return h, current_value
    
    # Try local optimization from initial point
    h_local, val_local = local_optimize(h0)
    if val_local < best_value:
        best_h = h_local
        best_value = val_local
    
    # Return best result
    return (best_h, best_value, n_points)
