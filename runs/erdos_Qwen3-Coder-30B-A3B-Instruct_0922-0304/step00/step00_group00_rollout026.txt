    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute correlation for performance
    def evaluate_c5(h_vec):
        h_vec = np.array(h_vec)
        # Compute all overlaps with full mode
        correlation = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Extract valid overlaps
        n = len(h_vec)
        overlaps = correlation[n-1:] * dx
        return np.max(overlaps)
    
    # Gradient descent with adaptive step size
    step_size = 0.1
    last_improvement = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Stop if good enough or no improvement
        if c5 <= target or no_improvement_count > 50:
            break
            
        # Find worst overlap
        correlation = signal.correlate(h, 1 - h, mode='full')

        k_max = np.argmax(overlaps)
        k_shift = k_max - (n_points - 1)
        
        # Compute gradient by hand
        grad = np.zeros_like(h)

            i_shifted = (i + k_shift) % n_points

        # Renormalize
        h_sum = np.sum(h)

            h = h * (n_points / 2) / h_sum
            
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
