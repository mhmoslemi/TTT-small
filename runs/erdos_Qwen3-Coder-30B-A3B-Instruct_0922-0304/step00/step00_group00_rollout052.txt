import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 100000
    alpha = 0.1
    patience = 50
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Ensure h is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlap for all shifts
        overlaps = correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        # Only consider valid k values for avoiding boundary artifacts
        valid_indices = np.abs(k_indices) < n_points
        valid_overlaps = overlaps[valid_indices]
        k_max = k_indices[np.argmax(valid_overlaps)]
        
        # Compute gradient w.r.t. the worst shift
        grad = np.zeros(n_points)
        if k_max >= 0:
            # h[i] * (1 - h[i + k_max])
            grad[:n_points - k_max] += (1 - h[k_max:]) * dx
            grad[k_max:] -= h[:n_points - k_max] * dx
        else:
            # h[i] * (1 - h[i - k_max])
            k_abs = -k_max
            grad[k_abs:] += (1 - h[:n_points - k_abs]) * dx
            grad[:n_points - k_abs] -= h[k_abs:] * dx
        
        # Update h
        h = h - alpha * grad
        
        # Clamp h into [0, 1]
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Check C5
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        if c5 < target_c5:
            break
            
        if iteration - last_improvement > patience:
            alpha *= 0.9  # Adaptive learning rate
            
    return (best_h.tolist(), best_c5, n_points)
