import numpy as np
import time
from scipy import optimize

def evaluate_c5(h):
    """Evaluate the maximum overlap integral C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value
    max_corr = np.max(correlations)
    return max_corr * dx

def is_feasible(h):
    """Check if h satisfies the constraints."""
    n = len(h)
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    # Check constraint sum(h) = n / 2
    if not np.isclose(np.sum(h), n / 2):
        return False
    return True

def project_feasible(h):
    """Project h onto the feasible set."""
    n = len(h)
    # Clip values
    h = np.clip(h, 0, 1)
    # Adjust sum to n/2
    current_sum = np.sum(h)
    if current_sum != n / 2:
        # Adjust by adding the difference to each entry
        adjustment = (n / 2 - current_sum) / n
        h = h + adjustment
        # Clip again
        h = np.clip(h, 0, 1)
    return h

def generate_individual(n_points):
    """Generate a random feasible individual."""
    h = np.random.random(n_points)
    h = project_feasible(h)
    return h

def ga_optimize(initial_h, n_points, budget_s):
    """Optimize h using genetic algorithm."""
    start_time = time.time()
    
    # Parameters
    population_size = 100
    generations = 1000
    elite_size = population_size // 2
    mutation_rate = 0.01
    
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        population.append(generate_individual(n_points))
    
    best_c5 = float('inf')
    best_h = initial_h.copy()
    
    # Keep track of generations without improvement
    no_improvement = 0
    max_no_improvement = 100
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h)
            fitnesses.append(1.0 / (c5 + 1e-10))
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)[::-1]
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best
        current_best = evaluate_c5(population[0])
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > max_no_improvement:
            break

        # Selection (top 50%)
        selected = population[:elite_size]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(len(selected))]
            parent2 = selected[np.random.randint(len(selected))]
            
            # Crossover: 1-point crossover
            point = np.random.randint(1, n_points)
            child1, child2 = parent1.copy(), parent2.copy()
            child1[point:], child2[point:] = parent2[point:], parent1[point:]
            
            # Apply mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project to feasible set
            child1 = project_feasible(child1)
            child2 = project_feasible(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        population = new_population[:]
        
    return best_h, best_c5
