import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Parameters
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    max_iter = 10000
    gamma = 0.01
    eps = 1e-4
    patience = 50
    
    # Initialize
    h = np.array(initial_h_values, dtype=float)
    h /= np.sum(h) * dx  # Normalize to sum = 1
    h *= n_points / 2.0  # Adjust to desired sum
    
    # Tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    patience_counter = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode="full")
        # Convolution is of size 2*n_points - 1
        # Corresponds to lags from -(n_points - 1) to (n_points - 1)
        C = corr * dx  # Scale by dx
        max_c = np.max(C)
        max_indices = np.where(np.abs(C - max_c) < eps)[0]
        max_lags = [i - (n_points - 1) for i in max_indices]
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for lag in max_lags:
            if lag <= 0:
                # h[i] * (1 - h[i-lag]) term
                for i in range(n_points):
                    if i - lag >= 0:
                        gradient[i] += 1 - h[i - lag]
            else:
                # h[i] * (1 - h[i+lag]) term  
                for i in range(n_points):
                    if i + lag < n_points:
                        gradient[i] += 1 - h[i + lag]
        
        # Update h
        h = h - gamma * gradient
        
        # Project to [0,1] and normalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h *= (n_points / 2.0) / h_sum  # Adjust to integral = 1

        # Evaluate
        current_c5 = evaluate_erdos_solution(h, dx)
        
        # Check improvement
        if current_c5 < best_c5 - eps:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        # Adaptive learning rate
        if patience_counter > 20:
            gamma *= 0.9
            patience_counter = 0
            
    return (best_h.tolist(), best_c5, n_points)
