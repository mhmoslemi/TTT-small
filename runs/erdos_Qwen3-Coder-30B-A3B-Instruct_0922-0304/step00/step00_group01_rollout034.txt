import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with symmetric construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[10:37] = 1.0  # Increase some values
        h[0:10] = 0.0
        h[37:47] = 0.0
        # Normalize to meet sum constraint
        h = h * (n_points / 2.0) / np.sum(h)
    
    # Precompute index mapping for circular correlation
    indices = np.arange(n_points)
    
    def compute_C5(h):
        # Compute all overlaps using convolution
        h_rev = np.flip(h)
        conv = np.convolve(h, h_rev, mode='full')
        # We need to compute \int h(x)(1 - h(x+k)) dx, which is 
        # sum(h[i] * (1 - h[(i+k)%n_points])) for all k
        # This is equivalent to sum(h[i] - h[i]*h[(i+k)%n_points])
        #                  = sum(h[i]) - sum(h[i]*h[(i+k)%n_points])
        # This is sum(h) - conv(k + n_points - 1) for shift k
        # But the original task specifies correlate with 1-h
        # So we compute correlate(h, 1-h, mode="full")
        one_minus_h = 1 - h
        corr = np.correlate(h, one_minus_h, mode="full")
        # Only valid for shifts [0, 2*n_points-2], we need [0, n_points-1] 
        # We actually want max of all valid shifts k from 0 to n_points-1
        # But h is periodic, so for shift k=0, it's sim (h * (1-h))
        # For k=1, it's sum(h[i] * (1-h[i+1])) etc.
        # The valid correlations from correlate(h, 1-h, mode="full") 
        # are at 0 to 2*n_points-2.
        # The k-th shift value is at index = k + n_points - 1
        # We want k from 0 to n_points-1
        max_corr = max(corr[n_points-1 : 2*n_points-1])
        return max_corr * (2.0 / n_points)
    
    def project_feasible(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint sum(h) == n_points/2
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2.0) > 1e-10:
            # Find scalar adjustment
            delta = (n_points/2.0 - current_sum) / n_points
            h = h + delta
            h = np.clip(h, 0, 1)
            # Final adjustment
            h_sum = np.sum(h)
            scale = (n_points/2.0) / h_sum
            h = h * scale
            h = np.clip(h, 0, 1)
        return h
    
    best_c5 = compute_C5(h)
    best_h = h.copy()
    
    # Iteration count limiting
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all possible shifts k
        # Convolution will dominate in performance if we do it for all shifts
        # We can directly compute for each k
        h1 = 1.0 - h
        max_overlaps = []
        shifts = []

        # Compute all C(k) = sum(h[i] * (1 - h[(i+k)%n_points]))
        # for k=0 to n_points-1
        for k in range(n_points):
            dot = 0
            for i in range(n_points):
                dot += h[i] * h1[(i + k) % n_points]
            max_overlaps.append(dot)
            shifts.append(k)
            
        # Find the maximum overlap and corresponding shift
        k_max = shifts[np.argmax(max_overlaps)]
        c_max = max_overlaps[np.argmax(max_overlaps)]
        
        # Problem: minimize C(k_max) subject to bounds on h and sum constraint
        # We can write this in convex form. 
        # The impact of change in h on C(k) = sum h[i] * (1 - h[(i+k)%n_points]) 
        # equals: (1 - h[(i+k)%n_points]) for the i-th term, and -h[i] for (i-k)%n_points term
        
        # Rebuild constraint for that shift only
        # Minimize (1 - h[(i+k)%n_points]) h[i] = (1 - h[(i+k)%n_points]) h[i]
        # for i from 0 to n_points-1
        # As a linear approximation, we linearize around current h
        # We don't build full convex programs here to reduce computation load.
        # Instead of full optimization, we simply try to improve current k_max
        
        # A simplified improvement step:
        # Try to decrease coefficients where they increase overlap
        # For k_max, we want to decrease h[i] (if possible) if h[(i+k)%n_points] > 0.5
        # Or increase h[i] if h[(i+k)%n_points] < 0.5
        
        # Sharing the fractional part
        # Binary: 1 iff corresponding h[i] should be reduced to reduce overlap.
        # For very simple approach: try perturbing
        # Perturb the h values by shift k_max to reduce overlap
        # Shift by comparing current overlap
        old_h = h.copy()
        dh = np.zeros(n_points)
        # Let's perturb along the gradient in direction of maximum overlap
        # i-th component of gradient dC/dh_i at k_max
        # equals: 1 - h[(i+k)%n_points] for all i such that k is related
        # But now we better enhance with the full per-iteration gradient descent approach
        # But that's too complex for time budget
        
        # Simpler approach: try with adjusting h values
        # Try adjusting h to reduce max overlap
        # Basic idea from variational derivative and potential improvement enhancement
        # Look at blocks around h"indices" and reduce their effect
        # Important: Making it convex.
        
        # Let us just try moving some small steps to perturb h
        # this is not fully rigorous, but effective for this problem with 47 points
        try:
            h_tent = h.copy()
            corresponding_indices = []
            for i in range(n_points):
                j = (i + k_max) % n_points
                if h_tent[i] > 0.001 and h[j] > 0.001:
                    corresponding_indices.append(i)
            # Try small changes in such indices
            if len(corresponding_indices) > 0:
                for i in range(max(1, len(corresponding_indices)//2)):
                    idx = corresponding_indices[i % len(corresponding_indices)]
                    h_tent[idx] = h_tent[idx] * 0.995  # reduce
                    h_tent[(idx + k_max) % n_points] = h_tent[(idx + k_max) % n_points] * 1.005  # try compensate

                # Project back
                h_tent = project_feasible(h_tent)
                
                # Check improved C5
                c5_new = compute_C5(h_tent)
                if c5_new < best_c5:
                    h = h_tent
                    best_c5 = c5_new
                    best_h = h.copy()
        except:
            pass

    # Final check
    h = project_feasible(h)
    final_c5 = compute_C5(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    # At least ensure integral constraint
    h = project_feasible(best_h)
    final_c5 = compute_C5(h)
    
    return (h.tolist(), final_c5, len(h))
