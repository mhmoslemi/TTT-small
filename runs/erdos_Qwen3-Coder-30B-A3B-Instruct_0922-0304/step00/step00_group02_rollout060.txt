import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # The max overlaps are from indices n-1 to 2*n-2
    max_overlap = np.max(overlaps[n-1:])
    return max_overlap

def project_to_binary(h_relaxed, target_sum):
    n = len(h_relaxed)
    # Threshold at 0.5
    h_binary = (h_relaxed > 0.5).astype(float)
    current_sum = np.sum(h_binary)
    
    if current_sum == target_sum:
        return h_binary
    
    # Adjust by flipping values to reach target_sum
    diff = int(current_sum - target_sum)
    if diff > 0:
        # Need to set some 1s to 0s
        # Find indices where h_binary is 1 but h_relaxed is less than 0.5
        indices = np.where(h_binary == 1)[0]
        # Sort by distance from 0.5
        distances = [abs(h_relaxed[i] - 0.5) for i in indices]
        sorted_indices = [i for _, i in sorted(zip(distances, indices), reverse=False)]
        for idx in sorted_indices[:diff]:
            h_binary[idx] = 0
    elif diff < 0:
        # Need to set some 0s to 1s
        # Find indices where h_binary is 0 but h_relaxed is greater than 0.5
        indices = np.where(h_binary == 0)[0]
        # Sort by distance from 0.5
        distances = [abs(h_relaxed[i] - 0.5) for i in indices]
        sorted_indices = [i for _, i in sorted(zip(distances, indices), reverse=False)]
        for idx in sorted_indices[:abs(diff)]:
            h_binary[idx] = 1
    
    return h_binary

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iterations += 1
        
        # Compute current overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlaps[n_points-1:])
        
        # Create convex relaxation problem
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == target_sum
        ]
        
        # Objective: Minimize max overlap
        max_obj = cp.Variable()
        constraints.extend([max_obj >= overlaps[i] for i in range(n_points-1, 2*n_points-1)])
        
        # Solve convex problem
        objective = cp.Minimize(max_obj)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            h_relaxed = h_vars.value
        except Exception:
            # If convex solver fails, use simple projection
            h_relaxed = h.copy()
        
        # Project back to feasible binary solution with exact sum
        h_new = project_to_binary(h_relaxed, target_sum)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new)
        
        if new_C5 < best_C5:
            best_h = h_new.copy()
            best_C5 = new_C5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Add perturbations for exploration
        perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for idx in perturb_indices:
            h[idx] = h_new[idx]
        
        # Small random mutations
        if np.random.rand() < 0.1:
            mutation_indices = np.random.choice(n_points, size=3, replace=False)
            for idx in mutation_indices:
                if h[idx] == 1.0:
                    h[idx] = 0.0
                else:
                    h[idx] = 1.0

    return (best_h.tolist(), best_C5, n_points)
