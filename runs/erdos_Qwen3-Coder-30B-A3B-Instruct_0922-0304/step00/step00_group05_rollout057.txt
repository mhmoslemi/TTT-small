import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a reasonable initial construction
        n_points = 70
        h = np.zeros(n_points)
        # Set first half to 1, second half to 0, then adjust
        h[:n_points//4] = 1.0
        h[n_points//4:3*n_points//4] = 0.5
        h[3*n_points//4:] = 0.0
        # Ensure sum = n_points/2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        h += (target_sum - current_sum) / n_points
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate(h, dx)
    
    # Parameters for simulated annealing
    T_max = 1.0
    T_min = 1e-4
    iterations = 10000
    alpha = np.exp(np.log(T_min / T_max) / iterations)
    T = T_max
    delta = 0.1
    
    start_time = time.time()
    for i in range(iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturbation: try both swap and shift
        if np.random.random() < 0.5:
            # Swap perturbation
            idx1, idx2 = np.random.randint(0, n_points, 2)
            if idx1 != idx2:
                h_new = h.copy()
                # Move mass from idx1 to idx2 (small amount)
                move_amount = delta * np.random.random()
                move_amount = min(move_amount, h[idx1], 1.0 - h[idx2])
                if move_amount > 0:
                    h_new[idx1] -= move_amount
                    h_new[idx2] += move_amount
        else:
            # Shift perturbation (adjacent transfer)
            idx = np.random.randint(0, n_points-1)
            h_new = h.copy()
            move_amount = delta * np.random.random()
            move_amount = min(move_amount, h[idx], 1.0 - h[idx+1])
            if move_amount > 0:
                h_new[idx] -= move_amount
                h_new[idx+1] += move_amount
                
        # Ensure constraints are maintained
        # Rescale to keep sum = n_points/2
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        scaling = target_sum / current_sum
        h_new *= scaling
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate
        C5_new = evaluate(h_new, dx)
        
        # Accept with probability
        delta_C = C5_new - best_C5
        accept = True
        if delta_C > 0 and T > 0:
            p_accept = np.exp(-delta_C / T)
            if np.random.random() > p_accept:
                accept = False
                
        if accept:
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
        
        # Adaptive temperature and delta
        T *= alpha
        if T < T_min:
            T = T_min
            
    # Final evaluation of best h
    C5_result = evaluate(best_h, dx)
    
    return (best_h.tolist(), float(C5_result), n_points)

def evaluate(h, dx):
    # Compute overlap integral
    # C5 = max_k int h(x)(1 - h(x+k)) dx
    # Equivalent to maximum correlation
    corr = correlate(h, 1-h, mode="full")
    max_corr = np.max(np.abs(corr))
    C5 = max_corr * dx
    return C5
