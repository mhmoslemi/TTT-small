import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Default construction: simple step function
        n_points = 100
        h0 = np.ones(n_points) * 0.5
        # Initialize with a simple piecewise constant function
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        # Rescale to meet constraint
        h0 *= n_points / (2 * np.sum(h0))
    
    dx = 2.0 / n_points
    h = h0.copy()
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    start_time = time.time()
    
    # Gradient-based optimization
    max_iter = 10000
    tol = 1e-6
    step_size = 0.01
    decay = 0.9999
    
    # Ensure constraints are met
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlations to find max overlap
        overlap = np.correlate(h, 1-h, mode="full")
        C5 = np.max(overlap) * dx
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Find the shift k that gives maximum overlap
        max_shift_idx = np.argmax(overlap)
        max_shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient of overlap w.r.t h for the active shift
        # For shift k, the overlap is sum of h[i] * (1 - h[i + k])
        grad = np.zeros_like(h)
        # Derivative of h[i] * (1 - h[i + k]) wrt h[i] is (1 - h[i + k])
        # For each i in range adjusted for shift
        if max_shift >= 0:
            for i in range(len(h) - max_shift):
                grad[i] += (1 - h[i + max_shift])
        else:
            for i in range(-max_shift, len(h)):
                grad[i] += (1 - h[i + max_shift])
        
        # Apply gradient descent update
        h -= step_size * grad
        # Project h to [0,1] bounds
        h = np.clip(h, 0, 1)
        
        # Renormalize h to satisfy integral constraint
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        # Adjust step size
        step_size *= decay
        
        # Check for convergence
        if np.allclose(grad, 0, atol=tol):
            break
    
    # Final evaluation
    final_C5 = evaluate_C5(best_h, dx)
    
    return (best_h, final_C5, n_points)

def evaluate_C5(h, dx):
    """Evaluate the C5 constant for given h values."""
    overlap = np.correlate(h, 1-h, mode="full")
    return np.max(overlap) * dx
