import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = np.clip(h0, 0, 1)
        n_points = len(h0)
        dx = 2.0 / n_points
        # Repair sum constraint
        current_sum = np.sum(h0) * dx
        target_sum = 1.0
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust by scaling or flipping
            diff = target_sum - current_sum
            if diff > 0:
                # Add values to smaller entries
                indices = np.argsort(h0)
                for i in range(int(diff/dx)):
                    if indices[i] < n_points:
                        h0[indices[i]] = min(1.0, h0[indices[i]] + dx)
            else:
                # Remove from larger entries
                indices = np.argsort(h0)[::-1]
                for i in range(int(-diff/dx)):
                    if indices[i] < n_points:
                        h0[indices[i]] = max(0.0, h0[indices[i]] - dx)
        # Ensure exact sum
        h0 = h0 * (target_sum / np.sum(h0) * dx)
    else:
        # Default construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        dx = 2.0 / n_points
    
    # Binary search setup
    t_low = 0.0
    t_high = 0.5
    tolerance = 1e-4
    
    best_h = None
    best_c5 = float('inf')
    
    def evaluate_h(h):
        # Use signal correlate for efficiency instead of manual loop
        corr = correlate(h, 1 - h, mode="full")
        max_val = np.max(corr) * dx
        return max_val
    
    def solve_feasibility(t_val):
        """
        Solve feasibility problem:
        min 0 s.t. sum(h) = n_points / 2,  h[i] in [0,1], 
        sum(h[i] * (1 - h[i + k])) <= t_val for all valid k
        """
        h = cp.Variable(n_points)
        # Variables w[i, k] = h[i] * h[i + k] where k is shift
        # However, directly linearizing is complex. Instead we solve via relaxation.
        # We split each constraint: sum over shift k that intersecting range
        constraints = [cp.sum(h) == n_points/2.0]
        for k in range(-n_points+1, n_points):  # all shifts
            if k >= 0:
                # shift forward: h[i+k] index at i+k, origin at i
                # this is a dot product of h[i] and (1-h[i+k])
                # sum_{i=0}^{n-k-1} h[i]*(1-h[i+k])
                dots = 0
                for i in range(n_points - k):
                    dots += h[i] * (1 - h[i + k])
            else:
                # shift backward
                dots = 0
                for i in range(-k, n_points):
                    dots += h[i] * (1 - h[i + k])
            # Add constraint: dots <= t
            constraints.append(dots <= t_val)
        
        # Box constraints
        constraints += [h >= 0, h <= 1]
        
        # Problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INACCURATE
        except Exception:
            return False
    
    # Binary search for optimal t
    while t_high - t_low > tolerance and time.time() - start_time < budget_s - 1:
        t_mid = (t_low + t_high) / 2.0
        if solve_feasibility(t_mid):
            t_high = t_mid
            # At this point we know critical value, try to find a feasible solution
            h_sol = cp.Variable(n_points)
            constraints = [cp.sum(h_sol) == n_points/2.0]
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    dots = cp.sum([h_sol[i] * (1 - h_sol[i + k]) for i in range(n_points - k)])
                else:
                    dots = cp.sum([h_sol[i] * (1 - h_sol[i + k]) for i in range(-k, n_points)])
                constraints.append(dots <= t_mid)
            constraints += [h_sol >= 0, h_sol <= 1]
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=cp.GLPK, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_feasible = h_sol.value
                    c5_val = evaluate_h(h_feasible)
                    if c5_val < best_c5:
                        best_h = h_feasible.copy()
                        best_c5 = c5_val
            except Exception:
                pass
        else:
            t_low = t_mid
    
    # Final evaluation on best solution found
    if best_h is not None:
        final_h = best_h.copy()
        final_c5 = best_c5
    else:
        # Reconstruct with optimal t from binary search
        final_t = t_high
        h_final = cp.Variable(n_points)
        constraints = [cp.sum(h_final) == n_points/2.0]
        for k in range(-n_points+1, n_points):
            if k >= 0:
                dots = cp.sum([h_final[i] * (1 - h_final[i + k]) for i in range(n_points - k)])
            else:
                dots = cp.sum([h_final[i] * (1 - h_final[i + k]) for i in range(-k, n_points)])
            constraints.append(dots <= final_t)
        constraints += [h_final >= 0, h_final <= 1]
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            final_h = h_final.value
            final_c5 = evaluate_h(final_h)
        except:
            final_h = h0
            final_c5 = evaluate_h(final_h)

    # Ensure final_h is in valid range and sum
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (1.0 / np.sum(final_h) * dx)
    
    return (final_h, final_c5, n_points)
