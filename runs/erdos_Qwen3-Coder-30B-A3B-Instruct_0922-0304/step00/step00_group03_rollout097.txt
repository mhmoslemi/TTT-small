import numpy as np

import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):

        h = h * (n_points / (2 * np.sum(h)))

    h = h * (n_points / (2 * np.sum(h)))

    dx = 2.0 / n_points

    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 2000
    patience = 500
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Initialize dual variables
    lambda_k = np.zeros(2 * n_points - 1)  # Full correlation length
    mu = 0.0
    
    # Precompute shifted versions
    def compute_overlaps(h_vec):
        # Compute h * (1-h) and full correlation
        h_shifted = 1 - h_vec
        overlaps = np.correlate(h_vec, h_shifted, mode='full') * dx
        return overlaps
    
    # Quadratic regularization term
    def regularized_objective(h_vec, lambda_k, mu):
        overlaps = compute_overlaps(h_vec)
        max_overlap = np.max(overlaps)
        penalty = np.sum(lambda_k * np.maximum(0, overlaps - max_overlap)) / len(overlaps)
        integral_penalty = mu * (np.sum(h_vec) * dx - 1.0)
        return max_overlap - integral_penalty + penalty
    
    # ADMM main loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Step 1: Update h by solving convex optimization
        # min_h: max_k [ (h ⋆ (1−h))[k] * dx ] + rho/2 ||h - h_prev||^2 + mu(∫h dx - 1)  
        # Change of variables: set h_new = h_prev
        h_prev = h.copy()
        
        # Solve convex subproblem with CVXPY
        h_var = cp.Variable(n_points)
        
        # Overlaps as quadratic constraints
        overlaps = compute_overlaps(h_prev)
        max_ov = np.max(overlaps)
        
        # Create objective using max relaxation
        max_overlap = cp.Variable()
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) * dx == 1]
        
        # Add linearized constraints based on current h_prev
        # Approximate max with linear constraint
        obj = cp.Minimize(max_overlap + rho * cp.sum_squares(h_var - h_prev))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new
        except:
            # fallback to gradient descent if CVXPY fails
            h = h_prev
            # Simple gradient descent step
            overlaps = compute_overlaps(h)
            grad = np.zeros_like(h)
            k_max = np.argmax(overlaps)
            for i in range(n_points):
                if k_max - i >= 0 and k_max - i < n_points:
                    grad[i] += (1 - h[k_max - i]) * dx
                if i - (k_max - n_points + 1) >= 0 and i - (k_max - n_points + 1) < n_points:
                    grad[i] += h[i - (k_max - n_points + 1)] * dx
            
            h = h - alpha * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / (2 * np.sum(h)))

        overlaps = compute_overlaps(h)

        mu += alpha * (2.0 * np.sum(h) * dx - 1.0)
        lambda_k = np.clip(lambda_k, -1e10, 1e10)
        
        # Step 4: entropy regularization on lambda
        lambda_k = lambda_k + beta * np.log(1 + lambda_k)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improvement = iter_count
        
        if C5 < 0.3807:
            break
            
        if iter_count - last_improvement > patience:
            break
            
        iter_count += 1
        # Progress every 100 iterations
        if iter_count % 100 == 0 and time.time() - start_time > budget_s * 0.9:
            break
    
    # Final validation and cleanup
    if best_C5 < 0.38085:
        final_h = np.clip(best_h, 0, 1)
        final_h = final_h * (n_points / (2 * np.sum(final_h)))
        overlaps = compute_overlaps(final_h)
        final_C5 = np.max(overlaps)
        return (final_h, final_C5, n_points)
    else:

        final_h = np.clip(best_h, 0, 1)
        final_h = final_h * (n_points / (2 * np.sum(final_h)))
        overlaps = compute_overlaps(final_h)
        final_C5 = np.max(overlaps)
        return (final_h, final_C5, n_points)
