import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.zeros(n_points)
        h[:int(target_sum)] = 1.0
        
    # Ensure sum constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    current_C5 = evaluate_erdos_solution(h)
    best_C5 = current_C5
    
    # Main optimization loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iterations += 1
        current_C5 = evaluate_erdos_solution(h)
        
        if current_C5 < best_C5 - 1e-10:
            best_C5 = current_C5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Compute overlaps as needed for relaxation
        overlap_vals = correlate(h, 1-h, mode='full')
        overlap_vals *= dx
        max_overlap = np.max(overlap_vals)
        
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == target_sum
        ]
        
        # Objective: minimize max overlap
        max_var = cp.Variable()
        constraints += [max_var >= overlap_vals[i] for i in range(len(overlap_vals))]
        objective = cp.Minimize(max_var)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
            else:
                # Fallback to previous if optimization fails
                h_new = h.copy()
        except Exception:
            h_new = h.copy()
            
        # Project back to feasible binary-like solution
        # Threshold and adjust sum
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        
        if abs(current_sum - target_sum) > 1e-6:
            # Apply simple adjustment
            diff = target_sum - current_sum
            if diff > 0:
                # Increase some values
                indices = np.argsort(h_new)
                for i in range(int(diff)):
                    if i < len(indices):
                        h_new[indices[i]] = min(1.0, h_new[indices[i]] + 1e-6)
            else:
                # Decrease some values
                indices = np.argsort(h_new)[::-1]
                for i in range(int(-diff)):
                    if i < len(indices):
                        h_new[indices[i]] = max(0.0, h_new[indices[i]] - 1e-6)
        
        # Final adjustment to ensure sum exactly
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Perturb randomly for exploration
        perturb_indices = np.random.choice(n_points, size=int(n_points*0.05), replace=False)
        for i in perturb_indices:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, 0.01), 0, 1)
            
        h = h_new.copy()
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), float(final_C5), n_points)

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1-h, mode="full") * dx
    return np.max(overlaps)
