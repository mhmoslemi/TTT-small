import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    dx = 2.0 / n_points
    
    # Ensure sum(h) * dx = 1, i.e. sum(h) = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute overlap integral
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    def perturb_h(h_vec, delta=0.1):
        # Randomly perturb some elements within [0, 1]
        h_new = h_vec.copy()
        num_perturb = max(1, int(0.05 * len(h_vec)))
        indices = np.random.choice(len(h_vec), size=num_perturb, replace=False)
        for i in indices:
            perturbation = np.random.uniform(-delta, delta)
            h_new[i] += perturbation
            h_new[i] = max(0, min(1, h_new[i]))
        # Renormalize to maintain sum(h) = n_points / 2
        h_new *= (n_points / 2) / np.sum(h_new)
        return h_new
    
    best_h = h.copy()
    c5_bound = compute_c5(h)
    best_c5 = c5_bound
    start_time = time.time()
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    delta = 0.1
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        h_new = perturb_h(h, delta)
        c5_new = compute_c5(h_new)
        
        delta_c5 = c5_new - c5_bound
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            c5_bound = c5_new
        
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        
        # Cool temperature
        T *= cooling_rate
        
        # Early stopping
        if best_c5 < 0.3807:
            break
        
        # Speed up convergence: decrease delta slightly
        delta = max(0.001, delta * 0.9999)
    
    return (best_h.tolist(), best_c5, n_points)
