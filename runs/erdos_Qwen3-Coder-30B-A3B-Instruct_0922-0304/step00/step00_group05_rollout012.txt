import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h *= (n_points / 2.0) / np.sum(h)
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    learning_rate = 0.01
    gamma_patience = 100
    patience_counter = 0
    
    max_iterations = int(budget_s * 1000)
    iteration = 0
    
    while iteration < max_iterations:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all indices where max overlap occurs
        max_lags = np.where(overlap == max_overlap)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for m in max_lags:
            if m < n_points:
                # overlapping part
                valid = np.arange(0, n_points - m)
                gradient[valid] += (1 - h[valid + m]) * dx
            else:
                # overlapping part on the end
                valid = np.arange(m - n_points + 1, n_points)
                gradient[valid] += (1 - h[valid + m - n_points + 1]) * dx
                
        gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h to [0, 1] and rescale
        h = np.clip(h, 0, 1)
        h *= (n_points / 2.0) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Adaptive learning rate
        if patience_counter >= gamma_patience:
            learning_rate *= 0.1
            patience_counter = 0
            
        iteration += 1
    
    return (best_h, best_c5, n_points)
