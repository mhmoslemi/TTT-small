import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 47
    dx = 2.0 / n_points
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    max_iter = 1000
    patience = 100
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:
        h = np.full(n_points, 0.5)
    
    # Rescale to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_overlap(h, k):
        # Compute overlap for shift k
        h_shifted = np.concatenate([np.zeros(k), h[:-k]]) if k < len(h) else np.zeros(len(h))
        return np.sum(h * (1 - h_shifted)) * dx
    
    def compute_all_overlaps(h):
        # Use np.correlate to compute overlaps for all k
        h_rev = h[::-1]
        corr = np.correlate(h, 1 - h_rev, mode='full')
        overlaps = corr[len(h) - 1:] * dx
        return overlaps
    
    def compute_gradient(h, k_max):
        # Compute gradient of overlap with respect to h for shift k
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # First term
            if i + k_max < len(h):
                grad[i] += 1 - h[i + k_max]
            else:
                grad[i] += 0
            # Second term
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]
            else:
                grad[i] -= 0
        return grad
    
    def line_search(h, grad, k_max, alpha_max=1.0):
        # Backtracking line search for step size
        alpha = alpha_max
        current_overlap = compute_overlap(h, k_max)
        
        for _ in range(30):  # max backtrack steps
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet constraint sum(h) = n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_overlap = compute_overlap(h_new, k_max)
            
            if new_overlap <= current_overlap - eta * alpha * np.dot(grad, grad):
                return h_new, alpha
            alpha *= rho
        return h, alpha
    
    iteration = 0
    no_improvement_count = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        overlaps = compute_all_overlaps(h)
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max]
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_c5 < 0.3807 or no_improvement_count > patience:
            break
        
        # Gradient of overlap for k_max
        grad = compute_gradient(h, k_max)
        grad_norm = np.linalg.norm(grad)
        
        if grad_norm == 0:
            break
            
        # Normalize gradient
        grad = grad / grad_norm
        
        # Line search
        h_new, step_size = line_search(h, grad, k_max)
        
        if step_size == 0:
            break
            
        h = h_new
        iteration += 1
    
    # Final evaluation
    final_overlaps = compute_all_overlaps(best_h)
    final_c5 = np.max(final_overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
