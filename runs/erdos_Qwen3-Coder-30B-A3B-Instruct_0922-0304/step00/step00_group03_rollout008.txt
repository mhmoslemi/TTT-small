import numpy as np
from scipy import correlate
import time
from copy import deepcopy

def evaluate_C5(h, dx):
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The correlation is computed with the flipped version of h 
    # which gives us the correct overlap integral
    return np.max(corr) * dx

def generate_initial_h(n_points):
    # Start with a simple binary construction: 0s followed by 1s
    # This ensures sum(h) = n_points / 2
    h = np.zeros(n_points)
    h[:n_points//2] = 1
    return h

def tabu_search(initial_h, n_points, dx, max_iter=1000, tabu_tenure=100, budget_s=600):
    # Initialize tabu list as a set of tuples (config, iteration)
    tabu_list = set()
    current_h = deepcopy(initial_h)
    best_h = deepcopy(current_h)
    best_C5 = evaluate_C5(current_h, dx)
    
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break

        # Generate candidates by flipping 2 or 3 bits
        candidates = []
        for i in range(n_points):
            for j in range(i+1, n_points):
                # Flip two bits
                candidate = deepcopy(current_h)
                candidate[i] = 1 - candidate[i]
                candidate[j] = 1 - candidate[j]
                if np.sum(candidate) == n_points / 2:
                    candidates.append((candidate, (i, j)))
                # Try flipping 3 bits if possible
                for k in range(j+1, n_points):
                    candidate = deepcopy(current_h)
                    candidate[i] = 1 - candidate[i]
                    candidate[j] = 1 - candidate[j]
                    candidate[k] = 1 - candidate[k]
                    if np.sum(candidate) == n_points / 2:
                        candidates.append((candidate, (i, j, k)))

        # Evaluate all candidates
        best_candidate = None
        best_candidate_C5 = np.inf
        for candidate_h, flip_indices in candidates:
            # Check if candidate is in tabu list
            if tuple(candidate_h.astype(int)) in tabu_list:
                continue

            candidate_C5 = evaluate_C5(candidate_h, dx)
            if candidate_C5 < best_candidate_C5:
                best_candidate = candidate_h
                best_candidate_C5 = candidate_C5

        # If no improvement found, break early
        if best_candidate_C5 >= best_C5:
            break

        # Update best solution if needed
        if best_candidate_C5 < best_C5:
            best_h = best_candidate
            best_C5 = best_candidate_C5

        # Move to new solution
        current_h = best_candidate

        # Update tabu list
        tabu_list.add(tuple(current_h.astype(int)))
        if len(tabu_list) > tabu_tenure:
            # Remove oldest entries
            tabu_list.pop()

    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70  # Use the same number as provided by parent construction
    dx = 2.0 / n_points

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        # Ensure exact sum to n/2
        target_sum = n_points / 2
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-5:
            # Adjust values to meet sum constraint
            sorted_indices = np.argsort(h)[::-1]
            for i in sorted_indices:
                if current_sum > target_sum:
                    if h[i] > 0:
                        dec = min(h[i], current_sum - target_sum)
                        h[i] -= dec
                        current_sum -= dec
                        if current_sum <= target_sum:
                            break
                else:
                    if h[i] < 1:
                        inc = min(1 - h[i], target_sum - current_sum)
                        h[i] += inc
                        current_sum += inc
                        if current_sum >= target_sum:
                            break
    else:
        h = generate_initial_h(n_points)
    
    # Use tabu search to optimize
    best_h, best_C5 = tabu_search(h, n_points, dx, budget_s=budget_s)
    
    return (best_h.tolist(), best_C5, n_points)
