import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h."""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Take maximum over all lags (excluding zero lag which is the full correlation)
    max_corr = np.max(corr)
    return max_corr * dx

def normalize_h(h, n_points):
    """Ensure integral of h equals 1 (sum(h) * dx = 1)."""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    target_sum = n_points / 2.0  # Because dx = 2/n_points and integral should be 1
    scale = target_sum / current_sum
    return h * scale

def perturb_h(h, fraction=0.05, delta=0.1):
    """Perturb a fraction of the entries in h."""
    n_points = len(h)
    n_perturb = max(1, int(n_points * fraction))
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    new_h = h.copy()
    for i in indices:
        delta_change = np.random.uniform(-delta, delta)
        new_h[i] += delta_change
        new_h[i] = np.clip(new_h[i], 0, 1)
    return new_h

    best_h = h.copy()
    current_C5 = evaluate_h(h, dx)
    best_C5 = current_C5

    # Annealing parameters
    T = 0.1  # Initial temperature
    cooling_rate = 0.995
    min_T = 1e-6
    max_iterations = int(budget_s * 1000)  # Rough estimation of iterations
    iteration = 0
    start_time = time.time()
    
    # Initialize random perturbation parameter
    delta = 0.1
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s:
        # Perturb h
        new_h = perturb_h(h, fraction=0.05, delta=delta)
        # Normalize
        new_h = normalize_h(new_h, n_points)
        
        # Compute new C5
        new_C5 = evaluate_h(new_h, dx)
        
        # Accept or reject the new configuration
        delta_E = new_C5 - current_C5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):

            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Update temperature
        T = max(min_T, T * cooling_rate)
        
        # Adjust delta adaptively based on current temperature
        if T < 0.01:
            delta = 0.01
        else:
            delta = 0.1
