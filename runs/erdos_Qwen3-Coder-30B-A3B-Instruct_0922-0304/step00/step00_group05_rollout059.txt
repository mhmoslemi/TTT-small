import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Start with a balanced construction
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[n_points//4:3*n_points//4] = 0.5
        h[3*n_points//4:] = 0.0
        # Make sure sum equals target_sum
        current_sum = np.sum(h)
        if current_sum > target_sum:
            # Reduce values from the middle
            delta = current_sum - target_sum
            h[n_points//4:3*n_points//4] = np.maximum(0, h[n_points//4:3*n_points//4] - delta/(n_points//2))
        elif current_sum < target_sum:
            delta = target_sum - current_sum
            h[n_points//4:3*n_points//4] = np.minimum(1, h[n_points//4:3*n_points//4] + delta/(n_points//2))
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Rescale if needed
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    
    # Precompute
    def compute_c5(h_array):
        # Use scipy signal correlation
        corr = signal.correlate(h_array, 1 - h_array, mode='full')
        max_corr = np.max(np.abs(corr)) * dx
        return max_corr
    
    # Initialize
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    total_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / total_iter)
    
    # Perturbation parameters
    delta_scale = 0.1
    accept_count = 0
    accept_threshold = 50
    
    start_time = time.time()
    for i in range(total_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new candidate
        new_h = h.copy()
        
        # Choose perturbation type and apply
        if random.random() < 0.5:
            # Swap two random positions
            i1 = random.randint(0, n_points - 1)
            i2 = random.randint(0, n_points - 1)
            new_h[i1], new_h[i2] = new_h[i2], new_h[i1]
        else:
            # Adjacent shift
            i1 = random.randint(0, n_points - 1)
            i2 = (i1 + 1) % n_points
            # Calculate shift amount
            shift_amount = random.uniform(-delta_scale, delta_scale)
            new_h[i1] += shift_amount
            new_h[i2] -= shift_amount
        
        # Ensure bounds
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to maintain sum exactly
        current_sum = np.sum(new_h)
        if current_sum > 0:
            new_h = new_h * target_sum / current_sum
        
        # Compute new C5
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        T = T_max * (alpha ** i)
        
        if delta_c5 < 0 or random.random() < np.exp(-delta_c5 / T):
            h = new_h.copy()
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
            accept_count += 1
        else:
            accept_count = max(0, accept_count - 1)
            
        # Adaptive: adjust delta_scale based on acceptance rate
        if i > 0 and i % accept_threshold == 0:
            if accept_count < accept_threshold // 2:
                delta_scale *= 0.9
            elif accept_count > accept_threshold * 3 // 4:
                delta_scale = min(1.0, delta_scale * 1.1)
            accept_count = 0
    
    return (best_h, best_c5, n_points)
