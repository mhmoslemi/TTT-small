def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h"""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    max_overlap = 0
    for k in range(n):
        shifted_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shifted_h)) * dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap
