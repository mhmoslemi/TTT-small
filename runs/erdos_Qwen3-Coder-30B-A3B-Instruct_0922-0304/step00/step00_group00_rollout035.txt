import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n = len(h0)
    except NameError:
        # Fallback: create initial step function
        n = 45
        h0 = np.ones(n) * 0.5
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n / 2) / np.sum(h0)
    
    # Initialize
    h = h0.copy()
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    max_iter = 5000
    iter_no_improve = 0
    max_no_improve = 50

    # Start optimizing
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute all overlaps
        full_correlation = np.correlate(h, 1 - h, mode='full')
        overlaps = full_correlation[len(h)-1:]
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # If already good enough - return
        if max_overlap < 0.3807:
            break
        
        # Set up optimization problem
        h_var = cp.Variable(n)
        C5_var = cp.Variable()
        
        # Objective: minimize C5
        obj = cp.Minimize(C5_var)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n / 2,
            # Overlap constraint
            cp.sum(h_var * np.roll(1 - h_var, k_max)) <= C5_var
        ]
        
        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new
                    new_C5 = max_overlap
                    if new_C5 < best_C5:
                        best_C5 = new_C5
                        best_h = h.copy()
                        iter_no_improve = 0
                    else:
                        iter_no_improve += 1
                else:
                    iter_no_improve += 1
            else:
                iter_no_improve += 1
        except:
            iter_no_improve += 1
            
        if iter_no_improve >= max_no_improve:
            break

    # Final refinement
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_C5, len(best_h))
