        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == target_sum]
        
        # For each shift k, add constraint that integral <= t_mid
        for k in range(-n_points + 1, n_points):
            # Indices for h[i] and h[i+k], considering wraparound in correlation
            i_indices = np.arange(n_points)
            j_indices = i_indices + k
            valid_indices = (j_indices >= 0) & (j_indices < n_points)
            
            if not np.any(valid_indices):
                continue

            j_valid = j_indices[valid_indices]
            
            # This constraint says sum(h[i] * (1-h[i+k])) <= t_mid
            # Let w[i] = h[i] * h[i+k] for i with valid j
            w = cp.Variable(len(i_valid))
            # w[i] = h[i] * h[j]
            # upper bounds on products
            for i, j in zip(i_valid, j_valid):
                constraints.append(w[i - i_valid[0]] <= h[i])
                constraints.append(w[i - i_valid[0]] <= h[j])
            # This enforces sum(w) >= sum(h[i]) - t_mid (i.e., sum(h[i]*1) - sum(h[i]*h[i+k]) <= t_mid)
            constraints.append(cp.sum(w) >= cp.sum(h[i_valid]) - t_mid)
            
        prob = cp.Problem(cp.Minimize(0), constraints)

                t_low = t_mid

            t_low = t_mid

            reduce_amount = min(1.0, diff)

            increase_amount = min(1.0 - h_rounded[i], diff)

        to_adjust = np.argsort(h_rounded)

            while adjust_i < len(to_adjust) and diff > 1e-6:

                diff -= to_reduce
                adjust_i += 1

            # Increase just enough
            while adjust_i < len(to_adjust) and diff < -1e-6:

                to_add = min(1.0 - h_rounded[idx], -diff)
                h_rounded[idx] += to_add
                diff += to_add
                adjust_i += 1

    # One final validation
    final_eval = evaluate_erdos_solution(h_rounded, n_points)
    return (h_rounded.tolist(), final_eval, n_points)

def evaluate_erdos_solution(h_values, n_points):
    """Return maximum overlap for the given h over all shifts k"""
    h = np.array(h_values)
    dx = 2.0 / n_points
    # np.correlate handles circular shifts
    # 1-h is the flipped h 
    corr = np.correlate(h, 1 - h, mode="full")
    # max overlap over all shifts
    return np.max(corr) * dx
