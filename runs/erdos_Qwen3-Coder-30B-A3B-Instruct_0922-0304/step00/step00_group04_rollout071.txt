def evaluate_erdos_solution(h_vals, n_points):
    dx = 2.0 / n_points
    # compute all correlations for all lags
    corr = np.correlate(h_vals, 1 - h_vals, mode="full")
    # take the maximum value over all k (the lags)
    # the "full" mode gives us 2*n - 1 correlations, centered about the middle
    # but we only care about the actual causal lags, which is from n-1 to n-1
    # so we take elements from index n-1 to 2*n-2 (inclusive)
    max_corr = np.max(corr[n_points-1:])
    return max_corr * dx
