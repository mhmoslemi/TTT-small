import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    
    # Normalize to sum to 1
    h *= (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vals):
        # Compute overlap integral for all shifts
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Take only the relevant part (shifts k from 0 to n_points-1)
        c5_val = np.max(corr[len(corr)//2:])
        return c5_val * dx
    
    def project_feasible(h_vals):
        # Ensure all values are in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Renormalize
        h_vals *= (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def perturb_h(h_vals, frac=0.05, delta=0.1):
        # Select indices to perturb
        n_perturb = max(1, int(frac * len(h_vals)))
        indices = np.random.choice(len(h_vals), size=n_perturb, replace=False)
        # Make random perturbations
        new_h = h_vals.copy()
        for i in indices:
            # Random change with clip
            change = np.random.uniform(-delta, delta)
            new_h[i] += change
        # Project to feasible region
        new_h = project_feasible(new_h)
        return new_h
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    best_c5_history = [best_c5]
    
    # Annealing settings
    temp = 0.1
    cooling_rate = 0.995
    min_temp = 1e-6
    max_iter = int(budget_s * 1000)  # ~1000 iterations per second
    iter_count = 0
    last_improvement = 0
    patience = 10000
    
    start_time = time.time()
    
    while iter_count < max_iter:
        # Check timeout
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb
        new_h = perturb_h(best_h, frac=0.05, delta=0.1)
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        delta_c5 = new_c5 - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            best_h = new_h
            best_c5 = new_c5
            best_c5_history.append(best_c5)
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Cool temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Early exit if target met
        if best_c5 < 0.3807:
            break
            
        # Early exit if no improvement for a while
        if last_improvement > patience:
            break
            
        iter_count += 1

    return (best_h.tolist(), best_c5, n_points)
