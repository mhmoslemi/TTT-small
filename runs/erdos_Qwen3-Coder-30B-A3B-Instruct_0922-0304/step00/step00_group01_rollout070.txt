import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps for shifts 0 to n-1
    overlaps = np.correlate(h_values, 1 - h_values, mode="full")
    # Extract the n overlapping parts (shifts 0 to n-1)  
    # The full correlation has 2*n-1 elements, we want middle n elements
    overlaps = overlaps[n - 1:]
    # Normalize by dx
    overlaps *= dx
    return np.max(overlaps)
