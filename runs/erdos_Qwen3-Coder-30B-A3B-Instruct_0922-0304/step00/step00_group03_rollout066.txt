import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
    else:
        h = np.ones(n_points) * 0.5
        # Ensure sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Constraints checking
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    learning_rate = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute all correlations
        full_corr = correlate(h, 1 - h, mode="full")
        max_corr_idx = np.argmax(full_corr)
        max_corr = full_corr[max_corr_idx]
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(-n_points + 1, n_points):
                if 0 <= i + k < n_points:
                    grad[i] += (1 - h[i + k])
        
        # Apply velocity
        velocity = momentum * velocity - learning_rate * grad
        h_new = h + velocity
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        h = h_new
        
        # Evaluate C5
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Adapt learning rate
        if iteration % 100 == 0:
            learning_rate *= 0.9
        
        # Termination condition
        if best_c5 <= target_bound:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx
