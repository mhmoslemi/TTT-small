import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default: uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h
    h = h * target_sum / np.sum(h)
    
    # Projected gradient descent
    max_iter = 10000
    gamma = 0.01
    gamma_decay = 0.1
    patience = 100
    patience_counter = 0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        # h(x) * (1 - h(x+k))
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        ones_extended = np.concatenate([np.ones(n_points), np.zeros(n_points - 1)])
        conv = correlate(h_extended, ones_extended - h, mode='full')
        c_values = conv[:2 * n_points - 1] * dx
        
        # Find maximum overlap and its indices
        max_c5 = np.max(c_values)
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            gamma *= gamma_decay
            patience_counter = 0
            
        # Compute gradient
        grad = np.zeros_like(h)
        max_indices = np.where(np.abs(c_values - max_c5) < 1e-10)[0]
        
        # For each valid lag, accumulate gradient contribution
        for idx in max_indices:
            if idx < n_points:
                # h[i] contributes to overlap at lag idx
                # ∂C/∂h[i] = (1 - h[i + idx]) 
                # But we don't want to index out of bounds,
                # so we compute partial derivatives correctly
                if idx == 0:
                    # Just h[i]*1 case, all h[i] contribute 1
                    grad += 1.0
                else:
                    # h[i] * (1 - h[i+idx]) -> ∂/∂h[i] = 1 - h[i+idx], 
                    # but we also compute ∂/∂h[i+idx] = -h[i]
                    # We treat h[i+idx] as fixed for this term, so:
                    # ∂C/∂h[i] = (1 - h[i+idx]) where i+idx is valid
                    if i + idx < n_points:
                        grad[i] += 1.0 - h[i + idx]
                    # We want to compute penalty gradient on h[i]'s contribution
                    # BUT h[i] can't contribute to overlap where index doesn't exist
                    pass
                
        # Just use simplified direction: sum over all max overlap positions
        grad = np.zeros_like(h)
        for idx in max_indices:
            if idx < n_points:
                # Delayed partial derivative approach:
                if idx == 0:
                    grad += 1.0
                elif idx <= n_points - 1:
                    for i in range(n_points):
                        if i + idx < n_points:
                            grad[i] += 1.0 - h[i + idx]
        
        # Gradient ascent (we want to minimize C5)
        h = h - gamma * grad
        
        # Project
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
    return (best_h.tolist(), float(best_c5), n_points)
