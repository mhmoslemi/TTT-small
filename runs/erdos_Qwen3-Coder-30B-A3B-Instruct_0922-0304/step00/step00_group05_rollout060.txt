import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for the given step function h
    # This returns the maximum overlap value C_5
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available
    h = initial_h_values.copy()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000
    
    # Track improvement
    last_improvement = 0
    iteration = 0
    start_time = time.time()
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s:
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iteration
            
        # Compute all overlaps to find max
        overlaps = correlate(h, 1 - h, mode="full") * dx
        
        # Find lags corresponding to maximum overlap
        max_overlap = np.max(overlaps)
        max_lags = np.where(np.abs(overlaps - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # i-th element from h contributes to (i + lag)-th overlap
                # So for the overlap at m=lag, we look at all i such that i + lag < n_points
                valid_indices = np.arange(n_points - lag)
                gradient[valid_indices] += (1 - h[valid_indices + lag]) * dx
            else:
                # This happens when lag > n_points - 1; we can ignore these
                pass
        
        # Normalize gradient by number of max lags
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h onto feasible region [0,1] and enforce sum = n_points / 2
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= (n_points / 2) / current_sum
        
        # Adapt learning rate based on improvement
        if iteration - last_improvement >= gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
        
        iteration += 1
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h, final_c5, n_points)
