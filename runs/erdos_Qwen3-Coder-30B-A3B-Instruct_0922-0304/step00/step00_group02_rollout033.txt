import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h with the provided solution if available, otherwise random binary
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
    else:
        h = np.zeros(n_points)
        indices = np.random.choice(n_points, int(target_sum), replace=False)
        h[indices] = 1.0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Early stopping condition
    consecutive_no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        # Try to improve the current solution
        h_backup = h.copy()
        
        # Binary approximation step - try to set each position to 1 if it helps
        for i in range(n_points):
            if h[i] < 0.5:  # Only consider changing from 0 to 1
                h_test = h.copy()
                h_test[i] = 1.0
                # Check if activating this position improves anything
                c5_test = evaluate_c5(h_test, dx)
                if c5_test < best_c5:
                    h = h_test
                    best_c5 = c5_test
                    if best_c5 < 0.3807:
                        return (best_h.tolist(), best_c5, n_points)
                    consecutive_no_improvement = 0
                elif c5_test < best_c5 + 1e-10:  # Small improvement
                    consecutive_no_improvement = 0
                else:
                    consecutive_no_improvement += 1
                        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            scale_factor = target_sum / current_sum
            h = np.clip(h * scale_factor, 0, 1)
        
        # Check if current solution is improved
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Early stopping if no improvement
        if consecutive_no_improvement >= max_no_improvement:
            break
        
        # Reinitialize if stuck with a bad solution
        if current_c5 > 0.45 or np.all(h == 0.5):
            h = np.zeros(n_points)
            indices = np.random.choice(n_points, int(target_sum), replace=False)
            h[indices] = 1.0
            consecutive_no_improvement = 0
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Using full cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # We want maximal overlap
    max_overlap = np.max(corr) * dx
    return max_overlap
