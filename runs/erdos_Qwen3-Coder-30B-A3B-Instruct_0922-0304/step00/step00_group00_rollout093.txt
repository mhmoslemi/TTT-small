import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    
    # Initialize h from initial_h_values if available, else create a reasonable starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = 50000
    patience = 50
    patience_count = 0
    
    # Use a coarse optimization loop to explore
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            shift = k if k >= 0 else n_points + k
            overlap = np.sum(h * np.roll(1 - h, shift))
            overlaps[k + n_points - 1] = overlap
        
        # Find maximum overlap
        max_k = np.argmax(overlaps)
        k_shift = max_k - n_points + 1
        
        # Compute gradient for the worst shift
        grad = np.zeros(n_points)
        if k_shift >= 0:
            grad[:-k_shift] += (1 - h[k_shift:])
            grad[k_shift:] -= h[:-k_shift]
        else:
            grad[-k_shift:] += (1 - h[:k_shift])
            grad[:k_shift] -= h[-k_shift:]
            
        # Update h using gradient descent
        h_new = h - alpha * grad * (2.0 / n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        h = h_new.copy()
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Adaptive learning rate
        if patience_count > patience:
            alpha *= 0.9
            patience_count = 0
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    max_overlap = 0
    
    # Compute overlap for all possible shifts
    for k in range(-n + 1, n):
        shift = k if k >= 0 else n + k
        rolled_h = np.roll(h, shift)
        overlap = np.sum(h * (1 - rolled_h))
        max_overlap = max(max_overlap, overlap)
        
    return max_overlap * dx
