import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps for all shifts
    overlap = correlate(h_values, 1 - h_values, mode="full")
    # The overlap values are for shifts from -(n-1) to (n-1)
    # We're interested in shifts from 1 to n-1 (positive shifts)
    # The maximum overlap is the maximum value in the middle part excluding the first n-1 elements
    max_overlap = np.max(overlap[n-1:])
    return max_overlap * dx

        g = np.zeros(n_points)

            # shift by max_k: h[(i - max_k) % n] 
            j = (i - max_k) % n_points
            g[i] = (1 - h[j]) - h[i] + lambda_
        
        # Update h
        h = h - alpha * g
        # Project h to [0,1]

        lambda_ = lambda_ + alpha * (np.sum(h) - n_points / 2)

            no_improve_count = 0
