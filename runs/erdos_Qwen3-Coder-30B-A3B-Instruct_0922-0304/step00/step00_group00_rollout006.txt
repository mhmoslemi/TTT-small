import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
        # Normalize to sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
    
    # Discretization step
    dx = 2.0 / n_points
    
    # Inequality constraints and linearization
    def compute_c5(h_vec):
        # Compute the full correlation for all shifts with 1-h
        conv_result = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take the maximum (excluding the 0 shift, which is not valid)
        max_overlap = np.max(conv_result[1:]) * dx
        return max_overlap

    def add_linear_constraints(model, h_vars, n, dx):
        # Add McCormick relaxation for h[i] * (1 - h[i+k]) terms
        for k in range(1, n):
            for i in range(n):
                j = (i + k) % n
                # We formulate h[i] * (1 - h[j]) 
                # Using McCormick envelopes:
                # z >= h[i] + h[j] - 1
                # z >= 0
                # z <= h[i]
                # z <= h[j]
                z = cp.Variable()
                model += [z >= h_vars[i] + h_vars[j] - 1]
                model += [z >= 0]
                model += [z <= h_vars[i]]
                model += [z <= h_vars[j]]
        return model
        
    # Objective function and solver loop
    max_iter = 1000
    improvement_threshold = 1e-6
    last_c5 = compute_c5(h)
    improvement_count = 0
    best_h = h.copy()
    best_c5 = last_c5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Identify worst shift - this is actually more complex in convex form
        # We'll use an approximate gradient-based approach with small perturbations
        # Identify most constraining correlations and add constraints
        
        # Perturb for gradient direction
        # For numerical stability, start with small perturbations
        gradient_h = np.empty_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            
            gradient_h[i] = (c5_plus - c5_current) / eps
        
        # Simple update using gradient
        new_h = h - 0.01 * gradient_h
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Reset if violation too severe
        if np.any(new_h < 0) or np.any(new_h > 1):
            new_h = np.clip(h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
        h = new_h
        
        if abs(c5_current - last_c5) < improvement_threshold:
            improvement_count += 1
            if improvement_count > 20:
                break
        else:
            improvement_count = 0
        last_c5 = c5_current
        
        if c5_current < 0.3807:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
