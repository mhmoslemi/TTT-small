import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_lag = n_points - 1

    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Generate an initial step function
        h = np.ones(n_points) * 0.5
        # Add some variation
        h = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
        # Normalize to correct sum
        h = h * target_sum / np.sum(h)

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    elapsed = time.time() - start_time

    # Try SDP relaxation approach first
    if elapsed < 0.8 * budget_s:
        try:
            # SDP approach
            h_sdp = solve_sdp_relaxation(n_points, dx, target_sum)
            if h_sdp is not None:
                c5_sdp = evaluate_c5(h_sdp, dx)
                if c5_sdp < best_c5:
                    best_h = h_sdp
                    best_c5 = c5_sdp
                    elapsed = time.time() - start_time
        except:
            pass

    # If not much time left or SDP failed, do gradient-based optimization
    if elapsed < 0.95 * budget_s:
        # Optimize with bounds and constraints using scipy
        bounds = [(0, 1) for _ in range(n_points)]
        # Project to feasible set
        h_proj = np.clip(h, 0, 1)
        h_proj = h_proj * target_sum / np.sum(h_proj)

        # Define constraint for integral constraint
        def constraint_func(x):
            return np.sum(x) - target_sum

        constr = {'type': 'eq', 'fun': constraint_func}

        # Objective function
        def obj_func(x):
            return evaluate_c5(x, dx)

        # Optimization
        result = minimize(obj_func, h_proj, method='SLSQP', bounds=bounds, constraints=constr, 
                          options={'maxiter': 1000, 'ftol': 1e-6}, callback=lambda x: None)
        
        if result.success:
            c5_opt = evaluate_c5(result.x, dx)
            if c5_opt < best_c5:
                best_h = result.x
                best_c5 = c5_opt

    # Final check
    best_h = np.clip(best_h, 0, 1)
    normalized_h = best_h * target_sum / np.sum(best_h)
    final_c5 = evaluate_c5(normalized_h, dx)
    
    return (normalized_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    corr = np.correlate(h, 1 - np.array(h), mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def solve_sdp_relaxation(n_points, dx, target_sum):
    # Set up SDP relaxation variables
    y = cp.Variable((n_points, n_points), symmetric=True)
    h = cp.Variable(n_points)
    
    # Objective
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        y >> 0,  # y is PSD
        cp.trace(y) == target_sum ** 2,  # trace constraint
        h >= 0,
        h <= 1,
        cp.diag(y) == h,  # y[i,i] = h[i]
    ]
    
    # For lag k, the correlation is that of h with (1-h) at lag k,
    # so sum of (h[i] - y[i, i+k]) must be <= C5/dx
    
    # Add the simulation of error bound constraints
    for k in range(n_points):
        # Use a simplified version for computational tractability
        lag_sum = cp.sum([h[i] - y[i, min(i+k, n_points-1)] for i in range(n_points)])
        constraints.append(lag_sum <= C5 / dx)
    
    # Try to reformulate it more carefully
    constraints = []
    
    # Direct formulation
    # The objective is to minimize max_k sum(h[i] * (1-h[i+k]))
    # Which is equivalent to minimizing max_k (sum(h[i]) - sum(h[i]*h[i+k]))
    # where lag k, sum is along i, h is 0 when out of bounds
    
    # Reparameterize into y = h * h^T
    h_vars = cp.Variable(n_points)
    C5 = cp.Variable()
    
    obj = cp.Minimize(C5)
    
    constraints = [
        0 <= h_vars,
        h_vars <= 1,
        cp.sum(h_vars) == target_sum,
    ]
    
    # PSD matrix constraint
    Y = cp.Variable((n_points, n_points), symmetric=True)
    constraints.append(Y >> 0)
    constraints.append(cp.diag(Y) == h_vars)
    
    # Rephrasing: constraint that cross correlations are bounded
    # Iterate over all possible lags
    for k in range(2*n_points-1):
        # Construct indices correctly (considering periodicity or padding)
        corr_sum = 0.0
        # The direct sum formulation:
        # For lag k, we compute sum h[i] * (1 - h[i+k])
        # which equals sum h[i] - sum h[i] * h[i+k]
        # But since we're doing this over the full correlation,
        # it may be helpful to double-check interpretation.
        # Let's be more precise and simulate Lag k = 0 to n_points-1 manually
        lag_indices = []
        for i in range(n_points):
            j = i + k
            if j < n_points:
                # h[i] * (1 - h[j])
                corr_sum += h_vars[i] - Y[i, j]
            # Don't count beyond last index for lower lag indices
        # Constructing the inequality for lag k
        constraints.append(corr_sum <= C5 / dx)
    
    # Final constraints
    problem = cp.Problem(obj, constraints)
    
    try:
        problem.solve(solver=cp.ECOS, verbose=False)
        if problem.status == cp.OPTIMAL:
            return np.array(h_vars.value)
    except Exception:
        pass
    
    # Fallback to simpler
    return None
