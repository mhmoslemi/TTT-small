def evaluate_c5(h, dx):
    # Calculate overlap integral for all lags and return max
    h_extended = np.concatenate([h, h])  # for circular convolution
    n = len(h)
    overlaps = np.zeros(n)
    for k in range(n):
        overlap = 0
        for i in range(n):
            overlap += h[i] * (1 - h_extended[i + k])
        overlaps[k] = overlap * dx
    return np.max(overlaps)
