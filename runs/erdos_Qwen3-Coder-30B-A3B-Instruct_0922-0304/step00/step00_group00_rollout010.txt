import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    # Convex optimization variables
    h_var = cp.Variable(n_points)
    C5_var = cp.Variable()
    
    # Initialize constraints list
    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2,
        C5_var >= 0
    ]
    
    # Define the objective
    objective = cp.Minimize(C5_var)
    
    # Solver settings for speed
    solver = cp.ECOS
    
    while time.time() - start_time < budget_s * 0.95:
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        iteration += 1
        
        # Find maximum overlap shift
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Build quadratic constraint for k_max
        # sum h[i] * (1 - h[(i + k_max) % n_points])
        constraint_expr = 0
        for i in range(n_points):
            j = (i + k_max) % n_points
            constraint_expr += h_var[i] * (1 - h_var[j])
        
        constraints.append(constraint_expr <= C5_var)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    # Renormalize to sum to n_points / 2
                    new_h = new_h * (n_points / 2) / np.sum(new_h)
                    new_c5 = evaluate_erdos_solution(new_h)
                    
                    if new_c5 < best_c5:
                        best_h = new_h.copy()
                        best_c5 = new_c5
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                        
                    h = new_h
        except Exception:
            pass
            
        # Termination conditions
        if best_c5 < 0.3807:
            break
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
    
    if final_c5 >= 0.38085:
        # If we didn't get better, return original solution
        final_h = np.array(initial_h_values) if 'initial_h_values' in globals() else best_h
        final_c5 = evaluate_erdos_solution(final_h)
        return (final_h.tolist(), final_c5, len(final_h))
    else:
        return (best_h.tolist(), final_c5, len(best_h))

def evaluate_erdos_solution(h_values):
    """Simple evaluation based on the overlap integral using numpy."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for all shifts
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap
