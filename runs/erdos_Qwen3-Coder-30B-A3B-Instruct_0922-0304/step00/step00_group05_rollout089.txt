import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all lags
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    dx = 2.0 / n_points
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000
    
    last_improvement = 0
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find indices where max overlap occurs
        max_lags = np.where(overlap == max_overlap)[0]
        
        # Compute gradient by summing over all max_lag contributions
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # Contribution from h[i] to overlap at lag
                # When we compute overlap[lag], we do sum(h[i] * (1 - h[i + lag]))
                # So gradient w.r.t. h[i] is -(1 - h[i + lag]) when i + lag < n_points
                for i in range(n_points):
                    if i + lag < n_points:
                        gradient[i] -= (1 - h[i + lag]) * dx
            else:
                # For negative lags probably can't happen given mode="full"
                # But just to be safe:
                lag_neg = lag - (n_points - 1)
                for i in range(n_points):
                    if i - lag_neg >= 0:
                        gradient[i] -= (1 - h[i - lag_neg]) * dx
        
        # Normalize gradient
        gradient /= len(max_lags) if len(max_lags) > 0 else 1
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h to feasible region: [0,1] and sum = n_points / 2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Evaluate solution
        c5 = evaluate_erdos_solution(h, dx)
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
            
    return (best_h, best_c5, n_points)
