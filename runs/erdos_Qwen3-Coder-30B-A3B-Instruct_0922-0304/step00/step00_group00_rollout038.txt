import numpy as np
import cvxpy as cp
from time import time
import math

def evaluate_erdos_solution(h):
    """Compute C5 from the given h values"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps with full convolution
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    # The maximum overlap is at index n-1 to 2*n-2, so we get the last n-1 values
    # But actually since we're maximizing over all k, it's the maximum of the full
    # convolution which has 2*n - 1 values. The central index is n-1
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize with the parent construction if available, otherwise use a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    dx = 2.0 / n
    
    # Make sure it satisfies the integral constraint
    h_sum = np.sum(h)
    h = h * (n / 2) / h_sum
    
    # Ensure bounds are maintained
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Track the last few constraints to avoid overfitting
    recent_constraints = []

    # Optimizing loop
    iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    max_iter = 10000  # Prevent runaway loop
    while time() - start_time < budget_s * 0.95 and iterations < max_iter:
        iterations += 1

        # Compute all overlaps to find maximum
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        max_idx = np.argmax(overlaps)
        k_max = max_idx - (n - 1)  # This maps back to actual k shift
        
        # The max overlap for this shift is at the correct index
        max_overlap = overlaps[max_idx]
        
        # Add to recent constraints
        recent_constraints.append(k_max)
        if len(recent_constraints) > 10:
            recent_constraints.pop(0)

        # Optimization problem setup
        h_var = cp.Variable(n)
        C5 = cp.Variable()

        # Build objective and constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n / 2,
            C5 >= 0
        ]
        
        # Add constraint for this shift
        # The overlap constraint is sum(h[i] * (1 - h[(i + k) % n])) <= C5
        # We must be careful about indexing
        if k_max >= 0:
            shift_h = np.concatenate([h_var[k_max:], h_var[:k_max]])
        else:
            shift_h = np.concatenate([h_var[k_max:], h_var[:k_max]])
        constraint_expr = cp.sum(h_var * (1 - shift_h))
        constraints.append(constraint_expr <= C5)
        
        # Solve
        prob = cp.Problem(cp.Minimize(C5), constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h_var.value
                if h_new is not None:
                    h = h_new
                    current_c5 = evaluate_erdos_solution(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                else:
                    no_improvement_count += 1
            else:
                no_improvement_count += 1
        except:
            no_improvement_count += 1

        # Early stopping
        if best_c5 < 0.3807:
            break
        if no_improvement_count > max_no_improvement:
            break
            
        # Occasionally remove some old constraints
        if len(recent_constraints) > 5 and iterations % 10 == 0:
            recent_constraints = []

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
