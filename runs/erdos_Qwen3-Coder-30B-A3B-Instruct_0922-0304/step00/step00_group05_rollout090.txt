import numpy as np
from scipy import optimize
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value from all lags
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, dx):
    """Project h to satisfy the constraints."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to meet integral constraint
    integral = np.sum(h) * dx
    if integral > 0:
        h = h * (1.0 / integral) * 2.0 * dx
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    max_time = time.time() + budget_s - 1  # Leave 1s for cleanup
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use the provided initial construction
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, dx)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    else:
        # Random initialization
        h0 = np.random.rand(n_points)
        h0 = project_to_feasible(h0, dx)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Fill with random particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_c5 = [evaluate_c5(p, dx) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_c5)
    global_best = personal_best[global_best_idx].copy()
    global_best_c5 = personal_best_c5[global_best_idx]
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() > max_time:
            break
            
        for i in range(swarm_size):
            h = particles[i]
            # Evaluate current particle
            c5 = evaluate_c5(h, dx)
            
            # Update personal best
            if c5 < personal_best_c5[i]:
                personal_best[i] = h.copy()
                personal_best_c5[i] = c5
                
                # Update global best
                if c5 < global_best_c5:
                    global_best = h.copy()
                    global_best_c5 = c5
            
            # Update velocities and positions
            r1, r2 = np.random.rand(2)
            pbest = personal_best[i]
            gbest = global_best
            
            # PSO update equations
            velocities[i] = (inertia * velocities[i] +
                             cognitive * r1 * (pbest - h) +
                             social * r2 * (gbest - h))
            
            particles[i] = h + velocities[i]
            
            # Project to feasible domain
            particles[i] = project_to_feasible(particles[i], dx)
    
    # Final evaluation
    final_c5 = evaluate_c5(global_best, dx)
    
    return (global_best, final_c5, n_points)
