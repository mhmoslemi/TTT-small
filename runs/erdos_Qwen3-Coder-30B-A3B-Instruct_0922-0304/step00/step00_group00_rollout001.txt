import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure the constraint sum(h) * dx = 1 (i.e. sum(h) = n_points/2)
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Gradient descent parameters
    learning_rate = 0.1
    max_iter = 1000000
    
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For early stopping
    no_improvement_count = 0
    max_no_improvement = 50
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all shifts
        # Using correlate for full convolution
        overlap_full = correlate(h, 1 - h, mode="full")
        # Only valid overlaps (from -n+1 to n-1), but we only consider k in [0, n-1] for simplicity
        # since we're using mode='full', overlap_full has length 2*n-1
        overlaps = []
        for k in range(-n_points + 1, n_points):
            overlap = overlap_full[k + n_points - 1] * dx
            overlaps.append(overlap)
        
        # Find k that gives maximal overlap
        max_k = np.argmax(overlaps)
        max_k = max_k - (n_points - 1)  # Convert index to shift k
        
        # Compute the gradient
        grad = np.zeros_like(h)
        if max_k >= 0:
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k]) * dx
                grad[i + max_k] -= h[i] * dx
        else:
            for i in range(-max_k, n_points):
                grad[i] += (1 - h[i + max_k]) * dx
                grad[i + max_k] -= h[i] * dx
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Renormalize to sum to n_points / 2
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        # Evaluate current C5
        c5 = max(overlaps)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if no_improvement_count > max_no_improvement:
            break
            
        # Adaptive learning rate
        if i % 100 == 0 and i > 0:
            if no_improvement_count > 20:
                learning_rate *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
