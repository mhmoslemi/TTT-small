import numpy as np
from scipy import signal
import time
import math

def evaluate_solution(h, dx):
    # Convolution of h with (1-h) for all lags
    conv = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx
    conv *= dx
    # Return maximum value
    return np.max(conv)

def project_to_feasible(h, dx):
    # Ensure h is within [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum = 1/dx (so that integral = 1)
    total = np.sum(h)
    if total > 0:
        h = h * (1.0 / (dx * total))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    particles = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_to_feasible(h0, dx)
        particles.append(h0.copy())
        best_positions.append(h0.copy())
        v0 = np.random.uniform(-0.5, 0.5, n_points)
        velocities.append(v0)
        best_values.append(evaluate_solution(h0, dx))
    else:
        # Create initial random solution
        h0 = np.random.random(n_points)
        h0 = project_to_feasible(h0, dx)
        particles.append(h0.copy())
        best_positions.append(h0.copy())
        v0 = np.random.uniform(-0.5, 0.5, n_points)
        velocities.append(v0)
        best_values.append(evaluate_solution(h0, dx))
    
    # Add random particles to fill swarm
    for i in range(swarm_size - 1):
        h = np.random.random(n_points)
        h = project_to_feasible(h, dx)
        particles.append(h)
        v = np.random.uniform(-0.5, 0.5, n_points)
        velocities.append(v)
        best_values.append(evaluate_solution(h, dx))
        best_positions.append(h.copy())
    
    # Optimization loop
    start_time = time.time()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Evaluate particles
        for i in range(swarm_size):
            h = particles[i]
            current_value = evaluate_solution(h, dx)
            if current_value < best_values[i]:
                best_values[i] = current_value
                best_positions[i] = h.copy()
        
        # Find global best
        gb_index = np.argmin(best_values)
        gb_value = best_values[gb_index]
        gb_position = best_positions[gb_index]
        
        # Update velocities and positions
        for i in range(swarm_size):
            if i == gb_index:
                continue
                
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive_component = cognitive * r1 * (best_positions[i] - h)
            social_component = social * r2 * (gb_position - h)
            v = inertia * v + cognitive_component + social_component
            
            # Update position
            new_h = h + v
            
            # Project to feasibility
            new_h = project_to_feasible(new_h, dx)
            
            # Store
            particles[i] = new_h
            velocities[i] = v
    
    # Final evaluation
    final_best_idx = np.argmin(best_values)
    final_h = best_positions[final_best_idx].copy()
    final_c5 = best_values[final_best_idx]
    
    return (final_h.tolist(), final_c5, n_points)
