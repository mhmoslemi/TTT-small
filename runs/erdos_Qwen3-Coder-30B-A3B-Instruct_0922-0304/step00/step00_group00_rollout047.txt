import numpy as np
import time
from scipy.signal import correlate

def initial_h():
    # Use the initial construction if available
    try:
        return np.array(initial_h_values)
    except NameError:
        # Fallback to uniform distribution with sum = n/2
        n = 45
        h = np.full(n, 0.5)
        h[0] = 0.5 + (n/2 - np.sum(h)) / n
        return h

def evaluate(h, dx):
    # Compute overlap integral C5
    conv = correlate(h, 1 - h, mode='full')
    max_val = np.max(conv) * dx
    return max_val

def normalize(h, n_points):
    # Ensure sum(h) = n_points / 2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    scale = target_sum / current_sum
    h = h * scale
    return np.clip(h, 0, 1)

def perturb(h, n_points, delta=0.1):
    # Select random indices and perturb
    num_perturb = max(1, int(0.05 * n_points))
    indices = np.random.choice(n_points, num_perturb, replace=False)
    h_new = h.copy()
    for i in indices:
        # Perturb by delta, but keep within [0,1]
        h_new[i] += np.random.uniform(-delta, delta)
    return np.clip(h_new, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    h = initial_h()
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Normalize initial to sum = n_points / 2
    h = normalize(h, n_points)
    
    current_c5 = evaluate(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters
    T = 0.1
    cooling_rate = 0.995
    delta = 0.1
    
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        # Perturb
        h_new = perturb(h, n_points, delta)
        # Renormalize
        h_new = normalize(h_new, n_points)
        # Evaluate
        new_c5 = evaluate(h_new, dx)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
        
        # Early stopping
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)
