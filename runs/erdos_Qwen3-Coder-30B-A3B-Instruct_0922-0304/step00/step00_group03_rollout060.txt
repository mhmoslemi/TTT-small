import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if available, otherwise create a reasonable initial guess
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create an initial guess: uniform distribution with some randomness
        n_points = 70
        h_init = np.ones(n_points) * 0.5
        # Add some structure
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
        h_init = np.clip(h_init, 0, 1)
    
    # Ensure sum constraint
    dx = 2.0 / len(h_init)
    current_sum = np.sum(h_init)
    target_sum = 1.0
    scale_factor = target_sum / current_sum
    h_init = h_init * scale_factor
    
    # Round and repair sum constraint
    h_init = np.round(h_init)
    current_sum = np.sum(h_init)
    if abs(current_sum - target_sum/dx) > 1e-6:
        # Repair: adjust elements randomly to fix sum
        target_count = int(target_sum / dx)
        current_count = int(np.sum(h_init))
        diff = target_count - current_count
        if diff > 0:
            # Need to flip some zeros to ones
            zero_indices = np.where(h_init == 0)[0]
            flipped = np.random.choice(zero_indices, diff, replace=False)
            h_init[flipped] = 1.0
        elif diff < 0:
            # Need to flip some ones to zeros
            one_indices = np.where(h_init == 1)[0]
            flipped = np.random.choice(one_indices, -diff, replace=False)
            h_init[flipped] = 0.0
    
    # Binary search on C5
    tolerance = 1e-4
    low = 0.0
    high = 0.5
    best_h = None
    best_c5 = np.inf
    
    while (high - low) > tolerance and (time() - start_time) < budget_s * 0.9:
        mid = (low + high) / 2.0
        
        # Formulate convex problem (linear program)
        n = len(h_init)
        h = cp.Variable(n, name='h')
        
        # Building term: h(i) * (1 - h(i+k))
        # Constraint: sum(h(i) * (1 - h(i+k))) <= mid  for all k
        constraints = []
        constraints.append(cp.sum(h) == n/2.0)
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # For each k: 
        # We want sum_{i=0}^{n-1-|k|} h(i)(1-h(i+k)) <= mid
        # We know that:
        # h(i) * (1 - h(i+k)) = h(i) - h(i) * h(i+k)  
        # So we want sum_{i=0}^{n-1-|k|} (h(i) - h(i) * h(i+k)) <= mid
        
        # Introduce auxiliary variable w[i,k] for h[i] * h[i+k]
        for k in range(-n+1, n):
            # This is for correlation with shift k
            k_indices = np.arange(n - abs(k))
            if k >= 0:
                valid_h = h[k_indices]
                shifted = h[k_indices + k]
            else:
                valid_h = h[k_indices - k]
                shifted = h[k_indices]
            product_terms = valid_h * shifted
            expr = cp.sum(valid_h) - cp.sum(product_terms)
            constraints.append(expr <= mid)
        
        # Objective: minimize mid (this will be handled by binary search)
        # Actually, we don't minimize anything here; we just check feasibility
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s/10.0)
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLVED:
                h_sol = h.value
                # Ensure the constraints are actually met by numerical checking
                h_sol = np.clip(h_sol, 0, 1)
                # Evaluate C5 for this solution
                corr = np.correlate(h_sol, 1-h_sol, mode='full')
                c5_val = np.max(corr) * dx
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_sol.copy()
                high = mid
            else:
                low = mid
        except:
            low = mid
        
        if (time() - start_time) > budget_s * 0.95:
            break
    
    # Final polish and return
    if best_h is None:
        best_h = h_init
    else:
        h_init = best_h
    
    # After selection, final evaluation of C5
    corr = np.correlate(h_init, 1-h_init, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (h_init, final_c5, len(h_init))
