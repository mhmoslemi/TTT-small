import numpy as np
import time
from scipy import optimize

    start_time = time.time()

        n_points = len(h)
    except NameError:
        # Create initial binary construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vals):
        # Compute overlap integral
        h = np.array(h_vals)
        # Compute correlations for all shifts
        corr = np.correlate(h, 1-h, mode="full")
        # Normalize by dx
        c5 = max(corr) * dx
        return c5
    
    def project_h(h_vals):
        # Clip to [0,1]
        h = np.clip(h_vals, 0, 1)
        # Scale to sum = target_sum
        current_sum = np.sum(h)
        if current_sum > 0:
            h *= target_sum / current_sum
        return h
    
    def compute_gradient(h_vals, active_shift):
        h = np.array(h_vals)
        grad = np.zeros_like(h)
        # Compute gradient for active shift
        for i in range(n_points):
            # For overlap at shift k, derivative w.r.t. h[i]
            j = (i + active_shift) % n_points
            grad[i] = -(1 - h[j])
        return grad
    
    # Gradient descent parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 0
    patience_limit = 100
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1 and patience < patience_limit:
        # Compute overlaps for all shifts
        h = project_h(h)
        corr = np.correlate(h, 1-h, mode="full")
        k_active = np.argmax(corr)
        c5 = corr[k_active] * dx
        
        # Store best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Compute gradient
        grad = compute_gradient(h, k_active)
        
        # Update with gradient
        h_temp = h - alpha * grad
        
        # Project
        h = project_h(h_temp)
        
        # Adaptively reduce step size
        if c5 >= best_c5:
            alpha *= gamma
            if alpha < alpha_min:
                break
        else:
            alpha = min(alpha * 1.05, 1.0)  # Occasionally increase
            
        iteration += 1
        # Perturb every 500 iterations to escape local minima
        if iteration % 500 == 0:
            perturb_idx = np.random.choice(n_points, size=max(1, n_points//20), replace=False)
            h[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            h = project_h(h)
    
    # Final computation of C5
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
