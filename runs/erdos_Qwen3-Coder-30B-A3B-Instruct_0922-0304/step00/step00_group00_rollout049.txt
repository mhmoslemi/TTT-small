import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    current_c5 = evaluate_c5(h, dx)
    best_c5 = current_c5
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05
    delta = 0.1
    
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        iterations += 1
        # Perturb a fraction of points
        n_perturb = max(1, int(n_points * perturb_frac))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Save current h and compute energy
        old_h = h.copy()
        old_c5 = current_c5
        
        # Make perturbations
        for i in indices:
            # Random change with limit
            change = np.random.uniform(-delta, delta)
            h[i] += change
            h[i] = np.clip(h[i], 0, 1)
        
        # Renormalize to correct sum
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        # Compute new energy
        current_c5 = evaluate_c5(h, dx)
        
        # Accept or reject based on Metropolis criterion
        delta_E = current_c5 - old_c5
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            # Accept new state
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reject change, restore
            h = old_h
            current_c5 = old_c5
        
        # Cool down
        T *= cooling_rate
        
        # Early termination if target achieved
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k ∫ h(x)(1 - h(x+k)) dx """
    # Compute all correlations via FFT
    corr = correlate(h, 1-h, mode="full")
    # Select the max lags valid for [0,2] interval
    # The lags are in range [-n+1, n-1] so valid range is [0,n-1]
    # Forgot to check the exact range but we're just taking max
    max_corr = np.max(corr)
    return max_corr * dx
