import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        # Randomly set exactly n_points/2 to 1
        ones_indices = np.random.choice(n_points, n_points//2, replace=False)
        h[ones_indices] = 1.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count - last_improvement < 1000:
        iter_count += 1
        h_old = h.copy()
        
        # Compute auto-correlations with 1-h (the overlap)
        # Use correlation coefficient, which is equivalent to what we want for overlap
        # We calculate for all lags
        h_complement = 1 - h
        corr = correlate(h, h_complement, mode='full')
        corr = corr[len(corr)//2:]  # Take only the positive lags
        
        # For each lag k, the overlap integral is approximately:
        # C5_k = sum(h[i] * (1 - h[i+k]))  (mod n_points)
        # We use the correlation as proxy of the "badness" of shifting
        # So we work with the maximum correlation value over all shifts
        
        # The maximum of the correlation values (for negative lags: k>0)
        max_corr = np.max(corr)
        c5_candidate = max_corr * dx
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h.copy()
            
        # Perform an update step:
        # For each position, calculate how setting it to 1 affects the max overlap
        # Then update in a greedy way
        max_lag = len(corr)
        overlap_counts = np.zeros(n_points)
        for k in range(max_lag):
            # Count overlaps for each index i when h[i] = 1 and h[i+k] = 1 (mod n_points)
            shift_indices = (np.arange(n_points) + k) % n_points
            overlap_counts += h[shift_indices]
            
        # Now set h[i]=1 if it helps reduce the maximum overlap
        # But must maintain constraint that sum(h) = target_sum
        current_sum = np.sum(h)
        diff = current_sum - target_sum
        
        # Adjust greedily -- try to reset a few positions
        # If too many 1s, make some 0s
        # If too few 1s, make some 1s
        
        # Make corrections based on overlap counts (higher is worse)
        # This is a form of greedy update
        if abs(diff) > 1e-6:
            # Want to fix the sum constraint by swapping some elements
            if diff > 0:
                # Too many 1s, must make some 0s
                # Find indices where we should set to 0
                indices_ones = np.where(h > 0.5)[0]
                if len(indices_ones) > 0:
                    # Sort by overlap count (descending, to choose worst/offending ones)
                    sort_indices = np.argsort(-overlap_counts[indices_ones])
                    indices_to_zero = indices_ones[sort_indices[:int(diff)]]
                    for i in indices_to_zero:
                        h[i] = 0.0
            else:
                # Not enough 1s, must make some 1s
                # Find indices where we could increase to 1 (0s)
                indices_zeros = np.where(h < 0.5)[0]
                if len(indices_zeros) > 0:
                    # Priority again to indices causing less overlap
                    sort_indices = np.argsort(overlap_counts[indices_zeros])
                    indices_to_one = indices_zeros[sort_indices[:int(-diff)]]
                    for i in indices_to_one:
                        h[i] = 1.0
        
        # Clamp
        h = np.clip(h, 0, 1)
        
        # Adjust sum to exact value
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            # Rescale to get correct sum
            scale = target_sum / current_sum
            h = np.clip(h * scale, 0, 1)
        
        # Check intermediate termination criteria
        if best_c5 < 0.3807:
            break
    
    # Final evaluation with original function if needed
    # Try to get exact c5 calculation:
    h_complement = 1 - best_h
    corr = correlate(best_h, h_complement, mode='full')
    corr = corr[len(corr)//2:]
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
