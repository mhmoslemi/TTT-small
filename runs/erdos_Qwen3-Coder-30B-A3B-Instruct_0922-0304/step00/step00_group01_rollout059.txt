import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the constraint sum(h) * dx = 1 (i.e., sum(h) = n_points / 2)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    # Convex optimization using CVXPY
    while time.time() - start_time < budget_s - 1:
        # Compute C(k) for all k 
        correlations = np.correlate(h, 1 - h, mode="full")
        correlations *= dx
        k_max = np.argmax(correlations)
        max_overlap = correlations[k_max]
        
        # Early stopping if target reached
        if max_overlap <= 0.3807:
            break
            
        # Invert k_max to get the index in the reduced correlations
        # indices = [0..n_points-1] for h[i], [n_points-1..0] for 1-h[i-k] 
        # full conv is size 2*n_points-1
        shift = k_max - (n_points - 1)
        
        # Form the CVXPY variables & optimization problem
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        # Sum constraint
        constraints.append(sum(h_var) == target_sum)
        
        # Objective
        # C(k) = sum over i of h[i] * (1 - h[(i - shift) % n_points])
        # We need h[(i-shift) % n_points], which is equivalent to h[(i + (n_points - shift)) % n_points]
        # Thus we define the linear term as the correlation at shift position.
        # For simplicity, we focus on one fixed shift k_max.
        objective = Minimize(h_var @ (1 - h_var[[(i - shift) % n_points for i in range(n_points)]]))
        
        # Solve current convex program
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value)
                h = h_new
            else:
                break
        except:
            break
            
        # Project back to [0,1] and re-scale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (target_sum / h_sum)
            
    # Final evaluation of C5
    correlations = np.correlate(h, 1 - h, mode="full")
    correlations *= dx
    c5_bound = np.max(correlations)
    
    return (h.tolist(), c5_bound, n_points)
