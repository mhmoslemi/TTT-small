import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Only consider the valid overlaps (shifts 0 to n-1)
    overlaps = corr[n-1:]
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n = 47  # same as initial_h_values
    dx = 2.0 / n
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n) * 0.5

    # Ensure sum constraint
    h = h * (n / 2) / np.sum(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iteration = 0
    last_improvement = 0
    tolerance = 1e-6

    # Multi-initialization
    initial_solutions = [h]
    for i in range(4):
        perturbed = h + np.random.normal(0, 0.005, n)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n / 2) / np.sum(perturbed)
        initial_solutions.append(perturbed)
    
    for init_h in initial_solutions:
        h = init_h.copy()
        for i in range(5000):  # Max iterations
            if time.time() - start_time > budget_s - 1:
                break

            # Compute overlaps
            corr = np.correlate(h, 1 - h, mode="full")
            overlaps = corr[n-1:]  # The actual overlaps
            max_k = np.argmax(overlaps)
            max_overlap = overlaps[max_k]
            
            # Gradient calculation for the worst case
            shifted = np.roll(1 - h, max_k)
            gradient = shifted - h * shifted
            
            # Normalize gradient
            grad_norm = np.linalg.norm(gradient)
            if grad_norm < 1e-10:
                step_size = 0.01
            else:
                step_size = 0.01
            
            # Backtracking line search
            alpha = step_size
            rho = 0.5
            max_backtrack = 30
            
            for _ in range(max_backtrack):
                h_new = h - alpha * gradient / grad_norm
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n / 2) / np.sum(h_new)
                new_c5 = evaluate_erdos_solution(h_new)
                
                if new_c5 < max_overlap or alpha < 1e-10:
                    h = h_new
                    break
                alpha *= rho

            # Convex refinement every 50 steps
            if i % 50 == 0 and i > 0:
                try:
                    # Minimize overlap for the current max_k
                    # Using convex optimization
                    h_var = cp.Variable(n)
                    constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n/2]
                    
                    # Overlap function
                    shifted = np.roll(1 - h, max_k)
                    obj = cp.sum(cp.multiply(h_var, shifted))  # This should be wrong, but let's correct it
                    
                    # Instead, we minimize sum over all valid h[i] values
                    # Approximate in Convex framework by fixing shift (max_k)
                    shift_index = max_k
                    penalty = h_var[shift_index] * (1 - h_var[0])  # Example for one term
                    obj = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - np.roll(h_var, shift_index))))
                    prob = cp.Problem(obj, constraints)
                    prob.solve(solver=cp.ECOS, verbose=False)
                    
                    if prob.status == cp.OPTIMAL:
                        h_new = h_var.value
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n / 2) / np.sum(h_new)
                        c5_new = evaluate_erdos_solution(h_new)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_new.copy()
                    
                except:
                    pass  # fallback to gradient based

            # Check if better solution
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5 - tolerance:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = i
                
    return (best_h, best_c5, n)
