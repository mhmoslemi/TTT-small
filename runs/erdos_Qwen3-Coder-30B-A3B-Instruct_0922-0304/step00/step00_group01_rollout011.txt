import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute overlaps for all shifts using correlation
    full_correlation = correlate(h, 1 - h, mode='full')
    # Only consider positive shifts (shifts 0 to n-1)
    overlaps = full_correlation[len(h)-1:] * dx
    return np.max(overlaps)

def project_h(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points, dx):
    """Compute gradient of overlap w.r.t. h for shift k_max."""
    gradient = np.zeros_like(h)
    # For each i, compute dC/dh[i] where C = sum h[i] * (1 - h[i+k_max])
    for i in range(n_points):
        # Consider h[i+k_max] only if it's within bounds
        if i + k_max < n_points:
            gradient[i] += (1 - h[i + k_max]) * dx
        # Contribute to h[i] term by gradient of C w.r.t. h[i]
        # This requires negative hit due to structure
        if i - k_max >= 0:
            gradient[i] -= h[i - k_max] * dx
    return gradient

def line_search(h, grad, k_max, n_points, dx, alpha_initial=0.01, rho=0.5, eta=0.5):
    """Backtracking line search to find good step size."""
    alpha = alpha_initial
    current_c5 = evaluate_c5(h, dx)
    # Gradient magnitude
    grad_norm_sq = np.sum(grad**2)
    if grad_norm_sq == 0:
        return 0
    grad_norm = np.sqrt(grad_norm_sq)
    
    for _ in range(30):  # max backtracking steps
        h_new = h - alpha * grad / grad_norm
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new, dx)
        
        if new_c5 < current_c5 - eta * alpha * grad_norm_sq:
            # Armijo condition satisfied
            return alpha
        alpha *= rho
    return alpha  # fallback

    best_h = h.copy()

    last_improvement = 0
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        iteration += 1
        
        # Compute overlaps for all shifts
        full_corr = correlate(h, 1 - h, mode='full')
        overlaps = full_corr[len(h)-1:] * dx
        k_max = np.argmax(overlaps)
        current_c5 = overlaps[k_max]
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if best_c5 < 0.3807:

        # Compute gradient for the worst shift
        grad = compute_gradient(h, k_max, n_points, dx)
        
        # Line search
        alpha = line_search(h, grad, k_max, n_points, dx)
        if alpha <= 0:
            # Line search failed to find descent
            break
            
        # Update h
        h = h - alpha * grad / np.linalg.norm(grad)
        h = project_h(h, n_points)
        
        # check for stagnation
        if current_c5 >= best_c5:
            last_improvement += 1
            if last_improvement > 100:
                break
        else:
            last_improvement = 0
    
    # return best solution
    return (best_h.tolist(), best_c5, n_points)
