import numpy as np
from scipy import signal
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize particles and velocities
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    
    if 'initial_h_values' in globals():
        particles[0] = initial_h_values
        # Scale to sum = n_points / 2
        particles[0] *= (n_points / 2) / np.sum(particles[0])
    else:
        particles[0] = np.ones(n_points) * 0.5
    
    # Initialize remaining particles
    for i in range(1, swarm_size):
        particles[i] = np.random.rand(n_points)
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.5, 0.5, (swarm_size, n_points))
    
    # PSO parameters
    personal_best = particles.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    
    time_start = time()
    iter_count = 0
    
    while time() - time_start < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            
            # Compute overlap using convolution
            # We compute correlation of h with (1-h) over all lags
            h_reflected = 1 - h[::-1]
            conv_result = signal.correlate(h, h_reflected, mode='full')
            
            # The overlap for each lag m is a convolution element
            # The matrix is size 2*n_points - 1, corresponding to all lags
            # The lags are from -(n_points-1) to (n_points-1)
            # For our specific problem, we want the overlap over all positive lags 0 to 2*n_points-2
            # The actual overlap value for lag m is at index (n_points-1)+m
            # Let's extract the positive lags correctly
            overlaps = conv_result[n_points-1:] * dx
            
            # Maximum overlap
            score = np.max(overlaps)
            
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h.copy()
            
            if score < global_best_score:
                global_best_score = score
                global_best_idx = i
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            pbest = personal_best[i]
            gbest = personal_best[global_best_idx]
            
            # Update velocity
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest - particles[i]) +
                social * r2 * (gbest - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            # Clamp to [0, 1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Adjust to exact sum = n_points / 2
            current_sum = np.sum(particles[i])
            particles[i] *= (n_points / 2) / current_sum
    
    # Return best solution
    h_best = personal_best[global_best_idx]
    return (h_best, global_best_score, n_points)
