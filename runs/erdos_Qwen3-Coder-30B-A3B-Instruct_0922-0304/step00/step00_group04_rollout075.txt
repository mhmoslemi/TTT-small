import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n-1, we only care about lags from -n+1 to n-1
    # But we want max of the positive lags (as per the integral definition)
    # If we want max over k from -n+1 to n-1, this is the full corr
    # But since it's symmetric and we're computing the `max` over k, 
    # we take the maximum of all those values
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points, dx):
    """Project h onto feasible region with sum = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    # Adjust to meet constraint sum = n_points/2
    current_sum = np.sum(h) * dx
    if abs(current_sum - n_points/2) > 1e-10:
        # Rescale to target sum
        scale = (n_points/2) / current_sum
        h = h * scale
        # projection to [0,1]
        h = np.clip(h, 0, 1)
        # Ensure sum is correct with final projection
        current_sum = np.sum(h) * dx
        if abs(current_sum - n_points/2) > 1e-10:
            # Small linear adjustment if final scaling didn't hit target
            diff = (n_points/2) - current_sum
            adjust = diff / (dx * n_points)
            h += adjust
            h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial if available, otherwise create a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # High school construction from responses
        n_points = 42
        h = np.ones(n_points) * 0.5  # simple uniform
        
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    last_improvement = 0
    
    # Gradient descent with line search
    lr = 1e-4  # initial learning rate
    max_iter = 10000

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current C5
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Find dominant k
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr_val = corr[max_idx]
        
        # Compute gradient w.r.t. max lag k
        # Lag goes from -(n_points-1) to (n_points-1)
        k = max_idx - (n_points - 1)  # k in [-n+1, n-1]
        
        # We're taking derivative of rotation of h (1-h)
        # For lag k, at index i, integral is h[i] * (1 - h[i+k])
        # So gradient w.r.t. h[i] is (1 - h[i+k]) for i + k >= 0 
        # and (1 - h[i-k]) for i - k >= 0, etc.
        # That's the gradient of sum h[i] * (1-h[i+k])
        
        # Simplified gradient computation:
        # For each i : grad[i] = (1 - h[i + k]) if i + k < n_points and i + k >= 0
        #                    = (1 - h[i - k]) if i - k < n_points and i - k >= 0 if k < 0 and k > 0
        # Better to compute explicitly
        
        gradient = np.zeros_like(h)
        # Gradient part for h[i] involves: coefficient of (1 - h[i+k]) term 
        # derivative of ∫ h[x] * (1-h[x+k]) dx = ∫ (1 - h[x+k]) dx
        
        # Let's compute directional derivative more directly.
        # Do the full correlation for the +k part to extract influence of each variable
        # We know max_k corresponds to corr[max_idx] 
        
        if k >= 0:
            # Lag k is positive, meaning integration over h[i](1-h[i+k])
            for i in range(n_points - k):
                gradient[i] += (1 - h[i + k])
        else:
            # k is negative => h[i](1 - h[i+k]) means h[i+k](1 - h[i]) 
            for i in range(-k, n_points):
                gradient[i] += (1 - h[i + k])
                
        # We could do more careful gradient but we need to iterate loop
        # Just take one step in the negative gradient direction

        h_new = h - lr * gradient
        h_new = project_h(h_new, n_points, dx)
        
        h = h_new
        
        # Adjust learning rate
        # If no improvement in a while, decay
        if iteration - last_improvement > 50:
            lr *= 0.9
            last_improvement = iteration
            
    # Final validation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h, final_c5, n_points)
