import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    cross_corr = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(cross_corr)
    return max_overlap * dx

        h[ones] = 1
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Simple alternating optimization approach
    max_iter = 10000
    last_improvement = 0

    shifts = np.arange(n_points)

        improved = False

        indices = np.random.permutation(n_points)

            if c5_with_zero <= c5_with_one:
                h[i] = 0
