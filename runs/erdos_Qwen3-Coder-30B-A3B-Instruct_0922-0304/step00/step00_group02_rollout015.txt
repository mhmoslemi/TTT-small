import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the full correlation
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def is_feasible(h):
    """Check if h satisfies the constraints."""
    n = len(h)
    dx = 2.0 / n
    # Check bounds
    if np.any(h < 0) or np.any(h > 1):
        return False
    # Check sum constraint
    if abs(np.sum(h) * dx - 1.0) > 1e-10:
        return False
    return True

def repair_h(h):
    """Repair h to be feasible by clipping and rescaling."""
    n = len(h)
    dx = 2.0 / n
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        scale = 1.0 / current_sum
        h = h * scale
    return h

def mutate_h(h, mutation_rate=0.01):
    """Mutate h with given rate."""
    n = len(h)
    mask = np.random.random(n) < mutation_rate
    if not np.any(mask):
        # Ensure at least one mutation
        idx = np.random.randint(0, n)
        mask[idx] = True
    # Perturb selected elements
    delta = np.random.normal(0, 0.05, n) * mask
    h_new = h + delta
    return repair_h(h_new)

def crossover_h(parent1, parent2):
    """Single-point crossover."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    
    # Select crossover point
    cp = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:cp], parent2[cp:]])
    child2 = np.concatenate([parent2[:cp], parent1[cp:]])
    
    # Repair children
    child1 = repair_h(child1)
    child2 = repair_h(child2)
    
    return child1, child2
