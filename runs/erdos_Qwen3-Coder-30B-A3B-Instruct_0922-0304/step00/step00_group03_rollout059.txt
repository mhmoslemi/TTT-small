import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h."""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def is_valid_solution(h, n_points):
    """Check if solution satisfies constraints."""
    return (np.all((h >= 0) & (h <= 1)) and 
            abs(np.sum(h) - n_points / 2) < 1e-10)

def flip_bits(h, num_flips=2):
    """Generate neighbors by flipping num_flips bits."""
    h = np.array(h)
    n = len(h)
    neighbors = []
    
    # Generate all combinations of num_flips indices to flip
    from itertools import combinations
    indices = list(range(n))
    for flip_indices in combinations(indices, num_flips):
        new_h = h.copy()
        for i in flip_indices:
            new_h[i] = 1 - new_h[i]
        neighbors.append(new_h)
    
    return neighbors

def tabu_search(initial_h, n_points, max_time, tabu_tenure=100):
    """Perform tabu search to minimize C5."""
    start_time = time.time()
    
    # Initialize
    current_h = np.array(initial_h)
    current_c5 = evaluate_c5(current_h, 2.0 / n_points)
    best_h = current_h.copy()
    best_c5 = current_c5
    tabu_list = []
    tabu_tenure = min(tabu_tenure, max_time * 10)  # heuristic

    # Decide max iterations based on time budget
    max_iter = int(max_time * 200)  # Estimate iterations
    max_iter = max(1000, max_iter)
    iter_since_improvement = 0
    max_iter_no_improvement = max_iter // 4

    for iteration in range(max_iter):
        if time.time() - start_time > max_time - 1:
            break

        # Generate neighbors by flipping 2 bits
        neighbors = flip_bits(current_h, num_flips=2)
        
        # Filter valid neighbors (maintains sum)
        valid_neighbors = []
        for neighbor in neighbors:
            if np.sum(neighbor) == n_points / 2:
                valid_neighbors.append(neighbor)

        # Remove tabu solutions
        valid_neighbors = [n for n in valid_neighbors if tuple(n) not in tabu_list]

        # Evaluate neighbors
        if valid_neighbors:
            neighbor_scores = [evaluate_c5(n, 2.0 / n_points) for n in valid_neighbors]
            best_neighbor_idx = np.argmin(neighbor_scores)
            best_neighbor = valid_neighbors[best_neighbor_idx]
            neighbor_c5 = neighbor_scores[best_neighbor_idx]

            # Update best if improved
            if neighbor_c5 < best_c5:
                best_h = best_neighbor.copy()
                best_c5 = neighbor_c5
                iter_since_improvement = 0
            else:
                iter_since_improvement += 1

            # Move to the best neighbor
            current_h = best_neighbor.copy()
            current_c5 = neighbor_c5

            # Add previous solution to tabu list
            tabu_list.append(tuple(current_h))
            if len(tabu_list) > tabu_tenure:
                tabu_list.pop(0)
        else:
            iter_since_improvement += 1

        # Early termination if no improvement recently
        if iter_since_improvement > max_iter_no_improvement:
            break

    return best_h, best_c5
