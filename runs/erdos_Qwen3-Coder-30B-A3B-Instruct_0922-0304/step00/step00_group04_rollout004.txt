import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create random feasible h
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    # Gradient descent with adaptive step size
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap integrals for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5_vals = corr * dx
        k_max = np.argmax(c5_vals)
        max_c5 = c5_vals[k_max] 
        
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            
        # Compute gradient at k_max
        grad = np.zeros_like(h)
        if k_max < n_points:
            # Positive lag case
            grad[:n_points-k_max] += (1 - h[k_max:])
        else:
            # Negative lag case  
            k_neg = k_max - n_points
            grad[-k_neg:] += (1 - h[:-k_neg])
        
        # Update using gradient descent
        lr = 0.1 / (1.0 + i * 0.01)
        h_new = h - lr * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Adjust to satisfy integral constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
        
        # Early stopping if improvement negligible
        if i > 10 and abs(max_c5 - best_c5) < 1e-8:
            break
    
    return (best_h, best_c5, n_points)

def evaluate_overlap(h, dx):
    """Evaluate C5 using correlation"""
    corr = correlate(h, 1 - h, mode='full')
    c5_vals = corr * dx
    return np.max(c5_vals)
