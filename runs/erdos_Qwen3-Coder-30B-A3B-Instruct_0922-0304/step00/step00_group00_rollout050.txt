import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5, dtype=float)
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Gradient descent parameters
    alpha = 0.1
    max_iter = 50000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    # For timing
    est_iter_time = 0.001  # estimated time per iteration
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - est_iter_time:
            break
            
        # Compute overlaps for all shifts
        h_full = np.concatenate([h, h])  # for easier convolution
        overlaps = []
        for k in range(-n_points + 1, n_points):
            # Compute overlap integral for shift k
            if k >= 0:
                h_shift = h_full[k:k + n_points]
            else:
                h_shift = h_full[n_points + k:2 * n_points + k]
            overlap = np.sum(h * (1 - h_shift)) * dx
            overlaps.append(overlap)
        
        # Find k that gives max overlap
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # Update best if improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Compute gradient w.r.t. this k
        grad = np.zeros(n_points)
        if max_k >= 0:
            h_shift = h_full[max_k:max_k + n_points]
            for i in range(n_points):
                if i + max_k < n_points:
                    grad[i] += (1 - h[i + max_k]) * dx
                    grad[i + max_k] -= h[i] * dx
        else:
            h_shift = h_full[n_points + max_k:2 * n_points + max_k]
            for i in range(n_points):
                if i + max_k >= 0:
                    grad[i] += (1 - h[i + max_k]) * dx
                    grad[i + max_k] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        
        # Clipping
        h = np.clip(h, 0, 1)
        
        # Renormalize
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2.0) / h_sum
        
        # Early stopping if no improvement for a while
        if iteration > patience and best_c5 < 0.3807:
            break
            
    # Final validation
    final_c5 = max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    return (best_h.tolist(), float(final_c5), n_points)
