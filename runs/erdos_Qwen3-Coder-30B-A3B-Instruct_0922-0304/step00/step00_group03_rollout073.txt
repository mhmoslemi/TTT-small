import numpy as np
from scipy import optimize
from cvxpy import *
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available
    try:
        h_init = initial_h_values.copy()
        n_points = len(h_init)
    except:
        # Create initial construction: step function with 1s in first half, 0s in second
        n_points = 70
        h_init = np.concatenate([np.ones(n_points//2), np.zeros(n_points - n_points//2)])
        # Randomize slightly 
        h_init += np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
    
    # Normalize so that sum(h) = n_points / 2
    h = h_init * (n_points / 2) / np.sum(h_init)
    
    # Variables for optimization
    h_var = Variable(n_points)
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 5000
    
    # Initialize dual variables
    lambda_vars = np.zeros(n_points * 2 - 1)  # For all overlap terms
    mu = 0
    
    # Timing
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Update h using cvxpy
        # Primal problem with augmented Lagrangian
        h_expr = h_var
        # Overlap terms
        overlaps = []
        for k in range(2 * n_points - 1):
            overlap = 0
            for i in range(n_points):
                if k - i >= 0 and k - i < n_points:
                    overlap += h_expr[i] * (1 - h_expr[k - i])
            overlaps.append(overlap)
        
        # Objective
        obj = Maximize(0)
        constraints = [
            h_expr >= 0,
            h_expr <= 1,
            sum(h_expr) == n_points / 2
        ]
        
        # Add penalty for overlaps
        q = sum([overlap * lambda_vars[k] for k, overlap in enumerate(overlaps)])
        obj = Minimize(-q + rho/2 * sum_squares(h_expr - h))
        
        # Solve the problem
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=OSQP, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h = h_var.value
            else:
                h = np.clip(h + np.random.normal(0, 1e-5, n_points), 0, 1)
        except:
            h = np.clip(h + np.random.normal(0, 1e-5, n_points), 0, 1)
        
        # Compute current overlaps
        overlaps = []
        for k in range(2 * n_points - 1):
            overlap = 0
            for i in range(n_points):
                if k - i >= 0 and k - i < n_points:
                    overlap += h[i] * (1 - h[k - i])
            overlaps.append(overlap)
        current_C5 = np.max(overlaps)
        
        # Update dual variables
        # Update lambda
        lambda_old = lambda_vars.copy()
        for k in range(len(overlaps)):
            lambda_vars[k] += alpha * (current_C5 - overlaps[k])
        
        # Update mu
        mu += alpha * (np.sum(h) - n_points / 2)
        
        # Apply entropy regularization to lambda
        lambda_vars = lambda_vars + beta * np.log(1 + lambda_vars)
        lambda_vars = np.clip(lambda_vars, -1e6, 1e6)
        
        # Update for next iteration
        iteration += 1
        
        # Break if we've found a solution within threshold
        if current_C5 <= 0.3807:
            break
            
    # Final normalization and evaluation
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute final C5 
    dx = 2.0 / n_points
    overlaps_final = []
    for k in range(2 * n_points - 1):
        overlap = 0
        for i in range(n_points):
            if k - i >= 0 and k - i < n_points:
                overlap += h[i] * (1 - h[k - i])
        overlaps_final.append(overlap)
    final_C5 = np.max(overlaps_final) * dx
    
    # Ensure it's valid
    if np.sum(h) != n_points / 2:
        h = h * (n_points / 2) / np.sum(h)
        
    return (h.tolist(), final_C5, n_points)
