import numpy as np
from scipy import signal
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute convolution of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    return np.max(corr) * dx

def project_to_feasible(h, dx):
    """Project h to satisfy integral constraint and bounds"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to sum = 1/dx (since we want integral = 1)
    scale = 1.0 / (dx * np.sum(h))
    h *= scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    
    # Initialize swarm
    swarm_size = 100
    particles = np.random.rand(swarm_size, n_points)
    if 'initial_h_values' in globals():
        particles[0] = initial_h_values
    particles = np.array([project_to_feasible(p, dx) for p in particles])
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.5
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track best solutions
    personal_best = particles.copy()
    personal_best_scores = np.array([evaluate_c5(p, dx) for p in particles])
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    start_time = time.time()
    iterations = 0
    
    while time.time() - start_time < budget_s - 0.1 and iterations < 10000:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], dx)
            
            # Evaluate
            score = evaluate_c5(particles[i], dx)
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = score
                
                # Update global best
                if score < global_best_score:
                    global_best = particles[i].copy()
                    global_best_score = score
        
        iterations += 1
    
    # Ensure final solution is feasible
    final_h = project_to_feasible(global_best, dx)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h, final_c5, n_points)
