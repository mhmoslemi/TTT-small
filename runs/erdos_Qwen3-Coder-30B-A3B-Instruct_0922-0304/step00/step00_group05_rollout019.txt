import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    n = len(h)
    # Compute all correlations
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only the valid lags (from -n+1 to n-1)
    valid_corr = corr[n-1:2*n-1]
    # Scale by dx
    max_overlap = np.max(valid_corr) * dx
    return max_overlap

def project_h(h, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to correct sum
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    # Clip again after scaling
    h = np.clip(h, 0, 1)
    return h

    for i in range(len(particles)):

    # Initialize velocities
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    personal_best = [particles[i].copy() for i in range(swarm_size)]
    personal_best_scores = [evaluate_h(particles[i], dx) for i in range(swarm_size)]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5

        iteration += 1
        
        # Evaluate current particles
        scores = [evaluate_h(particles[i], dx) for i in range(swarm_size)]
        
        # Update personal bests
        for i in range(swarm_size):
            if scores[i] < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = scores[i]
        
        # Update global best
        best_idx = np.argmin(personal_best_scores)
        if personal_best_scores[best_idx] < global_best_score:
            global_best = personal_best[best_idx].copy()
            global_best_score = personal_best_scores[best_idx]
            last_improvement = iteration
            
        # Check for stagnation
        if iteration - last_improvement > 1000:
            # Add some diversity
            for i in range(swarm_size):
                if np.random.rand() < 0.1:
                    particles[i] = np.random.rand(n_points)
                    particles[i] = project_h(particles[i], dx)
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)

            particles[i] = project_h(particles[i], dx)

            if time.time() - start_time > budget_s - 0.1:
