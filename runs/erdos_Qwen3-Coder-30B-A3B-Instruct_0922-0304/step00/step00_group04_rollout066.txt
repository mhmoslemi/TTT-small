import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 42
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
    h = h / np.sum(h) * (n_points / 2)
    
    # FFT settings
    fft_len = 2 * n_points - 1
    dx = 2.0 / n_points
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx) if 'evaluate_erdos_solution' in globals() else float('inf')
    
    # Optimization iterations
    iter_limit = 10000
    for iteration in range(iter_limit):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute Fourier transform
        H = fft.fft(h, fft_len)
        H_conj = np.conj(H)
        
        # Compute |H|^2
        S = np.abs(H)**2
        
        # Update H to flatten S (this is a simplified version)
        # We try to minimize max_k |H(k)(1 - H^*(k))|
        # This is approximated as minimizing max_k |H(k)|^2 * |1 - H^*(k)|^2  
        # In practice, we try to converge towards a uniform distribution in frequency domain
        
        # Simple update rule to try to make |H|^2 uniform
        new_H = H / (np.sqrt(S) + 1e-8)
        new_H[0] = H[0]  # Keep DC component unchanged
        
        # Inverse FFT to get new h
        new_h = np.real(fft.ifft(new_H, fft_len))[:n_points]
        
        # Project into feasible region [0,1] and adjust sum
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / np.sum(new_h) * (n_points / 2)
        
        # Check if this improved the C5
        c5 = evaluate_erdos_solution(new_h, dx) if 'evaluate_erdos_solution' in globals() else float('inf')
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
            
        h = new_h.copy()
        
    return (best_h, best_c5, n_points)
