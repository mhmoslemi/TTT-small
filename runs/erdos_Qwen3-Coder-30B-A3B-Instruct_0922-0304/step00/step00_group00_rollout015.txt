import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default initialization if no initial_h_values is available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    def evaluate(h_vec):
        """Evaluate C5 for current h"""
        dx = 2.0 / len(h_vec)
        overlaps = np.correlate(h_vec, 1 - h_vec, mode="full") * dx
        return np.max(overlaps)
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    alpha = 0.1
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s * 0.9 and iter_count < max_iter:
        # Calculate all overlaps (for finding worst case)
        dx = 2.0 / len(h)
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient for k_max shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + k_max) % len(h)
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Renormalize
        
        # Convex optimization refinement (QP)
        # Formulate a simple QP to improve condition
        if k_max < len(h):
            # Apply QP refinement
            w = cp.Variable(len(h))
            objective = cp.Minimize(cp.sum(w))
            constraints = [
                w[i] >= h[i] * (1 - h[(i + k_max) % len(h)]) * dx
                for i in range(len(h))
            ]
            constraints.append(cp.sum(w) <= best_c5)
            constraints += [0 <= w[i], w[i] <= 1 for i in range(len(h))]
            constraints.append(cp.sum(w) == n_points / 2)
            
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    improved_h = np.array(w.value)
                    improved_h = np.clip(improved_h, 0, 1)
                    improved_h = improved_h * (n_points / 2) / np.sum(improved_h)
                    c5_improved = evaluate(improved_h)
                    if c5_improved < best_c5:
                        best_h = improved_h.copy()
                        best_c5 = c5_improved
            except Exception:
                pass
        
        iter_count += 1
        if iter_count % 10 == 0:
            c5_val = evaluate(h)
            if c5_val < best_c5:
                best_h = h.copy()
                best_c5 = c5_val
            # Adjust learning rate
            if iter_count > 50 and iter_count % 50 == 0:
                alpha = max(0.01, alpha * 0.95)
        
        if iter_count % 50 == 0 and time.time() - start_time > budget_s * 0.8:
            break
    
    # Final validation
    final_c5 = evaluate(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, len(best_h))
