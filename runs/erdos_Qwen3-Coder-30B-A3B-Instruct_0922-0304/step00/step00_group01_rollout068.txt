import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_erdos_solution(h):
    """Evaluates the C5 constant for given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap for all shifts
    overlap = np.correlate(h, 1 - h, mode='full')
    # Take the maximum overlap (excluding the first n-1 elements which are not valid)
    max_overlap = np.max(overlap[n-1:])
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    if 'initial_h_values' in globals():

        n = len(h_initial)

        h_initial = np.full(n, 0.5)
    
    # Ensure sum constraint
    h_initial = h_initial * (n / 2) / np.sum(h_initial)
    
    # Track best solution
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(h_initial)
    
    # Gradient-based optimization
    max_iter = 10000
    iter_count = 0
    start_time = time.time()
    
    # Multi-start with perturbed versions
    initial_hs = [h_initial]
    for _ in range(4):
        perturbed = h_initial + np.random.normal(0, 0.01, n)
        # Clip to [0,1] and rescale
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n / 2) / np.sum(perturbed)
        initial_hs.append(perturbed)
    
    for h_start in initial_hs:
        h = h_start.copy()
        last_improvement = 0
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute current overlaps
            overlap = np.correlate(h, 1 - h, mode='full')[n-1:]
            k_max = np.argmax(overlap)
            current_c5 = overlap[k_max] * (2.0 / n)
            
            # Update best if improved
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                last_improvement = iteration
            
            # Compute gradient for k_max shift only
            shifted = np.roll(1 - h, k_max)
            gradient = shifted - h * shifted
            
            # Normalize gradient for step size
            grad_norm = np.linalg.norm(gradient)
            if grad_norm > 1e-10:
                gradient = gradient / grad_norm
            
            # Backtracking line search
            alpha = 0.005
            rho = 0.5
            max_steps = 20
            
            for _ in range(max_steps):
                h_new = h - alpha * gradient
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n / 2) / np.sum(h_new)
                
                # Evaluate new C5
                new_c5 = evaluate_erdos_solution(h_new)
                
                if new_c5 < current_c5:
                    h = h_new
                    break
                else:
                    alpha *= rho
            
            # Occasionally do convex refinement
            if iteration % 25 == 0:
                # Use active constraint approach to refine for k_max
                k_max = np.argmax(overlap)
                
                # Solving with CVXPY for convex subproblem
                try:
                    import cvxpy as cp
                    h_var = cp.Variable(n)
                    constraints = [
                        h_var >= 0,
                        h_var <= 1,
                        cp.sum(h_var) == n/2
                    ]
                    
                    # Objective: minimize overlap with shift k_max
                    shifted_indices = [(i - k_max) % n for i in range(n)]
                    obj = cp.sum(cp.multiply(h_var, 1 - h_var[shifted_indices]))
                    
                    prob = cp.Problem(cp.Minimize(obj), constraints)
                    prob.solve(solver=cp.GLPK, verbose=False)
                    
                    if prob.status == cp.OPTIMAL:
                        h_refined = np.array(h_var.value).flatten()
                        h = h_refined
                except:
                    pass
            
            # Check if we haven't improved for a while
            if iteration - last_improvement > 1000 and iteration > 1000:
                # Add noise to escape local minima
                noise = np.random.normal(0, 0.001, n)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n / 2) / np.sum(h)
        
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
