import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    # Compute all correlation terms
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take the maximum (excluding full overlap which is not constrained)
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points, dx):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to correct integral
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create a default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Gradient-based optimization
    max_iter = 20000  # Rough estimate of iterations we can do in 600s
    learning_rate = 0.01
    patience = 100
    patience_counter = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C₅
        c5_current = evaluate_c5(h, dx)
        
        # Update best if improved
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Stop if we're stuck
        if patience_counter > patience:
            break
        
        # Compute all correlations
        conv = signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(conv)
        max_positions = np.where(conv == max_corr)[0]
        
        # Compute gradient (for lag indices where corr is max)
        gradient = np.zeros_like(h)
        for pos in max_positions:
            # pos represents the lag in the correlation
            lag = pos - (n_points - 1)  # Convert to lag
            if lag >= 0:
                # Gradient contribution from h[i] term
                gradient[:n_points - lag] += (1 - h[lag:])
            else:
                # Gradient contribution from h[i] term
                gradient[-lag:] += (1 - h[:n_points + lag])
        
        # Apply gradient descent
        h = h - learning_rate * gradient
        
        # Project h back to feasible region
        h = project_h(h, n_points, dx)
        
        # Reduce learning rate if plateauing
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.9
    
    # Final evaluation of best solution
    final_c5 = evaluate_c5(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
