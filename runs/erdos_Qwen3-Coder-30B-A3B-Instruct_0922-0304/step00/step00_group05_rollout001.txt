import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Returns the maximum overlap integral C5 for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (max lag is n_points-1)
    max_corr = np.max(corr[n_points-1:])
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create from scratch
    if 'initial_h_values' in globals():

    else:

        h = np.ones(n_points) * 0.5

        h[1] = 0.7
        h[2] = 1.0
        h[3] = 0.8
        h[4] = 0.2
        h[5] = 0.6
        h[6] = 0.9
        h[7] = 0.4
        h[8] = 0.7
        h[9] = 0.3
        h = np.clip(h, 0, 1)
        h = h * (np.sum(h) / (np.sum(h) * 2.0 / len(h)))
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(budget_s * 10)  # Estimate iterations per second
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    prev_c5 = best_c5 + 1.0
    no_improvement_count = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        
        # Find maximum value and corresponding indices (lags)
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        
        # Identify matching lags where correlation == max_val
        max_lags = np.where(np.abs(corr - max_val) < 1e-8)[0]
        # Need to map lag indices to actual lag values
        lags = max_lags - (n_points - 1)
        
        # Compute gradient with respect to h
        gradient = np.zeros_like(h)
        for lag in lags:
            # Shift for 1-h
            offset = -lag
            if offset > 0:
                # Right shift
                gradient[offset:] += (1 - h[:-offset]) * dx
            elif offset < 0:
                # Left shift
                gradient[:offset] += (1 - h[-offset:]) * dx
            else:
                # No shift
                gradient += (1 - h) * dx

        # Update h
        h -= learning_rate * gradient
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        
        # Scale to meet constraint sum(h) * dx = 1
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Compute loss
        c5 = evaluate_erdos_solution(h)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        # Adaptive learning rate
        if no_improvement_count > 50:
            learning_rate *= 0.9
            no_improvement_count = 0

    return (best_h.tolist(), best_c5, n_points)
