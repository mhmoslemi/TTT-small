import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h_vals):
        # Compute the full cross-correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The maximum is the middle element of the correlation result
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_to_feasible(h_vals):
        # Ensure values are in [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h_vals) * dx
        if abs(current_sum - 1.0) > 1e-10:
            adjust = (1.0 - current_sum) / dx
            # Simple proportional adjust: distribute the error
            h_vals += adjust / n_points
            h_vals = np.clip(h_vals, 0, 1)
        return h_vals
    
    def compute_gradient(h_vals, k):
        # For fixed k, gradient of sum h[i] * (1 - h[i+k]) wrt h[i]
        # This is 1 - h[i+k] for i where i+k is valid
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            j = i + k
            if j < len(h_vals):
                grad[i] = 1.0 - h_vals[j]
        return grad
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Main optimization loop
    iter_count = 0
    while True:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        iter_count += 1
        
        # Evaluate current C5
        current_c5 = evaluate_c5(best_h)
        # Get correlation values
        corr = np.correlate(best_h, 1 - best_h, mode="full")
        # Get the maximum overlapping k (center)
        max_k = np.argmax(corr)
        # Adjust to get actual k (from -n+1 to n-1)   
        k = max_k - (n_points-1)
        
        grad = compute_gradient(best_h, k)
        # Update using gradient descent
        learning_rate = 0.01
        new_h = best_h - learning_rate * grad
        new_h = project_to_feasible(new_h)
        new_c5 = evaluate_c5(new_h)
        
        # Accept if improvement
        if new_c5 < best_c5:

            best_c5 = new_c5

    # Final validation
    final_c5 = evaluate_c5(best_h)
    final_h = project_to_feasible(best_h)
