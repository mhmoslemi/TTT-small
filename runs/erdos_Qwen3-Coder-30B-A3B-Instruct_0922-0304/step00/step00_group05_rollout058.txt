import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable initial guess
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[n_points//4:2*n_points//4] = 0.0
        h[2*n_points//4:3*n_points//4] = 1.0
        h[3*n_points//4:] = 0.0
        # Adjust to meet constraint
        current_sum = np.sum(h)
        scale_factor = target_sum / current_sum
        h *= scale_factor
    
    # Ensure valid bounds
    h = np.clip(h, 0, 1)
    
    # Evaluate initial solution
    def evaluate(h):
        # Compute correlation
        correlation = correlate(h, 1 - h, mode='full')
        # Take maximum
        max_correlation = np.max(np.abs(correlation))
        # Scale by dx
        return max_correlation * dx
    
    # Initial evaluation
    best_h = h.copy()
    current_c5 = evaluate(h)
    best_c5 = current_c5
    
    # Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = 100000
    start_time = time.time()
    
    # Adaptive perturbation
    delta = 0.1
    accept_count = 0
    total_count = 0
    
    # Main loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Adaptive temperature
        progress = elapsed / budget_s
        T = T_max * np.exp(-np.log(T_max / T_min) * progress)
        
        # Create new candidate
        h_new = h.copy()
        
        # Try a perturbation
        action = np.random.choice(['swap', 'shift'])
        if action == 'swap':
            i, j = np.random.choice(n_points, size=2, replace=False)
            # Swap values with probability proportional to their magnitude
            val_i, val_j = h_new[i], h_new[j]
            h_new[i] = val_j
            h_new[j] = val_i
        else:
            # Shift mass between adjacent points
            i = np.random.randint(0, n_points - 1)
            # Reduce one point, increase the next
            change = np.random.uniform(-delta, delta)
            h_new[i] += change
            h_new[i + 1] -= change
            # Ensure bounded
            h_new[i] = np.clip(h_new[i], 0, 1)
            h_new[i + 1] = np.clip(h_new[i + 1], 0, 1)
        
        # Rescale to maintain constraint - ensure sum = target_sum
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-10:
            scale_factor = target_sum / current_sum
            h_new *= scale_factor
            
        # Ensure bounds again after rescaling
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate the new solution
        new_c5 = evaluate(h_new)
        
        # Accept or reject
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
        
        # Update delta based on acceptance rate
        total_count += 1
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            accept_count += 1
        if total_count % 100 == 0:
            acceptance_rate = accept_count / total_count
            if acceptance_rate < 0.2:
                delta *= 0.9
            elif acceptance_rate > 0.4:
                delta *= 1.1
            accept_count = 0
            total_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
