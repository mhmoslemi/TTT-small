import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    N = 45
    pop_size = 100
    max_gens = 1000
    mutation_rate = 0.01
    elite_size = pop_size // 2
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(N) * 0.5
    
    # Ensure constraint: sum(h) * dx = 1 => sum(h) = N/2
    h0 = h0 * (N / 2) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    
    # dx for integration
    dx = 2.0 / N
    
    def evaluate_h(h):
        # Compute overlap integral
        overlaps = correlate(h, 1 - h, mode="full")
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    def is_feasible(h):
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - N/2) < 1e-10)
    
    def create_feasible_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - N/2) > 1e-10:
            # Adjust to make sum = N/2
            adjustment = (N/2 - current_sum) / N
            h = h + adjustment
            h = np.clip(h, 0, 1)
        return h
    
    def mutate(h):
        h = h.copy()
        num_mutations = int(mutation_rate * len(h))
        indices = np.random.choice(len(h), num_mutations, replace=False)
        for i in indices:
            h[i] = np.random.uniform(0, 1)
        return create_feasible_h(h)
    
    def crossover(parent1, parent2):
        # Simple one-point crossover
        point = np.random.randint(1, len(parent1))
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return create_feasible_h(child1), create_feasible_h(child2)
    
    # Initialize population
    population = []
    population.append(h0)
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, N)
        h = create_feasible_h(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h) for h in population]
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    start_time = time.time()
    gen = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and gen < max_gens:
        # Sort population by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > 100:
            break
            
        # Select top 50% for mating
        selected = population[:elite_size]
        
        # Generate new population
        new_population = [best_h]  # Elitism
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        
        # Evaluate new population
        fitness = [evaluate_h(h) for h in population]
        
        gen += 1
    
    # Final check
    final_fitness = [evaluate_h(h) for h in population]
    final_best_idx = np.argmin(final_fitness)
    if final_fitness[final_best_idx] < best_c5:
        best_c5 = final_fitness[final_best_idx]
        best_h = population[final_best_idx].copy()
    
    return (list(best_h), best_c5, N)
