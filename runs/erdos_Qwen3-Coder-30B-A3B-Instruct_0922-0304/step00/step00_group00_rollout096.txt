import numpy as np
import random
import time
from scipy import correlate

def evaluate_h(h, dx):
    # Compute C5 = max_k integral h(x)(1-h(x+k)) dx
    # Using convolution with flipped h
    conv = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (k from 0 to 2)
    # Since h is defined on [0,2], and we're looking at h(x+k)
    # k ranges from 0 to 2, we get correlation for lags from -n to n
    # We only care about the part where correlation corresponds to k in [0,2]
    # Folding: we take the middle n elements, but actually we need 
    #    correlations for k=0 to k=2 over 2*dx intervals
    # Because conv[i] corresponds to lag (i - (n-1))  
    # for i=0,...,2n-2
    # conv with 1-h, so the lags: -n+1,...,n-1
    # We need to extract the k=0 to k=2 part
    # lag 0 corresponds to i = n-1
    # lag 2 corresponds to i = n-1+2 
    # but with 'full', we get indices i = 0 to 2n-2
    # The middle n values, but actually need for k from 0 to 2 in steps of dx
    # we compute:
    # for k in 0 to 2---dx to 2-dx, get value at appropriate i
    # precisely: i ranges from n-1 to 2n-2 for lags 0 to n-1 
    #             That would be indices from n-1 to 2n-2, that's n elements
    # So first n elements are for lags from -n+1 to 0 to n-1.
    # We want to take elements that correspond to lags from 0 to n-1
    # in case n=45, we need to take elements at indices (n-1) to (2n-2) = 44 to 88
    # 45 elements: that's lags from 0 to 44, we need k=0 to 2  
    # But we must account correctly; convolution gives:
    # i=0 (lag=n-1)   to i=2n-2( lag = n-1+2(n-1) ) = 2n-2 
    # i=n-1 (lag=0) to i=2n-2 (lag=n-1) => but we want k in [0,2]
    # Let's quickly compute as for full convolution.
    # Let i=0 to 2n-2 give lags from -(n-1) to (n-1) -- indices 0  to 2n-2
    # So i=n-1 (when lag is 0) to i=2n-2 (when lag is n-1)
    # We want overlapping (k = 0 to 2) 
    # That means we go from index 0 to 2n-2, take:
    # i from n-1 to 2n-2  
    # For k from 0 to 2: k = i - (n-1) for i from n-1 to 2n-2  = 0 to n-1
    # want i from n-1 to 2n-2 = n elements, to get lags 0 to n-1
    # But how to map from kernel size n to desired k range [0,2]
    # kernel size 45: so dx = 2.0/45, k = lags = 0 to 44, 
    # 45 values = (n*ds) = 45 * (2/45) = 2 in terms of x
    # What we want is to extract partition into
    # [0, 2] in steps dx => 0 to 2-dx, step dx
    # which amounts to 0 to 2/dx = 45 (indices)  
    # Or just: take conv, look at indices corresponding to k in [0, 2]
    #---------Start anew---------- 
    # Let's manually compute using:
    # C5 = max_k integral h(x) (1 - h(x+ k)) dx 
    # For a given k, which in the discrete case becomes discrete sum
    # Let's define another function for explicit discrete sum
    # Here, x runs from 0 to 2, step dx = 2.0/n
    # So samples are at x_i = i*dx 
    # To compute for a given k, we compute
    # C5_k = sum_{i=0}^{n-1} h[i] * (1 - h[ (i+k) % n ]) * dx
    # What matters is (i+k) % n = 0,..., n-1  
    # But for a fixed k, this maps to: i=0 to n-1
    # So we have n values like:
    # C_k = sum_{i=0}^{n-1} h[i] * (1-h[ (i+k) % n]) * dx
    # and we want max_k over k in [0, 2] which discretization makes k=0 to n-1
    # So we take the maximum of these n values
    n = len(h)
    max_C5 = 0
    for k in range(n):
        C_k = 0
        for i in range(n):
            C_k += h[i] * (1 - h[(i+k) % n])
        val = C_k * dx
        if val > max_C5:
            max_C5 = val
    return max_C5

def normalize_h(h):
    # Ensure sum = n/2
    current_sum = np.sum(h)
    n = len(h)
    target_sum = n / 2.0
    scaling = target_sum / current_sum if current_sum != 0 else 1.0
    h *= scaling
    return h

    start_time = time.time()

    n = len(h)

    h = normalize_h(h)
    
    # For tracking best solution
    best_h = h.copy()
    old_c5 = evaluate_h(h, dx)
    best_c5 = old_c5

    # Parameters
    T = 0.1
    cooling_rate = 0.995
    delta = 0.1
    adapt_delta = True
    min_delta = 0.001
    max_delta = 0.5
    
    # Perturbation count
    n_perturb = max(1, int(0.05 * n))
    
    # Keep cool down
    T_min = 1e-5
    
    while time.time() - start_time < budget_s - 0.1:
        # Select random indices to perturb
        indices = random.sample(range(n), n_perturb)
        # Make backup
        h_old = h.copy()
        c5_old = old_c5
        
        # Perturb selected indices
        for i in indices:
            # Change by +/- delta
            dh = 2 * (random.random() - 0.5) * delta
            # A small change
            h[i] += dh
            # Clip to bounds
            h[i] = max(0, min(1, h[i]))
        
        # Renormalize
        h = normalize_h(h)
        
        # Compute new C5
        c5_new = evaluate_h(h, dx)
        
        # Accept if better or with probability
        delta_E = c5_new - c5_old
        if delta_E < 0 or random.random() < np.exp(-delta_E / T):
            old_c5 = c5_new
        else:
            # Revert
            h = h_old
            old_c5 = c5_old
        
        # Update best
        if old_c5 < best_c5:
            best_c5 = old_c5
            best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
        if T < T_min:
            T = T_min
        
        # Adaptive delta change
        if adapt_delta:
            if abs(delta_E) < 0.0001:
                delta *= 0.99
            else:
                delta *= 1.01
            delta = max(min_delta, min(max_delta, delta))
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    return (best_h.tolist(), best_c5, n)
