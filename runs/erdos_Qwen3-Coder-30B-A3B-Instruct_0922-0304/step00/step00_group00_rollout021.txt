import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (k from 0 to 2)
    # The full correlation has length 2*n-1
    # We need the part where k=0 to k=2 (corresponding to indices n-1 to 2*n-2)
    start_idx = len(h) - 1
    end_idx = 2 * len(h) - 1
    # Take max of all k values (correlations with all shifts)
    max_corr = np.max(corr[start_idx:end_idx]) * dx
    return max_corr

def renormalize(h, n_points):
    """Ensure sum(h) = n_points/2"""
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    scale = target_sum / current_sum
    h_scaled = h * scale
    return h_scaled

def perturb_h(h, n_points, delta, frac_perturb):
    """Perturb a fraction of h values"""
    h_new = h.copy()
    n_perturb = max(1, int(n_points * frac_perturb))
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        # Add noise
        noise = np.random.uniform(-delta, delta)
        h_new[i] += noise
        # Clamp
        h_new[i] = np.clip(h_new[i], 0, 1)
    # Renormalize
    h_new = renormalize(h_new, n_points)
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    global initial_h_values
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 45
    dx = 2.0 / n_points
    max_iter = 1000000  # Will be limited by time
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = renormalize(h, n_points)
    
    # Current value
    C_current = evaluate_h(h, dx)
    C_best = C_current
    h_best = h.copy()
    
    # Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    delta = 0.1
    frac_perturb = 0.05
    
    # Time tracking
    start_t = time.time()
    iter_count = 0
    
    # Optimization loop
    while time.time() - start_t < budget_s:
        iter_count += 1
        
        # Perturb
        h_new = perturb_h(h, n_points, delta, frac_perturb)
        
        # Evaluate
        C_new = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta_E = C_new - C_current
        if delta_E < 0 or np.random.rand() < np.exp(-delta_E / T):
            h = h_new
            C_current = C_new
            
            if C_current < C_best:
                C_best = C_current
                h_best = h.copy()
        
        # Cool
        T *= cooling_rate
        
        # Early exit
        if C_best < 0.3807:
            break
            
        # Occasionally decrease delta to fine-tune
        if iter_count % 1000 == 0 and delta > 0.001:
            delta *= 0.999
    
    return h_best.tolist(), C_best, n_points
