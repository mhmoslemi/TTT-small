        if c5 < best_c5:

            best_h = h.copy()

                # partial derivative of overlap w.r.t. h[i] is:
                # -1 for terms where (1-h[i]) is in a sum

            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h, best_c5, n_points)
