import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    n_points = len(h)
    dx = 2.0 / n_points
    target = 0.3807
    
    # Ensure initial h is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all overlaps
        h_rev = np.flip(h)
        overlaps = signal.convolve(h, h_rev, mode='full')
        overlaps = overlaps[0:2*n_points-1]
        # Correct for the integral
        overlaps = overlaps * dx
        
        # Find k that gives max overlap
        max_idx = np.argmax(overlaps)
        k_max = max_idx - (n_points - 1)
        
        # Compute gradient for the worst k
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + k_max
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Line search to find optimal step size
        step_size = 0.1
        current_c5 = evaluate_erdos_solution(h)
        
        # Try several step sizes to improve
        for _ in range(10):
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)  # Renormalize
            
            new_c5 = evaluate_erdos_solution(h_new)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                last_improvement = 0
                break
            step_size *= 0.5
            if step_size < 1e-10:
                break
        
        # Early stopping if no improvement
        last_improvement += 1
        if last_improvement > 50 and best_c5 < target:
            break
            
        # Stop if we've found a target solution
        if best_c5 <= target:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    h = np.array(h)
    # Compute all correlations
    n = len(h)
    dx = 2.0 / n
    h_rev = np.flip(h)
    overlaps = np.correlate(h, h_rev, mode='full')
    overlaps = overlaps[0:2*n-1]
    overlaps = overlaps * dx
    return np.max(overlaps)
