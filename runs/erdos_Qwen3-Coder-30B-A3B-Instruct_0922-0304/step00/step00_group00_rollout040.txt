import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Fallback initialization
        n_points = 45
        h = np.full(n_points, 0.5)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    iteration = 0
    alpha = 0.1
    stagnation_count = 0
    last_improvement = best_c5
    
    def compute_overlaps(h_vec):
        # Compute all overlaps using correlation
        h_rev = np.flip(h_vec)
        overlaps = np.correlate(h_vec, h_rev, mode='full')
        return overlaps * dx
    
    def get_k_max(overlaps):
        # Find k that maximizes the overlap
        # Overlaps array corresponds to k = -n+1 to n-1
        max_idx = np.argmax(overlaps)
        k_max = max_idx - (n_points - 1)
        return k_max, overlaps[max_idx]
    
    def gradient_step(h_vec, k_max):
        grad = np.zeros_like(h_vec)
        # For shift k_max, compute gradient
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h_vec[j]) * dx
            grad[j] -= h_vec[i] * dx
        return grad
    
    def project_to_feasible(h_vec):
        # Clip and renormalize
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Optimization loop
    while time.time() - start_time < budget_s * 0.9:
        iteration += 1
        overlaps = compute_overlaps(h)
        k_max, max_overlap = get_k_max(overlaps)
        
        # Gradient descent step with multiple iterations
        for _ in range(5):
            grad = gradient_step(h, k_max)
            h = h - alpha * grad
            h = project_to_feasible(h)
            
        # Convex optimization refinement
        # Relevant for k_max, minimize sum_{i} h[i] such that overlap_k_max <= current
        # This is a more complex constraint, but we will try robust QP
        try:
            # Create QP to ensure achievement of worst case
            h_i = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(h_i))
            
            # Set boundary constraints
            constraints = [
                h_i >= 0,
                h_i <= 1,
                cp.sum(h_i) == n_points / 2,
                cp.sum(cp.multiply(h_i, np.array([1 - h[(i + k_max) % n_points] for i in range(n_points)]))) 
                    <= max_overlap
            ]
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_i.value
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass  # Skip failed QP

        # Evaluate and update if better
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            if stagnation_count % 50 == 0 and alpha > 1e-6:
                alpha *= 0.5
                
        # Early exit if improvement is minimal
        if stagnation_count > 200:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
