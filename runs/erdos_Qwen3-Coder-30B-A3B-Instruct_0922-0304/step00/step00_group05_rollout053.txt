import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available

        # Fallback to a simple construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6

    dx = 2.0 / n_points
    max_iter = int(budget_s * 10)  # Rough estimate
    gamma = 0.01
    eps = 1e-4
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps using convolution
        # h(x) * (1 - h(x + k)) for all k
        # We compute this via convolution
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        h_reflected = np.concatenate([np.zeros(n_points - 1), 1 - h])
        
        # For full convolution, we compute:
        # C_k = sum h[i] * (1 - h[i+k]) for all i
        # This is conv(h, 1 - h) [full mode]
        # But we care about C_k = sum h[i] * (1 - h[i+k]) for valid indices
        
        # Use signal.correlate with mode='full' which computes:
        # result[k] = sum_{i} h[i] * (1 - h[i+k]) for i, k such that 0 <= i < n, 0 <= i+k < n
        
        # Actually, to be precise, we want:
        # C_k = integral h(x) * (1 - h(x+k)) dx
        # Using discrete convolution,
        # with shifting as in standard correlation
        # So we compute:
        # C_k = sum h[i] * (1 - h[i - k]) for valid i
        # Which is equivalent to: sum h[i] * (1 - h[- (k-i)])  
        # Note that h is periodic in a sense, but we're using overlap
        
        # Concretely:
        # in full mode, we compute a matrix with entries:
        #    corr[i][k] = sum_{j} h[j] * h[j - i + k] only if 0 <= j < n and 0 <= j - i + k < n  
        # Let's compute using the standard signal.convolve
        
        conv_result = signal.correlate(h, 1 - h, mode='full')
        # Calculate which lags are valid
        lags = np.arange(-n_points + 1, n_points)
        C_values = conv_result * dx  # Scale correct
        
        # Find maximum value
        max_C = np.max(C_values)
        if max_C < best_C5:
            best_C5 = max_C
            best_h = h.copy()
        
        # Get the indices where max is achieved
        max_indices = np.where(np.abs(C_values - max_C) < 1e-10)[0]
        # Calculate gradient
        gradient = np.zeros(n_points)
        # For each valid lag k, where C_k = max_C
        for k_idx in max_indices:
            # Compute derivative w.r.t. h[i] in the sum:
            # C_k = sum_i h[i] * (1 - h[i-k]) 
            # dC_k / dh[i] = (1 - h[i-k]) if i-k >= 0 and i-k < n  
            # = 1 if i-k is in valid range
            if k_idx < n_points:
                # lag from i to i+k: v = i
                # i-k = 0 => i = k, 
                # i-k = n-1 => i = n-1+k
                lag = k_idx - (n_points - 1)  # Converts index to actual lag shift
                # So correlation at lag = k_idx corresponds to looking at h[i] * (1 - h[i+lag])
                # Gradient contribution to h[i] is (1 - h[i+lag])  
                # But in convolution we have h[i], look at term in (1-h[j])
                # where j = i + lag  
                # So contribution: 1 - h[i + lag] 
                # if 0 <= i + lag < n and 0 <= i < n
                for i in range(n_points):
                    j = i + lag
                    if 0 <= j < n_points:
                        gradient[i] += 1 - h[j]
        
        # Update h
        h_new = h - gamma * gradient
        
        # Projection: clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum = n_points / 2
        current_sum = np.sum(h_new) * dx
        if abs(current_sum - 1.0) > 1e-10 and abs(current_sum) > 1e-10:
            h_new = h_new * (1.0 / current_sum)
        
        h = h_new
        
        # Adaptive learning rate
        if it > 0 and it % 100 == 0:
            gamma *= 0.1
    
    return (best_h, best_C5, n_points)
