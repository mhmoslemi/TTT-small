import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 47  # Use the same size as initial_h_values
    max_iter = 10000
    alpha = 0.01
    min_alpha = 1e-5
    patience = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    lambda_mult = 0.0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(n_points):
            # Compute convolution with flipped 1-h
            shifted_h = np.roll(h, k)
            overlap = np.dot(h, 1 - shifted_h)
            overlaps.append(overlap)
        
        # Find index of maximum overlap
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute subgradient
        g = np.zeros(n_points)
        shifted_h = np.roll(h, k_max)
        for i in range(n_points):
            g[i] = (1 - shifted_h[i]) - h[i] + lambda_mult
        
        # Update h
        h = h - alpha * g
        h = np.clip(h, 0, 1)
        
        # Project back to feasible sum
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        
        # Update Lagrange multiplier
        lambda_mult = lambda_mult + alpha * (np.sum(h) - n_points / 2)
        
        # Compute current C5
        current_c5 = max_overlap / n_points * 2  # Convert back to continuous scale
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive step size
        if iteration > 0 and iteration - last_improvement > patience:
            alpha *= 0.5
            last_improvement = iteration
            if alpha < min_alpha:
                break
    
    # Final evaluation
    # Remap to discrete points using full convolution
    h_extended = np.concatenate([h, h])  # Extend for 'full' correlation
    one_minus_h = 1 - h
    one_minus_h_extended = np.concatenate([one_minus_h, one_minus_h])
    conv_result = correlate(h_extended, one_minus_h_extended, mode='full')
    full_overlap = conv_result[len(h)-1:]
    c5_bound = np.max(full_overlap) * 2.0 / n_points
    c5_bound = max(c5_bound, 0)
    
    return (best_h.tolist(), c5_bound, n_points)
