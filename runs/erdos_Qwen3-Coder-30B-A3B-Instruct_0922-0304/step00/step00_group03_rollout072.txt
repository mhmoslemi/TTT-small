```python
import numpy as np
from scipy import optimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    """Compute C5 from the given h values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all overlaps
    h = np.array(h_values)
    overlaps = np.correlate(h, 1 - h, mode='full')
    # Compute the maximum overlap
    C5 = np.max(overlaps) * dx
    return C5

def initialize_h(n_points, seed=42):
    """Initialize h with a reasonable starting point"""
    np.random.seed(seed)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
    else:
        # Initialize like a step function
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1
        # Add small randomness
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        # Scale to ensure integral constraint
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
    return h

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    start_time = time.time()
    n_points = 70
    # Initialize h 
    h = initialize_h(n_points, seed)
    
    # Convergence tracking
    best_C5 = float('inf')
    best_h = h.copy()
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 10000
    patience = 500
    
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Solve convex subproblem for h
        h_var = Variable(n_points)
        h_vec = np.array(h)
        # Constraint: sum(h) = n_points / 2
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)
        
        # Objective: minimize max_k (h * (1 - h))[k] + (rho/2) * ||h - h_vec||_2^2
        overlaps = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(overlaps)
        
        h_reg = h_var - h_vec
        obj_reg = cp.sum_squares(h_reg) / 2
        objective = cp.Minimize(max_overlap + rho * obj_reg)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                new_h = h_var.value
                h = new_h
        except:
            # Fall back to simple gradient descent if needed
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h = h * scale_factor

        # Update C5 and check if we improved
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            last_improvement = iter_count
            
        if current_C5 < 0.3807:
            break
            
        iter_count += 1
        
        if iter_count - last_improvement > patience:
            break

    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)
    if final_C5 < 0.38085:
        return (list(best_h), final_C5, n_points)
    else:
        return (list(best_h), final_C5, n_points)
```
