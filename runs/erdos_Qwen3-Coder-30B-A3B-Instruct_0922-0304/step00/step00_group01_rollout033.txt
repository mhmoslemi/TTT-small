import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Fallback: symmetric construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Slight perturbation to avoid degeneracy
        h[0] = 0.45
        h[1] = 0.55

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Precompute full cross-correlation indices
    def compute_overlaps(h_vals):
        # Create 1-h as a numpy array
        one_minus_h = 1 - np.array(h_vals)
        # Use numpy's correlate with mode='full'
        corr = np.correlate(one_minus_h, h_vals, mode='full')
        # The overlap for shift k is stored at index k + (n-1)
        # So we extract from index n-1 to 2*n-2 (excluding the last one)
        # The shift indices are from -(n-1) to (n-1)
        k_indices = np.arange(- (n_points - 1), n_points)
        # Extract maximum overlap
        max_overlap = np.max(corr)
        return max_overlap, corr
    
    def project_feasible(h_vals):
        # Clip to [0,1]
        h_clipped = np.clip(h_vals, 0, 1)
        # Adjust to satisfy sum constraint
        current_sum = np.sum(h_clipped)
        if current_sum > 0:
            h_new = h_clipped * (n_points / 2) / current_sum
        else:
            h_new = np.ones(n_points) * 0.5
        return h_new
    
    # Inner convex optimization helper
    def minimize_overlap(h_vals, k_max):
        # Define the objective as a function to minimize
        # Minimize: sum over i of h[i] * (1 - h[(i-k)%n])
        # Recall: shift k means h[i] * (1 - h[i-k])
        # So, for k_max, the overlap is sum over all i of h[i] * (1 - h[(i-k_max)%n])
        def objective(h_vec):
            total = 0.0
            for i in range(n_points):
                j = (i - k_max) % n_points
                total += h_vec[i] * (1 - h_vec[j])
            return total * dx
        
        # Constraints
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        
        # Initial guess
        x0 = h_vals.copy()
        
        # Optimize
        res = minimize(objective, x0, bounds=bounds, constraints=cons, method='SLSQP', options={'maxiter': 100})
        new_h = res.x
        return new_h
    
    # Main loop
    best_h = h.copy()
    best_C5 = float('inf')
    
    iteration = 0
    max_iter = 100
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        # Compute all overlaps
        max_overlap, corr = compute_overlaps(best_h)
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            # Optional: track best h here if needed, or just update at end
        if best_C5 <= 0.3807:
            break  # Target reached

        # Find maximum index
        k_indices = np.arange(- (n_points - 1), n_points)
        max_idx = np.argmax(corr)
        k_max = k_indices[max_idx]

        # Optimize for this shift
        h_opt = minimize_overlap(best_h, k_max)
        # Project to feasible region
        best_h = project_feasible(h_opt)
        
    # Final compute of C5
    final_C5, _ = compute_overlaps(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
