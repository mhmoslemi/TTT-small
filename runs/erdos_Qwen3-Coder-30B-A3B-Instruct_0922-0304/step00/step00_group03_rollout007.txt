import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute correlation
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def generate_random_h(n_points):
    """Generate a random binary vector with exactly n_points/2 ones"""
    h = np.zeros(n_points, dtype=int)
    ones_indices = np.random.choice(n_points, n_points // 2, replace=False)
    h[ones_indices] = 1
    return h

def repair_h(h, n_points):
    """Ensure h has exactly n_points/2 ones"""
    current_sum = np.sum(h)
    target_sum = n_points // 2
    
    if current_sum > target_sum:
        # Remove extra ones
        diff = current_sum - target_sum
        zero_indices = np.where(h == 0)[0]
        if len(zero_indices) >= diff:
            indices_to_flip = np.random.choice(zero_indices, diff, replace=False)
            h[indices_to_flip] = 1
    elif current_sum < target_sum:
        # Add missing ones
        diff = target_sum - current_sum
        one_indices = np.where(h == 1)[0]
        if len(one_indices) >= diff:
            indices_to_flip = np.random.choice(one_indices, diff, replace=False)
            h[indices_to_flip] = 0
    
    return h

            # Perform flips
            neighbor_h[flip_from] = 0
            neighbor_h[flip_to] = 1
            
            # If it's in tabu list, skip
            if tuple(neighbor_h) in tabu_list:
                continue
            else:
                tabu_list.append(tuple(neighbor_h))
                if len(tabu_list) > tabu_tenure:
                    tabu_list.pop(0)  # Remove oldest
                
                # Evaluate neighbor
                neighbor_c5 = evaluate_h(neighbor_h, dx)
                
                # Update best if better
                if neighbor_c5 < best_c5:
                    best_h = neighbor_h.copy()
                    best_c5 = neighbor_c5
                    improved = True
                    no_improvement_count = 0
                else:
                    improved = False
                    no_improvement_count += 1
                    
                # Move to neighbor
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
                
                # Early termination if no improvement in last 25%
                if no_improvement_count > max_iterations // 4:
                    break
        else:
            continue
    
    return best_h, best_c5
