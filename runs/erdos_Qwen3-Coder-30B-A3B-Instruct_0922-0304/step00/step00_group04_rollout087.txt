import numpy as np
from scipy import signal
import cvxpy as cp
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Problem parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if abs(np.sum(h) - target_sum) > 1e-6:
            h = h * target_sum / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # Default initialization

    # Precompute lags to consider
    lag_max = n_points - 1
    lags = list(range(-lag_max, lag_max + 1))
    
    # Find best solution within time budget
    best_h = h.copy()
    best_c5 = np.inf
    
    # Attempt to solve directly with convex optimization
    solver_timeout = min(100, budget_s // 3)
    try:
        h_var = cp.Variable(n_points, name="h")
        C5 = cp.Variable(name="C5")
        
        # Constraints
        constraints = [
            cp.sum(h_var) == target_sum,
            h_var >= 0,
            h_var <= 1,
            C5 >= 0
        ]
        
        # Overlap constraints for each lag
        for k in lags:
            if k == 0:
                # Special case: k=0, correlation is sum(h*(1-h))
                corr = cp.sum(cp.multiply(h_var, 1 - h_var))
            elif k > 0:
                # Positive lag
                corr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
            else:
                # Negative lag
                corr = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:n_points+k]))
            
            constraints.append(corr <= C5)
        
        # Problem
        obj = cp.Minimize(C5)
        prob = cp.Problem(obj, constraints)
        
        # Solve with timeout
        prob.solve(solver=cp.GLPK, max_iters=1000, time_limit=solver_timeout)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h_var.value
            evaluated_c5 = compute_c5(h_opt, dx)
            if evaluated_c5 < best_c5:
                best_c5 = evaluated_c5
                best_h = h_opt.copy()
            
    except Exception as e:
        # Fall back to simple approach if optimization fails
        pass
    
    # Apply additional optimization heuristics if time still allows
    remaining_time = budget_s - (time.time() - start_time)
    
    if remaining_time > 10:
        # Simple gradient-free optimization approach (Nelder-Mead)
        try:
            from scipy.optimize import differential_evolution
            
            def objective(h_vec):
                # Normalize h_vec to target sum
                h_norm = h_vec * target_sum / np.sum(h_vec)
                # Add small penalty if out of bounds
                penalty = 0
                if np.any(h_norm < 0) or np.any(h_norm > 1):
                    penalty = 1e6
                
                c5_val = compute_c5(h_norm, dx)
                return c5_val + penalty
            
            bounds = [(0, 1)] * n_points
            result = differential_evolution(
                objective,
                bounds,
                maxiter=max(50, int(remaining_time * 5)),
                seed=seed,
                popsize=10,
                disp=False
            )
            
            if result.success:
                h_final = result.x * target_sum / np.sum(result.x)
                computed_c5 = compute_c5(h_final, dx)
                if computed_c5 < best_c5:
                    best_c5 = computed_c5
                    best_h = h_final
        except:
            # If optimization fails, just use the best solution so far
            pass
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * target_sum / np.sum(best_h)
    
    # Double check final c5
    final_c5 = compute_c5(best_h, dx)
    
    return (list(best_h), final_c5, n_points)

def compute_c5(h, dx):
    corr_full = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr_full)
    return max_corr * dx
