import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        current_sum = np.sum(h_init)
        if abs(current_sum - target_sum) > 1e-6:
            # Repair sum constraint by flipping some values
            need_flip = int(abs(current_sum - target_sum))
            indices = np.arange(n_points)
            flip_indices = np.random.choice(indices, size=need_flip, replace=False)
            h_init[flip_indices] = 1 - h_init[flip_indices]
    else:
        # Default initialization
        h_init = np.ones(n_points) * 0.5  # Start with uniform half

    # Binary search bounds
    t_low = 0.0
    t_high = 0.5
    tolerance = 1e-4

    best_h = h_init.copy()
    best_c5 = float('inf')

    # Binary search loop
    while t_high - t_low > tolerance and (time() - start_time) < budget_s - 1:
        t_mid = (t_low + t_high) / 2.0
        
        # Optimization variables
        h = cp.Variable(n_points)
        w = [cp.Variable((n_points, n_points), nonneg=True) for _ in range(2*n_points-1)]
        
        # Constraints
        constraints = []
        constraints.append(cp.sum(h) == target_sum)
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        
        # Overlay constraints
        for k in range(-n_points+1, n_points):
            if k >= 0:
                shift_idx = k
                # Compute indices for h[i] * h[i+k] terms
                i_indices = np.arange(n_points - k)
                k_indices = i_indices + k
                # Use predicate to aggregate the sum
                # We compute the correlation using the definition:
                # sum over i of h(i) * (1 - h(i+k)) = sum h(i) - sum h(i)*h(i+k)
                # So we need to model h(i)*h(i+k) with variables w[i,k]
                tmp = cp.sum(h[i_indices] - w[shift_idx][i_indices, k_indices])
                constraints.append(tmp <= t_mid)
            else:
                shift_idx = n_points + k - 1
                # same logic for negative k
                i_indices = np.arange(n_points + k)
                k_indices = i_indices - k
                tmp = cp.sum(h[i_indices] - w[shift_idx][i_indices, k_indices])
                constraints.append(tmp <= t_mid)
        
        # Objective: minimize t (but we're solving a feasibility problem)
        # In this case, the constraint reflects the minimum t
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s - (time() - start_time))
            if prob.status == cp.OPTIMAL or prob.status == cp.SOLUTION_PRESENT:
                h_sol = h.value
                if np.all(h_sol >= 0) and np.all(h_sol <= 1):
                    h_bin = np.round(h_sol)
                    # Repair sum
                    h_sum = np.sum(h_bin)
                    diff = int(target_sum - h_sum)
                    if diff != 0:
                        # Flip minimal number of entries to adjust sum
                        indices = np.argsort(np.abs(h_sol - 0.5))[:abs(diff)]
                        if diff > 0:
                            h_bin[indices] = 1
                        else:
                            h_bin[indices] = 0
                    temp_c5 = np.max(np.correlate(h_bin, 1-h_bin, mode='full')) * dx
                    if temp_c5 < best_c5:
                        best_c5 = temp_c5
                        best_h = h_bin.copy()
                    t_high = t_mid
                else:
                    t_low = t_mid
            else:
                t_low = t_mid
        except Exception:
            t_low = t_mid
    
    # Final evaluation of best solution found
    # Evaluate the final solution
    final_h = best_h
    try:
        c5 = np.max(np.correlate(final_h, 1-final_h, mode='full')) * dx
        if c5 < best_c5:
            best_c5 = c5
    except:
        pass

    return (final_h.tolist(), best_c5, n_points)
