import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    dx = 2.0 / len(h)
    overlaps = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():

    else:

        h[:n_points//2] = 1.0
        
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    patience = 0
    
    while time.time() - start_time < budget_s - 1 and patience < 50:
        # Evaluation
        current_C5 = evaluate_erdos_solution(h)
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            patience = 0
        else:
            patience += 1
            
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        
        # For convex optimization, we need to use the current solution to guide target
        overlaps = correlate(h, 1 - h, mode="full") * dx
        objective = cp.Minimize(cp.max(overlaps))  # Must compute with actual values
        
        # Solve the program
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=0.5)
        except:
            pass  # Fallback to simple projection if needed
            
        # Get relaxed solution
        if prob.status == cp.OPTIMAL:
            h_new = h_relaxed.value
        else:
            # Fallback: keep current h if no solution
            h_new = h.copy()
            
        # Project back to feasible region
        # Threshold and adjust for sum constraint
        h_new = np.clip(h_new, 0, 1)
        diff = n_points/2 - np.sum(h_new)
        if abs(diff) > 1e-8:
            # Use a simple adjustment method
            if diff > 0:
                # Need to increase values
                idx = np.argsort(h_new)
                for i in range(int(diff)):
                    if h_new[idx[i]] < 1:
                        h_new[idx[i]] = min(1, h_new[idx[i]] + 1e-6)
                        h_new[idx[i]] = min(1, h_new[idx[i]])
            else:
                # Need to decrease values
                idx = np.argsort(-h_new)
                for i in range(int(-diff)):
                    if h_new[idx[i]] > 0:
                        h_new[idx[i]] = max(0, h_new[idx[i]] - 1e-6)
        
        # Ensure sum is exactly n_points/2
        if abs(np.sum(h_new) - n_points/2) > 1e-6:
            # Do a simple thresholding search
            sorted_indices = np.argsort(h_new)[::-1]
            target_sum = n_points / 2
            current_sum = np.sum(h_new)
            if current_sum > target_sum:
                for i in range(int(current_sum - target_sum)):
                    h_new[sorted_indices[i]] = 0
            else:
                for i in range(int(target_sum - current_sum)):
                    h_new[sorted_indices[i]] = 1

        # Final adjustment to get exact sum
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-6:
            diff = target_sum - current_sum
            if diff > 0:
                indices = np.argsort(h_new)
                for i in indices:
                    if h_new[i] < 1 and diff > 0:
                        step = min(1 - h_new[i], diff)
                        h_new[i] += step
                        diff -= step
            else:
                indices = np.argsort(-h_new)
                for i in indices:
                    if h_new[i] > 0 and diff < 0:
                        step = max(0, diff + h_new[i])
                        h_new[i] -= step
                        diff += step

        h = h_new
        
        # Perturb slightly
        if patience > 0:

            to_change = max(1, int(n_points * 0.05))
            indices = np.random.choice(n_points, to_change, replace=False)
            for i in indices:
                h[i] = np.clip(h[i] + np.random.normal(0, 0.02), 0, 1)
                
        # Re-normalize to sum exactly
        h = h * (n_points / 2) / np.sum(h)

    return (best_h.tolist(), best_C5, n_points)
