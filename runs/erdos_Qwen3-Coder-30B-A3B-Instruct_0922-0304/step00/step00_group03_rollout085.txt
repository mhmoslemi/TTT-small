import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
    
    # If we start with a valid h0, we can make a reasonable guess
    dx = 2.0 / n_points
    
    def evaluate(h):
        if abs(np.sum(h) * dx - 1.0) > 1e-6:
            h = h * (1.0 / (np.sum(h) * dx))  # normalize
        h = np.array(h)
        # Compute all correlations
        corr = np.correlate(h, 1-h, mode="full")
        # Take only the relevant shifts (k in [-n_points+1, n_points-1])
        relevant_indices = slice(n_points-1, 2*n_points-2)
        max_corr = np.max(corr[relevant_indices]) * dx
        return max_corr
    
    # Binary search on the bound
    tol = 1e-4
    low = 0.0
    high = 0.5
    best_h = None
    best_c5 = float('inf')
    
    # Max iterations to prevent hanging
    max_iter = 30
    iter_count = 0
    
    while iter_count < max_iter and high - low > tol:
        t = (low + high) / 2.0
        if time() - start_time > budget_s * 0.95:
            break
        iter_count += 1
        
        # Build CVXPY optimization problem
        h = cp.Variable(n_points)
        w = cp.Variable((n_points, 2*n_points - 1))
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2.0
        ]
        
        # Add overlap constraints
        # For each shift k in [-n_points+1, n_points-1]
        for k in range(-n_points+1, n_points):
            # Compute the corresponding index in the full correlation
            shift_index = k + n_points - 1
            # Start indices in h and 1-h that are involved in overlap
            if k >= 0:
                i_start = 0
                j_start = k
                length = n_points - k
            else:
                i_start = -k
                j_start = 0
                length = n_points + k
            
            # Using the relaxation from the strategy:
            # w[i, shift_index] represents h[i] * h[i + k]
            # The sum is <= t
            overlap_sum = 0
            for i in range(length):
                # Add constraint that w[i, shift_index] <= h[i_start + i]
                constraints.append(w[i, shift_index] <= h[i_start + i])
                # Add constraint that w[i, shift_index] <= h[j_start + i]
                constraints.append(w[i, shift_index] <= h[j_start + i])
                overlap_sum += w[i, shift_index]

            # The constraint: sum(h[i]) - sum(w[i]) <= t => sum(h[i]) - sum(w[i]) - t <= 0
            # We compute sum(h[i]) trivially as 1.0 for a normalized h
            # We compute sum(w[i]) from the vector w for this shift
            sum_h = cp.sum(h[i_start:i_start+length])
            #sum_h = 1.0  # No, it's sum(h[i]) * dx = 1.0
            # More accurate way: instead of using 1.0 for sum(h) in overlap,
            # we just sum the entries over which we're overlapping
            # The overlap integral contribution is (sum_h) - sum(w_i)
            # So we enforce  sum_h - sum(w_i) - t <= 0 or sum(w_i) - sum_h + t >= 0
            # But we want it to be <= t, i.e. sum_h - sum(w_i) <= t, which is sum(w_i) >= sum_h - t
            # Let's tighten this by adding the constraint directly:
            # sum(h) - sum(w) <= t (but must be careful with indices)
            # We are looking at the sum over valid indices for overlap
            corsum = cp.sum(w[:length, shift_index])
            # The effect of 1-h is 1 - h[i], so the dot product is
            # sum(h[i] * (1 - h[i+k])) = sum(h[i]) - sum(h[i]*h[i+k])
            # So we can define eulerian overlap sum = sum(h[i]) - sum(w[i])
            # For shift k, we formulate sum(w[i]) <= sum(h[i]) - t:
            # sum(w[i]) + t <= sum(h[i])
            constraints.append(corsum - sum_h + t <= 0)
        
        # Objective is to minimize t, so we just check feasibility
        # Note that t is not a separate variable— it's used to define constraints. 
        # We'll keep the same t in binary search and just check if we can satisfy
        # all constraints using a CVXPY solver.
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_val = h.value
                c5_val = evaluate(h_val)
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_val.copy()
                high = t
            else:
                low = t
        except Exception as e:
            print(f"Problem solving failed: {e}")
            low = t
    
    # Check final solution
    if best_h is None:
        best_h = h0
    
    # Final refinement step
    # Try to minimize using Nonlinear Programming if still needed, but for now just
    # use the best approximation from CVXPY
    final_h = best_h if best_h is not None else h0
    final_c5 = evaluate(final_h)
    
    # Refine the binary solution if needed
    # Strip any numerical inaccurancies
    final_h = np.clip(final_h, 0, 1)
    
    # Repair the mean to be exactly 1.0
    sum_h = np.sum(final_h)
    if abs(sum_h * dx - 1.0) > 1e-6:
        # Adjust to sum to n_points / 2 exactly
        scale = (n_points / 2.0) / sum_h
        final_h = final_h * scale
    
    # Convert to binary rounding by rounding all values to 0 or 1
    final_h = np.round(final_h)
    # Repair sum
    actual_sum = np.sum(final_h)
    diff = actual_sum - n_points / 2.0
    if abs(diff) > 0.5:
        # flip some values to repair the sum
        # make smallest differences invert
        diff = int(round(diff))
        if diff > 0:
            # too many 1s, flip some 1s to 0
            flip_indices = np.argsort(final_h)[::-1][:diff]
            final_h[flip_indices] = 0
        else:
            # too few 1s, flip some 0s to 1
            flip_indices = np.argsort(final_h)[:abs(diff)]
            final_h[flip_indices] = 1
    
    # Final evaluation
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
