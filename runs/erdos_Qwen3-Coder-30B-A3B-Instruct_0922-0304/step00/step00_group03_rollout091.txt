import numpy as np
import time
from scipy import signal

    # Parameters
    n_points = 70
    dx = 2.0 / n_points

    max_iter = 5000
    alpha = 0.01
    beta = 0.9
    decay_every = 100
    min_alpha = 1e-6
    
    # Initialize h

    else:

    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Initialize momentum
    momentum = np.zeros_like(h)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        # Take only the relevant part (shifts from -n+1 to n-1)
        full_corr = corr[len(corr)//2:]
        
        # Find maximum overlap (excluding the 0 shift which is not relevant)
        max_idx = np.argmax(full_corr[1:]) + 1
        max_overlap = full_corr[max_idx]
        
        # Compute gradient: for each position i, 
        # contribution to gradient from overlap at shift max_idx
        grad = np.zeros_like(h)
        for j in range(len(h)):
            if j + max_idx < len(h):
                grad[j] += (1 - h[j + max_idx])
        
        # Update with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h_new = h - alpha * momentum
        
        # Project back onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Decay learning rate
        if (i + 1) % decay_every == 0:
            alpha = max(alpha * 0.9, min_alpha)
    
    # Final computation of C5
    corr = signal.correlate(h, 1 - h, mode='full')
    full_corr = corr[len(corr)//2:]
    final_c5 = np.max(full_corr[1:])
    
    return (h.tolist(), final_c5, n_points)
