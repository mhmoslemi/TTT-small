import numpy as np
from scipy.optimize import minimize
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define the initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable initial guess (symmetric step function)
        n_points = 42
        h = np.ones(n_points) * 0.5
        # Make it a step function
        for i in range(n_points):
            if i < n_points // 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Ensure it sums to 2.0 (since dx=2/n_points, sum(h)*dx=1)
        h = h * (n_points / 2.0) / np.sum(h)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)

    # Optimization stages
    # Stage 1: Gradient descent refinement
    # We'll iterate with gradient-based optimization
    max_iter = 1000
    learning_rate = 0.1
    iter_count = 0

    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute C5 value
        current_c5 = compute_c5(h, dx)
        # Get the dominant lag
        lags, correlation_values = compute_correlations(h, dx)
        dominant_lag_idx = np.argmax(correlation_values)
        dominant_lag = lags[dominant_lag_idx]
        
        # Compute gradient ascent direction for max correlation
        grad = compute_gradient_for_lag(h, dominant_lag, dx)
        h_new = h + learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2.0) / np.sum(h_new)
        h_new *= scale_factor
        
        # Update if better
        new_c5 = compute_c5(h_new, dx)
        if new_c5 < current_c5:
            h = h_new.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                break

        iter_count += 1

    # Stage 2: Try larger resolution if time allows
    if time.time() - start_time < budget_s - 2:
        # Resample to 84 points
        n_points_large = 84
        dx_large = 2.0 / n_points_large
        h_large = np.interp(np.linspace(0, 2, n_points_large),
                            np.linspace(0, 2, len(h)),
                            h)
        h_large = np.clip(h_large, 0, 1)
        # Make sum correct
        h_large *= (n_points_large / 2.0) / np.sum(h_large)

        # Optimize this one too
        best_large_c5 = compute_c5(h_large, dx_large)
        best_large_h = h_large.copy()

        for _ in range(100):
            if time.time() - start_time >= budget_s - 1:
                break
            lags, correlation_values = compute_correlations(h_large, dx_large)
            dominant_lag_idx = np.argmax(correlation_values)
            dominant_lag = lags[dominant_lag_idx]
            grad = compute_gradient_for_lag(h_large, dominant_lag, dx_large)
            h_large_new = h_large + 0.1 * grad
            h_large_new = np.clip(h_large_new, 0, 1)
            scale_factor = (n_points_large / 2.0) / np.sum(h_large_new)
            h_large_new *= scale_factor
            new_c5 = compute_c5(h_large_new, dx_large)
            if new_c5 < best_large_c5:
                h_large = h_large_new.copy()
                best_large_c5 = new_c5
                best_large_h = h_large.copy()
            else:
                break

        # Choose between 42 or 84 point versions
        if best_large_c5 < best_c5:
            best_h = best_large_h.copy()
            best_c5 = best_large_c5
            n_points = n_points_large
            dx = dx_large

    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # compute overlap integral for given h
    n_points = len(h)
    # Compute correlations for all lags
    correlations = np.correlate(h, 1-h, mode='full')
    # Slice to the actual values (not the full correlation)
    center = len(correlations) // 2
    actual_correlations = correlations[center : center + n_points]
    # Scale by dx
    actual_correlations *= dx
    # Return maximum
    return np.max(actual_correlations)

def compute_correlations(h, dx):
    # returns lags and correlation values
    n_points = len(h)
    all_correlations = np.correlate(h, 1-h, mode='full')
    # domains of cross correlation
    lags = np.arange(-n_points + 1, n_points)
    # center value corresponds to no lag
    center = len(all_correlations) // 2
    valid_correlations = all_correlations[center - (n_points-1) : center + n_points]
    return lags[center-n_points+1:center+n_points], valid_correlations * dx

def compute_gradient_for_lag(h, k, dx):
    # calculate the gradient w.r.t. h for the k-th lag
    n = len(h)
    grad = np.zeros_like(h)
    
    # Evaluate gradient for all positions according to convolution
    # Contribution is from lags k..k+n-1, from index 0..n-1
    if k >= 0:  
        # We're looking at h[i] * (1 - h[i+k]) for indices valid
        # i from k to n-1
        for i in range(k, n):
            grad[i] -= dx  # due to derivative of 1-h[i+k] 
    else:  
        # k < 0
        # We're looking at h[i] * (1 - h[i+k]) for i from 0 to n+ k - 1
        # i from 0 to n + k - 1
        for i in range(n + k):
            grad[i] -= dx  # derivative of 1-h[i+k]

    return grad
