import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < 0.95 * budget_s:
        iter_count += 1
        
        # Compute all overlaps with full convolution
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Constrain this specific shift
        # Let's set up the constraint that this overlaps must be <= C5
        # We make a convex relaxation of this constraint:
        # sum(h[i] * (1 - h[(i+k)%n])) <= C5 => sum(h[i] - h[i]*h[(i+k)%n]) <= C5
        # so we need to linearize or bound the quadratic term.
        
        # For this approach, we try direct quadratic constraint handling via CVXPY
        # Create variables for optimization
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Define constraints
        constraints = [
            0 <= h_var,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            c5_var >= 0
        ]
        
        # Add the specific constraint using CVXPY to handle quadratic terms
        # We'll build a quadratic constraint manually and do it exactly
        # Specifically, we take the effective constraint at k_max
        k_shift = k_max - (n_points - 1)
        # h[i] * (1 - h[i + k]) should be bounded
        # The constraint we add is:
        # sum_{i} h[i] * (1 - h[(i+k)%n]) <= C5 
        if k_shift >= 0:
            to_add = 0
            for i in range(n_points):
                j = (i + k_shift) % n_points
                to_add += h_var[i] * (1 - h_var[j])
        else:
            to_add = 0
            for i in range(n_points):
                j = (i + k_shift + n_points) % n_points
                to_add += h_var[i] * (1 - h_var[j])
        
        constraints.append(to_add <= c5_var)
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Solve problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                new_h = h_var.value
                if new_h is not None:
                    h = np.clip(new_h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    c5 = prob.value
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
            else:
                # If this fails, we can reset to a previous good solution
                pass
        except:
            pass
            
        # Early exit conditions
        if best_c5 <= 0.3807:
            break
        if no_improvement_count > max_no_improvement:
            break
        
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, n_points)
        
def evaluate_erdos_solution(h):
    """Evaluate the C5 bound given a step function h"""
    h = np.array(h)
    n = len(h)
    # The overlap integral for shift k is 
    # integral_0^2 h(x) * (1 - h(x + k)) dx
    # Discretized with dx = 2.0 / n
    dx = 2.0 / n
    overlaps = []
    for k in range(n):  # k from 0 to n-1 (shifts)
        # Compute overlap for shift k
        # h[i] * (1 - h[i+k])
        shifts = np.roll(h, -k)
        overlap = np.sum(h * (1 - shifts)) * dx
        overlaps.append(overlap)
    return max(overlaps)
