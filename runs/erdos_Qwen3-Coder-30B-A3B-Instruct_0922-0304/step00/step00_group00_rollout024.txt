import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.full(n_points, 0.5)
        
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize temperature and cooling
    T = 0.1
    cooling_rate = 0.995
    delta = 0.1
    
    # Normalize to ensure sum(h) * dx = 1 (i.e. sum(h) = n_points / 2)
    target_sum = n_points / 2.0
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        h *= target_sum / h_sum
        
    def evaluate(h):
        # Compute full correlation of h with (1 - h)
        corr = np.correlate(h, 1 - h, mode="full")
        # The maximum is taken over all lags, scaled by dx
        max_val = np.max(corr) * dx
        return max_val
        
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Perturb some indices
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Create a copy to try the perturbation
        h_new = h.copy()
        
        # Apply small random changes
        for i in indices:
            perturb = np.random.uniform(-delta, delta)
            h_new[i] += perturb
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to sum(h) = n_points / 2
        h_new_sum = np.sum(h_new)
        h_new *= target_sum / h_new_sum
        
        # Evaluate
        c5_new = evaluate(h_new)
        
        # Accept or reject based on Metropolis criterion
        diff = c5_new - best_c5
        if diff < 0 or np.random.rand() < np.exp(-diff / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                
        # Cool down
        T *= cooling_rate
        
        if best_c5 < target_bound:
            break
            
    return (best_h.tolist(), best_c5, n_points)
