import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    overlap = correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(overlap[len(h_values)-1:])
    return max_overlap * dx

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    no_improvement_count = 0
    
    # Parameters
    max_iter = 10000
    step_size = 0.05
    peak_count = 5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1 - h, mode='full')
        overlap_vals = overlap[n_points-1:]
        
        # Identify top peaks
        peak_indices = np.argsort(overlap_vals)[::-1][:peak_count]
        peak_vals = overlap_vals[peak_indices]
        
        # Create update vector
        update = np.zeros(n_points)
        
        # For each peak, adjust h to reduce that overlap
        for k in peak_indices:
            for i in range(n_points):
                j = (i + k) % n_points
                if i < n_points - k:
                    # h[i] and h[j] contribute to overlap at lag k
                    # Increase h[i] and decrease h[j] to reduce the product
                    update[i] += step_size * (1 - 2*h[i])
                    if j < n_points:
                        update[j] -= step_size * (1 - 2*h[j])
        
        # Apply update with bounds
        h_new = h + update
        h_new = np.clip(h_new, 0, 1)
        
        # Project to satisfy sum constraint
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
        else:
            h_new = np.full(n_points, 1.0 / n_points)
        
        # Evaluate new solution
        c5_new = evaluate_erdos_solution(h_new, n_points)
        
        # Accept if better
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early exit if target reached
        if best_c5 < 0.3807:
            break
            
        if no_improvement_count > 100:
            # Random perturbation
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * target_sum / np.sum(h)
            no_improvement_count = 0

    return (best_h.tolist(), best_c5, n_points)
