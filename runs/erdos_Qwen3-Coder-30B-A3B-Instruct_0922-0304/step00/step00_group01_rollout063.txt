import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iterations = 1000
    line_search_max_steps = 30
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Ensure sum(h) = n_points / 2
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate_c5(h_vec):
        # Compute overlap for all shifts
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = corr[n_points-1:] * dx
        return np.max(overlaps)
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for shift k_max
        grad = np.zeros_like(h_vec)
        # For each i, compute partial derivative of C(k_max) w.r.t. h[i]
        # grad[i] = d/dh[i] int h(x)(1-h(x+k_max)) dx
        # For i in range(n_points), compute h[i] * (1 - h_clipped[i+k_max])
        # and same for i-k_max where it makes sense
        for i in range(n_points):
            # Terms involving h[i] in the integral:
            # - Take derivative of h[x]* (1-h[x+k_max]) w.r.t. h[i]
            # For window 0 to 2, terms are:
            # h[i] * (1-h[i+k_max]) if i+k_max < n_points
            # and 0 otherwise
            if i + k_max < n_points:
                grad[i] += (1 - h_vec[i + k_max])
        return grad
    
    def line_search(h_vec, grad_vec, k_max, alpha_max=1.0):
        alpha = alpha_max
        for _ in range(line_search_max_steps):
            new_h = h_vec - alpha * grad_vec
            # Clip to [0, 1]
            new_h = np.clip(new_h, 0, 1)
            # Rescale to maintain sum = n_points / 2
            new_sum = np.sum(new_h)
            scaling = (n_points / 2) / new_sum
            new_h *= scaling
            # Evaluate C5 for this step
            new_c5 = evaluate_c5(new_h)
            old_c5 = evaluate_c5(h_vec)
            if new_c5 < old_c5:
                return new_h, new_c5
            alpha *= rho
        return h_vec, evaluate_c5(h_vec)
    
    iteration = 0
    no_improvement_count = 0
    
    while iteration < max_iterations:
        if time.time() - start_time > budget_s - 1:
            break
            
        C5_current = evaluate_c5(h)
        if C5_current < best_c5:
            best_c5 = C5_current
            best_h = h.copy()
            
        if best_c5 < target_bound:
            break
            
        # Find shift k that causes maximum overlap
        corr = correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        
        # Compute gradient for k_max
        grad = compute_gradient(h, k_max)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm == 0:
            break  # No progress possible
        grad /= grad_norm
        
        # Line search
        new_h, new_c5 = line_search(h, grad, k_max)
        
        if new_c5 >= C5_current:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break
        else:
            no_improvement_count = 0
            h = new_h
            
        iteration += 1
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    # Clip h to [0,1] and rescale
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    if best_h_sum > 0:
        best_h = best_h * (n_points / 2) / best_h_sum
    
    return (best_h.tolist(), best_c5, n_points)
