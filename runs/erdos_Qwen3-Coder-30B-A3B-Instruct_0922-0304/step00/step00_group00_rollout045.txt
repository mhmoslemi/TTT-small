import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Start with the default
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Ensure sum(h) = n_points / 2
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    
    # Compute initial C5
    def compute_c5(h, dx):
        h_rev = np.flip(h)
        correlation = signal.convolve(h, h_rev, mode='full')
        max_corr = np.max(correlation)
        return max_corr * dx
    
    current_c5 = compute_c5(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    perturbation_fraction = 0.05
    perturb_steps = max(1, int(perturbation_fraction * n_points))
    delta = 0.1
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Perturb some indices
        indices = np.random.choice(n_points, size=perturb_steps, replace=False)
        old_h = h.copy()
        old_sum = np.sum(h)
        
        # Apply random perturbations
        for i in indices:
            change = np.random.uniform(-delta, delta)
            h[i] += change
            h[i] = np.clip(h[i], 0, 1)
        
        # Renormalize
        new_sum = np.sum(h)
        if new_sum > 0:
            h *= (old_sum / new_sum)
        
        h = np.clip(h, 0, 1)
        
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        
        # Accept or reject
        if new_c5 < current_c5:
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            # Accept with probability
            delta_e = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta_e / temp):
                current_c5 = new_c5
            else:
                h = old_h.copy()
        
        # Cool down
        temp *= cooling_rate
        
        iteration += 1
        if best_c5 < target_bound:
            break
    
    return (best_h.tolist(), best_c5, n_points)
