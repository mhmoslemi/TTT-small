import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    # This is a placeholder; in practice, this would be imported
    n = len(h_values)
    dx = 2.0 / n
    overlap = signal.correlate(h_values, 1 - np.array(h_values), mode='full')
    max_overlap = np.max(overlap[n-1:]) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 47
    max_iter = 10000
    k_top_peaks = 5
    alpha = 0.05
    tolerance = 1e-6
    min_c5_target = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improve_count = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        dx = 2.0 / n_points
        overlap = signal.correlate(h, 1 - h, mode='full')
        overlap_vec = overlap[n_points-1:] * dx
        
        # Find top peaks
        peak_indices = np.argsort(overlap_vec)[::-1][:k_top_peaks]
        
        # Make a copy of h to update
        h_new = h.copy()
        
        # Update h based on the peaks
        for k in peak_indices:
            if k == 0:
                continue
            
            # Adjust h where both h[i] and h[i+k] contribute to overlap
            for i in range(n_points - k):
                if h[i] > 0 and h[i + k] > 0:
                    # Reduce both h[i] and h[i+k] to decrease overlap
                    h_new[i] = max(0, h_new[i] - alpha * 0.5)
                    h_new[i + k] = max(0, h_new[i + k] - alpha * 0.5)
        
        # Project h_new back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        
        # Accept improvement
        if new_c5 < best_c5 - tolerance:
            h = h_new
            best_c5 = new_c5
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early stop conditions
        if no_improve_count > 100 or best_c5 < min_c5_target:
            break
            
    return (best_h.tolist(), best_c5, n_points)
