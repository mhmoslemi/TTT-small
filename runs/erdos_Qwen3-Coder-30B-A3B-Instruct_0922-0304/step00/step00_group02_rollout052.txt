import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a reasonable starting point
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_c5(h_vals):
        # Compute the maximum overlap integral
        # h_vals * (1 - h_vals) convolution with itself
        # but we want max_k int h(x)(1 - h(x+k)) dx
        # This is equivalent to max_k int h(x) - h(x)*h(x+k) dx
        # Which is max_k [int h(x) dx - int h(x)*h(x+k) dx] 
        # = max_k [1 - int h(x)*h(x+k) dx]
        # We compute the full cross-correlation of h with 1-h
        # The cross-correlation at lag k is int h(x) (1 - h(x+k)) dx
        # This is equivalent to: corr = correlate(h, 1-h, mode='full')
        # Step-by-step formation:
        corr = correlate(h, 1 - h, mode='full')
        # The central part corresponds to lags in [-n+1, n-1]
        # The result length is 2*n - 1
        # Central value is at index n-1
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vals):
        # Project h onto feasible set
        # Maintain sum constraint and bounds [0,1]
        h_proj = h_vals.copy()
        h_proj = np.clip(h_proj, 0, 1)
        current_sum = np.sum(h_proj)
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust values to meet sum constraint
            # Simple strategy: scale to meet sum constraint
            # But keep values clipped
            if current_sum > 0:
                h_proj = h_proj * target_sum / current_sum
                h_proj = np.clip(h_proj, 0, 1)
        return h_proj
    
    def update_h(h_vals, i, j, delta):
        # Update h by transferring delta from i to j
        # i: increase, j: decrease
        h_new = h_vals.copy()
        h_new[i] += delta
        h_new[j] -= delta
        return h_new
    
    def valid_h(h_vals):
        # Check if h satisfies constraints
        return (np.all(h_vals >= 0) and 
                np.all(h_vals <= 1) and 
                abs(np.sum(h_vals) - target_sum) < 1e-10)
    
    # Compute initial
    start_time = time.time()
    c5_current = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_current
    elapsed = time.time() - start_time
    
    # Iteration counter
    iter_count = 0
    delta = 0.1
    decay_rate = 0.9
    iter_since_decay = 0
    
    while elapsed < budget_s:
        iter_count += 1
        if iter_since_decay >= 100:
            delta *= decay_rate
            iter_since_decay = 0
        
        # Choose two indices to modify
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Try small random change
        new_delta = (np.random.rand() - 0.5) * delta
        h_try = update_h(h, i, j, new_delta)
        
        # Ensure feasible
        h_projected = project_h(h_try)
        if not valid_h(h_projected):
            continue
        
        # Compute new C5
        c5_try = compute_c5(h_projected)
        if c5_try < best_c5:
            best_h = h_projected.copy()
            best_c5 = c5_try
        
        # Accept if better
        if c5_try < c5_current:
            h = h_projected.copy()
            c5_current = c5_try
        
        elapsed = time.time() - start_time
        iter_since_decay += 1
    
    # Final fixup
    corrected_h = project_h(best_h)
    final_c5 = compute_c5(corrected_h)
    
    return (corrected_h.tolist(), final_c5, n_points)
