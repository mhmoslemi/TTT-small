import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Create a simple binary construction
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper to compute C5
    def compute_c5(h_values):
        dx = 2.0 / len(h_values)
        # Compute correlation
        corr = signal.correlate(h_values, 1 - h_values, mode='full')
        # Maximize over all lags
        max_val = np.max(corr) * dx
        return max_val
    
    # Gradient-based optimization with projections
    step_size = 0.1
    decay_factor = 0.95
    min_step = 1e-6
    max_iter = 1000
    patience = 100
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    no_improvement = 0
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps efficiently
        dx = 2.0 / len(h)
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute negative gradient using finite differences
        # For each k, the gradient of sum h[i](1-h[i+k]) w.r.t h[i] is:
        # (1 - h[i+k]) - (h[i-k] if i-k is valid)
        # We'll compute this for the active shift
        max_shift_idx = np.argmax(corr)
        max_shift = max_shift_idx - (len(h) - 1)
        
        # Compute gradient using finite difference
        grad = np.zeros_like(h)
        for j in range(len(h)):
            k = (j + max_shift) % len(h)
            grad[j] = (1 - h[k])
            # Add contribution from h[i-k] where i-k is valid
            # This is non-standard, we compute a correct gradient approximation
            j_neg = (j - max_shift) % len(h)
            if max_shift > 0:
                # For k = j + shift, derivative of h[j] * h[k] wrt h[j] is h[k]
                grad[j] -= h[j_neg]
        
        # Take a step
        h_new = h - step_size * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Scale back to sum = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new value
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            no_improvement = 0
            best_h = h.copy()
        else:
            no_improvement += 1
            step_size *= decay_factor
            if step_size < min_step:
                step_size = min_step
            # Perturb every 500 iterations or if no improvement for a while
            if no_improvement % 500 == 0 or no_improvement > 50:
                h = h + np.random.normal(0, 0.001, len(h))
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        if best_c5 < 0.3807:
            break
            
        if no_improvement > patience:
            break
    
    # Final validation and return
    final_c5 = compute_c5(best_h)
    return (best_h, final_c5, len(best_h))
