import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Default parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 10000
    T_max = 1.0
    T_min = 1e-4
    alpha = np.exp(-np.log(T_max / T_min) / max_iter)
    delta_init = 0.1
    min_accept_rate = 0.05
    max_accept_rate = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a balanced initialization
        h = np.zeros(n_points)
        half = n_points // 2
        h[:half] = 1.0
        # Shuffle to avoid simple pattern
        indices = np.random.permutation(n_points)
        h = h[indices]
        
    # Ensure sum constraint
    current_sum = np.sum(h)
    h *= (n_points / 2) / current_sum
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    
    # Initial C5 calculation
    current_C5 = np.max(signal.correlate(h, 1-h, mode='full')) * dx
    best_C5 = current_C5
    best_h = h.copy()
    
    # Simulated Annealing parameters
    T = T_max
    delta = delta_init
    accept_count = 0
    total_iter = 0
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create new solution
        new_h = h.copy()
        
        # Perturb h
        if np.random.rand() < 0.5:
            # Swap two elements
            idx1, idx2 = np.random.randint(0, n_points, 2)
            new_h[idx1], new_h[idx2] = new_h[idx2], new_h[idx1]
        else:
            # Shift mass between adjacent points
            idx = np.random.randint(0, n_points - 1)
            shift_amount = np.random.rand() * delta
            shift_amount = min(shift_amount, new_h[idx], 1 - new_h[idx + 1])
            
            new_h[idx] -= shift_amount
            new_h[idx + 1] += shift_amount
        
        # Ensure that new_h remains within [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(new_h)
        new_h *= (n_points / 2) / current_sum
        
        # Ensure exact sum constraint
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Adjust to meet exactly
            diff = (n_points / 2) - current_sum
            idx = np.argmax(new_h) if diff > 0 else np.argmin(new_h)
            new_h[idx] = min(1.0, max(0.0, new_h[idx] + diff))
            
        # Compute new C5
        new_C5 = np.max(signal.correlate(new_h, 1 - new_h, mode='full')) * dx
        
        # Accept with probability
        delta_C5 = new_C5 - current_C5
        if delta_C5 < 0 or np.random.rand() < np.exp(-delta_C5 / T):
            h = new_h
            current_C5 = new_C5
            accept_count += 1
            
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Adapt delta based on acceptance rate
        total_iter += 1
        if total_iter % 100 == 0 and total_iter > 0:
            accept_rate = accept_count / 100.0
            if accept_rate < min_accept_rate:
                delta *= 0.9
            elif accept_rate > max_accept_rate:
                delta *= 1.1
            accept_count = 0
    
    return (best_h.tolist(), best_C5, n_points)
