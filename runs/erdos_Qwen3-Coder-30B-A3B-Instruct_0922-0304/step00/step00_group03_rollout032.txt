import numpy as np
from scipy import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    # Compute overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx to get actual integral value
    return np.max(corr) * dx

def is_valid_solution(h):
    """Check if solution is valid: binary values and correct sum"""
    return np.all((h == 0) | (h == 1)) and abs(np.sum(h) - len(h)/2) < 1e-10

def generate_neighbors(h, tabu_list, n_points):
    """Generate neighbors by flipping 2-3 bits while maintaining sum"""
    neighbors = []
    # Generate neighbors by flipping 2 bits
    for i in range(n_points):
        for j in range(i+1, n_points):
            if h[i] == h[j]:  # Only flip if they're the same
                h_new = h.copy()
                h_new[i] = 1 - h_new[i]
                h_new[j] = 1 - h_new[j]
                if is_valid_solution(h_new):
                    # Convert to tuple for hashing
                    key = tuple(h_new)
                    if key not in tabu_list:
                        neighbors.append(h_new)
    # Can add 3-bit flips if needed but keep it simple for now
    return neighbors

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a binary construction with 50% ones
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Ensure sum is exactly correct
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Modify to have exactly target_sum ones
        ones_idx = np.where(h == 1)[0]
        zeros_idx = np.where(h == 0)[0]
        n_to_flip = int(target_sum - current_sum)
        if n_to_flip > 0:
            # Flip some 0s to 1s
            for i in range(n_to_flip):
                if i < len(zeros_idx):
                    h[zeros_idx[i]] = 1
        else:
            # Flip some 1s to 0s
            for i in range(-n_to_flip):
                if i < len(ones_idx):
                    h[ones_idx[i]] = 0

    dx = 2.0 / n_points
    current_c5 = evaluate_h(h, dx)
    
    # Tabu search parameters
    tabu_tenure = 100
    tabu_list = set()
    best_h = h.copy()
    best_c5 = current_c5
    start_time = time.time()
    
    max_iter = 1000  # maximum iterations based on starting point
    max_iter = min(max_iter, int(budget_s * 50) if budget_s > 5 else 1000)
    
    # Track last improvement iteration
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:  # Leave some buffer
            break
            
        neighbors = generate_neighbors(h, tabu_list, n_points)
        if not neighbors:
            continue
            
        # Evaluate neighbors
        neighbor_scores = []
        for neighbor in neighbors:
            score = evaluate_h(neighbor, dx)
            neighbor_scores.append((score, neighbor))
        
        if not neighbor_scores:
            continue
            
        # Select best neighbor
        best_score, best_neighbor = min(neighbor_scores, key=lambda x: x[0])
        
        if best_score < best_c5:
            best_c5 = best_score
            best_h = best_neighbor.copy()
            last_improvement = iteration
            
        # Update tabu list
        tabu_list.add(tuple(h))
        if len(tabu_list) > tabu_tenure:
            # Remove oldest entry
            tabu_list.pop()
            
        h = best_neighbor
        
        # Early termination: if no improvement in last 25% of iterations
        if iteration - last_improvement > max_iter // 4:
            break
    
    # Final check of best solution
    best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
