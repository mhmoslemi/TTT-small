import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute full correlation with 1-h
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap values for k = 0, 1, ..., 2n-1
    overlap = corr[n-1:]
    # Compute C5 as max of overlaps scaled by dx
    c5 = np.max(overlap) * dx
    return c5

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum constraint."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

    h = h_init.copy()
    h = project_to_feasible(h, n_points / 2)
    
    t0 = time.time()

    best_c5 = evaluate_erdos_solution(h)
    
    # Parameters for optimization
    max_iterations = 3000
    backtracking_rho = 0.5
    backtracking_max_steps = 30
    refinement_frequency = 50
    
    for iteration in range(max_iterations):
        if time.time() - t0 > budget_s:
            break
            
        # Compute overlaps for all shifts
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:]
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max] * (2.0 / n_points)
        
        # If we improved, update best
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current

        # Compute gradient for the worst shift
        shifted_h = np.roll(1 - h, k_max)
        gradient = shifted_h - h * shifted_h
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            continue
            
        # Line search
        step_size = 0.01
        for _ in range(backtracking_max_steps):
            h_new = h - step_size * gradient / grad_norm
            h_new = project_to_feasible(h_new, n_points / 2)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5_current:
                h = h_new
                break
            step_size *= backtracking_rho
        else:
            # If no improvement after backtracking, still use current step
            pass

        # Convex refinement every few iterations
        if iteration % refinement_frequency == 0:
            # Focus on the current maximum overlap shift
            # Minimize overlap for k_max by solving convex problem
            # Create variables
            h_vars = cp.Variable(n_points)
            # Objective: minimize the overlap for shift k_max
            # We want to minimize sum_i h[i] * (1 - h[(i-k_max)%n])
            obj_term = cp.sum(cp.multiply(h_vars, 1 - cp.hstack([h_vars[k_max:], h_vars[:k_max]])))
            objective = cp.Minimize(obj_term)
            # Constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            # Solve optimization problem
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL:
                h = np.array(h_vars.value).flatten()
                h = np.clip(h, 0, 1)
                # Rescale to maintain sum constraint
                h = project_to_feasible(h, n_points / 2)

    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
