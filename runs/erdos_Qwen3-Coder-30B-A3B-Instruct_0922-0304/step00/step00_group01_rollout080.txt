        """Compute overlap for shift k"""

        return np.sum(h_vals * (1 - shifted))
        
    def compute_all_overlaps(h_vals):
        """Compute all overlaps for each shift k"""
        overlaps = []
        for k in range(n_points):
            overlaps.append(compute_overlap(h_vals, k))
        return np.array(overlaps)

        """Project h to feasible region"""
        # Clip to [0,1] and reweight to match sum constraint
        h_new = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points / 2) > 1e-10:
            # Use simple redistribution to shift mass
            diff = n_points / 2 - current_sum
            if diff > 0:
                # Need to distribute positive diff
                # Find indices with room to increase and increase proportionally
                room_to_increase = 1.0 - h_new
                total_room = np.sum(room_to_increase)
                if total_room > 0:
                    increase = diff * room_to_increase / total_room
                    h_new = np.minimum(1.0, h_new + increase)
            else:
                # Need to distribute negative diff
                room_to_decrease = h_new
                total_room = np.sum(room_to_decrease)
                if total_room > 0:
                    decrease = -diff * room_to_decrease / total_room
                    h_new = np.maximum(0.0, h_new - decrease)
