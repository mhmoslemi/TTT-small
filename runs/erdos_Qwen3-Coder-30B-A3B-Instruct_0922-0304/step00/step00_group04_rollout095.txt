        dx = 2.0 / len(h)
        # Compute correlations where h is convolved with (1-h)
        corr = np.correlate(h, 1 - h, mode='full') 
        # Extract the overlapping parts (k from -n+1 to n-1) - valid range
        valid_corr = corr[len(h)-1 : 2*len(h)-1]
        return np.max(valid_corr) * dx
    
    def project_and_rescale(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to unit integral
        scale_factor = (n_points / 2) / np.sum(h)
        h *= scale_factor
        return h

    def optimize_step(h, lr=0.05, max_iter=50):
        dx = 2.0 / len(h)
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            c5_vals = []
            corr_full = np.correlate(h, 1 - h, mode='full')
            # Extract valid range of lags (from -n+1 to n-1)
            corr = corr_full[len(h)-1 : 2*len(h)-1]
            max_k = np.argmax(corr)
            # Compute gradient
            g = np.zeros_like(h)
            # finite difference estimate of gradient for the dominant k
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = project_and_rescale(h_plus)
                c_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c_plus = c_plus[len(h_plus)-1 : 2*len(h_plus)-1]
                h_minus = h.copy()
                h_minus[i] -= 1e-5

                g[i] = (c_plus[max_k] - c_minus[max_k]) / (2e-5)
            # update with gradient and projection
            h_new = h - lr * g
            h_new = project_and_rescale(h_new)
            c5_new = evaluate_c5(h_new)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                h = h_new.copy()

        return best_h, best_c5

            h, current_c5 = optimize_step(h_up)
    except Exception:

    # Ensure final projection
    h = project_and_rescale(h)
    current_c5 = evaluate_c5(h)
