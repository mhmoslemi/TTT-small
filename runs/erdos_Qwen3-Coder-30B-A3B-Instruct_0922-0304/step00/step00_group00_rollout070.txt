    # Optimization parameters
    max_iter = int(1e6)  # Prevent infinite loop

    temperature = 0.1
    cooling_rate = 0.995
    perturb_frac = 0.05  # 5% of points perturbed per iteration
    delta_h = 0.1  # Perturbation magnitude

        # Compute overlap integral using convolution
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum across all lags and normalize by dx
        return np.max(conv) * dx

        # Adjust h to satisfy sum(h) = n/2
        current_sum = np.sum(h_vec)

        # Clamp values to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec
    
    # Initial evaluation
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()

    # Annealing loop
    for iteration in range(max_iter):
