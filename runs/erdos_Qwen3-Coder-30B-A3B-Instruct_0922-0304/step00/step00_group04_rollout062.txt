import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    # compute overlap integral using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # this is the max overlap over all lags
    max_overlap = np.max(correlation)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # initialize from seed if provided
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # create simple initial step function
        n_points = 42
        h = np.ones(n_points) * 0.5
        # make it a valid step function with integral 1
        h[0:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # discretization step
    dx = 2.0 / n_points
    
    # track best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # optimization loop: try different step sizes
    # use local optimization with gradient-based methods
    
    # make bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h_vec):
        return evaluate_erdos_solution(h_vec, dx)
    
    def constraint_func(h_vec):
        return np.sum(h_vec) - n_points/2.0
    
    # we constrain the integral to be 1, so we must ensure integral is 1
    # but integral is h*dx = 1, so sum(h) = n_points/2.0
    # Now, also ensure bounds are
    # allow time for some iterations
    iters = 0
    max_iters = 1000
    max_time = budget_s * 0.95
    
    # simplex optimization maybe faster with 1st order 
    # but let's use SLSQP which handles bounds better
    try:
        # warm start from current h
        result = optimize.minimize(objective, 
                                   h, 
                                   method='SLSQP', 
                                   bounds=bounds,
                                   constraints={'type': 'eq', 'fun': constraint_func},
                                   options={'maxiter': max_iters, 'ftol': 1e-8, 'gtol': 1e-6})
        if result.success:
            h = result.x
            C5 = evaluate_erdos_solution(h, dx)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
    except:
        pass
    
    # try a few more random initializations
    for tries in range(10):
        if time.time() - start_time > max_time:
            break
        h_init = np.random.rand(n_points)
        # normalize to proper integral
        h_init = h_init / np.sum(h_init) * (n_points/2.0)
        # clip to bounds
        h_init = np.clip(h_init, 0, 1)
        # constrain sum
        h_init = h_init * (n_points/2.0) / np.sum(h_init)
        
        try:
            result = optimize.minimize(objective, 
                                       h_init, 
                                       method='SLSQP', 
                                       bounds=bounds,
                                       constraints={'type': 'eq', 'fun': constraint_func},
                                       options={'maxiter': 200, 'ftol': 1e-8, 'gtol': 1e-6})
            if result.success:
                h_trial = result.x
                C5_trial = evaluate_erdos_solution(h_trial, dx)
                if C5_trial < best_C5:
                    best_C5 = C5_trial
                    best_h = h_trial.copy()
        except:
            pass
                
    return (best_h, best_C5, n_points)
