def evaluate_erdos_solution(h):
    """Evaluates the Erdős constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode='full')
    # Take only the valid ones
    valid_overlaps = overlaps[n-1:]  # corresponds to k from 0 to n-1
    return np.max(valid_overlaps) * dx
