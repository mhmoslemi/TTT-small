import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Number of samples
    n_points = 45
    dx = 2.0 / n_points
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Initialize as binary with exactly n_points/2 ones
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial solution
    overlaps = correlate(h, 1 - h, mode="full") * dx
    best_c5 = np.max(overlaps)
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 500
    patience = 50
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        current_c5 = np.max(overlaps)
        
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        
        # Constraint: 0 <= h[i] <= 1
        constraints = [h_relaxed >= 0, h_relaxed <= 1]
        
        # Constraint: sum(h) = n_points/2
        constraints.append(cp.sum(h_relaxed) == n_points / 2)
        
        # Objective: minimize max of overlaps
        # We use the fact that we're minimizing max_k of integral terms
        # We can relax this to minimize a large constant that dominates all overlaps
        
        # Solve convex problem
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            new_h = h_relaxed.value
        except:
            # Fallback to current solution if problem fails
            new_h = h
            
        # Project back to feasible set while maintaining sum constraint
        # Threshold at 0.5
        projected_h = np.where(new_h > 0.5, 1, 0)
        current_sum = np.sum(projected_h)
        
        if np.abs(current_sum - n_points / 2) > 1e-10:
            # Adjust to maintain sum n_points/2
            # Find candidates to change
            diff = int(n_points / 2 - current_sum)
            if diff > 0:
                # We need to convert some 0s to 1s
                indices_to_change = np.argsort(new_h)[::-1][:diff]
                for i in indices_to_change:
                    if projected_h[i] == 0:
                        projected_h[i] = 1
            elif diff < 0:
                # We need to convert some 1s to 0s
                indices_to_change = np.argsort(new_h)[:abs(diff)]
                for i in indices_to_change:
                    if projected_h[i] == 1:
                        projected_h[i] = 0
                        
        # Ensure the final solution is valid
        assert np.abs(np.sum(projected_h) - n_points / 2) < 1e-10
        h = projected_h
        
        # Evaluate new solution
        new_overlaps = correlate(h, 1 - h, mode="full") * dx
        new_c5 = np.max(new_overlaps)
        
        # Update best if improved
        if new_c5 < best_c5 - 1e-10:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Occasionally perturb solution to escape local optima
        if iteration % 20 == 0 and time.time() - start_time < budget_s - 1:
            # Perturb a small number of entries
            num_perturb = max(1, n_points // 20)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                # Add small noise to prevent getting stuck
                noise = np.random.normal(0, 0.05)
                h[i] = np.clip(h[i] + noise, 0, 1)
            
            # Re-normalize sum constraint
            h = h * (n_points / 2) / np.sum(h)
            
        # Early stop if no improvement
        if no_improve_count > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
