import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Beyond 100 iterations
    max_iter = 500
    no_improvement_count = 0
    last_best = best_c5
    
    for _ in range(max_iter):
        if time() - start_time > budget_s * 0.95:
            break
            
        # Find current maximum overlap shift
        correlations = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(correlations)
        max_corr = correlations[k_max]
        
        # Add quadratic constraint for the k_max shift
        # Sum h[i] * (1 - h[(i + k_max) % n_points])
        # is constrained to be <= C5
        
        # Define optimization variables
        h_var = cp.Variable(n_points)
        c5_var = cp.Variable()
        
        # Objective
        objective = cp.Minimize(c5_var)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
            c5_var >= 0
        ]
        
        # Add constraint for the max overlap shift
        # sum_i h[i] * (1 - h[(i + k_max) % n_points]) <= C5
        if k_max < n_points:
            idxs = np.arange(n_points)
            idxs_shift = (idxs + k_max) % n_points
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_var[idxs_shift]))
        else:
            # Wrap around case
            idxs = np.arange(n_points)
            idxs_shift = (idxs + k_max - n_points) % n_points
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_var[idxs_shift]))
            
        constraints.append(constraint_expr <= c5_var)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    new_h = new_h * (n_points / 2) / np.sum(new_h)  # Normalize again
                    
                    c5_bound = max_corr
                    for k in range(2 * n_points - 1):
                        corr = np.correlate(new_h, 1 - new_h, mode="full")[k]
                        c5_bound = max(c5_bound, corr)
                    c5_bound *= 2.0 / n_points
                    
                    if c5_bound < best_c5:
                        best_h = new_h.copy()
                        best_c5 = c5_bound
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
                        
                    h = new_h
        except Exception:
            pass
            
        if no_improvement_count > 50 or best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)


# Helper to compute C5 using the evaluator
def evaluate_erdos_solution(h_vals):
    h = np.array(h_vals)
    n_points = len(h)
    dx = 2.0 / n_points
    correlations = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations)
    c5 = max_corr * dx
    return c5
