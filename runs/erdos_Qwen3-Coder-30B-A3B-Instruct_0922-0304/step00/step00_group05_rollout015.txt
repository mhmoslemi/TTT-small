import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate C5 for given h values"""
    # Compute overlap integral for all lags
    conv = signal.correlate(h, 1 - h, mode='full')
    # Take the maximum over all lags
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) * dx = 1"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to match sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100

    # Initialize swarm
    particles = np.random.rand(swarm_size, n_points)  # Initialize randomly
    if 'initial_h_values' in globals():
        particles[0] = initial_h_values
    
    # Scale all particles to satisfy sum constraint
    for i in range(swarm_size):
        particles[i] = project_h(particles[i], dx)
    
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.5
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_values = np.full(swarm_size, np.inf)
    
    # Initialize global best
    global_best_idx = 0
    global_best_value = np.inf
    
    start_time = time.time()
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            c5_val = evaluate_h(h, dx)
            
            if c5_val < personal_best_values[i]:
                personal_best_values[i] = c5_val
                personal_best[i] = h.copy()
            
            if c5_val < global_best_value:
                global_best_value = c5_val
                global_best_idx = i
                
        # Update velocities and positions
        inertia = 0.7
        cognitive = 1.5
        social = 1.5
        
        for i in range(swarm_size):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (personal_best[i] - particles[i]) +
                social * r2 * (personal_best[global_best_idx] - particles[i])
            )
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_h(particles[i], dx)
    
    # Return best solution
    best_h = personal_best[global_best_idx]
    c5_bound = global_best_value
    
    return (best_h, c5_bound, n_points)
