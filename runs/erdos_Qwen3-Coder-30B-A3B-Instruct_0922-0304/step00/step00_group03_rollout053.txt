import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
    
    # Ensure integral constraint is met: sum(h) * dx = 1
    dx = 2.0 / n_points
    h *= (1.0 / np.sum(h)) * (n_points / 2.0)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent parameters
    max_iter = 10000
    step_size = 0.001
    tolerance = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 and identify worst-case shift
        correlations = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(correlations)
        c5 = correlations[k_max] * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

        # Compute gradient for max overlap
        grad = np.zeros(n_points)
        # The gradient for correlation with shift k_max
        shift = k_max - (n_points - 1)
        if shift <= 0:
            # h[i] contributes to correlation with (1-h[i-shift]) 
            # through term h[i] * (1-h[i-shift])
            for i in range(n_points):
                if i-shift >= 0 and i-shift < n_points:
                    grad[i] += (1 - h[i-shift])
        else:
            # shift > 0: Add to contribution from h[i+shift]
            for i in range(n_points):
                if i+shift < n_points:
                    grad[i] += (1 - h[i+shift])
        grad *= dx
        
        # Update h
        h_new = h - step_size * grad
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce integral constraint: sum(h) = n_points / 2
        sum_h = np.sum(h_new)
        h_new *= (n_points / 2.0) / sum_h
        
        # Check convergence
        if np.linalg.norm(h - h_new) < tolerance:
            break
            
        h = h_new
        
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))

def compute_c5(h, dx):
    correlations = np.correlate(h, 1 - h, mode="full")
    return np.max(correlations) * dx
