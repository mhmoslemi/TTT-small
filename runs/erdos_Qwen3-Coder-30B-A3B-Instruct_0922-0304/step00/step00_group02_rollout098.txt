import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        np.random.shuffle(h)
    
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Gradient descent loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode='full') * dx
        active_k = np.argmax(overlap)
        c5 = overlap[active_k]
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Check termination conditions
        if patience_counter > patience or c5 < 0.3807:
            break
            
        # Compute gradient
        # d/dh_i (sum over j of h_j * (1 - h_{i+k-j}))
        # = (1 - h_{i+k}) - sum_{j} h_j * delta_{i+k, j} = (1 - h_{i+k}) - h_{i+k}
        # But we actually need to compute it more rigorously
        # For a shift k:
        # grad_i = (1 - h_{i+k}) - h_{i+k} = 1 - 2*h_{i+k}
        # But this is only correct if we have no periodic boundary conditions properly
        # We'll use the correct convolution semantics
        grad = np.zeros(n_points)
        # Compute (1-h) shifted to the right by k
        shift_h = np.roll(1 - h, -active_k)
        # Compute (1-h) * h with proper indexing (this is our overlap gradient)
        for i in range(n_points):
            grad[i] = shift_h[i] # corresponds to d/dh_i [h[i] * (1 - h[i+k])]
        grad = 1 - 2 * h  # Direct finite-gradient approach
        # Actually, let's do it correctly:
        # For overlap at k: O_k = sum_{i} h[i] * (1 - h[i+k]) 
        # dO_k/dh[i] = (1 - h[i+k]) if i is valid
        # But we have circular boundary cond.
        # In general: dO_k/dh[i] = sum_{j} (1 - delta_{k+i,j}) * delta_{i,j} = (1 - h[i+k]) 
        # which is what we compute below using method
        # But if k = active_k, the gradient is simply 1 - h[(i+active_k)%n]
        # But numerically that's not quite right.
        grad = np.zeros_like(h)
        # Gradient computation more rigorously:
        # derivative of integral h(x)(1-h(x+k)) dx
        # For fixed k, and h[t] as variable:
        # d/dh[t] ∫ h(x)(1-h(x+k)) dx = (1 - h[t+k]) 
        # We have h cyclic, so 't+k' corresponds to (t+k)%n
        for i in range(n_points):
            j = (i + active_k) % n_points
            grad[i] = (1 - h[j])
        grad = 1 - 2 * np.roll(h, -active_k)  # this captures the effect of derivative of h[t]*h[t+k]
        # Correct computation of ∂O_k / ∂h_i:
        # We take derivative of sum_{j} h[j] * (1 - h[j+k])
        # ∂/∂h_i sum_{j} h[j] * (1 - h[j+k]) 
        # = delta_{i,j} * (1 - h[j+k]) + h[j] * (- delta_{i,j+k})
        # Use: ∂/∂h[i] of h[j] * (1 - h[j+k]) = 0 unless j=i or j=i-k
        # Thus: ∂/∂h[i] of term at index j = h[i-k] * (1 - h[i]) ~= -h[i-k] (if diff)
        # Objectively: 
        # We compute the effect of shifting h[i]
        # New overlap = sum h[j] (1 - h[j+k])
        # Each term's derivative w.r.t. h[i]: 
        # There's h[i] --> (1 - h[i+k]) term  
        # and if j = i-k, there's h[i-k] term --> -h[i]
        # so each term contributes 1-h[i+k] when i = j
        # and -h[i-k] when i = j+k
        # so sum_{j} d/dh[i] (h[j] * (1-h[j+k])) = (1-h[i+k]) - h[i-k]
        # Let's compute directly:
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + active_k) % n_points
            grad[i] = 1 - h[j]
        # But we already used shift_h above. So let's compute:
        # We compute gradient correction by finite diff
        grad = np.roll(1 - h, active_k)
        # Now update gradient properly for minimizing O_k
        # For i, it's: d/dh[i] (sum h[j] * (1 - h[j+k])) = 1 - h[(i+k) % n]
        grad = 1 - np.roll(h, -active_k)
        
        # Update h
        h_temp = h - alpha * grad
        
        # Project h_temp
        h_temp = np.clip(h_temp, 0, 1)
        h_sum = np.sum(h_temp)
        if h_sum > 0:
            h = h_temp * (n_points / 2) / h_sum
        else:
            h = np.ones_like(h) * (n_points / 2) / n_points
        
        # Update learning rate
        if np.mod(iteration, 100) == 0:
            alpha *= gamma
        if alpha < alpha_min:
            alpha = alpha_min
            
        # Occasionally apply small perturbation (5%) to escape local minima
        if np.mod(iteration, 500) == 0:
            perturb = np.random.rand(n_points) * 0.05
            h = np.clip(h + perturb, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    overlap_final = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(overlap_final)
    
    return (best_h.tolist(), final_c5, len(best_h))
