import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Computes C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using correlate with mode='full'
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)

    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = budget_s * 1000  # Rough estimate
    patience_count = 0
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all lags that achieve the maximum overlap
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient w.r.t. h
        gradient = np.zeros(n_points)
        for m in max_lags:
            # Compute contribution to gradient from overlap at lag m
            # overlap[m] = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+m])
            # Gradient contribution = 1 - h[i+m] for valid i
            if m < n_points:
                # i from 0 to n_points-1-m
                valid_indices = np.arange(n_points - m)
                gradient[valid_indices] += (1 - h[valid_indices + m])
            else:
                # i from 0 to m-n_points
                valid_indices = np.arange(m - n_points + 1)
                gradient[valid_indices + (n_points - m)] += (1 - h[valid_indices])
        
        # Average over all maximizers
        gradient /= len(max_lags)
        
        # Update h
        h = h - learning_rate * gradient
        
        # Projection: 0 <= h <= 1 and sum(h) * dx = 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if abs(h_sum) > 1e-10:
            h = h * (1.0 / h_sum)
        
        # Compute new C5
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience_count = 0
        else:
            patience_count += 1
            
        # Adaptive learning rate
        if patience_count >= gamma_patience:
            learning_rate *= gamma_decrease_factor
            patience_count = 0
    
    return (best_h, best_c5, n_points)
