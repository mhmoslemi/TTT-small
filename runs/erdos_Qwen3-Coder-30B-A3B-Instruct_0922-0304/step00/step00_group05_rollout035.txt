import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all lags
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iterations = int(budget_s * 1000)
    
    # Setup for learning rate adaptation
    last_improvement = 0
    last_c5 = best_c5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1.0:

        # Compute overlaps for all lags
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all lags where maximum overlap occurs
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient for max overlap lags
        gradient = np.zeros(n_points)
        for m in max_lags:
            # Compute contribution to gradient from this lag
            # For each valid i: grad[i] += (1 - h[i + m]) if i+m < n_points
            if m < n_points:
                for i in range(n_points - m):
                    gradient[i] += (1 - h[i + m]) * dx
            else:
                # For m >= n_points, we need to account for wrap-around
                for i in range(m - n_points + 1, n_points):
                    gradient[i] += (1 - h[i - (m - n_points + 1)]) * dx

        # Average over number of max lags
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h = h - learning_rate * gradient
        
        # Project h to feasible region (0 <= h <= 1)
        h = np.clip(h, 0, 1)
        
        # Rescale to correct sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iteration
            last_c5 = new_c5
        
        # Adapt learning rate if no improvement for several steps
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
    
    return (best_h, best_c5, n_points)
