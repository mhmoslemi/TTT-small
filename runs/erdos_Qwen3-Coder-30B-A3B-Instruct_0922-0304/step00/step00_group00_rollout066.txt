def evaluate_erdos_solution(h, dx):
    """
    Compute the maximum overlap integral for the given h and dx
    """
    h = np.array(h)
    n = len(h)
    max_val = 0
    for k in range(-n+1, n):
        shift_h = np.roll(h, k)
        overlap = np.sum(h * (1 - shift_h)) * dx
        max_val = max(max_val, overlap)
    return max_val
