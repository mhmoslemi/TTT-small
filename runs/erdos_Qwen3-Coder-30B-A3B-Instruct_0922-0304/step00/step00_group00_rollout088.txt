import numpy as np
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1 - h, mode='full')
    # Take the maximum (including all shifts)
    max_overlap = np.max(correlations)
    # Scale by dx
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
        
    # Normalize so that sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    max_iter = 1000
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 50
    last_c5 = best_c5 + 1
    
    # Use additional constraint tracking
    active_constraints = set()
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.95:
        iter_count += 1
        
        # Check for early termination
        if best_c5 < 0.3807:
            break
            
        # Evaluate all overlaps
        correlations = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(correlations)
        # The actual shift is (max_idx - n+1) to (n-1) 
        shift = max_idx - (n_points - 1)
        max_overlap = correlations[max_idx]
        
        # Add constraint for this shift
        if shift not in active_constraints:
            active_constraints.add(shift)
        
        # Convert to the form we want for CVXPY:
        # op = sum(h[i] * (1 - h[(i + shift) % n])) <= C5
        
        # Build a constraint and solve warm-starting with `h`
        # Set up variables
        h_var = cp.Variable(n_points)
        C5 = cp.Variable()
        
        # Objective: minimize C5
        objective = cp.Minimize(C5)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        
        # Quadratic constraint: sum_{i} h[i] * (1 - h[(i + shift) % n]) <= C5
        # This is a quadratic constraint: 
        # h_i * (1 - h_{i+shift}) <= C5 for all i where shift is negative, or
        # h_{i-shift} * (1 - h_i) <= C5 for all i where shift is positive
        
        # Construct the constraint term-by-term
        if shift >= 0:
            # h_{i-shift} * (1 - h_i) <= C5
            quad_expr = sum(h_var[i] * (1 - h_var[(i + shift) % n_points]) for i in range(n_points))
        else:
            # h_i * (1 - h_{i+shift}) <= C5
            shift_abs = -shift
            quad_expr = sum(h_var[i] * (1 - h_var[(i + shift_abs) % n_points]) for i in range(n_points))
            
        constraints.append(quad_expr <= C5)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h = h_var.value
                if new_h is not None:
                    new_h = np.clip(new_h, 0, 1)
                    h = new_h.copy()
                    # Check new C5
                    current_c5 = evaluate_erdos_solution(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
                        no_improvement_count = 0
                    else:
                        no_improvement_count += 1
            else:
                # If fails to solve, fall back to simple update or continue
                no_improvement_count += 1
        except Exception:
            no_improvement_count += 1
            
        # Periodic cleanup of constraint set
        if len(active_constraints) > 2 * n_points or no_improvement_count >= 30:
            active_constraints.clear()  # Rebuild constraints
            
        if np.abs(last_c5 - best_c5) < 1e-6:
            no_improvement_count += 1
        last_c5 = best_c5
            
        if no_improvement_count >= max_no_improvement:
            break
    
    # Ensure final result is valid
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
