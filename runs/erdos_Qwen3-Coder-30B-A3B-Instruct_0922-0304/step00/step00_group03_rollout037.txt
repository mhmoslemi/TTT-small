import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if abs(np.sum(h) - target_sum) > 1e-6:
            # Repair the sum constraint
            diff = target_sum - np.sum(h)
            if diff > 0:
                indices = np.argsort(h)
                for i in range(int(diff)):
                    h[indices[i]] = min(1.0, h[indices[i]] + 1.0)
            else:
                indices = np.argsort(h)[::-1]
                for i in range(int(-diff)):
                    h[indices[i]] = max(0.0, h[indices[i]] - 1.0)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Binary search over t
    low_t = 0.0
    high_t = 0.5
    tolerance = 1e-4
    
    def check_feasible(t_guess):
        # Formulate the convex feasibility problem
        h_vars = cp.Variable(n_points)
        w = cp.Variable((n_points, n_points))  # auxiliary variables for products
        constr = []
        
        # Bounds on h
        constr += [h_vars[i] >= 0 for i in range(n_points)]
        constr += [h_vars[i] <= 1 for i in range(n_points)]
        
        # Sum constraint
        constr += [cp.sum(h_vars) == target_sum]
        
        # Product constraints
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                if i + k >= 0 and i + k < n_points:
                    # w[i, k] should approximate h[i] * h[i + k]
                    # Use McCormick envelopes or bound constraints
                    constr += [w[i, k] <= h_vars[i]]
                    constr += [w[i, k] <= h_vars[i + k]]
                    constr += [w[i, k] >= h_vars[i] + h_vars[i + k] - 1]
                    
        # Overlap constraints for all shifts
        for k in range(-n_points+1, n_points):
            # For shift k, non-zero terms are for i from 0 to n_points - |k| - 1
            if k >= 0:
                total = cp.sum(h_vars[:n_points-k]) - cp.sum(w[:n_points-k, k])
            else:
                total = cp.sum(h_vars[-k:]) - cp.sum(w[-k:, k])
            constr += [total <= t_guess]
            
        # Objective is to check feasibility (no objective needed)
        prob = cp.Problem(cp.Minimize(0), constr)
        
        try:
            prob.solve(solver=cp.GLPK, time_limit=budget_s / 20)  # Use a small timeout to be safe
            return prob.status == cp.OPTIMAL or prob.status == cp.OPTIMIZED_INFEASIBLE
        except:
            return False
    
    # Binary search
    while high_t - low_t > tolerance and (time() - start_time) < budget_s * 0.9:
        mid_t = (low_t + high_t) / 2
        if check_feasible(mid_t):
            high_t = mid_t
        else:
            low_t = mid_t
    
    # Now solve a final problem with the tightest t to obtain h
    t_final = high_t
    
    # Solve with CVXPY to get a better h
    h_opt = cp.Variable(n_points)
    w = cp.Variable((n_points, n_points))
    constr = []
    
    # Constraints
    for i in range(n_points):
        constr += [h_opt[i] >= 0]
        constr += [h_opt[i] <= 1]
    constr += [cp.sum(h_opt) == target_sum]
    
    # Product constraints
    for i in range(n_points):
        for k in range(-n_points+1, n_points):
            if i + k >= 0 and i + k < n_points:
                constr += [w[i, k] <= h_opt[i]]
                constr += [w[i, k] <= h_opt[i + k]]
                constr += [w[i, k] >= h_opt[i] + h_opt[i + k] - 1]
    
    # Overlap constraints
    for k in range(-n_points+1, n_points):
        if k >= 0:
            total = cp.sum(h_opt[:n_points-k]) - cp.sum(w[:n_points-k, k])
        else:
            total = cp.sum(h_opt[-k:]) - cp.sum(w[-k:, k])
        constr += [total <= t_final]
    
    # Minimize a dummy objective (just feasibility)
    prob = cp.Problem(cp.Minimize(0), constr)
    
    try:
        prob.solve(solver=cp.GLPK, time_limit=budget_s * 0.1)
        if prob.status == cp.OPTIMAL:
            h_sol = h_opt.value
        else:
            h_sol = h
    except:
        h_sol = h
    
    # Final reconstruction to ensure all bounds and sum
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * target_sum / np.sum(h_sol)
    
    # Evaluate C5
    c5_bound = np.max(np.correlate(h_sol, 1-h_sol, mode="full") * dx)
    
    return (h_sol.tolist(), float(c5_bound), n_points)
