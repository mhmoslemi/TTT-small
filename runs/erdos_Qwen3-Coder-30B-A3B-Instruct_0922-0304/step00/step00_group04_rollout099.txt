import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default construction: piecewise constant
        n_points = 42
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Convex relaxation helper
    def solve_convex_relaxation(h_prev, n_points):
        h_var = cp.Variable(n_points)
        # Constraint: sum of h = n_points / 2
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Convex relaxation of correlation term
        # Maximize ||h||_2^2 - h^T (1 - h) = sum(h^2) - sum(h) + sum(h^2) = 2*sum(h^2) - sum(h)
        obj = cp.Minimize(-cp.sum_squares(h_var))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                return h_var.value
        except:
            pass
        return h_prev
    
    # Gradient-based refinement
    def compute_grad(h, k):
        n = len(h)
        # Precompute 1 - h
        h_complement = 1 - h
        
        # Compute correlation for this shift k
        corr = np.correlate(h, h_complement, mode="full")
        grad = np.zeros_like(h)
        
        # For lag k, partial derivative wrt h[i] is 1 - h[i - k] if k >= 0
        # Shift indices
        for i in range(n):
            j = i - k
            if 0 <= j < n:
                grad[i] += 1 - h_complement[j]
        return grad
    
    def project_and_rescale(h, dx):
        h = np.clip(h, 0, 1)
        # Rescale to match integral constraint
        current_sum = np.sum(h) * dx
        desired_sum = 1.0
        if abs(current_sum) > 1e-10:
            scale = desired_sum / current_sum
            h *= scale
        return h
    
    # Main optimization loop
    max_iter = 100
    lr = 0.1
    min_lr = 1e-5
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convex relaxation step
        h = solve_convex_relaxation(h, n_points)
         
        # Compute overlapping correlation
        corr_full = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(corr_full)
        corr_k = corr_full[k_max]
        c5 = corr_k * dx
        
        # Gradient descent step
        grad = compute_grad(h, k_max)
        h_new = h - lr * grad
        h_new = project_and_rescale(h_new, dx)
        
        # Check improvement
        corr_full_new = np.correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(corr_full_new) * dx
        
        # Accept if better
        if c5_new <= c5:
            h = h_new
            if c5_new < 0.3807:
                break
        else:
            lr *= 0.5
            if lr < min_lr:
                break

    # Final evaluation
    corr_full = np.correlate(h, 1 - h, mode="full")
    final_c5 = np.max(corr_full) * dx
    
    return (h, final_c5, n_points)
