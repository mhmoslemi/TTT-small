import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initial constraint check
    assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Sum constraint not satisfied"
    
    # Precompute cross-correlation indices
    full_corr_len = 2 * n_points - 1
    shift_indices = np.arange(full_corr_len)
    
    # Time tracking
    start_time = time.time()
    
    # Basic gradient descent parameters
    learning_rate = 0.01
    max_iter = 10000
    tol = 1e-6
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        
        # Early stopping
        if abs(c5 - last_c5) < tol:
            break
        last_c5 = c5
        
        # Find the shift that maximizes overlap (active constraint)
        max_idx = np.argmax(corr)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient of the active constraint w.r.t h
        # The constraint is sum(h[i] * (1 - h[i + k])) * dx <= t
        # For the active constraint, we compute partial derivative
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_k
            if 0 <= j < n_points:
                grad[i] -= (1 - h[j]) * dx
                grad[j] -= h[i] * dx
        
        # Apply gradient descent
        h_new = h - learning_rate * grad
        
        # Project to [0,1] bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Project to constraint sum(h) = 1/dx
        h_new = h_new * (1 / (np.sum(h_new) * dx)) * (1 / dx)
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to exactly satisfy the constraint
        h_new = h_new * (1 / (np.sum(h_new) * dx)) * (1 / dx)
        h_new = np.clip(h_new, 0, 1)
        
        # Defensive check for energy condition
        h = h_new.copy()
        
    # Final evaluation
    corr = np.correlate(h, 1 - h, mode="full")
    final_c5 = np.max(corr) * dx
    
    return (h.tolist(), final_c5, n_points)
