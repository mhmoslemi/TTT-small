import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 10000
    peak_count = 5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2 (i.e., integral = 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s and iteration < max_iter:
        # Compute overlaps
        overlap = correlate(h, 1 - h, mode='full')[n_points-1:]
        # Find top peaks (ignore trivial peaks at index 0)
        peak_indices = np.argsort(overlap[1:])[::-1][:peak_count] + 1
        
        # Update h to reduce these peaks
        for k in peak_indices:
            # Invert the update logic to reduce overlap
            # Identify where reduction will help: locations where h[i] is high and h[i+k] is low
            # Or where we could reduce the total overlap with sharper peaks
            update = np.zeros(n_points)
            
            # We want to reduce h[i] where overlap is due to both h[i] and h[i+k] being 1
            # but increase h where both are 0 or one is 0
            for i in range(n_points):
                if i + k < n_points:
                    # Check contribution to overlap at lag k
                    if h[i] > 0.1 and h[i+k] > 0.1:  # both are active, reduce
                        update[i] -= 0.01
                    elif h[i] > 0.1 and h[i+k] < 0.1:  # only h[i] active, keep or slightly increase
                        update[i] += 0.005
                    elif h[i] < 0.1 and h[i+k] > 0.1:  # only h[i+k] active, keep or slightly increase
                        update[i] += 0.005
                        
            # Apply updates subject to feasibility
            h_new = h + update
            h_new = np.clip(h_new, 0, 1)
            # Project back to the correct integral
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Evaluate current C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Early stopping if we've reached our target bound
        if best_c5 < target_c5:
            break
            
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute the overlaps via FFT correlation (more efficient)
    n = len(h)
    h_extended = np.concatenate([h, np.zeros(n)])
    h_reflected = np.concatenate([np.zeros(n), 1 - h[::-1]])
    # The full correlation
    overlap_full = correlate(h, 1 - h, mode='full')
    # Extract non-negative lags
    overlap = overlap_full[n-1:]
    # Return max scaled by dx
    return np.max(overlap) * dx
