import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial construction
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[1] = 0.4
        h[2] = 0.7
        h[3] = 0.3
    
    # Normalize to satisfy integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute correlation
    def compute_c5(h_vals):
        # Compute correlation like in the problem description
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Only consider the relevant parts (that correspond to k=0 to 2)
        # The full correlation has length 2*n_points-1
        # The center values (k=0 to 2) are for indices n_points-1 to n_points+1
        # So we take the central part of size 2*n_points-1
        # index of k=0 in the full correlation is n_points-1
        # so we take from n_points-1 to n_points+1 to max
        # but only up to 2*2+1 = 5 part (from k=0 to k=2) 
        # Actually compute max for all k from 0 to 2
        max_val = -np.inf
        for k in range(-n_points+1, n_points):
            # This is an inefficient approach, better to just select valid correlations
            # Let's fix with a simpler approach:
            if k >= 0:
                # right shift case
                intg = np.sum(h_vals[k:] * (1 - h_vals[:-k]))
            else:
                # left shift case
                intg = np.sum(h_vals[:k] * (1 - h_vals[-k:]))
            if intg > max_val:
                max_val = intg
        return max_val * dx
    
    # Optimization
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    iterations = 0
    delta = 0.1
    step_size = 0.1
    
    while time.time() - start_time < budget_s:
        iterations += 1
        
        # Perturb two random indices 
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Clip values so they stay in [0,1]
        if i == j:
            continue
            
        old_i = h[i]
        old_j = h[j]
        
        # Try a small move
        new_i = old_i + step_size
        new_j = old_j - step_size
        
        # Ensure bounds, and that the total sum remains approximately the same
        # since dx is constant and sum(h)*dx = 1 => sum(h) = n_points/2
        # so small corrections are okay if we keep sum consistent
        # Just make sure they are within [0,1]
        if new_i < 0:
            new_i = 0
            new_j = old_j - (old_i - new_i)
        elif new_i > 1:
            new_i = 1
            new_j = old_j - (old_i - new_i)
        
        if new_j < 0:
            new_j = 0
            new_i = old_i + (old_j - new_j)
        elif new_j > 1:
            new_j = 1
            new_i = old_i + (old_j - new_j)
        
        if not (0 <= new_i <= 1 and 0 <= new_j <= 1):
            continue
            
        # Check that we're within tolerance for sum constraint (quite loose due to floating point)
        original_sum = np.sum(h)
        new_sum = original_sum - old_i - old_j + new_i + new_j
        if abs(new_sum - n_points/2) > 1e-5:
            continue
            
        h[i] = new_i
        h[j] = new_j
        
        # calculate new C5
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # revert change
            h[i] = old_i
            h[j] = old_j
            
        # decrease step size occasionally
        if iterations % 100 == 0:
            step_size *= 0.9
            
        if step_size < 1e-6:
            step_size = 1e-6
    
    return (best_h.tolist(), float(best_c5), n_points)
