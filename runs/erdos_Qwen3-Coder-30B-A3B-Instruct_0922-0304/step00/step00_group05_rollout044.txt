import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        # Ensure it sums to n_points / 2
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    else:
        particles.append(np.random.rand(n_points))
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Fill rest of swarm
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Clipping and scaling function
    def project(h):
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Scale to correct sum
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Compute overlap for a particle
    def compute_c5(h):
        # Convert to convolution form
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h = 1.0 - h
        one_minus_h_extended = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
        # Perform cross-correlation
        cross_corr = correlate(h_extended, one_minus_h_extended, mode='full')
        # Take the maximum value
        max_overlap = np.max(cross_corr[:n_points]) * dx
        return max_overlap
    
    # Initialize personal best
    pbest = particles.copy()
    pbest_scores = np.array([compute_c5(h) for h in particles])
    
    # Global best
    best_idx = np.argmin(pbest_scores)
    gbest = pbest[best_idx].copy()
    gbest_score = pbest_scores[best_idx]
    
    iter_count = 0
    perturbation_interval = 500
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (inertia * velocities[i] + 
                            cognitive * r1 * (pbest[i] - particles[i]) + 
                            social * r2 * (gbest - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i])
            
            # Evaluate
            current_score = compute_c5(particles[i])
            
            # Update personal best
            if current_score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = current_score
                
                # Update global best
                if current_score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = current_score
        
        # Periodic perturbation
        if iter_count % perturbation_interval == 0 and iter_count > 0:
            for i in range(swarm_size):
                if np.random.rand() < 0.1:  # 10% chance to perturb
                    particles[i] += np.random.normal(0, 0.01, n_points)
                    particles[i] = project(particles[i])
        
        iter_count += 1
    
    # Final evaluation
    final_c5 = compute_c5(gbest)
    
    return (gbest, final_c5, n_points)
