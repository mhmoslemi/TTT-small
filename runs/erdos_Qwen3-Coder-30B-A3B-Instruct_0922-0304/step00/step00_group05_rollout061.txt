import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integrals for all lags
    overlap = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = initial_h_values.copy()
    else:
        # Create a starting guess
        n_points = 70
        h = np.ones(n_points) * 0.5
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    gamma_patience = 100
    gamma_decrease_factor = 0.1
    max_iterations = int(budget_s * 1000)
    last_improvement = 0
    start_time = time.time()
    
    # Project h to satisfy constraints
    def project_h(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * (n_points / 2) / current_sum
        return h
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        overlap_full = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap_full)
        
        # Find indices where maximum overlap occurs
        max_indices = np.where(np.abs(overlap_full - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in max_indices:
            # lag is the shift index
            lag = i - (n_points - 1)  # correlate lag indices
            if lag < 0:
                # shift h forward by -lag
                gradient[:-lag] += (1 - h[-lag:]) * dx
            else:
                # shift h backward by lag
                gradient[lag:] += (1 - h[:-lag]) * dx
                
        # Normalize gradient
        if len(max_indices) > 0:
            gradient = gradient / len(max_indices)
        
        # Update h
        h -= learning_rate * gradient
        h = project_h(h)
        
        # Evaluate new solution
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
            
    return (best_h, best_c5, n_points)
