def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    corr = np.correlate(h, 1 - h, mode='full')
    overlaps = corr[n-1:]  # Only consider positive shifts
    max_overlap = np.max(overlaps) * dx
    return max_overlap
