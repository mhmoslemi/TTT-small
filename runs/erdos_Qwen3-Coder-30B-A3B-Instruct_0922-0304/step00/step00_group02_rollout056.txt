import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Random binary initialization with exactly n_points/2 ones
        h = np.zeros(n_points)
        ones_idx = np.random.choice(n_points, size=n_points//2, replace=False)
        h[ones_idx] = 1.0
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Binary refinement loop
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            # Compute correlation: h[i] * (1 - h[(i+k)%n_points])
            shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted))
            overlaps[k] = overlap
        
        # Find maximum overlap (ignoring zero-shift)
        max_overlap_idx = np.argmax(overlaps[1:]) + 1  # exclude index 0 (no shift)
        max_overlap_val = overlaps[max_overlap_idx]
        
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            
        # If we've hit target, stop early
        if best_c5 <= target_bound:
            break
            
        # Greedy update: for each position, determine if flipping improves the maximum overlap
        # We'll flip the position where the gain in max overlap is the greatest
        best_gain = -np.inf
        flip_pos = -1
        
        for i in range(n_points):
            # Temporarily flip
            h_test = h.copy()
            h_test[i] = 1 - h_test[i]
            
            # Check if sum constraint is violated (we won't let sum differ by more than 1)
            current_sum = np.sum(h)
            desired_sum = n_points / 2
            
            # If violating constraint, skip
            if abs(np.sum(h_test) - desired_sum) > 0.5:
                continue
                
            # Compute new overlaps
            new_overlaps = np.zeros(n_points)
            for k in range(n_points):
                shifted = np.roll(h_test, k)
                new_overlaps[k] = np.sum(h_test * (1 - shifted))
            
            # Compute gain (positive means improvement)
            new_max_overlap = np.max(new_overlaps[1:])  # exclude 0 shift
            gain = max_overlap_val - new_max_overlap
            
            if gain > best_gain:
                best_gain = gain
                flip_pos = i
                
        # If no gain or no flip, break
        if flip_pos == -1 or best_gain <= 0:
            break
            
        # Flip the position
        h[flip_pos] = 1 - h[flip_pos]
        
        # Re-normalize (maintain sum to n_points/2)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-8:
            # Adjust to keep sum correct (equally distribute the difference)
            diff = current_sum - n_points / 2
            # Distribute among positions
            for idx in range(n_points):
                if h[idx] > 0:
                    delta = min(h[idx], abs(diff))
                    h[idx] -= delta
                    diff += delta
                    if abs(diff) < 1e-10:
                        break
                        
            # If still difference, adjust to meet sum
            if abs(np.sum(h) - n_points / 2) > 1e-8:
                h = (n_points / 2) * h / np.sum(h)
                
        h = np.clip(h, 0, 1)
        
        # Ensure exact sum constraint
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        h = np.clip(h, 0, 1)
        
    # Ensure final result respects constraints
    final_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2) / final_sum
    best_h = np.clip(best_h, 0, 1)
    
    # Final computation
    final_c5 = compute_c5(best_h, dx)
    
    return best_h.tolist(), final_c5, n_points

def compute_c5(h, dx):
    """Computes the max overlap as per the problem definition."""
    full_correlation = correlate(h, 1 - h, mode='full')
    max_val = np.max(full_correlation)
    # Convert to integral form with proper scaling
    # The correct integral form is direct evaluation as: sum h[i] * (1 - h[(i+k)%n])
    # Scalar h * (vector 1 - h shift) will not work because h is a vector
    # We compute max of: sum(h_i * (1 - h_{i+k})) for all valid k
    # For full correlation, we start from the full sum of all shifts
    return max_val * dx
