import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points)
        h[:n_points//2] = 0
        h = np.clip(h, 0, 1)
    
    # Rescale h to meet integral constraint
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 1000
    target_C5 = 0.3807
    
    # For tracking the best solution
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    start_time = time.time()
    iter_count = 0
    
    # Dual variables initialization
    lambda_k = np.zeros(2 * n_points - 1)  # for all shifts k
    mu = 0.0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Step a: Update h using convex optimization (simplified as proximal gradient)
        h_new = h.copy()
        
        # Define the optimization problem to minimize the overlap
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(
            cp.sum(cp.multiply(
                cp.convolution(h_var, 1 - h_var, mode="full"),
                np.ones(2 * n_points - 1)
            )) * dx + 
            rho / 2 * cp.sum_squares(h_var - h)
        )
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) * dx == 1
        ]
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
            else:
                # fallback if optimization failed
                h_new = h.copy()
        except:
            # fallback to previous h
            h_new = h.copy()
        
        # Step b: Compute overlaps
        overlaps = np.correlate(h_new, 1 - h_new, mode='full') * dx
        
        # Step c: Update dual variables
        max_overlap = np.max(overlaps)
        lambda_k += alpha * (max_overlap - overlaps)
        mu += alpha * (np.sum(h_new) * dx - 1)
        
        # Step d: Entropy regularization
        lambda_k = lambda_k + beta * np.log(1 + np.abs(lambda_k))
        
        # Step e: Maintain feasibility
        lambda_k = np.clip(lambda_k, -1e10, 1e10)
        h = h_new.copy()
        
        # Check if we improved
        current_C5 = compute_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Check early termination
        if best_C5 < target_C5:
            return (best_h.tolist(), best_C5, n_points)
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    """Compute C5 from the step function h."""
    overlaps = np.correlate(h, 1 - h, mode='full')
    return np.max(overlaps) * dx
