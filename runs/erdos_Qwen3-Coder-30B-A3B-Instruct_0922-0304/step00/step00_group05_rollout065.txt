import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    # Compute overlap integrals for all lags
    overlap = signal.correlate(h, 1-h, mode='full')
    # The overlap integral is computed for all lags, but we only want the
    # maximum over the valid range of lags, which corresponds to the middle
    # part of the full convolution (excluding the padding)
    n = len(h)
    # The full correlation has length 2*n - 1, so the valid range is from n-1 to n-1 + n-1 = 2*n-2
    # But we want the middle of it, which corresponds to lags from 0 to n-1
    # Actually, it's the value of correlation at all lags, so we want the max of the full
    # correlation result, scaled by dx
    return np.max(overlap) * dx

def project_to_feasible(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1."""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        target_sum = 1.0 / dx
        h = h * target_sum / h_sum
    return h

    global_best = np.inf
    global_best_position = np.zeros(n_points)

    for i in range(swarm_size):

            personal_best_positions[i] = particles[i].copy()

            global_best_position = particles[i].copy()
