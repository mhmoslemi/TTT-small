import numpy as np
from scipy import optimize
import time

def compute_c5(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    # The full correlation has length 2*n_points - 1
    # We only care about lags from -n_points+1 to n_points-1
    # The actual overlap integral for lag k is corr[n_points-1+k] * dx
    # We want max over all valid lags
    max_val = -np.inf
    for k in range(-len(h)+1, len(h)):
        idx = len(h) - 1 + k
        val = corr[idx] * dx
        if val > max_val:
            max_val = val
    return max_val

def project_h(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum(h)*dx=1"""
    h = np.clip(h, 0, 1)
    # Adjust h to make sum(h)*dx == 1
    current_sum = np.sum(h) * dx
    if abs(current_sum) > 1e-10:
        h = h / current_sum * (n_points / 2)
    h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    # Use the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points, dx)
    else:
        # Start with uniform distribution and some randomness
        h = np.ones(n_points) * 0.5
        h = project_h(h, n_points, dx)
    
    # Gradient based optimization
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    iters = 0
    
    while time.time() - start_time < budget_s - 1.0:
        iters += 1
        h_current = h.copy()
        
        # Compute cross-correlation and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_val = -np.inf
        max_k = 0
        for k in range(-n_points+1, n_points):
            idx = n_points - 1 + k
            val = corr[idx] * dx
            if val > max_val:
                max_val = val
                max_k = k
        
        # Compute gradient for current max_k
        # Gradient term for each point i:
        # If k >= 0, then h[i] contributes to corr[n_points-1+k] = sum_j h[j](1-h[j+k])
        # So for i where j = i, d/dh[i] = (1 - h[i+k])
        # For i-k >= 0, j=i-k, so the term is the rate of change of h[i] in corr[n_points-1+k]
        # [WARNING] We need to be careful: gradient only applies to valid lags.
        # For lag k, we can only update h[i] for i where i+k is a valid index
        
        grad = np.zeros(n_points)
        if max_k >= 0:
            # This corresponds to h[i] in the sum: h[i](1-h[i+k])
            # When we differentiate, we get 1 - h[i+k]
            for i in range(n_points - max_k):
                grad[i] += (1.0 - h[i + max_k])
        else:
            # Lag is negative, handle properly
            abs_k = -max_k
            for i in range(abs_k, n_points):
                grad[i] += (1.0 - h[i - abs_k])
        
        # Update with gradient descent
        step_size = 0.005
        h_new = h_current - step_size * grad
        
        # Project back to feasible region
        h_new = project_h(h_new, n_points, dx)
        h = h_new
        
        # Evaluate new C5
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
