import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_vals, n_points):
    # Evaluate the overlap integral for this h
    dx = 2.0 / n_points
    # Compute overlap for all shifts
    overlap = correlate(h_vals, 1 - h_vals, mode='full')
    # The maximum overlap is the maximum value (excluding the 0-shift case)
    # But since we're looking for max_k int h(x)(1-h(x+k)) dx,
    # it's the maximum of the full correlation
    max_overlap = np.max(overlap)
    # Normalize the overlap sum by dx
    c5 = max_overlap * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction
        n_points = 47
        h = np.full(n_points, 0.5)
    
    # Ensure the constraint sum(h) * dx = 1 = n_points/2
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute some constants for efficiency
    # Create the shift index mapping for fast correlation computation
    # Precompute the correlation for all shifts
    def compute_overlap(h):
        # use full correlation
        overlap = correlate(h, 1 - h, mode='full')
        # Find max k
        max_overlap = np.max(overlap)
        return max_overlap
    
    # Use adaptive gradient steps
    alpha = 0.01  # Initial step size
    max_iter = 10000
    best_h = h.copy()
    best_c5 = compute_overlap(h) * dx
    start_time = time.time()
    
    # Gradient descent with projection
    lambda_lag = 0.0
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        overlap_vals = correlate(h, 1 - h, mode='full')
        # the maximum overlap and its shift index
        max_idx = np.argmax(overlap_vals)
        max_overlap = overlap_vals[max_idx]
        
        # Compute the derivative for the worst case
        # The overlap for shift k is: sum(h[i] * (1 - h[i-k]))
        # According to the Lagrangian derivative:
        # g[i] = (1 - h[(i-k)%n]) - h[i] + lambda
        k = max_idx - (n_points - 1)
        # Build gradient for k shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k) % n_points
            grad[i] = (1 - h[j]) - h[i] + lambda_lag
        
        # Update h
        h_new = h - alpha * grad
        
        # Projection onto [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Project onto the feasible set: sum(h) = n_points/2
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Update lambda
        lambda_lag += alpha * (np.sum(h_new) - n_points / 2)
        
        # Update h
        h = h_new
        
        # Evaluate the new solution
        c5_new = compute_overlap(h) * dx
        
        iter_count += 1
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce step size
            if alpha >= 1e-5:
                alpha *= 0.95  # reduce step size gradually
        
        # Early stopping
        if c5_new < 0.3807:
            break
        if no_improvement_count > max_no_improvement:
            break
    
    # Final check of feasibility
    h_final = best_h.copy()
    sum_h = np.sum(h_final)
    dx_final = 2.0 / n_points
    c5_final = evaluate_erdos_solution(h_final, n_points)
    
    # Ensure sum is correct
    h_final = h_final * (n_points / 2) / sum_h
    
    return (h_final.tolist(), c5_final, n_points)
