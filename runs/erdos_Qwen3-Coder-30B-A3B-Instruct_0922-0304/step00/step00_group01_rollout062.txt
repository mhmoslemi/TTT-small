import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_max_iter = 30
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_overlap(h_vec, k):
        """Compute overlap for shift k using correlation"""
        h_clipped = np.concatenate([h_vec, np.zeros(k)])
        return np.sum(h_vec * (1 - h_clipped[k:]))
    
    def compute_all_overlaps(h_vec):
        """Compute all overlaps and return max and index"""
        h_rev = np.flip(h_vec)
        correlations = np.correlate(h_vec, h_rev, mode='full')
        n = len(h_vec)
        overlaps = correlations[n-1:] * dx  # Only positive shifts
        k_max = np.argmax(overlaps)
        return overlaps[k_max], k_max
    
    def compute_gradient(h_vec, k_max):
        """Compute gradient for overlap at k_max"""
        n = len(h_vec)
        gradient = np.zeros(n)
        for i in range(n):
            # h[i] * (1 - h[i + k_max]) for direct term
            # (1 - h[i - k_max]) for shift term (we need special handling for boundary)
            if i + k_max < n:
                gradient[i] += 1 - h_vec[i + k_max]
            if i - k_max >= 0:
                gradient[i] += h_vec[i - k_max]
            gradient[i] -= 1
        return gradient
    
    def line_search(h_vec, k_max, gradient, alpha_initial):
        """Perform backtracking line search with appropriate scaling"""
        alpha = alpha_initial
        current_c5, _ = compute_all_overlaps(h_vec)
        
        for _ in range(line_search_max_iter):
            h_new = h_vec - alpha * gradient
            # Projection to feasibility
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points/2) / np.sum(h_new)
            
            new_c5, _ = compute_all_overlaps(h_new)
            
            if new_c5 < current_c5:
                return h_new, alpha
            alpha *= rho
            
        # If all failed, return original h
        return h_vec, 0
    
    iteration = 0
    no_improvement_count = 0
    
    while iteration < max_iter:
        elapsed = time.time() - start_time
        remaining_budget = budget_s - elapsed
        if remaining_budget < 10:
            break
            
        # Compute all overlaps to find k_max
        c5_current, k_max = compute_all_overlaps(h)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        if c5_current < target_bound:
            break
            
        # Compute gradient
        gradient = compute_gradient(h, k_max)
        gradient_norm = np.linalg.norm(gradient)
        
        if gradient_norm < 1e-10:
            break  # Gradient is nearly zero
            
        # Normalize gradient
        gradient = gradient / gradient_norm
        
        # Line search
        h_new, alpha = line_search(h, k_max, gradient, alpha_initial)
        
        if alpha <= 1e-10:
            # If no successful step, try smaller learning rate
            alpha_initial *= 0.1
            if alpha_initial < 1e-6:
                break
        else:
            h = h_new
            
        # Check for no improvement
        if iteration > 0 and abs(best_c5 - c5_current) < 1e-8:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count > 100:
            break
            
        iteration += 1
    
    # Final evaluation of the best solution
    c5_eval, _ = compute_all_overlaps(best_h)
    
    return (best_h.tolist(), c5_eval, n_points)
