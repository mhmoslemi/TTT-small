import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_max_iter = 30
    alpha_init = 0.01
    rho = 0.5
    eta = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute indices for correlation
    full_len = 2 * n_points - 1
    
    def compute_c5(h_vec):
        # Compute all overlaps
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = corr[n_points-1:] * dx  # Maximum overlap indices
        return np.max(overlaps)
    
    def compute_gradient(h_vec, k_max):
        grad = np.zeros_like(h_vec)
        for i in range(n_points):
            # Forward contribution
            if i + k_max < n_points:
                grad[i] += 1 - h_vec[i + k_max]
            # Backward contribution
            if i - k_max >= 0:
                grad[i] -= h_vec[i - k_max]
        return grad
    
    def line_search(h_vec, grad, k_max, alpha_initial):
        alpha = alpha_initial
        current_c5 = compute_c5(h_vec)
        for _ in range(line_search_max_iter):
            h_new = h_vec - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                return h_new
            alpha *= rho
        return h_vec
    
    # Iterative optimization
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        current_c5 = overlaps[k_max]
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early stopping
        if current_c5 < target_bound:
            break
            
        # Check for stagnation
        if iteration - last_improvement > 100:
            break
            
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm < 1e-10:
            break
            
        grad = grad / grad_norm
        
        # Line search
        h = line_search(h, grad, k_max, alpha_init)
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    if best_c5 >= compute_c5(h):
        final_c5 = compute_c5(h)
        final_h = h.copy()
    else:
        final_c5 = best_c5
        final_h = best_h.copy()
        
    return (final_h.tolist(), final_c5, n_points)
