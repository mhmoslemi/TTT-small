import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the last n_points elements (shifts 0 to n_points-1)
    max_overlap = np.max(overlap[n_points-1:])
    return max_overlap * dx
