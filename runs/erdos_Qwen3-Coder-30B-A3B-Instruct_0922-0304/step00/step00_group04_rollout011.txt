def evaluate_erdos_solution(h_values):
    """Evaluates the C_5 constant for given h_values."""

    # Compute all correlations for lags from -n+1 to n-1
    corr = np.correlate(h_values, 1 - np.array(h_values), mode='full')
    # Extract the valid correlations for lags -n+1 to n-1
    # We only consider lags where the convolution is valid (not padding)
    max_corr = 0.0
    for k in range(-n + 1, n):
        start_idx = max(0, -k) if k < 0 else 0
        end_idx = min(n, n - k) if k >= 0 else n + k
        if end_idx > start_idx:
            corr_val = np.sum(h_values[start_idx:end_idx] * (1 - h_values[start_idx+k:end_idx+k]))
            max_corr = max(max_corr, corr_val)
    return max_corr * dx
