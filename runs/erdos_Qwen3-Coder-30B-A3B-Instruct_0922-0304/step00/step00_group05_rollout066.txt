import numpy as np
from scipy import signal
import time
import random

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute all overlaps using full convolution
    overlap = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx to get continuous integral values
    overlap *= dx
    # Return the maximum overlap (excluding index 0, which corresponds to no shift)
    return np.max(overlap)

def project_feasible(h, dx):
    """Project h to satisfy the constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1."""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet the integral constraint
    current_sum = np.sum(h)
    target_sum = 1.0 / dx
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):

    random.seed(seed)
    np.random.seed(seed)

    # Parameters

    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000  # Will limit by time instead
    
    # Initialize positions and velocities
    if 'initial_h_values' in globals():
        # Use the provided starting point
        particles = np.array([initial_h_values] + [np.random.rand(n_points) for _ in range(swarm_size - 1)])
    else:
        # Create a random initial population
        particles = [np.random.rand(n_points) for _ in range(swarm_size)]
    
    # Project each particle to feasible set
    particles = [project_feasible(p, dx) for p in particles]
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    # Track personal bests and global best
    personal_bests = [evaluate_overlap(p, dx) for p in particles]
    personal_best_positions = [p.copy() for p in particles]
    global_best_idx = np.argmin(personal_bests)
    global_best = personal_bests[global_best_idx]
    global_best_pos = personal_best_positions[global_best_idx].copy()
    
    # PSO parameters
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    start_time = time.time()
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:  # Leave 2s for cleanup
            break
            
        for i in range(swarm_size):
            # Evaluate current particle
            current_overlap = evaluate_overlap(particles[i], dx)
            
            # Update personal best
            if current_overlap < personal_bests[i]:
                personal_bests[i] = current_overlap
                personal_best_positions[i] = particles[i].copy()
                
                # Update global best if needed
                if current_overlap < global_best:
                    global_best = current_overlap
                    global_best_pos = particles[i].copy()
            
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive_component = cognitive * r1 * (personal_best_positions[i] - particles[i])
            social_component = social * r2 * (global_best_pos - particles[i])
            
            velocities[i] = (inertia * velocities[i] + 
                            cognitive_component + 
                            social_component)
            
            particles[i] += velocities[i]
            
            # Project to feasible set after update
            particles[i] = project_feasible(particles[i], dx)
            
            # Occasionally jitter particles if no improvement
            if iteration % 100 == 0 and iteration > 0:
                for j in range(n_points):
                    if np.random.rand() < 0.05:
                        particles[i][j] = np.clip(
                            particles[i][j] + np.random.normal(0, 0.05), 0, 1)
                particles[i] = project_feasible(particles[i], dx)

    # Final evaluation and return
    final_h = global_best_pos
    final_c5 = evaluate_overlap(final_h, dx)
    final_h = np.clip(final_h, 0, 1)
    return (final_h.tolist(), final_c5, n_points)
