    best_c5 = float('inf')

        H_conj = np.conj(H)

        # Create spectrum of 1 - h
        mod_one_minus_sq = np.abs(H_one_minus) ** 2

        corr = fft.ifft(H * H_one_minus.conj(), fft_size)
        corr_vals = np.abs(corr[:n_points])
        
        # Find max overlap
        c5 = np.max(corr_vals) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Optimization step: adjust H to minimize maximum overlap
        # Update H to flatten the magnitude spectrum
        # This is a simplified gradient-like step
        # We want to minimize max |H(k)(1 - H^*(k))|
        # which is equivalent to minimizing max |H(k) - |H(k)|^2|
        
        # Direction: approximate gradient wrt the spectrum
        # We use simple update: replace |H(k)|^2 with average
        H_mag_sq = np.abs(H) ** 2

        # But keep same sum constraint
        scale_factor = avg_mag_sq / (H_mag_sq + 1e-12)

        # But compute new h from inverse FFT
