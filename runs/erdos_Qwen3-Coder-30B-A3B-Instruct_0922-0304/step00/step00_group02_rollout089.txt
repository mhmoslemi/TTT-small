import numpy as np
import time
from scipy import optimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        n_points = 45

        np.random.shuffle(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # For tracking improvement
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 50:
        iteration += 1
        # Compute overlaps for current h
        overlaps = np.correlate(h, 1 - h, mode="full") * dx
        C5 = np.max(overlaps)
        
        # Formulate convex problem
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_relaxed.value
        except:
            # Fallback to previous solution
            h_new = h.copy()
        
        # Project h_new back to binary
        # First threshold at 0.5
        h_binary = (h_new > 0.5).astype(float)
        current_sum = np.sum(h_binary)
        desired_sum = n_points / 2
        
        # Adjust if needed
        if current_sum > desired_sum:
            # Zero out the largest values
            indices_to_zero = np.argpartition(h_new, -int(current_sum - desired_sum))[-int(current_sum - desired_sum):]
            h_binary[indices_to_zero] = 0
        elif current_sum < desired_sum:
            # Set the smallest values to 1
            indices_to_one = np.argpartition(h_new, int(desired_sum - current_sum))[:int(desired_sum - current_sum)]
            h_binary[indices_to_one] = 1
        
        # Ensure the final vector satisfies the constraint
        current_sum = np.sum(h_binary)
        if current_sum != desired_sum:
            # Fix discrepancy by flipping
            diff = int(desired_sum - current_sum)
            if diff > 0:
                # Find smallest values that are 1 and set to 0
                ones = np.where(h_binary == 1)[0]
                if len(ones) > 0:
                    flip_indices = np.random.choice(ones, size=min(diff, len(ones)), replace=False)
                    h_binary[flip_indices] = 0
            else:
                # Find largest values that are 0 and set to 1
                zeros = np.where(h_binary == 0)[0]
                if len(zeros) > 0:
                    flip_indices = np.random.choice(zeros, size=min(-diff, len(zeros)), replace=False)
                    h_binary[flip_indices] = 1
        
        # Evaluate new solution
        h = h_binary
        new_C5 = evaluate_erdos_solution(h)
        
        if new_C5 < best_C5:
            best_h = h.copy()
            best_C5 = new_C5
            last_improvement = iteration
            
        # Perturb a small percentage of values
        perturb_count = int(n_points * 0.05)
        if perturb_count > 0:
            indices = np.random.choice(n_points, size=perturb_count, replace=False)
            for i in indices:
                if np.random.rand() < 0.5:
                    h[i] = 1.0
                else:
                    h[i] = 0.0
    
    return (best_h.tolist(), best_C5, n_points)
