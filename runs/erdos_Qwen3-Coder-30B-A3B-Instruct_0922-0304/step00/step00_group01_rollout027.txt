import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:

        n_points = 47
        h = np.ones(n_points) * 0.5
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    # Optimization settings
    max_iter = 10000
    patience = 100
    patience_counter = 0
    alpha = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap
        overlap = correlate(h, 1-h, mode='full')[n_points-1:]
        max_k = np.argmax(overlap) + 1  # lag 1 to n_points
        max_overlaps = [(i, overlap[i]) for i in range(len(overlap)) if overlap[i] > 0]
        max_overlaps.sort(key=lambda x: x[1], reverse=True)
        
        # Get top peaks
        top_k = min(5, len(max_overlaps))
        peaks_to_reduce = [x[0] for x in max_overlaps[:top_k]]
        
        # Update h
        h_new = h.copy()
        for k in peaks_to_reduce:
            # Apply gradient-like update with decreasing step size
            update_magnitude = alpha * (1.0 - iteration / max_iter)
            
            # For each index i, reduce h[i] if h[i] and h[i+k] contribute to overlap
            for i in range(len(h)):
                if i + k < len(h):
                    # Increase h[i] and decrease h[i+k] to reduce overlap
                    if overlap[k] > 0.001:
                        h_new[i] = np.clip(h_new[i] + update_magnitude * 0.5, 0, 1)
                        h_new[i + k] = np.clip(h_new[i + k] - update_magnitude * 0.5, 0, 1)
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this new h improves C5
        new_C5 = evaluate_C5(h_new, dx)
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early termination if stable
        if patience_counter >= patience or best_C5 < 0.3807:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    """Compute the C5 value for given h"""
    n = len(h)
    overlap = correlate(h, 1-h, mode='full')[n-1:] * dx
    return np.max(overlap)
