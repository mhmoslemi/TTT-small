import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Max overlap over all shifts
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple binary starting point
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0  # First half ones, second half zeros
        
    # Ensure correct integral constraint
    dx = 2.0 / len(h)
    current_sum = np.sum(h)
    # Normalize to sum = n_points / 2 (since dx = 2/n)
    h = h * (len(h) / 2) / current_sum
    
    # Keep best solution
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Timer
    start_time = time.time()
    elapsed = 0
    
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while elapsed < budget_s - 0.1 and no_improvement_count < max_no_improvement:
        # Compute current overlaps
        current_overlaps = correlate(h, 1 - h, mode="full") * dx
        
        # Formulate convex relaxation
        n = len(h)
        h_relaxed = cp.Variable(n, name="h_relaxed")
        
        # Objective: minimize maximum overlap
        max_overlap = cp.Variable(name="max_overlap")
        objective = cp.Minimize(max_overlap)
        
        # Constraints
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n/2,
            max_overlap >= current_overlaps[i] for i in range(len(current_overlaps))
        ]
        
        # Create problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_relaxed.value.copy()
                
                # Project back to feasible set
                # Project to the binary simplex using thresholding
                h_binary = (h_new > 0.5).astype(float)
                h_sum = np.sum(h_binary)
                
                if abs(h_sum - n/2) > 1e-6:
                    # Adjust thresholds
                    threshold = 0.5
                    while True:
                        h_binary = (h_new > threshold).astype(float)
                        h_sum = np.sum(h_binary)
                        if abs(h_sum - n/2) <= 1e-6:
                            break
                        if h_sum < n/2:
                            threshold -= 0.01
                        else:
                            threshold += 0.01
                        if threshold < 0 or threshold > 1:
                            break
                
                # If still not perfect, adjust to make it sum to n/2
                diff = h_sum - n/2
                if abs(diff) > 1e-6:
                    # Adjust by changing some 1s to 0s or 0s to 1s
                    # Find indices with greatest error in binary approximation
                    errors = np.abs(h_new - h_binary)
                    sorted_indices = np.argsort(errors)[::-1]
                    adjust = int(abs(diff))
                    adjustment_direction = -1 if diff > 0 else 1
                    
                    for i in range(adjust):
                        if i < len(sorted_indices):
                            idx = sorted_indices[i]
                            if h_binary[idx] + adjustment_direction >= 0 and h_binary[idx] + adjustment_direction <= 1:
                                h_binary[idx] += adjustment_direction
                
                # Ensure sum is exactly correct
                h_binary = h_binary.astype(float)
                h_sum = np.sum(h_binary)
                h_binary = h_binary * (n/2) / h_sum  # Normalize, but keep in [0,1]
                
                # Clip to [0, 1]
                h_binary = np.clip(h_binary, 0, 1)
                h_new = h_binary
                
                # Compute new quality
                new_C5 = evaluate_erdos_solution(h_new)
                
                if new_C5 < best_C5:
                    best_h = h_new.copy()
                    best_C5 = new_C5
                    no_improvement_count = 0
                else:
                    no_improvement_count += 1
                    
                h = h_new.copy()
                
                # Minor perturbation
                for _ in range(max(1, int(n * 0.05))):
                    i = np.random.randint(n)
                    h[i] = np.random.random()
                # Re-normalize to satisfy integral constraint
                h = h * (n/2) / np.sum(h)
                
            else:
                # If convex optimization fails, just randomize a bit
                for _ in range(max(1, int(n * 0.05))):
                    i = np.random.randint(n)
                    h[i] = np.random.random()
                h = h * (n/2) / np.sum(h)
                no_improvement_count += 1
                
        except:
            # Fallback if convex optimization fails
            for _ in range(max(1, int(n * 0.05))):
                i = np.random.randint(n)
                h[i] = np.random.random()
            h = h * (n/2) / np.sum(h)
            no_improvement_count += 1
            
        # Early term
        elapsed = time.time() - start_time
        iteration += 1

    return (best_h.tolist(), best_C5, len(best_h))
