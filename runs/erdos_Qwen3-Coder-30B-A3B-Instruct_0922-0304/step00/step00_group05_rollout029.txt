import numpy as np
from scipy import optimize
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Fallback initialization: piecewise constant function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[1] = 0.4
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project h to have correct sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_C5 = np.inf
    start_time = time()
    
    # Gradient-based optimization with clipping
    gamma = 0.01
    patience = 0
    patience_limit = 20
    min_gamma = 1e-6
    
    for iteration in range(100000):
        if time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations for current h
        # convolve h with (1-h) using full mode
        h_padded = np.pad(h, (0, n_points - 1), 'constant')
        one_minus_h = 1.0 - h
        one_minus_h_padded = np.pad(one_minus_h, (0, n_points - 1), 'constant')
        
        # This is equivalent to np.correlate(h, 1-h, mode='full')
        full_conv = np.convolve(h_padded, one_minus_h_padded, mode='full')
        full_conv = full_conv[:2*n_points - 1] * dx
        
        # Extract the relevant values from full convolution
        # There are 2*n_points - 1 values, corresponding to lags from -(n_points-1) to (n_points-1)
        # The effective lags for our C_m are from 0 to 2*n_points - 2
        # But we actually want lags from 0 to n_points-1 (corresponding to indexes 0 to n_points-1)
        # However the way indices work is complex; let's compute it more carefully
        # C_m = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+m]) * dx, m=0 to n_points-1
        # We'll compute it using efficient matrix operations or manual loops
        
        C_values = np.zeros(n_points)
        for l in range(n_points):
            # Compute integral for lag l
            integral = 0.0
            # In terms of convolution, l corresponds to the lag where the second function shifts by l
            # i goes from 0 to n_points-1-l, and term is h[i] * (1 - h[i+l])
            for i in range(n_points - l):
                integral += h[i] * (1 - h[i + l])
            C_values[l] = integral * dx
            
        C5 = np.max(C_values)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > patience_limit and gamma > min_gamma:
            gamma *= 0.5
            patience = 0
            
        # Compute gradient w.r.t. h
        gradient = np.zeros(n_points)
        max_C_index = np.argmax(C_values)
        
        # For the max C index m, compute gradient:
        # d/dh_k = {1 - h[k+m], if k<m}
        #          {0, otherwise}
        # Instead, we really want derivative of sum_{l=0}^{n_points-1} h_i * (1-h_{i+l})
        # For fixed l, the derivative of the part is (1 - h[i+l]) for term h[i]
        # So for each i, we want sum_{l=0}^{n_points-i-1} (1 - h[i+l]) 
        # This is since we're differentiating over h_i, and i+l can range from i to n_points-1
        # But we want to find where that term contributes, i.e., where 0 <= i <= n_points - 1 - l, so l >= 0 and l <= n_points - 1 - i
        # So for each i, contribution from index i+l, we add (1-h[i+l]) when l=0 to n_points-1-i
        
        # Derivative for each h[i]:
        # derivative[i] = sum_{l=0}^{n_points-1-i} (1 - h[i+l])
        # This is just convolution of a vector of 1s with (1-h) starting at i
        # Let's compute more explicitly:
        # The Lagrange multiplier method or gradient descent:
        # We simply update h_i := h_i - gamma * gradient_i
        
        # Experimental approach: use the principle
        # For each m that achieves max, the corresponding contributions are:
        gradient = np.zeros(n_points)
        # For lag m, we contribute:
        # d/dh_i (h[i] * (1 - h[i+m])) = (1 - h[i+m]) if i+m < n_points, 0 otherwise
        # So for i=0 to n_points-1, gradient[i] += sum of all (1-h[i+m]) where m=0 to n_points-1 and i+m < n_points
        # But that's not right again, because a single point doesn't result in a sum.
        # A correct approach is that C[k] = sum over i of h[i] * (1 - h[i+k]), so:
        # dC[k]/dh[i] = (1 - h[i+k]) for all valid i such that i+k < n_points, 
        # which means k < n_points-i => k ranges from 0 to n_points-1-i
        # But that's hard to get a true gradient. 
        # 
        # We are trying to minimize max_{m=0..n_points-1} C_m = max_m sum h[i] (1-h[i+m]) 
        # Use optimality condition for max, or simply: 
        # if we are minimizing a maximum, a valid approach is to take the one contributing the most at the current point.
        
        # Simplified approach: Target the case that gives max C_value only
        max_indices = np.where(C_values == C5)[0]  # This is a mistake, we need to find contributing terms properly
        # Actually: approximate approach
        max_lag_index = max_C_index  # threshold to start with
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if max_lag_index < n_points - i:
                # This term contributes to the boundary of convolution
                gradient[i] += (1 - h[i + max_lag_index]) # only if not out of bounds
                
        # Correct gradient calculation, "subgradient" simple version:
        # Treat max as an implicit constraint and gradient much simpler. 
        # Sample the one that achieves high overlap
        # Ideally check mode-full
        # For a gradient step in cost, given that C_k = sum_i h_i (1 - h_{i+k}), 
        # then gradient[i] = 1 - h[i+k] if valid, (taking k as the max index).
        # But that is still not right. Better to do finite difference approach.
        
        # Fix issue: let's take the full set of effective lags that achieve the max to compute gradient
        max_indices = np.where(np.abs(C_values - np.max(C_values)) < 1e-8)[0]
        gradient = np.zeros(n_points)
        
        for m in max_indices:
            # Control the gradient contribution from lag m
            # We compute sum of d/dh_i [h_i * (1 - h_{i+m})] for all i with i+m < n_points
            for i in range(n_points - m):
                gradient[i] += 1.0 - h[i + m]
        
        # Update
        h = h - gamma * gradient
        
        # Projection
        h = np.clip(h, 0, 1)
        
        # Re-scale to sum condition
        h_sum = np.sum(h)
        if np.abs(h_sum) > 1e-8:
            h = h * target_sum / h_sum
            
    return (best_h.tolist(), best_C5, n_points)
