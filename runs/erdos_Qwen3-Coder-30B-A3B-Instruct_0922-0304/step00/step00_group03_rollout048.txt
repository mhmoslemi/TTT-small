import numpy as np
from scipy import optimize
import time
import cvxpy as cp

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution h * (1-h)
    conv = np.correlate(h, 1-h, mode="full")
    # The overlap integral for each shift k
    overlaps = conv * dx
    # Compute C5 as max of all overlaps
    return np.max(overlaps)

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        # Add a little randomness
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    h = initialize_h(n_points)
    dx = 2.0 / n_points
    best_c5 = evaluate_c5(h)
    best_h = h.copy()

    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 1000
    patience = 500
    no_improve_count = 0

    # Initialize dual variables
    lambda_k = np.zeros(2 * n_points - 1)  # full convolution size
    mu = 0.0

    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break

        # Solve convex subproblem for h
        h_var = cp.Variable(n_points)
        # Overlaps approximation for fixed duals
        # Use a regularized version
        obj = cp.Minimize(
            cp.max(cp.multiply(lambda_k, cp.conv({i: 1 for i in range(2*n_points-1)}, 
                                                    h_var, mode='full'))) +
            rho * cp.sum_squares(h_var) / 2 +
            mu * (cp.sum(h_var) - n_points / 2)
        )

        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        # Solve
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status != cp.OPTIMAL:
                print("Subproblem not solved optimally.")
            h_new = h_var.value
        except Exception as e:
            # Fallback to previous h
            h_new = h

        h = h_new

        # Evaluate new solution
        c5_new = evaluate_c5(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        # Update dual variables   
        # Re-compute overlaps for this h
        overlaps = np.correlate(h, 1-h, mode="full") * dx
        # Update lambda_k with subgradient ascent
        # This is an approximation to using constraints
        for k in range(len(lambda_k)):
            lambda_k[k] += alpha * np.maximum(0, best_c5 - overlaps[k])
        
        # Update mu based on mean constraint
        mu += alpha * (np.sum(h) - n_points / 2)

        # Entropy regularization (optional but can help set λ)
        lambda_k = lambda_k + beta * np.log(1 + np.exp(lambda_k))
        
        # Keep dual variables bounded
        lambda_k = np.clip(lambda_k, -1e6, 1e6)
        
        # Decay entropy term
        beta *= 0.995

    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
