import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h_vals, n_points):
    """Evaluates the Erdos constant for given step function h_vals."""
    # Normalize h_vals to have sum = n_points / 2
    h = np.array(h_vals)
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute overlaps for all shifts
    h_rev = np.flip(1 - h)
    overlap = np.correlate(h, h_rev, mode='full')
    overlap = overlap[len(h)-1:]  # Only the valid overlaps
    c5 = np.max(overlap) * 2.0 / n_points
    
    return c5

def compute_gradient(h, k_max, n_points):
    """Compute gradient for maximizing overlap with shift k_max."""
    shifted_h = np.roll(1 - h, k_max)
    gradient = shifted_h - h * shifted_h
    return gradient

def backtracking_line_search(h, grad, n_points, max_iter=30, alpha_init=0.01, rho=0.5):
    """Backtracking line search for step size."""
    alpha = alpha_init
    h_prev = h.copy()
    
    # Normalize gradient for step size
    grad_norm = np.linalg.norm(grad)
    if grad_norm > 0:
        grad_normalized = grad / grad_norm
    else:
        return alpha, h_prev
    
    for _ in range(max_iter):
        h_new = h - alpha * grad_normalized
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = evaluate_erdos_solution(h_new, n_points)
        c5_old = evaluate_erdos_solution(h, n_points)
        
        if c5_new <= c5_old:
            return alpha, h_new
        alpha *= rho
    
    return alpha, h_new

def convex_refinement(h, k_max, n_points):
    """Use convex optimization to refine h for specific k_max."""
    # This is a convex optimization problem
    h_cvx = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum(cp.multiply(h_cvx, 1 - cp.roll(h_cvx, k_max))))
    
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]
    
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.GLPK, verbose=False)
        h_refined = h_cvx.value
        return np.array(h_refined)
    except Exception:
        # Fallback to proximal projection if optimization fails
        h_refined = np.array(h)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        return h_refined

    best_c5 = evaluate_erdos_solution(h, n_points)
    best_h = h.copy()
    
    iterations = 0
    while elapsed < budget_s - 1.0:
        if iterations % 50 == 0:
            # Run convex refinement every 50 iterations
            overlap = np.correlate(h, 1 - h, mode='full')
            k_max = np.argmax(overlap[len(h)-1:])
            h = convex_refinement(h, k_max, n_points)
            c5 = evaluate_erdos_solution(h, n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Compute currents overlap
        overlap = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlap[len(h)-1:])
        current_c5 = overlap[k_max] * 2.0 / n_points
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h, k_max, n_points)
        
        # Apply line search
        alpha, h_new = backtracking_line_search(h, grad, n_points)
        h = h_new
        
        iterations += 1
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    if final_c5 < best_c5:
        best_c5 = final_c5
