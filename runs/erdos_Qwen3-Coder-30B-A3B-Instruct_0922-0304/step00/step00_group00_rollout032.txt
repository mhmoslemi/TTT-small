import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h = h * 2 / h.sum()  # normalize to sum to 1
    
    dx = 2.0 / n_points
    
    # Iterative refinement
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    # Add some randomness to avoid local minima
    h = h + np.random.normal(0, 0.01, n_points)
    h = np.clip(h, 0, 1)
    h = h * 2 / h.sum()
    
    max_iter = 50000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute all overlaps
        overlaps = []
        for k in range(n_points):
            # Circular convolution for overlap at shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        
        c5 = max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if c5 <= 0.3807:
            break
            
        if last_improvement > 100:
            # Try random perturbation
            h = best_h.copy()
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * 2 / h.sum()
            last_improvement = 0
            continue
            
        # Simple gradient ascent on most constraining shift
        max_k = np.argmax(overlaps)
        # try to adjust the maximum overlap by updating h at some positions
        # this is a simple heuristic to improve the current worst case
        delta = 0.01
        # Find positions that contribute to the overlap
        contrib = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            contrib[i] += (1 - h[j]) * delta
            contrib[j] -= h[i] * delta
        # Reduce h where contributing positively to overlap
        update_dir = contrib
        update_dir /= np.sum(np.abs(update_dir)) + 1e-8
        h += update_dir * 0.005
        h = np.clip(h, 0, 1)
        h = h * 2 / h.sum()
        
    return (best_h.tolist(), best_c5, n_points)
