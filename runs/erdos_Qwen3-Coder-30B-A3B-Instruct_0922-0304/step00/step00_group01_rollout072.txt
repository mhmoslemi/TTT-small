import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Compute C5 bound."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k=0..n-1
    overlaps = []
    for k in range(n):
        # Circular convolution with (1-h)
        shifted = np.roll(1 - h, k)
        overlap = np.sum(h * shifted) * dx
        overlaps.append(overlap)
    return max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 47
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum to n/2
    h = h * (n_points / 2) / np.sum(h)
    
    # For tracking
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    last_improvement = 0
    iterations = 0
    
    # Optimizer parameters
    step_size = 0.01
    max_iter = 10000
    min_step_size = 1e-5
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        iterations += 1
        
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            overlap = np.sum(h * shifted)
            overlaps.append(overlap)
        
        # Find worst case shift
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Compute subgradient for this shift
        # g[i] = (1 - h[(i - k_max) % n]) - h[i] + lambda
        # But lambda is not directly used in update, just keep old approach
        g = np.zeros(n_points)
        for j in range(n_points):
            idx = (j - k_max) % n_points
            g[j] = (1 - h[idx]) - h[j]
        
        # Update h
        h_new = h - step_size * g
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain integral constraint
        scale = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale
        
        # Check improvement
        old_max_overlap = max_overlap
        h = h_new
        
        # Recompute max overlap
        overlaps = []
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            overlap = np.sum(h * shifted)
            overlaps.append(overlap)
        current_max_overlap = max(overlaps)
        
        c5_current = current_max_overlap
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Update step size
        if current_max_overlap >= old_max_overlap:
            step_size *= 0.5
        step_size = max(step_size, min_step_size)
        
        if best_c5 < 0.3807:
            break
            
        if last_improvement > 100:
            break
    
    return (best_h.tolist(), best_c5, len(best_h))
