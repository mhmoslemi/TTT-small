import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h0)
    except:

        n_points = 42

    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    # Try SDP approach first
    try:
        # SDP relaxation approach
        h = cp.Variable(n_points)
        Y = cp.Variable((n_points, n_points), symmetric=True)
        C5 = cp.Variable()
        
        # Constraints
        constraints = [
            cp.diag(Y) == h,
            Y >> 0,  # PSD constraint
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2,
            C5 >= 0
        ]
        
        # Add correlation constraints for all lags k
        for k in range(n_points):
            corr_sum = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr_sum += h[i] - Y[i, i + k]
            constraints.append(corr_sum <= C5 / dx)
        
        # Objective
        objective = cp.Minimize(C5)
        prob = cp.Problem(objective, constraints)
        
        # Try solving SDP
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_bound = C5.value
            if c5_bound < target_c5:
                return (h_opt, c5_bound, n_points)
        else:
            raise Exception("SDP not solved")
            
    except:
        pass
    
    # Fallback: use gradient-based optimization
    
    def corr_value(h_vals):
        # Compute correlation for all lags and return max
        conv = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Take maximum over all lags
        return np.max(conv) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def objective(h):
        return corr_value(h)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial point
    h_init = h0.copy()
    
    # Optimization
    def obj_func(h):
        return objective(h)
    
    def constr_func(h):
        return constraint_sum(h)
    
    constraints = {'type': 'eq', 'fun': constr_func}
    
    # Use L-BFGS-B for optimization
    result = minimize(obj_func, h_init, method='L-BFGS-B', bounds=bounds, constraints=constraints,
                      options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-6})
    
    # Return if within budget
    if time.time() - start_time < budget_s - 1:
        h_final = result.x
        c5 = corr_value(h_final)
        if c5 < target_c5:
            return (h_final, c5, n_points)
    
    # If not successful, return the best we have
    h_final = result.x
    c5 = corr_value(h_final)
    return (h_final, c5, n_points)
