import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import random

def evaluate_c5(h, n_points):
    """Evaluate C5 for given h values"""
    dx = 2.0 / n_points
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # The valid overlaps are from indices n_points-1 to 2*n_points-2
    # This corresponds to the range where we slide the window
    valid_overlaps = overlaps[n_points - 1:2 * n_points - 1]
    c5 = np.max(valid_overlaps) * dx
    return c5

def initialize_population(n_points, n_pop=100, initial_h=None):
    """Initialize population with feasible step functions"""
    population = []
    
    # Add initial solution if provided
    if initial_h is not None:
        h = np.array(initial_h)
        # Normalize to ensure sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Add random feasible solutions
    while len(population) < n_pop:
        h = np.random.rand(n_points)
        # Ensure sum = n_points/2 by scaling
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Make sure no negative values
        h = np.maximum(h, 0)
        population.append(h.copy())
    
    return population

def mutate(h, mutation_rate=0.01):
    """Mutate a solution with given rate"""
    h = h.copy()
    n_points = len(h)
    n_mutations = max(1, int(n_points * mutation_rate))
    
    # Select indices to mutate
    indices = np.random.choice(n_points, n_mutations, replace=False)
    
    for idx in indices:
        # Add small random change
        delta = np.random.normal(0, 0.05)
        h[idx] += delta
        # Clip to [0,1]
        h[idx] = np.clip(h[idx], 0, 1)
    
    # Rescale to maintain sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    return h

def crossover(parent1, parent2):
    """Perform crossover between two parents"""
    n_points = len(parent1)
    # Random crossover point
    point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Rescale to maintain sum = n_points/2
    child1 = child1 * (n_points / 2) / np.sum(child1)
    child2 = child2 * (n_points / 2) / np.sum(child2)
    
    # Clip to [0,1]
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    return child1, child2

def genetic_algorithm(n_points, initial_h=None, budget_s=600):
    """Run genetic algorithm to minimize C5"""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, n_pop=100, initial_h=initial_h)
    
    best_c5 = float('inf')
    best_h = None
    
    n_generations = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        n_generations += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, n_points)
            fitness.append(1 / (c5 + 1e-12))  # Avoid division by zero
        
        # Sort by fitness (descending)
        sorted_indices = np.argsort(fitness)[::-1]
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Track best individual
        current_best_c5 = 1 / fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Selection (take top 50%)
        n_selected = len(population) // 2
        selected = population[:n_selected]
        
        # Create new population
        new_population = selected.copy()
        
        # Crossover + mutation
        while len(new_population) < len(population):
            # Select parents
            parents = random.sample(selected, 2)
            child1, child2 = crossover(parents[0], parents[1])
            
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to correct size
        population = new_population[:len(population)]
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5"""
    np.random.seed(seed)
    
    # Use the provided initial construction if available
    initial_h = None
    if 'initial_h_values' in globals():

    # Use 45 points as in the initial construction

    # Run genetic algorithm
    best_h, best_c5 = genetic_algorithm(n_points, initial_h, budget_s)
    
    # Final validation of solution
    if best_h is not None:
        # Ensure it satisfies constraints
        assert np.all(best_h >= 0) and np.all(best_h <= 1)
        assert abs(np.sum(best_h) - n_points/2) < 1e-6
        
        c5_final = evaluate_c5(best_h, n_points)
        return (best_h.tolist(), c5_final, n_points)
    else:
        # Fallback to initial solution
        if initial_h is not None:
            h = initial_h
            dx = 2.0 / n_points
            overlaps = correlate(h, 1 - np.array(h), mode='full')
            valid_overlaps = overlaps[n_points - 1:2 * n_points - 1]
            c5_cross = np.max(valid_overlaps) * dx
            return (h, c5_cross, n_points)
        else:
            # Basic starting point
            h = np.ones(n_points) * 0.5  # All 0.5's
            dx = 2.0 / n_points
            overlaps = correlate(h, 1 - np.array(h), mode='full')
            valid_overlaps = overlaps[n_points - 1:2 * n_points - 1]
            c5_cross = np.max(valid_overlaps) * dx
            return (h.tolist(), c5_cross, n_points)
