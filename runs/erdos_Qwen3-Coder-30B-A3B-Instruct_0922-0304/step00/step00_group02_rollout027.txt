from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlation between h and (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (excluding zero-padding)
    n = len(h)
    offset = n - 1
    corr_vals = corr[offset:2*n-1]
    # Return max value scaled by dx
    return np.max(corr_vals) * dx
