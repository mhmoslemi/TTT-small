import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: staggered step function
        n_points = 70
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 2 == 0 else 0.5
        h = h * (n_points / 2) / np.sum(h)  # normalize

    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization parameters
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    max_iters = int(budget_s * 1000)
        
    last_improvement = 0
    for iteration in range(max_iters):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find indices where maximum overlap occurs
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        max_lags = max_lags.tolist()
        
        # Compute gradient
        grad = np.zeros_like(h)
        for lag in max_lags:
            # Determine valid indices based on lag
            if lag < len(h):
                # lag represents shift from h to (1-h)
                # so we take h[i] * (1-h[i+lag]) 
                valid_indices = np.arange(len(h) - lag)
                if len(valid_indices) > 0:
                    grad[valid_indices] += (1 - h[valid_indices + lag]) * dx
            else:
                # lag > len(h), equivalent to negative lag
                valid_indices = np.arange(lag - len(h), len(h)) 
                if len(valid_indices) > 0:
                    grad[valid_indices] += (1 - h[valid_indices - lag]) * dx
                    
        grad /= max(1, len(max_lags))
        
        # Update h with gradient descent
        h -= learning_rate * grad
        
        # Apply box constraints and normalization
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (len(h) / 2) / h_sum  # ensure correct sum = n_points / 2
            
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
            
    return (best_h, best_c5, len(best_h))

def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral for given step function h"""
    n = len(h)
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode="full") * dx
    max_overlap = np.max(overlaps)
    return max_overlap
