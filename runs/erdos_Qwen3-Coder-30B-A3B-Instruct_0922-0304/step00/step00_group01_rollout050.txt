import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Early stopping threshold
    target_c5 = 0.3807
    
    # Optimization parameters
    max_iter = 10000
    k_peaks = 5
    step_size = 0.05
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for current h
        overlap_full = correlate(h, 1 - h, mode='full')
        overlap = overlap_full[len(h)-1:]  # Only the "shifted" overlaps
        max_overlap_idx = np.argmax(overlap)
        max_overlap_val = overlap[max_overlap_idx]
        
        # If we found a better solution, update our best
        if max_overlap_val < best_c5:
            best_c5 = max_overlap_val
            best_h = h.copy()
            
        if best_c5 <= target_c5:
            break
            
        # Identify peaks to reduce
        sorted_indices = np.argsort(overlap)[::-1][:k_peaks]
        
        # Gradient-like update to reduce peaks
        h_new = h.copy()
        for i in sorted_indices:
            # Target peak at index i
            # Try to reduce height at indices around i
            for j in range(n_points):
                k_shift = i
                if j + k_shift < n_points:
                    # Reduce j and j+k_shift to decrease overlap
                    if h_new[j] > 0:
                        h_new[j] = max(0, h_new[j] - step_size)
                    if h_new[j + k_shift] > 0:
                        h_new[j + k_shift] = max(0, h_new[j + k_shift] - step_size)
                        
        # Project to feasible region: [0,1], sum to target_sum
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (target_sum / current_sum)
            
        h = h_new
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    dx = 2.0 / n_points
    # Compute overlap integral
    overlap_full = correlate(h, 1 - h, mode='full')
    overlap = overlap_full[n_points-1:]
    max_overlap = np.max(overlap)
    return max_overlap * dx
