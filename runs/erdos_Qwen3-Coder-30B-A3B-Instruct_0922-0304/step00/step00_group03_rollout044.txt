import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Initialize with alternating 0.5s
        n_points = 70
        h = np.full(n_points, 0.5)
    
    # Ensure constraint: sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Short-term gradient descent parameters
    lr = 0.01
    momentum = 0.9
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    max_iter = 5000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute all correlations for all shifts
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        h_flip = 1 - h
        h_flip_extended = np.concatenate([h_flip, np.zeros(n_points - 1)])
        
        # Full cross-correlation
        correlations = signal.correlate(h_extended, h_flip_extended, mode='full')
        correlations = correlations[n_points - 1:]  # Only positive shifts

        # Find maximum correlation
        max_idx = np.argmax(correlations)
        max_corr = correlations[max_idx]
        
        # If we've found a better bound
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
        if max_corr <= 0.3807:
            break
            
        # Compute gradient w.r.t. max_corr
        # For k=0 to n_points-1, the contribution of h[i] to the maximum
        # is 1 - h[i + k] for each k where the correlation part includes h[i]
        grad = np.zeros_like(h)
        for i in range(n_points):
            # For each shift k that contributes to max_corr
            for k in range(n_points):
                idx = i + k
                if idx < n_points:
                    grad[i] += h_flip[idx]
        
        # Apply gradient descent with momentum
        velocity = momentum * velocity - lr * grad
        h_new = h + velocity
        
        # Project onto constraints
        h_new = np.clip(h_new, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new *= scale_factor
        
        h = h_new
        
        iter_count += 1
        
    # Final evaluation
    h_extended = np.concatenate([best_h, np.zeros(n_points - 1)])
    h_flip = 1 - best_h
    h_flip_extended = np.concatenate([h_flip, np.zeros(n_points - 1)])
    correlations = signal.correlate(h_extended, h_flip_extended, mode='full')
    correlations = correlations[n_points - 1:]  # Only positive shifts
    final_c5 = np.max(correlations)
    
    return (list(best_h), final_c5, n_points)
