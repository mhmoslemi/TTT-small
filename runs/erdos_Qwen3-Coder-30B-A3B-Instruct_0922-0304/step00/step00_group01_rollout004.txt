import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute current overlap
        overlap_full = signal.correlate(h, 1 - h, mode='full')
        overlap = overlap_full[n_points - 1:]
        c5 = np.max(overlap) * dx
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= max_no_improvement or best_c5 < 0.3807:
            break
            
        # Identify top peaks to reduce
        peak_indices = np.argsort(overlap)[::-1][:5]
        
        # Update h to reduce overlap at peaks
        h_new = h.copy()
        for k in peak_indices:
            if k == 0:
                continue  # skip the first (no lag)
                
            # Reduce h where overlap is high
            for i in range(n_points - k):
                if overlap[k] > 0.01:  # threshold to avoid noise
                    # Reduce both h[i] and h[i+k]
                    h_new[i] = max(0, h_new[i] - 0.01)
                    h_new[i + k] = max(0, h_new[i + k] - 0.01)
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * target_sum / current_sum
            
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
