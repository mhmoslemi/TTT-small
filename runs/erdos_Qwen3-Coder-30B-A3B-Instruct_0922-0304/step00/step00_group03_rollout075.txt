import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a basic starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    tol = 1e-6
    step_size = 0.01
    
    start_time = time.time()
    
    # Normalize h to satisfy ∫h(x)dx = 1
    h = h * (1.0 / np.sum(h) * 2.0 / dx)
    
    def compute_c5(h_vals):
        # Compute correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Maximum correlation
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vals):
        # Keep h in [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to maintain constraint
        sum_h = np.sum(h_vals) * dx
        if abs(sum_h - 1.0) > 1e-8:
            # Scale h to satisfy integral constraint
            scale = 1.0 / sum_h
            h_vals = h_vals * scale
        return h_vals
    
    def get_gradient(h_vals, k_max):
        # k_max is the index of the maximum correlation
        # Simplified gradient estimate
        grad = np.zeros_like(h_vals)
        # For simplicity, approximate gradients with finite differences
        # This section builds on basic approximations (takes last constraints)
        # Just to 'look' at where the peak is, add more precise updates
        # Using finite differences for estimating gradient of the overlap
        # We sample only 10% of gradient 통해 Efficient updates

        for i in range(len(h_vals)):
            # A small finite difference approximation 
            eps = 1e-8
            h_pos = h_vals.copy()
            h_neg = h_vals.copy()
            h_pos[i] += eps
            h_neg[i] -= eps if i > 0 else 0
            h_pos = project_h(h_pos)
            h_neg = project_h(h_neg)
            c5_pos = compute_c5(h_pos)
            c5_neg = compute_c5(h_neg)
            grad[i] = (c5_pos - c5_neg) / (2 * eps)
            
        return grad
    
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    it = 0
    while it < max_iter and time.time() - start_time < budget_s - 1.0:
        it += 1
        # Compute c5 and the index of max overlap
        corr = np.correlate(h, 1-h, mode="full")
        k_max = np.argmax(corr)
        
        # Estimated gradient w.r.t the maximum overlap
        grad = get_gradient(h, k_max)
        
        # Improve by moving along negative gradient
        h_new = h - step_size * grad
        
        # Project H to satisfy constraints
        h_new = project_h(h_new)
        
        # If improvement, keep new h
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

        # Update h
        h = h_new
        c5 = c5_new
        
        # Early stopping
        if abs(c5 - best_c5) < tol:
            break
    
    # Final projection to maintain the constraints
    h_final = project_h(h)
    c5_final = compute_c5(h_final)
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_final
    
    return (best_h.tolist(), best_c5, len(best_h))
