import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with alternating 0.5s
        n_points = 70
        h = np.full(n_points, 0.5)
        # Set boundary values
        h[0] = 0.0
        h[-1] = 1.0
        h[1] = 0.0
        h[-2] = 1.0
    
    n_points = len(h)
    
    # Scale to meet constraint: sum(h) = n_points / 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    alpha = 0.01
    beta = 0.9  # momentum
    momentum = np.zeros_like(h)
    
    # Time tracking
    start_time = time.time()
    
    # Optimization loop
    last_improvement = 0
    best_h = h.copy()
    best_c5 = float('inf')
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations
        correlation = signal.correlate(h, 1 - h, mode='full')
        correlation = correlation * (2.0 / n_points)  # scale by dx
        
        max_corr = np.max(correlation)
        max_idx = np.argmax(correlation)
        max_k = max_idx - (n_points - 1)
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
        if max_corr < 0.3807:
            break
            
        # Compute gradient
        # Note: For simplicity, a gradient approximation is used
        # Let's construct the gradient for max_k shift
        g = np.zeros_like(h)
        k_start = max_k
        k_end = min(k_start + n_points, 2 * n_points - 1)
        
        # Fill in the gradient manually for the most relevant part
        for i in range(n_points):
            # Only considering overlapping indices from the relevant correlation
            try:
                # For the k-th shift, we look at h[i] * (1 - h[i+k])
                k_pos = i + k_start
                if 0 <= k_pos < n_points:
                    g[i] = (1 - h[k_pos])
            except:
                pass
        
        # Apply momentum
        momentum = beta * momentum + (1 - beta) * g
        # Update
        h_new = h - alpha * momentum
        
        # Clamp and rescale
        h_new = np.clip(h_new, 0, 1)
        h_sum_new = np.sum(h_new)
        if h_sum_new > 0:
            h = h_new * (n_points / 2) / h_sum_new
        else:
            h = np.full_like(h, 0.5)
            
        iter_count += 1
        
        # Decrease learning rate occasionally
        if iter_count % 100 == 0 and alpha > 0.0001:
            alpha *= 0.5
            
    return best_h.tolist(), best_c5, n_points
