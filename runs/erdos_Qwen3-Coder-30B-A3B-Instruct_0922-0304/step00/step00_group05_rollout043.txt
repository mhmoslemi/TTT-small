import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute C5 as max of overlap integrals
    n = len(h)
    # Correlate h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Take only the relevant part (negative lags)
    overlaps = corr[n-1:] * dx
    return np.max(overlaps)

def project_to_feasible(h, dx):
    # Ensure h is in [0,1] and sum(h) * dx = 1
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if total > 0:
        h = h * (1.0 / total)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Initialize parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    
    # Initialize swarm
    particles = np.random.rand(swarm_size, n_points)
    if 'initial_h_values' in globals():
        particles[0] = initial_h_values
    for i in range(swarm_size):
        particles[i] = project_to_feasible(particles[i], dx)
    
    velocities = (np.random.rand(swarm_size, n_points) - 0.5) * 0.5
    personal_best = particles.copy()
    personal_best_scores = np.array([evaluate_h(p, dx) for p in particles])
    
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5

    iter_count = 0
    while time.time() - start_time < budget_s:
        for i in range(swarm_size):
            if time.time() - start_time >= budget_s:
                break
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (w * velocities[i] 
                           + c1 * r1 * (personal_best[i] - particles[i])
                           + c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible set
            particles[i] = project_to_feasible(particles[i], dx)
            
            # Evaluate
            current_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best_scores[i] = current_score
                personal_best[i] = particles[i].copy()
                
                # Update global best
                if current_score < global_best_score:
                    global_best_score = current_score
                    global_best = particles[i].copy()
            
            iter_count += 1
        
        # Occasionally perturb particles to avoid local minima
        if iter_count % 100 == 0 and time.time() - start_time < budget_s:
            for i in range(swarm_size):
                if np.random.rand() < 0.05:
                    particles[i] += np.random.normal(0, 0.01, n_points)
                    particles[i] = project_to_feasible(particles[i], dx)
    
    return (global_best, global_best_score, n_points)
