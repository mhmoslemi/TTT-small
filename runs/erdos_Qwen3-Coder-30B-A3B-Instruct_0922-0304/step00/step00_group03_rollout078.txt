import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a reasonable starting vector if no initial_h_values
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Precompute correlation indices
    max_k = n_points - 1
    
    # Timing
    start_time = time.time()
    
    def compute_c5(h):
        # Compute the overlap integral using convolution
        # h is a step function, meaning h(x) is constant on intervals
        # We compute the maximum of all possible overlaps
        corr = signal.correlate(h, 1 - h, mode='full')
        # The correlation is of length 2*n - 1, and we want the maximum
        # between 0 and n-1 shifts (represented in the array)
        #s = len(corr)
        c5 = np.max(corr) * dx
        return c5
    
    def proj_h(h):
        # Project h to satisfy the constraint sum(h) * dx = 1
        # This means we need sum(h) = target_sum
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            scale = target_sum / current_sum
            h = h * scale
        return h
    
    def update_h(h, step_size=0.01, max_iter=20):
        c5 = compute_c5(h)
        # Find the shift with maximum overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        # The maximum is in the middle of the correlation array,
        # which represents shifts from -max_k to max_k
        shift_idx = np.argmax(corr)
        max_k_idx = n_points - 1
        shift_k = shift_idx - max_k_idx
        # For granularity of which samples are used
        # Let's compute the gradient at the shift that causes maximum overlap
        # For shift k, the contribution to C5 is sum h[i] * (1 - h[i+k])
        # For a fixed k, the gradient w.r.t. h[i] is (1 - h[i+k]) for i=0..n-k-1
        # For shift k, we compute the gradient of sum h[i] (1 - h[i+k])
        # This is (1 - h[i+k]) for each i where i+k is valid.
        # The indices for i such that h[i+k] makes sense are i = 0..n_points-k-1
        # But for updating h[i], the gradient is 1 - h[i+k]
        # The gradient of the full overlap with respect to h[i] is:
        # If i+k < n_points, then gradient += (1-h[i+k]) * dx
        # But we also need to consider only the part that's affected by this k
        
        # Compute the gradient explicitly
        grad = np.zeros(n_points)
        # Compute which i values contribute to shift k
        for i in range(n_points):
            if i + shift_k >= 0 and i + shift_k < n_points:
                grad[i] += 1 - h[i + shift_k]
        
        grad = grad * dx
        # Gradient descent update
        h_new = h - step_size * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Project to enforce the constraint
        h_new = proj_h(h_new)
        
        return h_new
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Update until hitting time limit or no meaningful progress
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Normal update with gradient descent on the max overlap
        h = update_h(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5

    # Final projection to satisfy constraint
    final_h = proj_h(best_h)
    final_c5 = compute_c5(final_h)
            
    return (final_h, final_c5, n_points)
