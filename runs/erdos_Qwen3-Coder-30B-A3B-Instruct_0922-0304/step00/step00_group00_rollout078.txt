import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # Compute C5 using full correlation
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    corr = correlate(h, 1 - h, mode="full")
    # Max overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: uniform distribution
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to ∫h = 1
    dx = 2.0 / n_points
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum

    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    alpha = 0.1
    max_iter = 1000000
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement_count = 0

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        # Find k that maximizes overlap
        max_k = np.argmax(corr)
        max_overlap = corr[max_k]
        # k is from -n+1 to n-1, centered at n-1
        k = max_k - (n_points - 1)
        
        # Compute gradient w.r.t h for this k
        grad = np.zeros_like(h)
        # Add contributions to grad
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
                
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Renormalize
        h_new_sum = np.sum(h_new)
        if h_new_sum > 0:
            h_new = h_new * (n_points / 2) / h_new_sum
            
        h = h_new
        
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break

    return (best_h, best_c5, n_points)
