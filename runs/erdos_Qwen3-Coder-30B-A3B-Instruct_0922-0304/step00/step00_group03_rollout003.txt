import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if exists
    try:
        h0 = initial_h_values.copy()
    except NameError:
        # Fallback: create simple construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        h0 = h0 * 2.0 / np.sum(h0)  # normalize to area 1
    
    n_points = len(h0)
    dx = 2.0 / n_points
    max_iter = 20000
    tol = 1e-6
    step_size = 0.01
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all shifts
        cross_corr = np.correlate(h, 1 - h, mode='full')
        c5_val = np.max(cross_corr) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Find which shift gives maximum overlap
        max_idx = np.argmax(cross_corr)
        shift_len = len(cross_corr)
        k = max_idx - (shift_len // 2)
        
        # Compute gradient of overlap for the active constraint
        # For shift k, overlap = sum h[i] * (1 - h[i+k]) 
        # Gradient w.r.t. h[i] = 1 - h[i+k] if i+k is valid, else 0
        # But we need to consider the case where this is an internal term
        # We compute it as the derivative term we can numerically get
        
        # It's easier to work with gradient. We use finite differences to get gradient
        # of overlap w.r.t. h for the shift that gives the max overlap
        # But for efficiency, let's compute analytically. For k-th shift:
        # overlap = sum_{i=0}^{n-1} h[i] * (1 - h[i+k]) = sum h[i] - sum h[i]*h[i+k]
        # Gradient w.r.t. h[j] is: 1 if j is not shifted, -h[j+k] if j+k exists, 0 otherwise
        # loss = -overlap, so gradient is: h[j+k] - 1 if such j+k exists, 0 otherwise
        
        # For our setup: we consider the weights that contribute to overlap with shift k
        # Compute gradient of overlap w.r.t h, accounting for the k-shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] = h[i + k] - 1.0
            elif i + k < 0:
                grad[i] = -1.0  # For cases where i+k < 0 but we add from non-zero h
                
        # Clip gradient to ensure valid updates
        grad = np.clip(grad, -1.0, 1.0)
        
        # Make update step
        h_new = h - step_size * grad
        
        # Project back to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Force the sum constraint to maintain area = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Ensure we don't go below tolerance too much (numerically)
        h = h_new
        
    # Final validity check
    final_sum = np.sum(h) * dx
    if abs(final_sum - 1.0) > 1e-3:
        h = h * (1.0 / final_sum)
        
    # Compute final C5
    cross_corr = np.correlate(h, 1-h, mode='full')
    final_c5 = np.max(cross_corr) * dx
    
    return (best_h, final_c5, n_points)
