import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    # Optimization parameters
    alpha = 0.01
    beta = 0.9
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode="full")
        n_corr = len(corr)
        c5 = 0
        max_k = n_points - 1
        for k in range(n_corr):
            # k ranges from 0 to 2*n_points-2 but we only consider valid shifts
            shift = k - (n_points - 1)
            if abs(shift) >= n_points:
                continue
            corr_val = corr[k] * dx
            if corr_val > c5:
                c5 = corr_val
        
        # Early termination
        if c5 < 0.3807:
            break
        
        # If this is better, store it
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(-n_points + 1, n_points):
            # Create correlation shifted properly
            # We're computing derivatives for each element w.r.t max overlap
            if k >= 0:
                # h[i] appears in term: h[i] * (1 - h[i+k])
                # Gradient from h[i] is (1 - h[i+k])
                for i in range(n_points - k):
                    grad[i] += (1 - h[i + k])
            else:
                # k < 0: h[i] appears as h[i+k] term, which is 1-h[i]
                # So if h[i] interacts with h[i+k], 
                # gradient = derivative of h[i]=(1 - h[i+k])
                for i in range(-k, n_points):
                    grad[i] += -h[i + k]
                    
        # Normalize gradient
        grad = grad / np.linalg.norm(grad) * alpha if np.linalg.norm(grad) > 0 else grad
        
        # Apply momentum
        momentum = beta * momentum + grad
        h_new = h - momentum
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
        
        h = h_new
        iteration += 1
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode="full")
    n_corr = len(corr)
    c5 = 0
    for k in range(n_corr):
        shift = k - (n_points - 1)
        if abs(shift) >= n_points:
            continue
        corr_val = corr[k] * dx
        if corr_val > c5:
            c5 = corr_val
            
    return (best_h.tolist(), c5, n_points)
