import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2.0) / h_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Optimizer parameters
    lr = 0.01
    beta = 0.9
    momentum = np.zeros_like(h)
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        # Compute overlaps for all shifts
        # This computes correlation of h with (1-h)
        corr = correlate(h, 1 - h, mode='full')
        # The full correlation has length 2*n_points - 1
        # The valid overlaps are for shifts -n_points+1 to n_points-1
        # But we're interested in max of positive shifts (k in [1, n_points-1]) 
        # and negative shifts (k in [-n_points+1, -1])
        # We want max of all these values
        
        # In summary, corr contains [k=-n+1 to n-1] for full correlation
        # We exclude k=0 (no shift) -- because h(x) (1 - h(x+0)) is h(x)(1-h(x))
        # But k=0 is not in our constraints
        # However, for the first overlap calculation:
        # To make sure we compute the integral over [0,2] and not [0,2) we handle k from 1 to n_points-1 and -n_points+1 to -1
        
        # Extract relevant values: 
        # For k from 1 to n_points-1, indices are n_points to 2*n_points-2
        # For k from -n_points+1 to -1, indices are 0 to n_points-2
        left_part = corr[:n_points-1]   # k=-n+1 to -1, indices 0 to n_points-2
        right_part = corr[n_points+1:]  # k=1 to n_points-1, indices n_points+1 to 2*n_points-2
        max_k = np.argmax(np.concatenate([left_part, right_part]))
        max_shift = max_k - (n_points-1)  # Convert back to shift index
        
        if max_shift < 0:
            corr_id = max_k
        else:
            corr_id = max_k + n_points
         
        # For more accurate gradient:
        # The overlap for shift k is:
        # sum(h[i] * (1 - h[i+k])) for i = 0 to n-k-1 
        # Gradient of overlap w.r.t h[j]:
        #   For j = i, it's (1-h[i+k]) 
        #   For j = i+k, it's -h[i]
        #   Everything else is 0
        
        # So, we need to compute the gradient wrt all h_i at once.
        # For shift k, the overlap is sum(h[i] * (1 - h[i+k]))
        # So for i = 0 to n-1-k:
        #   d overlap / d h[i] = (1 - h[i+k]) if i+k < n, else 0
        #   d overlap / d h[i+k] = -h[i] if i >= 0, else 0
        # Let's track the indices that matter:

        grad = np.zeros_like(h)
        # Now compute gradient for current max overlap shift
        k = max_shift
        if k > 0:
            # h[i] terms for i = 0 to n-k-1 get (1 - h[i+k])
            # h[i+k] terms for i = 0 to n-k-1 get -h[i]
            # Add to gradient:
            for i in range(n_points - k):
                grad[i] += 1 - h[i+k]
            for i in range(n_points - k):
                grad[i+k] -= h[i]
        elif k < 0:
            k_abs = -k
            for i in range(n_points - k_abs):
                grad[i] += 1 - h[i+k_abs]
            for i in range(n_points - k_abs):
                grad[i+k_abs] -= h[i]
        else:
            # k == 0:
            # Gradient of overlap = 0 for all h, so this shouldn't happen as i > 0
            # Let's not update if we are in this case
            pass
        
        # Weighted descent with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h = h - lr * momentum
        
        # Project to feasible set: [0,1] and sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2.0) / h_sum
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        left_part = corr[:n_points-1]
        right_part = corr[n_points+1:]
        current_c5 = np.max(np.concatenate([left_part, right_part]))
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        iter_count += 1
        if current_c5 <= 0.3807:
            break
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    left_part = corr[:n_points-1]
    right_part = corr[n_points+1:]
    final_c5 = np.max(np.concatenate([left_part, right_part]))
    
    return (best_h.tolist(), final_c5, n_points)
