import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure the constraint sum(h) * dx = 1, i.e., sum(h) = n_points / 2
    target_sum = n_points / 2.0
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        h = h * target_sum / h_sum

    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for simulated annealing
    T = 0.1
    cooling_rate = 0.995
    min_T = 1e-6
    perturb_frac = 0.05
    delta = 0.1
    max_perturb = int(n_points * perturb_frac)
    
    iteration = 0
    while time.time() - start_time < budget_s and T > min_T:
        # Randomly select indices to perturb
        n_perturb = min(max_perturb, n_points - 1)
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Make a copy of h to test the perturbation
        h_new = h.copy()
        
        # Perturb the selected indices
        for i in indices:
            # Add a small random perturbation within [0,1]
            delta_val = np.random.uniform(-delta, delta)
            h_new[i] += delta_val
            # Clamp to [0,1]
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Renormalize to meet sum(h) = n_points / 2
        h_sum_new = np.sum(h_new)
        h_new = h_new * target_sum / h_sum_new

        # Evaluate new C5 value
        new_c5 = evaluate_erdos_solution(h_new, dx)
        
        # Accept or reject the change
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.exp(-delta_c5 / T) > np.random.rand():
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Cool the system
        T *= cooling_rate
        
        iteration += 1
        if iteration % 1000 == 0:
            if best_c5 < 0.3807:
                break

    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute the full correlation, which corresponds to max_k \int h(x)(1 - h(x+k)) dx
    correlation = correlate(h, 1 - h, mode="full")
    max_correlation = np.max(correlation)
    return max_correlation * dx
