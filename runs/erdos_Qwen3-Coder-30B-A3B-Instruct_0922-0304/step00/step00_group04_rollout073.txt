import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h_values):
    # This is a stub - in the actual environment this function is pre-imported
    # It computes the maximum overlap integral
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        n_points = 42
        # Create initial step function
        h0 = np.ones(n_points) * 0.5
        
    h_best = np.array(h0, dtype=float)
    c5_best = evaluate_erdos_solution(h_best)
    
    # Convex relaxation helper
    def convex_relaxation(h0, n_points):
        dx = 2.0 / n_points
        
        # Variables
        h = Variable(n_points)
        
        # Objective variable
        C5 = Variable()
        
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2.0,
        ]
        
        # For the maximum over all lags: add constraints for all possible lags
        # But since we optimize greedily on one lag, we'll compute it only when needed
        # This is done with a simplified approach in the refinement step
        
        # Create the problem
        objective = Minimize(C5)
        prob = Problem(objective, constraints)
        
        # Try to solve
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                return h.value
        except:
            pass
        return h0
    
    # Gradient-based refinement
    def gradient_descent_refine(h_start, n_points, max_iter=100):
        dx = 2.0 / n_points
        h = np.array(h_start, dtype=float)
        
        for _ in range(max_iter):
            # Compute correlations
            corr = np.correlate(h, 1 - h, mode='full')
            # Get max correlation index
            max_idx = np.argmax(corr)
            k = max_idx - (n_points - 1)  # corresponding lag
            
            # Compute gradient with respect to h for this specific lag
            # Derivative of h(x) * (1 - h(x + k)) with respect to h(x):
            # This is more complex; we'll estimate with finite difference
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_minus = h.copy()
                h_plus[i] += eps
                h_minus[i] -= eps
                
                # Clip to [0,1]
                h_plus = np.clip(h_plus, 0, 1)
                h_minus = np.clip(h_minus, 0, 1)
                
                # Recompute the specific correlation term
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')[max_idx]
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')[max_idx]
                
                grad[i] = (corr_plus - corr_minus) / (2 * eps)
                
            # Update h in direction of negative gradient
            lr = 0.1
            h_new = h - lr * grad
            
            # Project back to [0,1] and rescale
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum to n_points/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            h = h_new
            current_c5 = evaluate_erdos_solution(h)
            
            if current_c5 < c5_best:
                c5_best = current_c5
                h_best = h.copy()
                
        return h
    
    # Stage 1: Low-resolution convex optimization
    if time.time() - start_time < budget_s * 0.5:
        h_convex = convex_relaxation(h0, n_points)
        h_best = gradient_descent_refine(h_convex, n_points, 50)
        c5_best = evaluate_erdos_solution(h_best)
    
    # Stage 2: High-resolution refinement (upsample to 84 points if time allows)
    if time.time() - start_time < budget_s * 0.8:
        upsample_ratio = 2
        n_upsample = n_points * upsample_ratio
        dx_upsample = 2.0 / n_upsample
        
        # Interpolate
        h_upsample = np.interp(
            np.linspace(0, 2, n_upsample),
            np.linspace(0, 2, n_points),
            h_best
        )
        
        # Convex relaxation
        h_convex = convex_relaxation(h_upsample, n_upsample)
        h_best = gradient_descent_refine(h_convex, n_upsample, 50)
        c5_best = evaluate_erdos_solution(h_best)
    
    # Return best solution
    return (h_best.tolist(), c5_best, len(h_best))
