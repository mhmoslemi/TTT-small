import numpy as np

import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial construction if available

    except NameError:
        # Fallback to random construction
        n_points = 42

        h0 = h0 / np.sum(h0) * (n_points / 2)

    h0 = np.clip(h0, 0, 1)
    
    # Scale to correct integral
    h0 = h0 / np.sum(h0) * (n_points / 2)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    t = 0
    
    while t < budget_s:
        # Compute cross-correlation
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        dx = 2.0 / n_points
        corr *= dx
        
        # Find maximum
        max_corr = np.max(corr)
        max_k = np.argmax(corr) - (len(best_h) - 1)
        
        # Compute gradient w.r.t. the maximum lag
        grad = np.zeros_like(best_h)
        if max_k >= 0:
            for i in range(len(best_h) - max_k):
                grad[i] += (1 - best_h[i + max_k])
        else:
            for i in range(-max_k, len(best_h)):
                grad[i] += (1 - best_h[i + max_k])
        
        # Gradient descent step
        lr = 0.01
        new_h = best_h - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h / np.sum(new_h) * (n_points / 2)
        
        # Evaluate
        new_c5 = evaluate_erdos_solution(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h
            best_c5 = new_c5
        
        t = time.time() - start_time
        if t > budget_s:
            break
    
    return (best_h, best_c5, n_points)


def evaluate_erdos_solution(h):
    """Evaluates the Erdős overlap constant for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
