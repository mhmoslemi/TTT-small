import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    # Initialize parameters
    n_points = 42
    dx = 2.0 / n_points
    lag_max = n_points - 1
    
    # Get initial solution if available
    try:
        h_start = np.array(initial_h_values)
        if len(h_start) != n_points:
            h_start = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except NameError:
        # Create a simple initial guess
        h_start = np.ones(n_points) * 0.5
    
    # Set up MILP optimization
    start_time = time.time()
    
    # Variables
    h = cp.Variable(n_points, bounds=(0, 1))
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        C5 >= 0
    ]
    
    # Overlap constraints for all lags
    for k in range(-lag_max, lag_max + 1):
        if k < 0:
            # Right shift
            corr = cp.sum(cp.multiply(h[:n_points + k], 1 - h[-k:]))
        elif k > 0:
            # Left shift
            corr = cp.sum(cp.multiply(h[k:], 1 - h[:n_points - k]))
        else:
            # No shift
            corr = cp.sum(cp.multiply(h, 1 - h))
        constraints.append(corr <= C5 * dx)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Create and solve problem
    prob = cp.Problem(objective, constraints)
    
    # Solve using CBC solver (fast and reliable for this scale)
    try:
        prob.solve(solver=cp.CBC, timeout=budget_s - 10, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
        else:
            h_opt = h_start
    except:
        h_opt = h_start
    
    # Round to binary values (maintaining sum constraint)
    h_binary = np.zeros(n_points)
    
    # Use greedy method to round
    indices = np.argsort(-h_opt)  # Sort by descending values
    ones_count = n_points // 2
    
    # Set the highest values to 1
    h_binary[indices[:ones_count]] = 1
