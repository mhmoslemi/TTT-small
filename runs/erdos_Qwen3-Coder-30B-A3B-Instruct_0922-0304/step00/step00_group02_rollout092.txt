import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C5 for a given h."""
    dx = 2.0 / len(h)
    # Compute overlaps using full convolution
    overlaps = correlate(h, 1 - h, mode="full")
    # The first (n-1) elements correspond to k=0,1,...,n-2 in the problem.
    # But the actual overlap is computed with h[i] * (1 - h[i+k])
    # which we compute using correlate with flipped 1-h.
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def make_feasible(h):
    """Ensure h is feasible: values in [0,1] and sum = n/2."""
    n = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n / 2
    if abs(current_sum - target_sum) > 1e-6:
        # Adjust values to make sum correct
        diff = target_sum - current_sum
        # Distribute the difference to make the sum correct
        h = h + diff / n
        h = np.clip(h, 0, 1)
    return h

def mutate_h(h, rate=0.01):
    """Mutate h with given rate."""
    n = len(h)
    mask = np.random.random(n) < rate
    if np.any(mask):
        # Randomly change masked elements
        h = h.copy()
        h[mask] += np.random.normal(0, 0.05, np.sum(mask))
        h = make_feasible(h)
    return h

def crossover_h(h1, h2):
    """Perform crossover between two vectors."""
    n = len(h1)
    # Random crossover point
    point = np.random.randint(1, n)
    child1 = np.concatenate([h1[:point], h2[point:]])
    child2 = np.concatenate([h2[:point], h1[point:]])
    child1 = make_feasible(child1)
    child2 = make_feasible(child2)
    return child1, child2

def initialize_population(n_points, pop_size=100):
    """Initialize a population of feasible h vectors."""
    pop = []
    # Use the initial construction
    if 'initial_h_values' in globals():
        h0 = make_feasible(np.array(initial_h_values))
        pop.append(h0)
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = make_feasible(h0)
        pop.append(h0)
    
    # Add random individuals
    for _ in range(pop_size - len(pop)):
        h = np.random.random(n_points)
        h = make_feasible(h)
        pop.append(h)
    return pop

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    if 'initial_h_values' in globals():
        n_points = len(initial_h_values)
    
    # Initialize population
    population = initialize_population(n_points, pop_size=100)
    best_h = None
    best_c5 = float('inf')
    
    gen_no_improve = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and gen_no_improve < 100:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(1 / (c5 + 1e-10))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Selection: top 50% based on fitness
        sorted_idx = np.argsort(fitnesses)[::-1]
        top_idx = sorted_idx[:len(population)//2]
        selected = [population[i] for i in top_idx]
        
        # Create new population
        new_pop = selected.copy()
        while len(new_pop) < len(population):
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            child1, child2 = crossover_h(parent1, parent2)
            
            # Apply mutation with 1% rate
            child1 = mutate_h(child1, rate=0.01)
            child2 = mutate_h(child2, rate=0.01)
            
            new_pop.extend([child1, child2])
        
        population = new_pop[:len(population)]
        generation += 1
        
        # Check no improvement
        if generation > 1 and best_c5 == float('inf'):
            gen_no_improve += 1
        else:
            gen_no_improve = 0
            
    # Final update to correct best_h
    if best_h is None:
        best_h = population[0]
        best_c5 = evaluate_h(best_h)
    else:
        # Re-evaluate best to ensure it is accurate
        best_c5 = evaluate_h(best_h)
    
    # Make sure best_h is fully feasible
    best_h = make_feasible(best_h)
    return (best_h.tolist(), best_c5, len(best_h))
