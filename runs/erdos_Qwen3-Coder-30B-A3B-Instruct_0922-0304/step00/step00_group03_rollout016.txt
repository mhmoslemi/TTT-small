import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # If no initial construction, create one
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    # Ensure the integral constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Learnable parameters
    alpha = 0.01
    beta = 0.9
    momentum = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization loop
    for iteration in range(10000):
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        # The full correlation has length 2*n_points - 1, but we only care about shift indices
        # for non-zero shifts from -n_points+1 to n_points-1
        shifts = range(-n_points + 1, n_points)
        # Take middle part corresponding to non-zero shifts
        half_len = len(corr) // 2
        corr_vals = corr[half_len - (n_points - 1):half_len + (n_points - 1)]
        
        # Find the maximum overlap
        max_overlap = np.max(corr_vals)
        max_k = shifts[np.argmax(corr_vals)]
        
        # Compute gradient of max_overlap w.r.t. h
        # Gradient: for each h[i], count how many times it's involved
        # in a product in the overlap sum
        grad = np.zeros_like(h)
        for i in range(n_points):
            # We compute how many times h[i] appears in a product for a shift k
            # For overlap at shift k, we sum over i: h[i] * (1 - h[i + k])
            # The gradient w.r.t. h[i] is the sum over k of (1 - h[i+k]) if k exists  
            for k in range(-n_points+1, n_points):
                # If k is valid (i+k is within bounds)
                if 0 <= i + k < n_points:
                    grad[i] += (1 - h[i + k])
        
        # Update with momentum
        momentum = beta * momentum + (1 - beta) * grad
        h = h - alpha * momentum

        # Project onto feasible set
        h = np.clip(h, 0, 1)
        # Rescale to meet integral constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum

        # Evaluate and update best
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early termination if target reached
        if best_c5 <= 0.3807:
            break
            
        # Decrease learning rate every 100 iterations
        if iteration % 100 == 0:
            alpha *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    # Perform the correlation again
    corr = correlate(h, 1 - np.array(h), mode="full")
    n = len(h)
    half_len = len(corr) // 2
    corr_vals = corr[half_len - (n - 1):half_len + (n - 1)]
    return np.max(corr_vals)
