import numpy as np
from scipy.optimize import minimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlations for all lags
    correlations = np.correlate(h, 1 - h, mode="full")
    # Take the maximum value and multiply by dx
    max_corr = np.max(correlations)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        # Generate a reasonable starting point
        h = np.ones(n_points) * 0.5  # uniform distribution
        h = h / np.sum(h) * target_sum
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Convex relaxation approach first
    # Only proceed if we have enough time
    if time.time() - start_time < budget_s * 0.8:
        try:
            # Formulate the SDP relaxation
            # Use variables for correlations
            h_vars = Variable(n_points)
            
            # Constraint 1: sum(h) = target_sum
            constr1 = sum_entries(h_vars) == target_sum
            
            # Constraint 2: 0 <= h <= 1 
            constr2 = [h_vars >= 0, h_vars <= 1]
            
            # For correlation constraints, approximate with linear constraints on
            # the linearized version of max over correlations.
            
            # Shortcircuit to a reasonable SDP approach
            # This approach is intended to be rough, i.e., approximation or heuristics
            # The real computation might take too long
            
            # Instead, let's start by optimizing with gradient descent
            # to get a better local improvement step
            def obj_func(h_vals):
                h_vals = np.maximum(0, np.minimum(1, h_vals))
                return evaluate_erdos_solution(h_vals)
            
            def constraint_func(h_vals):
                return np.sum(h_vals) - target_sum
                
            constraints = [{'type': 'eq', 'fun': constraint_func}]
            
            # Try a simple gradient descent with bounds
            opt = minimize(
                obj_func,
                h,
                method='SLSQP',
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=constraints,
                options={'maxiter': 500, 'ftol': 1e-6}
            )

            # Check if the optimization improved
            if not opt.failed and opt.fun < best_c5:
                best_h = opt.x.copy()
                best_c5 = opt.fun
                
        except Exception as e:
            pass
    
    # Now at most 20% of time left, perform quick local search 
    # If the previous did not work or time is left
    if time.time() - start_time < budget_s * 0.95:
        # Try a few random starts to escape local minima
        for _ in range(20):
            if time.time() - start_time > budget_s * 0.95:
                break

            h_trial = h + np.random.normal(0, 0.02, n_points)
            h_trial = np.clip(h_trial, 0, 1)
            h_trial = h_trial / np.sum(h_trial) * target_sum

            c5_trial = evaluate_erdos_solution(h_trial)
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()

    # Return best result found
    return (best_h.tolist(), best_c5, n_points)
