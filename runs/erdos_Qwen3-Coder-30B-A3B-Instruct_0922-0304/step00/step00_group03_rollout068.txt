import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with alternating values
        n_points = 70
        h = np.full(n_points, 0.5)
        
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    alpha = 0.01
    beta = 0.9
    momentum = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute correlation indices
    n = len(h)
    dx = 2.0 / n
    
    def compute_c5(h_vals):
        # Compute full correlation of h with (1-h)
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # The shifts go from -n+1 to n-1, so we need to extract valid overlaps
        # The maximum overlap corresponds to the largest value among valid correlations
        # For shift k=0, k=1,...,k=n-1 and k=-1,...,-(n-1)
        # Correlation indices are from 0 to 2*n-2
        max_corr = 0
        for k in range(1, n):
            # k=1 to n-1: shifting h forward
            # k=-n+1 to -1: shifting h backward
            # The correlation index for shift k is n-1+k
            idx = n-1+k
            if 0 <= idx < len(corr):
                max_corr = max(max_corr, corr[idx])
            idx = n-1-k
            if 0 <= idx < len(corr):
                max_corr = max(max_corr, corr[idx])
        return max_corr * dx

    def compute_c5_full(h_vals):
        # Computes full correlation, including all possible shifts
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
        # The cross-correlation for shifts from -n+1 to n-1 is stored at indices n-1 to 2*n-2
        # For positive shifts, indices [n, 2*n-2] and negative shifts [0, n-2]
        # But actually indices from [0, 2*n-2]
        # The correlation at index i corresponds to k = i - (n-1)
        # k ranges from -(n-1) to n-1, so valid k values are from 0 to n-1 and from -(n-1) to -1
        # Correlation for non-zero shift k is at index (n-1)+k = n-1+k
        # Thus we consider terms for all valid k
        # Specifically:
        # k in [1, n-1] (shifts), idx = n-1+k 
        # k in [-n+1, -1] (shifts), idx = n-1+k  
        # Only the upper and lower half of self-correlation will matter for overlaps
        # Total size is 2*n - 1, only one maximum value
        # extract maximum, subtracting 1 due to indexing
        # Let's directly find the maximum valid value
        max_val = 0.0
        # Equivalent to all values except the center
        for i in range(len(corr)):
            # For shift k = i - (n-1) = i - n + 1
            # We consider valid indices (inner overlap)
            if 0 <= i < len(corr):
                max_val = max(max_val, corr[i])
        return max_val * dx

    def compute_gradient(h_vals, k_star):
        # Compute gradient for a given shift k_star
        # This is where h[i] contributes to correlation at shift k
        # Gradient w.r.t. h[i] of overlap integral at shift k is 1 - h[i+k] for k>=0, and 1 - h[i-k] for k<0 (but index handling)
        # This is complex, more efficient approach:
        # For shift k and valid overlap (i,k), the contribution to overlap is h[i] * (1 - h[i+k]) 
        # So d/dh[i] = 1 - h[i+k] if k >= 0 and i+k < n
        # For positive k, gradient at i = 1 - h[i+k], for negative k, gradient at i = 1 - h[i-k]
        # Differentiate sum_over_all_shifts = sum_over_all_i h[i](1-h[i+k])  
        # Then d/dh[i] = (1-h[i+k]) if index i+k is valid. But assignments have to be for each valid overlapping pair
        # Better approach:   gradient for shift k is -h[i+k]  for k at and -h[i-k] for k negative         # complex to ...
        # For simplicity, approximate gradient w.r.t overall C5
        # But with current structure we can return the entire gradient
        # Let's compute for shift k_star by extracting the relevant indices
        # and how contributions separate to form the gradient
        grad = np.zeros_like(h_vals)
        # Compute raw cross-correlation implemented differently
        # Full cross-correlation length might be 2*n - 1
        # For shift k in [-n+1, n-1], index in cross-corr = n-1+k
        # The overlap for shift k at positions {i} where h[i], h[i+k] are both valid is sum h[i](1-h[i+k])
        # So we're looking for gradient of that quantity w.r.t. h
        # Each term contributes: d/dh[i] (h[i] (1-h[i+k])) = (1-h[i+k])
        # for positions where correlation exists, which implies (i,i+k) are within [0,n) 
        # This translates to 0 <= i <= n-1 and 0 <= i+k <= n-1
        # So: i >= 0, i+k <= n-1 => i <= n-1-k, i >= -k => i >= max(0, -k)
        # When k >= 0, i >= 0, i <= n-1-k => 0 <= i <= n-1-k
        # When k < 0, i >= -k, i <= n-1 => -k <= i <= n-1
        # Thus: for k >= 0, i ranges from 0 to n-1-k, for k < 0, i ranges from -k to n-1
        # So:  
        # Want sum_{i} h[i] (1-h[i+k]) which relates to positions with overlap
        # For the derivative we get:
        # d/dh[i] sum_{k, i valid} h[i] (1-h[i+k]) = 
        # For each i: 
        #    -1 if i ≥ -k and i' = i+k valid
        #    etc
        # More specific: treat k-th shift explicitly and find matching terms
        # For shift k (meaning h[i] multiplied with h[i+k]), Jacobian element for h[i'] is nonzero only if i'=i-k  (for k positive)
        # Better we compute convolution explicitly with symbolic specifics:
        # To differentiate total sum over all k: sum_k (sum_i h[i] (1 - h[i+k]))
        # Grad_k = sum_i (1 - h[i+k]) where this term contributes.  
        # But that’s an arbitrary shift
        # Let’s directly principle: compute cross-correlation, extract for k_max
        # Differentiate function at k_star for maximum overlap
        # This is the transformed correlation value.
        # Since C5_k = sum_i h[i] (1 - h[i+k]) = sum_i h[i] - sum_i h[i] h[i+k] = 1 - sum h[i] h[i+k]
        # So dC5/dh[i] = - sum_k (d/dh[i] h[i] h[i+k])  
        # d/dh[i] h[i] h[i+k] = h[i+k] if k >= 0 else h[i-k]  
        # ❗ +
        # x  = h.forward
    
        # This is hard to compute manually; correct it with a numerical hybrid approach:
        # Instead, recover the actual k that maximizes the sum. k_star determines which terms contribute.
        # So draw of every i, if term overlaps with i+k_star (that is 0 <= i, i+k_star < n)  
        # then dC/d h[i] = (1 – h[i+k_star])    # if i+k_star < n 
        # and if k_star < 0, then (1 – h[i-k_star])
        # 
        # Using that, replace with direct approach:
        # View the full cross-correlation with full indices: 2n-1 
        # Compute derivatives directly for true I(k), sum of overlaps 
        # Gradient.  For k>0, i+k <= n-1, so d/dh[i] I[k] = 1 - h[i+k],  (i <= n-1-k) 
        # For k<0, i >= -k and i < n, so d/dh[i] I[k] = 1 - h[i-k] for i >= -k and i < n
        # Under this approach:
        # Compute for shift k_star (at max overlap), sum over all i in valid range:
        # accumulated gradient from shift k_star  
        # Gradient from position i is 1 - h[i+k_star] if k_star >= 0 and i+k_star < n  
        # For k_star < 0, gradient from i is 1 - h[i-k_star] if i-k_star < n, i >= 0
        # So this becomes:
        # For k >= 0, we have i up to n-1-k, and element w.r.t. h[i] = 1 - h[i+k]  
        # For k < 0, we have i down to -k, and element w.r.t. h[i] = 1 - h[i-k]  
        # Thus for k_star definitely:
        # rank 1 p = 1 - h[i + k_star] if k_star >= 0, i in [0, n-1-k_star]
        # q = 1 - h[i - k_star] if k_star < 0, i in [-k_star, n-1]  
        # Actually same formula for both, since:
        #   For index i = 0..n-1-k_star
        #   shift + k = i + k_star, = i' = i + k_star .  
        #    For k_star >= 0:  
        #      i' >= i, i' <= n-1  means: i' is valid index so we take 1 - h[i']
        #      
        # Alternate idea:
        # Let’s write gradient explicitly
        # Run smallest k with maximum overlap, find number of indices  
        # Fastولي correct approach:  
        # When we're done with the loop, we have correlation at indices...
        # A simpler way (for now) : just run numerical derivative or finite diff to explore effect
        # Try convoluational numerical solution.
        
        # Let's compute for each i what values of j contribute to h[i] term in max overlap
        # Say k_star = k > 0, the term h[i](1-h[i+k]) contributes to C5_k_star = \sum_i h[i](1 - h[i+k])
        # So dC5_k_star/dh[i] = 1 - h[i+k], k >= 0.
        #
        # if we grab the largest term at index k_star:
        # Define vx = h[i]
        # Then vx * (1 - h[i+k_star]) so grad[vx] = 1 - h[i+k_star] for valid i+k_star
        # i local or global - we are differentiating at a fixed shot of h
        if k_star >= 0:
            # gradient at i = 1 - h[i + k_star] if i + k_star < n
            for i in range(n):
                if i + k_star < n:
                    # v[i] points to 1-h[i+k], add sign to accumulate so
                    # grad = dC5_kstar / dh[i] = 1 - h[i + k_star]
                    grad[i] += 1 - h[i + k_star]
        else:
            # k_star < 0, so we shift later 
            # For i+k_star >= 0, and i < n (so that k_star >= -i)
            # build geneate list of valid i where k_star + i >= 0
            # h[i - |k_star|] when i >= -k_star  
            for i in range(-k_star, n):   # i from -k_star to n-1
                # How we map: i_k = i + k_star  >= 0,  -> i_k = i + k_star, so i = i_k - k_star 
                # i_k such that i = i_k - k_star >= 0 => i_k >= -k_star = |k_star| 
                # So we are really adding d/dh[i] = 1 - h[i - k_star]
                # grad[i] += 1 - h[i - k_star]
                # But h[i - k_star] = h[i + |k_star|]
                if i - k_star < n and i >= 0:
                    grad[i] += 1 - h[i - k_star]
        return grad

    grad = np.zeros_like(h)
    iters = 0
    while time.time() - start_time < budget_s - 1:
        iters += 1
        if iters > 100000:
            break
        # Compute current C5
        c5 = compute_c5_full(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()  
        if c5 < 0.3807:
            break
            
        # Compute cross-correlation across all offsets
        corr = signal.correlate(h, 1 - h, mode='full')
        # Get the most contributing shift k
        # Correlation indices from 0 to 2*n-2
        # shift range k in [-n+1, n-1]
        # index in corr corresponds to shift k = i - (n-1)  
        # so i for shift k = n-1 + k
        max_corr_idx = np.argmax(corr)
        k_star = max_corr_idx - (n-1)
        
        # Compute the analytic gradient
        grad = compute_gradient(h, k_star)
        
        # Momentum-based update
        momentum = beta * momentum + (1 - beta) * grad
        h = h - alpha * momentum
        
        # Clamp to feasible range
        h = np.clip(h, 0, 1)
        
        # Renormalize to sum constraint
        h = h * (n / 2) / np.sum(h)
        
        # Decrease learning rate
        if iters % 100 == 0:
            alpha *= 0.9

    # Final validation on result
    final_c5 = compute_c5_full(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
