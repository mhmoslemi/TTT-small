import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    correlation = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts 0 to 2*n-2)
    overlaps = correlation[n-1:]  # This is the max shift values
    # The actual overlap integral is scaled by dx
    C5 = np.max(overlaps) * dx
    return C5

def compute_gradient(h, k_max):
    n = len(h)
    # Create shifted version of (1-h) with shift k_max
    shifted_h = np.roll(1 - h, k_max)
    # Gradient for minimizing overlap at shift k_max
    # derivative of h[i] * (1 - h[(i-k_max)%n]) wrt h[i]
    # We need: d/dh[i] [h[i] * (1 - h[(i-k_max)%n])] = (1 - h[(i-k_max)%n]) - h[i] * (0)
    grad = shifted_h - h * shifted_h
    return grad

def project_to_feasible(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def is_feasible(h, n_points):
    return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - n_points / 2) < 1e-10

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create starting point
    n_points = 47
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values, dtype=float)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
    
    best_h = h0.copy()
    best_C5 = evaluate_erdos_solution(h0)
    best_time = start_time
    
    # Multi-start initialization
    initial_points = [h0]
    for _ in range(4):
        perturbed = h0 + np.random.normal(0, 0.005, n_points)
        initial_points.append(project_to_feasible(perturbed, n_points))
    
    # For each start point
    best_local_h = h0.copy()
    best_local_C5 = best_C5
    current_h = h0.copy()
    
    iteration_count = 0
    max_iter = 10000  # max allowed iterations
    line_search_steps = 30
    refinement_every = 50
    
    while time.time() - start_time < budget_s * 0.98 and iteration_count < max_iter:
        # compute overlap values
        dx = 2.0 / n_points
        correlation = np.correlate(current_h, 1 - current_h, mode="full")
        overlaps = correlation[n_points-1:]  # Shifts 0,1,...,n-1
        k_max = np.argmax(overlaps)
        current_C5 = overlaps[k_max] * dx
        
        # update best if needed
        if current_C5 < best_local_C5:
            best_local_C5 = current_C5
            best_local_h = current_h.copy()
        
        # Compute gradient
        gradient = compute_gradient(current_h, k_max)
        if np.linalg.norm(gradient) == 0:
            # If no progress, try random step
            step = np.random.normal(0, 0.01, n_points)
        else:
            step = gradient / np.linalg.norm(gradient)
        
        # Backtracking line search
        alpha = 0.01
        rho = 0.5
        step_size = alpha
        for _ in range(line_search_steps):
            new_h = current_h - step_size * step
            new_h = project_to_feasible(new_h, n_points)
            new_C5 = evaluate_erdos_solution(new_h)
            
            if new_C5 <= current_C5:
                current_h = new_h
                break
            else:
                step_size *= rho
        
        # Convex refinement every few steps
        if iteration_count % refinement_every == 0:
            # We want to minimize the overlap at worst shift k_max
            # Set up convex problem (approximately)
            h_var = cp.Variable(n_points)
            # Objective: minimize sum over i of h[i] * (1 - h[(i-k_max)%n])
            objective = cp.Minimize(cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, k_max))))
            
            # Constraints
            constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
            
            # Solve
            try:
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.GLPK, verbose=False)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    concave_h = h_var.value
                    if concave_h is not None:
                        current_h = project_to_feasible(concave_h, n_points)
            except Exception:
                pass  # Fall back to gradient steps
        
        iteration_count += 1
    
    # Final check of the best solution found
    final_C5 = evaluate_erdos_solution(best_local_h)
    
    return (best_local_h.tolist(), final_C5, n_points)
