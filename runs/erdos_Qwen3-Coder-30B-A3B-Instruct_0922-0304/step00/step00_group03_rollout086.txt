import numpy as np
import cvxpy as cp
from math import floor, ceil
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Define initial guess
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h0 = np.array(initial_h_values)
        if np.sum(h0) != target_sum:
            # Repair the sum
            diff = target_sum - np.sum(h0)
            if diff > 0:
                # Add to zeros
                zero_indices = np.where(h0 < 1e-6)[0]
                if len(zero_indices) > 0:
                    indices = np.random.choice(zero_indices, size=min(int(diff/dx), len(zero_indices)), replace=False)
                    for i in indices:
                        h0[i] = min(1.0, h0[i] + dx)
            else:
                # Remove from ones
                ones_indices = np.where(h0 > 1 - 1e-6)[0]
                if len(ones_indices) > 0:
                    indices = np.random.choice(ones_indices, size=min(int(-diff/dx), len(ones_indices)), replace=False)
                    for i in indices:
                        h0[i] = max(0.0, h0[i] - dx)
        h0 = np.clip(h0, 0, 1)
    else:
        # Create a simple binary step function
        h0 = np.zeros(n_points)
        mid = n_points // 2
        h0[:mid] = 1
        # Adjust to make sum equal to target_sum
        h0 = np.clip(h0 * target_sum / np.sum(h0), 0, 1)

    # Search bounds
    low_bound = 0.0
    high_bound = 0.5
    tolerance = 1e-5
    best_h = h0.copy()
    best_C5 = evaluate_solution(h0, dx)

    def evaluate_solution(h, dx):
        # Using the provided evaluation method
        h = np.array(h)
        correlation = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(correlation) * dx
        return max_overlap

    def solve_feasibility(h0, t, dx):
        # Variables
        h = cp.Variable(n_points)
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == target_sum)
        # Overlap constraints for all shifts
        for k in range(-n_points + 1, n_points):
            # Correlation sum for shift k
            # Sum over i where both h[i] and h[i+k] are valid
            if k >= 0:
                valid_indices = range(n_points - k)
                corr_sum = cp.sum(
                    cp.multiply(h[valid_indices],
                                1 - h[valid_indices + k])
                )
            else:
                valid_indices = range(-k, n_points)
                corr_sum = cp.sum(
                    cp.multiply(h[valid_indices],
                                1 - h[valid_indices - k])
                )
            # The overlap for shift k should be <= t * dx
            # So corr_sum * dx <= t * dx => corr_sum <= t
            constraints.append(corr_sum <= t)  
        # Objective: minimize t (just checking feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, max_iters=10000)
        except Exception as e:
            return False, None

        if prob.status in ["optimal", "optimal_inaccurate"]:
            return True, h.value
        else:
            return False, None

    # Binary search for minimal C5
    while high_bound - low_bound > tolerance:
        if time.time() - start_time > budget_s - 5:
            break
        t = (low_bound + high_bound) / 2.0
        feasible, h_solved = solve_feasibility(h0, t, dx)
        if feasible:
            high_bound = t
            if h_solved is not None:
                best_h = h_solved
        else:
            low_bound = t

    # Final evaluation of best solution
    final_h = np.clip(best_h, 0, 1)
    # Repair sum
    if abs(np.sum(final_h) - target_sum) > 1e-3:
        diff = target_sum - np.sum(final_h)
        if diff > 0:
            zero_indices = np.where(final_h < 1e-6)[0]
            if len(zero_indices) > 0:
                indices = np.random.choice(zero_indices, size=min(int(diff/dx), len(zero_indices)), replace=False)
                for i in indices:
                    final_h[i] = min(1.0, final_h[i] + dx)
        else:
            ones_indices = np.where(final_h > 1 - 1e-6)[0]
            if len(ones_indices) > 0:
                indices = np.random.choice(ones_indices, size=min(int(-diff/dx), len(ones_indices)), replace=False)
                for i in indices:
                    final_h[i] = max(0.0, final_h[i] - dx)
        final_h = np.clip(final_h, 0, 1)
    final_C5 = evaluate_solution(final_h, dx)

    return (final_h.tolist(), final_C5, n_points)
