import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_c5 = 0.3807
    max_iter = 50000
    alpha = 0.1
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute all overlaps
        h_padded = np.pad(h, (n_points-1, n_points-1), mode='constant', constant_values=0)
        h_rev = np.flip(h)
        overlaps = correlate(h_padded, h_rev, mode='full') * dx
        
        # Find k that maximizes overlap
        k_max = np.argmax(overlaps[n_points-1:]) - (n_points-1)
        max_overlap = overlaps[n_points-1+k_max]
        
        # Check if we've achieved target
        if max_overlap <= target_c5:
            break
            
        # Compute gradient for this k
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
                grad[j] -= h[i] * dx
                
        # Update h
        h = h - alpha * grad
        
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Update best solution
        current_c5 = max_overlap
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    # Final evaluation
    h_final = best_h
    h_padded = np.pad(h_final, (n_points-1, n_points-1), mode='constant', constant_values=0)
    h_rev = np.flip(h_final)
    overlaps = correlate(h_padded, h_rev, mode='full') * dx
    final_c5 = np.max(overlaps[n_points-1:])
    
    return (h_final.tolist(), final_c5, n_points)
