def evaluate_erdos_solution(h, dx):
    """Evaluate the overlap integral for the given step function."""
    # Compute the full correlation (equivalent to the max overlap)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the correlation
    return np.max(corr) * dx

def project_to_feasible(h, n_points, dx):
    """Project h to feasible region and ensure sum = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to sum = n_points / 2
    current_sum = np.sum(h) * dx
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
