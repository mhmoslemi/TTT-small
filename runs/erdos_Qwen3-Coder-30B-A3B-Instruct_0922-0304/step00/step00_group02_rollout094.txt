import numpy as np
import time
from scipy import correlate

def evaluate_c5(h):
    """Evaluate C5 for given step function h."""
    dx = 2.0 / len(h)
    # Compute correlations for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Maximum overlap
    c5 = np.max(corr) * dx
    return c5

def normalize_h(h):
    """Ensure h sums to n/2 and stays within [0,1]."""
    n = len(h)
    target_sum = n / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Scale to meet sum constraint
    scaled = h * target_sum / current_sum
    # Clip to [0,1]
    clipped = np.clip(scaled, 0, 1)
    # Adjust to exact sum
    actual_sum = np.sum(clipped)
    diff = target_sum - actual_sum
    if abs(diff) > 1e-10:
        # Find index to add diff
        idx = np.argmax(clipped)
        clipped[idx] = min(1.0, clipped[idx] + diff)
    return clipped

def mutate_h(h, mutation_rate=0.01):
    """Mutate h vector with given rate."""
    n = len(h)
    mutated = h.copy()
    num_mutations = int(n * mutation_rate)
    indices = np.random.choice(n, num_mutations, replace=False)
    for i in indices:
        # Add small random perturbation
        mutated[i] += np.random.normal(0, 0.05)
    # Ensure [0,1] bounds
    mutated = np.clip(mutated, 0, 1)
    # Normalize to sum constraint
    mutated = normalize_h(mutated)
    return mutated

def crossover_h(h1, h2):
    """Crossover two h vectors."""
    n = len(h1)
    # Select random crossover point
    cp = np.random.randint(1, n)
    child1 = np.concatenate([h1[:cp], h2[cp:]])
    child2 = np.concatenate([h2[:cp], h1[cp:]])
    # Normalize children
    child1 = normalize_h(child1)
    child2 = normalize_h(child2)
    return child1, child2

def selection(pop, fitnesses):
    """Tournament selection."""
    n = len(pop)
    selected = []
    for _ in range(n // 2):
        # Tournament size = 3
        indices = np.random.choice(n, 3, replace=False)
        tournament_fitness = [fitnesses[i] for i in indices]
        winner = indices[np.argmax(tournament_fitness)]
        selected.append(pop[winner])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    population_size = 100
    mutation_rate = 0.01
    generations = 1000
    
    # Initialize population
    pop = []
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        pop.append(h_start)
    else:
        h_start = np.full(n_points, 0.5)
        pop.append(h_start)
    
    # Add random samples
    for _ in range(population_size - len(pop)):
        h = np.random.rand(n_points)
        h = normalize_h(h)
        pop.append(h)
    
    best_h = None
    best_c5 = float('inf')
    last_improvement = 0
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in pop:
            c5 = evaluate_c5(h)
            fitnesses.append(1.0 / (c5 + 1e-15))
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for convergence
        if gen > 0 and gen - last_improvement > 100:
            break
            
        # Select
        selected = selection(pop, fitnesses)
        
        # Generate new population
        new_pop = selected.copy()
        
        # Crossover and mutate
        while len(new_pop) < population_size:
            # Select parents
            idx1, idx2 = np.random.choice(len(selected), 2, replace=False)
            par1, par2 = selected[idx1], selected[idx2]
            
            # Crossover
            child1, child2 = crossover_h(par1, par2)
            
            # Mutate
            child1 = mutate_h(child1, mutation_rate)
            child2 = mutate_h(child2, mutation_rate)
            
            new_pop.extend([child1, child2])
        
        # Keep population size
        pop = new_pop[:population_size]
        
        # Check for improvement
        if gen > 0 and gen % 10 == 0:
            if abs(best_c5 - best_c5) < 1e-5 * 0.38085:
                last_improvement = gen

    if best_h is None:
        # Fallback
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
        else:
            best_h = np.full(n_points, 0.5)
        best_c5 = evaluate_c5(best_h)
        
    # Final check
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
