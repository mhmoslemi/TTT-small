import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Fallback: binary construction with 1s and 0s
        n_points = 45
        h = np.ones(n_points)
        h[n_points//2:] = 0
    
    # Ensure correct sum
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Parameters
    max_iter = 10000
    alpha = 0.1
    alpha_min = 1e-6
    decay = 0.95
    perturb_freq = 500
    perturb_fraction = 0.05
    target_c5 = 0.3807
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Precompute constants
    dx = 2.0 / n_points
    half_size = n_points // 2
    
    # Precompute indices for convolution
    indices = np.arange(n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        overlap_full = correlate(h, 1-h, mode='full')
        overlaps = overlap_full * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Update best if improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        if best_c5 <= target_c5:
            break

        # Compute gradient for active shift
        # For shift k_max, we have O_k = sum h[i] * (1 - h[(i+k_max) % n_points])
        # So dO_k/dh[i] = 1 - h[(i+k_max) % n_points] - h[(i-k_max) % n_points]?
        # Actually let's use finite differences:
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_pert = h.copy()
            h_pert[j] += eps
            overlap_plus = correlate(h_pert, 1-h_pert, mode='full')[k_max] * dx
            
            h_pert[j] = h[j] - eps
            overlap_minus = correlate(h_pert, 1-h_pert, mode='full')[k_max] * dx
            
            grad[j] = (overlap_plus - overlap_minus) / (2 * eps)
            
        # Update step
        h_new = h - alpha * grad
        
        # Project
        h_new = np.clip(h_new, 0, 1)
        scale = (n_points / 2.0) / np.sum(h_new)
        h_new *= scale
        
        # Evaluate and accept
        overlap_new = correlate(h_new, 1-h_new, mode='full')[k_max] * dx
        if overlap_new <= max_overlap:
            h = h_new.copy()
        else:
            alpha *= decay
            if alpha < alpha_min:
                alpha = alpha_min
                
        # Perturb occasionally
        if i % perturb_freq == 0 and i > 0:
            idxs = np.random.choice(n_points, int(n_points * perturb_fraction), replace=False)
            h[idxs] += np.random.normal(0, 0.02, len(idxs))
            h = np.clip(h, 0, 1)
            scale = (n_points / 2.0) / np.sum(h)
            h *= scale
    
    # Final check
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, len(best_h))
