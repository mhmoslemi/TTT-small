import numpy as np
import scipy.optimize as opt
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from available construction
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5)
    
    # Normalize to correct sum
    h_current = h_current * target_sum / np.sum(h_current)
    best_h = h_current.copy()
    best_c5 = float('inf')
    
    # Helper function to compute C5
    def compute_c5(h):
        # Compute all overlaps
        corr = np.correlate(h, 1-h, mode="full")
        overlaps = corr[len(h)-1:]
        return np.max(overlaps) * dx
    
    # Helper function to compute gradient for a specific shift
    def compute_gradient(h, shift_idx):
        # Shift 1-h by shift_idx
        shifted = np.roll(1 - h, shift_idx)
        # Compute gradient
        gradient = shifted - h * shifted
        return gradient
    
    def project_h(h):
        # Clip
        h = np.clip(h, 0, 1)
        # Rescale
        h = h * target_sum / np.sum(h)
        return h
    
    # Multi-initialization
    initial_solutions = [h_current]
    for _ in range(4):
        perturbation = np.random.normal(0, 0.01, n_points)
        h_pert = h_current + perturbation
        h_pert = project_h(h_pert)
        initial_solutions.append(h_pert)
    
    best_c5 = float('inf')
    best_h = None
    
    # Iterate over all initializations
    for h_initial in initial_solutions:
        h = h_initial.copy()
        h_old = h.copy()
        
        # Early stopping criteria
        iter_since_improvement = 0
        last_improvement = 0
        
        for iteration in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlaps
            corr = np.correlate(h, 1-h, mode="full")
            overlaps = corr[len(h)-1:]
            k_max = np.argmax(overlaps)
            current_c5 = overlaps[k_max] * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                iter_since_improvement = 0
                last_improvement = iteration
            else:
                iter_since_improvement += 1
                if iter_since_improvement > 20 and iteration > last_improvement + 50:
                    break
            
            # Compute gradient for worst shift
            grad = compute_gradient(h, k_max)
            
            # Normalize gradient
            norm_grad = np.linalg.norm(grad)
            if norm_grad < 1e-10:
                break
                
            grad = grad / norm_grad
            
            # Backtracking line search
            alpha = 0.01
            rho = 0.5
            max_backtrack = 30
            
            for _ in range(max_backtrack):
                h_new = h - alpha * grad
                h_new = project_h(h_new)
                new_c5 = compute_c5(h_new)
                
                if new_c5 <= current_c5:
                    h = h_new
                    break
                else:
                    alpha *= rho
            
            # Occasionally do convex refinement
            if iteration % 50 == 0:
                # Formulate convex optimization for the worst shift
                k = k_max
                h_cvx = Variable(n_points)
                objective = Minimize(sum(h_cvx[i] * (1 - h_cvx[(i - k) % n_points]) for i in range(n_points)))
                constraints = [0 <= h_cvx[i] <= 1 for i in range(n_points)]
                constraints.append(sum(h_cvx) == target_sum)
                prob = Problem(objective, constraints)
                
                try:
                    prob.solve(solver=ECOS, verbose=False)
                    if prob.status == 'optimal':
                        h_new = np.array([h_cvx[i].value for i in range(n_points)])
                        h = project_h(h_new)
                except:
                    pass
    
    # Final validation
    if best_h is not None:
        c5_final = compute_c5(best_h)
        best_c5 = min(c5_final, best_c5)
        return (list(best_h), best_c5, n_points)
    else:
        return (list(h_current), compute_c5(h_current), n_points)
