import numpy as np
from scipy import fft, optimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, else create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    max_iter = 10000
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    # Precompute FFT size for efficiency
    fft_size = 2 * n_points - 1
    half_fft = fft_size // 2 + 1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # FFT computation
        H = fft.fft(h, fft_size)
        H_conj = np.conj(H)
        
        # Compute correlation in frequency domain
        corr_spectrum = H * H_conj
        
        # Compute the maximum value in correlation (C5)
        corr_full = fft.ifft(corr_spectrum, fft_size).real
        c5 = np.max(corr_full[:n_points]) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient ascent in frequency domain to reduce max correlation
        # Gradient direction: d/dh (max correlation)
        # Approximate using gradient of power spectrum with respect to h
        
        # Update H based on flattening the magnitude squared spectrum
        # This aims to reduce the maximum correlation
        H_magnitude = np.abs(H)
        H_magnitude = np.maximum(H_magnitude, 1e-8)
        weight = 1.0 / (H_magnitude**2 + 1e-10)
        
        # Simple gradient ascent step
        H_new = H * np.exp(-0.01 * np.real(H_conj * fft.fft(weight * H, fft_size)))
        H_new /= np.abs(H_new[0])  # Normalize DC
        h = np.real(fft.ifft(H_new, fft_size)[:n_points])
        
        # Project h to feasible region
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        
        # Ensure exact sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Check converge
        if iteration > 100 and abs(c5 - best_c5) < 1e-8:
            break
    
    # Final evaluation
    H = fft.fft(best_h, fft_size)
    corr_spectrum = H * np.conj(H)
    corr_full = fft.ifft(corr_spectrum, fft_size).real
    c5_final = np.max(corr_full[:n_points]) * dx
    
    return (list(best_h), c5_final, n_points)
