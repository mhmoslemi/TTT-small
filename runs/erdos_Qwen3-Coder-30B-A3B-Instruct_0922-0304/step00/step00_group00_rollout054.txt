import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 50000
    alpha = 0.1
    last_improvement_iter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = correlate(h, 1 - h, mode='full') * dx
        
        # Find the shift k with maximum overlap
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement_iter = i
            
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient with respect to k_max
        grad = np.zeros_like(h)
        for j in range(n_points):
            k_idx = k_max - n_points + 1 + j
            if 0 <= k_idx < n_points:
                grad[j] += (1 - h[k_idx]) * dx
                grad[k_idx] -= h[j] * dx
                
        # Update h using gradient descent
        h = h - alpha * grad
        
        # Ensure feasibility
        h = np.clip(h, 0, 1)
        
        # Renormalize to satisfy constraint sum(h) * dx = 1
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Small learning rate adjustment
        if i % 100 == 0 and i > 0:
            alpha *= 0.99
            
    # Final validation
    overlaps_final = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(overlaps_final)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
