import numpy as np
import time
from scipy import signal

def evaluate_h(h, n_points):
    dx = 2.0 / n_points
    # Compute all overlaps
    overlaps = signal.correlate(h, 1 - h, mode='full')
    # The last n_points-1 elements contain the relevant overlaps
    max_overlap = np.max(overlaps[n_points-1:])
    return max_overlap * dx

def init_population(n_points, pop_size, initial_h):
    population = []
    population.append(initial_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        # Apply constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def mutate(h, mutation_rate):
    n_points = len(h)
    num_mutations = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        h[i] = np.random.rand()
    # Adjust to maintain sum = n_points / 2
    current_sum = np.sum(h)
    diff = (n_points / 2) - current_sum
    # Add adjustment uniformly to all elements
    h += diff / n_points
    h = np.clip(h, 0, 1)
    return h

def crossover(h1, h2):
    n_points = len(h1)
    # Simple single-point crossover
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    # Adjust to maintain sum = n_points / 2
    for child in [child1, child2]:
        current_sum = np.sum(child)
        diff = (n_points / 2) - current_sum
        # Add adjustment uniformly to all elements
        child += diff / n_points
        child = np.clip(child, 0, 1)
    return child1, child2

        selected = [population[i] for i in sorted_indices[:pop_size//2]]
