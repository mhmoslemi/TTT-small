def evaluate_erdos_solution(h):
    """Evaluate the overlap integral C5 for a given step function h."""
    n = len(h)

    # Compute correlations for all lags
    correlations = np.correlate(h, 1 - h, mode="full")
    # Return the maximum
    return np.max(correlations) * dx
