import numpy as np
import cvxpy as cp
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points // 2)
        h = np.concatenate([h, np.zeros(n_points - n_points // 2)])
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    start_time = time.time()
    target = 0.3807
    max_iter = 5000
    time_per_iter = budget_s / max_iter
    
    lambda_k = np.zeros(n_points * 2 - 1)
    mu = 0.0
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Solve convex subproblem
        h_new = solve_subproblem(h, lambda_k, mu, rho, dx, n_points)
        
        # Evaluate overlaps
        overlaps = np.correlate(h_new, 1 - h_new, mode='full') * dx
        
        # Update dual variables
        c5 = np.max(overlaps)
        lambda_k += alpha * np.maximum(0, c5 - overlaps)
        mu += alpha * (np.sum(h_new) * dx - 1)
        
        # Apply entropy regularization to lambda
        lambda_k = lambda_k + beta * np.log(1 + lambda_k)
        
        # Check progress
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            
        if c5 < target:
            break
            
        # Update h
        h = h_new.copy()
        
        # Decay entropy coefficient
        if it % 100 == 0:
            beta *= 0.9
    
    return (best_h.tolist(), best_c5, n_points)

def solve_subproblem(h, lambda_k, mu, rho, dx, n_points):
    h_var = cp.Variable(n_points)
    h_old = h.copy()
    
    # Objective with regularization
    # min 0.5 * ||h - h_old||^2 + sum lambda_k * overlap_k + mu * (sum(h) - 1)
    overlap_term = 0
    for k in range(2 * n_points - 1):
        # Get indices
        offset = k - n_points + 1
        if offset >= 0:
            # shift right
            h_shift = h_var[offset:]
            h_old_shift = h_old[:n_points-offset]
        else:
            # shift left
            h_shift = h_var[:n_points+offset]
            h_old_shift = h_old[-offset:]
        # Only consider terms when both vectors have elements
        if len(h_shift) > 0 and len(h_old_shift) > 0:
            overlap_term += lambda_k[k] * cp.sum(h_shift * (1 - h_old_shift))
    
    # Add mu term
    mu_term = mu * cp.sum(h_var)
    
    # Regularization and objective
    objective = cp.Minimize(0.5 * cp.sum_squares(h_var - h_old) + overlap_term + mu_term + 0.5 * rho * cp.sum_squares(h_var))
    
    # Constraints
    constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) * dx == 1]
    
    # Solve
    prob = cp.Problem(objective, constraints)
    try:
        prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=50)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            return h_var.value
    except:
        pass
    
    # Fallback to old h if solver fails
    return h_old

def evaluate_c5(h, dx):
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)
