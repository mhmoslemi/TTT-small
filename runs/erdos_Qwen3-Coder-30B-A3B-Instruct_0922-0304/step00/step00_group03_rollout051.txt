import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic construction: piecewise constant with alternating values
        n_points = 70
        h = np.ones(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
        h = h / np.sum(h) * n_points / 2  # normalize sum to n_points / 2
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # Compute all correlations
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Shift indices to be from 0 to 2*dx*(n_points-1)
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_to_feasible(h_vec):
        # Project to [0,1] and adjust sum to be n_points/2
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        target_sum = n_points / 2
        h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Gradient-based optimization with projection
    max_iter = 50000
    lr = 0.01
    tol = 1e-6
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Find the most constraining shift
        corr = np.correlate(h, 1 - h, mode="full")
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient of the overlap with respect to h
        grad = np.zeros(n_points)
        # Derivative of sum h[i]*(1-h[i+k]) 
        # For the active constraint, the gradient is 1 - h[i+k] where k = max_k - n_points + 1
        # This is valid since we're in a full correlation setup
        shift_offset = max_k - (n_points - 1)
        for i in range(n_points):
            j = i - shift_offset
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) * dx
            # Also account for the term -h[i+k] * h[i] -> -h[i] * h[i+k] derivative gives -h[i]
            # But we are doing correlative, so the term that actually matters is just the one above:
            # Since h[i] is treated as scalar multiplying 1-h[i+k], derivative w.r.t h[i] is (1-h[i+k]) 
            # We corrected above
        
        # Update h
        h = h - lr * grad
        h = project_to_feasible(h)
        
        # Early stopping
        if iteration > 100 and abs(current_C5 - best_C5) < tol:
            break
    
    current_C5 = compute_C5(best_h)
    if current_C5 < best_C5:
        best_C5 = current_C5
        best_h = best_h.copy()
    
    # Final project
    best_h = project_to_feasible(best_h)
    final_C5 = compute_C5(best_h)
    
    return (list(best_h), float(final_C5), int(n_points))
