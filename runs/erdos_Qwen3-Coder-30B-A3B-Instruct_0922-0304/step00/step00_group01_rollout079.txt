import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    overlap = correlate(h_values, 1 - h_values, mode='full')
    max_overlap = np.max(overlap[n_points-1:]) * dx
    return max_overlap

def project_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 47
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    h = project_h(h, n_points)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    start_time = time.time()

    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100

    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        dx = 2.0 / n_points
        overlap = correlate(h, 1 - h, mode='full')[n_points-1:]
        peaks = np.argsort(overlap)[::-1][:5]
        update = np.zeros(n_points)

        for k_idx in peaks:
            k = k_idx
            for i in range(n_points - k):
                if h[i] > 0 and h[i + k] > 0:
                    update[i] -= 0.05
                elif h[i] < 1 and h[i + k] < 1:
                    update[i] += 0.05
                    
        h_new = h + update
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_erdos_solution(h_new, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        h = h_new.copy()
        
        if no_improvement_count >= max_no_improvement or best_c5 < 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
