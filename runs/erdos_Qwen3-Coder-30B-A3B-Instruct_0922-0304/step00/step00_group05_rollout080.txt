import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 70
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Default construction
        h = np.ones(n_points//2) * 0.5
        h = np.append(h, np.zeros(n_points//2))
    
    # Ensure sum condition
    h_sum = np.sum(h)
    dx = 2.0 / n_points
    scale_factor = (n_points / 2) / h_sum
    h = h * scale_factor
    
    # Initial evaluation
    def evaluate(h_vec):
        # Compute C5 using full correlation
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        c5_val = np.max(corr) * dx
        return c5_val
    
    current_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    n_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / n_iter)
    
    # Perturbation scale
    delta = 0.1
    accept_count = 0
    total_count = 0
    
    start_time = time.time()
    
    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create neighbor
        h_new = h.copy()
        idx1 = np.random.randint(0, n_points)
        idx2 = np.random.randint(0, n_points)
        
        # Try swap or shift
        if np.random.random() < 0.5:
            # Swap approach: move some mass between two points
            # Ensure no overflow/underflow
            diff = np.random.random() * delta
            if h_new[idx1] >= diff:
                h_new[idx1] -= diff
                h_new[idx2] = min(1.0, h_new[idx2] + diff)
            else:
                h_new[idx2] = min(1.0, h_new[idx2] + h_new[idx1])
                h_new[idx1] = 0.0
        else:
            # Adaptive shift: redistribute mass to neighbors
            shift_dir = np.random.choice([-1, 1])
            idx_shift = (idx1 + shift_dir) % n_points
            diff = np.random.random() * delta
            if h_new[idx1] >= diff:
                h_new[idx1] -= diff
                h_new[idx_shift] = min(1.0, h_new[idx_shift] + diff)
        
        # Rescale to meet sum constraint
        h_sum_new = np.sum(h_new)
        if abs(h_sum_new) > 1e-10:
            h_new = h_new * (n_points / 2) / h_sum_new
        
        # Clip boundary values
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        new_c5 = evaluate(h_new)
        delta_c5 = new_c5 - current_c5
        
        # Accept or reject
        prob = np.exp(-delta_c5 / T_max) if delta_c5 > 0 else 1.0
        if np.random.random() < prob:
            h = h_new
            current_c5 = new_c5
            accept_count += 1
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        total_count += 1
        
        # Update temperature
        T_max *= alpha
        
        # Adjust delta based on acceptance rate
        if total_count > 0 and total_count % 100 == 0:
            accept_rate = accept_count / total_count
            if accept_rate < 0.2:
                delta *= 0.9
            elif accept_rate > 0.5:
                delta *= 1.1
            accept_count = 0
            total_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
