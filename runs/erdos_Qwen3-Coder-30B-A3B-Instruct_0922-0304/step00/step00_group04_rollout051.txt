import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):

    best_c5 = evaluate_erdos_solution(h)

    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # The lag indices correspond to [-n_points+1, ..., n_points-1]
        lags = np.arange(-n_points + 1, n_points)
        # We only want the lags in [0, 2] range (n_points-long segment)
        valid_lags = np.where((lags >= 0) & (lags <= 2 * n_points - 2))[0]
        # Force fewer samples for speed when needed
        if len(valid_lags) >= 200:
            valid_lags = np.arange(0, min(200, len(valid_lags)), 1)
            
        c5_vals = corr[valid_lags] * dx
        max_idx = np.argmax(c5_vals)
        max_lag = lags[valid_lags[max_idx]]
        
        # Compute gradient and update
        grad = np.zeros(n_points)
        if max_lag <= n_points - 1:
            # h[i] contributes to the term h[i](1 - h[i + max_lag]) 
            for i in range(n_points - max_lag):
                grad[i] += (1 - h[i + max_lag]) 
                if i + max_lag < n_points:
                    grad[i + max_lag] -= h[i]
            grad *= dx
        else:
            # Lag is too big, skip
            continue
            
        # Apply a small learning rate
        lr = 0.001
        h_new = h - lr * grad
        
        # Project back to feasible region
        # 1. Normalize sum to n_points / 2
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # 2. Clip values to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            h = h_new
        else:
            # Try another local update with a random perturbation
            # Add some noise
            noise = np.random.normal(0, 0.01, n_points)
            h_new = h + noise
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            h_new = np.clip(h_new, 0, 1)
            new_c5 = evaluate_erdos_solution(h_new)
            
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                h = h_new
                
        # Stop early if no useful progress
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), best_c5, n_points)
