import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def eval_c5(h_vals):
        # Compute cross-correlation with 1-h
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Extract the overlapping parts (valid mode)
        valid_corr = corr[len(h_vals)-1:]
        return np.max(valid_corr) * dx
    
    def optimize_convex(h_in, n_points_in):
        """Convex relaxation approach."""
        h = cp.Variable(n_points_in)
        # Constraints
        constraints = [h >= 0, h <= 1, cp.sum(h) == n_points_in / 2]
        # Objective: convex relaxation of max correlation
        # We'll maximize a single correlation (i.e., lag 0 for simplicity)
        # This is approximate but fast and constructs a feasible point
        obj = cp.Maximize(cp.sum(cp.multiply(h, 1 - h)))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_sol = h.value
            if h_sol is not None:
                return h_sol
        except:
            pass
        return h_in
    
    def project_h(h_in):
        """Project h into [0,1] and ensure sum constraint."""
        h_proj = np.clip(h_in, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    def gradient_descent_step(h_in, lr=0.1):
        """Single gradient descent step."""
        c5 = eval_c5(h_in)
        # Gradient approximation
        delta = 1e-3
        grad = np.zeros_like(h_in)
        for i in range(len(h_in)):
            h_plus = h_in.copy()
            h_plus[i] += delta
            h_plus = project_h(h_plus)
            c5_plus = eval_c5(h_plus)
            grad[i] = (c5_plus - c5) / delta
        # Update step
        h_new = h_in - lr * grad
        h_new = project_h(h_new)
        return h_new
    
    # Stage 1: Convex relaxation refined
    h = optimize_convex(h, n_points)
    h = project_h(h)
    
    # Stage 2: Gradient descent
    # Number of iterations and learning rate schedule
    max_iter = 100
    lr = 0.1
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        h_old = h.copy()
        h = gradient_descent_step(h, lr)
        # Adaptive learning rate
        new_c5 = eval_c5(h)
        old_c5 = eval_c5(h_old)
        if new_c5 >= old_c5:
            lr *= 0.5
        else:
            lr = min(lr * 1.1, 0.5)
    
    # Final evaluation and constraint check
    h = project_h(h)
    c5_result = eval_c5(h)
    
    # Stage 3: High resolution refinement if time allows
    if time.time() - start_time < budget_s - 5:
        n_points_high = 84
        dx_high = 2.0 / n_points_high
        h_up = np.interp(np.linspace(0, 2, n_points_high), 
                         np.linspace(0, 2, n_points), 
                         h)
        h_up = project_h(h_up)
        # Re-run gradient descent at higher resolution
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            h_up_old = h_up.copy()
            h_up = gradient_descent_step(h_up, 0.1)
        h = h_up
        c5_result = eval_c5(h) * dx_high / dx  # Scale back
    
    return (h.tolist(), c5_result, len(h))
