import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    max_backtrack = 30
    patience = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        h_rev = np.flip(1 - h)
        overlaps = correlate(h, h_rev, mode='full')
        n = len(h)
        overlaps = overlaps[n-1:] * dx
        
        k_max = np.argmax(overlaps)
        C5_current = overlaps[k_max]
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if best_C5 <= target_bound:
            break
            
        if no_improvement_count > patience:
            break
            
        # Compute gradient for k_max
        gradient = np.zeros(n)
        for i in range(n):
            if i + k_max < n:
                gradient[i] += (1 - h[i + k_max])
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
                
        # Line search
        alpha = alpha_initial
        grad_norm = np.linalg.norm(gradient)
        
        if grad_norm > 1e-12:
            direction = gradient / grad_norm
        else:
            iter_count += 1
            continue
            
        backtrack_count = 0
        improved = False
        
        while backtrack_count < max_backtrack:
            h_new = h - alpha * direction
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to satisfy sum constraint
            current_sum = np.sum(h_new)
            if abs(current_sum) > 1e-12:
                scale = (n_points / 2) / current_sum
                h_new = h_new * scale
                
            # Compute new C5
            h_new_rev = np.flip(1 - h_new)
            new_overlaps = correlate(h_new, h_new_rev, mode='full')
            new_C5 = new_overlaps[n-1:] * dx
            new_C5_max = np.max(new_C5)
            
            if new_C5_max < C5_current:
                h = h_new
                improved = True
                break
            else:
                alpha *= rho
                backtrack_count += 1
                
        if not improved:
            break
            
        iter_count += 1
            
    if best_C5 < float('inf'):
        return (best_h.tolist(), best_C5, n_points)
