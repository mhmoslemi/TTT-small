def evaluate_erdos_solution(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx over all shifts k."""
    h = np.array(h)
    n = len(h)
    max_overlap = 0
    for k in range(n):
        corr = 0
        for i in range(n):
            j = (i + k) % n
            corr += h[i] * (1 - h[j])
        if corr > max_overlap:
            max_overlap = corr
    return max_overlap * dx
