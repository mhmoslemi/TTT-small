import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdős constant C_5 for given h."""
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all indices where overlap equals max_overlap
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient in one pass
        gradient = np.zeros(n_points)
        for lag_idx in max_lags:
            if lag_idx < n_points:
                # Forward difference for lag
                valid = slice(0, n_points - lag_idx)
                if valid.start < valid.stop:
                    gradient[valid] += (1 - h[valid.start + lag_idx:valid.stop]) * dx
            else:
                # Backward difference for lag > n_points - 1
                valid = slice(lag_idx - n_points + 1, n_points)
                if valid.start < valid.stop:
                    gradient[valid] += (1 - h[valid.start:valid.stop]) * dx
                
        # Normalize by number of contributing lags
        if len(max_lags) > 0:
            gradient /= len(max_lags)
            
        # Update h using gradient descent
        h -= learning_rate * gradient
        
        # Project h to feasible set [0, 1] and rescale to integral=1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h *= (n_points / 2) / h_sum
            
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adapt learning rate based on performance
        if patience_counter >= gamma_patience:
            learning_rate *= 0.1
            patience_counter = 0
            
    return best_h, best_c5, n_points
