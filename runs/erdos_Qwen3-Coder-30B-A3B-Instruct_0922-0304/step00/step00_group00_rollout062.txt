import numpy as np
from scipy import correlate
import cvxpy as cp
import time

def evaluate_c5(h):
    """Evaluate the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps with the shifted version
    overlap = np.correlate(h, 1-h, mode='full')
    # Only consider the valid overlaps in [0,2]
    max_overlap = np.max(overlap[n-1:])
    return max_overlap * dx

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    
    # Normalize to have integral = 1
    dx = 2.0 / n
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Initialize constraints to track worst shifts
    max_k = []
    max_overlaps = []
    
    iteration = 0
    last_improvement = 0
    tolerance = 1e-5
    
    while time.time() - start_time < budget_s * 0.95 and (iteration - last_improvement) < 50:
        if time.time() - start_time > budget_s * 0.95:
            break
            
        iteration += 1
        
        # Compute all overlaps
        overlap = np.correlate(h, 1-h, mode='full')
        # Only consider relevant shifts [0,2]
        max_k_idx = np.argmax(overlap[n-1:])
        current_overlap = overlap[max_k_idx] * dx
        
        if current_overlap <= best_c5 + tolerance:
            break
            
        # Add this constraint if not already in view
        if max_k_idx not in max_k:
            max_k.append(max_k_idx)
            max_overlaps.append(current_overlap)
            
        # Try to optimize by solving convex problem with constraints
        h_new = optimize_with_constraints(h, max_k, max_overlaps, n, dx)
        
        if h_new is not None:
            h = h_new.copy()
            c5_new = evaluate_c5(h)
            # Update best if improved
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                last_improvement = iteration
                
        # Check if we've found a good enough solution
        if best_c5 < 0.3807:
            break
            
        # Periodically reset max_k to avoid constraint inflation
        if iteration % 20 == 0:
            max_k = []
            max_overlaps = []
    
    # Final validation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))

def optimize_with_constraints(h, k_indices, max_overlaps, n, dx):
    h_vars = cp.Variable(n)
    C5 = cp.Variable()
    
    # Constraints: 0 <= h[i] <= 1
    constraints = [h_vars[i] >= 0 for i in range(n)]
    constraints.extend([h_vars[i] <= 1 for i in range(n)])
    # Sum to 1
    constraints.append(cp.sum(h_vars) == n/2)
    
    # Add quadratic constraints
    for k_idx in k_indices:
        # This is a quadratic constraint for a specific shift k
        # Sum over i: h[i] * (1 - h[(i+k) % n]) <= C5
        # We approximate this with the convex relaxation
        # But strictly speaking, we need to compute for the actual shift
        # For this we simulate and compute the worst-case scenario
        shift = k_idx - (n-1)
        constraint = 0
        for i in range(n):
            j = (i + shift) % n
            constraint += h_vars[i] * (1 - h_vars[j])
        constraints.append(constraint <= C5)
    
    # Objective
    objective = cp.Minimize(C5)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == 'optimal':
            h_new = h_vars.value
            # Clip and renormalize
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n / 2) / np.sum(h_new)
            return h_new
    except:
        pass
    return None
