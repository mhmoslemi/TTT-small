import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / len(h)
    # Compute correlation with (1-h) using full mode
    corr = correlate(h, 1-h, mode='full')
    # The overlap integral for each shift k
    max_overlap = np.max(corr) * dx
    return max_overlap

def get_neighbors(h, tabu_list, n_flip=2):
    """Generate neighboring solutions by flipping n_flip bits"""
    neighbors = []
    n = len(h)
    
    # Generate all combinations of n_flip indices to flip
    from itertools import combinations
    
    # Make a copy to avoid modifying original
    h_copy = h.copy()
    
    # Try all combinations of positions to flip
    for flip_positions in combinations(range(n), n_flip):
        # Make a copy for this neighbor
        neighbor = h_copy.copy()
        
        # Flip the bits
        for pos in flip_positions:
            neighbor[pos] = 1 - neighbor[pos]
            
        # Check if this neighbor is valid (sum is n/2)
        if np.sum(neighbor) == n // 2:
            # Check tabu list
            neighbor_tuple = tuple(neighbor)
            if neighbor_tuple not in tabu_list:
                neighbors.append(neighbor)
    
    return neighbors

        h[:n_points//2] = 1
        np.random.shuffle(h)
    
    # Ensure we have exactly n/2 ones
    while np.sum(h) != len(h) // 2:
        if np.sum(h) > len(h) // 2:
            # Find a 1 and flip to 0
            idx = np.where(h == 1)[0][0]
            h[idx] = 0
        else:
            # Find a 0 and flip to 1
            idx = np.where(h == 0)[0][0]
            h[idx] = 1
    
    # Current best, and tabu list
    best_h = h.copy()
    best_value = evaluate_h(h)
    
    tabu_list = set()
    tabu_tenure = 100
    
    # Starting time
    start_time = time.time()
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    max_iter = 10000  # Conservative limit
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Generate neighbors
        neighbors = get_neighbors(best_h, tabu_list, n_flip=2)
        
        if len(neighbors) == 0:
            no_improvement_count += 1
            if no_improvement_count > 100:
                break
            continue
            
        no_improvement_count = 0
        
        # Evaluate all neighbors
        neighbor_values = [evaluate_h(n) for n in neighbors]
        
        # Find the best neighbor
        best_neighbor_idx = np.argmin(neighbor_values)
        best_neighbor_value = neighbor_values[best_neighbor_idx]
        best_neighbor = neighbors[best_neighbor_idx]
        
        # Update best if improved
        if best_neighbor_value < best_value:
            best_value = best_neighbor_value
            best_h = best_neighbor.copy()
            
        # Update tabu list
        if len(tabu_list) >= tabu_tenure:
            # Remove oldest entry
            tabu_list.pop()
            
        # Add the previous best_h to tabu list
        tabu_list.add(tuple(best_h))
    
    # Final evaluation
    final_value = evaluate_h(best_h)
    
    return (best_h, final_value, len(best_h))
