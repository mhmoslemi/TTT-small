import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Returns C5 value for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute overlaps for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (shifts from 0 to n_points-1)
    overlaps = corr[n_points-1:] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h; use provided initial if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure that sum == n_points / 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    max_iter = 1000
    patience = 100
    best_C5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    no_improvement_count = 0
    
    # Line search parameters
    alpha_initial = 0.01
    rho = 0.5
    eta = 0.5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        corr = np.correlate(h, 1 - h, mode='full')
        overlaps = corr[n_points-1:] * dx
        k_max = np.argmax(overlaps)
        C5_current = overlaps[k_max]
        
        if C5_current < target_bound:
            break
            
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Compute gradient for the worst shift
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # h[i] (1 - h[i + k_max]) part
            if i + k_max < n_points:
                gradient[i] += 1 - h[i + k_max]
            else:
                gradient[i] += 0
                
            # h_clipped[i - k_max] part
            if i - k_max >= 0:
                gradient[i] -= h[i - k_max]
            else:
                gradient[i] -= 0
                
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
        gradient = gradient / grad_norm
        
        # Perform line search
        alpha = alpha_initial
        current_C5 = C5_current
        for _ in range(30):
            h_new = h - alpha * gradient
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum to n/2
            h_new_sum = np.sum(h_new)
            if h_new_sum != 0:
                h_new = h_new * (n_points / 2) / h_new_sum
            # Evaluate new C5
            new_C5 = evaluate_erdos_solution(h_new)
            if new_C5 < current_C5:
                h = h_new
                break
            alpha *= rho
            
        if alpha == 0:
            break
            
    # Final check
    final_C5 = evaluate_erdos_solution(best_h)
    if final_C5 < target_bound:
        return (best_h.tolist(), final_C5, n_points)
    else:
        return (best_h.tolist(), final_C5, n_points)
