import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1-h, mode='full')
    # Take the latter half which corresponds to positive shifts
    overlap = overlap[n-1:]
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to meet the sum constraint
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
        # Final adjustment
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Simple rescaling isn't enough; use a more careful approach
            # Make sure sum equals exactly target_sum
            h = np.clip(h, 0, 1)
            diff = target_sum - np.sum(h)
            if diff != 0:
                # Add diff to some element or distribute it
                # For simplicity, just adjust first element (if not at bounds)
                if h[0] < 1:
                    h[0] = min(1, h[0] + diff)
            # Ensure final correction
            h = h * target_sum / np.sum(h)
    return h

            for i in range(n_points - k):
                if k == 0:

                    update[i] += 0.005 * (1 - h[i+k])
