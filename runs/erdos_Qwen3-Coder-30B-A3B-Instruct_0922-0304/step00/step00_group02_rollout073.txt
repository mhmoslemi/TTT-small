import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Default construction: binary pattern
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    # Ensure constraints
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_no_improvement = 100
    perturb_freq = 500
    perturb_strength = 0.05
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    no_improvement_count = 0
    
    # Precompute convolution for efficiency
    def compute_overlaps(h_vec):
        # Compute overlaps using full convolution
        conv = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Normalize by dx = 2.0 / n_points = 2.0 / n_points
        dx = 2.0 / n_points
        overlaps = conv * dx
        return overlaps
    
    def project_h(h_vec):
        # Clip to [0,1]
        h_clipped = np.clip(h_vec, 0, 1)
        # Scale to match sum = n_points / 2
        scale_factor = (n_points / 2) / np.sum(h_clipped)
        return h_clipped * scale_factor
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute current overlaps
        overlaps = compute_overlaps(h)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if iteration == 1:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute finite difference gradients
        grad = np.zeros(n_points)
        dx = 2.0 / n_points
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            overlap_plus = compute_overlaps(h_plus)[max_k]
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            overlap_minus = compute_overlaps(h_minus)[max_k]
            
            grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
        
        # Update h
        h_new = h - alpha * grad
        
        # Project
        h_new = project_h(h_new)
        
        # Evaluate new solution
        new_overlaps = compute_overlaps(h_new)
        new_c5 = np.max(new_overlaps)
        
        # Accept if better
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            alpha *= gamma
            if alpha < alpha_min:
                alpha = alpha_min
                
        # Optional perturbation to escape local minima
        if iteration % perturb_freq == 0 and no_improvement_count > 50:
            perturb_mask = np.random.random(n_points) < perturb_strength
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h = project_h(h)
            
    return (best_h.tolist(), float(best_c5), n_points)
