import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        # Start with a binary vector with exactly n_points/2 ones
        h = np.zeros(n_points)
        h[:n_points//2] = 1
        
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Time tracking
    start_time = time.time()
    elapsed = 0
    last_improvement = 0
    iteration = 0
    
    while elapsed < budget_s - 1 and last_improvement < 50:
        iteration += 1
        # Compute current overlaps
        overlaps = correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(overlaps)
        
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
        objective = cp.Minimize(max_overlap)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_relaxed.value
            else:
                h_new = h
        except:
            h_new = h
        
        # Project back to binary
        h_proj = np.zeros(n_points)
        h_proj[np.argsort(h_new)[::-1][:n_points//2]] = 1
        
        # Ensure sum is exactly n_points / 2
        current_sum = np.sum(h_proj)
        if current_sum < n_points / 2:
            # Set lowest values to 1
            low_indices = np.argsort(h_proj)
            for i in range(int(n_points / 2) - int(current_sum)):
                h_proj[low_indices[i]] = 1
        elif current_sum > n_points / 2:
            # Set highest values to 0
            high_indices = np.argsort(h_proj)[::-1]
            for i in range(int(current_sum) - int(n_points / 2)):
                h_proj[high_indices[i]] = 0
                
        # Or, smarter projection with assignment
        if np.sum(h_proj) != n_points / 2:
            # Use simple thresholding
            sorted_idx = np.argsort(h_new)[::-1]
            h_proj = np.zeros(n_points)
            h_proj[sorted_idx[:n_points//2]] = 1
        
        # Evaluate
        C5 = evaluate_erdos_solution(h_proj)
        
        # Update best if improved
        if C5 < best_C5:
            best_h = h_proj.copy()
            best_C5 = C5
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Perturb
        h = h_proj.copy()
        perturb_indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for i in perturb_indices:
            h[i] = 1 - h[i]
            
        # Adjust to maintain integral constraint
        diff = np.sum(h) - n_points / 2
        if abs(diff) > 1e-5:
            # Adjust slightly to correct the sum
            if diff > 0:
                zero_indices = np.where(h == 1)[0]
                if len(zero_indices) > 0:
                    idx = np.random.choice(zero_indices)
                    h[idx] = 0
            else:
                one_indices = np.where(h == 0)[0]
                if len(one_indices) > 0:
                    idx = np.random.choice(one_indices)
                    h[idx] = 1
                    
        elapsed = time.time() - start_time
        
    return (best_h.tolist(), best_C5, n_points)
