import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = np.correlate(h, 1 - h, mode="full")
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Normalize initial h if needed
    current_sum = np.sum(h_init)
    if current_sum != target_sum:
        h_init = h_init * target_sum / current_sum
    
    # Helper function to optimize
    def objective(h):
        h = np.abs(h)  # Ensure non-negative
        h = np.clip(h, 0, 1)  # Clip to [0, 1]
        # Apply constraint that sum should equal target_sum
        h = h * target_sum / np.sum(h)
        return evaluate_c5(h, dx)
    
    # Constraints for optimization
    def constraint_sum(h):
        return np.sum(h) - target_sum
    
    def constraint_bounds(h):
        return np.minimum(h, 1 - h)  # Ensures h[i] ∈ [0,1]
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: constraint_bounds(h)},
    ]
    
    # Initial evaluation
    current_h = h_init.copy()
    current_c5 = objective(current_h)
    
    # Perform optimization using scipy minimize
    method = 'SLSQP'
    
    try:
        options = {'maxiter': 1000, 'ftol': 1e-6, 'gtol': 1e-6}
        result = minimize(
            objective,
            current_h,
            method=method,
            constraints=constraints,
            options=options,
            callback=lambda x: None,
        )
        
        if result.success:
            current_h = result.x
        else:
            # Fallback: try a few more iterations
            pass

        # Final processing
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * target_sum / np.sum(current_h)
        current_c5 = evaluate_c5(current_h, dx)
        
        # Ensure correctness
        assert np.isclose(np.sum(current_h), target_sum, rtol=1e-4), "Sum constraint violated"
        assert all(0 <= h <= 1 for h in current_h), "Bounds constraint violated"
        
    except Exception as e:
        # Fallback to initial construction
        current_h = h_init.copy()
        current_c5 = evaluate_c5(current_h, dx)

    # Ensure we don't run out of time - return at least a valid solution
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        pass

    return (current_h, current_c5, n_points)
