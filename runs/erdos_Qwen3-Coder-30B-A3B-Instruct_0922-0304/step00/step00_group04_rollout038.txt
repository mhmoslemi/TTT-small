import numpy as np
from scipy import optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    # Initialize parameters
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    lag_max = n_points - 1
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create a reasonable starting point
        h0 = np.ones(n_points) * 0.5
        h0[0:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        h0 = h0 / np.sum(h0) * n_points / 2

    # Check if constraint is satisfied initially
    if abs(np.sum(h0) * dx - 1.0) > 1e-6:
        # Adjust to satisfy mass constraint
        h0 = h0 * (1.0 / np.sum(h0) * n_points / 2)
    
    best_h = h0.copy()
    best_C5 = evaluate_erdos_solution(h0)
    
    t0 = time.time()
    elapsed = lambda: time.time() - t0

    while elapsed() < budget_s * 0.9:
        # Use CVXPY for the optimization
        h = cp.Variable(n_points, name='h')
        C5 = cp.Variable(name='C5')

        # Define constraints
        constraints = [
            cp.sum(h) == n_points / 2,
            h >= 0,
            h <= 1,
            C5 >= 0
        ]

        # Add overlap constraints
        # For each lag k, we compute the overlap integral
        # We set up only constraints for meaningful lags
        for k in range(-lag_max, lag_max + 1):
            if k >= 0:
                # We form the convolution explicitly
                # For positive lags, we compute:
                # sum_{i=0}^{n-k-1} h[i] * (1 - h[i + k])
                start_idx = 0
                end_idx = n_points - k
            else:
                # For negative lags, we compute:
                # sum_{i=0}^{n+ k -1} h[i + (-k)] * (1 - h[i])
                start_idx = -k
                end_idx = n_points

            if end_idx > start_idx:
                lags_sum = cp.sum(cp.multiply(h[start_idx:end_idx], 1 - h[start_idx + k:end_idx + k]))
                constraints.append(lags_sum <= C5 / dx)

        # Define the objective
        objective = cp.Minimize(C5)

        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_val = h.value
                C5_val = C5.value
                if C5_val < best_C5:
                    # Rescale if needed to satisfy mass constraint
                    if abs(np.sum(h_val) - n_points/2) > 1e-4:
                        h_val = h_val * (n_points / 2) / np.sum(h_val)
                    
                    # Ensure bounds
                    h_val = np.clip(h_val, 0, 1)
                    best_C5 = C5_val
                    best_h = h_val
            
        except Exception:
            # In case of failure, fallback to gradient-based method
            pass
        
        # Gradient-based smoothing optimization
        try:
            def obj_func(h_vals):
                return evaluate_erdos_solution(h_vals)
            
            # Use L-BFGS-B
            result = optimize.minimize(
                fun=obj_func,
                x0=h0,
                method='L-BFGS-B',
                bounds=[(0, 1)] * n_points,
                options={'maxiter': 10}
            )
            
            if (not result.success) or np.sum(result.x) < 1e-5:
                continue
                
            # Adjust to satisfy sum constraint
            h_res = result.x * (n_points / 2) / np.sum(result.x)
            h_res = np.clip(h_res, 0, 1)
            
            # Recompute
            c5_current = evaluate_erdos_solution(h_res)
            if c5_current < best_C5:
                best_h = h_res
                best_C5 = c5_current
                
        except Exception:
            pass
        
        h0 = best_h.copy()
        
        # Check if time is up
        if elapsed() > budget_s * 0.9:
            break

    return (best_h, best_C5, n_points)

def evaluate_erdos_solution(h):
    """Evaluates the C5 bound for given h array"""
    # Ensure it's a numpy array
    h = np.array(h)
    # Compute correlation using numpy's correlate
    corr = np.correlate(h, 1 - h, mode='full')
    # Scale by dx = 2/n_points
    corr = corr * (2.0 / len(h))
    # Return maximum value
    return np.max(corr)
