import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create random feasible initial point
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    max_iter = 1000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations and find the maximum
        corr = signal.correlate(h, 1 - h, mode='full')
        # The lag indices range from -n+1 to n-1
        lags = np.arange(-n_points + 1, n_points)
        max_idx = np.argmax(corr)
        max_lag = lags[max_idx]
        max_corr = corr[max_idx] * dx
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        if max_lag >= 0:
            # Positive lag: h[i] * (1 - h[i + max_lag])
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
            for i in range(max_lag, n_points):
                grad[i] -= h[i - max_lag]
        else:
            # Negative lag: h[i] * (1 - h[i + max_lag])
            neg_lag = -max_lag
            for i in range(neg_lag, n_points):
                grad[i - neg_lag] += 1 - h[i]
            for i in range(n_points - neg_lag):
                grad[i] -= h[i + neg_lag]
        
        # Gradient descent with adaptive step size
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Scale to satisfy constraint sum(h) = n_points/2
        current_sum = np.sum(h_new)
        if current_sum > 0:
            h_new = h_new * (n_points / 2) / current_sum
            
        h = h_new
        
    return (best_h, best_c5, n_points)

def compute_c5(h, dx):
    corr = signal.correlate(h, 1 - h, mode='full')
    lags = np.arange(-len(h) + 1, len(h))
    max_lag_idx = np.argmax(corr)
    max_corr = corr[max_lag_idx]
    return max_corr * dx
