import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C_5 for given h"""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h), full mode
    corr = correlate(h, 1 - h, mode="full")
    # Extract the valid overlaps (the center part)
    center_start = len(h) - 1
    overlaps = corr[center_start:center_start + len(h)]
    # Scale by dx and return maximum
    return max(overlaps) * dx

def generate_initial_h(n_points):
    """Generate a binary starting point with exactly n_points/2 ones"""
    h = np.zeros(n_points, dtype=int)
    h[:n_points//2] = 1
    np.random.shuffle(h)
    return h

    for i in range(n):

            if h[i] == h[j]:
                continue
            # Create neighbor
            neighbor = h.copy()
            neighbor[i] = 1 - neighbor[i]
            neighbor[j] = 1 - neighbor[j]
            
            # Check tabu
            key = tuple(neighbor)
            if key in tabu_list:
                continue
            # Check score
            score = evaluate_h(neighbor)
            if score < best_score:
                best_h = neighbor
                best_score = score
