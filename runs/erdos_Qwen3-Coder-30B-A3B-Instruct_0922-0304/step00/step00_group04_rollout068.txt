import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    
    dx = 2.0 / n_points
    max_iterations = 5000
    tolerance = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute FFT
        H = fft.fft(h)
        H_conj = np.conj(H)
        
        # Compute the overlap integral using convolution theorem
        # We compute max over all lags of int h(x)(1-h(x+k)) dx
        # This corresponds to the maximum value of |H(k)(1-H*(k))|^2
        # Let's compute the full correlation.
        overlap_spectrum = np.abs(H * (1 - H_conj))**2
        # Normalize by dx to match the integral definition
        overlap_spectrum = overlap_spectrum / dx**2
        
        # Update H to decrease overlap (flatten spectrum)
        # Reduce all frequency components except DC to be close to the minimum
        target = np.min(overlap_spectrum[1:]) if len(overlap_spectrum) > 1 else 0
        if target > 0:
            # Create a new H that pushes to decrease overlap
            # Not a full optimization but a crystal ball direction update
            new_H = H.copy()
            # Simple heuristic: reduce non-DC components to limit the max
            # If this is too big, let it go back to a balance
            ratio = np.clip(target / (overlap_spectrum[1:] + 1e-12), 0, 1) if len(overlap_spectrum) > 1 else np.array([0])
            ratio = np.concatenate([[1], ratio])  # DC stays same
            new_H *= ratio
            new_h = np.real(fft.ifft(new_H))
            # Project new_h back to correct domain
            new_h = np.clip(new_h, 0, 1)
            # Adjust sum to match constraint
            new_h = new_h / np.sum(new_h) * (n_points / 2.0)
            h = new_h.copy()
        else:
            # Another way to increase performance using gradient
            # Modified H to force a smaller max overlap
            # In frequency domain, smaller peaks in spectral product = better C5
            h_freq = fft.fft(h)
            h_freq_real = h_freq.real
            h_freq_imag = h_freq.imag
            # Adjust to dampen spectrum fluctuations slightly
            # But still preserve important values
            # Use an exponential decay instead of hard threshold
            freqs = np.arange(n_points)
            weight = np.exp(-0.02 * freqs**2)
            weight[0] = 1.0  # DC component unchanged
            h_freq_real *= weight
            h_freq_imag *= weight
            h_freq_corrected = h_freq_real + 1j * h_freq_imag
            h_new = np.real(fft.ifft(h_freq_corrected))
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2.0)
            h = h_new.copy()
        
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Check for convergence
        if i > 10 and abs(c5 - best_c5) < tolerance:  # reasonable for local improvement
            break

    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_vals, dx):
    h = np.array(h_vals)
    n = len(h)
    # Convolution with (1 - h) with full mode
    # Use forward FFT to compute correlation efficiently
    h_fft = np.fft.fft(h)
    h_conj = np.conj(h_fft)
    one_minus_h = 1 - h
    one_minus_h_fft = np.fft.fft(one_minus_h)
    # convolution = fft ifft(fft(h) * fft(1-h))
    conv = np.fft.ifft(h_fft * one_minus_h_fft)
    # Take the maximum over all shifts
    conv = np.real(conv)
    # Compute overlap values
    max_overlap = np.max(conv)
    c5 = max_overlap * dx
    return c5
