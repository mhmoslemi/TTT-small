def evaluate_erdos_solution(h):
    h = np.array(h)
    n = len(h)
    dx = 2.0 / n
    # This uses the standard way to compute C5 from a step function
    c5_val = 0.0
    for k in range(n):
        # Shift h by k positions to the left (circular shift)
        shifted = np.roll(h, k)
        # Compute inner product of h and (1-shifted)
        overlap = np.sum(h * (1 - shifted)) * dx
        if overlap > c5_val:
            c5_val = overlap
    return c5_val
