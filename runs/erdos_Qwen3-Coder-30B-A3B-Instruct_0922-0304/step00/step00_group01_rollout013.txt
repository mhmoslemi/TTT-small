import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx=2.0 / len(h)):
    """Evaluate the Erdős constant C₅ for given h."""
    overlaps = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps[len(h)-1:])
    return max_overlap * dx

def project_to_feasible(h, target_sum):
    """Project h to satisfy sum(h) = target_sum and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        h = np.full_like(h, target_sum / len(h))
    else:
        h = h * (target_sum / current_sum)
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient of overlap for shift k_max."""
    gradient = np.zeros(n_points)
    for i in range(n_points):
        # h[i] contributes to overlap at shift k_max if i+k_max < n_points
        if i + k_max < n_points:
            gradient[i] += 1 - h[i + k_max]
        # Consider the term -h[i] * (1 - h_clipped[i - k_max])
        if i - k_max >= 0:
            gradient[i] -= h[i - k_max]
    return gradient

def line_search(h, grad, k_max, n_points, dx, alpha_init=0.01, rho=0.5, eta=0.5):
    """Perform backtracking line search to find optimal step size."""
    alpha = alpha_init
    current_c5 = evaluate_erdos_solution(h, dx)
    
    for _ in range(30):
        h_new = h - alpha * grad
        h_new = project_to_feasible(h_new, n_points / 2)
        new_c5 = evaluate_erdos_solution(h_new, dx)
        if new_c5 < current_c5 - eta * alpha * np.dot(grad, grad):
            return alpha, h_new
        alpha *= rho
    return alpha, h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem setup
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    h = project_to_feasible(h, target_sum)
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Compute overlaps for all shifts
        overlaps = correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps[n_points-1:]) + (n_points - 1)
        
        # Get current overlap value and check if acceptable
        current_c5 = overlaps[k_max] * dx
        if current_c5 < 0.3807:
            break
            
        # Compute gradient for k_max
        gradient = compute_gradient(h, k_max, n_points)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < 1e-12:
            break
        gradient = gradient / grad_norm
        
        # Line search
        alpha, h_new = line_search(h, gradient, k_max, n_points, dx)
        if not np.isfinite(alpha) or alpha <= 0:
            break
            
        h = h_new
        h = project_to_feasible(h, target_sum)
        
        # Update best solution
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        iteration += 1
        
    return (best_h.tolist(), best_c5, n_points)
