import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points // 2) * 0.5
        h = np.concatenate([h, np.zeros(n_points - len(h))])
        h = h / np.sum(h) * target_sum
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Compute initial C5
    def compute_C5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
    
    current_C5 = compute_C5(h)
    best_C5 = current_C5
    best_h = h.copy()
    
    # Parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = 10000
    time_start = time.time()
    
    # Cooling schedule
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    T = T_max
    delta = 0.1
    
    # Track accept rate
    accept_count = 0
    total_attempts = 0
    
    for i in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
            
        # Make a move
        h_new = h.copy()
        move_type = np.random.choice(['swap', 'shift'])
        
        if move_type == 'swap':
            i1, i2 = np.random.choice(n_points, 2, replace=False)
            # swap h[i1] and h[i2]
            h_new[i1], h_new[i2] = h_new[i2], h_new[i1]
        else:
            # adjacent shift
            i1 = np.random.randint(0, n_points - 1)
            i2 = i1 + 1
            # shift a small amount from i1 to i2
            d = np.random.uniform(0, delta)
            if h_new[i1] >= d:
                h_new[i1] -= d
                h_new[i2] += d
            else:
                d = h_new[i1]
                h_new[i1] = 0
                h_new[i2] += d
        
        # Rescale to maintain constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * target_sum / np.sum(h_new)
        
        new_C5 = compute_C5(h_new)
        delta_C = new_C5 - current_C5
        
        # Accept or reject
        total_attempts += 1
        if delta_C < 0 or np.random.rand() < np.exp(-delta_C / T):
            h = h_new
            current_C5 = new_C5
            accept_count += 1
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Update temperature
        if T > T_min:
            T *= alpha
            
        # Adaptive delta
        if total_attempts > 0 and total_attempts % 100 == 0:
            accept_rate = accept_count / total_attempts
            if accept_rate < 0.1:
                delta *= 0.9
            elif accept_rate > 0.3:
                delta = min(delta * 1.1, 0.5)
            accept_count = 0
            total_attempts = 0
    
    return (best_h.tolist(), best_C5, n_points)
