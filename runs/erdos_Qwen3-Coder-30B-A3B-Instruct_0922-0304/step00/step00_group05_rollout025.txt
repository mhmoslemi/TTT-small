import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except:
        # Fallback: create a basic symmetric construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        # Add a bump in the center
        center = n_points // 2
        h[center-5:center+5] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 100000
    learning_rate = 0.01
    eps = 1e-4
    patience = 200
    patience_counter = 0
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        # correlate(h, 1-h, mode="full") produces all lags
        conv = correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        c5_idx = np.argmax(conv)
        
        # Compute gradient
        grad = np.zeros_like(h)
        # For each lag that gives max overlap, add gradient contributions
        # The full convolution has length 2*n_points - 1
        # The lags are from -(n_points-1) to (n_points-1)
        lag = c5_idx - (n_points - 1)  # this is the lag term that's max
        
        # Add contribution from this lag
        if lag >= 0:
            # 1-h[i+lag] contributes to h[i] (this is how derivative goes)
            grad[:n_points-lag] += (1 - h[lag:])
        else:
            # lag is negative, so h[i] contribution is from h[i+|lag|] 
            grad[-lag:] += (1 - h[:n_points+lag])
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to correct sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, n_points)
        
        if new_c5 < best_c5 - eps:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if patience_counter > 100 and learning_rate > 1e-5:
            learning_rate *= 0.5
            patience_counter = 0
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_values, n_points):
    h = np.array(h_values)
    dx = 2.0 / n_points
    # Compute all integrals C_m
    conv = correlate(h, 1 - h, mode="full")
    c5 = np.max(conv) * dx
    return c5
