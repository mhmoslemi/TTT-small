import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    # Compute all overlaps using full convolution
    overlap = correlate(h, 1 - h, mode="full")
    # Scale by dx to get continuous integral
    overlap *= dx
    # Take maximum over all lags (excluding zero lag which is not considered for overlap)
    return np.max(overlap)

def project_to_feasible(h, dx):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    # Rescale to meet the integral constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h *= (1.0 / (h_sum * dx))
    return h

    swarm_size = 100
    max_iter = int(1e6)
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # One particle from initial construction
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Fill rest with random feasible points
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_h(p, dx) for p in particles]
    
    # Track global best
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # PSO iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        for i in range(swarm_size):
            # Compute current score
            current_score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_score < personal_best_scores[i]:
                personal_best[i] = particles[i].copy()
                personal_best_scores[i] = current_score
            
            # Update global best if needed
            if current_score < global_best_score:
                global_best = particles[i].copy()
                global_best_score = current_score
                
            # Update velocity and position
            r1, r2 = np.random.rand(2)
            # Compute cognitive and social components
            cognitive_component = cognitive * r1 * (personal_best[i] - particles[i])
            social_component = social * r2 * (global_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], dx)
            
    # Final evaluation
    best_h = global_best
    best_c5 = global_best_score
    return (best_h, best_c5, n_points)
