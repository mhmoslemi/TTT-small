def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 constant for given h"""
    n = len(h)

    conv = correlate(h, 1 - h, mode="full")
    # The max is over all shifts
    max_val = np.max(conv) * dx
    return max_val

def project_h(h, n_points):
    """Project h to feasible region with correct sum"""

    # Scale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
