import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create initial h that satisfies constraints
        h = np.ones(n_points // 2) * 0.5
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0)
    
    # Ensure constraint ∫h dx = 1
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    # Helper function to compute C5
    def compute_c5(h_vec):
        # Compute correlation for all lags
        corr = correlate(h_vec, 1 - h_vec, mode="full")
        # Correct the scaling to match dx
        return np.max(corr) * dx
    
    # Compute initial C5
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Parameters for simulated annealing
    T_max = 1.0
    T_min = 1e-4
    max_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    T = T_max
    delta = 0.1
    
    # Track acceptance rate for adaptive delta
    accept_count = 0
    total_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try perturbation
        new_h = h.copy()
        
        # Randomly choose perturbation type
        if np.random.random() < 0.5:
            # Swap two points
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            # Swap the values with a small change
            delta_val = np.random.random() * delta * 0.5
            if new_h[idx1] - delta_val >= 0 and new_h[idx2] + delta_val <= 1:
                new_h[idx1] -= delta_val
                new_h[idx2] += delta_val
        else:
            # Adjacent shift
            idx = np.random.randint(0, n_points - 1)
            delta_val = np.random.random() * delta * 0.5
            if new_h[idx] - delta_val >= 0 and new_h[idx + 1] + delta_val <= 1:
                new_h[idx] -= delta_val
                new_h[idx + 1] += delta_val
                # Also try to balance the sum constraint
                # Find a point to decrease
                idx3 = np.random.randint(0, n_points)
                if new_h[idx3] + delta_val <= 1:
                    new_h[idx3] += delta_val
                    new_h[idx] -= delta_val
        
        # Ensure sum constraint using linear programming approach
        current_sum = np.sum(new_h)
        scale_factor = target_sum / current_sum
        new_h = new_h * scale_factor
        
        # Clamp values to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Recompute C5
        new_c5 = compute_c5(new_h)
        
        # Accept or reject
        dC = new_c5 - current_c5
        if dC < 0 or np.random.random() < np.exp(-dC / T):
            h = new_h
            current_c5 = new_c5
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Update temperature
        T = max(T_min, T * alpha)
        
        # Adaptive delta based on acceptance rate
        total_count += 1
        if dC < 0:
            accept_count += 1
        if total_count % 100 == 0 and total_count > 0:
            accept_rate = accept_count / total_count
            if accept_rate < 0.2:
                delta *= 0.9
            elif accept_rate > 0.5:
                delta *= 1.1
            accept_count = 0
            total_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
