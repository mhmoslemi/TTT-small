import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    dx = 2.0 / n_points
    max_iter = int(100000)
    gamma = 0.01
    eps = 1e-6
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute C5 (overlap integral)
        # This is effectively convolution with (1-h)
        conv_result = signal.correlate(h, 1 - h, mode='full')
        # The overlap integral values for each lag
        c5_vals = conv_result * dx
        # Exclude trivial cases and get the maximum
        c5_vals = c5_vals[len(h)-1:]  # only valid lags
        current_c5 = np.max(c5_vals)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        # We only consider the "active" lags where max was achieved
        max_lags = np.where(np.abs(c5_vals - current_c5) < eps)[0]
        gradient = np.zeros(n_points)
        
        for lag in max_lags:
            # lag in [0, 2*n_points - 2], but we need [0, n_points - 1]
            actual_lag = lag
            # Compute gradient contribution
            for j in range(n_points):
                if j + actual_lag < n_points:
                    gradient[j] += 1 - h[j + actual_lag]
        
        # Update
        h = h - gamma * gradient
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        
        # Rescale to sum = 1
        h_sum = np.sum(h) * dx
        if h_sum != 0:
            h = h * (1.0 / h_sum)
        
        # Reduce learning rate occasionally
        if i > 0 and i % 100 == 0:
            gamma *= 0.9
    
    # Final evaluation of best solution
    conv_result = signal.correlate(best_h, 1 - best_h, mode='full')
    c5_vals = conv_result * dx
    c5_vals = c5_vals[len(best_h)-1:]
    final_c5 = np.max(c5_vals)
    
    return (best_h.tolist(), final_c5, n_points)
