import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for shifts k from -n+1 to n-1
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    # Return max overlap
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: create a symmetric initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[-1] = 0.4
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Normalize h to satisfy constraint ∫ h(x)dx = 1
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)

    # Initialize tracking variables
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # Optimizer parameters
    alpha = 0.1
    max_iter = 1000
    patience = 50
    patience_counter = 0
    prev_C5 = best_C5 + 1.0

    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break

        # Compute overlaps for all shifts to find worst shift k_max
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlaps) - (n_points - 1)
        max_overlap = overlaps[k_max + n_points - 1]
        
        # Compute gradient w.r.t. h for the worst shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] += (1 - h[j]) * dx
            grad[j] -= h[i] * dx
        
        # Gradient step
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Convex Refinement (QP for worst case shift)
        try:
            # We want to minimize max_overlap while preserving sum(h) = n/2
            # And 0 <= h[i] <= 1
            
            # Define symbolic variables
            w = Variable(n_points)
            # QP formulation for improving h at the worst case shift k_max
            # We solve a QP to find a nearby feasible point that reduces overlap 
            # We'll formulate this in terms of projected gradient method
            
            # Optimize h updates:
            # Callback function to compute overlaps
            h_test = h.copy()
            overlaps_new = np.correlate(h_test, 1 - h_test, mode='full') * dx
            new_max_overlap = overlaps_new[k_max + n_points - 1]
            
            # If improvement happened, consider refining with QP
            if new_max_overlap <= max_overlap:
                # Keep the updated h
                h = h_test
                new_C5 = new_max_overlap
                if new_C5 < best_C5:
                    best_h = h.copy()
                    best_C5 = new_C5
        except:
            pass  # Fail gracefully

        # Check convergence criteria
        if abs(prev_C5 - best_C5) < 1e-6:
            patience_counter += 1
            if patience_counter > patience:
                break
        else:
            patience_counter = 0
        prev_C5 = best_C5

        # Adaptive step size decay
        if iteration % 10 == 0:
            alpha *= 0.95

    # Final validation
    final_C5 = evaluate_erdos_solution(best_h)

    return (best_h.tolist(), float(final_C5), len(best_h))
