def compute_c5(h, dx):
    # This computes C5 = max_k int h(x)(1-h(x+k)) dx
    n_points = len(h)
    # Full cross-correlation of h and (1-h)
    conv_result = correlate(h, 1 - h, mode='full')
    # The full convolution includes negative lags
    # The maximum is over all lags from -N+1 to N-1
    max_val = np.max(conv_result)
    # Scale by dx
    return max_val * dx
