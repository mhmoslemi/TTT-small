import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n = len(h)
    else:
        n = 45
        h = np.full(n, 0.5)
    
    # Normalize h to have sum = n/2
    h = h * (n / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Ensure sum constraint is exactly met
    h = h * (n / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iteration += 1
        
        # Compute all overlaps and find maximum
        full_correlation = np.correlate(h, 1 - h, mode="full")
        overlaps = full_correlation[len(h)-1:]
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # If we've achieved the target or no improvement for a while, stop
        if max_overlap < 0.3807 or iteration - last_improvement > 50:
            break
            
        # Formulate and solve convex optimization problem
        h_var = cp.Variable(n)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n/2,
            cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, -k_max))) <= C5_var
        ]
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Solve
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                C5_new = C5_var.value
                
                if C5_new < best_C5:
                    best_C5 = C5_new
                    best_h = h_new.copy()
                    last_improvement = iteration
                    
                h = h_new.copy()
            else:
                # Use the existing h if optimization fails
                pass
        except:
            # If CVXPY fails, just keep current h
            pass
    
    # Final validation with full computation
    final_C5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_C5, len(best_h))

def evaluate_erdos_solution(h):
    """Compute the C5 bound using correlation method"""
    h = np.array(h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr[len(h)-1:])
    return max_overlap
