import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameter setup
    n_points = 45
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 5000
    patience = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.full(n_points, 0.5)
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    patience_count = 0
    iteration = 0
    
    # Optimization loop
    while iteration < max_iter and time.time() - start_time < budget_s * 0.95:
        # Compute overlaps
        full_corr = correlate(h, 1 - h, mode="full")
        # Extract the relevant part of correlation (from -n+1 to n-1)
        # This gives overlaps for shifts k in range [-n+1, n-1]
        shifts = np.arange(-n_points + 1, n_points)
        overlaps = full_corr * dx
        # Only consider valid shifts where the overlap is computed correctly
        valid_overlap_indices = (shifts >= 0) & (shifts < n_points)
        valid_shifts = shifts[valid_overlap_indices]
        valid_overlaps = overlaps[valid_overlap_indices]
        
        # Find maximum overlap
        idx_max = np.argmax(valid_overlaps)
        k_max = valid_shifts[idx_max]
        c5_current = valid_overlaps[idx_max]
        
        # Early termination if target reached
        if c5_current <= target_bound:
            break
            
        # Update best solution if improved
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            break

        # Build new constraint
        # Constraint: sum_i h[i] * (1 - h[(i + k_max) % n_points]) <= C5
        # Let's define the quadratic constraint for this shift
        # This is quotable as: sum_i h[i] - sum_i h[i] * h[(i + k_max) % n_points] <= C5
        # So it is: -sum_i h[i] * h[(i + k_max) % n_points] <= C5 - sum_i h[i]
        # Which is: sum_i h[i] * h[(i + k_max) % n_points] >= sum_i h[i] - C5
        
        # We'll do correlative approach: let C5 be a variable and add a constraint such that 
        # max_k int h(x) * (1-h(x+k)) dx <= C5  
        # We take this k_max explicitly
        
        # Now a direct convex optimization using CVXPY
        # Variable h and C5
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Define quadratic constraint
        # This is ellipse-like structure with specific shift.
        # For shift k_max:
        # sum h[i] * (1 - h[(i+k_max)%n]) <= C5 
        # Which is:
        # sum h[i] - sum h[i] * h[(i+k_max)%n] <= C5 
        # =>
        # sum h[i] * h[(i+k_max)%n] >= sum h[i] - C5
        
        # Build constraint: sum_{i} h[i]*h[(i+k_max)%n] >= sum_{i} h[i] - C5
        
        # Compute the inner product for shift k_max
        h_shifted = np.roll(h, k_max)
        quadratic_expr = cp.sum(cp.multiply(h_var, h_shifted))
        
        # Enforce coalition
        constraint1 = cp.sum(h_var) == n_points / 2
        constraint2 = cp.sum(cp.multiply(h_var, 1 - h_shifted)) <= C5_var
        constraint3 = h_var >= 0
        constraint4 = h_var <= 1
        
        # Minimize C5
        objective = cp.Minimize(C5_var)
        prob = cp.Problem(objective, [constraint1, constraint2, constraint3, constraint4])
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                new_h_vals = h_var.value
                if new_h_vals is not None:
                    h = np.clip(new_h_vals, 0, 1)
                    # Renormalize
                    h = h / np.sum(h) * (n_points / 2)
        except:
            pass  # In case of solver failure, proceed manually
            
        iteration += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    if final_c5 < 0.3807:  # Tighter check
        return (best_h, final_c5, n_points)
    else:  # fallback to initial
        return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps for shifts 0 to n_points-1
    full_corr = correlate(h, 1 - h, mode="full")
    shifts = np.arange(-n_points + 1, n_points)
    overlaps = full_corr * dx
    
    # Extract the relevant overlaps (shifts from 0 to n_points-1)
    valid_shifts = (shifts >= 0) & (shifts < n_points)
    valid_overlaps = overlaps[valid_shifts]
    
    return np.max(valid_overlaps)
