import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Otherwise initialize randomly
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, n_points)
    
    # Gradient descent parameters
    learning_rate = 0.01
    max_iter = int(budget_s / 0.6)  # conservatively estimate 0.6s per iteration
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        # Adjust for proper indexing
        full_corr = corr.copy()
        max_idx = np.argmax(full_corr)
        max_k = max_idx - (len(h) - 1)
        max_corr_val = full_corr[max_idx]
        
        # Compute gradient w.r.t. h for the maximum lag
        grad = np.zeros_like(h)
        if max_k >= 0:
            # h[i] contributes to h[i] * (1 - h[i + k])
            # Gradient: 1 - 2 * h[i + k] 
            for i in range(len(h) - max_k):
                grad[i] += 1 - 2 * h[i + max_k]
        else:
            # h[i] contributes to h[i - k] * (1 - h[i])
            # Gradient:  -(1 - 2 * h[i]) 
            for i in range(-max_k, len(h)):
                grad[i] += -(1 - 2 * h[i - max_k])
        
        # Update h using gradient descent
        h_new = h - learning_rate * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        c5_new = compute_c5(h_new, n_points)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
    
    return (best_h.tolist(), float(best_c5), int(n_points))

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
