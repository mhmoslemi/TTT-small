import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 constant for given step function h"""
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create simple initial guess if no parent construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6

    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure initial sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    
    # Optimization params
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    last_improvement = 0
    
    start_time = time.time()
    max_iterations = int(budget_s * 1000)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current solution
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find all indices where maximum overlap occurs
        max_lags = np.where(overlap == max_overlap)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for m in max_lags:
            if m < n_points:
                # Forward differences
                valid_indices = np.arange(min(m, n_points - m))
                if len(valid_indices) > 0:
                    gradient[valid_indices] += (1 - h[valid_indices + m]) * dx
            if m >= n_points:
                # Backward differences
                valid_indices = np.arange(m - n_points + 1, n_points)
                if len(valid_indices) > 0:
                    gradient[valid_indices] += (1 - h[valid_indices + m - n_points]) * dx
        
        if len(max_lags) > 0:
            gradient /= len(max_lags)
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h, dx)
        
        # Update best if improved
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration - last_improvement > gamma_patience:
            learning_rate *= gamma_decrease_factor
            last_improvement = iteration
    
    return (best_h.tolist(), best_c5, n_points)
