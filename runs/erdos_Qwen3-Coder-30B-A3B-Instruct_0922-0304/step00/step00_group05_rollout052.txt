import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: step function with 2 segments
        n_points = 70
        h = np.ones(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 1.5
        
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial h satisfies constraint
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Helper: project h to constraints
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Objective function to evaluate C5
    def evaluate_c5(h_vec):
        # Compute overlaps using full correlation with flipped h
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Consider only valid overlaps (shifts from -n+1 to n-1)
        shifts = np.arange(-n_points + 1, n_points)
        # Find where overlap > 0 (positive)  
        valid_overlaps = corr[shifts >= 0]
        return np.max(valid_overlaps) * dx
    
    # Initialize optimization
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Parameters
    gamma = 0.01
    eps = 1e-4
    iter_count = 0
    last_improvement = 0
    max_iters = int(budget_s * 500)  # Rough estimate for iterations
    start_time = time.time()
    
    while iter_count < max_iters:
        if time.time() - start_time > budget_s - 1:
            break
            
        iter_count += 1
        
        # Evaluate current loss
        current_c5 = evaluate_c5(h)
        
        if current_c5 < best_c5 - eps:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iter_count
        
        # Compute gradient via finite differences or direct computation
        # Using one step of computation of each correlation component
        # To compute gradient at each point, we check which lagged correlations 
        # are contributing to max value
        
        # Evaluate all correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        # Only consider valid lags (shift ≥ 0) otherwise invalid
        valid_corr = corr[lags >= 0]
        max_val = np.max(valid_corr)
        max_indices = np.where(np.abs(valid_corr - max_val) < eps)[0]
        max_lags = lags[max_indices + (n_points - 1)]
        
        # Compute gradient estimate
        grad = np.zeros_like(h)
        for lag in max_lags:
            if lag >= 0:
                grad[:-lag] += (1 - h[lag:])
            else:
                grad[-lag:] += (1 - h[:lag])
        
        # Update h
        h = h - gamma * grad
        
        # Project
        h = project_h(h)
        
        # Adapt learning rate
        if iter_count > last_improvement + 100 and gamma > 1e-5:
            gamma *= 0.1
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_c5, len(best_h))
