import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 45
        h = np.full(n_points, 0.5, dtype=float)
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Ensure the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Optimization parameters
    alpha = 0.1
    max_iter = 50000
    patience = 50
    patience_counter = 0
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts k
        # Using convolution to compute overlaps efficiently
        h_padded = np.concatenate([h, h])  # periodic padding for full convolution
        overlaps = signal.correlate(h_padded, 1 - h, mode='full') * dx
        # Extract the valid overlaps (k from -n+1 to n-1)
        overlaps = overlaps[n_points-1:2*n_points-1]
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # If the overlap is too small, skip gradient updates
        if max_overlap < 1e-10:
            break
            
        # Compute gradient at k_max
        grad = np.zeros_like(h)
        k = k_max - (n_points - 1)
        for j in range(n_points):
            j_k = (j + k) % n_points
            if 0 <= j_k < n_points:
                grad[j] += (1 - h[j_k]) * dx
                grad[j_k] -= h[j] * dx
                
        # Update h
        h = h - alpha * grad
        
        # Enforce bounds and sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Update best solution if better
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stop if no improvement for patience steps
        if patience_counter >= patience:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return best solution
    return (list(best_h), final_c5, n_points)

def evaluate_erdos_solution(h_values):
    h = np.array(h_values, dtype=float)
    n = len(h)
    dx = 2.0 / n
    
    # Compute all overlaps for k in [-n+1, ..., n-1]
    h_padded = np.concatenate([h, h])  # periodic padding for full convolution
    overlaps = signal.correlate(h_padded, 1 - h, mode='full') * dx
    
    # Extract overlaps for k from -n+1 to n-1
    overlaps = overlaps[n - 1:2 * n - 1]
    
    # Return maximum overlap
    return np.max(overlaps)
