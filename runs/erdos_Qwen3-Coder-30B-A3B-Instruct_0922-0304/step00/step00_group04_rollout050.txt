import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a random feasible initial point
        n_points = 42
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.01
    tol = 1e-6
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    dx = 2.0 / n_points
    
    # Helper to compute C5
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        # Find maximum over all lags
        max_val = np.max(corr)
        return max_val * dx
    
    # Helper to compute gradient w.r.t. max lag
    def compute_gradient(h, max_k):
        grad = np.zeros_like(h)
        # Correlation: for each i, consider sum_j h[i+j] * (1 - h[j]) where j runs over valid indices
        # Exact gradient for max_k:
        # For each i, we have: contribution from h[i] to the term with lag max_k is (1 - h[i + max_k])
        # So gradient for h[i] is - (1-h[i + max_k]) (if i+max_k < len) else 0 if not within bounds
        # That's 1 - h[i + max_k] * 1_{i + max_k < n} 
        # But actually, we consider: h[i] * (1 - h[i + k]) * dx where i ranges to 0 to n-1-k
        for i in range(len(h)):
            j = i + max_k
            if 0 <= j < len(h):
                grad[i] -= (1 - h[j])
            # Also symmetric term due to h[j] * (1 - h[i]) but we don't double-count
        return grad
    
    # Helper to project h back into feasible region
    def project_h(h, dx):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h) * dx
        if abs(current_sum - 1.0) > 1e-6:
            # Scale to meet exact sum
            scale = 1.0 / current_sum
            h = h * scale
        return h
    
    # Current iteration
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Positions where the correlation is defined and has value
        # Full correlation dlen = 2*n - 1 => indices from -(n-1) to (n-1)
        # But we only consider the positive lags (since h is on [0,2], lag k from 0 to 2-k is our matrix)
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        # Adjust index to get real lag k
        lag = max_k - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = i + lag
            if 0 <= j < len(h):
                grad[i] -= (1 - h[j])
        
        # Normalize gradient (optional step for numerical stability)
        norm_grad = np.linalg.norm(grad)
        if norm_grad > 0:
            grad = grad / norm_grad
        
        # Perform step
        h_new = h - lr * grad
        h_new = project_h(h_new, dx)
        
        # Evaluate new C5
        new_c5 = compute_c5(h_new)
        
        # Update if improvement
        if new_c5 < best_c5 - tol:
            best_c5 = new_c5
            best_h = h_new.copy()
        
        h = h_new
    
    # Final check of bounds and sum
    best_h = project_h(best_h, dx)
    final_c5 = compute_c5(best_h)
    
    # Make sure final C5 is indeed the best
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
