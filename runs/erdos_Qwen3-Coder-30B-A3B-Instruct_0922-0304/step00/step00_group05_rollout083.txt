import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
    except:
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//4] = 1.0
        h[n_points//4:2*n_points//4] = 0.0
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure h sums to target_sum
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Initial evaluation
    def compute_C5(h_vec):
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Simulated annealing parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = 100000
    cooling_rate = np.exp(np.log(T_min / T_max) / max_iter)
    temp = T_max
    perturb_scale = 0.1
    
    # Track acceptance rate for adaptive perturbation
    accept_count = 0
    total_count = 0
    accept_rate = 0.5
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Create perturbation
        new_h = h.copy()
        
        # Try adjacent mass shift (more effective for step functions)
        idx1 = np.random.randint(0, n_points - 1)
        idx2 = idx1 + 1
        delta = np.random.uniform(-perturb_scale, perturb_scale)
        
        # Apply change but keep within bounds
        old_val1, old_val2 = new_h[idx1], new_h[idx2]
        new_h[idx1] = np.clip(old_val1 + delta, 0, 1)
        new_h[idx2] = np.clip(old_val2 - delta, 0, 1)
        
        # Rescale to maintain sum
        new_sum = np.sum(new_h)
        new_h = new_h * target_sum / new_sum
        
        # Compute new C5
        new_C5 = compute_C5(new_h)
        
        # Compute energy difference
        delta_C = new_C5 - best_C5
        
        # Accept with probability
        if delta_C < 0 or np.random.rand() < np.exp(-delta_C / temp):
            h = new_h
            if new_C5 < best_C5:
                best_h = new_h.copy()
                best_C5 = new_C5
                
        # Update temperature
        temp = max(T_min, temp * cooling_rate)
        
        # Adapt perturbation scale
        total_count += 1
        if delta_C < 0:
            accept_count += 1
        if total_count % 100 == 0:
            accept_rate = accept_count / total_count
            if accept_rate < 0.2:
                perturb_scale *= 0.95
            elif accept_rate > 0.4:
                perturb_scale *= 1.05
            accept_count = 0
            total_count = 0
    
    return (best_h.tolist(), best_C5, n_points)
