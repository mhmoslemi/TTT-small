import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 42
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a step function
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    def optimize_convex():
        h_var = Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
        
        # Create a rough approximation of the problem using convex constraints
        objective = Minimize(1.0)  # Will be updated
        
        prob = Problem(objective, constraints)
        
        # Try solving with different solvers
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return np.array(h_var.value)
        except:
            pass
        return h.copy()
    
    def project_and_rescale(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * (n_points / 2) / current_sum
        return h_vec
    
    def get_gradient(h_vec, k):
        # Finite difference gradient for lag k
        n = len(h_vec)
        grad = np.zeros(n)
        dx = 2.0 / n
        for i in range(n):
            h_plus = h_vec.copy()
            h_minus = h_vec.copy()
            h_plus[i] += 1e-6
            h_minus[i] -= 1e-6
            h_plus = project_and_rescale(h_plus)
            h_minus = project_and_rescale(h_minus)
            
            # Compute overlap for k
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
            grad[i] = (corr_plus[k + n - 1] - corr_minus[k + n - 1]) / (2e-6 * dx)
        return grad
    
    # Stage 1: Convex relaxation and coarse optimization
    print("Stage 1: Convex relaxation...")
    h = optimize_convex()
    h = project_and_rescale(h)
    c5 = evaluate_erdos_solution(h, n_points)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Stage 2: Gradient descent refinement, up to 100 iterations
    print("Stage 2: Gradient descent...")
    learning_rate = 0.1
    for iteration in range(100):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        # Identify dominant lag
        corr = np.correlate(h, 1 - h, mode="full")
        k_best = np.argmax(corr)
        # Gradient update
        grad = get_gradient(h, k_best - n_points + 1)
        h = h - learning_rate * grad
        h = project_and_rescale(h)
        c5 = evaluate_erdos_solution(h, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        # Reduce learning rate if not improving
        if iteration > 0 and iteration % 10 == 0:
            learning_rate *= 0.5
    
    # Stage 3: High resolution fine-tuning
    if time.time() - start_time < budget_s - 2:
        print("Stage 3: High resolution refinement...")
        n_points_high = 84
        dx_h = 2.0 / n_points_high
        # Upsample
        h_upsampled = np.interp(np.linspace(0, 2, n_points_high), 
                                np.linspace(0, 2, n_points), 
                                h)
        h = h_upsampled
        h = project_and_rescale(h)
        c5 = evaluate_erdos_solution(h, n_points_high)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Fine-tune high-res version
        learning_rate = 0.05
        for iteration in range(50):
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
            corr = np.correlate(h, 1 - h, mode="full")
            k_best = np.argmax(corr)
            grad = get_gradient(h, k_best - n_points_high + 1)
            h = h - learning_rate * grad
            h = project_and_rescale(h)
            c5 = evaluate_erdos_solution(h, n_points_high)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            if iteration % 5 == 0: 
                learning_rate *= 0.9
    
    # Final check
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    final_h = best_h.tolist()
    return final_h, best_c5, len(final_h)
