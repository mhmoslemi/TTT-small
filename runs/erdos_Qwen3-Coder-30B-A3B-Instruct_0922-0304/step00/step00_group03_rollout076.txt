import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    h = np.array(initial_h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Precompute cross-correlation indices
    indices = np.arange(n_points)
    
    def compute_C5(h):
        # Compute all cross-correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Reshape appropriately
        corr = corr[len(corr)//2:]  # Take only positive shifts
        return np.max(corr) * dx
    
    def project_h(h):
        # Project onto feasible set: [0,1] and integral = 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_grad(h, k_max):
        # k_max is the index of the shift that gives maximum overlap
        # we compute gradient of S_k_max with respect to h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # derivative of h[i] * (1 - h[i + k_max]) 
            # w.r.t. h[i] is 1 - h[i + k_max]
            # but only at index i - k_max in the correlation
            if i >= k_max:
                grad[i] = 1 - h[i - k_max]
        return grad
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    start_time = time.time()
    
    # Simple gradient descent with projection
    max_iter = 100000
    learning_rate = 0.01
    
    for i in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # compute current C5
        current_C5 = compute_C5(h)
        
        # Find the shift that maximizes overlap
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[len(corr)//2:]
        k_max = np.argmax(corr)
        
        # compute gradient
        grad = compute_grad(h, k_max)
        
        # update step
        h_new = h - learning_rate * grad
        
        # Project onto constraints
        h_new = project_h(h_new)
        
        # update
        h = h_new
        
        # check improvement
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
    # Final evaluation
    final_C5 = compute_C5(best_h)
    return (best_h, final_C5, len(best_h))
