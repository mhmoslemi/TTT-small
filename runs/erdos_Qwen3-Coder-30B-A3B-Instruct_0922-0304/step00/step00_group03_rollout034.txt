import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    # Compute the overlap integral C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    # Using full correlation mode to get all lags
    corr = correlate(h, 1 - h, mode='full')
    # The correlation includes negative lags, but we're interested in the maximum
    # over all valid lags. The correlation length is 2*n-1; we take the middle n-1 values
    # corresponding to positive shifts. But in full mode, we have 2*n-1 values.
    # For a shift of k (in the range [0, n-1]), the correlation index in full mode is k + (n-1)
    # This will be from (n-1) to 2*(n-1) inclusive.
    # But since we have a circular buffer nature of our problem, we need to look at std correlation
    # Maybe seeing integers works best:
    # The actual overlap corresponds to the correlation shifted by n-1 positions
    # For some purists, the appropriate corresponding shift in indices would be:
    # 0 at index offset_equiv = n-1, 1 at index offset_equiv+1, ..., n-1 at offset_equiv + (n-1)
    # But because of periodicity in correlation or because of
    # direct position interpretation, let's take the absolute maximum of the valid correlations
    # Indices from n-1 to 2*(n-1)  (inclusive)
    half_len = len(h)
    start = half_len - 1
    end = 2 * half_len - 2
    # Take the max among all the valid/meaningsful overlaps
    return np.max(corr[start:end+1]) * dx

def move_h(h, i, j, copy=True):
    # Flip bit i and bit j to generate neighbor.
    if copy:
        h = h.copy()
    if h[i] == h[j]:
        # Cannot flip same bits
        return h, False
    else:
        h[i] = 1 - h[i]
        h[j] = 1 - h[j]
        return h, True

def generate_neighbor(h, n_points):
    # Generate a neighbor by flipping two bits with equal number of 1s
    # Create copy
    h = h.copy()
    # Get indices of 1s and 0s
    ones_idx = [i for i in range(n_points) if h[i] == 1]
    zeros_idx = [i for i in range(n_points) if h[i] == 0]

    # If it's too unbalanced, then construct differently
    # We keep the constraint: must maintain exactly n_points/2 ones
    if len(ones_idx) >= n_points // 2:  # have more 1s or equal
        # We need to turn 2 ones into 0s and 2 zeros into 1s
        # Select 2 1s to make 0
        idxs1 = random.sample(ones_idx, 2)
        # Select 2 0s to make 1
        idxs2 = random.sample(zeros_idx, 2)
    else: # have less than n_points/2 ones
        # Need to turn 2 0s into 1s and 2 1s into 0s
        idxs1 = random.sample(ones_idx, 2)
        idxs2 = random.sample(zeros_idx, 2)

    for i in idxs1:
        h[i] = 0
    for i in idxs2:
        h[i] = 1

    return h, True

    # Ensure it's binary and correct sum
    h = (h > 0.5).astype(int)
    current_sum = np.sum(h)
    target_sum = n_points // 2
    if current_sum != target_sum:
        # Adjust to have exactly target_sum ones
        positions = list(np.where(h == 1)[0])
        needed = target_sum - current_sum
        if needed > 0:
            zeros_positions = list(np.where(h == 0)[0])
            # flip needed zeros to ones
            for i in range(needed):
                if i < len(zeros_positions):
                    h[zeros_positions[i]] = 1
        else:
            needed_flip = -needed
            for i in range(needed_flip):
                if i < len(positions):
                    h[positions[i]] = 0

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    tabu_list = set()
    tabu_tenure = 100
    iterations = 0
    max_iterations = 1000
    last_improvement_iter = 0

    while time.time() - start_time < budget_s - 1 and iterations < max_iterations:
        iterations += 1
        new_h, valid = generate_neighbor(h, n_points)
        if not valid:
            continue

        # Create a hashable representation for tabu check
        key = tuple(new_h)
        if key in tabu_list:
            continue

        c5_new = evaluate_h(new_h, dx)
        if c5_new < best_c5:
            best_h = new_h.copy()
            best_c5 = c5_new
            last_improvement_iter = iterations

        # Add current into tabu
        tabu_list.add(tuple(h))
        if len(tabu_list) > tabu_tenure:
            # Remove oldest item (we can do with pop; only caveat is
            # that ordering is undefined. If we have performance issues,
            # can use a deque or rework)
            try:
                tabu_list.pop()
            except KeyError:
                pass  # ignore if empty
        h = new_h

        # Early termination check
        if iterations - last_improvement_iter > max_iterations // 4:
            break

    # Final validation
    final_sum = np.sum(best_h)
    if final_sum != n_points // 2:
        # Patch up if necessary
        positions_ones = np.where(best_h == 1)[0]
        while len(positions_ones) > n_points // 2:
            pos = positions_ones[-1]
            best_h[pos] = 0
            positions_ones = np.where(best_h == 1)[0]

            best_h[pos] = 1
            positions_ones = np.where(best_h == 1)[0]

    final_c5 = evaluate_h(best_h, dx)
    return (list(best_h), final_c5, n_points)
