def evaluate_erdos_solution(h):

    n = len(h)

    conv = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (corresponding to shifts k from -n+1 to n-1)
    c5 = np.max(conv) * dx
    return c5
