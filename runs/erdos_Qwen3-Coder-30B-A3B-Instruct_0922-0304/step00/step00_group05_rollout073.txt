    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    start_time = time()

        # Get all correlations with 1-h to compute C5
        corr = np.correlate(h, 1-h, mode="full")
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        # This is a very coarse evaluation, we'll compute from scratch for better accuracy
        C5_current = max_corr_val * dx

        # Try to solve relaxated version using CVXPY
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1]
        
        # Approximate h[i] * (1 - h[i+k]) with convex relaxation
        # We use first-order Taylor:
        # h[i] * (1 - h[i+k]) <= h[i] + (1 - h[i+k]) * (h[i] - h[i]) = h[i]
        # Better: h[i] * (1 - h[i+k]) <= h[i] - h[i]*h[i+k]

        # But in our case it's h[i] * (1 - h[i+k])
        # Let's test with linear upper bounds

        # For now, a simple approach using quadratic constraint (overly conservative, but convex)

        # Try projective approach instead of optimization

        h_new = h.copy()
        # Perturb slightly with gradient descent on estimated gradients
        # This is a heuristic but can work in presence of noise

def evaluate_erdos_solution(h, dx):
    # Compute the maximum overlap as maximum of correlations
    h_arr = np.array(h)
    corr = np.correlate(h_arr, 1-h_arr, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx
