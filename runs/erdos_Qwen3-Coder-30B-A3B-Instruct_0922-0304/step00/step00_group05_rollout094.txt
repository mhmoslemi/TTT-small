import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral C₅ for given h."""
    # Compute correlations for all lags and find maximum
    conv = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h, dx):
    """Project h onto feasible region: [0,1] and sum(h) = 1/dx."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize to sum = 1/dx
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (1.0 / dx) / sum_h
    return h

    np.random.seed(seed)

    dx = 2.0 / n_points
    swarm_size = 100

    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        particles = [np.array(initial_h_values)]
        particles.extend([np.random.rand(n_points) for _ in range(swarm_size - 1)])
    else:
        particles = [np.random.rand(n_points) for _ in range(swarm_size)]
    
    # Normalize initial particles
    for i in range(len(particles)):
        particles[i] = project_h(particles[i], dx)
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    p_best = [p.copy() for p in particles]
    p_best_vals = [evaluate_h(p, dx) for p in particles]
    
    # Initialize global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = p_best[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    # Main optimization loop
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            cognitive_component = cognitive * r1 * (p_best[i] - particles[i])
            social_component = social * r2 * (g_best - particles[i])
            velocities[i] = inertia * velocities[i] + cognitive_component + social_component
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], dx)
            
            # Evaluate current particle
            current_val = evaluate_h(particles[i], dx)
            
            # Update personal best
            if current_val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = current_val
                
                # Update global best
                if current_val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = current_val
    
    # Final evaluation
    final_c5 = evaluate_h(g_best, dx)
    
    return (g_best, final_c5, n_points)
