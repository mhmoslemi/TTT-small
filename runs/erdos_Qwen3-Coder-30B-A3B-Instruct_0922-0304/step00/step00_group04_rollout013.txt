import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 42
    dx = 2.0 / n_points
    lag_max = n_points - 1
    
    # Load initial construction if available
    try:
        h_init = initial_h_values
        if len(h_init) != n_points:
            h_init = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except:
        # Generate a reasonable starting point
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
    
    # Set up optimization problem
    h = cp.Variable(n_points, lower_bound=0, upper_bound=1)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [cp.sum(h) == n_points / 2, C5 >= 0]
    
    # Add overlap constraints for all lags
    # For each lag k, we compute the overlap:
    # sum_{i=0}^{n-|k|-1} h[i] * (1 - h[i+k])
    for k in range(-lag_max, lag_max + 1):
        if k < 0:
            left = h[:n_points + k]
            right = h[-k:]
        elif k > 0:
            left = h[:n_points - k]
            right = h[k:]
        else:
            left = h
            right = h
        
        # Overlap is sum of h[i] * (1 - h[i+k])
        overlap = cp.sum(cp.multiply(left, 1 - right))
        constraints.append(overlap <= C5 * dx)
        
        if time.time() - start_time > 0.8 * budget_s:
            break
    
    # Objective: minimize C5
    prob = cp.Problem(cp.Minimize(C5), constraints)
    
    try:
        prob.solve(solver=cp.GLPK_MI, verbose=False)
        if prob.status in ['optimal', 'optimal_inaccurate']:
            h_opt = h.value
        else:
            h_opt = h_init
    except:
        h_opt = h_init

    if time.time() - start_time > budget_s * 0.95:
        # Final evaluation
        pass
    else:
        # Try local refinement with gradient-based method if not too late
        try:
            from scipy.optimize import minimize
            
            def obj(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                overlaps = []
                for k in range(-lag_max, lag_max + 1):
                    corr = correlate(h_vec, 1-h_vec, mode='full')
                    if k >= 0:
                        overlaps.append(corr[n_points-1+k])
                    else:
                        overlaps.append(corr[n_points-1+k])
                return max(overlaps) * dx
            
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points/2
            
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            res = minimize(obj, h_opt, method='SLSQP', bounds=bounds, constraints=cons, 
                           options={'maxiter': 1000, 'ftol': 1e-8})
            if res.success:
                h_opt = np.clip(res.x, 0, 1)
        except:
            pass

    h_final = np.clip(h_opt, 0, 1)
    # Final check
    overlaps = []
    for k in range(-lag_max, lag_max + 1):
        corr = correlate(h_final, 1-h_final, mode='full')
        if k >= 0:
            overlaps.append(corr[n_points-1+k])
        else:
            overlaps.append(corr[n_points-1+k])
    c5_bound = max(overlaps) * dx
    
    if time.time() - start_time > budget_s * 0.99:
        return (initial_h_values, 0.3809, n_points)
    
    return (h_final, c5_bound, n_points)
