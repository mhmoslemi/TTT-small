import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create a simple starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 0.0
        h[1::2] = 1.0
        h[::2] = 0.0
        h = np.clip(h, 0, 1)
        
    # Normalize to meet integral constraint
    integral = np.sum(h) * 2.0 / n_points
    if integral != 1.0:
        h = h * (1.0 / integral)
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Parameters
    max_iter = 5000
    learning_rate = 0.01
    momentum = 0.9
    decay_every = 100
    
    v = np.zeros_like(h)
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Extract valid correlations (shifts from -n+1 to n-1)
        # But we care only about shifts that result in a valid window
        # Shape of corr is 2*n - 1
        n = len(h)
        # corr indices correspond to shifts from -n+1 to n-1
        # so the max is at index n-1
        c5_val = np.max(corr)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
            
        # Early termination
        if best_c5 <= 0.3807:
            break
            
        # Compute gradient of max overlap with respect to h
        argmax_shift = np.argmax(corr)
        shift = argmax_shift - (n - 1)
        
        # For the overlap at shift k, the derivative w.r.t. h[i] is 1 - h[i+k] if i+k is valid, 
        # and 0 otherwise. But we want to avoid the explicit derivative computation, 
        # instead we'll use a modified gradient descent using a pseudo-gradient approach
        
        # A smarter approach: estimate gradient directly by perturbing h
        # Let's compute gradient as a direction that would decrease the current peak overlap
        # since computing true gradient of max of correlation is hard, we will do 
        # gradient descent with direct handling of overlapping areas
        
        # Simplified gradient: gradient w.r.t to max overlap is given by direct correlation parts
        # If for h[i] we have contribution to overlap = h[i] * (1 - h[i+shift]), 
        # then derivative w.r.t. h[i] is (1 - h[i+shift])
        # But we handle all indices in the overlap
        grad = np.zeros_like(h)
        # Use correlation again but only for the shift that gave max overlap
        if shift >= 0:
            # shift positive implies we look at h[j] * (1 - h[j+shift]) for all j 
            grad[:n-shift] += 1 - h[shift:]
        else:
            # shift negative implies we look at h[j-shift] * (1 - h[j]) for all j  
            grad[-shift:] += 1 - h[:-shift]
        
        # Apply momentum
        v = momentum * v - learning_rate * grad
        h = h + v
        
        # Project back to feasible set (bounds and integral constraint)
        h = np.clip(h, 0, 1)
        
        # Normalize again
        integral = np.sum(h) * 2.0 / n_points
        h = h * (1.0 / integral)
        
        # Ensure sum constraint
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Adjust learning rate
        if iteration % decay_every == 0 and iteration > 0:
            learning_rate *= 0.9
    
    return (best_h.tolist(), best_c5, n_points)
