import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values, dtype=float)
    except NameError:
        # Create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.6
        h[1] = 0.4
        h[-1] = 0.6
        h[-2] = 0.4
        h = h / np.sum(h) * (n_points / 2)
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # Compute the overlap integral
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def is_valid(h_vals):
        # Check constraints
        if np.any(h_vals < 0) or np.any(h_vals > 1):
            return False
        if abs(np.sum(h_vals) - n_points / 2) > 1e-10:
            return False
        return True
    
    def update_h(h_vals, i, j, delta):
        # Update h_vals by moving delta from i to j
        new_h = h_vals.copy()
        if new_h[i] >= delta and new_h[j] + delta <= 1:
            new_h[i] -= delta
            new_h[j] += delta
            return new_h
        return None
    
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    delta = 0.1
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Perturb two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        if i == j:
            continue
            
        # Try to update
        new_h = update_h(h, i, j, delta)
        if new_h is None:
            if iter_count % 10 == 0:
                delta *= 0.9
            continue
            
        new_C5 = compute_C5(new_h)
        if new_C5 < current_C5:
            h = new_h
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
        else:
            # Occasionally accept worse solutions to escape local minima
            if np.random.rand() < 0.01:
                h = new_h
                current_C5 = new_C5
        
        # Adaptive delta
        if iter_count % 100 == 0:
            delta *= 0.9
            
    return (best_h.tolist(), best_C5, n_points)
