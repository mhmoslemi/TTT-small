import numpy as np
import time
from scipy.signal import correlate
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5)
    
    # Normalize to sum constraint
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate_c5(h_vec):
        # Compute all overlaps
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = corr[n_points-1:]
        return np.max(overlaps) * dx
    
    def project_h(h_vec):
        # Clip to [0, 1] and rescale to sum to target_sum
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def compute_gradient(h_vec, k_max):
        # Compute gradient for shift k_max
        shifted_h = np.roll(1 - h_vec, k_max)
        grad = shifted_h - h_vec * shifted_h
        return grad
    
    def backtracking_line_search(h_vec, grad, k_max, alpha_init=0.01, rho=0.5, max_steps=30):
        alpha = alpha_init
        current_c5 = evaluate_c5(h_vec)
        
        for _ in range(max_steps):
            new_h = h_vec - alpha * grad
            new_h = project_h(new_h)
            new_c5 = evaluate_c5(new_h)
            if new_c5 < current_c5:
                return alpha, new_h
            alpha *= rho
        return alpha, h_vec
    
    # Multi-initialization
    initial_points = [h.copy()]
    for _ in range(4):
        noise = np.random.normal(0, 0.005, n_points)
        perturbed = h + noise
        perturbed = project_h(perturbed)
        initial_points.append(perturbed)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for h_init in initial_points:
        h = h_init.copy()
        h = project_h(h)
        
        for iter_count in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute overlaps and find k_max
            corr = correlate(h, 1 - h, mode='full')
            overlaps = corr[n_points-1:]
            k_max = np.argmax(overlaps)
            
            # Compute gradient
            grad = compute_gradient(h, k_max)
            grad_norm = np.linalg.norm(grad)
            
            if grad_norm > 1e-10:
                grad = grad / grad_norm
            
            # Line search
            alpha, new_h = backtracking_line_search(h, grad, k_max)
            h = new_h
            
            # Convex refinement every 50 steps
            if iter_count % 50 == 0:
                # Use CVXPY to solve convex subproblem
                h_var = cp.Variable(n_points)
                constraints = [0 <= h_var, h_var <= 1, cp.sum(h_var) == target_sum]
                
                # Sliding window over overlapping indices
                objective = cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, k_max)))
                prob = cp.Problem(cp.Minimize(objective), constraints)
                
                try:
                    prob.solve(solver=cp.GLPK, verbose=False)
                    if prob.status in [cp.OPTIMAL, cp.OPTIMIZED_INFEASIBLE]:
                        h = h_var.value
                        h = project_h(h)
                except:
                    pass
            
            # Update best solution
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
