def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all shifts and return max
    h = np.array(h)
    n_points = len(h)
    overlaps = []
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            if i + k < n_points:
                overlap += h[i] * (1 - h[i + k]) * dx
        overlaps.append(overlap)
    return max(overlaps)
