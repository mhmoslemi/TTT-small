import numpy as np
from scipy import optimize
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Initial construction from the paper 
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    start_time = time.time()
    
    def evaluate_c5(h_vals):
        # Compute C5 using numpy's correlate (full mode)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(corr)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0
    
    def project_h(h):
        # Project onto [0,1] and rescale to sum = n_points/2
        h = np.clip(h, 0, 1)
        s = np.sum(h)
        if s > 0:
            h = h * (n_points / 2.0) / s
        return h
    
    # Convex relaxation helper
    def convex_relaxation_step(h_init, n):
        # Simple convex envelope solver (using cvxpy)
        h = cp.Variable(n)
        obj = cp.Minimize(cp.sum(cp.multiply(h, h))) + 1e-8  # Dummy objective for feasibility
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n/2.0
        ]
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                return np.maximum(0, np.minimum(1, h_sol))
        except:
            pass
        return h_init
    
    # Gradient-based refinement
    def gradient_refine(h_orig, n_iter=100, learning_rate=0.1):
        h = h_orig.copy()
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        
        for i in range(n_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute all correlations and find max
            corr = np.correlate(h, 1 - h, mode='full')
            max_idx = np.argmax(corr)
            max_k = max_idx - (len(h) - 1)  # lag index
            
            # Compute gradient w.r.t. each h[i] for this specific lag
            grad = np.zeros_like(h)
            for idx in range(len(h)):
                # Gradient wrt h[idx] matches the corr kernel (for specific lag)
                if 0 <= idx + max_k < len(h):
                    grad[idx] = -1 * (1 - h[idx + max_k]) if max_k >= 0 else -1 * h[idx - max_k]
                if 0 <= idx - max_k < len(h):
                    grad[idx] += -1 * h[idx - max_k] if max_k <= 0 else -1 * (1 - h[idx + max_k])
            
            # Update
            h_new = h - learning_rate * grad
            h_new = project_h(h_new)
            
            # Check improvement
            new_c5 = evaluate_c5(h_new)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_new.copy()
                h = h_new.copy()
            else:
                learning_rate *= 0.5  # Reduce learning rate if no improvement
                if learning_rate < 1e-6:
                    break
                    
        return best_h
    
    h_current = h0.copy()
    c5_min = evaluate_c5(h_current)
    
    # Stage 1: convex relaxation
    if time.time() - start_time < budget_s / 2:
        h_proj = convex_relaxation_step(h_current, n_points)
        h_proj = project_h(h_proj)
        h_current = h_proj
        c5_min = evaluate_c5(h_current)
        
    # Stage 2: gradient descent refinement
    refined = gradient_refine(h_current, 100, 0.1)  
    c5_refined = evaluate_c5(refined)
    
    if c5_refined < c5_min:
        h_current = refined
        c5_min = c5_refined
        
    # Stage 3: Upgrade to higher resolution if time permits
    if time.time() - start_time < budget_s - 5:
        n_high = n_points * 2
        dx_high = 2.0 / n_high
        h_high = np.interp(np.linspace(0, 2, n_high, endpoint=False), 
                          np.linspace(0, 2, n_points, endpoint=False), 
                          h_current)
        h_high = project_h(h_high)
        c5_high = evaluate_c5(h_high)
        if c5_high < c5_min:
            h_current = h_high
            c5_min = c5_high
            
        # Refine high resolution
        h_fine = gradient_refine(h_high, 100, 0.1)
        c5_fine = evaluate_c5(h_fine)
        if c5_fine < c5_min:
            h_current = h_fine
            c5_min = c5_fine
            
    return (h_current, c5_min, len(h_current))
