import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def evaluate_c5(h, dx):
    # Compute overlap integral for all lags
    conv = np.correlate(h, 1 - h, mode="full")
    max_conv = np.max(conv)
    return max_conv * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 42
    dx = 2.0 / n_points
    
    # Initial solution from parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create basic step function
        h = np.ones(n_points) * 0.5
        # Normalize to have integral = 1
        h = h / np.sum(h) * (n_points * dx / 2.0)
    
    # Time tracking
    start_time = time.time()
    
    # Define a continuous optimization using scipy
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return evaluate_c5(h_vec, dx)
    
    # Constraint function for sum(h) = 1
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2.0
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop with time limit
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Define a simple local search for improvement
    current_h = h.copy()
    current_c5 = evaluate_c5(current_h, dx)
    
    # Iterative improvement with gradient descent type approach
    for iteration in range(5000):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Basic hill climbing neighborhood
        # Perturb one element at a time
        i = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.01, 0.01)
        new_h = current_h.copy()
        new_h[i] = np.clip(new_h[i] + delta, 0, 1)
        
        # Enforce the sum constraint
        current_sum = np.sum(current_h)
        new_sum = np.sum(new_h)
        scale_factor = current_sum / new_sum if new_sum > 0 else 1.0
        new_h = new_h * scale_factor
        
        # Make sure it's still normalized
        new_h = new_h * (n_points / 2.0) / np.sum(new_h)
        
        # Evaluate and accept with probability proportional to improvement
        new_c5 = evaluate_c5(new_h, dx)
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return (best_h, best_c5, n_points)
