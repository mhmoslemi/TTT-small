import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 100
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * target_sum
        population.append(h0)
    
    # Add random feasible solutions
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
        # Ensure constraints 0 <= h[i] <= 1
        h = np.clip(h, 0, 1)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        # Compute overlaps
        corr = signal.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        fitnesses.append(1.0 / (max_overlap + 1e-12))  # Avoid division by zero
    
    best_h = population[np.argmax(fitnesses)]
    best_c5 = 1.0 / np.max(fitnesses)
    
    # Genetic Algorithm
    generations = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Selection: keep top 50%
        sorted_indices = np.argsort(fitnesses)[::-1]
        top_indices = sorted_indices[:population_size // 2]
        selected_population = [population[i] for i in top_indices]
        selected_fitnesses = [fitnesses[i] for i in top_indices]
        
        # Create new population
        new_population = selected_population[:]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_population, 2, replace=False)
            
            # Crossover: uniform random exchange
            mask = np.random.rand(n_points) > 0.5
            child1 = np.where(mask, parent1, parent2)
            child2 = np.where(mask, parent2, parent1)
            
            # Mutation: 1% of values changed
            for child in [child1, child2]:
                for i in np.random.choice(n_points, int(0.01 * n_points), replace=False):
                    child[i] = np.random.rand()
                
                # Ensure feasibility
                child_sum = np.sum(child)
                child = child / child_sum * target_sum
                child = np.clip(child, 0, 1)
                
                new_population.append(child)
                
                # Early termination if time is running out
                if time.time() - start_time >= budget_s - 1:
                    break
                    
            if time.time() - start_time >= budget_s - 1:
                break
        
        # Ensure exact population size
        new_population = new_population[:population_size]
        
        # Evaluate new population
        new_fitnesses = []
        for h in new_population:
            corr = signal.correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            fitnesses.append(1.0 / (max_overlap + 1e-12))
            new_fitnesses.append(1.0 / (max_overlap + 1e-12))
            
            # Update best solution
            if max_overlap < best_c5:
                best_h = h.copy()
                best_c5 = max_overlap
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        population = new_population
        fitnesses = new_fitnesses
        generations += 1
        
        if time.time() - start_time >= budget_s - 1:
            break
    
    # Final evaluation of best solution
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
