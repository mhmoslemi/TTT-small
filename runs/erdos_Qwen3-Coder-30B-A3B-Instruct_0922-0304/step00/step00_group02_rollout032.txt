from scipy.signal import correlate

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Only take the valid lags: from -n+1 to n-1
    # But we're looking at the "full" correlation, so for a periodic signal:
    # corr[k] = sum(h[i] * (1 - h[(i+k)%n])) for i=0..n-1
    # This corresponds to lag k, where k is from 0 to 2*n-2
    # To get "shifts" k from -n+1 to n-1, indexing should be 0 to 2*n-2
    # The actual meaningful overlaps are the middle entries.
    mid = len(corr) // 2
    max_overlap = max(corr[mid - (n-1) : mid + n])
    return max_overlap * dx
