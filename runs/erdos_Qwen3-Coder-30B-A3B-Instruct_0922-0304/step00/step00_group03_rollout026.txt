        h = np.array(h)
        # Compute convolution with (1 - h)
        conv = np.correlate(h, 1 - h, mode='full')
        # Return maximum value scaled by dx
        return np.max(conv) * dx
    
    def project(h):
        """Project h to satisfy constraints"""
        # Clamp to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to make sum equal to n_points / 2 (since dx = 2/n_points)
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-10:
            # Simple projection: adjust by a constant
            adjustment = (target_sum - current_sum) / n_points
            h = np.clip(h + adjustment, 0, 1)
        return h
    
    def gradient_descent_step(h, step_size=1e-3):
        """Perform one gradient descent step"""
        C5 = compute_C5(h)
        # Find which shift gives maximum overlap
        h_arr = np.array(h)
        corr = np.correlate(h_arr, 1 - h_arr, mode='full')
        max_k = np.argmax(corr)
        # Gradient vector for the specific shift:
        grad = np.zeros_like(h)
        n = len(h)
        # For shift k, the gradient is the derivative of sum_i h[i] * (1 - h[i+k])
        # Derivative w.r.t. h[i] is (1 - h[i+k]) if k >= 0 and i+k < n
        # But need to map back properly
        # For simplicity, compute gradient of active constraint
        active_shift = max_k - (n - 1)  # Convert full convolution index to shift
        for i in range(n):
            k = active_shift
            if 0 <= i - k < n:
                grad[i] += 1 - h[i - k]
            else:
                # out of bounds; nothing to do for this term
                pass
                
        # Update
        h_new = h - step_size * grad
        # Project back
        h_new = project(h_new)
        return h_new
