import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap for all lags
    overlap = correlate(h, 1 - h, mode='full')
    # Get only the positive lags (first n-1 elements)
    overlap = overlap[n-1:]
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def get_peaks(overlap_vals):
    """Get indices of top peaks in descending order"""
    peak_indices = np.argsort(overlap_vals)[::-1]
    return peak_indices

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Default to match initial construction
    max_iter = 10000
    alpha = 0.05  # Update step size
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        current_c5 = evaluate_h(h)
        
        if current_c5 < 0.3807:
            break
            
        # Compute full overlap to find peaks
        dx = 2.0 / n_points
        overlap_full = correlate(h, 1 - h, mode='full')
        overlap_positive = overlap_full[n_points-1:]
        
        # Get top 5 peaks
        peak_indices = get_peaks(overlap_positive)[:5]
        
        # Create new h by reducing peak overlaps
        new_h = h.copy()
        for idx in peak_indices:
            if idx >= n_points: continue
            # Typical approach: reduce h in areas that contribute most to overlap
            # Search around index idx for local influence
            for i in range(max(0, idx-3), min(n_points, idx+4)):
                if i < len(new_h):
                    new_h[i] -= alpha * 0.1  # reduce
        new_h = project_h(new_h, n_points)
        
        new_c5 = evaluate_h(new_h)
        if new_c5 < current_c5:
            h = new_h
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n_points)
