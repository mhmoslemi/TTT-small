import numpy as np
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Set up discretization
    n_points = 45  # Use the initial construction size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Normalize to integral = 1
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Optimize
    alpha = 0.1
    last_improvement = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    while time() - start_time < budget_s - 0.1:
        # Compute overlaps for all shifts
        h_ext = np.concatenate([h, h])  # double for easier convolution
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            # Extract h[i] * (1 - h[i+k])
            if k >= 0:
                valid_h = h[:n_points-k]
                valid_hk = h_ext[k:k+n_points-k]
            else:
                valid_h = h[-k:]
                valid_hk = h_ext[-k:2*n_points+k]
            overlap = np.sum(valid_h * (1 - valid_hk)) * dx
            overlaps[k + n_points - 1] = overlap

        # Find max overlap shift
        max_k = np.argmax(overlaps)
        k_max = max_k - (n_points - 1)
        
        # Compute gradient for this k
        grad = np.zeros(n_points)
        if k_max >= 0:
            for i in range(n_points - k_max):
                grad[i] += (1 - h[i + k_max]) * dx
                grad[i + k_max] -= h[i] * dx
        else:
            for i in range(-k_max, n_points):
                grad[i] += (1 - h[i + k_max]) * dx
                grad[i + k_max] -= h[i] * dx
        
        # Update h
        h = h - alpha * grad
        
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        
        # Evaluate current C5
        c5_current = np.max(overlaps)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        # Early termination
        if c5_current < 0.3807 or last_improvement > 50:
            break
    
    # Final validation
    final_h = best_h.copy()
    final_c5 = best_c5
    
    return (final_h.tolist(), final_c5, n_points)
