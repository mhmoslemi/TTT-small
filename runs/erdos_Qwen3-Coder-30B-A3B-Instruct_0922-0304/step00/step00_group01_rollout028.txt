import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h"""
    overlap = correlate(h, 1-h, mode="full")
    c5 = np.max(overlap[len(h)-1:]) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise initialize
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    iter_count = 0
    no_improve_count = 0
    max_no_improve = 100
    
    while time.time() - start_time < budget_s - 1 and no_improve_count < max_no_improve:
        iter_count += 1
        
        # Compute overlap
        overlap = correlate(h, 1-h, mode="full")
        overlap_vals = overlap[len(h)-1:]  # Upper half
        
        # Identify peaks by sorting in descending order
        peak_indices = np.argsort(overlap_vals)[::-1][:5]  # Top 5 peaks
        
        # Create a modified version of h
        h_new = h.copy()
        update_magnitude = 0.05
        
        # Modify h based on peak locations
        for k in peak_indices:
            # Target indices for this peak
            for i in range(n_points):
                idx_shifted = i + k
                if idx_shifted < n_points:
                    # Reduce h[i] to decrease overlap
                    h_new[i] = max(0, h_new[i] - update_magnitude)
                    # Also adjust nearby points, avoid too aggressive changes
                    if i > 0:
                        h_new[i-1] = min(1, h_new[i-1] + update_magnitude/2)
                    if idx_shifted < n_points - 1:
                        h_new[idx_shifted+1] = min(1, h_new[idx_shifted+1] + update_magnitude/2)
        
        # Ensure constraints
        h_new = project_h(h_new, n_points)
        
        # Check improvement
        new_c5 = evaluate_c5(h_new, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improve_count = 0  # Reset counter on improvement
        else:
            no_improve_count += 1
            
        h = h_new
        
        if best_c5 < target_c5:
            break
            
    return (best_h, best_c5, n_points)
