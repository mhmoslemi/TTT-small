import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Binary construction with 1s in first half
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Adjust to ensure sum = n_points / 2 (22.5)
    h = h * (22.5 / np.sum(h))
    
    # Time tracking
    start_time = time.time()
    
    # Gradient-based optimization
    step_size = 0.1
    min_step_size = 1e-6
    decay_factor = 0.95
    max_iter = 10000
    no_improvement_count = 0
    max_no_improvement = 100
    perturbation_freq = 500
    perturbation_magnitude = 0.05
    
    def compute_c5(h_vec):
        # Using a full convolution
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Take only valid shifts (from -N+1 to N-1), but only up to N-1
        # Properly indexed to get the right indexing
        # For overlap at shift k: k from 0 to 2*N-2 but we're just interested
        # in the circular convolution that gives max overlap = overlay of h with 1-h
        # Maximum overlap as convolution values
        max_overlap = np.max(conv) * 2.0 / len(h_vec)
        return max_overlap
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 0.1:
            break
        
        # Compute overlaps and find maximum
        h_convol = correlate(h, 1 - h, mode='full')
        full_overlaps = h_convol * (2.0 / len(h))
        
        # Find the shift that gives max overlap (also the max value in conv)
        max_idx = np.argmax(full_overlaps)
        max_overlap = np.max(full_overlaps)
        
        # Compute gradient for this specific shift
        # Gradient of the term for shift k is: 
        # 1 - h[(i+k)%N], summed over all valid k: we just work with one k
        # That corner case where the gradient of overlap w.r.t. h_i is:
        # For shift k, ∂O_k/∂h_i = (1 - h[(i+k)%N]) because 
        # O_k = sum_i h[i] * (1 - h[(i+k)%N])
        # ∂O_k/∂h_i = (1 - h[(i+k)%N]) when i is in the sum
        # This is the simplest interpretation for the gradient WRT h_i
        # Let's compute ∂C5 / ∂h from the active overlap which is a_max
        # Gradient of the single term (slope) for minimizing max O_k:
        # We take the index shift active, and compute gradient 
        # based on the max overlap +
        # Assume C5 = max_k O_k, so I take the maximum to compute the gradient for the active shift.
        # At point i in determination of gradient for max k:
        # d(O_k)/d(h_i) = (1 - h[(i+k)%N]) if i < N-k, and  
        # (1 - h[(i+shift)%N]) otherwise
        max_shift = max_idx - (len(h) - 1)
        
        # Actually, let's iterate over the oposite logic
        # 0..N-1 for each x, which corresponds to conv applying at k (shift k)
        # Actually math for the kernel: 
        # h * 1-h shift k => int h(x) (1 - h(x+k)) dx, so gradient is
        # for the shift k that gives the max, we compute gradient 
        # at h[i] of the overlap for shift k
        # = (1 - h[(i+k)%N]) 
        # We'll recompute all gradients to make sure we stick to accurate math
        # So compute for shift corresponding to max overlap
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i + max_shift) % len(h)
            grad[i] = (1.0 - h[j])
            
        # Update
        h_temp = h - step_size * grad
        
        # Project onto feasible set
        clipped = np.clip(h_temp, 0, 1)
        scale = 22.5 / np.sum(clipped)
        h = clipped * scale
        
        # Evaluate new overlap
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Dynamic step size
        if no_improvement_count % 100 == 0 and step_size > min_step_size:
            step_size *= decay_factor
            
        # Perturbation to escape local minima
        if iteration % perturbation_freq == 0 and iteration > 0:
            mask = np.random.rand(len(h)) < perturbation_magnitude
            h[mask] += np.random.normal(0, 0.01, np.sum(mask))
            # Re-project after perturbations
            clipped = np.clip(h, 0, 1)
            scale = 22.5 / np.sum(clipped)
            h = clipped * scale
            
        # Early termination
        if best_c5 <= 0.3807:
            break
        
        if no_improvement_count > max_no_improvement:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), float(final_c5), len(best_h))
