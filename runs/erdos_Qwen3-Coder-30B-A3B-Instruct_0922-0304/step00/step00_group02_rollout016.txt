import numpy as np
import time
from scipy import correlate

def evaluate_solution(h):
    """Evaluate the C5 bound for a given step function h"""
    dx = 2.0 / len(h)
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode="full")
    # The max overlap is the maximum of all these values
    c5 = np.max(overlaps) * dx
    return c5

def is_feasible(h):
    """Check if h satisfies constraints: 0 <= h <= 1, sum = 1"""
    n = len(h)
    dx = 2.0 / n
    return (np.all(h >= 0) and np.all(h <= 1) and
            abs(np.sum(h) * dx - 1.0) < 1e-10)

def initialize_population(pop_size, n_points, initial_h_values):
    """Initialize population with a mix of initial and random solutions"""
    population = []
    # Add initial solution
    if initial_h_values is not None and len(initial_h_values) == n_points:
        population.append(np.array(initial_h_values))
    else:
        # Default initialization if no initial provided
        h = np.ones(n_points) * 0.5
        population.append(h)

    # Add random solutions
    for i in range(pop_size - 1):
        h = np.random.rand(n_points)
        # Normalize to satisfy sum constraint
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def mutate(h, mutation_rate=0.01):
    """Mutate a solution by slightly adjusting values"""
    n = len(h)
    h_new = h.copy()
    mutations = int(n * mutation_rate)
    indices = np.random.choice(n, size=mutations, replace=False)
    for idx in indices:
        # Add small random change
        h_new[idx] += np.random.normal(0, 0.05)
        # Clip to [0,1]
        h_new[idx] = np.clip(h_new[idx], 0, 1)
    # Adjust sum to maintain constraint
    dx = 2.0 / n
    total = np.sum(h_new) * dx
    if abs(total - 1.0) > 1e-10:
        # Scale to fix sum
        scale_factor = 1.0 / total
        h_new = h_new * scale_factor
    return h_new

def crossover(parent1, parent2):
    """Create child from two parents by randomly swapping segments"""
    n = len(parent1)
    # Choose crossover point
    cp = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:cp], parent2[cp:]])
    child2 = np.concatenate([parent2[:cp], parent1[cp:]])
    # Normalize sum to meet constraint
    dx = 2.0 / n
    for child in [child1, child2]:
        total = np.sum(child) * dx
        if abs(total - 1.0) > 1e-10:
            # Scale to fix sum
            scale_factor = 1.0 / total
            child[:] = child * scale_factor
    return child1, child2

def run(seed=42, budget_s=600, **kwargs):
    """
    Run genetic algorithm to optimize h for minimizing C5
    Returns (h_values, c5_bound, n_points)
    """
    np.random.seed(seed)
    
    n_points = 45
    pop_size = 100
    max_generations = 1000
    patience = 100
    mutation_rate = 0.01
    
    # Initialize population
    population = initialize_population(pop_size, n_points, initial_h_values)
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    generations_without_improvement = 0
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all individuals
        fitness_scores = []
        for h in population:
            if not is_feasible(h):
                # Reinitialize infeasible individuals
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                population[population.index(h)] = h
            c5 = evaluate_solution(h)
            fitness_scores.append(1.0 / (c5 + 1e-15))  # Avoid division by zero
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)[::-1]
        population = [population[i] for i in sorted_indices]
        
        # Track best
        current_best_c5 = evaluate_solution(population[0])
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
            
        if generations_without_improvement > patience:
            break
            
        # Selection: top half become parents
        selected = population[:pop_size // 2]
        
        # Create new population
        new_population = selected[:]
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Select two parents
            p1 = selected[np.random.randint(0, len(selected))]
            p2 = selected[np.random.randint(0, len(selected))]
            
            # Crossover
            c1, c2 = crossover(p1, p2)
            
            # Mutate
            c1 = mutate(c1, mutation_rate)
            c2 = mutate(c2, mutation_rate)
            
            new_population.extend([c1, c2])
            
        # Trim to pop_size
        population = new_population[:pop_size]
    
    if best_h is None:
        # If no improvement, use the best from initial run
        best_h = population[0]
        best_c5 = evaluate_solution(best_h)
        
    return (best_h, best_c5, n_points)
