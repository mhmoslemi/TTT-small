def evaluate_erdos_solution(h, dx):

    n = len(h)
    # Create the flipped version of h
    flipped = 1 - h
    # Compute correlation with mode="full"
    corr = np.correlate(h, flipped, mode="full")
    # The correlation values are indexed from -(n-1) to (n-1)
    # We want the maximum value among those with 0 <= k <= n-1 (for the shift)
    # Since we're correlating h with (1-h), correlation for shift k is at index n-1+k
    # So k ranges from 0 to n-1, which corresponds to indices n-1 to 2*n-2
    max_corr = np.max(corr[n-1:2*n-1])
    return max_corr * dx
