import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_bound = 0.3807

        h[indices] = 1.0
    
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)  # Ensure sum is exactly n_points/2
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 10000
    inner_iter = 100
    cooldown_iter = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        old_h = h.copy()
        
        # For each position, compute how many overlaps it contributes to
        # and adjust based on minimizing the maximum overlap
        # This is simplified to a binary greedy approach
        
        # Binary approximation: update to 0 or 1 if improving overlap
        for i in range(n_points):
            if time.time() - start_time > budget_s - 1:
                break
                
            old_val = h[i]
            
            # Try 0 or 1 for this position to see which gives better overlap
            current_sum = np.sum(h)
            
            # Try making it 0
            h[i] = 0.0
            # Compute new C5
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr) * dx
            c5_0 = max_corr
            
            # Try making it 1
            h[i] = 1.0
            corr = correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr) * dx
            c5_1 = max_corr
            
            # Restore old value
            h[i] = old_val
            
            # Choose new value to minimize C5
            if c5_0 <= c5_1:
                new_val = 0.0
            else:
                new_val = 1.0
                
            h[i] = new_val
            
            # Rebalance to maintain sum = n_points/2
            current_sum = np.sum(h)
            if abs(current_sum - n_points / 2) > 1e-6:
                # If we're over or under, adjust
                diff = current_sum - n_points / 2
                if diff > 0:
                    # Need to reduce
                    indices = np.argsort(h)[::-1]  # descending
                    decrease_count = int(np.ceil(abs(diff)))
                    for k in range(decrease_count):
                        if h[indices[k]] > 0:
                            h[indices[k]] -= 1e-10
                elif diff < 0:
                    # Need to increase
                    indices = np.argsort(h)  # ascending
                    increase_count = int(np.ceil(abs(diff)))
                    for k in range(increase_count):
                        if h[indices[k]] < 1:
                            h[indices[k]] += 1e-10
                
                # Regulate to [0,1]
                h = np.clip(h, 0, 1)
                # Remake sum
                scale = (n_points / 2) / np.sum(h)
                if abs(scale - 1) > 1e-10:
                    h = h * scale
                h = np.clip(h, 0, 1)
        
        # Objective check and update best
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Check stopping criteria
        if c5 < target_bound:
            break
                
    # Final evaluation and normalization
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / h_sum
    h = np.clip(h, 0, 1)
    
    # Final result
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        c5 = best_c5
        h = best_h.copy()
    
    return (h.tolist(), float(c5), n_points)
