import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Ensure sum is exactly n_points / 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient descent
    alpha = 0.01
    beta = 0.9  # momentum
    v = np.zeros_like(h)
    
    iteration = 0
    while time.time() - start_time < budget_s:
        if iteration % 100 == 0 and iteration > 0:
            alpha *= 0.1
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        
        # Max shift index
        max_idx = np.argmax(corr_full)
        max_k = max_idx - (len(h) - 1)
        max_overlap = corr_full[max_idx]
        
        if max_overlap <= best_c5:
            best_h = h.copy()
            best_c5 = max_overlap
        
        if best_c5 <= target_bound:
            break
            
        # Compute gradient w.r.t h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            for k in range(-len(h)+1, len(h)):
                if 0 <= i + k < len(h):
                    grad[i] += (1 - h[i + k])
        
        # Update using momentum
        v = beta * v + (1 - beta) * grad
        h_new = h - alpha * v
        
        # Clip to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_sum_new = np.sum(h_new)
        if h_sum_new > 0:
            h_new = h_new * (n_points / 2) / h_sum_new
            
        h = h_new
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h):
    dx = 2.0 / len(h)
    corr_full = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr_full)
    return max_overlap * dx
