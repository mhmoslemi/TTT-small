import numpy as np
from scipy import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 100000
    timeout = time.time() + budget_s
    
    # Initialize h from the given initial construction or create a new one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create an initial step function
        h = np.ones(n_points) * 0.5

    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    delta = 0.1
    decay = 0.9
    decay_interval = 100
    
    for i in range(max_iter):
        if time.time() > timeout:
            break
            
        # Perturb two random indices
        idx1 = np.random.randint(n_points)
        idx2 = np.random.randint(n_points)
        
        # Make sure indices are not the same
        if idx1 == idx2:
            continue
        
        # Try small perturbation
        perturb = np.random.uniform(-delta, delta)
        
        new_h = h.copy()
        new_h[idx1] += perturb
        new_h[idx2] -= perturb
        
        # Ensure bounds
        if new_h[idx1] < 0 or new_h[idx1] > 1 or new_h[idx2] < 0 or new_h[idx2] > 1:
            continue

        # Ensure sum constraint
        if abs(np.sum(new_h) - n_points / 2) > 1e-8:
            # Adjust for normalization
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute new C5
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept improvement
        if new_c5 < current_c5:
            h = new_h
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
                
        # Decay delta every few iterations
        if (i + 1) % decay_interval == 0:
            delta *= decay
        
    return (best_h.tolist(), best_c5, n_points)
