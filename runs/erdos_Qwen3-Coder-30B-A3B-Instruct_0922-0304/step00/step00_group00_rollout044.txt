def evaluate_erdos_solution(h):
    """Evaluate the overlap integral C5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    max_overlap = 0
    for k in range(-n+1, n):
        overlap = 0
        for i in range(n):
            j = (i + k) % n
            overlap += h[i] * (1 - h[j])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap
