import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a symmetric starting function
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Slightly perturb to avoid degeneracy
        h += np.random.normal(0, 0.01, n_points)
    
    # Ensure sum constraint
    h *= (n_points / 2) / np.sum(h)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode='full')
        overlaps = overlaps[len(h)-1:]  # Only the valid shifts
        k_max = np.argmax(overlaps)
        C5_max = overlaps[k_max]
        
        if C5_max <= 0.3807:
            break
            
        # Convex relaxation to minimize the maximum overlap
        # We optimize to minimize C(k_max)
        def objective(h_new):
            # The overlap is the dot product of h_new and 1 - h_shifted_kmax
            h_shifted = np.roll(h_new, k_max)
            return np.dot(h_new, 1 - h_shifted)
        
        # Constraints
        cons = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2},
            {'type': 'ineq', 'fun': lambda x: x},
            {'type': 'ineq', 'fun': lambda x: 1 - x}
        ]
        
        # Bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        res = minimize(objective, h.copy(), method='SLSQP', bounds=bounds, constraints=cons, 
                       options={'disp': False, 'maxiter': 100})
        
        if res.success:
            h_new = res.x
            h_new = np.clip(h_new, 0, 1)
            # Renormalize to sum to n_points / 2
            h_new *= (n_points / 2) / np.sum(h_new)
            h = h_new
                
            current_C5 = evaluate_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        else:
            # Fallback: small perturbation and projection
            h += np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            
            current_C5 = evaluate_C5(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
                
    # Final evaluation
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), int(n_points))

def evaluate_C5(h):
    # Compute the maximum overlap
    h = np.array(h)
    overlaps = np.correlate(h, 1 - h, mode='full')
    overlaps = overlaps[len(h)-1:]  # Only valid shifts
    return np.max(overlaps)
