import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: create a simple initial guess
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    # Ensure constraint sum(h) * dx = 1 <=> sum(h) = n_points / 2
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Start timing
    start_time = time.time()
    
    # Optimization parameters
    max_iter = 100000
    delta = 0.1
    delta_decay = 0.9
    delta_decay_interval = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb two random indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Calculate variation
        h_new = h.copy()
        variation = delta * (np.random.rand() - 0.5)
        
        # Adjust h[i] and h[j]
        if h[i] + variation > 1:
            variation = 1 - h[i]
        elif h[i] + variation < 0:
            variation = -h[i]
            
        if h[j] - variation > 1:
            variation = h[j] - 1
        elif h[j] - variation < 0:
            variation = h[j]
            
        if abs(variation) < 1e-10:
            continue
            
        h_new[i] += variation
        h_new[j] -= variation
        
        # Maintain sum constraint
        if i != j:
            # Adjust one more variable to maintain exact sum
            idx_to_adjust = (i + 1) % n_points
            h_new[idx_to_adjust] += (target_sum - np.sum(h_new)) / (n_points - 1)
        
        # Ensure all elements in [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Recompute sum to ensure accuracy
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new, n_points)
        
        # Accept or reject
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            
        # Adaptive delta
        if iteration % delta_decay_interval == 0:
            delta *= delta_decay
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, n_points):
    # Evaluate C5 = max_k int h(x)(1-h(x+k)) dx
    dx = 2.0 / n_points
    h_complement = 1.0 - h
    
    # Compute full correlation (cross-correlation)
    correlation = signal.correlate(h, h_complement, mode='full')
    
    # Normalize and find maximum
    max_corr = np.max(correlation) * dx
    
    return max_corr
