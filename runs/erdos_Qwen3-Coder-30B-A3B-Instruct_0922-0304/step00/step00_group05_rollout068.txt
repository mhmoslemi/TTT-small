import numpy as np
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (shifts from -n+1 to n-1)
    n = len(h)
    corr = corr[n-1:2*n-1]  # Focus on valid shifts
    return np.max(corr) * dx

def project_to_feasible(h, dx):
    """Project h to satisfy sum(h) * dx = 1 and 0 <= h[i] <= 1."""
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        scale = 1.0 / current_sum
        h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5

    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    if 'initial_h_values' in globals():
        h0 = initial_h_values.copy()
        h0 = project_to_feasible(h0, dx)
        swarm.append(h0)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        personal_best.append(h0.copy())
        personal_best_values.append(evaluate_h(h0, dx))
    else:
        h0 = np.random.rand(n_points)
        h0 = project_to_feasible(h0, dx)
        swarm.append(h0)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        personal_best.append(h0.copy())
        personal_best_values.append(evaluate_h(h0, dx))
    
    # Initialize remaining particles
    for i in range(1, swarm_size):
        h = np.random.rand(n_points)
        h = project_to_feasible(h, dx)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
        personal_best.append(h.copy())
        personal_best_values.append(evaluate_h(h, dx))

    # Initialize global best
    best_idx = np.argmin(personal_best_values)
    global_best = personal_best[best_idx].copy()
    global_best_value = personal_best_values[best_idx]

    # PSO main loop
    for iteration in range(max_iter):
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time < 1.0:
            break
            
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive_component = cognitive * r1 * (personal_best[i] - h)
            social_component = social * r2 * (global_best - h)
            v = inertia * v + cognitive_component + social_component
            
            # Update position
            new_h = h + v
            # Ensure it stays within bounds before projection
            new_h = np.clip(new_h, 0, 1)
            
            # Project to feasibility (enforce sum constraint)
            new_h = project_to_feasible(new_h, dx)
            
            # Evaluate new solution
            new_value = evaluate_h(new_h, dx)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best[i] = new_h.copy()
                personal_best_values[i] = new_value
            
            # Update global best
            if new_value < global_best_value:
                global_best = new_h.copy()
                global_best_value = new_value
            
            # Update swarm and velocity
            swarm[i] = new_h
            velocities[i] = v
        
        # Occasionally perturb velocity to escape local optima
        if iteration % 100 == 0 and iteration > 0:
            for i in range(swarm_size):
                if np.random.rand() < 0.2:
                    velocities[i] = np.random.uniform(-0.5, 0.5, n_points)

    # Final evaluation of global best
    final_value = evaluate_h(global_best, dx)
    
    return (global_best, final_value, n_points)
