import numpy as np
import cvxpy as cp
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target = 0.3807
    
    # Normalize to integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate_C5(h):
        h_ext = np.concatenate([h, h])
        overlaps = np.array([np.sum(h_ext[i:i+n_points] * (1 - h)) for i in range(n_points)])
        return np.max(overlaps) * dx

    def solve_with_constraints(h, max_overlaps, n_points, dx):
        h_vars = cp.Variable(n_points)
        constraints = []
        
        # Constraint: 0 <= h[i] <= 1
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        
        # Constraint: sum(h) = n_points/2
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Add McCormick constraints for products
        for k, max_overlap in enumerate(max_overlaps):
            if max_overlap is not None:
                # We approximate max_k integral with McCormick
                # h[i] * (1 - h[i+k]) = h[i] - h[i]*h[i+k]
                # Use McCormick envelopes: let z[i] = h[i]*h[i+k] 
                # then we want to maximize sum h[i] - z[i]
                # z[i] >= h[i] + h[i+k] - 1
                # z[i] >= 0 
                # z[i] <= h[i]
                # z[i] <= h[i+k]
                # But that's too conservative for optimization.
                # For a specific k, we add a hard constraint on this overlap
                z = cp.Variable(n_points)
                constraints.extend([z[i] >= h_vars[i] + h_vars[(i+k)%n_points] - 1 for i in range(n_points)])
                constraints.extend([z[i] >= 0 for i in range(n_points)])
                constraints.extend([z[i] <= h_vars[i] for i in range(n_points)])
                constraints.extend([z[i] <= h_vars[(i+k)%n_points] for i in range(n_points)])
                
                # And we bound the total overlap
                total_overlap = cp.sum(h_vars) - cp.sum(z)
                constraints.append(total_overlap <= max_overlap * n_points / 2)
        
        objective = cp.Minimize(cp.sum(h_vars))  # This is a dummy - we just want to satisfy constraints
        
        # Check remaining time
        if time.time() - start_time > budget_s - 1:
            return None
        
        try:
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h_vars.value
        except:
            pass
        return None

    # Main optimization loop
    best_C5 = evaluate_C5(h)
    best_h = h.copy()
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        max_overlaps = np.full(n_points, 0.0)
        
        for k in range(n_points):
            shift_sum = 0
            for i in range(n_points):
                shift_sum += h[i] * (1 - h[(i+k)%n_points])
            max_overlaps[k] = shift_sum * dx
        
        max_k = np.argmax(max_overlaps)
        C5_value = np.max(max_overlaps)
        
        if C5_value < best_C5:
            best_C5 = C5_value
            best_h = h.copy()
            
        if C5_value <= target:
            break
            
        # Add a hard constraint for the highest overlap
        # Use a more efficient approach: just try small perturbations
        if iteration % 5 == 0:  # Avoid over-constraining
            # Try a small perturbation 
            perturbed_h = h.copy()
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            perturbed_h[idx] += delta
            
            # Make sure it stays within bounds
            perturbed_h = np.clip(perturbed_h, 0, 1)
            # Renormalize to sum to n_points/2
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            
            test_C5 = evaluate_C5(perturbed_h)
            if test_C5 < C5_value:
                h = perturbed_h
                best_C5 = test_C5
                best_h = h.copy()
            
        iteration += 1
        # Throttle down the frequency 
        if iteration % 100 == 0:
            time.sleep(0.001)  # To keep the loop responsive
    
    # Ensure final solution is within bounds and constraint
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_C5 = evaluate_C5(final_h)
    
    return (final_h.tolist(), final_C5, n_points)
