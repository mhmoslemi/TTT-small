import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[n_points//4:2*n_points//4] = 0.0
        # Ensure sum is n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        delta = target_sum - current_sum
        if delta != 0:
            # Distribute remaining mass evenly
            portion = delta / (n_points - 2*n_points//4)
            h[2*n_points//4:3*n_points//4] += portion
            h[3*n_points//4:] = 0.5
        
    # Discretization step
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = _compute_C5(h, dx)
    
    # Simulated Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    n_iter = int(1e4)
    alpha = np.exp(np.log(T_min/T_max) / n_iter)
    T = T_max
    delta = 0.1
    
    # Track acceptance rate for adaptive delta
    accept_count = 0
    total_count = 0 
    
    start_time = time.time()
    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate perturbed solution
        h_new = h.copy()
        
        # Choose perturbation type
        if np.random.random() < 0.5:
            # Swap two points
            idx1, idx2 = np.random.choice(n_points, 2, replace=False)
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        else:
            # Shift mass between adjacent points
            idx = np.random.randint(0, n_points-1)
            # Pick small amount to shift
            shift = np.random.uniform(-delta, delta)
            h_new[idx] += shift
            h_new[idx+1] -= shift
            
        # Clip to bounds
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraints
        current_sum = np.sum(h_new)
        target_sum = n_points / 2
        if current_sum > 0:
            scale = target_sum / current_sum
            h_new *= scale
        
        # Compute new C5
        C5_new = _compute_C5(h_new, dx)
        
        # Accept or reject
        delta_C = C5_new - best_C5
        if delta_C < 0 or np.random.random() < np.exp(-delta_C / T):
            h = h_new
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h.copy()
                
        # Update temperature and delta
        T *= alpha
        total_count += 1
        if delta_C < 0:
            accept_count += 1
            
        if total_count % 100 == 0 and total_count > 0:
            accept_rate = accept_count / total_count
            if accept_rate < 0.1:
                delta *= 0.9
            elif accept_rate > 0.3:
                delta *= 1.1
            accept_count = 0
            total_count = 0
    
    return (best_h.tolist(), best_C5, n_points)

def _compute_C5(h, dx):
    # Compute the C5 score for the given h
    conv = correlate(h, 1 - h, mode='full')
    max_val = np.max(np.abs(conv))
    return max_val * dx
