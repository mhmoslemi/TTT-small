import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    n_points = 45
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create binary-like initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        
    h = h / np.sum(h) * (n_points / 2)
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Optimization parameters
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    max_no_improvement = 100
    no_improvement_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_C5(h_array):
        # Compute overlaps
        # Using numpy correlate with mode='full'
        h1 = h_array
        h2 = 1 - h_array
        corr = np.correlate(h1, h2, mode='full')
        # Only take the relevant part (shifts from -N+1 to N-1)
        # But actually we want to interpret the maximum correlation at any shift
        # The full correlation has length 2*N-1
        # Result indices go from -N+1 to N-1
        # We want to get the max
        max_corr = np.max(corr)
        return max_corr * dx
    
    max_iter = 5000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        c5 = compute_C5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if c5 < 0.3807 or no_improvement_count > max_no_improvement:
            break
            
        # Estimate gradient via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        h_backup = h.copy()
        
        for i in range(n_points):
            h[i] += eps
            c5_plus = compute_C5(h)
            h[i] -= 2 * eps
            c5_minus = compute_C5(h)
            h[i] += eps  # restore
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Step update with constraints
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if h_sum > 0:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Check new value
        c5_new = compute_C5(h_new)
        if c5_new < c5:
            h = h_new
            alpha = min(1.0, alpha * 1.02)  # Increase step size if successful
        else:
            alpha *= gamma
            if alpha < alpha_min:
                break
                
    return (best_h.tolist(), best_c5, n_points)
