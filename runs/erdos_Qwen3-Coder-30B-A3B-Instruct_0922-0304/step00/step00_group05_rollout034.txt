import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a balanced initial configuration with some randomness
        h = np.ones(n_points) * 0.5
        # Ensure sum constraint
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if diff > 0:
            # Add to lower values
            sorted_indices = np.argsort(h)
            for i in range(int(diff / 0.1)):
                if i >= len(sorted_indices):
                    break
                idx = sorted_indices[i]
                h[idx] = min(1.0, h[idx] + 0.1)
        elif diff < 0:
            # Reduce from higher values
            sorted_indices = np.argsort(h)[::-1]
            for i in range(int(-diff / 0.1)):
                if i >= len(sorted_indices):
                    break
                idx = sorted_indices[i]
                h[idx] = max(0.0, h[idx] - 0.1)
        # Normalize sum
        h *= target_sum / np.sum(h)

    # Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    n_iterations = 10**4
    alpha = np.exp(np.log(T_min / T_max) / n_iterations)
    T = T_max
    delta = 0.1
    
    # Track best solution
    best_h = h.copy()
    current_C5 = compute_C5(h, dx)
    best_C5 = current_C5
    
    # Cooling schedule is exponential
    temp_schedule = T_max * (alpha ** np.arange(n_iterations))
    
    # For adaptive delta
    accept_count = 0
    total_count = 0
    
    for i in range(n_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a proposal
        h_new = h.copy()
        method = np.random.choice(["swap", "shift"])
        
        if method == "swap":
            # Pick two random indices
            i1 = np.random.randint(n_points)
            i2 = np.random.randint(n_points)
            if i1 != i2:
                # Swap but preserve sum
                val1 = h_new[i1]
                val2 = h_new[i2]
                # Adjust magnitudes
                new_val1 = min(1.0, max(0.0, val1 + delta * (np.random.rand() - 0.5)))
                new_val2 = min(1.0, max(0.0, val2 + delta * (np.random.rand() - 0.5)))
                # Ensure sum constraint via adjustment
                change1 = new_val1 - val1
                change2 = new_val2 - val2
                # Shift other values to maintain sum
                total_change = change1 + change2
                if abs(total_change) > 1e-8:
                    # Find index to adjust
                    for idx in range(n_points):
                        if idx != i1 and idx != i2:
                            if abs(h_new[idx] - 1.0) > 1e-8 and change1 + change2 > 0:
                                h_new[idx] = max(0.0, h_new[idx] - min(change1 + change2, 1.0 - h_new[idx]))
                                break
                            elif abs(h_new[idx] - 0.0) > 1e-8 and change1 + change2 < 0:
                                h_new[idx] = min(1.0, h_new[idx] - max(change1 + change2, -h_new[idx]))
                                break
                h_new[i1] = new_val1
                h_new[i2] = new_val2
                
        else:  # shift
            # Shift between adjacent points and balance
            i1 = np.random.randint(n_points - 1)
            i2 = i1 + 1
            delta_shift = delta * (np.random.rand() - 0.5)
            h_new[i1] = min(1.0, max(0.0, h_new[i1] + delta_shift))
            h_new[i2] = min(1.0, max(0.0, h_new[i2] - delta_shift))
            
        # Clip values
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum constraint
        current_sum_new = np.sum(h_new)
        scale_factor = target_sum / current_sum_new
        h_new *= scale_factor
    
        # Evaluate
        new_C5 = compute_C5(h_new, dx)
        delta_C = new_C5 - current_C5
        
        # Accept or reject
        if delta_C < 0 or np.random.rand() < np.exp(-delta_C / T):
            h = h_new
            current_C5 = new_C5
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Adaptive delta adjustment
        total_count += 1
        if delta_C < 0:
            accept_count += 1
        if total_count == 100:
            accept_rate = accept_count / total_count
            # Adjust delta based on acceptance rate
            if accept_rate > 0.7:
                delta *= 1.05
            elif accept_rate < 0.3:
                delta /= 1.05
            accept_count = 0
            total_count = 0
            
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Correlate h and 1 - h
    h_rev = np.flip(h)
    corr = correlate(h, h_rev, mode='full')
    # The cross-correlation for overlap is h * (1 - h)
    # We need correlation of h with 1-h shifted by k steps
    c5_vals = []
    n = len(h)
    for k in range(2 * n - 1):
        # Compute dot product of h with reversed (1-h) shifted by k
        shifted_one_minus_h = np.zeros_like(h)
        # Create shifted version of (1-h)
        if k < n:
            temp = np.concatenate([np.zeros(k), 1 - h])
        else:
            temp = np.concatenate([1 - h, np.zeros(k - n + 1)])
        shifted_one_minus_h = temp[:n]
        # Compute dot product
        dot_prod = np.dot(h, shifted_one_minus_h)
        c5_vals.append(dot_prod * dx)
    return max(c5_vals)
