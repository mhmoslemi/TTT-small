import time

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()

        # Compute overlap integral
        corr = np.correlate(h, 1-h, mode='full')
        max_val = np.max(corr) * dx
        return max_val
    
    def project_h(h, dx):
        # Keep values in [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to maintain sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Convex relaxation helper (simplified)
    def convex_solve(h0, max_iter=100):
        h = h0.copy()
        h = project_h(h, dx)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            # Simple gradient ascent step based on correlation
            corr = np.correlate(h, 1-h, mode='full')
            max_idx = np.argmax(corr)
            # Apply small update to reduce that correlation
            lag = max_idx - (len(h) - 1)
            # Approximate gradient
            grad = np.zeros_like(h)
            for i in range(len(h)):
                if i + lag >= 0 and i + lag < len(h):
                    grad[i] += (1 - h[i + lag])
                if i - lag >= 0 and i - lag < len(h):
                    grad[i] += (1 - h[i - lag])
            h = h - 0.01 * grad
            h = project_h(h, dx)
        return h
    
    # Try convex relaxation
    h = convex_solve(h0)
    best_C5 = eval_C5(h)
    best_h = h.copy()

    def obj_and_grad(h):
        h = np.array(h)
        corr = np.correlate(h, 1-h, mode='full')
        max_k = np.argmax(corr)
        k = max_k - (len(h) - 1)
        # Approximate gradient w.r.t. h at lag k.
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k >= 0 and i + k < len(h):
                grad[i] += (1 - h[i + k])
            if i - k >= 0 and i - k < len(h):
                grad[i] += (1 - h[i - k])
        return -corr[max_k] * dx, grad

        return -eval_C5(h)

        return np.sum(h) - n_points / 2

                if j + k >= 0 and j + k < len(h_up):
                    grad[j] += (1 - h_up[j + k])
                if j - k >= 0 and j - k < len(h_up):
                    grad[j] += (1 - h_up[j - k])
            h_up = h_up - 0.01 * grad
            h_up = project_h(h_up, dx_up)
        
        up_C5 = eval_C5(h_up)
        if up_C5 < best_C5:
            best_C5 = up_C5
            best_h = h_up
            n_points = n_up
            
    # Make sure we return valid solution
    best_h = project_h(best_h, dx)
    return (best_h.tolist(), best_C5, n_points)
