import numpy as np
from scipy import signal
import time

def evaluate_C5(h, dx):
    # Compute all overlaps using convolution
    conv = signal.convolve(h, 1-h, mode='full')
    # Need to slice to get valid lags
    n = len(h)
    # Max overlap corresponds to the max of the convolution (excluding edge effects)
    # The overlap integral for lag k is int h(x) (1-h(x+k)) dx
    # which corresponds to conv[k] * dx for appropriate k
    # But we need only the max value, which we get from full convolution
    # max between k=0 and k=n-1, so k ranges from 0 to 2*n-2 
    # But the appropriate range feels tricky, 
    # better to check all possible lags directly
    
    # Since h(2)=h(0) in period, we can use k=0,..,n-1 for unique lags 
    # But freq domain conv is not quite what we want
    # Let's compute manually  
    max_overlap = 0.0  
    for k in range(n):
        s = 0.0
        for i in range(n):
            j = (i + k) % n
            s += h[i] * (1 - h[j]) * dx
        if s > max_overlap:
            max_overlap = s
    return max_overlap


def project_h(h, dx):
    # Project to feasible set: clip and scale to sum = 1/dx = n_points/2
    n_points = len(h)
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    target_sum = n_points / 2.0
    if current_sum > 0:
        scale = target_sum / current_sum  
        h = h * scale
    return h


def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise make a simple one
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        n_points = 70
        initial_h = np.random.rand(n_points) * 0.5
    
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 10000
    
    # Initialize swarm
    particles = []
    velocities = []
    for i in range(swarm_size):
        if i == 0 and 'initial_h_values' in globals():
            h = initial_h.copy()
        else:
            h = np.random.rand(n_points) * 0.5
            h = project_h(h, dx)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.5, 0.5, n_points))
    
    # Track personal bests
    pbest = [None]*swarm_size
    pbest_scores = [float('inf')]*swarm_size
    
    # Global best
    gbest = None
    gbest_score = float('inf')
    
    start_time = time.time()
    current_iter = 0
    
    while time.time() - start_time < budget_s - 1 and current_iter < max_iter:
        current_iter += 1
        for i in range(swarm_size):
            h = particles[i]
            # Evaluate
            score = evaluate_C5(h, dx)
            
            # Update personal best
            if score < pbest_scores[i]:
                pbest[i] = h.copy()
                pbest_scores[i] = score
            
            # Update global best
            if score < gbest_score:
                gbest = h.copy()
                gbest_score = score
            
            # Update velocities and positions
            w = 0.7
            c1 = 1.5
            c2 = 1.5
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            
            # Update velocity
            v = w * velocities[i] + c1*r1*(pbest[i] - h) + c2*r2*(gbest - h)
            velocities[i] = v
            
            # Update position
            new_h = h + velocities[i]
            new_h = project_h(new_h, dx)
            
            particles[i] = new_h
            
    # Return best solution found
    if gbest is None:
        return initial_h, evaluate_C5(initial_h, dx), n_points
    
    # Re-evaluate gbest to make sure it's clean
    final_score = evaluate_C5(gbest, dx)
    return gbest, final_score, n_points
