```python
import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 bound for given h values."""
    n = len(h)
    # Compute correlation for all lags
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract valid correlations and get maximum
    max_corr = np.max(corr[n-1:])
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Fallback: create initial guess
        n_points = 42
        h0 = np.ones(n_points) * 0.5

    dx = 2.0 / n_points

    # Objective function to minimize
    def objective(h):
        return evaluate_erdos_solution(h, dx)

    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    def constraint_bounds(h):
        return np.min(h), np.max(h) - 1.0

    # Define constraints for scipy
    constraints = [
        {'type': 'eq', 'fun': constraint_sum},
        {'type': 'ineq', 'fun': lambda h: np.min(h)},
        {'type': 'ineq', 'fun': lambda h: 1.0 - np.max(h)}
    ]

    # Try to find initial better solution using scipy
    h_best = h0.copy()
    c5_best = objective(h_best)
    
    # Optimizer with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        result = minimize(objective, h_best, method='SLSQP', bounds=bounds,
                          constraints=constraints, options={'maxiter': 10000})
        if result.success:
            h_opt = result.x
            c5_opt = objective(h_opt)
            if c5_opt < c5_best:
                h_best = h_opt
                c5_best = c5_opt
    except:
        pass

    # Try SDP relaxation approach
    try:
        # Build a convex optimization using CVXPY
        # Variables
        y = Variable((n_points, n_points), symmetric=True)
        h = Variable(n_points)
        C5 = Variable()

        # Constraints
        constraints = [
            # PSD constraint
            y >> 0,
            # Constraint that rows sum to correct value
            trace(y) == (n_points / 2.0)**2,
            # Individual h variable constraint
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2.0
        ]
        
        # Add bounds on correlations
        for k in range(n_points):
            corr = 0
            for i in range(n_points):
                if i + k < n_points:
                    corr += h[i] - y[i, i + k]
                else:
                    corr += h[i] - y[i, i + k - n_points]
            constraints.append(corr <= C5 / dx)

        # Objective
        prob = Problem(Minimize(C5), constraints)

        # Try to solve via SDP with available solver
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        
        if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
            h_sdp = h.value
            if h_sdp is not None and np.all(h_sdp >= 0) and np.all(h_sdp <= 1):
                c5_sdp = objective(h_sdp)
                if c5_sdp < c5_best:
                    h_best = h_sdp
                    c5_best = c5_sdp
    except:
        pass

    # Final check
    if c5_best > 0.38085:
        # Use previously computed h0 as fallback if needed
        c5_best = objective(h0)
        if c5_best < c5_best:
            h_best = h0.copy()

    return h_best, c5_best, n_points
```
