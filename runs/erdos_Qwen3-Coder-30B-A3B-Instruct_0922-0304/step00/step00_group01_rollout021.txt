import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, n_points):
    # compute the overlap integral for all shifts
    dx = 2.0 / n_points
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial h values if available, else create a uniform one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        h = np.full(n_points, 0.5)
    
    # Normalize to satisfy the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # cache for performance
    dx = 2.0 / n_points
    target = 0.3807
    
    # parameters
    alpha = 0.01
    max_iter = 1000
    patience = 100 
    min_alpha = 1e-5
    prev_best = float('inf')
    patience_counter = 0
    
    # main optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute overlaps for all shifts
        overlaps = np.correlate(h, 1 - h, mode='full') * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # early stopping
        if max_overlap <= target:
            break
            
        # compute subgradient for the worst case
        g = np.zeros(n_points)
        for i in range(n_points):
            j = (i - k_max) % n_points
            g[i] = (1 - h[j]) - h[i]
            
        # update h
        h -= alpha * g
        h = np.clip(h, 0, 1)
        
        # enforce sum constraint
        h *= (n_points / 2) / np.sum(h)
        
        # adjust alpha if no improvement
        if max_overlap >= prev_best:
            patience_counter += 1
            if patience_counter > patience:
                alpha *= 0.5
                if alpha < min_alpha:
                    break
                patience_counter = 0
        else:
            prev_best = max_overlap
            patience_counter = 0
            
    # final evaluation
    c5_bound = evaluate_erdos_solution(h, n_points)
    return (h.tolist(), c5_bound, n_points)
