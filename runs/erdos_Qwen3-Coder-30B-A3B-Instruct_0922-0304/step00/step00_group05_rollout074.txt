def evaluate_erdos_solution(h, dx):

    corr = np.correlate(h, 1 - h, mode='full')

    # The max overlap is from shifts k = 0 to n-1 for the first part
    c5 = 0.0
    for k in range(n):

        # This is just the kth element of the full correlation from index n-1
        c5 = max(c5, corr[n-1+k] * dx)  # corresponds to shift k
    return c5
