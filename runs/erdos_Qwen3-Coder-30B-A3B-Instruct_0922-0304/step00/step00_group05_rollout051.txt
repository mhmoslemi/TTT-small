import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create a basic one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n_points = 70
        h = np.ones(n_points, dtype=float) * 0.5

    dx = 2.0 / len(h)
    n_points = len(h)
    target_sum = n_points / 2.0
    max_iter = 50000  # conservative limit
    learning_rate = 0.01
    eps = 1e-4
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        h_padded = np.concatenate([h, np.zeros(n_points-1)])
        one_minus_h = 1.0 - h
        one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n_points-1)])
        
        # Convolution: h(x) * (1 - h(x + k))
        correlation = correlate(h_padded, one_minus_h_padded, mode='full')
        # Extract valid overlaps (k from 0 to 2*len(h)-2)
        overlaps = correlation[len(h)-1:] * dx
        
        # Compute max overlap (C5)
        c5 = np.max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find indices where maximum occurs
        max_indices = np.where(np.abs(overlaps - c5) < eps)[0]
        
        # Compute gradient
        gradient = np.zeros_like(h)
        for idx in max_indices:
            # This is a simplified calculation, using np.roll for simplicity
            # and model the derivative of the convolution
            shift = idx - (n_points - 1)
            if shift >= 0:
                # h[i] * (1 - h[i + shift])
                if shift < len(h):
                    gradient += one_minus_h[shift:]
                    # We're approximating gradient w.r.t h[i]
                    gradient[:len(h)-shift] += -h[shift:]
            else:
                # h[i] * (1 - h[i - shift]) => h[i]*1 - h[i]*h[i-shift]
                if -shift < len(h):
                    gradient = gradient + 1.0
                    gradient[shift:] = gradient[shift:] - h[shift:]
                # "Forward" part
                gradient[:len(h)+shift] = gradient[:len(h)+shift] - h[shift:]
        
        # Update rule with gradient, projected
        h = h - learning_rate * gradient
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-6:
            h = h * target_sum / current_sum
    
    # Final C5 evaluation
    h_padded = np.concatenate([best_h, np.zeros(n_points-1)])
    one_minus_h = 1.0 - best_h
    one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n_points-1)])
    correlation = correlate(h_padded, one_minus_h_padded, mode='full')
    overlaps = correlation[len(h)-1:] * dx
    final_c5 = np.max(overlaps)
    
    return (best_h.tolist(), float(final_c5), n_points)
