import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given h"""
    dx = 2.0 / len(h)
    overlaps = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

def project_binary(h_relaxed, target_sum):
    """Project relaxed solution to binary with exact sum constraint"""
    n = len(h_relaxed)
    # Sort indices by value (descending)
    sorted_indices = np.argsort(h_relaxed)[::-1]
    
    # Start with all ones
    h_binary = np.ones(n)
    
    # Adjust to target sum
    diff = int(target_sum - np.sum(h_binary))
    if diff > 0:
        # Need to turn some 1's into 0's
        for i in range(diff):
            h_binary[sorted_indices[i]] = 0
    elif diff < 0:
        # Need to turn some 0's into 1's
        for i in range(-diff):
            h_binary[sorted_indices[i]] = 1
    
    return h_binary
