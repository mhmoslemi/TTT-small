import numpy as np
from scipy import optimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 5000
    target_bound = 0.3807
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    
    # Dual variables
    lambda_vals = np.zeros(2 * n_points - 1)
    mu = 0.0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex subproblem for h
        h_new = h.copy()
        h_var = Variable(n_points)
        
        # Objective: minimize max overlap + quadratic penalty
        overlaps = []
        for k in range(2 * n_points - 1):
            overlap = 0
            for i in range(n_points):
                j = i + k - (n_points - 1)
                if 0 <= j < n_points:
                    overlap += h_var[i] * (1 - h_var[j])
            overlaps.append(overlap * dx)
        
        # Maximize overlap constraint
        max_overlap = 0
        for o in overlaps:
            max_overlap = max(max_overlap, o)
        
        # Add penalty term
        obj = max_overlap + rho * sum_squares(h_var - h)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
        
        prob = Problem(Minimize(obj), constraints)
        prob.solve(solver=ECOS, max_iter=1000, verbose=False)
        
        if prob.status == "optimal":
            h_new = h_var.value
        
        h = h_new
        
        # Compute overlaps and update dual variables
        overlap_vals = np.correlate(h, 1 - h, mode='full') * dx
        max_overlap_val = np.max(overlap_vals)
        
        # Update dual variables
        lambda_vals = lambda_vals + alpha * (overlap_vals - max_overlap_val)
        mu = mu + alpha * (np.sum(h) * dx - 1)
        
        # Entropy regularization for lambda
        lambda_vals = lambda_vals + beta * np.log(1 + np.exp(lambda_vals))
        lambda_vals = np.clip(lambda_vals, -1e6, 1e6)
        
        # Termination check
        if max_overlap_val <= target_bound:
            break
    
    c5_bound = np.max(np.correlate(h, 1 - h, mode='full') * dx)
    
    return (h.tolist(), c5_bound, n_points)
