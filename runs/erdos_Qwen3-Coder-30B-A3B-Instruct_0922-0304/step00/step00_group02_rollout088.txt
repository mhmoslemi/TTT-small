import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    overlaps = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)

        iteration += 1

        overlaps = correlate(h, 1 - h, mode="full") * dx
        C5 = np.max(overlaps)

        h_relaxed = cp.Variable(n)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n / 2
        ]
        obj = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)

                threshold = 0.5

                        proj_h[indices] = 1
                    else:
                        # Need to set some 1s to 0s
