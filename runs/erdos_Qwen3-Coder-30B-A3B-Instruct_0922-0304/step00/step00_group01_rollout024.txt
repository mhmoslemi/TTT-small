def evaluate_erdos_solution(h_values):
    # This is a helper from the harness
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    c5_values = []
    for k in range(n):
        # Compute overlap for shift k
        overlap = 0.0
        for i in range(n):
            j = (i - k) % n
            overlap += h[i] * (1.0 - h[j])
        c5_values.append(overlap * dx)
    return max(c5_values)
