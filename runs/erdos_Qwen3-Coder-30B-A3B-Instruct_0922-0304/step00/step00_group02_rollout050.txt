import numpy as np
from scipy import correlate
import time

def evaluate_h(h_values):
    """Evaluate C5 for given h values."""
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute full cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The correlation values are indexed from -len(h)+1 to len(h)-1
    # The maximum is over all valid lags, but we only consider overlapping parts
    # The maximum value of the integral computed as corr * dx
    max_corr = np.max(corr)
    return max_corr * dx
