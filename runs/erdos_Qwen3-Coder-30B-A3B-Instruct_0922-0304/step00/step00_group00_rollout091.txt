import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Initialize with uniform distribution
        n_points = 45
        h0 = np.full(n_points, 0.5)
        h0 = np.clip(h0, 0, 1)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    h = h0.copy()
    dx = 2.0 / n_points
    best_C5 = float('inf')
    best_h = h.copy()
    
    # Optimization parameters
    alpha = 0.1
    max_iter = 10000
    patience = 50
    patience_count = 0
    last_C5 = float('inf')
    
    # Gradient descent + convex optimization hybrid
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.9:
            break
            
        # Evaluate current C5
        # Compute overlaps for all possible shifts
        overlapping = np.zeros(n_points * 2 - 1)
        for k in range(1 - n_points, n_points):
            shift = k if k >= 0 else n_points + k
            if k >= 0:
                # Negative shift (right shift)
                h_shift = np.concatenate([h[shift:], h[:shift]])
            else:
                # Positive shift (left shift)
                h_shift = np.concatenate([h[-k:], h[:-k]])
            overlap = np.sum(h * (1 - h_shift)) * dx
            overlapping[k + n_points - 1] = overlap
            
        C5 = np.max(overlapping)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Gradient computation for the maximum overlap shift
        max_k_index = np.argmax(overlapping)
        max_k = max_k_index - (n_points - 1)
        
        grad = np.zeros(n_points)
        if max_k >= 0:
            h_shift = np.concatenate([h[max_k:], h[:max_k]])
        else:
            h_shift = np.concatenate([h[-max_k:], h[:-max_k]])
            
        # Compute gradient for overlap with shift k
        for i in range(n_points):
            grad[i] += (1 - h_shift[i]) * dx
            if max_k >= 0:
                grad[(i + max_k) % n_points] -= h[i] * dx
            else:
                grad[(i - max_k) % n_points] -= h[i] * dx
                
        # Update step
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Convex optimization refinement
        if iteration % 5 == 4:
            # Focus on the current max shift to improve it
            k_max_index = np.argmax(overlapping)
            k_max = k_max_index - (n_points - 1)
            
            # Use QP solver for this step
            # Formulate as QP to minimize s.t. constraints at k_max
            # Minimize 0.5 * x^T Q x + c^T x subject to Ax <= b, A_eq x = b_eq
            
            # Very simple direct refinement using CVXPY
            w = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(w))
            constraints = [
                w[i] >= h[i] * (1 - h[(i + k_max) % n_points]) * dx 
                for i in range(n_points)
            ]
            
            # Added constraints
            constraints += [
                0 <= w[i] for i in range(n_points)
            ]
            constraints += [
                np.sum(w) == n_points / 2
            ]
            prob = cp.Problem(objective, constraints)
            
            try:
                # Solve using ECOS
                prob.solve(solver=cp.ECOS, verbose=False)
                if prob.status == cp.OPTIMAL:
                    w_sol = w.value
                    # Ugly heuristics to continue making update
                    h = np.array(w_sol)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except:
                pass  # Fall back to gradient descent if solver fails
                
        # Adaptive step size adjustment
        if iteration > 0 and iteration % patience == 0:
            if abs(C5 - last_C5) < 1e-6:
                patience_count += 1
                if patience_count > 3:
                    alpha *= 0.5
                    patience_count = 0
            else:
                patience_count = 0
            last_C5 = C5
            
    # Final evaluation
    overlapping_final = np.zeros(n_points * 2 - 1)
    for k in range(1 - n_points, n_points):
        shift = k if k >= 0 else n_points + k
        if k >= 0:
            h_shift = np.concatenate([best_h[shift:], best_h[:shift]])
        else:
            h_shift = np.concatenate([best_h[-k:], best_h[:-k]])
        overlap = np.sum(best_h * (1 - h_shift)) * dx
        overlapping_final[k + n_points - 1] = overlap
    final_C5 = np.max(overlapping_final)
    
    return (best_h.tolist(), float(final_C5), int(n_points))
