import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_steps = 30
    alpha_init = 0.01
    rho = 0.5
    eta = 0.5
    patience = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute 1-h for efficiency
    h1 = 1 - h
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        # correlate(h, h1, mode='full') gives correlation of length 2*n-1
        # We only care about positive shifts (k from 0 to n-1)
        # But we want shift indices from 0 to n-1, so we extract the correct part
        full_corr = correlate(h, h1, mode='full')
        overlaps = full_corr[n_points-1:] * dx  # This is C(k) for k=0 to n-1
        
        k_max = np.argmax(overlaps)
        c5_current = overlaps[k_max]
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if c5_current < target_bound:
            break
            
        if no_improvement_count > patience:
            break
            
        # Compute gradient for shift k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Term 1: 1 - h[i + k_max] if i + k_max < n_points else 0
            term1 = h1[i + k_max] if i + k_max < n_points else 0
            # Term 2: h[i] * (1 - h[i - k_max]) if i - k_max >= 0 else 0
            term2 = h[i] * (h1[i - k_max] if i - k_max >= 0 else 0)
            grad[i] = term1 - term2
        
        # Line search
        alpha = alpha_init
        failed = False
        
        for _ in range(line_search_steps):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale to match constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new C5
            h1_new = 1 - h_new
            full_corr_new = correlate(h_new, h1_new, mode='full')
            overlaps_new = full_corr_new[n_points-1:] * dx
            c5_new = overlaps_new[k_max]
            
            if c5_new < c5_current:
                h = h_new
                break
            else:
                alpha *= rho
        else:
            # If line search fails, terminate early
            failed = True
            
        iter_count += 1
        if failed:
            break
    
    if best_c5 >= 0.3807:
        # Could not improve, return current best
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values, dtype=float)
            h = h * (n_points / 2) / np.sum(h)
            h1 = 1 - h
            full_corr = correlate(h, h1, mode='full')
            overlaps = full_corr[n_points-1:] * dx
            best_c5 = np.max(overlaps)
        else:
            # Just return what we have
            pass
    
    # Final validation
    if not (np.all(h >= 0) and np.all(h <= 1)):
        raise ValueError("h values out of bounds")
    if abs(np.sum(h) - n_points / 2) > 1e-6:
        raise ValueError("h does not sum to correct value")
    
    return best_h.tolist(), best_c5, n_points
