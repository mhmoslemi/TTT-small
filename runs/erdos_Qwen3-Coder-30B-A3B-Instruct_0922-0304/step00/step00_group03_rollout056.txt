import numpy as np
from scipy import correlate
import time
import random

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h."""
    correlation = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def check_feasibility(h):
    """Check if h satisfies constraints."""
    return np.all((h >= 0) & (h <= 1)) and abs(np.sum(h) - 1.0) < 1e-10

def repair_sum(h, target_sum):
    """Repair the sum to match target by flipping bits."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    h_copy = h.copy()
    diff = int(current_sum - target_sum)
    
    if diff > 0:
        # Need to reduce sum, flip 1s to 0s
        ones_indices = np.where(h_copy == 1)[0]
        flip_indices = random.sample(list(ones_indices), abs(diff))
        for i in flip_indices:
            h_copy[i] = 0
    else:
        # Need to increase sum, flip 0s to 1s
        zeros_indices = np.where(h_copy == 0)[0]
        flip_indices = random.sample(list(zeros_indices), abs(diff))
        for i in flip_indices:
            h_copy[i] = 1
            
    return h_copy

            if tuple(h_new) not in tabu_list and check_feasibility(h_new):
                neighbors.append(h_new)
        except Exception:
