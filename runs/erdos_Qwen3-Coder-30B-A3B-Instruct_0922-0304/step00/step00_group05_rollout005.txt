import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create initial configuration with some randomness
        h = np.zeros(n_points)
        # Place 35 ones in random positions
        ones_indices = random.sample(range(n_points), 35)
        h[ones_indices] = 1.0
        # Add some intermediate values to allow for smoothing
        for i in range(n_points):
            if h[i] == 0:
                h[i] = np.random.random() * 0.5
            elif h[i] == 1:
                h[i] = 1.0 - np.random.random() * 0.5
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    # Optimization parameters
    T_max, T_min = 1.0, 1e-4
    max_iter = int(1e4)
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    
    # Perturbation scale
    delta = 0.1
    
    # Track best solution
    best_h = h.copy()
    current_C5 = evaluate_C5(h, dx)
    best_C5 = current_C5
    
    # Simulated annealing
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Make a copy of current h
        new_h = h.copy()
        
        # Try a random perturbation
        if random.random() < 0.5:
            # Swap two random indices
            i, j = random.randint(0, n_points-1), random.randint(0, n_points-1)
            new_h[i], new_h[j] = new_h[j], new_h[i]
        else:
            # Shift small amount between adjacent points
            i = random.randint(0, n_points-1)
            j = (i + 1) % n_points
            # Slide some mass
            amount = random.random() * delta
            if amount > new_h[i]:
                amount = new_h[i]
            new_h[i] -= amount
            new_h[j] += amount
        
        # Ensure feasibility after perturbation
        # Clip values to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to maintain sum constraint
        new_sum = np.sum(new_h)
        if new_sum > 0:
            new_h = new_h * (target_sum / new_sum)
        
        # Evaluate new solution
        new_C5 = evaluate_C5(new_h, dx)
        
        # Accept with probability based on Metropolis criterion
        if new_C5 < current_C5:
            h = new_h.copy()
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
        else:
            # Accept worse solution with probability
            T = T_max * (alpha ** iteration)
            if T > T_min:
                prob = np.exp(-(new_C5 - current_C5) / T)
                if random.random() < prob:
                    h = new_h.copy()
                    current_C5 = new_C5
        
        # Adapt delta based on acceptance rate
        # We do not implement a full adaptation here due to simplicity,
        # but delta could be updated based on the ratio of accepted moves
        # For now we keep it constant, as our main exploration is via perturbations
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h, dx):
    # Compute 1-h for correlation
    one_minus_h = 1 - np.array(h)
    # Perform full convolution
    corr = signal.correlate(h, one_minus_h, mode='full')
    # Get the maximum value from the convolution (excluding padding)
    max_corr = np.max(np.abs(corr))
    # Scale by dx
    C5 = max_corr * dx
    return C5
