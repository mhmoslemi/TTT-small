import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all lags
    overlap = correlate(h, 1 - h, mode="full") * dx
    return np.max(overlap)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Default construction: evenly distribute 1s in a reasonable pattern
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[10:30] = 1.0
        h[40:60] = 0.0
        h = h * (n_points / 2.0) / np.sum(h)  # Normalize
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialize optimization parameters
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    learning_rate = 0.01
    gamma_decrease_factor = 0.1
    gamma_patience = 100
    patience_counter = 0
    last_best_c5 = best_c5
    
    # Track time
    start_time = time.time()
    max_iterations = int(budget_s * 1000)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate current overlap
        overlap = correlate(h, 1 - h, mode="full") * dx
        max_overlap = np.max(overlap)
        
        # Find lags where max overlap occurs
        max_lags = np.where(np.abs(overlap - max_overlap) < 1e-10)[0]
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for lag in max_lags:
            if lag < n_points:
                # Forward difference
                for i in range(n_points - lag):
                    gradient[i] += (1 - h[i + lag])
            else:
                # Backward difference
                for i in range(lag - n_points + 1, n_points):
                    gradient[i] += (1 - h[i - lag + n_points - 1])
        
        gradient /= len(max_lags) if len(max_lags) > 0 else 1.0
        
        # Update h
        h -= learning_rate * gradient
        
        # Project h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy constraint ∫ h dx = 1
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Evaluate new C5
        new_c5 = evaluate_erdos_solution(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Adjust learning rate with patience
        if abs(new_c5 - last_best_c5) < 1e-8:
            patience_counter += 1
        else:
            patience_counter = 0
            last_best_c5 = new_c5
            
        if patience_counter >= gamma_patience:
            learning_rate *= gamma_decrease_factor
            patience_counter = 0
        
    return (best_h.tolist(), best_c5, n_points)
