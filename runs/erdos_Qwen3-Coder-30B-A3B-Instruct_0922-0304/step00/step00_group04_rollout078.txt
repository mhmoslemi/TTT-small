import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 42
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_h(h_vals):
        # Project to [0,1] and enforce sum constraint
        h_proj = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_proj)
        if h_sum > 0:
            h_proj = h_proj * (n_points / 2) / h_sum
        return h_proj
    
    def compute_gradient(h_vals, k):
        # Compute gradient for lag k
        n = len(h_vals)
        grad = np.zeros(n)
        # k is actually the lag in cross-correlation
        # For a positive lag k, we want to compute d/dh[i] of sum h[i] * (1 - h[i+k])
        # This involves 1-difference of indices
        # We compute the derivative manually
        for i in range(n):
            if i + k >= 0 and i + k < n:
                grad[i] += (1 - h_vals[i + k]) * dx
            if i - k >= 0 and i - k < n:
                grad[i] += -h_vals[i - k] * dx
        return grad
    
    iterations = 0
    while time.time() - start_time < budget_s - 1:
        # Compute all correlations and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr = corr[max_corr_idx]
        c5_new = max_corr * dx
        
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            
        # Find the lag associated with max corr
        # In correlation, the middle index corresponds to k=0
        lag = max_corr_idx - (len(h) - 1)
        
        # Compute gradient
        grad = compute_gradient(h, lag)
        
        # Simple gradient descent with hand-tuned learning rate
        learning_rate = 0.05
        h_new = h - learning_rate * grad
        
        # Project back to feasible set
        h = project_h(h_new)
        
        # Early termination if no significant improvement
        if iterations > 500 and abs(best_c5 - c5_new) < 1e-6:
            break
            
        iterations += 1
    
    if best_c5 == float('inf'):
        best_c5 = compute_c5(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
