from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the Erdos constant C5 for given h."""
    h_full = np.concatenate([h, np.zeros_like(h)])
    conv = correlate(h_full, 1 - h, mode='full')
    max_conv = np.max(conv[:len(h)])
    return max_conv * dx
