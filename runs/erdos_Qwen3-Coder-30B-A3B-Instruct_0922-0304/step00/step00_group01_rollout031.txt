import numpy as np

    except NameError:
        # Construct initial solution
        h = np.ones(n_points) * 0.5

    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    iter_count = 0
    max_iter = 100
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute overlaps for all shifts
        overlaps = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if best_C5 > 0.3807:
            # We still need improvement, solve a convex problem
            # Minimize max overlap with linear constraints
            
            # This is simplified since we're solving for a single constraint
            
            # Determine shift index
            shift_idx = k_max - (n_points - 1)
            
            # Define a local change to reduce the overlap at k_max
            # Build Jacobian and solve to reduce that overlap
            A = np.zeros((1, n_points))
            for i in range(n_points):
                j = (i - shift_idx) % n_points
                A[0, i] = 1 - h[j]
            
            # Gradient of that overlap w.r.t h
            g = A[0]
            
            # Solve quadratic problem
            # min 0.5 * dt * G * dt + g^T * dt
            # s.t. sum(dt) = 0 (maintain sum)
            #     0 <= h[i] + dt[i] <= 1
            
            # Constraint matrix for bounds
            # G = I (for l2 norm approximate gradient)
            dt = -0.1 * g / np.linalg.norm(g)
            
            # Project to satisfy sum
            dt = dt - np.mean(dt)
            
            # Clip and project feasibility
            h_new = h + dt
            
            # Apply bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Re-adjust to maintain integral constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new

            if C5_new < best_C5:

                best_h = h.copy()
        else:
            break
        
        iter_count += 1
        
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_h.tolist(), float(best_C5), n_points)


def evaluate_erdos_solution(h):
    """Evaluate the C5 bound given h"""
    dx = 2.0 / len(h)
    overlaps = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(overlaps)
