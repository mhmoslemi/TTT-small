import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode='full')
    # Only take the relevant part (k from 0 to n_points-1)
    overlaps = overlaps[len(h)-1:]  # This gives us the correct computation for C5
    return np.max(overlaps) * dx

def project_h(h, dx):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    # Scale to sum = n_points / 2
    current_sum = np.sum(h)
    target_sum = 1.0 / dx  # Since dx = 2.0 / n_points, this equals n_points/2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.random.random(n_points) * 0.5  # Start with random values in [0, 0.5]
    
    dx = 2.0 / n_points
    swarm_size = 100
    
    # Initialize particles and velocities
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    
    # Set first particle to initial values
    particles[0] = h0
    velocities[0] = np.random.uniform(-0.5, 0.5, n_points)
    
    # Initialize others randomly
    for i in range(1, swarm_size):
        particles[i] = np.random.random(n_points) * 0.5
        velocities[i] = np.random.uniform(-0.5, 0.5, n_points)
        
    # Project initial particles to feasible region
    for i in range(swarm_size):
        particles[i] = project_h(particles[i], dx)
    
    # Track personal bests
    personal_best_h = particles.copy()
    personal_best_C5 = np.full(swarm_size, np.inf)
    
    # Initialize global best
    global_best_h = particles[0].copy()
    global_best_C5 = evaluate_h(global_best_h, dx)
    
    # PSO params
    w, c1, c2 = 0.7, 1.5, 1.5
    
    iter_count = 0
    while time.time() - start_time < budget_s:
        iter_count += 1
        # Evaluate all particles
        for i in range(swarm_size):
            C5 = evaluate_h(particles[i], dx)
            if C5 < personal_best_C5[i]:
                personal_best_C5[i] = C5
                personal_best_h[i] = particles[i].copy()
            
            if C5 < global_best_C5:
                global_best_C5 = C5
                global_best_h = particles[i].copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(n_points), np.random.random(n_points)
            # Update velocity
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best_h[i] - particles[i]) +
                c2 * r2 * (global_best_h - particles[i])
            )
            # Update position
            particles[i] += velocities[i]
            # Project to feasible set
            particles[i] = project_h(particles[i], dx)
    
    # Final evaluation of the best solution
    final_C5 = evaluate_h(global_best_h, dx)
    
    return (global_best_h.tolist(), final_C5, n_points)
