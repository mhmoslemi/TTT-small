import numpy as np
from scipy import optimize
import time

def evaluate_h(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    # Project h to [0,1] and scale to sum to n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

    best_h = h.copy()
    best_c5 = evaluate_h(h, n_points)
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Evaluate current state
        current_c5 = evaluate_h(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        
        # Find maximum lag
        max_idx = np.argmax(corr)
        max_lag = max_idx - (len(h) - 1)
        max_corr = corr[max_idx]
        
        # Compute gradient w.r.t. h[0] at max_lag
        grad = np.zeros(n_points)
        
        # For lag k, we consider terms h[i](1-h[i+k])
        # Gradient w.r.t. h[i] = (1 - h[i+k]) if i+k is valid, 0 otherwise
        # Also gradient w.r.t. h[i+k] = -h[i] if i+k is valid, 0 otherwise
        # We only focus on the dominant contribution to reduce dimensionality
        
        # Gradient for the maximum contributing lag (max_lag)
        if max_lag >= 0:
            # Update h[i] terms: d/dh[i] = (1 - h[i + max_lag])
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
            # Update h[i + max_lag] terms: d/dh[i + max_lag] = -h[i]  
            for i in range(n_points - max_lag):
                grad[i + max_lag] -= h[i]
        else:  # max_lag < 0
            # max_lag is negative: update h[i] terms (1-h[i+abs(max_lag)])
            abs_lag = -max_lag
            for i in range(n_points - abs_lag):
                grad[i] += 1 - h[i + abs_lag]
            # Update h[i - abs_lag] term (positive indices only)
            for i in range(abs_lag, n_points):
                grad[i] -= h[i - abs_lag]
                
        # Update h using gradient descent
        h_new = h - learning_rate * grad
        
        # Project the new point back into feasible region  
        h = project_h(h_new, n_points)
        
        # Early stopping if improvement is too small
        if abs(current_c5 - best_c5) < tol:
            break
            
    # Final confirmation
    final_c5 = evaluate_h(best_h, n_points)
    
    return (best_h, final_c5, n_points)
