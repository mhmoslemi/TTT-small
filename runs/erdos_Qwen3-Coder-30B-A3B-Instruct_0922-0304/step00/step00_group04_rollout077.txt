def evaluate_c5(h, dx):
    # compute overlaps C5 = max_k over all valid lag k
    corr = correlate(h, 1 - h, mode='full') 
    # Note corr has len 2*n - 1
    n = len(h)
    max_val = -np.inf
    
    # for lag k from -n+1 to n-1
    for lag_idx in range(len(corr)):
        k = lag_idx - (n - 1)
        valid_len = n - abs(k)
        if valid_len <= 0:
            continue
        # The corr value at that lag, scaled by dx
        integral_k = corr[lag_idx] * dx
        if integral_k > max_val:
            max_val = integral_k
    
    return max_val
