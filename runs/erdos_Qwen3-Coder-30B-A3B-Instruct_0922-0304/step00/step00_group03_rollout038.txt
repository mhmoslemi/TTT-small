import numpy as np
import cvxpy as cp
from time import perf_counter

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial solution if available
    try:
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    except NameError:
        # Construct a reasonable starting solution
        n_points = 70
        h_init = np.ones(n_points) * 0.5
        # Make it more structured
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
        # Ensure sum constraint
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Binary search setup
    t_low = 0.0
    t_high = 0.5
    tolerance = 1e-4
    elapsed = 0.0
    start_time = perf_counter()
    
    best_h = None
    best_c5 = float('inf')
    
    while t_high - t_low > tolerance and elapsed < budget_s * 0.95:
        t_mid = (t_low + t_high) / 2.0
        elapsed = perf_counter() - start_time
        
        # Create variables: h and w for products
        h = cp.Variable(n_points)
        w = {}
        constraints = []
        
        # Bounds and sum constraint
        constraints.extend([h >= 0, h <= 1])
        constraints.append(cp.sum(h) == n_points/2)
        
        # Add product constraints
        half_len = n_points // 2 + 1
        for k in range(-half_len, half_len):
            if k == 0:
                continue
            k_abs = abs(k)
            shift = 0 if k > 0 else k
            for i in range(n_points - k_abs):
                if k > 0:
                    idx1, idx2 = i, i + k
                else:
                    idx1, idx2 = i - k, i
                if idx2 >= n_points:
                    continue
                    
                w_key = (i, k)
                w[w_key] = cp.Variable()
                # w <= h[i] and w <= h[i+k]
                constraints.append(w[w_key] <= h[i])
                constraints.append(w[w_key] <= h[idx2])
                # w >= h[i] + h[i+k] - 1
                constraints.append(w[w_key] >= h[i] + h[idx2] - 1)
        
        # Add constraints for each shift k to ensure overlapping integral <= t_mid
        for k in range(-half_len, half_len):
            if k == 0:
                continue
            k_abs = abs(k)
            shift = 0 if k > 0 else k
            expr_k = 0
            for i in range(n_points - k_abs):
                if k > 0:
                    idx1, idx2 = i, i + k
                else:
                    idx1, idx2 = i - k, i
                if idx2 >= n_points:
                    continue
                w_key = (i, k)
                expr_k += h[i] - w[w_key]
            
            # Add constraint
            constraints.append(expr_k <= t_mid)
        
        # Solve feasibility problem
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False, time_limit=budget_s * 0.95 - elapsed)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h.value
                if h_sol is not None:
                    h_round = np.round(h_sol).clip(0, 1)
                    # Repair sum if needed
                    while abs(np.sum(h_round) - n_points/2) > 1e-6:
                        diff = int(np.sum(h_round) - n_points/2)
                        if diff > 0:
                            # Need to decrease
                            idxs = np.where(h_round == 1)[0]
                            if len(idxs) > 0:
                                idx = np.random.choice(idxs)
                                h_round[idx] = 0
                        else:
                            # Need to increase
                            idxs = np.where(h_round == 0)[0]
                            if len(idxs) > 0:
                                idx = np.random.choice(idxs)
                                h_round[idx] = 1
                    # Evaluate
                    c5 = max(np.correlate(h_round, 1-h_round, mode="full")) / n_points
                    if c5 < best_c5:
                        best_h = h_round.copy()
                        best_c5 = c5
                    t_high = t_mid
                else:
                    t_low = t_mid
            else:
                t_low = t_mid
        except Exception:
            t_low = t_mid
    
    # Use final search result
    if best_h is None:
        # fallback, just round the initial
        h_final = np.round(h_init).clip(0, 1)
        diff = int(np.sum(h_final) - n_points/2)
        if diff != 0:
            for i in range(abs(diff)):
                idx = np.random.choice(np.where(h_final == (1 if diff < 0 else 0))[0])
                h_final[idx] = 1 - h_final[idx]
        best_h = h_final
        best_c5 = max(np.correlate(best_h, 1-best_h, mode="full")) / n_points
    
    return (list(best_h), best_c5, n_points)
