import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # sum(h) * dx = 1 => sum(h) = n_points/2
    
    # Initialize h from existing construction or create a balanced one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
    else:
        # Create a balanced initial
        h = np.zeros(n_points)
        h[:n_points//4] = 1.0
        h[n_points//4:3*n_points//4] = 0.5
        h[3*n_points//4:] = 0.0
        # Rescale to match constraint
        current_sum = np.sum(h)
        h *= target_sum / current_sum
    
    # Ensure that sum matches exactly
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        h *= target_sum / h_sum
    
    # Function to compute C5
    def compute_C5(h_local):
        # Compute convolution: h * (1 - h)
        conv = signal.correlate(h_local, 1 - h_local, mode="full") * dx
        return np.max(conv)
    
    # Current best
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Simulated Annealing parameters
    T_max = 1.0
    T_min = 1e-4
    n_iter = int(1e4)  # estimated steps
    alpha = np.exp(np.log(T_min / T_max) / n_iter)
    T = T_max
    
    # Step size for perturbations
    delta = 0.1
    
    # Parameters for adaptive delta
    prev_accept = 0
    accept_count = 0
    total_iter = 0
    
    start_time = time.time()
    
    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
        total_iter += 1
        
        # Create a new candidate h
        h_new = h.copy()
        
        # Random perturbation: either a swap or adjacent shift
        if np.random.random() < 0.5:
            # Swap two elements
            idx1, idx2 = np.random.randint(0, n_points, 2)
            h_new[idx1], h_new[idx2] = h_new[idx2], h_new[idx1]
        else:
            # Adjacent shift
            idx = np.random.randint(0, n_points-1)
            shift = np.random.uniform(-delta, delta)
            h_new[idx] += shift
            h_new[idx+1] -= shift
            
        # Clip and rescale to keep in bounds and satisfy constraint
        h_new = np.clip(h_new, 0, 1)
        h_sum_new = np.sum(h_new)
        if abs(h_sum_new - target_sum) > 1e-10:
            h_new *= target_sum / h_sum_new
            
        # Compute new cost
        new_C5 = compute_C5(h_new)
        
        # Accept or reject
        delta_C = new_C5 - best_C5
        if delta_C < 0 or np.random.random() < np.exp(-delta_C / T):
            h = h_new.copy()
            if new_C5 < best_C5:
                best_h = h.copy()
                best_C5 = new_C5
                
        # Update temperature
        T *= alpha
        
        # Adapt delta based on acceptance rate
        if total_iter % 100 == 0:
            accept_rate = accept_count / 100.0 if total_iter > 100 else 0.5
            if accept_rate < 0.2:
                delta *= 0.9
            elif accept_rate > 0.5:
                delta *= 1.1
            accept_count = 0
        
    return (best_h.tolist(), best_C5, n_points)
