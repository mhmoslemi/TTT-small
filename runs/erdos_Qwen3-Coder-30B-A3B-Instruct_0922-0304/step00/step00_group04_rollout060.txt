import numpy as np
from scipy.signal import correlate
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 42
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
    else:

        h_initial = np.ones(n_points) * 0.5
        h_initial[:n_points//2] = 1.0
        h_initial[n_points//2:] = 0.0
    
    # Normalize to correct sum
    h_initial = h_initial * target_sum / np.sum(h_initial)
    
    # Best solution found so far
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    remaining_budget = budget_s
    
    # Use MILP to optimize
    # Variables: h[i] in [0,1], and C5 as objective
    h = cp.Variable(n_points, name="h")
    C5 = cp.Variable(name="C5")
    
    # Constraint: sum(h) = target_sum
    constraints = [cp.sum(h) == target_sum]
    
    # Add bound constraints
    constraints.extend([h[i] >= 0 for i in range(n_points)])
    constraints.extend([h[i] <= 1 for i in range(n_points)])
    
    # Lag indices
    lag_max = n_points - 1
    
    # Add constraints for each lag
    # We need: sum h[i] * (1 - h[i + k]) <= C5/dx for all valid k
    # This is equivalent to: sum h[i] - sum h[i] * h[i+k] <= C5/dx
    # Or: sum h[i] - sum h[i+k] * h[i] <= C5/dx
    
    # For each lag, compute the paired indices
    for k in range(-lag_max, lag_max + 1):
        # Compute the range of valid indices for this lag
        if k >= 0:
            valid_indices = range(n_points - k)
        else:
            valid_indices = range(-k, n_points)
        
        if valid_indices:
            corr_expr = 0
            for i in valid_indices:
                corr_expr += h[i] * (1 - h[i + k])
            
            constraints.append(corr_expr <= C5 / dx)
    
    # Objective: minimize C5
    objective = cp.Minimize(C5)
    
    # Create problem
    prob = cp.Problem(objective, constraints)
    
    # Solve repeatedly with different solvers to improve chances of finding
    # a better solution. We'll use 'GLPK_MI' which is good for small MILPs.
    solver = 'GLPK_MI'
    solver_options = {}
    
    # Time limit per solve
    solve_time_limit = remaining_budget / 10  # Half of budget to be safe
    
    try:
        prob.solve(solver=solver, verbose=False, time_limit=solve_time_limit)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = h.value
            c5_opt = C5.value
            
            # Round the solution to binary values, keeping sum target_sum
            # To ensure sum remains target_sum, we preserve top target_sum elements
            # as 1, and set rest to 0
            h_binary = np.zeros(n_points)
            sorted_indices = np.argsort(h_opt)[::-1]
            h_binary[sorted_indices[:int(target_sum)]] = 1.0
            
            # Re-normalize to ensure sum equals target_sum
            h_binary = h_binary * target_sum / np.sum(h_binary)
            
            c5_binary = evaluate_erdos_solution(h_binary)
            
            if c5_binary < best_c5:
                best_c5 = c5_binary
                best_h = h_binary
                
    except Exception as e:
        pass  # We'll continue to use the initial or other fallbacks
    
    # Simple local search in case we can improve further
    def local_improve(h_start, max_iter=100):
        h_best = h_start.copy()
        c5_best = evaluate_erdos_solution(h_best)
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            # Small random changes
            h_trial = h_best.copy()
            # Mutate randomly
            i = np.random.randint(0, n_points)
            # Small change
            change = np.random.uniform(-0.1, 0.1)
            h_trial[i] += change
            h_trial[i] = np.clip(h_trial[i], 0, 1)
            # Re-normalize
            h_trial = h_trial * target_sum / np.sum(h_trial)
            c5_trial = evaluate_erdos_solution(h_trial)
            if c5_trial < c5_best:
                c5_best = c5_trial
                h_best = h_trial
        return h_best, c5_best
    
    # Do a simple local improvement
    h_improve, c5_improve = local_improve(best_h, 50)
    if c5_improve < best_c5:
        best_c5 = c5_improve
        best_h = h_improve
    
    # Possibly use the initial solution anyway if it's better
    initial_c5 = evaluate_erdos_solution(h_initial)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = h_initial
    
    return (best_h.tolist(), best_c5, n_points)
