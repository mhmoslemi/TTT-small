import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        # Fallback: create binary-like vector
        n_points = 45
        h = np.ones(n_points)
        h[n_points//2:] = 0
        # Scale to have sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_time = start_time + budget_s
    
    # Function to evaluate C5
    def evaluate_C5(h_vals):
        h_vals = np.asarray(h_vals)
        # compute overlap for all shifts
        overlap = np.correlate(h_vals, 1-h_vals, mode="full")
        # adjust for discretization
        overlap = overlap * dx
        return np.max(overlap)
    
    # Function to project h to feasible set
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        if h_sum > 0:
            h_vals = h_vals * (n_points / 2) / h_sum
        return h_vals
    
    # Gradient-based optimization
    best_h = h.copy()
    best_c5 = evaluate_C5(best_h)
    
    # Gradient descent setup
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 0
    max_patience = 100
    
    def compute_gradient(h_vals, k):
        # Compute gradient of overlap at shift k
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            j = (i + k) % n_points
            grad[i] = (1 - h_vals[j]) - (1 - h_vals[i])  # Simplified approx
        return grad
    
    iteration = 0
    while time.time() < target_time:
        if iteration % 500 == 0:
            # Small random perturbation to escape local minima
            perturb = np.random.normal(0, 0.01, n_points)
            h = project_h(h + perturb)
            patience = 0

        # Evaluate overlaps for all shift values
        overlap_full = np.correlate(h, 1-h, mode="full")
        overlap_full = overlap_full * dx
        max_k = np.argmax(overlap_full)
        
        # Compute gradient of active overlap
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] = 1 - h[j] 
        grad = grad * 2 * dx / n_points  # scaled approx gradient
        
        # Update h
        h_new = h - alpha * grad
        
        # Project
        h_new = project_h(h_new)
        
        # Evaluate new C5
        new_c5 = evaluate_C5(h_new)
        
        # Accept if improved
        if new_c5 < best_c5:
            h = h_new
            best_c5 = new_c5
            patience = 0
            alpha = min(0.1, alpha * 1.01)  # slight increase if improving
        else:
            patience += 1
            if patience > max_patience:
                break
            alpha *= gamma  # reduce step size
            if alpha < alpha_min:
                break
        
        iteration += 1
        
        # Check early stopping
        if best_c5 < 0.3807:
            break
            
        if time.time() > target_time - 0.1:  # leave margin for final checks
            break
    
    # Final evaluation
    final_c5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
