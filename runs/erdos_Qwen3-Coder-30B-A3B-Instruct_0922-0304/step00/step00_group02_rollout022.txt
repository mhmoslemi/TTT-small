import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a binary one

        h = initial_h_values.copy()
        n_points = len(h)
    except NameError:
        n_points = 45
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
    
    # Normalize so sum = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    # Compute initial overlap
    def compute_c5(h_vals):
        dx = 2.0 / n_points
        overlaps = correlate(h_vals, 1 - h_vals, mode="full") * dx
        return np.max(overlaps)
    
    c5_bound = compute_c5(h)
    
    # Gradient-based optimization
    max_iter = 10000
    alpha = 0.1
    alpha_min = 1e-6
    gamma = 0.95
    patience = 0
    best_c5 = c5_bound
    best_h = h.copy()
    
    # Main loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s * 0.98:
            break
            
        # Compute overlaps
        dx = 2.0 / n_points
        overlaps = correlate(h, 1 - h, mode="full") * dx
        
        # Find k that maximizes overlap
        k_max = np.argmax(overlaps)
        k_max = k_max - (len(overlaps) - 1) // 2  # Convert full index back to shift
        
        # Gradient computation
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad[i] = (1 - h[j]) - h[i]  # Derivative w.r.t h[i]
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to desired sum
        h_new *= (n_points / 2) / np.sum(h_new)
        
        # Check if new h decreases C5
        new_c5 = compute_c5(h_new)
        
        if new_c5 < c5_bound:
            c5_bound = new_c5
            h = h_new.copy()
            patience = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            patience += 1
            alpha *= gamma
            
        # Reduce learning rate if no improvement
        if patience > 50:
            patience = 0
            alpha *= 0.9
            
        # Adaptive termination
        if alpha < alpha_min or new_c5 < 0.3807:
            break
            
        if iteration % 500 == 0 and iteration > 0:
            # Small random perturbations to escape local minima
            perturb = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + perturb, 0, 1)
            h *= (n_points / 2) / np.sum(h)
    
    # Final validation and return
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
