import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic distribution
        n_points = 42
        h = np.ones(n_points) * 0.5
        h[0] = 0.4  # slightly adjust to make it non-uniform
        h[1] = 0.6
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure initial sum is correct
    h_sum = np.sum(h)
    if h_sum != target_sum:
        h = h * target_sum / h_sum
    
    # Precompute fixed FFT data
    fft_len = 2 * n_points - 1
    fft_freq = np.fft.fftfreq(fft_len, d=dx)
    
    def evaluate(h_vec):
        # Compute cross-correlation and find maximum
        h_full = np.concatenate([h_vec, np.zeros(n_points - 1)])
        h_conj = np.concatenate([1 - h_vec, np.zeros(n_points - 1)])
        corr = np.correlate(h_full, h_conj, mode='full')
        max_corr = np.max(corr[:n_points])
        return max_corr * dx
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Scale to correct sum
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def optimize_step(h_vec):
        # Use frequency domain trick
        h_fft = fft.fft(h_vec, fft_len)
        h_conj_fft = fft.fft(1 - h_vec, fft_len)
        
        # Compute power spectrum and update
        power_spectrum = np.abs(h_fft) ** 2
        # Try to flatten the spectrum while preserving DC component
        # Normalize to avoid overflows
        power_spectrum = np.clip(power_spectrum, 1e-10, np.max(power_spectrum))
        # We want to make it flat, so we invert
        weighting = 1.0 / np.sqrt(power_spectrum)
        # Clip high frequencies for smoothing
        weighting = np.clip(weighting, 0, 10)
        
        # Apply smoothing to reduce high-frequency noise
        # Smooth by averaging with neighbors
        smooth_weight = np.convolve(weighting, np.ones(3)/3, mode='same')
        
        new_h_fft = h_fft * smooth_weight
        
        # Inverse FFT
        h_new = np.real(fft.ifft(new_h_fft))
        h_new = np.concatenate([h_new[:n_points], np.zeros(n_points - 1)])
        h_new = np.real(fft.ifft(new_h_fft, fft_len))[:n_points]
        
        # Project H back to feasible space
        h_new = project_feasible(h_new)
        return h_new
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    start_time = time.time()
    elapsed = 0
    
    step = 0
    
    while elapsed < budget_s * 0.95:
        step += 1
        # Alternate approach: simple gradient descent on shifted versions
        h_new = optimize_step(best_h)
        h_new = project_feasible(h_new)
        
        c5_new = evaluate(h_new)
        elapsed = time.time() - start_time
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Stop if already close to target
        if best_c5 < 0.3807:
            break
    
    # Final validation and return
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
