import numpy as np
import scipy.optimize as opt
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 47
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Ensure the constraint sum(h) * dx = 1
    h = initial_h.copy()
    h_sum = np.sum(h)
    h *= (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-start approach
    initial_guesses = [h]
    for _ in range(4):
        perturbed = h + np.random.normal(0, 0.005, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed_sum = np.sum(perturbed)
        perturbed *= (n_points / 2) / perturbed_sum
        initial_guesses.append(perturbed)
    
    for h_start in initial_guesses:
        h = h_start.copy()
        last_improvement = 0
        iter_count = 0
        
        while time.time() - start_time < budget_s - 1 and iter_count < 1000:
            # Compute overlaps
            overlap_full = np.correlate(h, 1 - h, mode="full")
            overlap = overlap_full[n_points - 1:]
            k_max = np.argmax(overlap)
            c5 = overlap[k_max] * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = 0
            
            # Compute gradient for worst shift
            shifted = np.roll(1 - h, k_max)
            grad = shifted - h * shifted  # element-wise gradient
            
            # Normalize gradient
            grad_norm = np.linalg.norm(grad)
            if grad_norm > 1e-10:
                grad_normed = grad / grad_norm
            else:
                iter_count += 1
                continue
            
            # Backtracking line search
            alpha = 0.01
            rho = 0.5
            max_steps = 30
            step_ok = False
            
            for _ in range(max_steps):
                h_new = h - alpha * grad_normed
                h_new = np.clip(h_new, 0, 1)
                h_new_sum = np.sum(h_new)
                h_new *= (n_points / 2) / h_new_sum
                
                # Evaluate new objective
                overlap_new_full = np.correlate(h_new, 1 - h_new, mode="full")
                overlap_new = overlap_new_full[n_points - 1:]
                c5_new = overlap_new[k_max] * dx
                
                if c5_new < c5:
                    h = h_new
                    step_ok = True
                    break
                else:
                    alpha *= rho
            
            if not step_ok:
                h = h_new  # Force update even if not better
                
            # Convex refinement every 50 iterations
            if iter_count % 50 == 0 and iter_count > 0:
                # Solve convex optimization for k_max
                var_h = Variable(n_points)
                constraints = [var_h >= 0, var_h <= 1, sum(var_h) == n_points / 2]
                
                # Overlap for the dominant shift k_max
                expr = 0
                for i in range(n_points):
                    j = (i - k_max) % n_points
                    expr += var_h[i] * (1 - var_h[j])
                
                prob = Problem(Minimize(expr), constraints)
                try:
                    prob.solve(solver=ECOS, verbose=False)
                    if prob.status in [ OPTIMAL, OPTIMAL_INACCURATE ]:
                        h_cvx = np.array([var_h[i].value for i in range(n_points)])
                        if np.any(np.isnan(h_cvx)):
                            pass
                        else:
                            h = h_cvx
                except:
                    pass
            
            iter_count += 1
            
            if last_improvement > 200:
                break
            last_improvement += 1
    
    # Final evaluation
    overlap_full = np.correlate(best_h, 1 - best_h, mode="full")
    overlap = overlap_full[n_points - 1:]
    final_c5 = np.max(overlap) * dx
    
    return (best_h.tolist(), final_c5, n_points)
