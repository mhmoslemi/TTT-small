import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 6000  # Rough estimate for 600s budget
    
    # PSO parameters
    inertia = 0.7
    cognitive = 1.5
    social = 1.5
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        particles = np.array([initial_h_values] + [np.random.rand(n_points) for _ in range(swarm_size-1)])
    else:
        particles = [np.random.rand(n_points) for _ in range(swarm_size)]
    
    # Scale to correct sum
    for i in range(swarm_size):
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    velocities = [np.random.uniform(-0.5, 0.5, n_points) for _ in range(swarm_size)]
    
    # Personal best
    pbest = [p.copy() for p in particles]
    pbest_c5 = [float('inf')] * swarm_size
    
    # Global best
    gbest_idx = 0
    gbest = particles[0].copy()
    gbest_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Evaluate current particle
            particle = particles[i]
            # Compute overlaps using convolution
            correlation = signal.correlate(particle, 1 - particle, mode='full')
            # Extract relevant part - full convolution is 2*n_points-1 long
            overlap_values = correlation[len(particle)-1:] * dx
            c5 = np.max(overlap_values)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particle.copy()
                pbest_c5[i] = c5
            
            # Update global best
            if c5 < gbest_c5:
                gbest = particle.copy()
                gbest_c5 = c5
                gbest_idx = i
        
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (
                inertia * velocities[i] +
                cognitive * r1 * (pbest[i] - particles[i]) +
                social * r2 * (gbest - particles[i])
            )
            
            particles[i] += velocities[i]
            
            # Projection to feasible region
            # Clip to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            # Scale to sum to n_points / 2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation of the best solution
    c5_bound = gbest_c5
    
    return (gbest, c5_bound, n_points)
