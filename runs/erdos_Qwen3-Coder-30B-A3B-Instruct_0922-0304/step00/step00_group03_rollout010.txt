import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Define parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Get initial construction if available
    try:
        h0 = initial_h_values.copy()
        # Normalize to correct sum
        h0 = h0 * target_sum / np.sum(h0)
    except:
        # Generate random starting point
        h0 = np.random.rand(n_points)
        h0 = h0 * target_sum / np.sum(h0)
        
    # Clamp to [0,1]
    h0 = np.clip(h0, 0, 1)
    
    # Binary search for minimum C_5
    t_low = 0.0
    t_high = 0.5
    tolerance = 1e-4
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    while t_high - t_low > tolerance and time() - start_time < budget_s - 5:
        t_mid = (t_low + t_high) / 2.0
        
        # Define optimization variables
        h = cp.Variable(n_points)
        w = cp.Variable((n_points, n_points))  # w[i, k] = h[i] * h[i+k]
        
        # Constraints
        constraints = [
            cp.sum(h) == target_sum,
            h >= 0,
            h <= 1
        ]
        
        # Add constraints for each shift k
        for k in range(-n_points + 1, n_points):
            # For shift k, need to compute correct indices
            if k >= 0:
                # h[i] * h[i+k] for i in range(0, n_points-k)
                expr_k = cp.sum([h[i] * h[i + k] for i in range(n_points - k)]) 
            else:
                # shift is negative; h[i] * h[i+k] for i in range(-k, n_points)
                expr_k = cp.sum([h[i] * h[i + k] for i in range(-k, n_points)])
            
            # This is a tighter approach for evaluating NOverlap
            # The key idea is that integral can be evaluated via discrete
            # correlation computation with reversed h.
            # Use linear relaxation: we can directly add constraints for 
            # individual constraints on correlations.
            
            # A better way: build it directly
            # We are computing sum of h(x)(1 - h(x + k)) 
            # So we compute sum of h[i]*(1 - h[i+k]) for valid pairs
            
            expr_total = 0
            count = 0
            for i in range(n_points):
                i_k = i + k
                if 0 <= i_k < n_points:
                    expr_total += h[i] * (1 - h[i_k])
                    count += 1
            
            # Add constraint: this sum should be at most t_mid
            constraints.append(expr_total <= t_mid)
            # Inflate equality to <= for relaxation
        
        # Set up the problem (we want feasibility, so just check
        # if there exists h satisfying all constraints)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.GLPK, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                # feasible solution
                h_val = h.value
                h_val = np.maximum(0, np.minimum(1, h_val))
                c5_test = evaluate_erdos_solution(h_val)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_val.copy()
                t_high = t_mid
            else:
                t_low = t_mid
        except:
            t_low = t_mid
            
    # Final evaluation
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, best_c5, n_points)
