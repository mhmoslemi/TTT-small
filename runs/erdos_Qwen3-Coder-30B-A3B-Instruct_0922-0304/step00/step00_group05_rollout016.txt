import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral C5 for given h."""
    # Compute correlations for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integrals are the correlation values scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasibility(h, dx):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h)*dx = 1."""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to unit sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (2.0 / (dx * h_sum))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available, otherwise create a random one
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    h0 = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
    
    # Initialize parameters
    dx = 2.0 / n_points
    swarm_size = 100
    max_iters = 10000
    inertia = 0.7
    cognitive = 1.5
    social = 1.5

    # Initialize swarm
    particles = np.random.rand(swarm_size, n_points) * 0.2 + 0.4  # Initial random particles around 0.5
    if 'initial_h_values' in globals():
        particles[0] = h0  # First particle is the initial solution
    for i in range(swarm_size):
        particles[i] = project_to_feasibility(particles[i], dx)
    
    velocities = np.random.rand(swarm_size, n_points) * 0.5 - 0.25  # Random velocities in [-0.25, 0.25]
    
    # Track best solution
    best_particle_idx = 0
    best_C5 = evaluate_h(particles[0], dx)
    
    # PSO main loop
    for it in range(max_iters):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate all particles
        for p in range(swarm_size):
            particle_c5 = evaluate_h(particles[p], dx)
            if particle_c5 < best_C5:
                best_C5 = particle_c5
                best_particle_idx = p
        
        # Update particles
        for p in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[p] = (
                inertia * velocities[p] +
                cognitive * r1 * (particles[best_particle_idx] - particles[p]) +
                social * r2 * (particles[best_particle_idx] - particles[p])
            )
            
            # Update position
            particles[p] += velocities[p]
            
            # Project to feasibility
            particles[p] = project_to_feasibility(particles[p], dx)
    
    # Return best solution
    best_h = particles[best_particle_idx]
    return (best_h.tolist(), best_C5, n_points)
