import numpy as np
import time
from scipy import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial solution
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values).copy()
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h0 = h0 * target_sum / np.sum(h0)
    
    # Fitness function
    def compute_c5(h):
        # Correlation
        corr = correlate(h, 1 - h, mode="full")
        # Find maximum
        max_corr = np.max(corr)
        return max_corr * dx
    
    def is_feasible(h):
        return (np.all(h >= 0) and np.all(h <= 1) and 
                abs(np.sum(h) - target_sum) < 1e-10)
    
    # Initialize
    population_size = 100
    population = []
    fitness_values = []
    
    # Generate initial population
    for _ in range(population_size):
        h = h0.copy()
        # Random perturbations on initial solution
        if np.random.rand() > 0.5:
            indices = np.random.choice(n_points, size=max(1, n_points//5), replace=False)
            for i in indices:
                h[i] += np.random.normal(0, 0.1)
        # Clip and rescale to maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        population.append(h)
        fitness_values.append(compute_c5(h))
    
    # Main evolution loop
    start_time = time.time()
    best_h = population[np.argmin(fitness_values)].copy()
    best_c5 = min(fitness_values)
    no_improvement_count = 0
    max_no_improvement = 100
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitness_values)
        population = [population[i] for i in sorted_indices]
        fitness_values = [fitness_values[i] for i in sorted_indices]
        
        new_best_c5 = fitness_values[0]
        if new_best_c5 < best_c5:
            best_c5 = new_best_c5
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Create new population
        new_population = [population[0].copy()]  # Keep best
        new_fitness = [fitness_values[0]]
        
        # Selection, crossover, mutation
        while len(new_population) < population_size:
            parent1_idx = np.random.randint(0, population_size // 2)
            parent2_idx = np.random.randint(0, population_size // 2)
            
            # Crossover (uniform, but simple two-point)
            h1 = population[parent1_idx].copy()
            h2 = population[parent2_idx].copy()
            
            # Simple random crossover
            crossover_point1 = np.random.randint(1, n_points)
            crossover_point2 = np.random.randint(crossover_point1, n_points)
            
            child1 = np.concatenate([h1[:crossover_point1], h2[crossover_point1:crossover_point2], h1[crossover_point2:]])
            child2 = np.concatenate([h2[:crossover_point1], h1[crossover_point1:crossover_point2], h2[crossover_point2:]])
            
            # Mutation (1% of genes)
            for child in [child1, child2]:
                if np.random.rand() < 0.1:
                    num_mutations = max(1, n_points // 100)
                    indices = np.random.choice(n_points, size=num_mutations, replace=False)
                    for i in indices:
                        child[i] += np.random.normal(0, 0.05)
                
                # Clip values
                child = np.clip(child, 0, 1)
                # Rescale to correct sum
                child = child * target_sum / np.sum(child)
                
                new_population.append(child)
                new_fitness.append(compute_c5(child))
        
        population = new_population[:population_size]
        fitness_values = new_fitness[:population_size]
    
    return (best_h.tolist(), float(best_c5), n_points)
