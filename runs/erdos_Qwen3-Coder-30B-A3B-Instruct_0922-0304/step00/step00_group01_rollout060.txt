import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 1000
    line_search_max = 30
    alpha_init = 0.01
    rho = 0.5
    eta = 0.5
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.full(n_points, 0.5, dtype=float)
    
    # Ensure integral constraint
    h *= (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    def compute_overlap(h_vec):
        # Compute overlaps for all shifts
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        overlaps = conv[n_points-1:] * dx
        return overlaps
    
    def compute_gradient(h_vec, k_max):
        grad = np.zeros_like(h_vec)
        # For shift k_max, gradient is d/dh_i of sum h_j (1 - h_{j+k_max}) for all j
        # which gives 1 - h_{i+k_max} (when i+k_max < n) or 0 (out of bounds)
        # In the real gradient for C(k_max), we only consider i such that 
        # j = i + k_max is valid.
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += 1 - h_vec[i + k_max]
        return grad
    
    def project_feasible(h_vec):
        # Clip to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to match sum constraint
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec *= (n_points / 2) / current_sum
        return h_vec
    
    def line_search(h_vec, grad, k_max, alpha_init):
        alpha = alpha_init
        for _ in range(line_search_max):
            h_new = h_vec - alpha * grad
            h_new = project_feasible(h_new)
            overlaps_new = compute_overlap(h_new)
            C5_new = np.max(overlaps_new)
            
            if C5_new < np.max(compute_overlap(h_vec)) - eta * alpha * np.linalg.norm(grad)**2:
                return h_new, C5_new
            alpha *= rho
        return h_vec, np.max(compute_overlap(h_vec))
    
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        if iteration % 50 == 0:
            # Occasionally recompute current solution
            pass
            
        overlaps = compute_overlap(h)
        k_max = np.argmax(overlaps)
        C5_current = overlaps[k_max]
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if C5_current < target_bound:
            break
            
        if no_improvement_count > 100:
            break
            
        # Compute gradient
        grad = compute_gradient(h, k_max)
        norm = np.linalg.norm(grad)
        if norm < 1e-10:
            break
        grad = grad / norm
        
        # Line search
        h_new, C5_new = line_search(h, grad, k_max, alpha_init)
        h = h_new
        
        # Verify feasibility steadily
        h = project_feasible(h)
        
        iteration += 1
        
    # Final check
    overlaps_final = compute_overlap(best_h)
    final_C5 = np.max(overlaps_final)
    
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h.tolist(), best_C5, n_points)
