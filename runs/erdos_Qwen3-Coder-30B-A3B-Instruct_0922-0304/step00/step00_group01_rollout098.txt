import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    # Compute overlap integral
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for different shifts
    overlaps = []
    for k in range(n):
        # Circular correlation
        corr = np.correlate(h, np.ones_like(h) - h, mode='full')
        overlap = corr[n-1-k] * dx
        overlaps.append(overlap)
    return max(overlaps)

    # Precompute dx
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.01
    min_alpha = 1e-5
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    no_improvement_count = 0
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = []
        corr_full = signal.correlate(h, np.ones_like(h) - h, mode='full')
        for k in range(n_points):
            overlap = corr_full[n_points-1-k] * dx
            overlaps.append(overlap)
        
        k_max = np.argmax(overlaps)
        max_overlap = max(overlaps)
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Early stopping
        if best_c5 <= 0.3807:

                alpha *= 0.5
                no_improvement_count = 0
            else:

        h_shifted = np.roll(h, k_max)

        # Enforce sum constraint
        scaling = (n_points / 2) / np.sum(h)

    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
