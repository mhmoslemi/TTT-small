import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available

        n_points = len(h)
    except NameError:
        # Fallback: create a symmetric step function
        n_points = 47
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
    
    dx = 2.0 / n_points
    target_c5 = 0.3807
    
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(conv) * dx

    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec) * dx
        if abs(total - 1.0) > 1e-10:
            h_vec = h_vec * (1.0 / total)
        return h_vec

    def solve_subproblem(h_vec, k_max):
        # Convex relaxation to minimize C(k_max)
        n = len(h_vec)
        # Create the linear system for overlap C(k_max) = sum h[i] * (1 - h[i-k])
        # We build the constraint matrix for sum h[i] = 1 and bounds [0,1]
        # For minimizing C(k_max), we need to compute the contribution of each h[i]
        # The constraint to minimize is simply: h^T * A * h where A is based on 
        # the kernel 1-h[i-k]
        # Use quadratic programming approach
        # Minimize: a^T * h + 0.5 * h^T * H * h
        # Subject to: sum(h) = 1, 0 <= h[i] <= 1
        # Let a = - (1 - h[i-k]), and H = 0 (since it's linear)
        # So just minimize a^T * h with constraints
        # For exact computation we compute dC/dh[i]
        # Using finite differences in the note:
        # dC(k)/dh[i] = 1 - h[i-k] + h[i+k] (mod n)
        # We take a = - (1 - h[i-k]) = h[i-k] - 1
        
        c = np.zeros(n)
        if k_max == 0:
            c[:] = 0
        else:
            # (i - k) % n
            for i in range(n):
                j = (i - k_max) % n
                c[i] = 1 - h_vec[j]
        
        # Not strictly needed but setup for projection
        # We can just directly solve linear system with bounds
        # But above is a better way to handle gradient
        # Let's instead minimize C(k_max) using scipy.optimize
        
        def obj(hv):
            return np.sum(hv * c)
        
        # Constraints
        def constraint_sum(hv):
            return np.sum(hv) * dx - 1.0
            
        def constraint_bounds(hv):
            return np.concatenate([
                hv,
                1.0 - hv
            ])
        
        cons = [{'type': 'eq', 'fun': constraint_sum},
                {'type': 'ineq', 'fun': lambda hv: hv}, 
                {'type': 'ineq', 'fun': lambda hv: 1.0 - hv}]
        
        bounds = [(0, 1) for _ in range(n)]
        res = minimize(obj, h_vec, method='SLSQP', bounds=bounds, constraints=cons,
                       options={'maxiter': 200, 'ftol': 1e-10})
        if res.success:
            return res.x
        else:
            # fallback to projection
            return project_to_feasible(h_vec)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    iterations = 0
    max_iter = 100
    # We reduce to using gradient descent instead of convex subproblems 
    # since that's faster and simpler. We still use iterative approach.
    
    while time.time() - start_time < budget_s - 1.0 and iterations < max_iter:
        if best_c5 <= target_c5:
            break
            
        c5_vals = np.zeros(2 * n_points - 1)
        for k in range(1 - n_points, n_points):
            # Compute value of overlap integral at shift k
            k_mod = k % n_points
            h_shifted = np.roll(h, k_mod)
            c5_vals[k + n_points - 1] = np.sum(h * (1.0 - h_shifted)) * dx
            
        k_max = np.argmax(c5_vals)
        max_val = c5_vals[k_max]
        
        if max_val <= target_c5:
            break
        
        # Optimize h to reduce overlap at k_max
        # This is approximated as minimizing linear function
        # Using gradient descent approach on shifted and clipped versions
        
        h_grad = h.copy()
        k = k_max - (n_points - 1)
        # At shift k, effect of h[i] on C(k) is 1 - h[(i - k) % n]
        # So gradient = 1 - h[(i - k) % n]
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            prev_index = (i - k) % n_points
            grad[i] = 1 - h[prev_index]
        
        # Take small step
        step_size = 0.01
        h_new = h - step_size * grad
        
        h_new = project_to_feasible(h_new)
        c5_attempt = compute_c5(h_new)
        if c5_attempt < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_attempt
            
        h = h_new.copy()
        iterations += 1
        
        if iterations % 10 == 0:
            if best_c5 <= target_c5:
                break

    # Final validation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
