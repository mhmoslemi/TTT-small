import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 42
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values)
    
    # Define objective function using correlation
    def compute_C5(h):
        # Normalize h to sum to n_points/2
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode="full")
        # Return max correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Gradient-based optimization
    def objective(h):
        return compute_C5(h)
    
    # Constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Initial guess
    h_init = initial_h.copy()
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    constraints = [
        {'type': 'eq', 'fun': constraint_sum}
    ]
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Perform optimization
    res = minimize(objective, h_init, method='SLSQP', bounds=bounds, constraints=constraints, 
                   options={'maxiter': 1000, 'ftol': 1e-6})
    
    best_h = res.x
    
    # Re-normalize
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final computation of C5
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
