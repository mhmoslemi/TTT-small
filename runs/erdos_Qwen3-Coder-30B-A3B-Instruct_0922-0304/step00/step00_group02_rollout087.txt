import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlaps)
