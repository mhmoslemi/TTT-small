import numpy as np
from scipy import correlate
import time

def evaluate_h(h, dx):
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Maximize over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial construction

    h = h / np.sum(h) * target_sum
    
    # Ensure constraints
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - target_sum) < 1e-10
    
    # Evaluate initial
    current_c5 = evaluate_h(h, dx)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 100000
    delta = 0.1
    decay_rate = 0.9
    decay_interval = 100
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Randomly select two indices
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Perturbation amount
        perturbation = np.random.uniform(-delta, delta)
        
        # Ensure feasibility
        if h[i] + perturbation < 0 or h[j] - perturbation < 0:
            continue
        if h[i] + perturbation > 1 or h[j] - perturbation > 1:
            continue
        
        # Apply perturbations
        h_new = h.copy()
        h_new[i] += perturbation
        h_new[j] -= perturbation
        
        # Adjust to maintain sum constraint
        diff = np.sum(h_new) - target_sum
        if abs(diff) > 1e-10:
            # Adjust the first element to restore sum
            h_new[0] -= diff
            
        # Ensure new values are within [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint again
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Evaluate
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
                
        # Decay learning rate periodically
        if (iteration + 1) % decay_interval == 0:
            delta *= decay_rate
    
    # Final validation
    final_c5 = evaluate_h(best_h, dx)
    
    return (list(best_h), final_c5, n_points)
