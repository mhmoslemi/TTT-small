        # Compute overlap integral using full convolution
        h_pad = np.concatenate([h_vals, np.zeros(n_points-1)])
        one_minus_h = 1 - h_vals
        one_minus_h_pad = np.concatenate([one_minus_h, np.zeros(n_points-1)])
        correlation = np.correlate(h_pad, one_minus_h_pad, mode="full")
        # Only valid overlaps are from shifts 0 to n_points-1
        overlaps = correlation[n_points-1:] * dx
        return np.max(overlaps)

        return np.sum(h_vals) - target_sum
