import numpy as np
from scipy import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def repair_sum(h, target_sum):
    """Ensure h has exactly target_sum ones"""
    current_sum = np.sum(h)
    if current_sum == target_sum:
        return h.copy()
    
    h = h.copy()
    if current_sum > target_sum:
        # Need to set some 1s to 0s
        indices_1 = np.where(h == 1)[0]
        to_flip = int(current_sum - target_sum)
        flip_indices = np.random.choice(indices_1, to_flip, replace=False)
        h[flip_indices] = 0
    else:
        # Need to set some 0s to 1s
        indices_0 = np.where(h == 0)[0]
        to_flip = int(target_sum - current_sum)
        flip_indices = np.random.choice(indices_0, to_flip, replace=False)
        h[flip_indices] = 1
        
    return h

def generate_neighbor(h, n_points, target_sum):
    """Generate a neighbor by flipping 2-3 bits"""
    h_new = h.copy()
    num_flips = random.randint(2, 3)
    indices = list(range(n_points))
    flip_indices = random.sample(indices, num_flips)
    
    for i in flip_indices:
        h_new[i] = 1 - h_new[i]
    
    return repair_sum(h_new, target_sum)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction or create a binary one
    if 'initial_h_values' in globals():

        h = (h > 0.5).astype(int)
    else:
        n_points = 70
        h = np.random.choice([0, 1], size=n_points, p=[0.5, 0.5])
    
    # Ensure binary and correct sum
    target_sum = int(len(h) / 2)
    h = repair_sum(h, target_sum)
    
    dx = 2.0 / len(h)
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Tabu search parameters
    tabu_tenure = 100
    tabu_list = []
    max_iter = 1000
    no_improvement_limit = max_iter // 4
    
    # Timing tracking
    elapsed = time.time() - start_time
    iter_count = 0
    last_improvement_iter = 0
    
    while elapsed < budget_s and iter_count < max_iter:
        elapsed = time.time() - start_time
        iter_count += 1
        
        # Generate neighbor
        h_neighbor = generate_neighbor(h, len(h), target_sum)
        
        # Skip if in tabu list
        neighbor_tuple = tuple(h_neighbor)
        if neighbor_tuple in tabu_list:
            continue
            
        # Evaluate neighbor
        c5_neighbor = evaluate_h(h_neighbor, dx)
        
        # Check if better than best
        if c5_neighbor < best_c5:
            best_h = h_neighbor.copy()
            best_c5 = c5_neighbor
            last_improvement_iter = iter_count
            
        # Move to neighbor
        h = h_neighbor
        
        # Update tabu list
        tabu_list.append(tuple(h))
        if len(tabu_list) > tabu_tenure:
            tabu_list.pop(0)
            
        # Early termination
        if iter_count - last_improvement_iter > no_improvement_limit:
            break
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h, final_c5, len(best_h))
