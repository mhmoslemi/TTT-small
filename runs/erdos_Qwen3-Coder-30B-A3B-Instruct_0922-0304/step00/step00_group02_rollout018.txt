import numpy as np
import time
from scipy import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    dx = 2.0 / len(h)
    # Compute correlations for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # Maximum correlation represents the overlap
    max_overlap = np.max(corr)
    return max_overlap * dx

def validate_h(h):
    """Ensure h satisfies constraints."""
    if not (0 <= h).all() or not (h <= 1).all():
        return False
    if abs(np.sum(h) - len(h) / 2) > 1e-10:
        return False
    return True

def repair_h(h):
    """Repair h to meet constraints if it violates them."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = len(h) / 2
    diff = target_sum - current_sum
    
    if abs(diff) < 1e-10:
        return h
    
    # Simple repair: adjust one value
    if diff > 0:
        h[0] = min(h[0] + diff, 1)
    else:
        h[0] = max(h[0] + diff, 0)
        
    return h

def mutate_h(h, rate=0.01):
    """Apply mutation to h."""
    h = h.copy()
    n = len(h)
    n_mutate = int(n * rate)
    indices = np.random.choice(n, n_mutate, replace=False)
    for i in indices:
        # Add small random perturbation
        h[i] += np.random.normal(0, 0.05)
    return repair_h(h)

def crossover_h(h1, h2):
    """Perform crossover between two step functions."""
    n = len(h1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([h1[:point], h2[point:]])
    child2 = np.concatenate([h2[:point], h1[point:]])
    return repair_h(child1), repair_h(child2)
