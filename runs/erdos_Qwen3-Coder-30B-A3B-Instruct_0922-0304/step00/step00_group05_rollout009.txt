import numpy as np
from scipy import signal
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points // 2) * 0.5
        h = np.pad(h, (0, n_points - len(h)), 'constant', constant_values=0.0)
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum != target_sum:
        h = h * target_sum / h_sum
    
    # Make sure bounds respected
    h = np.clip(h, 0, 1)
    
    def evaluate(h_vec):
        # Correlation with 1-h
        corr = signal.correlate(h_vec, 1 - h_vec, mode="full")
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    # Optimization parameters
    T_max = 1.0
    T_min = 1e-4
    max_iter = 10000
    alpha = np.exp(np.log(T_min / T_max) / max_iter)
    T = T_max
    delta = 0.1
    accept_count = 0
    total_count = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h.copy()
        
        # Randomly choose a perturbation type
        choice = random.random()
        
        if choice < 0.5:
            # Swap two random indices
            idx1, idx2 = random.sample(range(n_points), 2)
            val1, val2 = h_new[idx1], h_new[idx2]
            # Ensure new values remain in [0,1] and respect sum
            diff = val1 - val2
            if diff > 0:
                change = min(delta, diff, 1 - val2)
                h_new[idx1] -= change
                h_new[idx2] += change
            else:
                change = min(delta, -diff, val1)
                h_new[idx1] += change
                h_new[idx2] -= change
        else:
            # Shift mass between adjacent points
            idx = random.randint(0, n_points - 2)
            if random.random() < 0.5:
                # Move from idx to idx+1
                change = min(delta, h_new[idx], 1 - h_new[idx+1])
                if change > 0:
                    h_new[idx] -= change
                    h_new[idx+1] += change
            else:
                # Move from idx+1 to idx
                change = min(delta, h_new[idx+1], 1 - h_new[idx])
                if change > 0:
                    h_new[idx] += change
                    h_new[idx+1] -= change
        
        # Ensure bound constraints
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-8:
            scale_factor = target_sum / current_sum
            h_new *= scale_factor
        
        # Evaluate new solution
        new_c5 = evaluate(h_new)
        old_c5 = evaluate(h)
        
        # Accept or reject
        if new_c5 <= old_c5 or random.random() < np.exp(-(new_c5 - old_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Update temperature and delta
        T *= alpha
        total_count += 1
        if total_count > 0 and total_count % 50 == 0:
            accept_count = 0  # Reset for adaptive delta
        if total_count % 100 == 0:
            # Adapt delta based on recent acceptance rate
            if accept_count / 100 < 0.2:
                delta *= 0.9
            elif accept_count / 100 > 0.4:
                delta *= 1.1
            accept_count = 0
    
    return (best_h.tolist(), best_c5, n_points)
