import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def evaluate_erdos_solution(h, n_points):
    # Computes the C5 bound for given h
    dx = 2.0 / n_points
    # Full correlation
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the relevant part (shifts from 0 to 2n-2)
    overlaps = corr[len(h)-1:]  # Only the overlapping parts
    c5 = np.max(overlaps) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 47
    max_iter = 1000
    refinement_freq = 50
    lr = 0.01
    backtracking_rho = 0.5
    max_backtracking = 30
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
    else:
        # Simple initialization
        h_base = np.ones(n_points) * 0.5
    
    # Ensure sum constraint: ∫h = 1, i.e., sum(h)*dx = 1 => sum(h) = n/2
    h_base = h_base * (n_points / 2) / np.sum(h_base)
    
    best_c5 = float('inf')
    best_h = h_base.copy()
    
    h = h_base.copy()
    h = np.clip(h, 0, 1)
    
    # Multi-start approach
    best_overall = float('inf')
    best_h_overall = h_base.copy()
    
    # Perturb initial solution
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        h_perturbed = h_base + np.random.normal(0, 0.005, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        h_current = h_perturbed.copy()
        current_c5 = evaluate_erdos_solution(h_current, n_points)
        if current_c5 < best_overall:
            best_overall = current_c5
            best_h_overall = h_current.copy()
            
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Evaluate overlaps
            dx = 2.0 / n_points
            corr = np.correlate(h_current, 1 - h_current, mode='full')
            overlaps = corr[len(h_current)-1:]
            k_max = np.argmax(overlaps)
            c5_current = overlaps[k_max] * dx
            
            if c5_current < best_overall:
                best_overall = c5_current
                best_h_overall = h_current.copy()
                
            # Compute gradient w.r.t. the worst shift
            shifted_h = np.roll(1 - h_current, k_max)
            gradient = shifted_h - h_current * shifted_h
            
            # Normalize gradient
            grad_norm = np.linalg.norm(gradient)
            if grad_norm < 1e-10:
                break
                
            gradient = gradient / grad_norm
            
            # Line search
            alpha = lr
            for _ in range(max_backtracking):
                h_new = h_current - alpha * gradient
                h_new = np.clip(h_new, 0, 1)
                # Scale to satisfy sum constraint
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                c5_new = evaluate_erdos_solution(h_new, n_points)
                if c5_new < c5_current:
                    h_current = h_new
                    break
                alpha *= backtracking_rho
                
            # Convex refinement every refinement_freq iterations
            if i % refinement_freq == 0 and i > 0:
                # Solve convex relaxation for worst shift
                k_max = np.argmax(overlaps)
                # Fix k_max, minimize the corresponding overlap
                # Create convex problem
                h_vars = cp.Variable(n_points)
                objective = cp.Minimize(cp.sum(cp.multiply(h_vars, 1 - cp.roll(h_vars, k_max))))
                constraints = [
                    h_vars >= 0,
                    h_vars <= 1,
                    cp.sum(h_vars) == n_points / 2
                ]
                prob = cp.Problem(objective, constraints)
                
                try:
                    prob.solve(solver=cp.GLPK)
                    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                        h_new = h_vars.value
                        if h_new is not None:
                            h_current = np.clip(h_new, 0, 1)
                except:
                    pass
        if best_overall < best_c5:
            best_c5 = best_overall
            best_h = best_h_overall.copy()
            
    return (best_h, best_c5, n_points)
