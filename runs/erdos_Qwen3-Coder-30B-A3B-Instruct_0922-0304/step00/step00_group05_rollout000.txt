    np.random.seed(seed)

        mid = n_points // 2
        for i in range(n_points):
            if i < mid:
                h[i] = 0.2 + 0.3 * (i / mid)
            else:
                h[i] = 1.0 - 0.3 * ((i - mid) / (n_points - mid))
        # Normalize
        h *= n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.01
    patience = 100

    best_h = h.copy()

        current_c5 = evaluate_erdos_solution(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute all correlations
        corr = signal.correlate(h, 1 - h, mode="full")
        # Extract the valid range for shifts from -n_points+1 to n_points-1
        start_idx = n_points - 1
        end_idx = 2 * n_points - 2
        c_vals = corr[start_idx:end_idx+1] * dx
        
        # Find maximum C_m and its indices
        max_c = np.max(c_vals)
        max_indices = np.where(np.abs(c_vals - max_c) < 1e-8)[0]
        
        # Compute gradient w.r.t. h
        grad = np.zeros(n_points)
        for idx in max_indices:
            shift = idx - (n_points - 1)
            # Compute derivative at each position
            for i in range(n_points):
                j = i + shift
                if 0 <= j < n_points:
                    grad[i] += (1 - h[j])
        
        # Update h
        h -= learning_rate * grad
        
        # Project h to [0, 1] and normalize
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        
        # Reduce learning rate if no improvement for a while
        patience_counter += 1
        if patience_counter > patience:
            patience_counter = 0
            learning_rate *= 0.9
        
    return (best_h, best_c5, n_points)
