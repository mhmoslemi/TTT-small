```python
import numpy as np
from scipy import optimize
import time
from cvxpy import *
import math

def evaluate_erdos_solution(h):
    """Computes C5 from the step function h"""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, else create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        else:
            h = np.ones(n_points) * 0.5
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure the sum constraint and bounds
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    elapsed = time.time() - start_time
    
    # ADMM parameters
    rho = 1e-3
    alpha = 1e-2
    beta = 1e-4
    max_iter = 10000
    
    # Initialize dual variables
    lambda_k = np.zeros(2 * n_points - 1)
    mu = 0.0
    
    # Iteration counter
    iter_count = 0
    no_improvement_count = 0
    last_C5 = best_C5 + 1
    
    while elapsed < budget_s and iter_count < max_iter:
        iter_count += 1
        prev_C5 = best_C5
        
        # Update h using convex optimization
        h_var = Variable(n_points)
        h_new = np.array(h)
        
        # Set up and solve the convex optimization problem for h
        obj = Minimize(
            sum( [lambda_k[i] * h_var[i] for i in range(len(lambda_k))] ) 
            + mu * (sum(h_var) - n_points/2)
            + 0.5 * rho * sum((h_var[i] - h[i])**2 for i in range(n_points))
        )
        constraints = [h_var >= 0, h_var <= 1]
        prob = Problem(obj, constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_new = np.array(h_var.value).flatten()
            else:
                # fallback to previous h if solve fails
                pass
        except:
            # fallback - keep previous h
            pass
            
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlaps
        overlaps = np.correlate(h_new, 1 - h_new, mode="full")
        C5_new = np.max(overlaps) * dx
        
        # Update dual variables
        for k in range(len(lambda_k)):
            current_overlap = overlaps[k]
            lambda_k[k] += alpha * max(0, C5_new + 1e-8 - current_overlap)
            
        mu += alpha * (np.sum(h_new) - n_points/2)
        
        # Entropy regularization on lambda (not needed for convergence, but helps stability)
        # No actual entropy term, since the variable is already lagrangian, already penalizes deviance
        
        # Update h and best solution
        h = h_new
        
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Decay beta
        if iter_count % 100 == 0:
            beta *= 0.9
        
        # If we're below target or have too many iterations with no improvement, stop
        if best_C5 < 0.3807:
            break
        if no_improvement_count > 500:
            break
        
        elapsed = time.time() - start_time
    
    return (best_h.tolist(), best_C5, n_points)
```
