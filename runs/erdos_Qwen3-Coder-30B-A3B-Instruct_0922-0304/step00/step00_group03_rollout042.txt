import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 70
    dx = 2.0 / n_points
    target_bound = 0.3807
    max_iter = 5000
    lr = 0.01
    decay_every = 100
    momentum = 0.9
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h = np.full(n_points, 0.5)
    
    # Ensure constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # For gradient descent with momentum
    velocity = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all cross-correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        # corr length = 2*n_points - 1
        # We need to extract valid correlations
        max_overlap = 0
        max_k = 0
        
        # Iterate over all valid shifts
        for shift in range(-n_points + 1, n_points):
            # The shift k corresponds to correlation index
            # In correlate with mode='full', the shift k = -n_points+1 ... n_points-1
            # We compute correlation for this shift
            k_idx = shift + n_points - 1
            val = corr[k_idx] * dx
            if val > max_overlap:
                max_overlap = val
                max_k = shift
        
        # Early stopping if already good enough
        if max_overlap < target_bound:
            break

        # Compute gradient
        grad = np.zeros_like(h)
        # For shift k, cross-correlation involves terms h[i]*(1-h[i+k])
        # The gradient of the overlap integral w.r.t. h[i] includes terms
        # where (1 - h[i+k]) contributes for each shift k that includes i
        # So we compute for i, sum over k where i+k is valid and add (1-h[i+k])
        # But reversed, since corr[i] for shift k=-i corresponds to h[i]*(1-h[0])
        # We focus on the maximum overlap index and compute its gradient
        # Using dot product with 1-h shifted by max_k
        # Only for indices that are "active" in this overlap term
        k_start = max_k
        k_end = max_k + n_points
        
        for i in range(n_points):
            if i + k_start >= 0 and i + k_start < n_points:
                # If forward and backward
                temp_index = i + k_start
                if 0 <= temp_index < n_points:
                    grad[i] += 1 - h[temp_index]
        
        # Gradient descent with momentum
        velocity = momentum * velocity - lr * grad
        h = h + velocity
        
        # Project back to feasible set
        # Clip to [0, 1] for each h[i]
        h = np.clip(h, 0, 1)
        
        # Rescale so that sum(h)*dx = 1
        h_sum = np.sum(h) * dx
        h = h * (1 / h_sum)
        
        # Store best solution so far
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
