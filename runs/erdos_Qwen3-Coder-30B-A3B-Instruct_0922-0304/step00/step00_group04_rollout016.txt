import numpy as np
from scipy import fft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize to correct sum
    h *= target_sum / np.sum(h)
    
    # Precompute some structs
    fft_size = 2 * n_points - 1
    half_size = n_points
    
    # Optimization loop
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Run the main optimization
    iter = 0
    while time_left() > 1.0:
        iter += 1
        # FFT-based optimization
        H = fft.fft(h, fft_size)
        S = np.abs(H)**2
        
        # Compute optimal update to flatten spectrum while preserving DC
        # Use an iterative approach to reduce maximum value in spectrum
        scale_factor = np.max(S)
        if scale_factor > 0:
            w = 1.0 / (1e-12 + S)
            w = w / np.sum(w) * len(w)
            new_H = H * np.sqrt(w)
            new_H[0] = H[0]  # keep DC component
            new_h = np.real(fft.ifft(new_H, fft_size))
            # Take relevant part
            new_h = new_h[:n_points]
            
            # Clip and project back to feasible region
            new_h = np.clip(new_h, 0, 1)
            scale = target_sum / np.sum(new_h)
            new_h *= scale
            
            # Test the new candidate
            new_C5 = evaluate_erdos_solution(new_h, dx)
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = new_h.copy()
                
            h = new_h
        else:
            break
            
        # Occasionally add a random perturbation
        if iter % 10 == 0 and time_left() > 1.0:
            # Small random perturbation to avoid local minima
            pert = np.random.normal(0, 0.01, n_points)
            pert = np.clip(pert, -0.1, 0.1)
            h = np.clip(h + pert, 0, 1)
            h *= target_sum / np.sum(h)
            
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h_vals, dx):
    """Evaluate C5 using numpy's correlate."""
    h = np.array(h_vals)
    # Cross correlation with 1 - h
    corr = np.correlate(h, 1 - h, mode="full")
    # The result is indexed from -n+1 to n-1
    # Maximum value is in the middle, but we check all values
    max_val = np.max(corr)
    return max_val * dx
