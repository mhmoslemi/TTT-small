import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided initial construction if available
    try:
        h = initial_h_values.copy()
    except NameError:
        # Fallback: create a simple initial function
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    start_time = time.time()

    # Parameters
    delta = 0.1
    decay_factor = 0.9
    iter_count = 0

    while time.time() - start_time < budget_s:
        # Perturbation
        i = np.random.randint(0, n_points)
        j = np.random.randint(0, n_points)
        
        # Ensure we perturb different indices
        if i == j:
            j = (j + 1) % n_points
        
        new_h = h.copy()
        delta_change = delta * np.random.choice([-1, 1])
        
        # Adjust values
        new_h[i] += delta_change
        new_h[j] -= delta_change
        
        # Make sure new_h stays within bounds
        # Clip to ensure feasibility
        new_h = np.clip(new_h, 0, 1)
        
        # Force sum constraint
        current_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / current_sum
        
        # Evaluate new solution
        new_c5 = evaluate_c5(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            h = new_h
            best_c5 = new_c5
            best_h = h.copy()
        
        # Decay delta
        if iter_count % 100 == 0:
            delta *= decay_factor
            
        iter_count += 1
    
    return (best_h, best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute overlap integral
    # Using mode="full" to get all cross-correlations
    correlations = correlate(h, 1 - h, mode="full")
    max_corr = np.max(correlations)
    return max_corr * dx
