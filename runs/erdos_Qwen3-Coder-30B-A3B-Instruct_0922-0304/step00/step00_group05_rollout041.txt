import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    swarm_size = 100
    max_iter = 6000  # estimated based on budget
    
    # Initialize particles
    particles = np.zeros((swarm_size, n_points))
    velocities = np.zeros((swarm_size, n_points))
    
    # Set first particle to initial solution if available
    if 'initial_h_values' in globals():
        particles[0] = initial_h_values
    else:
        particles[0] = np.random.rand(n_points)
    
    # Initialize other particles randomly with correct sum
    for i in range(1, swarm_size):
        particles[i] = np.random.rand(n_points)
    
    # Normalize all particles to have correct integral
    for i in range(swarm_size):
        particles[i] *= (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.5, 0.5, (swarm_size, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Initialize best positions and values
    personal_best_positions = particles.copy()
    personal_best_values = np.full(swarm_size, np.inf)
    global_best_position = None
    global_best_value = np.inf
    
    def evaluate(h):
        # Compute overlap integral
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_flipped = np.flip(1 - h)
        correlation = signal.convolve(h_padded, h_flipped, mode='valid')
        c5_val = np.max(correlation[:n_points]) * dx
        return c5_val
    
    # Main PSO loop
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i in range(swarm_size):
            h = particles[i]
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h *= (n_points / 2) / np.sum(h)
            
            # Update particle
            particles[i] = h
            
            # Evaluate
            c5 = evaluate(h)
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_best_positions[i] = h.copy()
            
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best_position = h.copy()
        
        # Update velocities and positions
        for i in range(swarm_size):
            if global_best_position is not None:
                r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
                velocities[i] = (w * velocities[i] + 
                                c1 * r1 * (personal_best_positions[i] - particles[i]) + 
                                c2 * r2 * (global_best_position - particles[i]))
                
                particles[i] += velocities[i]
                
                # Project to feasibility
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] *= (n_points / 2) / np.sum(particles[i])
        
        iteration += 1
    
    # Ensure final solution is valid
    final_h = global_best_position.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h *= (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate(final_h)
    
    return (final_h, final_c5, n_points)
