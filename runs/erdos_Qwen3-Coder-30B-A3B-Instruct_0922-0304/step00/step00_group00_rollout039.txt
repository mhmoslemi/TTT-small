import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except:
        # Default initialization: step function
        n_points = 45
        h0 = np.ones(n_points) * 0.5
        # Normalize to sum to n_points/2
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Ensure h0 satisfies constraints
    h0 = np.clip(h0, 0, 1)
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    h_best = h0.copy()
    c5_best = evaluate_erdos_solution(h_best)
    
    # Track recent max shifts
    recent_shifts = []
    max_recent = 50
    
    # Iteration counter
    iteration = 0
    no_improve_count = 0
    
    while time() - start_time < budget_s * 0.95:
        iteration += 1
        if iteration > 1000:
            break

        # Compute overlaps for all shifts
        h = h_best.copy()
        overlaps = np.correlate(h, 1 - h, mode="full")
        k_max_idx = np.argmax(overlaps)
        k_max = k_max_idx - (n_points - 1)
        
        recent_shifts.append(k_max)
        if len(recent_shifts) > max_recent:
            recent_shifts = recent_shifts[-max_recent:]
        
        # Get the current C5 bound value
        current_c5 = overlaps[k_max_idx]
        
        # If we've improved, update the best
        if current_c5 < c5_best:
            c5_best = current_c5
            h_best = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        # Early stopping
        if c5_best < 0.3807:
            break
        if no_improve_count > 50:
            break
            
        # Solve optimization:
        # min C5
        # subject to: sum(h) = n_points/2
        #             0 <= h[i] <= 1
        #             sum h[i] * (1 - h[(i+k) mod n]) <= C5  for all k
        
        # Try to solve convex relaxation for most violated constraint
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Add constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            cp.sum(cp.multiply(h_var, 1 - cp.roll(h_var, k_max))) <= C5_var
        ]
        
        # Objective
        objective = cp.Minimize(C5_var)
        
        # Create and solve problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h_best = np.array(h_new)
                    c5_best = prob.value
        except:
            pass

    # Final check of the solution
    h_final = h_best.copy()
    c5_final = evaluate_erdos_solution(h_final)
    
    return (list(h_final), c5_final, n_points)

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap
