import numpy as np
import cvxpy as cp
from time import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx):
    """Evaluate the C5 constant for given h"""
    # Compute all correlations
    corr = correlate(h, 1-h, mode="full")
    # Take the maximum correlation value and scale by dx
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Fallback: create a uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h = h * (n_points / 2) / np.sum(h)  # normalize sum
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time()
    
    # Main optimization loop
    max_iter = 2000
    for iter_num in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        # Formulate the convex relaxation problem
        # We aim to minimize C5 = max_k sum h[i] * (1 - h[i+k]) dx
        # This is equivalent to minimizing an upper bound on the sum
        
        # Create variables
        h_var = cp.Variable(n_points)
        
        # Relax h[i] * (1 - h[i+k]) terms
        # For convex relaxation, we use the fact that:
        # h[i] * (1 - h[i+k]) <= max(h[i], 1 - h[i+k]) if we're maximizing
        # but to minimize, we need different bounds.
        # We actually want to bound above by a convex upper bound:
        # h[i] * (1 - h[i+k]) <= (h[i] + (1-h[i+k])) / 2 + (h[i] - (1-h[i+k]))^2 / 4
        # Actually, lets use simpler bound: 
        # h[i] * (1 - h[i+k]) <= 0.25 + (h[i] - 0.5) * (1 - h[i+k] - 0.5)
        # No, simpler and safe: we know h[i] <= 1, so h[i] * (1 - h[i+k]) <= 1 * (1 - h[i+k]) = 1 - h[i+k]
        # But this isn't good as it doesn't capture cross-terms. 
        # Try a more standard convex relaxation:
        # Define new variables for products, but here we consider an alternative.
        
        # Conservative linearization approach
        # Let's use a second order cone relaxation or linear one.
        # And instead, reduce this to solving a simpler subproblem.
        
        # For a bilinear term h[i] * (1 - h[i+k]), we add constraint  
        # if we have a variable z[i,k] approximating the product.
        # But simpler approach: use softer bounds via gaussian blur of product
        
        # Let's compute correlations as a small LP:
        # For each k, we compute sum_i h[i]*(1 - h[i+k]) // This is not convex in h in general
        
        # Because of the intrinsic non-convexity, let's try gradient descent on a proxy.
        
        # The main approach changes to this:
        # Use projected gradient descent on a relaxed model.
        # We now adopt a conservative approach:
        # Find the value of max_k sum_i h[i] * (1 - h[i+k]) dx
        # Assume we can upper bound the correlation via simple bounds.
        # Offer a convex surrogate: we state that we construct a new variable t = max_k ...  
        # But that is not directly setup with CVXPY, so resort to SGD on the bridge.
        
        # A more robust approach is to sample many iteration seeds.
        # But with timing constraints: apply a simple penalty approach.
        
        # For large enough n_points, that overlaps create lots.
        # Idea: Build constraint where h has sum = n_points / 2
        # Express the overlaps via correlations, projecting on feasible points,
        # improving in direction of steepest decsent on the trade off proxy.
        
        # Use projected gradient descent with updated objective surrogate.
        # Since CVXPY is too slow here, we relax and do stochastic projective descent approach.
        # But stick to direct approach using only cvxpy on a kiss approximation.
        
        # Instead, compute empirical c5, but direct minimization using cvxpy is now simpler:
        
        # Propose a convex relaxation:
        # Introduce variables s for overlap at k
        # s_k = sum h[i] * (1 - h[i+k]) * dx  
        # Constraint: s_k <= C5 (for all k)

        # Use standard gradient method, with toy relaxation approach

            problem_instance.solve(solver=cp.ECOS, verbose=False)

                h_proj = h_proj * (n_points / 2) / np.sum(h_proj)

                    best_c5 = c5_new

        for k in range(-n_points+1, n_points):
            if k >= 0:
                temp = 1 - h[k:]
                temp_pad = np.concatenate([temp, np.zeros(n_points - len(temp))])
            else:
                temp_pad = np.concatenate([np.zeros(-k), h[:n_points+k]])
                temp = 1 - temp_pad
            gradient += temp
        
        h = h - 0.001 * gradient  # step size 0.001
        h = h.clip(0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        c5_new = evaluate_erdos_solution(h, dx)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    # Return best solution found
    return (best_h.tolist(), best_c5, n_points)
