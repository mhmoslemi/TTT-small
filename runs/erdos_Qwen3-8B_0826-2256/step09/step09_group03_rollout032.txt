        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Strategy: Use a combination of genetic algorithm and SLSQP for refinement
    # First, run differential evolution with a good initial population
    def objective_de(h_values):
        return objective(h_values)

    # Generate a diverse initial population
    initial_population = np.zeros((30, n_points))
    for i in range(30):
        # Add some variation by shifting and scaling
        shift = np.random.uniform(-0.1, 0.1, n_points)

        h_variation = h.copy() + shift * scale
        h_variation = np.clip(h_variation, 0, 1)

        h_variation = h_variation / (np.sum(h_variation) * dx) if np.sum(h_variation) != 0 else h_variation

    result_slsqp = minimize(

        best_h,
        bounds=bounds,

        options={"ftol": 1e-9, "maxiter": 150}
    )
