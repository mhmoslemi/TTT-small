import numpy as np
import scipy.optimize as optimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points
    n_tries = 150  # Number of trials for diversification

    # Use the initial_h_values if provided
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initialize with a reasonable structure
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    for _ in range(n_tries):
        if time.time() - start_time > budget_s - 1:
            break

        # Random perturbation to explore new regions
        perturbation = np.random.uniform(-0.3, 0.3, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        x0 = h

        # Use SLSQP for local refinement with tighter tolerances
        result_slsqp = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )

        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x

        # Use COBYLA for robustness
        result_cobyla = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="COBYLA",
            options={"maxiter": 100}
        )

        if result_cobyla.success:
            current_c5 = result_cobyla.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x

    return best_h, best_c5, n_points
