import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import minres

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better resolution while keeping computation within budget
    dx = 2.0 / n_points
    max_overlap = 0.38092  # Current best known upper bound

    # Initialize with the provided initial solution or a balanced one
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.5
        h[n_points // 2:] = 0.5
        # Normalize to ensure the integral is exactly 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Define the overlap integral function
    def compute_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx

    # Constraint function to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Bounds for the optimization
    bounds = Bounds(0, 1)

    # Start with the initial guess
    best_h = h
    best_c5 = compute_overlap(h)
    start_time = time.time()

    # Use a global optimization approach for exploration
    def objective(h_values):
        return compute_overlap(h_values)

    # Use a combination of global search and local refinement
    for _ in range(20):  # Limit the number of global search rounds
        if time.time() - start_time > budget_s - 1:
            break
        # Use differential evolution for global exploration
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-4,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False,
            constraints={"type": "eq", "fun": constraint}
        )
        if result.success:
            h_candidate = result.x
            # Refine local minimum with SLSQP
            result_slsqp = minimize(
                objective,
                h_candidate,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

    # Final refinement with COBYLA
    result_cobyla = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="COBYLA",
        options={"maxiter": 150}
    )
    if result_cobyla.success:
        best_c5 = result_cobyla.fun
        best_h = result_cobyla.x

    return best_h, best_c5, n_points
