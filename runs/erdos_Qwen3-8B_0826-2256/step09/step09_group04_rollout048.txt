    def objective(h_values):

        perturbation = np.random.uniform(-0.3 * (1 - trial / trials), 0.3 * (1 - trial / trials), size=n_points)

        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

        diversity = np.abs(x0 - best_h) * diversity_penalty
