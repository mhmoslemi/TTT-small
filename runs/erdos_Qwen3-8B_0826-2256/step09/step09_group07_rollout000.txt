    h = np.clip(h, 0, 1)

    def objective(h_values):

        c5 = np.max(overlap) * dx
        return c5

    def constraints(h_values):

    best_h = h

    perturbation_range = 0.1  # Initial perturbation range

    # Perform multiple optimization trials with different initializations
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
