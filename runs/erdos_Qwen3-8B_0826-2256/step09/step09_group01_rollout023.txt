import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balanced number of points for speed and accuracy
    
    # Initialize from the provided initial_h_values
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of trials for exploration
    
    # Use a combination of global and local optimization with perturbations
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.05 * (1 - trial / trials), 0.05 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
    
        # Local refinement with SLSQP
        result_slsqp = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
        # Global exploration with differential evolution
        result_de = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"maxiter": 100, "ftol": 1e-9}
        )
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
    
    return best_h, best_c5, n_points
