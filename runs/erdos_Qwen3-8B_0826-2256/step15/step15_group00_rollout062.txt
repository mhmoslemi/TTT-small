import scipy.optimize as opt

from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import DifferentialEvolution

def run(seed=42, budget_s=100, **kwargs):

    else:

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 100  # Number of trials for exploration
    population_size = 15  # Population size for differential evolution
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random perturbation to avoid local minima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Define the optimization problem
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        
        # Use differential evolution for global exploration
        result_de = minimize(
            objective,
            h,
            bounds=bounds,
            constraints=constraint,
            method=DifferentialEvolution,
            options={"tol": 1e-6, "mutation": 0.5, "recombination": 0.7, "popsize": population_size, "maxiter": 100}
        )
        
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x

    return best_h, best_c5, n_points
