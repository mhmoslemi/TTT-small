import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Efficient number of points for speed and quality
    
    # Use initial_h_values if available for better starting point

        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Parameters for optimization
    trials = 100  # Number of trials
    perturbation_scale = 0.3  # Initial perturbation scale
    learning_rate = 0.01  # Learning rate for gradient-based methods
    maxiter = 200  # Maximum iterations for local optimization
    
    # Main optimization loop
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Local refinement with SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": maxiter}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Global exploration with differential evolution
        try:
            result_de = differential_evolution(
                objective,
                bounds=[(0, 1) for _ in range(n_points)],
                constraints=[integral_constraint],
                strategy='best1bin',
                popsize=15,
                maxiter=50,
                tol=1e-5,
                seed=trial
            )
            if result_de.success:
                current_c5 = result_de.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_de.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
