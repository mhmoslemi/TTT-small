import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of points for the budget
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 200  # Increase the number of trials for better exploration
    perturbation_scale = 0.15  # Initial perturbation scale
    learning_rate = 0.01  # Learning rate for gradient-based methods
    exploration_rate = 0.3  # Probability of exploring new regions

    # Add differential evolution for global exploration
    def differential_evolution_objective(h_values):
        return objective(h_values)

    # Differential evolution parameters
    de_popsize = 15
    de_maxiter = 50
    de_strategy = "best1bin"
    de_mutate = 0.5
    de_crossover = 0.9

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Randomly choose between perturbation and new random initialization
        if np.random.rand() < exploration_rate:
            # Random initialization
            h = np.random.rand(n_points)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        else:
            # Perturb the current best solution with decreasing magnitude
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0, 1)
            # Ensure the integral constraint is satisfied
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        x0 = h

        # SLSQP for local refinement with higher precision
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 200}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # COBYLA for global exploration with higher precision
        try:
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 200}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass
        
        # Run differential evolution for global exploration
        try:
            result_de = opt.differential_evolution(
                differential_evolution_objective,
                bounds=np.array([[0, 1] for _ in range(n_points)]),
                strategy=de_strategy,
                mutation=(de_mutate, de_mutate, de_mutate),
                crossover=de_crossover,
                maxiter=de_maxiter,
                popsize=de_popsize,
                tol=1e-10,
                disp=False
            )
            if result_de.success:
                current_c5 = result_de.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_de.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
