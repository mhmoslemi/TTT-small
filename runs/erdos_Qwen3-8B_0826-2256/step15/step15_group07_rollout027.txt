from scipy.optimize import LinearConstraint
from scipy.optimize import DifferentialEvolution

def run(seed=42, budget_s=100, **kwargs):

    # Use initial_h_values if available

    else:
        h = np.zeros(n_points)

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of trials for exploration

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Random initialization
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Differential evolution for global exploration
        result_de = DifferentialEvolution(
            objective,
            bounds=bounds,
            constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False
        )
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x

        # Local refinement with SLSQP
        try:
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            result_slsqp = minimize(
                objective,
                x0=best_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 200}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

    return best_h, best_c5, n_points
