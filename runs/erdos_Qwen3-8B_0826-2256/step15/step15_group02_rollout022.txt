import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    
    # Initialize with the provided initial_h_values
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Default pattern: symmetric function with a peak in the middle
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.5
        h[n_points // 2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Optimize using local and global strategies
    for _ in range(50):  # 50 trials for exploration
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb and clip
        perturbation = np.random.uniform(-0.03, 0.03, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize to meet integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Local optimization with SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 100}
            )
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        except:
            pass
        
        # Global optimization with differential evolution
        try:
            result_de = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="differential_evolution",
                options={"maxiter": 20, "popsize": 15}
            )
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
        except:
            pass
    
    return best_h, best_c5, n_points
