    h = np.clip(h, 0, 1)

    def objective(h_values):

        overlap = np.correlate(h, 1 - h, mode="full")

    def differential_evolution(population, bounds, objective, n_points, maxiter=100):
        for _ in range(maxiter):
            if time.time() - start_time > budget_s - 1:
                break
            # Mutation
            a, b, c = np.random.choice(len(population), 3, replace=False)
            mutant = population[a] + np.random.uniform(-1, 1, n_points) * (population[b] - population[c])
            mutant = np.clip(mutant, bounds.lb, bounds.ub)
            # Crossover
            trial = np.copy(population[np.random.randint(len(population))])

                    trial[i] = mutant[i]

            trial_obj = objective(trial)

    best_h_de = differential_evolution(initial_population, bounds, objective, n_points)

    best_h = best_h_de

    except:
        pass
    
    return best_h, best_c5, n_points
