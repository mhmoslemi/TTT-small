import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Moderate number of trials for thorough exploration
    perturbation_scale = 0.3  # Larger initial perturbation for exploration
    learning_rate = 0.01  # Learning rate for gradient-based methods
    
    # Add a genetic algorithm for global exploration
    def genetic_algorithm():
        population_size = 20
        generations = 20
        mutation_rate = 0.1
        mutation_scale = 0.05
        
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population / (np.sum(population, axis=1, keepdims=True) * 2.0 / n_points)
        population = np.clip(population, 0, 1)
        
        for _ in range(generations):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Evaluate fitness
            fitness = np.array([objective(ind) for ind in population])
            
            # Select parents
            elite_indices = np.argsort(fitness)[:population_size//2]
            parents = population[elite_indices]
            
            # Crossover
            offspring = []
            for _ in range(population_size - len(parents)):
                parent1, parent2 = np.random.choice(parents, size=2, replace=False)
                child = np.zeros(n_points)
                for i in range(n_points):
                    if np.random.rand() < 0.5:
                        child[i] = parent1[i]
                    else:
                        child[i] = parent2[i]
                offspring.append(child)
            
            # Mutation
            for i in range(len(offspring)):
                for j in range(n_points):
                    if np.random.rand() < mutation_rate:
                        offspring[i][j] += np.random.uniform(-mutation_scale, mutation_scale)
                        offspring[i][j] = np.clip(offspring[i][j], 0, 1)
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = population / (np.sum(population, axis=1, keepdims=True) * 2.0 / n_points)
            population = np.clip(population, 0, 1)
        
        best_individual = population[np.argmin(fitness)]
        return best_individual
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use genetic algorithm for global exploration
        try:
            h = genetic_algorithm()
            current_c5 = objective(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
        except:
            pass
        
        # Use SLSQP with adaptive learning
        try:
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            result_slsqp = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 200}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
