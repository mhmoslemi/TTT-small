    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of trials for differential evolution
    n_init = 10  # Number of initial points for differential evolution

    # Use differential evolution for global exploration
    result_de = differential_evolution(
        objective,
        bounds,
        constraints=integral_constraint,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        maxiter=200,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun

    # Local refinement with SLSQP
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints=constraint,
            method="SLSQP",

        )
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x

        pass
