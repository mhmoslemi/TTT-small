import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # balance between speed and accuracy

    else:
        # Fallback to a balanced pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h * (n_points / 2.0) / np.sum(h) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h) if np.sum(h) != 0 else h
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to ensure the integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 300  # More trials for a comprehensive search

    # Use SLSQP for refinement with tight tolerances
    def refine_solution(h_initial):
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        result_slsqp = minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints=constraint,
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 200}
        )
        if result_slsqp.success:
            return result_slsqp.x
        return h_initial

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.normal(0, 0.01 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h * (n_points / 2.0) / np.sum(h) if np.sum(h) != 0 else h
        h = np.clip(h, 0, 1)
        # Refine the perturbed solution
        h = refine_solution(h)
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

    return best_h, best_c5, n_points
