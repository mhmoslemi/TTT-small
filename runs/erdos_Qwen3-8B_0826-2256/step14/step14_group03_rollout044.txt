import scipy.optimize as optimize

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Use a combination of global search and local refinement
    # Step 1: Global search with COBYLA for diversity
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        x0 = np.random.uniform(0, 1, n_points)
        x0 = x0 / (np.sum(x0) * dx) if np.sum(x0) != 0 else h
        x0 = np.clip(x0, 0, 1)
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=constraint,
            method="COBYLA",
            options={"maxiter": 100}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Step 2: Local refinement with SLSQP
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        x0 = best_h
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=constraint,
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    # Step 3: Additional exploration with differential evolution
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:

        x0 = best_h

        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints=constraint,
            method="differential_evolution",
            options={"maxiter": 30, "popsize": 20}
        )
        if result.success and result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun
    
    return best_h, best_c5, n_points
