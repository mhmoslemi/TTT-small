def run(seed=42, budget_s=100, **kwargs):

        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Evaluate current solution
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Try differential evolution for global optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result_de = differential_evolution(
        objective,
        bounds,
        constraints={"type": "eq", "fun": constraint},
        strategy='best1bin',
        maxiter=50,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        workers=-1,
        disp=False,
        vectorized=True
    )
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun
    
    # Run SLSQP with tighter constraints for refinement
    bounds_slsqp = Bounds(0, 1)
    result_slsqp = minimize(
        objective,
        best_h,
        bounds=bounds_slsqp,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 150}
    )
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    # Run Nelder-Mead for further refinement
    result_nelder = minimize(
        objective,
        best_h,
        bounds=bounds_slsqp,
        constraints={"type": "eq", "fun": constraint},
        method="Nelder-Mead",
        options={"maxiter": 150}
    )
    if result_nelder.success:
        best_c5 = result_nelder.fun
        best_h = result_nelder.x
    
    # Check if we have enough time to do more trials
    if time.time() - start_time < budget_s - 1:
        # Try a few more trials with perturbations
        for _ in range(3):
            perturbation = np.random.uniform(-0.2, 0.2, size=n_points)
            h_perturb = best_h + perturbation

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
