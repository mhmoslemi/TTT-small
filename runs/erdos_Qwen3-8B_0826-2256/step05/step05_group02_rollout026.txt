import scipy.optimize as optimize
import time

from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):

    def objective(h_values):
        h_values = np.clip(h_values, 0, 1)
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraint to ensure the total area is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    # Optimization method
    def optimize_solution(h_initial):
        result = minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result.success:
            return result.x
        return h_initial

    def differential_evolution_search():
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-5,
            args=(dx,),
            polish=True,
            disp=False
        )
        if result.success:
            return result.x
        return np.random.rand(n_points)
    
    # Main optimization loop
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break

        h_opt = optimize_solution(best_h)

        best_c5 = compute_c5(best_h, dx)

            best_h = h_opt

            best_h = h_perturbed
    
    return best_h, best_c5, n_points

def compute_c5(h_values, dx):
    overlap = np.correlate(h_values, 1 - h_values, mode="full")
    return np.max(overlap) * dx
