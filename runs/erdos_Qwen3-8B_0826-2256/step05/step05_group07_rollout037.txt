import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and sufficient resolution
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start from a better initial guess (sawtooth-like pattern)
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 if i % 3 == 0 else 0.0
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint
    h = np.clip(h, 0.0, 1.0)

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initialize best solution
    best_h = np.copy(h)
    best_c5 = objective_function(best_h)

    # Use a more sophisticated optimizer for better convergence
    def optimize_with_perturbations(h_initial, n_iterations=5, perturbation_scale=0.05):
        h = np.copy(h_initial)
        for _ in range(n_iterations):
            perturbation = np.random.normal(0, perturbation_scale, size=n_points)
            h = np.clip(h + perturbation, 0.0, 1.0)
            h = h * (total_integral / np.sum(h))
            result = opt.minimize(
                fun=objective_function,
                x0=h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        return best_h, best_c5

    # First round of optimization
    best_h, best_c5 = optimize_with_perturbations(h, n_iterations=10, perturbation_scale=0.05)

    # Second round with smaller perturbations
    best_h, best_c5 = optimize_with_perturbations(best_h, n_iterations=10, perturbation_scale=0.01)

    # Final refinement
    for _ in range(5):
        perturbation = np.random.normal(0, 0.005, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result.fun < best_c5:
            best_h = result.x
            best_c5 = result.fun

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
