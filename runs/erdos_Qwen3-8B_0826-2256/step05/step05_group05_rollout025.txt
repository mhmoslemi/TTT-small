import numpy as np
import scipy.optimize as opt
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_n_points = len(initial_h_values)
    h_values = initial_h_values.copy()

    # Function to evaluate C5 for a given h
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)

    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2

    # Optimization configuration
    options = {'maxiter': 200, 'disp': False}

    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = opt.minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)

    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points
    best_time = time.time()

    # Particle swarm optimization for better exploration
    def pso_optimize(n_points, initial_h):
        n_particles = 30
        inertia = 0.8
        cognitive = 1.2
        social = 1.2
        particles = np.random.rand(n_particles, n_points) * 2 - 1
        particles = np.clip(particles, 0, 1)
        particles = particles * (n_points / 2) / np.sum(particles, axis=1, keepdims=True)
        velocities = np.random.rand(n_particles, n_points) * 0.1
        pbest = particles.copy()
        pbest_error = np.array([evaluate_c5(p, n_points) for p in particles])
        gbest = particles[np.argmin(pbest_error)]
        gbest_error = np.min(pbest_error)
        for _ in range(100):
            for i in range(n_particles):
                r1, r2 = np.random.rand(2, n_points)
                velocities[i] = inertia * velocities[i] + cognitive * r1 * (pbest[i] - particles[i]) + social * r2 * (gbest - particles[i])
                particles[i] += velocities[i]
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                error = evaluate_c5(particles[i], n_points)
                if error < pbest_error[i]:
                    pbest[i] = particles[i].copy()
                    pbest_error[i] = error
            gbest = pbest[np.argmin(pbest_error)]
            gbest_error = np.min(pbest_error)
        return gbest, gbest_error

    # Main optimization loop
    start_time = time.time()
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points

    while time.time() - start_time < budget_s - 5:
        # Run PSO on current solution for better exploration
        pso_h, pso_c5 = pso_optimize(best_n_points, best_h)
        if pso_c5 < best_c5:
            best_h, best_c5 = pso_h, pso_c5
            best_time = time.time()
            print(f"PSO Improved C5: {best_c5}")
        else:
            # Consider increasing n_points
            new_n_points = best_n_points + 10
            if new_n_points > 1000:
                break

            # Initialize with previous best values and fill the rest with random values
            new_h = np.zeros(new_n_points)
            new_h[:best_n_points] = best_h
            new_h[best_n_points:] = np.random.rand(new_n_points - best_n_points)
            new_h = new_h * (new_n_points / 2) / np.sum(new_h)

            # Run PSO on new_n_points
            new_pso_h, new_pso_c5 = pso_optimize(new_n_points, new_h)
            if new_pso_c5 < best_c5:
                best_h, best_c5 = new_pso_h, new_pso_c5
                best_n_points = new_n_points
                best_time = time.time()
                print(f"PSO Increased C5: {best_c5}")
            else:
                # Check if improvement is too small to continue
                if time.time() - best_time > 5:
                    break

    return best_h, best_c5, best_n_points
