import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
from scipy.stats import truncnorm

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)
    
    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Optimization configuration
    options = {'maxiter': 100, 'disp': False}
    
    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h):
        bounds = Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points
    best_time = time.time()
    best_c5_history = [best_c5]
    
    # Systematic parameter sweep with early stopping
    start_time = time.time()
    search_space = np.linspace(100, 500, 5, dtype=int)
    
    for n_points in search_space:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate random initial guess
        low = 0.0
        high = 1.0
        mean = 0.5
        std = 0.25
        size = n_points
        random_h = truncnorm.rvs((low - mean) / std, (high - mean) / std, loc=mean, scale=std, size=size)
        random_h = random_h * (n_points / 2) / np.sum(random_h)
        
        # Optimize
        current_h, current_c5 = optimize_h(n_points, random_h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h, best_c5 = current_h, current_c5
            best_n_points = n_points
            best_time = time.time()
            best_c5_history.append(best_c5)
            print(f"Improved C5: {best_c5}")
        else:
            # Check for improvement
            if len(best_c5_history) > 1 and current_c5 >= best_c5_history[-1]:
                break
    
    return best_h, best_c5, best_n_points
