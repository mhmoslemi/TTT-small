import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points
    
    # Initialize with provided initial_h_values if available, otherwise use a structured initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        # Initialize with a periodic pattern for better distribution
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Normalize to ensure integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to maintain integral requirement
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Key parameter: number of trials
    trials = 100  # Good balance between exploration and time
    # Use a more efficient method for local optimization
    method = "L-BFGS-B"  # Efficient for constrained local optimization
    maxiter = 100  # Sufficient for local refinement
    
    # Perform multiple trials with different initializations and perturbations
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Small perturbation to ensure diversity in search
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize to maintain integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Perform local optimization
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method=method,
                options={"ftol": 1e-8, "maxiter": maxiter}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass  # Gracefully handle exceptions
    
    return best_h, best_c5, n_points
