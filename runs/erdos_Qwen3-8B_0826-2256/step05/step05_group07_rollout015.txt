import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start from a better initial guess based on known optimal patterns (e.g., periodic or piecewise constant)
    h = np.zeros(n_points)
    # A more optimal initial guess: a 5-level piecewise constant function on [0, 2]
    # This pattern is inspired by known constructions and can be refined
    h[0:20] = 1.0
    h[20:40] = 0.5
    h[40:60] = 0.0
    h[60:80] = 0.5
    h[80:100] = 1.0
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint
    h = np.clip(h, 0.0, 1.0)

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Start with a more efficient optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Perform multiple rounds of perturbation with gradually decreasing step sizes
    for step_size in [0.05, 0.02, 0.01, 0.005, 0.002]:
        for _ in range(5):
            perturbation = np.random.normal(0, step_size, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if result_pert.fun < best_c5:
                best_h = result_pert.x
                best_c5 = result_pert.fun

    # Final evaluation and normalization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
