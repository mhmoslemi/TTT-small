import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (using provided or a better structured initial guess)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # # Function to perform optimization with perturbations
    def optimize_with_perturbations(h_start, max_iter=300, perturbation_factor=0.05):
        best_h = np.copy(h_start)
        best_c5 = objective_function(best_h)
        for _ in range(max_iter):
            perturbation = np.random.normal(0, perturbation_factor, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result.success and result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        return best_h, best_c5

    # # Function to perform gradient-based optimization with adaptive learning rates
    def gradient_based_optimization(h_start, max_iter=200, learning_rate=0.01):
        h = np.copy(h_start)
        best_h = np.copy(h)
        best_c5 = objective_function(h)
        for _ in range(max_iter):
            # Compute gradient numerically
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = np.copy(h)
                h_plus[i] += 1e-6
                h_plus = np.clip(h_plus, 0.0, 1.0)
                h_plus = h_plus * (total_integral / np.sum(h_plus))
                f_plus = objective_function(h_plus)
                
                h_minus = np.copy(h)
                h_minus[i] -= 1e-6
                h_minus = np.clip(h_minus, 0.0, 1.0)
                h_minus = h_minus * (total_integral / np.sum(h_minus))
                f_minus = objective_function(h_minus)
                
                grad[i] = (f_plus - f_minus) / (2 * 1e-6)
            
            # Update step
            h -= learning_rate * grad
            h = np.clip(h, 0.0, 1.0)
            h = h * (total_integral / np.sum(h))
            
            # Check for improvement
            current_c5 = objective_function(h)
            if current_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = current_c5
        return best_h, best_c5

    # # Function to perform simulated annealing
    def simulated_annealing(h_start, max_iter=500, T_initial=1.0, T_final=0.1, alpha=0.99):
        h = np.copy(h_start)
        best_h = np.copy(h)
        best_c5 = objective_function(h)
        T = T_initial
        for _ in range(max_iter):
            # Generate a perturbation
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbed_h = h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            
            # Evaluate the perturbed solution
            current_c5 = objective_function(perturbed_h)
            
            # Acceptance probability
            if current_c5 < best_c5 or np.random.rand() < np.exp((current_c5 - best_c5) / T):
                h = perturbed_h
                best_c5 = current_c5
            
            T *= alpha
        return best_h, best_c5

    # # Function to perform multi-start optimization
    def multi_start_optimization(h_start, num_starts=5, max_iter=100):
        best_h = np.copy(h_start)
        best_c5 = objective_function(h_start)
        for _ in range(num_starts):
            # Generate a random perturbation
            perturbation = np.random.normal(0, 0.1, size=n_points)
            perturbed_h = h_start + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            
            # Perform gradient descent
            h = np.copy(perturbed_h)
            for _ in range(max_iter):
                # Compute gradient numerically
                grad = np.zeros(n_points)
                for i in range(n_points):
                    h_plus = np.copy(h)
                    h_plus[i] += 1e-6
                    h_plus = np.clip(h_plus, 0.0, 1.0)
                    h_plus = h_plus * (total_integral / np.sum(h_plus))
                    f_plus = objective_function(h_plus)
                    
                    h_minus = np.copy(h)
                    h_minus[i] -= 1e-6
                    h_minus = np.clip(h_minus, 0.0, 1.0)
                    h_minus = h_minus * (total_integral / np.sum(h_minus))
                    f_minus = objective_function(h_minus)
                    
                    grad[i] = (f_plus - f_minus) / (2 * 1e-6)
                
                # Update step
                h -= 0.01 * grad
                h = np.clip(h, 0.0, 1.0)
                h = h * (total_integral / np.sum(h))
            
            current_c5 = objective_function(h)
            if current_c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = current_c5
        return best_h, best_c5

    # Initial optimization with gradient-based method
    start_time = time.time()
    best_h, best_c5 = optimize_with_perturbations(h, max_iter=200, perturbation_factor=0.05)
    elapsed_time = time.time() - start_time

    # Refinement with simulated annealing if time allows
    if elapsed_time < budget_s - 20:
        best_h, best_c5 = simulated_annealing(best_h, max_iter=300, T_initial=1.0, T_final=0.1, alpha=0.99)

    # Further refinement with multi-start optimization if time allows
    if elapsed_time < budget_s - 10:
        best_h, best_c5 = multi_start_optimization(best_h, num_starts=5, max_iter=100)

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
