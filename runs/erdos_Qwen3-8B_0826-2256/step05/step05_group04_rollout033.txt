        h = np.clip(h, 0, 1)

        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    for trial in range(max_trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current solution
        perturbation = np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Enforce integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Run optimization with SLSQP for local refinement
        try:
            result = minimize(
                objective,
                x0=h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Reduce perturbation magnitude for fine-tuning
        perturbation_factor *= 0.95
