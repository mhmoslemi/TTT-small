```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for efficiency
    n_points = 100
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initialize with a better structured guess
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to meet constraint
    
    # Use a different approach: combine random perturbations with gradient-based optimization
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx
    
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral
    
    # Create bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Function to perform optimization with random restarts and perturbations
    def optimize_with_restarts(h_start, max_restarts=20, max_iter=100):
        best_h = np.copy(h_start)
        best_c5 = objective_function(best_h)
        
        for _ in range(max_restarts):
            # Add random perturbation
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            
            # Optimize from perturbed initial guess
            result = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-8, 'maxiter': max_iter}
            )
            
            if result.success and result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        
        return best_h, best_c5
    
    # Initial optimization
    start_time = time.time()
    best_h, best_c5 = optimize_with_restarts(h, max_restarts=20, max_iter=100)
    elapsed_time = time.time() - start_time

    # Fine-tuning if time allows
    if elapsed_time < budget_s - 10:
        best_h, best_c5 = optimize_with_restarts(best_h, max_restarts=10, max_iter=50)
    
    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
```
