```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from sklearn.cluster import KMeans

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased points for better resolution while keeping optimization feasible
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with a structured initial guess
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    h = h * (total_integral / np.sum(h))

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # Structure search space using K-means to find cluster centers
    cluster_centers = KMeans(n_clusters=20, random_state=seed).fit(np.linspace(0, 2, n_points).reshape(-1, 1)).cluster_centers_
    cluster_centers = np.sort(cluster_centers.flatten())

    # Use a hybrid approach combining local optimization with global search
    for _ in range(20):  # Limit iterations to ensure we stay within budget
        # Perturb the current solution with structured noise
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))

        # Local optimization with SLSQP
        result = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Random restart using cluster centers
        restart_center = np.random.choice(cluster_centers)
        restart_h = np.zeros(n_points)
        restart_h[np.abs(np.linspace(0, 2, n_points) - restart_center) < 0.1] = 1.0
        restart_h = restart_h * (total_integral / np.sum(restart_h))
        result = opt.minimize(
            fun=objective_function,
            x0=restart_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Time check
        if time.time() - start_time > budget_s - 1:
            break

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
```
