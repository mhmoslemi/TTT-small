import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (using provided or a better structured initial guess)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Function to perform optimization with perturbations and simulated annealing
    def optimize_with_perturbations_and_annealing(h_start, max_iter=300, perturbation_factor=0.05, temp=1.0, temp_decay=0.995):
        best_h = np.copy(h_start)
        best_c5 = objective_function(best_h)
        current_temp = temp
        for _ in range(max_iter):
            # Perturb the current solution
            perturbation = np.random.normal(0, perturbation_factor, size=n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            
            # Evaluate the new solution
            new_c5 = objective_function(perturbed_h)
            
            # Simulated annealing decision
            if new_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - new_c5) / current_temp):
                best_h = perturbed_h
                best_c5 = new_c5
            
            # Cool down
            current_temp *= temp_decay

        return best_h, best_c5

    # Initial optimization
    start_time = time.time()
    best_h, best_c5 = optimize_with_perturbations_and_annealing(h, max_iter=200, perturbation_factor=0.05, temp=1.0, temp_decay=0.995)
    elapsed_time = time.time() - start_time

    # Fine-tuning with smaller perturbations if time allows
    if elapsed_time < budget_s - 10:
        best_h, best_c5 = optimize_with_perturbations_and_annealing(best_h, max_iter=100, perturbation_factor=0.01, temp=0.5, temp_decay=0.99)

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
