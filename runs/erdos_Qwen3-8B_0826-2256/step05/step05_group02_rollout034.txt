import scipy.optimize as optimize
import time

        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Define objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Main optimization loop
    for _ in range(20):  # Perform multiple optimization passes
        if time.time() - start_time > budget_s - 1:
            break
        # Create a perturbed version of the current best solution
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)

        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

        # Use SLSQP for local optimization with tighter tolerances
        try:
            result_slsqp = minimize(
                objective,
                x0,

                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",

            )

                constraints={"type": "eq", "fun": constraint},
                method="NelderMead",
                options={"maxiter": 150}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x
