import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, minimize, differential_evolution, SLSQP

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a slightly larger n_points for better resolution, but still under 1k
    n_points = 200
    dx = 2.0 / n_points
    
    # Initial guess based on provided initial_h_values or random construction
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//3] = 0.5
        h[n_points//3:2*n_points//3] = 1.0
        h[2*n_points//3:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Use differential evolution for global search
    def differential_evolution_run():
        nonlocal best_h, best_c5
        result = differential_evolution(
            objective,
            bounds,
            constraints={"type": "eq", "fun": constraint},
            strategy="best1bin",
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False
        )
        if result.success:
            candidate_h = result.x
            candidate_c5 = result.fun
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h
    
    # Use scaled local optimization with SLSQP
    def slsqp_refinement():
        nonlocal best_h, best_c5
        result = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result.success:
            candidate_h = result.x
            candidate_c5 = result.fun
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h
    
    # Run global search with DE
    differential_evolution_run()
    
    # Run local refinement
    slsqp_refinement()
    
    # Additional rounds of localized optimization
    for _ in range(3):
        if time.time() - start_time > budget_s - 2:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Local refinement
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result.success:
            candidate_h = result.x
            candidate_c5 = result.fun
            if candidate_c5 < best_c5:
                best_c5 = candidate_c5
                best_h = candidate_h
    
    return best_h, best_c5, n_points
