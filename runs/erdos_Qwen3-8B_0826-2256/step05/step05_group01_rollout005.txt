import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with a more structured initial guess
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0:
            h[i] = 1.0
        elif i % 5 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    # Define the objective function
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint function
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Function to perform optimization with structured perturbations
    def optimize_with_perturbations(h_start, max_iter=300, perturbation_stride=5):
        best_h = np.copy(h_start)
        best_c5 = objective_function(best_h)
        for _ in range(max_iter):
            # Apply structured perturbations
            perturbation = np.zeros(n_points)
            for i in range(n_points):
                if i % perturbation_stride == 0:
                    perturbation[i] = np.random.normal(0, 0.05)
            perturbed_h = best_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result.fun < best_c5:
                best_h = result.x
                best_c5 = result.fun
        return best_h, best_c5

    # Initial optimization
    start_time = time.time()
    best_h, best_c5 = optimize_with_perturbations(h, max_iter=200, perturbation_stride=5)
    elapsed_time = time.time() - start_time

    # Fine-tuning with smaller perturbations if time allows
    if elapsed_time < budget_s - 10:
        best_h, best_c5 = optimize_with_perturbations(best_h, max_iter=100, perturbation_stride=10)

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
