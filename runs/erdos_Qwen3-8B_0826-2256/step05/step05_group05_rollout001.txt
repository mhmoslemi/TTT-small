import numpy as np
import scipy.optimize as opt

from scipy.optimize import Bounds, NonlinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    def get_overlap_integral(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        return cross_corr

    def evaluate_robustness(h, n_points):
        overlap_integrals = get_overlap_integral(h, n_points)
        return np.max(overlap_integrals) * 2.0 / n_points

    def objective(x, n_points):
        return evaluate_robustness(x, n_points)

    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2

    def sparse_optimization(x, n_points):
        x = np.array(x)
        dx = 2.0 / n_points
        cross_corr = np.correlate(x, 1 - x, mode="full")
        max_overlap = np.max(cross_corr) * dx
        return max_overlap

    def optimization_with_nelder_mead(x, n_points):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(lambda x: sparse_optimization(x, n_points), x, method='Nelder-Mead', bounds=bounds, tol=1e-6)
        return result.x, sparse_optimization(result.x, n_points)

    def solve_with_scipy_optimization(n_points, initial_guess):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = opt.minimize(lambda x: evaluate_robustness(x, n_points), initial_guess, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
        return result.x, evaluate_robustness(result.x, n_points)

    def local_search(n_points, initial_guess, iterations=10):
        best_h = initial_guess.copy()
        best_c5 = evaluate_robustness(best_h, n_points)
        for _ in range(iterations):
            perturbation = np.random.normal(0, 0.01, n_points)
            perturbed_h = np.clip(best_h + perturbation, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            perturbed_c5 = evaluate_robustness(perturbed_h, n_points)
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
        return best_h, best_c5

    def optimize_with_crossover(n_points, h1, h2):
        child = np.random.rand(n_points)
        child = child * h1 + (1 - child) * h2
        child = np.clip(child, 0, 1)
        child = child * (n_points / 2) / np.sum(child)
        c5 = evaluate_robustness(child, n_points)
        return child, c5

    def genetic_algorithm(n_points, population_size=10, generations=100, mutation_rate=0.1):
        population = [np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points)) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) for p in population]
        fitness = [evaluate_robustness(p, n_points) for p in population]
        best_h = population[np.argmin(fitness)]
        best_c5 = np.min(fitness)
        
        for _ in range(generations):
            fitness = [evaluate_robustness(p, n_points) for p in population]
            indices = np.argsort(fitness)
            elite = [population[i] for i in indices[:int(population_size / 2)]]
            parents = [population[i] for i in indices[:population_size]]
            
            children = []
            for i in range(population_size // 2):
                p1, p2 = np.random.choice(parents, 2, replace=False)
                child = np.random.rand(n_points)
                child = child * p1 + (1 - child) * p2
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                children.append(child)
            
            for i in range(len(children)):
                if np.random.rand() < mutation_rate:
                    mutated = children[i] + np.random.normal(0, 0.01, n_points)
                    mutated = np.clip(mutated, 0, 1)
                    mutated = mutated * (n_points / 2) / np.sum(mutated)
                    children[i] = mutated
            
            population = elite + children
            population = [np.clip(p, 0, 1) for p in population]
            population = [p * (n_points / 2) / np.sum(p) for p in population]
            fitness = [evaluate_robustness(p, n_points) for p in population]
            
            best_index = np.argmin(fitness)
            best_h = population[best_index]
            best_c5 = fitness[best_index]
        
        return best_h, best_c5

    best_h = h_values
    best_c5 = evaluate_robustness(best_h, initial_n_points)
    best_n_points = initial_n_points
    best_time = time.time()

    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        if best_n_points < 200:
            # Try optimization with SLSQP
            optimized_h, optimized_c5 = solve_with_scipy_optimization(best_n_points, best_h)

                best_n_points = best_n_points
                best_time = time.time()

                # Try with local search
                local_h, local_c5 = local_search(best_n_points, best_h)

                    best_n_points = best_n_points

                    print(f"Improved C5: {best_c5}")

                    # Try with genetic algorithm
                    genetic_h, genetic_c5 = genetic_algorithm(best_n_points)
                    if genetic_c5 < best_c5:
                        best_h, best_c5 = genetic_h, genetic_c5
                        best_n_points = best_n_points
                        best_time = time.time()
                        print(f"Improved C5: {best_c5}")

            new_n_points = best_n_points + 10

            new_h = new_h * (new_n_points / 2) / np.sum(new_h)

                        best_n_points = new_n_points

                    else:
                        # Check if improvement is too small to continue
                        if time.time() - best_time > 5:
                            break

    return best_h, best_c5, best_n_points
