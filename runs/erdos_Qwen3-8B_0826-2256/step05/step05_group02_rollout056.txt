import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import NonlinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Use the initial_h_values if provided
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initialize with a reasonable structure
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 50
    cooling_factor = 0.95
    perturbation_scale = 0.2
    max_iter_per_trial = 50

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Define the constraints as a NonlinearConstraint
        constraints = NonlinearConstraint(integral_constraint, 0, 0, jac=None)

        # Run SLSQP with tight tolerances and fixed iterations
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraints,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": max_iter_per_trial}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Decrease perturbation scale for finer tuning
        perturbation_scale *= cooling_factor

    return best_h, best_c5, n_points
