def run(seed=42, budget_s=100, **kwargs):
    # Define the number of points
    n_points = 100
    n_points = max(n_points, len(initial_h_values))
    # Initialize the solution
    h = np.array(initial_h_values[:n_points], dtype=float)
    # Normalize the initial h to ensure the integral is 1
    h = h / (np.sum(h) * 2.0 / n_points)
    # Convert to a list for optimization
    h = h.tolist()
    # Track best solution
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Define the objective function to minimize
    def objective_func(h_values):
        h = np.array(h_values, dtype=float)
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * 2.0 / n_points)
        # Compute the overlap integral
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / n_points)
        return c5
    
    # Define constraints (h_i in [0, 1])
    def constraints_func(h_values):
        return [0 <= x <= 1 for x in h_values]
    
    # Define a function to perform optimization
    def optimize_solution(h_initial, max_iter=20):
        h = h_initial[:]
        for _ in range(max_iter):
            # Random perturbation around the current solution
            perturbation = np.random.normal(0, 0.05, size=len(h))
            h_perturbed = np.clip(h + perturbation, 0, 1)
            # Normalize the perturbed solution
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points)
            # Evaluate the new solution
            c5_perturbed = objective_func(h_perturbed)
            # Accept better solution
            if c5_perturbed < objective_func(h):
                h = h_perturbed
        return h
    
    # Run the optimization loop
    for _ in range(20):
        # Optimize the current solution
        h = optimize_solution(h)
        # Evaluate the current solution
        current_c5 = objective_func(h)
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
    
    return best_h, best_c5, n_points
