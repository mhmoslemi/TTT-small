```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Preloaded initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 45  # Start with the initial n_points
    max_trials = 10  # Limited trials due to time constraint
    
    # Try initial solution
    h_initial = initial_h_values.copy()
    c5_initial = evaluate_erdos_solution(h_initial)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = h_initial
    
    # Try increasing n_points to potentially find better solution
    for n_points in [60, 75, 90, 105, 120]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate random initial guess
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1
        
        # Use L-BFGS-B to optimize
        def objective(x):
            h = x.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            return np.max(overlap) * (2.0 / n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        
        if result.success:
            h_optimized = result.x
            c5_optimized = evaluate_erdos_solution(h_optimized)
            if c5_optimized < best_c5:
                best_c5 = c5_optimized
                best_h = h_optimized
        
        # Try small perturbations to avoid local minima
        for _ in range(3):
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h_optimized + perturbation, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            c5_perturbed = evaluate_erdos_solution(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed
    
    # Try convex optimization approach for better results
    if time.time() - start_time < budget_s - 1:
        n_points = 90
        h = cp.Variable(n_points)
        objective = cp.Maximize(cp.norm(signal.correlate(h.value, 1 - h.value, mode="full"), 1))
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver='SCS', verbose=False)
            h_optimized = h.value
            c5_optimized = evaluate_erdos_solution(h_optimized)
            if c5_optimized < best_c5:
                best_c5 = c5_optimized
                best_h = h_optimized
        except:
            pass
    
    # Final check for best solution
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        return (initial_h_values, c5_initial, 45)
```
