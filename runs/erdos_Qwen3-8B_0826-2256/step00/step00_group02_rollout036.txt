import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points_start = 50  # Start with a moderate number of samples
    max_n_points = 100   # Maximum number of samples to consider
    max_iterations = 100  # Max iterations for optimization
    
    best_c5 = float('inf')
    best_h = None
    best_n = n_points_start
    
    # Initial guess
    h_init = copy.copy(initial_h_values)
    h_init = np.array(h_init)
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        nonlocal n_points_start
        n_points = len(h)
        dx = 2.0 / n_points
        h_pad = np.zeros(n_points * 2)
        h_pad[:n_points] = h
        convolution = np.correlate(h_pad, 1 - h_pad, mode="full")
        overlap_integral = convolution[n_points - 1 : 2 * n_points - 1] * dx
        return np.max(overlap_integral)
    
    # Function to optimize h using gradient descent
    def optimize_h(h_initial, n_points, max_iterations=100):
        h = h_initial.copy()
        h = h / (np.sum(h) * 2 / n_points)  # Normalize to meet the constraint
        
        for _ in range(max_iterations):
            # Compute current C5
            current_c5 = evaluate_c5(h)
            
            # If we've exceeded our budget, break
            if time.time() - start_time > budget_s:
                return h, current_c5
            
            # Compute the derivative (approximate gradient)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                c5_plus = evaluate_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= 1e-5
                c5_minus = evaluate_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update using gradient descent
            h -= 0.1 * grad
            
            # Ensure h remains in [0, 1]
            h = np.clip(h, 0, 1)
            
            # Ensure the integral constraint is satisfied
            h = h * (n_points / 2) / np.sum(h)
        
        return h, current_c5
    
    # Start the optimization loop
    start_time = time.time()
    
    for n_points in range(n_points_start, max_n_points + 1):
        # Initialize h with a finer grid
        h = np.linspace(0, 2, n_points)
        h_values = np.zeros(n_points)
        h_values[:n_points_start] = initial_h_values[:n_points_start]
        h_values[n_points_start:] = np.random.rand(n_points - n_points_start)
        
        # Normalize h to meet the constraint
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Run the optimization
        h_opt, c5 = optimize_h(h_values, n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_points
        
        # If we've exceeded our budget, break
        if time.time() - start_time > budget_s:
            break
    
    return best_h, best_c5, best_n
