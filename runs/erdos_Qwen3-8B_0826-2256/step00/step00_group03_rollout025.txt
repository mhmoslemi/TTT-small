import numpy as np
import scipy.optimize as opt
import time
import random
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Predefined initial h_values (n=70)
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5],
                              dtype=np.float64)
    
    # Initial n_points
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    sum_h = np.sum(initial_h_values)
    if not np.isclose(sum_h, n_points / 2):
        initial_h_values = initial_h_values * (n_points / 2) / sum_h
    
    def objective_function(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, len(h) / 2):
            h = h * (len(h) / 2) / sum_h
        one_minus_h = 1 - h
        correlation = correlate(h, one_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    def objective_function_with_gradient(h):
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if not np.isclose(sum_h, len(h) / 2):
            h = h * (len(h) / 2) / sum_h
        one_minus_h = 1 - h
        correlation = correlate(h, one_minus_h, mode="full")
        c5 = np.max(correlation) * dx
        
        # Compute gradient numerically
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = np.clip(h_plus, 0, 1)
            sum_h_plus = np.sum(h_plus)
            if not np.isclose(sum_h_plus, len(h_plus) / 2):
                h_plus = h_plus * (len(h_plus) / 2) / sum_h_plus
            one_minus_h_plus = 1 - h_plus
            correlation_plus = correlate(h_plus, one_minus_h_plus, mode="full")
            c5_plus = np.max(correlation_plus) * dx
            grad[i] = (c5_plus - c5) / 1e-6
        return c5, grad
    
    # Optimize with initial guess
    bnds = [(0, 1) for _ in range(n_points)]
    
    result = minimize(
        objective_function,
        initial_h_values,
        bounds=bnds,
        method='L-BFGS-B',
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    best_h = result.x
    best_c5 = result.fun
    best_n_points = n_points
    
    # Try reducing n_points for faster optimization
    for n_points in [50, 40, 30, 20, 10]:
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        h = h * (n_points / 2) / sum_h
        
        result = minimize(
            objective_function,
            h,
            bounds=bnds,
            method='L-BFGS-B',
            jac=None,
            options={'maxiter': 20, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = result.x
                best_c5 = current_c5
                best_n_points = n_points
    
    # Evaluate the best solution
    best_h = np.clip(best_h, 0, 1)
    sum_h = np.sum(best_h)
    if not np.isclose(sum_h, best_n_points / 2):
        best_h = best_h * (best_n_points / 2) / sum_h
    one_minus_h = 1 - best_h
    correlation = correlate(best_h, one_minus_h, mode="full")
    c5 = np.max(correlation) * (2.0 / best_n_points)
    
    # Check time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        return (best_h, c5, best_n_points)
    
    return (best_h, c5, best_n_points)
