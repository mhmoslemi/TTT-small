import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = np.inf
    best_h = None
    n_points = 80  # Starting with 80 samples, can adjust if needed

    # Start with initial guess
    h = initial_h_values.copy()
    h = h[:n_points]
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h

    # Define grid
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points)

    # Evaluate initial solution
    print(f"Initial C5: {c5:.6f}")

    # Local search with simulated annealing
    def objective_function(h_values):
        h = h_values.copy()
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr * dx)
        return c5_val

    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2

    # Simulated Annealing
    def simulated_annealing(n_iter=1000, temp=100.0, cooling_rate=0.995):
        nonlocal best_c5, best_h
        h = initial_h_values.copy()
        h = h[:n_points]
        h = np.clip(h, 0, 1)
        current_c5 = objective_function(h)
        best_c5 = current_c5
        best_h = h.copy()

        for i in range(n_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate a random perturbation
            idx = np.random.randint(n_points)
            delta = np.random.uniform(-0.1, 0.1)
            h_new = h.copy()
            h_new[idx] += delta
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            # Ensure sum constraint is approximately satisfied
            if abs(h_new_sum - n_points / 2) < 1e-3:
                new_c5 = objective_function(h_new)
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_new.copy()
                # Accept based on simulated annealing
                if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                    h = h_new.copy()
                    current_c5 = new_c5
            temp *= cooling_rate

    simulated_annealing(n_iter=1000, temp=100.0, cooling_rate=0.995)

    # Try with more points if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 100
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * n_points / 2
        h = np.clip(h, 0, 1)
        current_c5 = objective_function(h)
        best_c5 = current_c5
        best_h = h.copy()

        def optimize_h():
            nonlocal best_c5, best_h
            h = cp.Variable(n_points)
            constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
            objective = cp.Maximize(cp.Max(signal.correlate(h.value, 1 - h.value, mode="full") * dx))
            prob = cp.Problem(objective, constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-4, max_iter=100)
                if prob.status == 'optimal':
                    best_h = h.value
                    best_c5 = objective.value
            except:
                pass

        optimize_h()

    # Final evaluation
    print(f"Best C5: {best_c5:.6f}")
    return (best_h, best_c5, n_points)
