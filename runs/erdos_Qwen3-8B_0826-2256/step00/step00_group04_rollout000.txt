import scipy.optimize as optimize
import scipy.signal as signal

import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.1542, 0.3121, 0.3307, 0.3277, 0.3553, 0.3304, 0.3306, 0.3407, 0.3508, 0.3541,
                             0.3649, 0.3493, 0.3543, 0.3591, 0.3599, 0.3538, 0.3654, 0.3605, 0.3482, 0.3501,
                             0.3579, 0.3546, 0.3533, 0.3515, 0.3604, 0.3636, 0.3539, 0.3609, 0.3568, 0.3512,
                             0.3555, 0.3532, 0.3592, 0.3549, 0.3546, 0.3538, 0.3543, 0.3521, 0.3514, 0.3596,
                             0.3584, 0.3533, 0.3557, 0.3536, 0.3601, 0.3563, 0.3536, 0.3561, 0.3552, 0.3582])

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Function to evaluate the overlap integral
    def evaluate(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        if np.sum(h) * (2.0 / len(h)) != 1:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Initial evaluation
    initial_c5 = evaluate(best_h)
    best_c5 = initial_c5
    
    # Try a few different starting points: double the samples, then half
    for scale in [2, 0.5]:
        n_points_new = int(len(best_h) * scale)
        if n_points_new < 20:
            continue
        h_new = np.zeros(n_points_new)
        h_new[:len(best_h)//2] = 0.5
        h_new[len(best_h)//2:] = 0.5
        h_new = np.random.permutation(h_new)
        h_new /= np.sum(h_new) * (2.0 / n_points_new)
        h_new = np.clip(h_new, 0, 1)
        c5_new = evaluate(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Perform a grid search over a few values
    for n in [100, 200, 300]:
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.rand(n)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (2.0 / n)
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Perform a gradient descent-like optimization
    def objective(x):
        h = np.clip(x, 0, 1)
        if np.sum(h) * (2.0 / len(h)) != 1:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))
    
    x0 = best_h.copy()
    result = optimize.minimize(objective, x0, method='L-BFGS-B', options={'maxiter': 50})
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    # Ensure we are within the time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, final_c5, len(best_h))
    
    # Additional refinement using CVXPY
    n = len(best_h)
    x = cp.Variable(n)
    constraints = [0 <= x, x <= 1]
    constraints.append(cp.sum(x) * (2.0 / n) == 1)
    
    def cvx_func(x):
        h = np.clip(x, 0, 1)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n)
    
    prob = cp.Problem(cp.Minimize(cvx_func(x)), constraints)
    prob.solve(solver=cp.SCS, eps=1e-4)
    if prob.status == 'optimal':
        best_h = np.clip(x.value, 0, 1)
        best_c5 = cvx_func(x.value)
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h, final_c5, len(best_h))
