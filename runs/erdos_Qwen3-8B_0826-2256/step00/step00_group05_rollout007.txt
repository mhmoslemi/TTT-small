import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    h_initial = initial_h_values / 2  # Normalize to [0, 1] if needed
    h_initial = np.clip(h_initial, 0, 1)
    
    # Ensure the integral constraint is satisfied
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        # Convert to a function for convolution
        h_func = np.interp(np.linspace(0, 2, n_points * 2), np.linspace(0, 2, n_points), h)
        correlation = np.correlate(h_func, 1 - h_func, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Define optimization problem
    h = cp.Variable(n_points)
    objective = cp.Minimize(0)
    
    # Constraints
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    
    # Objective function: minimize the maximum overlap
    # Since we cannot directly minimize the max, we use an iterative approach with bounds
    # We will use a heuristic optimization approach with local search
    
    # Initial guess
    h_values = h_initial.copy()
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    tolerance = 1e-4
    best_c5 = float('inf')
    best_h = h_values
    
    # Iterative improvement
    for _ in range(max_iterations):
        # Evaluate current C5
        current_c5 = evaluate_overlap(h_values)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # If we meet the tolerance, stop
        if current_c5 < tolerance:
            break
        
        # Update h_values using gradient descent (heuristic)
        # We use a simple gradient approximation around the current h_values
        # Create a small perturbation
        perturbation = np.random.normal(0, 0.01, size=n_points)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        h_perturbed = h_values + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate perturbed C5
        perturbed_c5 = evaluate_overlap(h_perturbed)
        
        # Update based on the difference
        if perturbed_c5 < current_c5:
            h_values = h_perturbed
        else:
            # Backtrack
            h_values = h_values - learning_rate * perturbation
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h)
    
    # Return the result
    return (best_h, final_c5, n_points)
