import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduce number of points for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def h_sampler(n_points, seed=42):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
        h = np.clip(h, 0, 1)
        h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

    def simulate_search(n_points, budget_s, seed):
        best_h = h_sampler(n_points, seed)
        best_c5 = objective_function(best_h)

        start_time = time.time()
        while time.time() - start_time < budget_s:
            h = h_sampler(n_points, seed + int(time.time()))
            c5 = objective_function(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        return best_h, best_c5

    # Run the simulation
    best_h, best_c5 = simulate_search(n_points, budget_s, seed)

    return best_h, best_c5, n_points
