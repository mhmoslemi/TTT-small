evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with 200 points for balance between speed and quality
    h = copy.deepcopy(initial_h_values[:n_points])
    h = h / (n_points / 2.0)  # Ensure the integral constraint is satisfied
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Optimization steps
    for _ in range(15):  # Max 15 iterations
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # First: Gradient descent-style update
        def objective_function(h):
            return evaluate_erdos_solution(h)

        # Use a simple gradient-based approach
        h = h.astype(float)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            grad[i] = (objective_function(h_plus) - objective_function(h_minus)) / (2 * 1e-4)

        # Update h in the direction of decreasing C5
        h -= 0.1 * grad

        # Enforce bounds and integral constraint
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) / (n_points / 2.0))

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Second: Simulated annealing-style exploration
        for _ in range(10):
            if elapsed >= budget_s:
                break
            i = np.random.randint(len(h))
            delta = np.random.uniform(-0.1, 0.1)
            h_try = h.copy()
            h_try[i] += delta
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try / (np.sum(h_try) / (n_points / 2.0))
            current_c5_try = evaluate_erdos_solution(h_try)
            if current_c5_try < best_c5:
                best_c5 = current_c5_try
                best_h = h_try.copy()
                h = h_try.copy()

        # Optionally, try a different n_points
        if _ == 0:
            n_points = 200
        elif _ == 1:
            n_points = 250
        elif _ == 2:
            n_points = 300
        elif _ == 3:
            n_points = 350
        elif _ == 4:
            n_points = 400
        elif _ == 5:
            n_points = 450
        elif _ == 6:
            n_points = 500
        elif _ == 7:
            n_points = 550
        elif _ == 8:
            n_points = 600
        elif _ == 9:
            n_points = 650
        elif _ == 10:
            n_points = 700
        elif _ == 11:
            n_points = 750
        elif _ == 12:
            n_points = 800
        elif _ == 13:
            n_points = 850
        elif _ == 14:
            n_points = 900

        # Reconstruct h at new n_points
        h = copy.deepcopy(best_h)
        h = h[:n_points]
        h = h / (np.sum(h) / (n_points / 2.0))
        h = np.clip(h, 0, 1)

    return (best_h, best_c5, n_points)
