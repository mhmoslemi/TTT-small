evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial construction is provided

def run(seed=42, budget_s=100, **kwargs):

    start_time = time.time()
    
    # Define the number of points
    n_points = 70  # Starting with 70 points (can be adjusted if needed)

    # Objective function to minimize the maximum overlap integral
    def objective(h_values):
        h = h_values.copy()
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraint: sum(h) * dx == 1 => sum(h) == n_points / 2
    # Define optimization variables (we'll use a simple gradient-based approach for now)
    h_opt = cp.Variable(n_points, nonneg=True, name='h')
    constraints = [cp.sum(h_opt) == n_points / 2]
    
    # Optimization problem: minimize the maximum overlap
    # Convert to a problem that can be optimized (this is a proxy)
    # For real optimization, we need to convert to a form that can be minimized
    def optimize_h(h_opt, h_vals):
        # Simple gradient descent optimizer
        for _ in range(100):  # Limited iterations for speed
            corr = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5 = np.max(corr) * dx
            # Gradient approximation
            grads = np.zeros_like(h_vals)
            for i in range(n_points):
                for k in range(n_points):

                    perturb = np.zeros_like(h_vals)

            h_vals -= 0.01 * grads

        return h_vals

    best_h = np.copy(h)
