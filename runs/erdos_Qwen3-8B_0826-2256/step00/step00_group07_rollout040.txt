```python
import numpy as np
import scipy.optimize as optimize
import random
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    
    # Initial guess
    h_initial = initial_h_values.copy()[:n_points]
    h_initial = np.array(h_initial) * (n_points / 2) / np.sum(h_initial)
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, np.zeros(n_points, dtype=h.dtype)])  # Pad for full correlation
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = np.max(corr * dx)
        return overlap
    
    # Objective function to minimize
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        return evaluate_overlap(h)
    
    # Bounds for each h[i]
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Optimization
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 100, 'ftol': 1e-8}
    )
    
    # Extract best solution
    best_h = result.x
    best_c5 = evaluate_overlap(best_h)
    
    # Refinement: Try a few more local improvements
    for _ in range(3):
        # Apply a small perturbation to avoid local minima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0.0, 1.0)
        perturbed_c5 = evaluate_overlap(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = perturbed_c5
    
    # Final check for time budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 10:
        # If time is tight, return the best found so far
        return (best_h, best_c5, n_points)
    
    # Optional: Try a slightly higher n_points if time allows
    n_points_next = min(200, n_points + 50)
    dx_next = 2.0 / n_points_next
    h_next = np.interp(np.linspace(0, 2, n_points_next), np.linspace(0, 2, n_points), best_h)
    h_next = np.clip(h_next, 0.0, 1.0)
    h_next = h_next * (n_points_next / 2) / np.sum(h_next)
    
    # Evaluate with the new n_points
    best_c5_next = evaluate_overlap(h_next)
    if best_c5_next < best_c5:
        best_h = h_next
        best_c5 = best_c5_next
        n_points = n_points_next
    
    return (best_h, best_c5, n_points)
```
