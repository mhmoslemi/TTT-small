def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(initial_h_values)
    
    def objective(h):
        nonlocal best_c5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    # Initial solution
    initial_h = initial_h_values.copy()
    initial_c5 = evaluate_erdos_solution(initial_h)
    best_c5 = initial_c5
    best_h = initial_h
    
    # Random perturbation
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        idx = np.random.choice(n_points)
        h[idx] = np.random.uniform(0, 1)
        h /= np.sum(h) * (2.0 / n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Simulated annealing
    temp = 1.0
    cooling_rate = 0.995
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        idx = np.random.choice(n_points)
        h[idx] = np.random.uniform(0, 1)
        h /= np.sum(h) * (2.0 / n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5 or np.random.rand() < np.exp((best_c5 - c5) / temp):
            best_c5 = c5
            best_h = h
        temp *= cooling_rate
    
    # Optimization with CVXOPT
    n = n_points
    h = cp.Variable(n)
    constraints = [h >= 0, h <= 1, cp.sum(h) == n / 2]
    obj = cp.Minimize(cp.Max(signal.correlate(h.value, 1 - h.value, mode="full") * (2.0 / n)))
    try:
        prob = cp.Problem(obj, constraints)
        result = prob.solve(solver=cp.CVXOPT)
        if result is not None:
            best_h = h.value
            best_c5 = evaluate_erdos_solution(best_h)
    except:
        pass
    
    # Check if we should try a different starting point
    if best_c5 > 0.38092:
        n_points = 100
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i % 3 == 0 else 0.0
        h /= np.sum(h) * (2.0 / n_points)
        best_h = h
        best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        # If we're over time, return the best solution found
        return best_h, best_c5, n_points
    
    # Check for further optimality
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        h = best_h.copy()
        idx = np.random.choice(n_points)
        h[idx] = np.random.uniform(0, 1)
        h /= np.sum(h) * (2.0 / n_points)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    return best_h, best_c5, n_points
