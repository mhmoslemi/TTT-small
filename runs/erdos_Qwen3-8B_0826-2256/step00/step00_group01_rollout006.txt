import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import sys

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction
    n_points = len(initial_h_values)
    h = np.array(initial_h_values, dtype=np.float64)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h / sum_h * (n_points / 2)
    
    def compute_c5(h_values):
        h = np.array(h_values, dtype=np.float64)
        dx = 2.0 / len(h)
        shifted = np.zeros(len(h) * 2 - 1)
        for i in range(len(h)):
            shifted[i:i+len(h)] += h
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def evaluate(h_values):
        return compute_c5(h_values)
    
    # Define optimization problem
    def optimize_h(h_values):
        h = cp.Variable(len(h_values), name="h", bounds=(0, 1))
        obj = cp.Minimize(0)
        constraints = [
            cp.sum(h) == len(h_values) / 2,
        ]
        
        # Evaluate C5 for the current h
        c5_val = compute_c5(h_values)
        
        return c5_val, h_values
    
    # Optimization loop
    best_c5 = float('inf')
    best_h = np.copy(h)
    iterations = 0
    max_iterations = 100
    
    while time.time() - start_time < budget_s and iterations < max_iterations:
        # Generate perturbations
        perturbations = np.random.normal(0, 0.05, len(h))
        new_h = h + perturbations
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (len(h) / 2) / np.sum(new_h)
        
        # Evaluate new_h
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h
            print(f"New best C5: {best_c5}")
        
        # Perform gradient descent-like update
        grad = np.zeros(len(h))
        for k in range(len(h)):
            for i in range(len(h)):
                overlap = np.correlate(h, 1 - h, mode="full")
                grad[k] += np.gradient(overlap[k], dx)
        
        h -= 0.01 * grad
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
        
        # Evaluate updated h
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h
            print(f"Updated best C5: {best_c5}")
        
        iterations += 1
    
    return best_h.tolist(), best_c5, len(best_h)
