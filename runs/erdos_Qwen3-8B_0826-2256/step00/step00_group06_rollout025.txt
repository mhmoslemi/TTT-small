import numpy as np
import scipy as sp
import scipy.optimize as optimize
import numpy.random as random
from scipy.linalg import norm
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for efficiency
    max_iterations = 100
    iteration = 0

    # Initial guess from provided initial_h_values
    h_initial = initial_h_values
    if len(h_initial) != n_points:
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h = h_initial.copy()
    h = h / (np.sum(h) * 2.0 / n_points)  # Normalize to satisfy integral constraint

    def objective(h_vec, dx):
        h = h_vec.reshape(-1)
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = corr * dx
        return np.max(overlap)

    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2

    def evaluate(h_vec):
        nonlocal best_c5
        h = h_vec
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = corr * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    def simulate(h_vec):
        nonlocal best_c5
        h = h_vec
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        overlap = corr * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Initial evaluation
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode="full")
    overlap = corr * dx
    best_c5 = np.max(overlap)
    best_h = h.copy()

    # Optimization loop
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
        # Perturb the current solution
        perturbation = 0.1 * np.random.randn(n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points)

        # Evaluate the perturbed solution
        corr = np.correlate(h_perturbed, 1 - h_perturbed, mode="full")
        overlap = corr * dx
        c5 = np.max(overlap)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()
        h = h_perturbed

        # Early stopping if improvement is minimal
        if best_c5 <= 0.380876:
            break

    # If more points might help, try increasing n_points
    if best_c5 > 0.380876:
        n_points = 200
        dx = 2.0 / n_points
        h = best_h
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        h = h / (np.sum(h) * 2.0 / n_points)

        # Further optimization with more points
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
            perturbation = 0.05 * np.random.randn(n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points)

            corr = np.correlate(h_perturbed, 1 - h_perturbed, mode="full")
            overlap = corr * dx
            c5 = np.max(overlap)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_perturbed.copy()
            h = h_perturbed

    # Final evaluation
    dx = 2.0 / n_points
    corr = np.correlate(best_h, 1 - best_h, mode="full")
    overlap = corr * dx
    best_c5 = np.max(overlap)

    return best_h, best_c5, n_points
