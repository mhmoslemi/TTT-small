```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 50  # Starting with 50 samples, can adjust if needed

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        dx = 2.0 / n_points
        shifted = np.zeros_like(h)
        for k in range(1, n_points):
            shifted[k:] = h[:-k]
        overlap = np.sum(h * (1 - shifted)) * dx
        return overlap

    # Initial guess
    h_initial = initial_h_values
    h_initial = h_initial[:n_points]  # Ensure correct length
    h_initial = np.array(h_initial)
    total_area = np.sum(h_initial) * (2.0 / n_points)
    h_initial = h_initial * (n_points / 2) / total_area  # Normalize to meet the constraint

    # Optimization function
    def objective(h_vec):
        h = h_vec.copy()
        total_area = np.sum(h) * (2.0 / n_points)
        h = h * (n_points / 2) / total_area  # Normalize to meet the constraint
        c5 = compute_c5(h)
        return c5

    def constraint_total_area(h_vec):
        h = h_vec.copy()
        total_area = np.sum(h) * (2.0 / n_points)
        return total_area - (n_points / 2)  # Enforce the constraint sum(h) = n_points / 2

    # Define bounds for h values
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial guess
    h_initial = h_initial.copy()

    # Run optimization
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_total_area},
        options={'maxiter': 100, 'ftol': 1e-6}
    )

    best_h = result.x
    best_c5 = compute_c5(best_h)

    # Check if we can improve by trying a different n_points
    for n_points in [60, 70, 80]:
        h_initial = initial_h_values[:n_points]
        h_initial = np.array(h_initial)
        total_area = np.sum(h_initial) * (2.0 / n_points)
        h_initial = h_initial * (n_points / 2) / total_area  # Normalize

        def objective_n(n):
            def obj(h_vec):
                h = h_vec.copy()
                total_area = np.sum(h) * (2.0 / n)
                h = h * (n / 2) / total_area  # Normalize to meet the constraint
                c5 = compute_c5(h)
                return c5

            def constraint_total_area_n(h_vec):
                h = h_vec.copy()
                total_area = np.sum(h) * (2.0 / n)
                return total_area - (n / 2)

            bounds_n = [(0.0, 1.0) for _ in range(n)]
            result = optimize.minimize(
                obj,
                h_initial,
                method='L-BFGS-B',
                bounds=bounds_n,
                constraints={'type': 'eq', 'fun': constraint_total_area_n},
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            return compute_c5(result.x)

        try:
            c5_candidate = objective_n(n_points)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = result.x
        except:
            pass

    # Final evaluation
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)

    return (best_h.tolist(), best_c5, n_points)
```
