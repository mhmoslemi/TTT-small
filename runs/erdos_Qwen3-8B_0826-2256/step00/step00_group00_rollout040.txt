```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Start with the given initial construction
    dx = 2.0 / n_points
    h = initial_h_values.copy()  # Initial guess
    
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    best_h = h.copy()
    
    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    tolerance = 1e-4
    
    # Try a few different starting points to avoid local minima
    for i in range(3):
        # Perturb the initial solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Evaluate the initial perturbed solution
        c5_perturbed = evaluate_erdos_solution(h_perturbed, n_points, dx)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        
        # Gradient descent optimization
        for _ in range(max_iterations):
            # Compute overlap integral
            overlap = np.correlate(h_perturbed, 1 - h_perturbed, mode="full") * dx
            c5 = np.max(overlap)
            
            # Compute gradient
            grad = np.zeros(n_points)
            for i in range(n_points):
                # Approximate gradient using finite differences
                h_plus = h_perturbed.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                overlap_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
                c5_plus = np.max(overlap_plus)
                
                h_minus = h_perturbed.copy()
                h_minus[i] -= 1e-5
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                overlap_minus = np.correlate(h_minus, 1 - h_minus, mode="full") * dx
                c5_minus = np.max(overlap_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
            
            # Update the solution
            h_perturbed -= learning_rate * grad
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Check for convergence
            if np.abs(c5 - best_c5) < tolerance:
                break
        
        # Update the best solution
        if evaluate_erdos_solution(h_perturbed, n_points, dx) < best_c5:
            best_c5 = evaluate_erdos_solution(h_perturbed, n_points, dx)
            best_h = h_perturbed.copy()
    
    # Check elapsed time
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded. Returning best solution found.")
    
    return (best_h, best_c5, n_points)
```
