import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import copy

# Initial h values provided
initial_h_values = np.array([...])  # <--- replace with actual initial_h_values array

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Helper function to compute C5 given a h_values array
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        one_minus_h = 1 - h
        correlation = signal.correlate(h, one_minus_h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def objective(h_values):
        return compute_c5(h_values)
    
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Start with the initial values
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    if np.sum(h_initial) != n_points / 2:
        h_initial = (h_initial / np.sum(h_initial)) * (n_points / 2)
    
    # Optimization settings
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use differential evolution for global optimization
    result = optimize.differential_evolution(
        objective,
        bounds,
        constraints=constraints,
        strategy='best1bin',
        maxiter=100,
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed,
        disp=False
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Ensure the best_h satisfies the sum constraint
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = (best_h / np.sum(best_h)) * (n_points / 2)
        best_c5 = compute_c5(best_h)
    
    # Check if we have exceeded the time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Try with a slightly higher number of points if time allows
    n_points_new = n_points + 50
    if n_points_new <= 1000:
        h_new = np.random.rand(n_points_new)
        h_new = (h_new / np.sum(h_new)) * (n_points_new / 2)
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return (best_h, best_c5, n_points)
