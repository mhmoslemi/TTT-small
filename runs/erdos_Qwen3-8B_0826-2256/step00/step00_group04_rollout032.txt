import numpy as np
import scipy.optimize as optimize
import scipy.interpolate
import time
import random
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, n_points=256):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None

    # Start with the initial guess
    h = initial_h_values.copy()
    h = np.resize(h, n_points)
    h /= (h.sum() / (n_points / 2))  # Normalize to meet the integral constraint

    # Function to compute C5
    def compute_c5(h_values):
        dx = 2.0 / n_points
        h = np.resize(h_values, n_points)
        h = np.pad(h, (0, n_points), mode='constant')  # Pad to full length for convolution
        result = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(result) * dx
        return c5

    # Function to evaluate and update the best solution
    def update_best(h_values):
        nonlocal best_c5, best_h
        c5 = compute_c5(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
        return best_c5

    # Initial evaluation
    update_best(h)

    # Optimization loop
    for _ in range(100):  # Number of iterations
        # Random perturbation
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h + perturb, 0, 1)
        h_perturbed = np.resize(h_perturbed, n_points)
        h_perturbed /= (h_perturbed.sum() / (n_points / 2))  # Normalize

        # Evaluate
        c5 = compute_c5(h_perturbed)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_perturbed.copy()

        # Simple gradient descent with random direction
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_test = np.zeros(n_points)
            h_test[i] = 1.0
            h_test /= (h_test.sum() / (n_points / 2))
            c5_test = compute_c5(h_test)
            grad[i] = (c5_test - best_c5) / 0.001  # Approximate gradient

        # Update with momentum
        direction = -grad / np.linalg.norm(grad)
        h = np.clip(h + 0.01 * direction, 0, 1)
        h = np.resize(h, n_points)
        h /= (h.sum() / (n_points / 2))

        # Evaluate
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Check time
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break

    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
