import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start from initial_h_values
    n_points_start = len(initial_h_values)
    dx = 2.0 / n_points_start
    h_initial = initial_h_values.copy()
    sum_h = np.sum(h_initial)
    if np.isclose(sum_h, n_points_start / 2, atol=1e-8):
        pass  # Already satisfies the constraint
    else:
        # Adjust initial values to meet the constraint
        delta = (n_points_start / 2 - sum_h) / n_points_start
        h_initial += delta
        h_initial = np.clip(h_initial, 0, 1)

    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        n_points = len(h)
        dx = 2.0 / n_points
        h_extended = np.concatenate([h, np.zeros_like(h)])
        conv = signal.correlate(h_extended, 1 - h, mode="full")
        max_overlap = np.max(conv * dx)
        return max_overlap

    # Function to optimize h using a local search
    def optimize_h(h_initial, budget_s_remaining, max_attempts=10):
        nonlocal best_c5, best_h, best_n_points
        h = h_initial.copy()
        n_points = len(h)
        dx = 2.0 / n_points
        sum_h = np.sum(h)
        if not np.isclose(sum_h, n_points / 2, atol=1e-8):
            # Adjust to meet the constraint
            delta = (n_points / 2 - sum_h) / n_points
            h += delta
            h = np.clip(h, 0, 1)
        best_h = h.copy()
        best_c5 = evaluate_c5(h)
        best_n_points = n_points
        attempts = 0
        while attempts < max_attempts and time.time() - start_time < budget_s_remaining:
            # Random perturbation
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturb = h + perturb
            h_perturb = np.clip(h_perturb, 0, 1)
            sum_h_perturb = np.sum(h_perturb)
            if np.isclose(sum_h_perturb, n_points / 2, atol=1e-8):
                pass  # Already satisfies the constraint
            else:
                # Adjust to meet the constraint
                delta = (n_points / 2 - sum_h_perturb) / n_points
                h_perturb += delta
                h_perturb = np.clip(h_perturb, 0, 1)
            c5_perturb = evaluate_c5(h_perturb)
            if c5_perturb < best_c5:
                best_c5 = c5_perturb
                best_h = h_perturb.copy()
                best_n_points = n_points
            attempts += 1
        return best_c5, best_h, best_n_points

    # Try different numbers of points
    n_points_list = [n_points_start, n_points_start - 5, n_points_start + 5]
    for n_points in n_points_list:
        if n_points <= 0:
            continue
        # Generate initial guess
        h_initial = np.random.uniform(0, 1, size=n_points)
        sum_h_initial = np.sum(h_initial)
        delta = (n_points / 2 - sum_h_initial) / n_points
        h_initial += delta
        h_initial = np.clip(h_initial, 0, 1)
        # Optimize
        c5, h, n = optimize_h(h_initial, budget_s - (time.time() - start_time))
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n

    # Final check for the initial construction
    c5_initial = evaluate_c5(initial_h_values)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_h_values.copy()
        best_n_points = len(initial_h_values)

    # Final evaluation
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print(f"Time exceeded budget: {elapsed_time:.2f}s")
    else:
        print(f"Completed in {elapsed_time:.2f}s")

    return best_h, best_c5, best_n_points
