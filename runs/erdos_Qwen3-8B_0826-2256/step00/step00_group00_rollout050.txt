def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_expanded = np.zeros(2 * n_points - 1)
    h_expanded[:n_points] = h
    correlation = correlate(h, 1 - h, mode="full")
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    max_time = budget_s
    
    # Initial guess
    n_points = len(initial_h_values)
    h_initial = initial_h_values.copy()
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = np.array(h_initial, dtype=np.float64)
    
    # Define the function to minimize
    def objective(h):
        h = h.reshape(n_points)
        h = np.clip(h, 0, 1)
        dx = 2.0 / n_points
        h_expanded = np.zeros(2 * n_points - 1)
        h_expanded[:n_points] = h
        correlation = correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h):
        h = h.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    bounds = [(0, 1) for _ in range(n_points)]
    
    start_time = time.time()
    result = optimize.minimize(
        objective,
        h_initial,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'ftol': 1e-8, 'maxiter': 1000}
    )
    
    elapsed = time.time() - start_time
    
    # Check if optimization finished within time
    if elapsed >= max_time:
        return (h_initial, evaluate_erdos_solution(h_initial), n_points)
    
    # Get best solution
    h_best = result.x
    c5_best = evaluate_erdos_solution(h_best)
    
    # Convert to a list
    h_values = h_best.tolist()
    
    return (h_values, c5_best, n_points)
