```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    max_n_points = 500  # Maximum number of points (adjust based on budget)
    initial_n_points = 95  # Starting with the initial construction
    max_iter = 100  # Maximum number of iterations
    n_points = initial_n_points
    
    # Initialize with the provided initial_h_values
    h = np.array(initial_h_values)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    for i in range(max_iter):
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        
        # Check if we should try increasing the number of points
        if i % 5 == 0 and n_points < max_n_points:
            n_points = min(n_points + 50, max_n_points)
            h = np.zeros(n_points)
            dx = 2.0 / n_points
            h[:int(n_points / 2)] = 1.0  # Initialize with a simple step function
            # Calculate sum to meet the constraint
            sum_h = np.sum(h)
            h = h * (n_points / 2) / sum_h if sum_h != 0 else h
            best_h = h.copy()
        
        # Check if we have exceeded the budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perform a grid search around the current solution
        for perturbation in [0.01, 0.02, 0.03]:
            h_perturbed = h.copy()
            # Randomly perturb 10% of the points
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            h_perturbed[indices] += np.random.uniform(-perturbation, perturbation, size=len(indices))
            h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
            
            # Ensure the integral constraint is satisfied
            sum_h_perturbed = np.sum(h_perturbed)
            if sum_h_perturbed != 0:
                h_perturbed = h_perturbed * (n_points / 2) / sum_h_perturbed
            
            c5_perturbed = evaluate_erdos_solution(h_perturbed)
            
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
    
    # Final check for the best solution
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h, final_c5, len(best_h))
```
