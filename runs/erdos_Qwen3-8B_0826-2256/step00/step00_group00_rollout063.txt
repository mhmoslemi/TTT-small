import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.4897, 0.476, 0.4725, 0.4691, 0.4437, 0.4495, 0.4464, 0.4431, 0.4177, 0.4164,
                             0.4108, 0.4174, 0.4186, 0.4136, 0.4148, 0.4193, 0.4114, 0.4138, 0.4025, 0.4131,
                             0.4143, 0.4121, 0.4105, 0.4161, 0.4135, 0.4109, 0.4136, 0.4148, 0.4044, 0.4153,
                             0.4207, 0.4139, 0.4177, 0.4092, 0.4199, 0.4148, 0.4212, 0.4125, 0.4222, 0.4074,
                             0.4189, 0.4117, 0.4191, 0.4146, 0.4186, 0.4171, 0.4188, 0.4127, 0.4175, 0.4113,

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial number of points
    n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    
    # Evaluate current C5
    current_c5 = evaluate_erdos_solution(h_values)
    best_h = h_values.copy()
    best_c5 = current_c5
    n_points = len(h_values)
    time_used = time.time() - start_time
    
    # Function to compute C5
    def compute_c5(h):
        return evaluate_erdos_solution(h)
    
    # Function to compute the constraint: integral of h over [0, 2] = 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Function to evaluate the objective and constraints
    def objective_and_constraints(h):
        return compute_c5(h), constraint_sum(h)
    
    # Optimization loop
    n_iterations = 100
    for _ in range(n_iterations):
        if time_used >= budget_s:
            break
        
        # Evaluate current h
        current_c5 = compute_c5(h_values)
        
        # If current C5 is better than best, update
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_values.copy()
        
        # Try to perturb h
        perturbations = np.random.uniform(-0.1, 0.1, size=len(h_values))
        perturbed_h = np.clip(h_values + perturbations, 0, 1)
        
        # Evaluate perturbed h
        perturbed_c5 = compute_c5(perturbed_h)
        
        # Update best solution if better
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h.copy()
        
        # Update time
        time_used = time.time() - start_time
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return best_h, best_c5, len(best_h)
