import numpy as np
import scipy.optimize as optimize
import time
import logging
import random

# Pre-imported functions
# evaluate_erdos_solution() is a function that computes the overlap integral
# initial_h_values is a starting point for the optimization

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with a manageable number of points

    # Initial guess from provided starting point
    initial_h = initial_h_values.copy()
    initial_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
    initial_h = (initial_h * n_points / 2).astype(float)

    # Try different numbers of points up to a maximum to balance speed and accuracy
    for n_points in [100, 200, 300, 400, 500, 600, 700, 800, 900, 1000]:
        if time.time() - start_time > budget_s - 1:
            break

        # Ensure the number of points is even for symmetry
        if n_points % 2 != 0:
            n_points += 1

        # Create grid
        x = np.linspace(0, 2, n_points)
        dx = 2.0 / n_points

        # Define the objective function: maximize the overlap integral
        def objective(h):
            h = np.clip(h, 0, 1)
            h = np.concatenate([h, h[:n_points//2]])  # Extend for full correlation
            h_shifted = np.concatenate([np.zeros(n_points//2), h[:-n_points//2]])
            overlap = np.correlate(h, 1 - h_shifted, mode="full")
            c5 = np.max(overlap) * dx
            return c5

        # Define the constraint: sum(h) == n_points / 2
        def constraint(h):
            return np.sum(h) - n_points / 2

        # Setup optimization problem
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        h0 = initial_h.copy()

        # Solve the optimization problem
        try:
            result = optimize.minimize(objective, h0, method='SLSQP', bounds=bounds, constraints=cons, tol=1e-6)
            if result.success:
                h_candidate = result.x
                c5_candidate = objective(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate
        except:
            pass

        # Try random perturbations to escape local minima
        for _ in range(5):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_perturbed = np.clip(h0 + perturbation, 0, 1)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed

    # Final check
    if best_h is not None:
        best_h = np.clip(best_h, 0, 1)
        c5_bound = best_c5
        n_points = len(best_h)
        return (best_h, c5_bound, n_points)
    else:
        return (initial_h, float('inf'), n_points)
