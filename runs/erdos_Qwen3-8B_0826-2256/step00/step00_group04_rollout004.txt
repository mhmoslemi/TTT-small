def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    start_time = time.time()

    # Heuristic: try starting with different n_points
    for n_points in [42, 64, 84, 100, 126, 140, 168, 196, 210, 224, 252, 280, 308, 336, 364, 392]:

            break

        if len(initial_h_values) != n_points:
            h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h_values = initial_h_values.copy()
        # Scale to satisfy the integral constraint
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        
        # Local search: random perturbations
        for _ in range(100):
            if time.time() - start_time > budget_s:
                break
            # Random perturbation
            idxs = np.random.choice(n_points, size=10, replace=False)
            h_values[idxs] += np.random.uniform(-0.1, 0.1, size=10)
            h_values = np.clip(h_values, 0, 1)
            # Enforce integral constraint
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            
            # Evaluate the C₅
            c5 = evaluate_erdos_solution(h_values)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_values.copy()
        
        # Genetic algorithm with tournament selection
        population_size = 20
        n_generations = 20
        mutation_rate = 0.1
        for gen in range(n_generations):
            if time.time() - start_time > budget_s:
                break
            # Generate population
            population = []
            for _ in range(population_size):
                h = h_values.copy()
                # Random perturbation
                idxs = np.random.choice(n_points, size=10, replace=False)
                h[idxs] += np.random.uniform(-0.1, 0.1, size=10)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
            
            # Evaluate fitness
            fitness = [evaluate_erdos_solution(h) for h in population]
            # Tournament selection
            new_population = []
            for _ in range(population_size):
                idxs = np.random.choice(population_size, size=2, replace=False)
                if fitness[idxs[0]] < fitness[idxs[1]]:
                    winner = population[idxs[0]]
                else:
                    winner = population[idxs[1]]
                new_population.append(winner)
            
            # Mutation
            for i in range(population_size):
                if np.random.rand() < mutation_rate:
                    h = new_population[i].copy()
                    idxs = np.random.choice(n_points, size=10, replace=False)
                    h[idxs] += np.random.uniform(-0.1, 0.1, size=10)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    new_population[i] = h
            
            population = new_population
            # Update best
            for h in population:
                c5 = evaluate_erdos_solution(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
    
    # Additional local optimization with convex optimization
    if time.time() - start_time < budget_s - 10:
        n_points = len(best_h)
        h = cp.Variable(n_points)
        constraints = [0 <= h, h <= 1, cp.sum(h) == n_points / 2]
        # Objective: minimize the maximum overlap
        # Note: We approximate the max overlap by evaluating it over all shifts
        # This is an NP-hard problem, so we use a heuristic
        overlap_values = []
        for k in range(1, n_points):
            overlap = cp.multiply(h, 1 - cp.roll(h, k)) 
            overlap_values.append(overlap)
        objective = cp.Minimize(cp.Max(overlap_values))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS)
            if prob.status == 'OPTIMAL':
                best_h = h.value
                best_c5 = evaluate_erdos_solution(best_h)
        except:
            pass
    
    return (best_h, best_c5, n_points)
