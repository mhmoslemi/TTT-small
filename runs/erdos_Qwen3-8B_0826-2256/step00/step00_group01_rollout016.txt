import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
from time import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder, actual implementation would load initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Start with the initial solution
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h
    
    # Try a few other starting points with different n_points
    for n_points in [50, 100, 150, 200]:
        h = np.random.rand(n_points)
        h = h * 2.0 / np.sum(h)  # Normalize to ensure sum is 1
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Optimization
    for n_points in [50, 100, 150, 200]:
        if time() - start_time > budget_s - 10:
            break
        
        # Try to find a better solution by optimizing with CVXPY
        h = np.random.rand(n_points)
        h = h * 2.0 / np.sum(h)  # Normalize to ensure sum is 1
        
        # Define variables
        x = cp.Variable(n_points)
        constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
        
        # Define objective function
        def objective(x):
            # Compute convolution and maximum value
            conv = signal.correlate(x.value, 1 - x.value, mode="full")
            return np.max(conv) * (2.0 / n_points)
        
        # Use a local optimization method to improve
        result = optimize.minimize(objective, x.value, method='L-BFGS-B', jac=None, options={'ftol': 1e-9, 'gtol': 1e-9})
        x.value = result.x
        
        # Check constraints and update best solution
        if np.all(0 <= x.value) and np.all(x.value <= 1) and np.isclose(np.sum(x.value), n_points / 2):
            c5 = evaluate_erdos_solution(x.value)
            if c5 < best_c5:
                best_c5 = c5
                best_h = x.value
    
    # Final result
    elapsed_time = time() - start_time
    return best_h, best_c5, len(best_h)
