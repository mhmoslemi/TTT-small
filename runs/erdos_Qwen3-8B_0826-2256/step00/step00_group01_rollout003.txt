```python
import numpy as np
import scipy.optimize as optimize
import time
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 100  # Use 100 points for faster optimization
    dx = 2.0 / n_points
    h_initial = initial_h_values.copy()[:n_points]  # Start with initial construction
    
    # Normalize the initial guess to satisfy the integral constraint
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def objective(h):
        # Ensure h is within [0, 1] and has the correct sum
        h = np.clip(h, 0, 1)
        if np.sum(h) * dx != 1:
            return float('inf')
        
        # Compute overlap integral
        h_pad = np.concatenate([h, np.zeros_like(h)])
        h_shifted = np.concatenate([np.zeros_like(h), h])
        overlap = np.correlate(h_pad, h_shifted, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization with bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    
    # Use Sequential Least Squares Programming (SLSQP) for optimization
    result = optimize.minimize(
        objective,
        h_initial,
        bounds=bounds,
        constraints=cons,
        method='SLSQP',
        options={'maxiter': 100, 'ftol': 1e-6, 'eps': 1e-6}
    )
    
    # Check if we have enough time to run additional iterations
    elapsed = time.time() - start_time
    if elapsed < budget_s - 5:
        # Refine with a local search
        result = optimize.minimize(
            objective,
            result.x,
            bounds=bounds,
            constraints=cons,
            method='L-BFGS-B',
            options={'maxiter': 50, 'ftol': 1e-6, 'eps': 1e-6}
        )
    
    # Extract the best solution
    best_h = result.x
    best_c5 = objective(best_h)
    
    # Evaluate and return the best solution
    evaluate_erdos_solution(best_h, n_points, dx)
    return best_h, best_c5, n_points
```
