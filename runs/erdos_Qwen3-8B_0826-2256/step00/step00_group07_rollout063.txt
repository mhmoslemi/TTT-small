```python
import numpy as np
import scipy.optimize as opt
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    max_n_points = 500  # Balance between quality and speed
    n_points = 500
    dx = 2.0 / n_points
    
    # Initial guess from provided or a random one
    if 'initial_h_values' in locals():
        h_values = copy.copy(initial_h_values)
    else:
        h_values = np.random.rand(n_points) * 2 - 1  # Random initial guess
        h_values = np.clip(h_values, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
    
    def objective(h):
        # Scale h to ensure integral is 1
        h_scaled = h * (n_points / 2) / np.sum(h)
        # Compute the overlap integral
        corr = np.correlate(h_scaled, 1 - h_scaled, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization
    bounds = [(0, 1) for _ in range(n_points)]
    result = opt.minimize(
        objective,
        h_values,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-8,
        options={'maxiter': 100}
    )
    
    h_optimized = result.x
    c5 = objective(h_optimized)
    
    # Early exit if time is up
    if time.time() - start_time > budget_s - 1:
        return h_optimized, c5, n_points
    
    # Perform a few more iterations with a different method
    result = opt.minimize(
        objective,
        h_optimized,
        method='SLSQP',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint},
        tol=1e-8,
        options={'maxiter': 50}
    )
    h_optimized = result.x
    c5 = objective(h_optimized)
    
    # Final check
    if time.time() - start_time > budget_s - 1:
        return h_optimized, c5, n_points
    
    # Use a local search with random perturbations
    for _ in range(10):
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = h_optimized + perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = objective(h_perturbed)
        if c5_perturbed < c5:
            h_optimized = h_perturbed
            c5 = c5_perturbed
    
    return h_optimized, c5, n_points
```
