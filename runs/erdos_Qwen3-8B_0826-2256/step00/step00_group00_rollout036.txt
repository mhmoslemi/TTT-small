import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import matplotlib.pyplot as plt

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution as eval_erdos
from evaluate_erdos_solution import initial_h_values as initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for performance
    
    # Initial guess from provided h_values
    initial_h = initial_h_values.copy()
    if len(initial_h) > n_points:
        initial_h = initial_h[:n_points]
    else:
        # Pad with zeros if needed
        initial_h += [0.0] * (n_points - len(initial_h))
    
    # Define objective function to minimize the maximum overlap
    def objective(h):
        h = np.array(h)
        # Ensure h is in [0, 1] and integral is 1
        h = np.clip(h, 0, 1)
        total = np.sum(h) * (2.0 / n_points)
        if not np.isclose(total, 1.0):
            return float('inf')
        # Compute overlap
        overlap = np.correlate(h, 1 - h, mode="full") * (2.0 / n_points)
        return np.max(overlap)
    
    # Optimization using differential evolution
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.differential_evolution(
        objective,
        bounds,
        popsize=15,
        strategy='best1bin',
        mutation=(0.5, 1.0),
        recombination=0.9,
        tol=1e-5,
        maxiter=100,
        disp=False,
        seed=seed
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Check if we can improve with more points (but within budget)
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        new_h = np.zeros(n_points)
        # Interpolate initial_h to new_h
        if len(initial_h_values) > n_points:
            initial_h = initial_h_values[:n_points]
        else:
            initial_h = initial_h_values + [0.0] * (n_points - len(initial_h_values))
        new_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h)), initial_h)
        # Optimize with more points
        result = optimize.differential_evolution(
            objective,
            [(0.0, 1.0) for _ in range(n_points)],
            popsize=15,
            strategy='best1bin',
            mutation=(0.5, 1.0),
            recombination=0.9,
            tol=1e-5,
            maxiter=100,
            disp=False,
            seed=seed
        )
        best_h = result.x
        best_c5 = result.fun
    
    # Convert to list and ensure integral is 1
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h) * (2.0 / n_points)
    if not np.isclose(total, 1.0):
        # Adjust to make integral exactly 1
        adjustment = (1.0 - total) / n_points
        best_h += adjustment
    
    # Convert to list and return
    best_h = best_h.tolist()
    return best_h, best_c5, n_points
