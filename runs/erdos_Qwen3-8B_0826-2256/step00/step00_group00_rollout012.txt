import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time
import os
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Try various initial configurations
    for i in range(3):
        if i == 0:
            h = initial_h_values.copy()
        elif i == 1:
            h = np.zeros(n_points)
            h[::2] = 1.0
        elif i == 2:
            h = np.zeros(n_points)
            h[range(n_points // 2 + 1)] = 1.0
        
        # Ensure the integral constraint is satisfied
        total_area = np.sum(h) * (2.0 / n_points)
        if np.abs(total_area - 1.0) > 1e-6:
            h = h * (1.0 / total_area) * (2.0 / n_points)
        
        # Optimize the function over a smaller number of points
        n_points = 100
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 4 == 0:
                h[i] = 1.0
            elif i % 2 == 0:
                h[i] = 0.5
            else:
                h[i] = 0.0
        
        # Use a simpler local optimization
        for _ in range(10):
            # Randomly perturb the function
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h = np.clip(h + perturbation, 0, 1)
            total_area = np.sum(h) * (2.0 / n_points)
            if np.abs(total_area - 1.0) > 1e-6:
                h = h * (1.0 / total_area) * (2.0 / n_points)
            
            # Evaluate the C5
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check if we've used up the budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Return the best solution found
    return (best_h, best_c5, n_points)
