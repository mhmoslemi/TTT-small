```python
import numpy as np
import scipy.optimize as opt
import time
import os
import sys
import random
import multiprocessing
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from scipy.sparse import lil_matrix
from scipy.sparse.linalg import spsolve
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the given initial solution
    n_points_init = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    h_initial = h_initial / (np.sum(h_initial) / n_points_init)  # Normalize to meet the constraint
    
    # Define parameters
    n_points = n_points_init  # Start with the initial number of points
    max_iterations = 10  # Max number of iterations for optimization
    n_parallel = 10  # Max number of parallel processes
    
    best_c5 = float('inf')
    best_h = h_initial.copy()
    
    def compute_c5(h):
        """Compute the C₅ value for the given function h."""
        dx = 2.0 / len(h)
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode='full')
        # Take the relevant part of the correlation
        corr = corr[len(h) - 1 : 2 * len(h) - 1]
        # Compute max value
        c5 = np.max(corr) * dx
        return c5
    
    # Function to optimize
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return compute_c5(h)
    
    # Function to evaluate with constraints
    def evaluate_with_constraints(h_flat):
        h = h_flat.copy()
        h = np.clip(h, 0.0, 1.0)
        # Ensure integral constraint is met
        if np.sum(h) < len(h) / 2 - 1e-6:
            h = h * (len(h) / 2) / np.sum(h)
        elif np.sum(h) > len(h) / 2 + 1e-6:
            h = h * (len(h) / 2) / np.sum(h)
        # Ensure values are within [0, 1]
        h = np.clip(h, 0.0, 1.0)
        return compute_c5(h)
    
    # Optimization settings
    bounds = Bounds(0.0, 1.0)
    
    for i in range(max_iterations):
        # Start optimization
        result = minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        h_candidate = result.x
        c5_candidate = compute_c5(h_candidate)
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        
        # Try perturbing the solution
        perturb = np.random.normal(0, 0.05, len(h_candidate))
        h_perturbed = np.clip(h_candidate + perturb, 0.0, 1.0)
        # Ensure integral constraint is met
        if np.sum(h_perturbed) < len(h_perturbed) / 2 - 1e-6:
            h_perturbed = h_perturbed * (len(h_perturbed) / 2) / np.sum(h_perturbed)
        elif np.sum(h_perturbed) > len(h_perturbed) / 2 + 1e-6:
            h_perturbed = h_perturbed * (len(h_perturbed) / 2) / np.sum(h_perturbed)
        # Ensure values are within [0, 1]
        h_perturbed = np.clip(h_perturbed, 0.0, 1.0)
        c5_perturbed = compute_c5(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()
        
        # Early stopping if time is up
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
    
    # Final check and return
    final_h = best_h.copy()
    final_c5 = compute_c5(final_h)
    return final_h, final_c5, len(final_h)
```
