```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h.npy')  # Assuming this is provided as a pre-loaded initial solution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can be increased if time permits
    
    # Try different numbers of points up to a maximum of 500
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on the provided initial solution or random
        if n_points <= len(initial_h_values):
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define variables
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective function: minimize the maximum overlap
        overlaps = []
        for k in range(1, n_points):
            shifted_h = np.roll(h, k)
            overlap = cp.sum(h * (1 - shifted_h))
            overlaps.append(overlap)
        objective = cp.Minimize(cp.Max(overlaps))
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.SCS, max_iter=1000, eps=1e-4, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_values = h.value
                c5 = evaluate_erdos_solution(h_values)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_values
        except:
            pass
        
        # Early exit if we've met the target
        if best_c5 <= 0.380876:
            break
    
    # Return the best solution found
    return (best_h, best_c5, len(best_h))
```
