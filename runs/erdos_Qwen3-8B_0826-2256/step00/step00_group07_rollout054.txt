import numpy as np
import scipy.optimize as opt
import time
import copy
from scipy.optimize import Bounds, minimize, LinearConstraint
from scipy.linalg import block_diag
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    max_n_points = 400  # Max number of samples allowed before timing out
    n_points = 160  # Start with 160 points for faster optimization
    
    # Initial construction
    initial_h = initial_h_values.copy()
    initial_h = initial_h[:n_points]  # Truncate to n_points
    initial_h = np.interp(initial_h, (initial_h.min(), initial_h.max()), (0, 1))
    initial_h = np.clip(initial_h, 0, 1)
    
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        dx = 2.0 / n_points
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Normalize by dx
        corr = corr * dx
        # Take max
        return np.max(corr)
    
    def constraint_sum(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) - n_points / 2
    
    # Define bounds
    bounds = Bounds(lb=0, ub=1)
    
    # Optimization
    start_time = time.time()
    result = minimize(
        objective,
        x0=initial_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint_sum}],
        tol=1e-6,
        options={'maxiter': 200}
    )
    
    best_h = result.x
    best_c5 = result.fun
    best_n_points = n_points
    
    # Check if we can increase n_points
    while time.time() - start_time < budget_s - 10 and n_points < max_n_points:
        n_points += 40
        # Interpolate the current best_h to new n_points
        new_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, best_n_points), best_h)
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h[:n_points]
        new_h = np.interp(new_h, (new_h.min(), new_h.max()), (0, 1))
        best_h = new_h
        best_n_points = n_points
        
        # Optimize the new h
        def objective_new(h_flat):
            h = h_flat.reshape(n_points)
            dx = 2.0 / n_points
            corr = np.correlate(h, 1 - h, mode="full")
            corr = corr * dx
            return np.max(corr)
        
        def constraint_sum_new(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - n_points / 2
        
        bounds_new = Bounds(lb=0, ub=1)
        result_new = minimize(
            objective_new,
            x0=new_h,
            method='L-BFGS-B',
            bounds=bounds_new,
            constraints=[{'type': 'eq', 'fun': constraint_sum_new}],
            tol=1e-6,
            options={'maxiter': 200}
        )
        
        if result_new.fun < best_c5:
            best_h = result_new.x
            best_c5 = result_new.fun
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
