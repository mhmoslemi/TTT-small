```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = None
initial_h_values = None

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Try with different n_points (smaller for faster optimization)
    n_points_list = [45, 60, 80, 100]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)
        
        # Initial guess
        if initial_h_values is not None and len(initial_h_values) == n_points:
            h = np.array(initial_h_values)
        else:
            h = np.random.uniform(0, 1, n_points)
        
        # Ensure constraint: sum(h) == n_points / 2
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the objective function
        def objective(h):
            h = np.array(h)
            overlap = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(overlap)
        
        # Define the constraints
        def constraints(h):
            h = np.array(h)
            return np.sum(h) - (n_points / 2)
        
        # Optimization using scipy.optimize
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            tol=1e-5,
            options={'maxiter': 100}
        )
        
        # Update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points
        
        # Optionally try a more refined optimization
        if best_c5 < 0.380876:
            break
    
    # Final evaluation
    if best_h is not None:
        h_values = best_h
        c5_bound = best_c5
        n_points = best_n_points
    else:
        # Default to initial guess
        h_values = initial_h_values
        c5_bound = evaluate_erdos_solution(h_values)
        n_points = len(h_values)
    
    return h_values, c5_bound, n_points
```
