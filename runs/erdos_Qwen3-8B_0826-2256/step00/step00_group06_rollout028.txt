```python
import numpy as np
import scipy.optimize as opt
import time
import copy
from scipy.optimize import minimize

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with the initial configuration
    h_values = copy.deepcopy(initial_h_values)
    n_points = len(h_values)
    dx = 2.0 / n_points
    current_c5 = evaluate_erdos_solution(h_values)
    best_c5 = current_c5
    best_h = h_values

    # Check if we can reduce the number of points
    if n_points > 500:
        n_points = 500
        dx = 2.0 / n_points
        h_values = np.zeros(n_points)
        # Initialize with a basic piecewise constant function that satisfies ∫h(x) dx = 1
        h_values[0:50] = 0.0
        h_values[50:250] = 0.02
        h_values[250:450] = 0.04
        h_values[450:500] = 0.08
        h_values = np.clip(h_values, 0, 1)
        sum_h = np.sum(h_values)
        h_values = h_values * (1 / sum_h) * (n_points / 2)
        h_values = np.clip(h_values, 0, 1)

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        return c5

    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)

    # Set up optimization
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones_like(h)}

    # Run optimization
    result = minimize(objective, h_values, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 50, 'ftol': 1e-6})
    h_opt = result.x
    c5_opt = result.fun

    # Check if the optimization improved the result
    if c5_opt < best_c5:
        best_c5 = c5_opt
        best_h = h_opt

    # Since we're in a limited time budget, do a few more local refinements
    for _ in range(3):
        # Perform a small perturbation
        perturbation = np.random.uniform(-0.05, 0.05, n_points)
        h_perturbed = np.clip(h_opt + perturbation, 0, 1)
        sum_h = np.sum(h_perturbed)
        h_perturbed = h_perturbed * (n_points / 2) / sum_h
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed

    # Return the best result found within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s:
        print("Time budget exceeded, returning best found so far")
    return best_h, best_c5, n_points
```
