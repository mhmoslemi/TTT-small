```python
import numpy as np
import scipy.optimize as opt
import time
import copy
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # We'll start with 100 points and increase if needed
    dx = 2.0 / n_points
    max_correlation = 0.0
    best_h = None
    
    # Initial guess from provided initial_h_values (n=95)
    h_initial = initial_h_values
    if len(h_initial) != n_points:
        # Resize to n_points by padding or truncating
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    
    # Function to compute the overlap integral
    def compute_overlap(h):
        nonlocal dx
        h = np.array(h)
        h_shifted = np.concatenate([h[-n_points+1:], np.zeros(n_points)])
        corr = np.correlate(h, 1 - h_shifted, mode="full")
        overlap = corr * dx
        return np.max(overlap)
    
    # Function to compute the objective function
    def objective(h):
        return compute_overlap(h)
    
    # Function to compute the constraint (integral over [0,2] equals 1)
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    h_bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial guess
    h_guess = h_initial
    
    # Optimization using SLSQP with bounds
    result = opt.minimize(
        objective,
        h_guess,
        method='SLSQP',
        bounds=h_bounds,
        constraints={'type': 'eq', 'fun': constraint},
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = compute_overlap(best_h)
    max_correlation = best_c5
    elapsed_time = time.time() - start_time
    
    # If we still have budget, try increasing n_points
    if elapsed_time < budget_s - 10:
        n_points_new = min(200, int(n_points * 1.5))
        dx_new = 2.0 / n_points_new
        h_initial_new = np.interp(np.linspace(0, 2, n_points_new), np.linspace(0, 2, len(h_initial)), h_initial)
        
        h_bounds_new = [(0, 1) for _ in range(n_points_new)]
        
        def compute_overlap_new(h):
            nonlocal dx_new
            h = np.array(h)
            h_shifted = np.concatenate([h[-n_points_new+1:], np.zeros(n_points_new)])
            corr = np.correlate(h, 1 - h_shifted, mode="full")
            overlap = corr * dx_new
            return np.max(overlap)
        
        def objective_new(h):
            return compute_overlap_new(h)
        
        def constraint_new(h):
            return np.sum(h) - n_points_new / 2
        
        result_new = opt.minimize(
            objective_new,
            h_initial_new,
            method='SLSQP',
            bounds=h_bounds_new,
            constraints={'type': 'eq', 'fun': constraint_new},
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        best_h = result_new.x
        best_c5 = compute_overlap_new(best_h)
        
        elapsed_time = time.time() - start_time
    
    # Final check to ensure we don't exceed time budget
    if elapsed_time >= budget_s:
        return best_h, max_correlation, n_points
    
    return best_h, best_c5, n_points
```
