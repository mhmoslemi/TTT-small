    def evaluate_c5(h):
        dx = 2.0 / n_points
        h = np.array(h).clip(0, 1)
        current_max = 0
        for k in range(1, n_points):
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx

                        child1[j] = p2[j]

                offspring.append(child1)

        for i in range(len(offspring)):

    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
