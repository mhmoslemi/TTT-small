    def objective(h):
        h = np.array(h)
        h = np.clip(h, 0, 1)
        h = np.concatenate((h, h[:1]))  # Extend for convolution
        conv = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(conv) * dx
        return c5

    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]

    # Optimization options
    options = {
        'disp': False,
        'maxiter': 100,
        'ftol': 1e-6,
        'gtol': 1e-6,
        'eps': 1e-8
    }

    # Initial solution
    best_h = initial_h.copy()
    best_c5 = objective(best_h)

    # Time tracking
    start_time = time.time()

    # Perform optimization
    try:
        result = minimize(
            objective,
            initial_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=[{'type': 'eq', 'fun': constraint_sum}],
            options=options
        )

        # Update best solution
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
    except:
        pass

    # Try a few random perturbations if stuck in local minima
    for _ in range(5):
        perturbed_h = best_h * (1 + 0.1 * np.random.randn(n_points))
        perturbed_h = np.clip(perturbed_h, 0, 1)

        current_c5 = objective(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5

    # Check time and return best result
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
