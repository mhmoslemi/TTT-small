import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    # Define initial number of points
    n_points = 95
    dx = 2.0 / n_points

    # Initial guess from pre-loaded values
    h_values = initial_h_values.copy()
    h_values = h_values[:n_points]

    # Ensure sum of h_values is n_points / 2
    total_sum = np.sum(h_values)
    if total_sum != n_points / 2:
        h_values = h_values / total_sum * (n_points / 2)

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr * dx)
        return c5

    # Function to update h and track the best solution
    def update_h(h):
        nonlocal best_h, best_c5
        current_c5 = evaluate_overlap(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5

    # Initialize best values
    best_h = h_values.copy()
    best_c5 = evaluate_overlap(best_h)

    # Optimization loop
    start_time = time.time()
    elapsed_time = 0

    # Optimization parameters
    max_iterations = 100
    learning_rate = 0.01
    momentum = 0.9
    decay = 0.999
    epsilon = 1e-8

    # Training loop
    for i in range(max_iterations):
        if elapsed_time >= budget_s:
            break
        # Perturb the current h_values
        perturbed_h = h_values + np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)

        # Evaluate the perturbed h_values
        current_c5 = evaluate_overlap(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h.copy()
            best_c5 = current_c5

        # Update h_values
        h_values = perturbed_h.copy()
        elapsed_time = time.time() - start_time

    return best_h, best_c5, n_points
