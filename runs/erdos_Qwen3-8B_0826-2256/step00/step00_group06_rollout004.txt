import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    random.seed(seed)
    
    # Define the number of points - we'll start with 256, which is under 1k
    n_points = 256
    dx = 2.0 / n_points
    
    # Initial guess from the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:n_points]  # Ensure it's the correct length
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2)  # Scale to satisfy the integral constraint
    
    # Function to evaluate the overlap integral
    def evaluate_c5(h):
        h = np.array(h, dtype=np.float64)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Scale to meet the integral constraint
        h_pad = np.concatenate([h, np.zeros(n_points)])
        h_shifted = np.zeros(2 * n_points)
        for k in range(n_points):
            h_shifted[k:n_points + k] = h
        h_fft = np.fft.fft(h_pad)
        h_shifted_fft = np.fft.fft(h_shifted)
        cross_corr_fft = h_fft * np.conj(h_shifted_fft)
        cross_corr = np.fft.ifft(cross_corr_fft).real
        cross_corr = cross_corr[n_points:]
        c5 = np.max(cross_corr * dx)
        return c5
    
    # Function to optimize h using a genetic algorithm
    def optimize_h(h_initial, budget_s):
        start_time = time.time()
        best_c5 = float('inf')
        best_h = h_initial.copy()
        
        # Define bounds for the variables (each h[i] in [0, 1])
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Use differential evolution for optimization
        result = optimize.differential_evolution(
            fun=evaluate_c5,
            bounds=bounds,
            x0=h_initial,
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-6,
            disp=False,
            callback=lambda x: None
        )
        
        # Check if the optimization finished within the time budget
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            print("Time budget exceeded, returning best found so far")
            return best_h, best_c5
        
        # Extract the best solution found
        best_h = result.x
        best_c5 = evaluate_c5(best_h)
        
        return best_h, best_c5
    
    # Run the optimization
    best_h, best_c5 = optimize_h(h_initial, budget_s)
    return best_h, best_c5, n_points
