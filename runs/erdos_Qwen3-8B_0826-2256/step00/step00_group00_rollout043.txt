import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45  # Start with the initial construction
    h_initial = initial_h_values.copy()
    h_initial = np.array(h_initial, dtype=np.float64)
    
    # Optimize with adaptive sampling and gradient-based methods
    def objective_function(h):
        h = np.clip(h, 0.0, 1.0)
        h = h.reshape(-1)
        dx = 2.0 / len(h)
        # Compute the overlap integral
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def constraint_sum(h):
        return np.sum(h) - len(h) / 2
    
    # Initial optimization with a gradient-based method
    h = h_initial.copy()
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='L-BFGS-B',
        bounds=[(0.0, 1.0) for _ in range(len(h))],
        tol=1e-6,
        options={'maxiter': 100}
    )
    
    best_h = result.x
    best_c5 = objective_function(best_h)
    
    # Try increasing n_points for better resolution
    for n_points in [60, 80, 100]:
        # Resample the best_h to n_points
        h_resampled = np.zeros(n_points)
        for i in range(len(best_h)):
            h_resampled[int(i * n_points / len(best_h))] = best_h[i]
        h_resampled = np.clip(h_resampled, 0.0, 1.0)
        # Add some noise to avoid local minima
        h_resampled += np.random.normal(0, 0.05, size=n_points)
        h_resampled = np.clip(h_resampled, 0.0, 1.0)
        # Ensure the integral constraint is satisfied
        h_resampled = h_resampled * (len(best_h) / 2) / np.sum(h_resampled)
        
        # Optimize further
        result = opt.minimize(
            fun=objective_function,
            x0=h_resampled,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            tol=1e-6,
            options={'maxiter': 100}
        )
        
        current_c5 = objective_function(result.x)
        if current_c5 < best_c5:
            best_h = result.x
            best_c5 = current_c5
            print(f"Improved C5: {best_c5} with {n_points} points")
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    h_values = best_h.tolist()
    c5_bound = best_c5
    return h_values, c5_bound, len(h_values)
