        h = np.array(h)

            h = np.resize(h, n_points)

        # Normalize so that the integral is 1
        h_norm = h / (np.sum(h) * 2.0 / n_points)
        # Compute the overlap integral
        overlap = correlate(h_norm, 1 - h_norm, mode="full")
        # Only consider the relevant part of the correlation
        # Since we are interested in shifts up to 2, we take indices up to 2 * n_points
        # But since the integral is over [0, 2], the maximum shift is 2
        # So the overlap for k=0 to 2 is in the first 2 * n_points + 1 elements

        # and the maximum value of the overlap integral for shift in [0, 2]
        # For our purposes, we take the maximum overlap for shift in [0, 2]
        # So the relevant part is the last n_points values of the correlation
        c5 = np.max(overlap[-n_points:])
        return c5
    
    def optimize_solution(h_initial, budget_s, max_iterations=100):
        best_h = h_initial
        best_c5 = evaluate_solution(best_h)
        start_time = time.time()

            perturbation = np.random.uniform(-0.2, 0.2, n_points)

            if current_c5 < best_c5:

                best_c5 = current_c5
            
            # Check time budget
            elapsed = time.time() - start_time
