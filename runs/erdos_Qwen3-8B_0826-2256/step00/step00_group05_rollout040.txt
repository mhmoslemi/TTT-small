```python
import numpy as np
import scipy.optimize as optimize
import sys
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is provided

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for faster evaluation

    # Define the objective function
    def objective(h):
        nonlocal best_c5
        h = h.reshape(-1)
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Compute overlap integral
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Define the constraints
    def constraint_sum(h):
        return np.sum(h) - (n_points / 2)

    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]

    # Initial guess from the provided initial_h_values
    initial_guess = initial_h_values[:n_points].copy()
    initial_guess = np.clip(initial_guess, 0, 1)

    # Optimization
    result = optimize.minimize(
        fun=objective,
        x0=initial_guess,
        method='L-BFGS-B',
        bounds=bounds,
        constraints={'type': 'eq', 'fun': constraint_sum},
        options={'maxiter': 100, 'ftol': 1e-6}
    )

    # Check if we can increase n_points within the budget
    if time.time() - start_time < budget_s - 5:
        n_points = 200
        bounds = [(0, 1) for _ in range(n_points)]
        initial_guess = np.random.rand(n_points)
        initial_guess = np.clip(initial_guess, 0, 1)
        initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)

        result = optimize.minimize(
            fun=objective,
            x0=initial_guess,
            method='L-BFGS-B',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_sum},
            options={'maxiter': 50, 'ftol': 1e-6}
        )

    # Final check
    h_values = best_h
    c5_bound = best_c5
    return (h_values, c5_bound, n_points)
```
