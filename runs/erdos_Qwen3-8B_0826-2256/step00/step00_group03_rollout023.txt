import numpy as np
import scipy.optimize as optimize
import multiprocessing
from functools import partial
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(70) * 2 - 1  # Placeholder for initial values

def run(seed=42, budget_s=100, n_points=200):
    start_time = time.time()
    
    # Set random seed
    np.random.seed(seed)
    
    # Initial solution (modified from given)
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1 if np.random.rand() < 0.5 else 0
    h = np.array(h, dtype=np.float64)
    total = np.sum(h)
    h = h * (n_points / 2) / total
    h = np.clip(h, 0, 1)
    
    # Function to compute the overlap integral for a shift
    def compute_overlap(h, shift):
        shifted = np.roll(h, shift)
        overlap = np.sum(h * (1 - shifted)) * (2.0 / n_points)
        return overlap
    
    # Function to compute all overlaps for a given h
    def compute_overlaps(h):
        overlaps = []
        for shift in range(1, n_points - 1):
            overlap = compute_overlap(h, shift)
            overlaps.append(overlap)
        return np.max(overlaps)
    
    # Objective function for optimization
    def objective_and_gradient(x):
        h = x.reshape((n_points,))
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        c5 = compute_overlaps(h)
        return c5, np.zeros_like(x)
    
    # Bounds for optimization
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization
    result = optimize.minimize(
        objective_and_gradient,
        x0,
        bounds=bounds,
        method='L-BFGS-B',
        options={'disp': False, 'ftol': 1e-9, 'maxiter': 100}
    )
    
    # Extract best solution
    best_h = result.x
    best_h = np.clip(best_h, 0, 1)
    total = np.sum(best_h)
    best_h = best_h * (n_points / 2) / total
    best_h = np.clip(best_h, 0, 1)
    c5 = compute_overlaps(best_h)
    
    # Ensure we stay within budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h, c5, n_points)
    
    # Multi-threaded fine-tuning
    def optimize_h(h_initial, budget_s):
        x0 = h_initial.copy()
        result = optimize.minimize(
            objective_and_gradient,
            x0,
            bounds=bounds,
            method='L-BFGS-B',
            options={'disp': False, 'ftol': 1e-9, 'maxiter': 100}
        )
        best_h = result.x
        best_h = np.clip(best_h, 0, 1)
        total = np.sum(best_h)
        best_h = best_h * (n_points / 2) / total
        best_h = np.clip(best_h, 0, 1)
        c5 = compute_overlaps(best_h)
        return best_h, c5
    
    # Start with initial solution
    best_h, best_c5 = optimize_h(best_h, budget_s - elapsed)
    
    # Start a pool of workers
    pool = multiprocessing.Pool(processes=4)
    worker_func = partial(optimize_h, budget_s=budget_s - elapsed)
    results = pool.map(worker_func, [best_h for _ in range(4)])
    pool.close()
    pool.join()
    
    # Extract best solution from all results
    best_h = results[0][0]
    best_c5 = results[0][1]
    for h, c5 in results:
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return (best_h, best_c5, n_points)
