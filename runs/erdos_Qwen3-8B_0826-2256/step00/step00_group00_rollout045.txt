import numpy as np
import scipy.optimize as optimize
import time

from scipy.optimize import Bounds
from scipy.optimize import minimize

# Pre-imported functions
# initial_h_values is a global variable with an initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 45
    h_initial = initial_h_values.copy()
    
    # Start with the initial solution
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    
    # Time tracking
    start_time = time.time()
    
    # Try different numbers of points (up to a maximum of 200)
    for n_points in [45, 60, 75, 90, 105, 120, 135, 150, 165, 180, 200]:
        # Ensure we don't exceed budget
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a new h with n_points samples
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        
        # Randomly distribute values to maintain the integral constraint
        # Start with a random initial guess
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Define objective function
        def objective(x):
            # Reshape x to h
            h_current = x.reshape(n_points)
            # Compute overlap integral
            # Ensure h_current is in [0, 1]
            h_current = np.clip(h_current, 0, 1)
            # Compute the correlation
            corr = np.correlate(h_current, 1 - h_current, mode="full")
            # Compute dx for the integral
            integral = corr * dx
            # Take the maximum value
            c5 = np.max(integral)
            return c5
        
        # Define bounds
        bounds = Bounds(0, 1)
        
        # Perform optimization
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, options={'ftol': 1e-7, 'maxiter': 100})
        
        # Get the best solution
        best_h_current = result.x
        c5_current = result.fun
        
        # Update best solution
        if c5_current < best_c5:
            best_h = best_h_current
            best_c5 = c5_current
            best_n_points = n_points
        
        print(f"n_points={n_points}, c5={best_c5:.6f}")
    
    # Final check
    if best_c5 < 0.380876:
        print("Improved bound found!")
    
    return best_h, best_c5, best_n_points
