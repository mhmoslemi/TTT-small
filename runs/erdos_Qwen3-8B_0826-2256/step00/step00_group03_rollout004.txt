    start_time = time.time()

    # Function to compute C5
    def compute_c5(h):
        h = h.copy()
        h = np.pad(h, (0, len(h)), 'constant')
        result = np.correlate(h, 1 - h, mode='full') * (2.0 / len(h))
        return np.max(result)
    
    # Function to evaluate the objective
    def objective(x):
        h = x.copy()
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * 2.0 / len(h))
        return compute_c5(h)
    
    # Function to optimize using the Nelder-Mead method
    def optimize_solution(h):
        result = optimize.minimize(objective, h, method='Nelder-Mead', 
                                  options={'xatol': 1e-6, 'fatol': 1e-6, 'maxiter': 100})
        return result.x
    
    # Run the optimization
    h_values = optimize_solution(h_values)
    
    # Ensure the values are within [0,1] and normalize
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / (np.sum(h_values) * 2.0 / len(h_values))
    
    # Calculate the final C5 value
    c5_bound = compute_c5(h_values)
    
    # Check if we have time to increase points and refine the solution
    elapsed = time.time() - start_time
