# Utility function to evaluate C5 for given h_values
def evaluate_c5(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros_like(h)
    max_overlap = 0.0
    for k in range(1, n_points):
        shifted_h = np.roll(h, k)
        overlap = np.convolve(h, 1 - shifted_h, mode="full")
        overlap = overlap * dx
        max_overlap = max(max_overlap, np.max(overlap))
    return max_overlap

# Strategy: Use a gradient-based optimization approach with simulated annealing
# Starting from an initial guess and iteratively improving the h_values to minimize C5

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Initial guess with n_points = 200 (a reasonable choice to balance quality and speed)
    n_points = 200
    h_initial = np.random.rand(n_points)
    h_initial = h_initial * 2 - 1  # Random initial guess within [0, 1]
    h_initial = np.clip(h_initial, 0, 1)
    sum_h_initial = np.sum(h_initial)
    # Scale to ensure the integral over [0, 2] is 1
    h_initial = h_initial * (n_points / 2) / sum_h_initial
    h_initial = np.clip(h_initial, 0, 1)
    
    best_h = h_initial.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Optimization parameters
    num_iterations = 100
    temp = 1.0
    temp_decay = 0.995
    for iteration in range(num_iterations):
        if time.time() - start_time > budget_s:
            break
        # Generate a neighbor solution by perturbing h_initial
        h_neighbor = h_initial.copy()
        idx = np.random.randint(0, n_points)
        delta = np.random.uniform(-0.1, 0.1)
        h_neighbor[idx] += delta
        h_neighbor = np.clip(h_neighbor, 0, 1)
        sum_h_neighbor = np.sum(h_neighbor)
        h_neighbor = h_neighbor * (n_points / 2) / sum_h_neighbor
        h_neighbor = np.clip(h_neighbor, 0, 1)
        
        # Evaluate C5 for both solutions
        c5_initial = evaluate_c5(h_initial)
        c5_neighbor = evaluate_c5(h_neighbor)
        
        # Accept the neighbor with probability based on simulated annealing
        if c5_neighbor < c5_initial or np.random.rand() < np.exp((c5_initial - c5_neighbor) / temp):
            h_initial = h_neighbor
            if c5_neighbor < best_c5:
                best_h = h_initial.copy()
                best_c5 = c5_neighbor
        
        temp *= temp_decay
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return best_h, final_c5, n_points
