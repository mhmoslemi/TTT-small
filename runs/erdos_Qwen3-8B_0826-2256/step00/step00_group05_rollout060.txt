import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder, actual path may vary

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 200  # Start with 200 points, adjust if needed
    
    # Function to evaluate the C5 value for a given h
    def evaluate_c5(h):
        nonlocal best_c5
        h = np.array(h)
        if len(h) != n_points:
            h = np.resize(h, n_points)
        corr = np.correlate(h, 1 - h, mode="full")
        c5_val = np.max(corr) * (2.0 / n_points)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        return c5_val
    
    # Initial guess
    h_initial = initial_h_values[:n_points] if len(initial_h_values) >= n_points else initial_h_values
    h_initial = np.resize(h_initial, n_points)
    h_initial = np.clip(h_initial, 0, 1)
    
    # Optimization with simulated annealing
    result = optimize.minimize(
        evaluate_c5,
        h_initial,
        method='SLSQP',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=[
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ],
        options={'ftol': 1e-8, 'maxiter': 100}
    )
    
    # Check if we need to refine with more points or other methods
    if time.time() - start_time < budget_s - 10:
        # Try increasing the number of points
        n_points = 400
        h_initial = np.resize(result.x, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        result = optimize.minimize(
            evaluate_c5,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=[
                {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
            ],
            options={'ftol': 1e-8, 'maxiter': 100}
        )
    
    # Final check
    if time.time() - start_time < budget_s - 5:
        # Try using CVXOPT for more precise optimization
        h = cp.Variable(n_points)
        objective = cp.Minimize(cp.Max(cp.multiply(h, 1 - h)))
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.CVXOPT)
        if prob.status == 'optimal':
            best_h = h.value
            best_c5 = evaluate_c5(best_h)
    
    # Ensure best_h is within bounds
    best_h = np.clip(best_h, 0, 1)
    return (best_h, best_c5, n_points)
