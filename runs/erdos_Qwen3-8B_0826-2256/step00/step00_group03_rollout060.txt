import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points
    n_points = 100  # Start with 100 points for optimization speed
    dx = 2.0 / n_points
    max_iterations = 100
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Initial solution from provided initial_h_values
    initial_h = np.array(initial_h_values[:n_points])
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h / np.sum(initial_h) * n_points / 2
    
    # Optimization loop
    for _ in range(max_iterations):
        # Evaluate the current h
        h = initial_h.copy()
        # Compute the correlation
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Adaptive update step
        if np.random.rand() < 0.3:
            # Random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
        else:
            # Gradient descent-like update
            # Compute gradients for all shifts
            gradients = np.zeros(n_points)
            for shift in range(n_points + 1):
                # Compute derivative of the overlap with respect to h
                # This is a simplified approximation
                # For real optimization, consider using a more sophisticated approach
                corr_shift = signal.correlate(h, 1 - h, mode="full")[shift]
                grad = -1 * (1 - h) * (np.roll(corr_shift, shift) - corr_shift) * dx
                gradients += grad
            
            # Update h using gradient descent
            h += -0.01 * gradients
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    # Final evaluation
    final_corr = signal.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
