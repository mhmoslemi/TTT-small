```python
import numpy as np
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode='full')) * (2 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 samples for performance
    max_iter = 100  # Limit iterations to stay within budget

    # Initial guess based on provided or random
    if initial_h_values is not None and len(initial_h_values) == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Normalize to [0, 1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Ensure integral is 1

    # Convert to CVXPY variables
    h_var = cp.Variable(n_points, nonneg=True)
    constraints = [cp.sum(h_var) == n_points / 2]

    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        # Objective: minimize the max overlap
        overlap = cp.Maximize(cp.max([cp.sum(h_var * (1 - h_var[k:] + h_var[:-k])) for k in range(1, n_points)]))
        prob = cp.Problem(overlap, constraints)
        try:
            prob.solve(solver='ECOS', eps=1e-6, verbose=False)
            if prob.status == 'OPTIMAL':
                h = h_var.value
                h = np.clip(h, 0, 1)
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            # If solver fails, use random perturbation
            h += 0.1 * np.random.rand(n_points) - 0.05
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

    # Try increasing n_points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 0.5 * budget_s:
        n_points = 150
        h = np.random.rand(n_points)
        h = h * 2 - 1  # Normalize to [0, 1]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_var = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h_var) == n_points / 2]
        for _ in range(50):
            if time.time() - start_time > budget_s:
                break
            try:
                overlap = cp.Maximize(cp.max([cp.sum(h_var * (1 - h_var[k:] + h_var[:-k])) for k in range(1, n_points)]))
                prob = cp.Problem(overlap, constraints)
                prob.solve(solver='ECOS', eps=1e-6, verbose=False)
                if prob.status == 'OPTIMAL':
                    h = h_var.value
                    h = np.clip(h, 0, 1)
                    current_c5 = evaluate_erdos_solution(h)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h.copy()
            except:
                h += 0.1 * np.random.rand(n_points) - 0.05
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)

    # Ensure we return the best result found
    if best_h is not None:
        return (best_h, best_c5, n_points)
    else:
        return (h, evaluate_erdos_solution(h), n_points)
```
