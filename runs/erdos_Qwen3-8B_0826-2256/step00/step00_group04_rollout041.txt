import numpy as np
import scipy.optimize as optimize
from scipy.optimize import Bounds

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    import time
    import numpy as np
    import scipy.optimize as optimize
    from scipy.optimize import Bounds
    from scipy.signal import correlate

    # Start timer
    start_time = time.time()

    # Set up initial parameters
    n_points = 100  # Start with 100 points for faster optimization
    max_iterations = 100
    initial_h = initial_h_values.copy()  # Use the initial construction if available

    # Ensure initial_h is properly scaled
    initial_h = initial_h[:n_points]
    initial_h /= np.sum(initial_h) * 2.0  # Rescale to satisfy ∫h(x)dx = 1

    # Function to compute the C5 value
    def compute_c5(h):
        h = h.reshape(-1)
        h = np.concatenate([h, h])  # Duplicate to handle full convolution
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * (2.0 / len(h))  # dx = 2.0 / n_points
        return c5

    # Optimization function
    def objective_function(h):
        return compute_c5(h)

    # Optimization constraints
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Set up bounds and constraints
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    constraints = [{'type': 'eq', 'fun': constraint_sum, 'jac': lambda h: np.ones(n_points)}]

    # Initial guess
    initial_guess = initial_h

    # Optimization
    result = optimize.minimize(
        objective_function,
        initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': max_iterations, 'disp': False}
    )

    # Extract best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)

    # Try with larger n_points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        initial_h = initial_h_values.copy()
        initial_h = initial_h[:n_points]
        initial_h /= np.sum(initial_h) * 2.0

        def compute_c5_large(h):
            h = h.reshape(-1)
            h = np.concatenate([h, h])
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / len(h))
            return c5

        def objective_function_large(h):
            return compute_c5_large(h)

        bounds = Bounds(np.zeros(n_points), np.ones(n_points))
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) - n_points / 2, 'jac': lambda h: np.ones(n_points)}]

        initial_guess = initial_h

        result_large = optimize.minimize(
            objective_function_large,
            initial_guess,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': max_iterations, 'disp': False}
        )

        best_h = result_large.x
        best_c5 = compute_c5_large(best_h)

    # Ensure h is within [0, 1] and scaled correctly
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)  # Rescale to ensure ∫h(x)dx = 1

    # Final check
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
