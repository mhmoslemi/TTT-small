```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial solution
    h_initial = initial_h_values()
    n_points = len(h_initial)
    dx = 2.0 / n_points
    h_initial = np.array(h_initial)
    
    # Function to evaluate C5
    def evaluate_C5(h):
        h = np.array(h)
        h = np.concatenate([h, h[:1]])  # To handle circular shifts
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Function to optimize h with constraints
    def optimize_h(h, n_points, budget_s, max_iter=100, tol=1e-3):
        best_h = h.copy()
        best_c5 = evaluate_C5(best_h)
        time_used = 0
        for _ in range(max_iter):
            # Convert to CVXPY variables
            h_var = cp.Variable(n_points)
            constraints = [
                cp.sum(h_var) == n_points / 2,
                0 <= h_var,
                h_var <= 1
            ]
            # Objective: minimize the maximum overlap
            # We use a convex relaxation by penalizing high overlaps
            # This is an approximation and may require refinement
            # For simplicity, we optimize for the maximum overlap directly
            # Using a heuristic approach
            h_guess = best_h.copy()
            h_guess += np.random.uniform(-0.1, 0.1, size=n_points)
            h_guess = np.clip(h_guess, 0, 1)
            
            # Use L-BFGS-B optimization with constraints
            def objective_func(h_vec):
                h = h_vec.copy()
                h = np.clip(h, 0, 1)
                return evaluate_C5(h)
            
            result = optimize.minimize(
                objective_func,
                h_guess,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'maxiter': 50, 'ftol': 1e-6}
            )
            
            if result.success:
                current_h = result.x
                current_c5 = evaluate_C5(current_h)
                if current_c5 < best_c5:
                    best_h = current_h
                    best_c5 = current_c5
                time_used = time.time() - start_time
                if time_used >= budget_s - 1:
                    break
            else:
                break
        return best_h, best_c5
    
    # Optimize the solution
    best_h, best_c5 = optimize_h(h_initial, n_points, budget_s)
    best_c5 = evaluate_C5(best_h)
    
    # Final refinement using CVXPY
    h_var = cp.Variable(n_points)
    constraints = [
        cp.sum(h_var) == n_points / 2,
        0 <= h_var,
        h_var <= 1
    ]
    # Objective: minimize the maximum overlap
    # Since this is non-convex, we use a convex relaxation
    # We penalize the maximum overlap by minimizing the maximum value of the correlation
    # This is a relaxation and may not yield the exact minimum
    # We optimize using a heuristic with L-BFGS-B
    h_guess = best_h.copy()
    h_guess += np.random.uniform(-0.1, 0.1, size=n_points)
    h_guess = np.clip(h_guess, 0, 1)
    
    def objective_func(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        return evaluate_C5(h)
    
    result = optimize.minimize(
        objective_func,
        h_guess,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        options={'maxiter': 50, 'ftol': 1e-6}
    )
    
    if result.success:
        best_h = result.x
        best_c5 = evaluate_C5(best_h)
    else:
        best_h = best_h
    
    best_c5 = evaluate_C5(best_h)
    
    return best_h.tolist(), best_c5, n_points
```
