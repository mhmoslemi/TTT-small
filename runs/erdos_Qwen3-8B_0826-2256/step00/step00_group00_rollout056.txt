import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Start with 100 points for faster computation

    h_values = h_values[:n_points]  # Truncate to n_points
    h_values = h_values / (np.sum(h_values) * (2.0 / n_points)) * (n_points / 2.0)  # Ensure sum(h) = n_points / 2
    
    def compute_c5(h):
        h = np.array(h)
        # Normalize h to the interval [0, 1]
        h_norm = h / np.max(h)
        # Compute (1 - h)
        one_minus_h = 1.0 - h_norm
        # Compute cross-correlation
        corr = np.correlate(h_norm, one_minus_h, mode="full")
        # We need to evaluate max over k of integral h(x)(1 - h(x+k)) dx
        # The cross-correlation result corresponds to the integral over the shifted function
        # The central part of the correlation corresponds to k = 0, 1, ..., n_points - 1
        # But we need to consider all k in the range [-n_points + 1, n_points - 1]
        # The full correlation has length 2*n_points - 1
        # The overlap integral for each k is corr[k + n_points - 1] * dx
        # The dx is 2.0 / n_points
        dx = 2.0 / n_points
        # We need to evaluate the maximum of corr[k + n_points - 1] * dx for all k
        # The indices of corr correspond to displacements from -n_points + 1 to n_points - 1
        # The correct overlap integral values are the elements starting from index n_points - 1
        # So we take the last n_points values in corr (corresponding to k = 0 to n_points - 1)
        # For each k = 0 to n_points - 1, the integral is corr[n_points - 1 + k] * dx
        # We must also consider the values for negative k (k < 0). These correspond to the first
        # n_points - 1 values in corr (for k = -n_points + 1 to -1)
        # We consider the overlap for k from -n_points + 1 to n_points - 1
        # So we take the entire corr array and multiply by dx
        c5 = np.max(corr * dx)
        return c5

    def objective(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2.0

    # Set up constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Optimize using scipy.optimize
    result = minimize(objective, h_values, method='SLSQP', bounds=bounds, constraints=cons, jac=None, tol=1e-6)
    best_h = result.x

    # Evaluate the best solution
    c5_bound = compute_c5(best_h)

    # Check if we still have time to refine further
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:
        # Try a higher resolution with the best solution found so far
        n_points = 200
        best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        best_h = best_h / np.max(best_h)
        best_h = best_h * (n_points / 2.0) / np.sum(best_h) * (np.sum(best_h) * (2.0 / n_points))
        # Ensure sum(h) = n_points / 2
        best_h = best_h / np.sum(best_h) * (n_points / 2.0)
        # Optimize again with higher resolution
        result = minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=cons, jac=None, tol=1e-6)
        best_h = result.x
        c5_bound = compute_c5(best_h)

    return (best_h, c5_bound, n_points)
