```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def initial_h_values(n_points):
    # Construct an initial guess with equal intervals and fractional occupancy
    h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with 100 points, can be increased if time permits
    dx = 2.0 / n_points
    h = initial_h_values(n_points)
    
    # Objective function to minimize C5
    def objective(h):
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Optimization setup
    h_vec = cp.Variable(n_points)
    objective_func = cp.Max(cp.multiply(cp.sum(cp.multiply(h_vec, 1 - h_vec)), dx))
    
    # Constraints
    constraints = [
        cp.sum(h_vec) == n_points / 2,  # Integral constraint
        0 <= h_vec,
        h_vec <= 1
    ]
    
    # Optimization problem
    prob = cp.Problem(cp.Minimize(objective_func), constraints)
    
    # Solve the problem
    try:
        result = prob.solve(solver=cp.ECOS, eps=1e-6, time_limit=budget_s - (time.time() - start_time))
        if result is not None:
            h_values = h_vec.value
        else:
            h_values = h
    except:
        h_values = h
    
    # Evaluate the solution
    c5_bound = evaluate_erdos_solution(h_values)
    
    # Return the best solution found
    return h_values, c5_bound, n_points
```
