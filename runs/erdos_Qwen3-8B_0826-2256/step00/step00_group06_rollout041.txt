def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100  # Start with 100 points for faster optimization
    dx = 2.0 / n_points
    h = initial_h_values[:n_points].copy()
    h = h / (np.sum(h) * dx) * (n_points / 2)
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Define the optimization problem
    def objective_function(h_vals):
        c5 = evaluate_erdos_solution(h_vals)
        return c5
    
    # Optimization loop
    for _ in range(5):  # Run a few iterations
        # Perform a local optimization
        h_vals = best_h.copy()
        h_vals = np.clip(h_vals, 0, 1)
        result = optimize.minimize(objective_function, h_vals, method='L-BFGS-B', bounds=[(0, 1)]*n_points, 
                                   options={'maxiter': 100, 'ftol': 1e-6})
        h_vals = result.x
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final check with more points
    n_points = 200
    dx = 2.0 / n_points
    h = np.linspace(0, 2, n_points)
    h_vals = np.zeros(n_points)
    # Interpolate from best_h to match n_points
    best_h_interp = np.interp(h, np.linspace(0, 2, len(best_h)), best_h)
    # Ensure the integral constraint is satisfied
    total = np.sum(best_h_interp) * dx
    if abs(total - 1) > 1e-6:
        # Adjust to meet the constraint
        scale = 1 / total
        best_h_interp *= scale
    # Evaluate
    c5 = evaluate_erdos_solution(best_h_interp)
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = best_h_interp
    
    return (best_h, best_c5, n_points)
