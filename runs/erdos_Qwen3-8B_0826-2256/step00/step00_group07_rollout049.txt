```python
import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # 80 samples
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    target_sum = n_points / 2  # To satisfy ∫h(x)dx = 1

    # Define the objective function to minimize
    def objective(h):
        return max(np.correlate(h, 1 - h, mode="full")) * dx

    # Define constraints
    def constraints(h):
        return np.sum(h) - target_sum

    # Start with the initial solution
    current_h = initial_h_values.copy()
    current_c5 = objective(current_h)
    best_h = current_h
    best_c5 = current_c5

    # Run a local optimization with brute force search
    for _ in range(100):
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = np.clip(current_h + perturbation, 0, 1)
        perturbed_h = np.round(perturbed_h, 5)  # Reduce floating point errors
        perturbed_sum = np.sum(perturbed_h)
        # Adjust to maintain the integral constraint
        if perturbed_sum != target_sum:
            scale_factor = target_sum / perturbed_sum
            perturbed_h = np.round(perturbed_h * scale_factor, 5)
        perturbed_c5 = objective(perturbed_h)
        if perturbed_c5 < best_c5:
            best_c5 = perturbed_c5
            best_h = perturbed_h.copy()
        current_h = perturbed_h.copy()
        current_c5 = perturbed_c5
        if time.time() - start_time > budget_s - 1:
            break

    # Try a few random initializations to avoid local optima
    for _ in range(5):
        # Randomly generate a valid function h
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        scale_factor = target_sum / h_sum
        h = np.round(h * scale_factor, 5)
        h_c5 = objective(h)
        if h_c5 < best_c5:
            best_c5 = h_c5
            best_h = h.copy()

    return best_h, best_c5, n_points
```
