```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
from scipy.linalg import toeplitz
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.correlate(h, 1 - h, mode="full") * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assuming initial setup is loaded

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    
    # Define the number of points (less than 1000 for faster computation)
    n_points = 200  # Adjust as needed to balance quality and speed
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Start with initial values or a simple uniform distribution
    if initial_h_values.shape[0] == n_points:
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure the integral equals 1
    
    # Ensure the integral constraint is satisfied
    sum_h = np.sum(h)
    scale = (n_points / 2) / sum_h if sum_h != 0 else 1.0
    h = h * scale
    
    # Optimization: minimize the maximum overlap integral
    def objective(x):
        h_vals = x
        # Compute the overlap
        corr = signal.correlate(h_vals, 1 - h_vals, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Constraints: sum(h) should be n_points / 2
    def constraint(x):
        return np.sum(x) - n_points / 2
    
    # Bounds: h[i] in [0, 1]
    bounds = Bounds(0, 1)
    
    # Set up the optimization problem
    x = cp.Variable(n_points)
    obj = cp.Minimize(objective(x))
    constraints = [constraint(x) == 0]
    
    # Solve the optimization problem
    prob = cp.Problem(obj, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-4, time_limit=budget_s - (time.time() - start_time))
    except:
        # Fallback: use gradient descent if SCS fails
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'maxiter': 1000, 'ftol': 1e-6},
            tol=1e-6
        )
        h = result.x
    
    # Ensure the integral constraint is still satisfied
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    
    # Evaluate the current solution
    c5_bound = evaluate_erdos_solution(h)
    
    # Ensure the solution is within the time budget
    if time.time() - start_time > budget_s - 1:
        return h, c5_bound, n_points
    
    return h, c5_bound, n_points
```
