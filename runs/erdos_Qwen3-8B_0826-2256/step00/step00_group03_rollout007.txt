```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
import copy

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros(2 * n_points - 1)
    for k in range(1, n_points):
        shifted[k - 1] = np.correlate(h, 1 - h, mode='full')[k]
    integral = shifted * dx
    return np.max(integral)

def initial_h_values():
    # Placeholder for initial construction
    return np.random.uniform(0, 1, 100)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial parameter setup
    n_points = 100  # Start with 100 samples
    max_iterations = 100
    improvement_threshold = 1e-5
    best_c5 = np.inf
    best_h = None
    
    # Initial guess
    h = initial_h_values()
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2)  # Ensure the integral is 1
    
    # Evaluate initial solution
    current_c5 = evaluate_erdos_solution(h)
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop
    for iteration in range(max_iterations):
        # Check remaining time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Define the function to minimize
        def objective(x):
            h_new = x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2)
            c5 = evaluate_erdos_solution(h_new)
            return c5

        # Use L-BFGS-B for optimization
        try:
            result = opt.minimize(
                objective, 
                x0=best_h, 
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)], 
                jac=None,
                options={'maxiter': 50, 'ftol': 1e-6}
            )
        except:
            result = opt.minimize(
                objective, 
                x0=best_h, 
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)], 
                jac=None,
                options={'maxiter': 50, 'ftol': 1e-6}
            )

        # Update best solution
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        # Adjust n_points if needed
        if iteration % 5 == 0:
            # Try increasing n_points
            n_points = min(n_points + 10, 500)
            h = np.random.uniform(0, 1, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Final evaluation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (len(final_h) / 2)
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), final_c5, len(final_h))
```
