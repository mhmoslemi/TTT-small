        def objective(h_flat):
            h = h_flat.reshape(n_points)

        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        # Gradient descent with line search
        h_flat = h.copy()
        learning_rate = 0.01
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break

            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr) * (2.0 / n_points)

            gradient = np.zeros(n_points)

            h_flat -= learning_rate * gradient

            h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
