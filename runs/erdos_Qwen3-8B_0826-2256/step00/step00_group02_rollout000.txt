```python
import numpy as np
import scipy.optimize as opt
import time
import copy
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial guess from pre-loaded data
    initial_h = initial_h_values
    n_initial = len(initial_h)
    dx_initial = 2.0 / n_initial

    # Approximate the current C5 value from the initial solution
    current_c5 = evaluate_erdos_solution(initial_h)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = initial_h
        best_n_points = n_initial

    # Early exit if already below target
    if best_c5 < 0.380876:
        return best_h, best_c5, best_n_points

    # Try improving with different n_points
    for n_points in [n_initial - 5, n_initial, n_initial + 5]:
        if n_points < 4:
            continue
        dx = 2.0 / n_points
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)  # Ensure h is in [0,1]
        if np.sum(h) * dx != 1:
            h = h * (1 / (np.sum(h) * dx))  # Normalize

        # Create a function to evaluate C5 for a given h
        def obj_func(h_candidate):
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            h_candidate = np.clip(h_candidate, 0, 1)
            corr = np.correlate(h_candidate, 1 - h_candidate, mode="full")
            return np.max(corr) * dx

        # Try a few iterations of optimization
        for _ in range(5):
            # Use a simple gradient descent with random perturbations
            h_perturb = h + np.random.rand(n_points) * 0.1 - 0.05
            h_perturb = np.clip(h_perturb, 0, 1)
            h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
            h_perturb = np.clip(h_perturb, 0, 1)
            if np.sum(h_perturb) * dx != 1:
                h_perturb = h_perturb * (1 / (np.sum(h_perturb) * dx))
            current_c5 = obj_func(h_perturb)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_perturb.copy()
                best_n_points = n_points

            # Early exit if budget is exceeded
            if time.time() - start_time > budget_s:
                break

        # After optimization, check if we can improve further
        if best_c5 < 0.380876:
            break

    return best_h, best_c5, best_n_points
```
