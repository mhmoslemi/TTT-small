```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Approximate initial solution from research

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 50  # Start with 50 points, can increase if needed
    dx = 2.0 / n_points
    h_initial = initial_h_values[:n_points]  # Take the first n_points from initial solution
    h_initial = np.clip(h_initial, 0, 1)  # Ensure values are in [0,1]
    h_initial = h_initial * (n_points / 2)  # Adjust to meet the integral constraint
    
    # Objective function to minimize C₅
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Ensure integral constraint
        c5_val = evaluate_erdos_solution(h)
        return c5_val
    
    # Constraints: sum(h) == n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Optimization setup
    x0 = h_initial
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Use Sequential Least Squares Programming (SLSQP) for optimization
    result = optimize.minimize(
        objective, 
        x0, 
        method='SLSQP', 
        bounds=bounds, 
        constraints=cons,
        options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
    )
    
    # Check if time is running out
    elapsed = time.time() - start_time
    if elapsed > budget_s - 5:
        # If time is tight, return the best solution so far
        best_h = result.x
    else:
        # If more time is available, try increasing n_points and optimizing again
        n_points_new = min(100, n_points + 20)  # Increase n_points if time allows
        dx_new = 2.0 / n_points_new
        h_initial_new = np.linspace(0, 1, n_points_new)
        h_initial_new = h_initial_new * (n_points_new / 2)
        
        # Re-optimize with new n_points
        def objective_new(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points_new / 2)
            c5_val = evaluate_erdos_solution(h)
            return c5_val
        
        def constraint_sum_new(h):
            return np.sum(h) - n_points_new / 2
        
        cons_new = {'type': 'eq', 'fun': constraint_sum_new}
        
        result_new = optimize.minimize(
            objective_new, 
            h_initial_new, 
            method='SLSQP', 
            bounds=[(0, 1) for _ in range(n_points_new)],
            constraints=cons_new,
            options={'maxiter': 100, 'ftol': 1e-8, 'disp': False}
        )
        
        best_h = result_new.x
        n_points = n_points_new
    
    # Final evaluation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2)
    c5_bound = evaluate_erdos_solution(best_h)
    
    return (best_h, c5_bound, n_points)
```
