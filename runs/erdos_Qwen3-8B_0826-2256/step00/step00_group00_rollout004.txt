```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
import random

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points = 100
    h = initial_h_values.copy()[:n_points]
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to meet integral constraint
    
    best_c5 = np.inf
    best_h = h
    
    # Try a few different n_points sizes
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Define the function to minimize
        def objective_func(h_values):
            h_values = np.clip(h_values, 0, 1)
            dx = 2.0 / n_points
            h = h_values.reshape(n_points)
            # Compute the correlation
            corr = np.correlate(h, 1 - h, mode="full")
            # Weighted by dx
            weighted_corr = corr * dx
            # Take the maximum value
            return np.max(weighted_corr)
        
        # Optimization: use local optimization with initial guess
        result = optimize.minimize(
            objective_func,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)]
        )
        
        # Check if we found a better solution
        if result.success:
            current_h = result.x
            current_c5 = objective_func(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h
        
        # Try a few random perturbations
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            perturbed_h = np.clip(h + perturbation, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            current_c5 = objective_func(perturbed_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h
    
    # Final check
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = objective_func(final_h)
    
    return final_h, final_c5, n_points
```
