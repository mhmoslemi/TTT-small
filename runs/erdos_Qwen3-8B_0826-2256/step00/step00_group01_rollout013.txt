import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Set the number of points for the function representation
    n_points = 100  # Start with 100 points, can be increased if needed
    dx = 2.0 / n_points
    
    # Initial guess
    h = initial_h_values[:]
    h = h[:n_points]  # Ensure it fits the n_points
    h = np.array(h)
    
    # Constraint: sum(h) * dx = 1
    sum_h = np.sum(h)
    if abs(sum_h * dx - 1) > 1e-6:
        h = np.ones(n_points) * (1 / n_points)
    
    # Function to evaluate the overlap integral
    def evaluate_C5(h_vals):
        h = h_vals.copy()
        # Normalize h to satisfy the constraint
        sum_h = np.sum(h)
        if abs(sum_h * dx - 1) > 1e-6:
            h = h / sum_h * 1 / dx
        h = np.clip(h, 0, 1)
        
        # Compute the overlap integral
        h_pad = np.concatenate([h, np.zeros(n_points)])
        one_minus_h = 1 - h
        one_minus_h_pad = np.concatenate([one_minus_h, np.zeros(n_points)])
        
        # Convolution to compute overlap
        conv = np.correlate(h_pad, one_minus_h_pad, mode="full")
        # Convolution is symmetric, take the middle part
        max_overlap = np.max(conv[n_points - 1 : 2 * n_points - 1])
        return max_overlap * dx
    
    # Define the optimization problem
    # We minimize the maximum overlap integral
    # This is a non-convex problem, so we use a heuristic approach
    # Here we use a genetic algorithm for optimization
    
    # Genetic algorithm parameters
    pop_size = 50
    num_generations = 50
    mutation_rate = 0.1
    elite_size = 5
    
    # Genetic algorithm
    def genetic_algorithm():
        population = np.random.rand(pop_size, n_points)
        population = np.clip(population, 0, 1)
        
        for _ in range(num_generations):
            # Evaluate fitness
            fitness = np.array([evaluate_C5(ind) for ind in population])
            
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = population[elite_indices]
            
            # Crossover and mutation
            new_population = elites.copy()
            for _ in range(pop_size - elite_size):
                # Select parents
                parent1 = population[np.random.choice(pop_size)]
                parent2 = population[np.random.choice(pop_size)]
                
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.uniform(0, 1)
                
                new_population = np.vstack([new_population, child])
            
            population = new_population[:pop_size]
            population = np.clip(population, 0, 1)
        
        # Get the best individual
        best_fitness = np.min([evaluate_C5(ind) for ind in population])
        best_h = population[np.argmin([evaluate_C5(ind) for ind in population])]
        return best_h, best_fitness
    
    # Start timer
    start_time = time.time()
    
    # Run the genetic algorithm
    best_h, best_c5 = genetic_algorithm()
    
    # Check the time budget
    elapsed = time.time() - start_time
    if elapsed >= budget_s:
        return (best_h, best_c5, n_points)
    
    # If time remains, perform local optimization
    # Use a local optimizer to refine the solution
    def objective(x):
        x = np.clip(x, 0, 1)
        sum_h = np.sum(x)
        if abs(sum_h * dx - 1) > 1e-6:
            x = x / sum_h * 1 / dx
        return evaluate_C5(x)
    
    x0 = best_h.copy()
    result = optimize.minimize(objective, x0, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
    best_h = result.x
    best_c5 = result.fun
    
    # Return the best solution
    return (best_h, best_c5, n_points)
