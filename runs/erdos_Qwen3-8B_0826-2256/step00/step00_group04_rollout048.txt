```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points, can increase if time allows
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    
    # Initialize with a simple step function
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    
    # Adjust to meet integral constraint
    sum_h = np.sum(h)
    target_sum = n_points / 2
    adjustment = (target_sum - sum_h) / n_points
    h += adjustment
    
    def evaluate_c5(h_values):
        h = h_values.copy()
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        conv = np.correlate(h, 1 - h, mode="full")
        conv = conv * dx
        return np.max(conv)
    
    def objective(x):
        h = x.copy()
        # Ensure values are within [0, 1]
        h = np.clip(h, 0, 1)
        # Compute the overlap integral
        conv = np.correlate(h, 1 - h, mode="full")
        conv = conv * dx
        return np.max(conv)
    
    # Define the optimization problem using CVXPY
    h_var = cp.Variable(n_points)
    obj = cp.Minimize(cp.Max(cp.sum(cp.multiply(h_var, 1 - h_var[1:n_points+1])) * dx))
    constraints = [cp.sum(h_var) == n_points / 2, 0 <= h_var, h_var <= 1]
    
    # Solve the problem
    start_time = time.time()
    prob = cp.Problem(obj, constraints)
    try:
        prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - (time.time() - start_time))
    except:
        pass  # If solving fails, fall back to optimization
    
    if prob.status == 'optimal':
        h = h_var.value
    else:
        # Use gradient descent if CVXPY fails
        def loss(x):
            return objective(x)
        
        result = optimize.minimize(loss, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)], tol=1e-6, options={'maxiter': 100})
        h = result.x
    
    c5 = evaluate_c5(h)
    return h, c5, n_points
```
