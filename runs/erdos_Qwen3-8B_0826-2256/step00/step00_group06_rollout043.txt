import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import sys
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    one_minus_h = 1.0 - h
    # Compute all correlations
    correlations = signal.correlate(h, one_minus_h, mode='full')
    # Normalize by dx
    correlations /= dx
    # Find the maximum value
    return np.max(correlations)

def initial_h_values():
    # Placeholder for the initial construction
    # For the purpose of this example, we'll return a simple uniform distribution
    n_points = 95
    return np.full(n_points, 1.0 / n_points)

# Strategy:
# 1. Start with the initial h_values and refine them using numerical optimization.
# 2. Use a gradient-based optimization approach to minimize the maximum overlap integral.
# 3. Use constraints to ensure the integral of h over [0, 2] is exactly 1.
# 4. Sample a few thousand h_values to keep the computation feasible within the time budget.
# 5. Use a simple initial guess and refine it with optimization algorithms to find a better solution.

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define the number of samples
    n_points = 1000
    dx = 2.0 / n_points
    
    # Initial guess (uniform distribution)
    initial_guess = np.full(n_points, 1.0 / n_points)
    
    # Objective function to minimize: maximum overlap integral
    def objective_function(h_values):
        h = np.array(h_values)
        one_minus_h = 1.0 - h
        correlations = signal.correlate(h, one_minus_h, mode='full')
        correlations /= dx
        return np.max(correlations)
    
    # Constraints: sum(h) * dx = 1
    # Equivalent to sum(h) = n_points / 2
    def constraint_function(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Define the optimization variables
    h = cp.Variable(n_points)
    objective = cp.Minimize(objective_function(h))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    # Create and solve the optimization problem
    start_time = time.time()
    prob = cp.Problem(objective, constraints)
    try:
        result = prob.solve(solver='SCIP', time_limit=budget_s - (time.time() - start_time))
    except:
        # If SCIP is not available, fall back to a gradient-based optimization
        result = opt.minimize(
            objective_function,
            initial_guess,
            method='L-BFGS-B',
            bounds=(0, 1),
            constraints={'type': 'eq', 'fun': constraint_function, 'jac': lambda x: np.ones_like(x)}
        )
    
    # Extract the best solution
    best_h_values = result.x if 'result' in locals() else h.value
    best_c5 = evaluate_erdos_solution(best_h_values)
    
    return best_h_values, best_c5, n_points
