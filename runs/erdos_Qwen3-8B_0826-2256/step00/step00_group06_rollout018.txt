def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 1000  # Adjust based on time budget
    dx = 2.0 / n_points
    max_iterations = 100
    tolerance = 1e-5
    
    # Initial guess
    h = np.zeros(n_points)
    h[:n_points // 2] = 1.0
    h = (h + np.random.rand(n_points) * 0.1) * (1.0 / np.sum(h)) * n_points / 2.0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for _ in range(max_iterations):
        t1 = time.time()
        
        # Compute the overlap integral for all shifts
        c5_values = np.zeros(n_points + 1)
        for k in range(n_points + 1):
            if k == 0:
                overlap = np.sum(h * (1 - h))
            else:
                overlap = np.sum(h * (1 - np.roll(h, k)))
            c5_values[k] = overlap * dx
        
        current_c5 = np.max(c5_values)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # If we are close enough, break early
        if current_c5 < 0.380876 + tolerance:
            break
        
        # Simple greedy update to minimize the maximum overlap
        max_overlap_index = np.argmax(c5_values)
        if max_overlap_index == 0:
            # No shift to consider, break
            break
        
        # Find the shift with maximum overlap
        shift = max_overlap_index
        # Find the indices where h is 1 and 0
        indices_1 = np.where(h == 1)[0]
        indices_0 = np.where(h == 0)[0]
        
        # Try to flip some 1s to 0s and 0s to 1s to reduce overlap
        if len(indices_1) > 0 and len(indices_0) > 0:
            # Choose a random index from each set
            idx_1 = np.random.choice(indices_1)
            idx_0 = np.random.choice(indices_0)
            
            # Flip the values
            h[idx_1] = 0.0
            h[idx_0] = 1.0
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > tolerance:
            # Adjust h to satisfy the constraint
            scale = 1.0 / (total)
            h *= scale * n_points / 2.0
        
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break
    
    return best_h, best_c5, n_points
