import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions and initial values
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Determine n_points based on budget and efficiency
    # We prioritize speed, so use a smaller n_points for faster evaluation
    n_points = 200  # Adjusted for better trade-off between speed and accuracy
    
    # Initial guess (using the provided initial construction)
    h_initial = np.array(initial_h_values[:n_points] + initial_h_values[:n_points])
    h_initial = h_initial[:n_points]
    
    # Ensure the initial guess satisfies the integral constraint
    h_initial = h_initial / (np.sum(h_initial) / (n_points / 2))
    
    # Optimization loop
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # Define the function to minimize
    def objective(h):
        # Ensure h stays within [0, 1]
        h_clamped = np.clip(h, 0, 1)
        # Compute the overlap integral using convolution
        conv = np.correlate(h_clamped, 1 - h_clamped, mode="full")
        # Compute the maximum value of the overlap integral
        c5 = np.max(conv) * (2.0 / n_points)
        return c5
    
    # Define the constraint: sum(h) == n_points / 2
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Set up the optimization problem
    h = cp.Variable(n_points, nonneg=True)
    obj = cp.Minimize(objective(h))
    cons = [constraint(h) == 0]
    prob = cp.Problem(obj, cons)
    
    # Initial guess for the optimization
    h_guess = h_initial.copy()
    
    # Run the optimization
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100, eps=1e-4, warm_start=True)
        if result is not None:
            best_h = h.value
            best_c5 = objective(best_h)
    except:
        pass
    
    # Additional refinement using local optimization
    def objective_func(h):
        h_clamped = np.clip(h, 0, 1)
        conv = np.correlate(h_clamped, 1 - h_clamped, mode="full")
        c5 = np.max(conv) * (2.0 / n_points)
        return c5
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    # Local optimization
    h_local = best_h.copy()
    result = optimize.minimize(
        objective_func,
        h_local,
        constraints={"type": "eq", "fun": constraint_func},
        bounds=[(0, 1) for _ in range(n_points)],
        method="L-BFGS-B",
        options={"ftol": 1e-6, "maxiter": 100}
    )
    best_h = result.x
    best_c5 = objective_func(best_h)
    
    # Ensure best_h stays within [0, 1]
    best_h = np.clip(best_h, 0, 1)
    
    # Final check for the constraint
    if np.abs(np.sum(best_h) - n_points / 2) > 1e-6:
        best_h = best_h / (np.sum(best_h) / (n_points / 2))
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
