import numpy as np

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    class ErdosOptimizer:
        def __init__(self, initial_h=None, n_points=45, target_c5=0.380876, max_evals=200, max_time=60):
            self.n_points = n_points
            self.dx = 2.0 / n_points
            self.target_c5 = target_c5
            self.max_evals = max_evals
            self.max_time = max_time
            self.c5_history = []
            self.best_h = None
            self.best_c5 = float('inf')
            self.initial_h = initial_h
            self.eval_count = 0
            self.warmup_samples = 10
            self.num_workers = min(10, multiprocessing.cpu_count())
            
        def _construct_h(self):
            h = np.zeros(self.n_points)
            if self.initial_h is not None:
                h = np.array(self.initial_h)
                h = np.clip(h, 0, 1)
                if np.abs(np.sum(h) - self.n_points / 2) > 1e-6:
                    h = np.random.rand(self.n_points)
                    h = np.clip(h, 0, 1)
                    h = h / np.sum(h) * self.n_points / 2
            else:
                h = np.random.rand(self.n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * self.n_points / 2
            return h
        
        def _evaluate(self, h):
            if self.eval_count >= self.max_evals:
                return self.best_c5
            if self.eval_count < self.warmup_samples:
                h = np.random.rand(self.n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * self.n_points / 2
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * self.n_points / 2
            self.eval_count += 1
            shifts = np.arange(-self.n_points + 1, self.n_points)
            max_overlap = 0.0
            for k in shifts:
                if k == 0:
                    continue
                x = np.arange(self.n_points)
                shifted = np.roll(x, k)
                if k > 0:
                    shifted = shifted[:self.n_points - k]
                else:
                    shifted = shifted[k:]
                overlap = np.sum(h[shifted] * (1 - h))
                max_overlap = max(max_overlap, overlap)
            c5 = max_overlap * self.dx
            self.c5_history.append(c5)
            if c5 < self.best_c5:
                self.best_c5 = c5
                self.best_h = h.copy()
            return c5
        
        def _optimize(self):
            pool = multiprocessing.Pool(self.num_workers)
            results = []
            for _ in range(self.max_evals):
                h = self._construct_h()
                result = pool.apply_async(self._evaluate, (h,))
                results.append(result)
                if time.time() - start_time > self.max_time:
                    break
            pool.close()
            pool.join()
            for result in results:
                result.get()
            return self.best_h, self.best_c5

    best_h, best_c5 = optimizer._optimize()
