```python
import numpy as np
import scipy.optimize as optimize
import time
import math
from scipy.optimize import minimize
from scipy.optimize.slsqp import minimize_slsqp
from scipy.signal import correlate
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda x: np.max(np.correlate(x, 1 - x, mode="full")) * (2.0 / len(x))
initial_h_values = np.load('initial_h.npy')  # For example, a 95-point solution

# Constants
MAX_POINTS = 1000
TARGET_C5 = 0.380876
DEFALUT_N_POINTS = 100
DEFAULT_BUDGET_S = 100

def run(seed=42, budget_s=DEFAULT_BUDGET_S, **kwargs):
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Function to evaluate the C5 value
    def evaluate(h):
        n_points = len(h)
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5

    # Function to optimize a single h vector
    def optimize_h(n_points, seed):
        np.random.seed(seed)
        # Initial guess based on uniform distribution
        h_initial = np.random.rand(n_points)
        # Scale to meet the integral constraint
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        # Use L-BFGS-B to minimize the maximum overlap
        def objective(h):
            return evaluate(h)
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, tol=1e-6)
        if result.success:
            return result.x, evaluate(result.x)
        else:
            return h_initial, evaluate(h_initial)

    # Function to evaluate a solution
    def evaluate_solution(h):
        nonlocal best_c5, best_h, best_n_points
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = len(h)
            print(f"New best C5: {best_c5} with {len(h)} points")

    # Try different numbers of points
    for n_points in [DEFALUT_N_POINTS, 150, 200, 250, 300, 350, 400, 450, 500, 550, 600, 650, 700, 750, 800, 850, 900, 950, 1000]:
        if time.time() - start_time > budget_s:
            break
        if n_points > MAX_POINTS:
            continue
        # Run the optimization
        h, c5 = optimize_h(n_points, seed)
        evaluate_solution(h)
        # Early stopping if we reach the target
        if best_c5 <= TARGET_C5 + 1e-5:
            break

    print(f"Final best C5: {best_c5} with {best_n_points} points")
    return best_h, best_c5, best_n_points
```
