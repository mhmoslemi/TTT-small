```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
from copy import deepcopy

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * 2.0 / len(h)
initial_h_values = np.load('initial_h.npy')

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for speed
    max_iter = 100  # Limit iterations to stay within time budget
    best_c5 = float('inf')
    best_h = None

    # Initial guess
    h = initial_h_values[:n_points].copy()
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    h /= (np.sum(h) * 2.0 / n_points)  # Normalize to ensure integral is 1
    h = np.clip(h, 0, 1)

    # Initial evaluation
    c5 = evaluate_erdos_solution(h)
    best_c5 = c5
    best_h = h.copy()

    # Optimization loop
    for i in range(max_iter):
        elapsed = i * 1.0  # Simulate time (adjust as needed)
        if elapsed >= budget_s:
            break
        
        # Try perturbing the current solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2.0) / np.sum(h_perturbed)
        
        # Evaluate
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed.copy()

        # Try a greedy improvement
        for j in range(n_points):
            for shift in range(1, 5):
                h_shifted = np.roll(h, shift)
                overlap = np.correlate(h, 1 - h_shifted, mode="full") * 2.0 / n_points
                max_overlap = np.max(overlap)
                if max_overlap < best_c5:
                    best_c5 = max_overlap
                    best_h = h_shifted.copy()

        # Try alternating between optimizing for different shifts
        for shift in range(1, 5):
            h_shifted = np.roll(h, shift)
            overlap = np.correlate(h, 1 - h_shifted, mode="full") * 2.0 / n_points
            max_overlap = np.max(overlap)
            if max_overlap < best_c5:
                best_c5 = max_overlap
                best_h = h_shifted.copy()

        # Try small convex optimization
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        objective = cp.Maximize(cp.max(cp.correlate(x, 1 - x, mode="full") * 2.0 / n_points))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-4)
            if prob.status == 'OPTIMAL':
                h_new = x.value
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
        except:
            pass

        # Try small local search
        for j in range(n_points):
            for dj in [-1, 1]:
                h_test = h.copy()
                h_test[j] += dj * 0.05
                h_test = np.clip(h_test, 0, 1)
                h_test = h_test * (n_points / 2.0) / np.sum(h_test)
                c5_test = evaluate_erdos_solution(h_test)
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_test.copy()

        h = best_h.copy()

    return best_h, best_c5, n_points
```
