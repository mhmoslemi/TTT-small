```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    h_initial = h_initial / np.sum(h_initial) * (n_points_initial / 2)
    
    # Try different n_points starting from initial n_points
    n_points_list = [n_points_initial, 90, 120, 150, 180, 210, 240, 270, 300]
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(h_initial)
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Rescale h_initial to n_points
        h_rescaled = np.linspace(0, 2, n_points)
        h_rescaled_values = np.interp(h_rescaled, np.linspace(0, 2, n_points_initial), h_initial)
        h_rescaled_values = h_rescaled_values / np.sum(h_rescaled_values) * (n_points / 2)
        
        # Define optimization variables
        h = cp.Variable(n_points, nonneg=True)
        constraints = [cp.sum(h) == n_points / 2]
        
        # Objective: minimize the maximum overlap
        max_overlap = cp.Maximize(cp.sum(cp.multiply(h, 1 - cp.roll(h, 1)) * 2.0 / n_points))
        prob = cp.Problem(max_overlap, constraints)
        
        # Solve the optimization problem
        result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - (time.time() - start_time))
        
        if result is not None:
            h_values = h.value
            c5 = evaluate_erdos_solution(h_values)
            
            if c5 < best_c5:
                best_h = h_values
                best_c5 = c5
        
        # Also try a random perturbation
        perturb = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturb = h_rescaled_values + perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        c5_perturb = evaluate_erdos_solution(h_perturb)
        
        if c5_perturb < best_c5:
            best_h = h_perturb
            best_c5 = c5_perturb
    
    return (best_h, best_c5, len(best_h))
```
