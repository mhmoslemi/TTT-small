def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial guess
    n_points = 80
    h_initial = initial_h_values.copy()
    h_initial = np.array(h_initial)
    h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
    h_initial = np.clip(h_initial, 0, 1)
    # Ensure the integral is 1
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Try different n_points to find a good solution
    for n_points in [80, 100, 120, 160]:
        elapsed = 0
        best_c5 = float('inf')
        best_h = h_initial.copy()
        
        # Define the optimization problem
        def evaluate_h(h):
            h = np.clip(h, 0, 1)
            # Ensure the integral is 1
            h = h * (n_points / 2) / np.sum(h)
            # Compute the overlap integral
            h_array = h.reshape(-1)
            h_shifted = np.zeros((n_points, n_points), dtype=np.float64)
            for i in range(n_points):
                h_shifted[i, :] = np.roll(h_array, i)
            integrals = np.sum(h_array * (1 - h_shifted), axis=0)
            integral_values = integrals[:n_points - 1]
            # Take the maximum integral
            c5 = np.max(integral_values) * (2.0 / n_points)
            return c5
        
        # Use a simple gradient descent approach
        def optimize_h(h_initial):
            h = h_initial.copy()
            for _ in range(100):
                # Compute current c5
                c5 = evaluate_h(h)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                    if best_c5 < 0.380876:
                        break
                # Gradient descent step
                h_array = h.reshape(-1)
                h_shifted = np.zeros((n_points, n_points), dtype=np.float64)
                for i in range(n_points):
                    h_shifted[i, :] = np.roll(h_array, i)
                integrals = np.sum(h_array * (1 - h_shifted), axis=0)
                integral_values = integrals[:n_points - 1]
                grad = np.zeros(n_points)
                for i in range(n_points):
                    # Compute derivative of c5 w.r.t h[i]
                    # Use finite difference
                    h_plus = h.copy()
                    h_plus[i] += 1e-6
                    c5_plus = evaluate_h(h_plus)
                    h_minus = h.copy()
                    h_minus[i] -= 1e-6
                    c5_minus = evaluate_h(h_minus)
                    grad[i] = (c5_plus - c5_minus) / (2 * 1e-6)
                # Update h
                h -= 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                elapsed += 0.01
                if elapsed >= budget_s:
                    break
            return best_h, best_c5
        
        start_time = time.time()
        h_opt, c5_opt = optimize_h(h_initial)
        elapsed = time.time() - start_time

            best_c5 = c5_opt
            best_h = h_opt.copy()
        if elapsed >= budget_s:
            break
    
    # Evaluate the best solution
    h_values = best_h
    c5_bound = best_c5
    return h_values, c5_bound, n_points
