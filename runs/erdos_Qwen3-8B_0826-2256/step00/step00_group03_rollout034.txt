```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_70.npy')  # Assume this file exists with the initial 70-point solution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, adjust as needed

    def evaluate_candidate(h_values):
        nonlocal best_c5
        c5 = evaluate_erdos_solution(h_values)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values.copy()
        return c5

    def optimize_h(n_points):
        h = cp.Variable(n_points)
        constraints = [
            0 <= h,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        obj = cp.Maximize(cp.max([cp.sum(h[i:i + n_points] * (1 - h[i + k:i + n_points + k])) for k in range(n_points)]))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.SCS, verbose=False)
            if prob.status == 'optimal':
                return h.value
        except:
            pass
        return None

    def simulate_and_evaluate(h_values):
        c5 = evaluate_erdos_solution(h_values)
        return c5, h_values

    # Start with the initial solution
    initial_h = initial_h_values.copy()
    initial_c5 = evaluate_erdos_solution(initial_h)
    if initial_c5 < best_c5:
        best_c5 = initial_c5
        best_h = initial_h.copy()

    # Try a few random perturbations of the initial solution
    for _ in range(5):
        perturbation = np.random.uniform(-0.1, 0.1, len(initial_h))
        h = initial_h + perturbation
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Refine with a small optimization
    refined_h = optimize_h(len(best_h))
    if refined_h is not None:
        best_h = refined_h
        best_c5 = evaluate_erdos_solution(best_h)

    # Try a few different n_points values
    for n_points in [50, 100, 150, 200]:
        if time.time() - start_time > budget_s - 10:
            break
        # Randomly sample h
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

        # Try optimization
        refined_h = optimize_h(n_points)
        if refined_h is not None:
            c5 = evaluate_erdos_solution(refined_h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = refined_h.copy()

    return best_h, best_c5, len(best_h)
```
