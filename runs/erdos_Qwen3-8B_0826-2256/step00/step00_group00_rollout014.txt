def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros(2 * n_points - 1)
    h_shifted[:n_points] = h
    h_shifted[n_points:] = h[::-1]
    overlap = correlate(h, 1 - h, mode='full')
    c5 = np.max(overlap) * dx
    return c5
