import numpy as np
import scipy.optimize as optimize
import time
import sys
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, i:] = h[i:]
    overlap = np.dot(h, 1 - shifted_h)
    max_overlap = np.max(overlap)
    return max_overlap * dx

def initial_h_values():
    return [0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
            0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]

def run(seed=42, budget_s=100, **kwargs):
    # Seed for reproducibility
    np.random.seed(seed)
    
    # Start timer
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 points, can increase if time allows
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]  # Trim to n_points
    h_initial = np.array(h_initial)
    sum_h = np.sum(h_initial)
    # Normalize to make sure the integral is 1
    h_initial = h_initial * (n_points / 2) / sum_h

    best_c5 = float('inf')
    best_h = h_initial.copy()

    # Define the objective function
    def objective_function(h):
        h = h.reshape((n_points,))
        # Ensure h is in [0, 1] and sum is correct
        h_clamped = np.clip(h, 0, 1)
        # Normalize to ensure the integral is 1
        h_normalized = h_clamped * (n_points / 2) / np.sum(h_clamped)
        # Compute the overlap integral
        overlap = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(h_normalized, k)
            overlap[k] = np.dot(h_normalized, 1 - shifted)
        c5 = np.max(overlap)
        return c5

    # Define the optimization problem
    # We will use a genetic algorithm to find the best h
    def genetic_optimization():
        nonlocal best_c5, best_h
        population_size = 50
        generations = 100
        mutation_rate = 0.1
        elite_size = 5

        # Initialize population
        population = np.random.rand(population_size, n_points)
        population = np.clip(population, 0, 1)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)

        for gen in range(generations):
            # Evaluate fitness
            fitness = np.array([objective_function(ind) for ind in population])
            # Find the best individual
            best_idx = np.argmin(fitness)
            current_best = population[best_idx]
            current_best_c5 = fitness[best_idx]
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = current_best.copy()

            # Select parents
            parents_idx = np.argsort(fitness)[:elite_size]
            parents = population[parents_idx]

            # Generate offspring
            offspring = []
            for _ in range(population_size - elite_size):
                # Select two parents
                parent1_idx = np.random.choice(parents_idx)
                parent2_idx = np.random.choice(parents_idx)
                parent1 = parents[parent1_idx]
                parent2 = parents[parent2_idx]
                # Crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] = np.random.rand()
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                offspring.append(child)
            # Update population
            population[elite_size:] = offspring
            # Check time budget
            if time.time() - start_time > budget_s:
                break

    # Run the genetic algorithm
    genetic_optimization()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    # Return the best solution found
    return best_h.tolist(), best_c5, n_points
