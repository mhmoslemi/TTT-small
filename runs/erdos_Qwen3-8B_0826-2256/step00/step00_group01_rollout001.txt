```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Function to compute the C5 value for a given h
    def compute_c5(h_values):
        n_points = len(h_values)
        dx = 2.0 / n_points
        h = np.array(h_values)
        shifted_h = np.concatenate([np.zeros_like(h), h[:-1]])
        correlation = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlation) * dx
        return c5

    # Function to evaluate the objective and constraints for CVXPy
    def objective_func(h_values):
        return compute_c5(h_values)

    # Function to enforce the integral constraint using CVXPy
    def integral_constraint(h_values):
        return cp.sum(h_values) == len(h_values) / 2

    # Optimization strategy
    def optimize_h(n_points, initial_guess=None):
        if initial_guess is None:
            initial_guess = initial_h_values.copy()
        h = cp.Variable(n_points)
        objective = cp.Minimize(compute_c5(h))
        constraints = [h >= 0, h <= 1, integral_constraint(h)]
        problem = cp.Problem(objective, constraints)
        try:
            result = problem.solve(max_time=10, solver=cp.SCS)
            if result is not None:
                return h.value
        except:
            pass
        return initial_guess

    # Start from the initial guess
    n_points = 100
    best_h = initial_h_values.copy()
    best_c5 = compute_c5(best_h)
    best_n_points = n_points

    # Try with different n_points to find the best solution within the budget
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        h = optimize_h(n_points, initial_guess=initial_h_values.copy())
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points

    # Refine the solution with a more detailed optimization
    if time.time() - start_time < budget_s - 1:
        h = optimize_h(best_n_points, initial_guess=best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    # Final check
    if time.time() - start_time < budget_s - 1:
        h = optimize_h(best_n_points, initial_guess=best_h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    return (best_h, best_c5, best_n_points)
```
