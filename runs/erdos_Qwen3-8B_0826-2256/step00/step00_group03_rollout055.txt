import numpy as np
import scipy.optimize as optimize
import time
import copy
import cvxpy as cp
import random
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with 200 points for balance between speed and quality
    h = copy.deepcopy(initial_h_values[:n_points])
    h = h / (n_points / 2.0)  # Ensure the integral constraint is satisfied
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Optimization steps
    for _ in range(15):  # Max 15 iterations
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # First: Gradient descent-style update
        def objective_function(h):
            return evaluate_erdos_solution(h)

        # Use a simple gradient-based approach
        h = h.astype(float)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            grad[i] = (objective_function(h_plus) - objective_function(h_minus)) / (2 * 1e-4)

        # Update h in the direction of negative gradient
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0)  # Re-normalize to maintain the integral constraint

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Second: Random perturbation to escape local maxima
        h = h.copy()
        for i in range(len(h)):
            if random.random() < 0.1:
                h[i] = np.random.uniform(0, 1)
        h = h * (n_points / 2.0)
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Third: Local optimization using CVXPY
        try:
            # Convert h to a variable
            h_var = cp.Variable(len(h), bounds=(0, 1))
            # Constraint: integral constraint
            constraint = cp.sum(h_var) == n_points / 2.0
            # Objective: minimize the maximum overlap
            # We approximate this by minimizing the maximum of several shifted overlaps
            # For simplicity, we evaluate the correlation for a few shifts
            # This is an approximation, but it's faster than full optimization
            overlap_values = []
            for k in range(1, 6):  # Try shifts from 1 to 5
                shifted_h = np.roll(h_var.value, k) if h_var.value is not None else np.roll(h, k)
                overlap = cp.sum(h_var * (1 - shifted_h))
                overlap_values.append(overlap)

            # Objective: minimize the maximum of these overlaps
            objective = cp.Minimize(cp.Max(overlap_values))
            problem = cp.Problem(objective, [constraint])
            problem.solve(solver=cp.SCS, eps=1e-4)
            if problem.status == 'optimal':
                h = h_var.value
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
        except:
            pass

        # Fourth: Try a new random initialization
        if random.random() < 0.1:
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2.0)
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

    # Final refinement
    h = best_h.copy()
    for _ in range(10):
        h = h.astype(float)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            grad[i] = (evaluate_erdos_solution(h_plus) - evaluate_erdos_solution(h_minus)) / (2 * 1e-4)
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0)

    final_c5 = evaluate_erdos_solution(h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()

    # Return the best solution
    return best_h, best_c5, n_points
