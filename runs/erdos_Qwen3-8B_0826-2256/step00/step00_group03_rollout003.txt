```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import spsolve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    initial_h_values = np.load('initial_h_values.npy')  # Simulated initial values
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    initial_sum = np.sum(initial_h_values)
    if abs(initial_sum - n_points / 2) > 1e-6:
        initial_h_values = initial_h_values / (initial_sum / (n_points / 2))
    
    # Try a lower number of points
    for n in [50, 60, 70, 80, 90, 100]:
        h = initial_h_values[:n].copy()
        h = h / np.sum(h) * (n / 2)
        h = np.clip(h, 0, 1)
        
        # Compute current C5
        def evaluate_c5(h):
            h = np.concatenate([h, h[:n-1]], axis=0)  # Extend to full correlation
            c5_values = np.correlate(h, 1 - h, mode="full") * dx
            return np.max(c5_values)
        
        c5_bound = evaluate_c5(h)
        if c5_bound < 0.380876:
            print(f"Found better C5 with {n} points: {c5_bound}")
            return (h[:n], c5_bound, n)
    
    # If not found, optimize with more points
    n_points = 100
    h = np.random.rand(n_points) * 2 - 1
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    def objective(h):
        h = np.concatenate([h, h[:n_points-1]], axis=0)
        c5_values = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(c5_values)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints={'type': 'eq', 'fun': constraint})
    
    if result.success:
        h_opt = result.x
        c5_bound = objective(h_opt)
        print(f"Optimized C5 with {n_points} points: {c5_bound}")
        return (h_opt, c5_bound, n_points)
    else:
        print("Optimization failed")
        return (h, objective(h), n_points)
```
