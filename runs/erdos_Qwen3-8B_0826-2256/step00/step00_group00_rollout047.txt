```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import random
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Starting with 100 points for faster optimization
    max_points = 200  # Max points to try if budget allows

    # Initial solution
    h = initial_h_values.copy()
    h = np.array(h)
    h = h[:n_points]
    h /= (h.sum() / (n_points / 2))  # Normalize to satisfy the integral constraint
    current_c5 = evaluate_erdos_solution(h)
    best_c5 = current_c5
    best_h = h.copy()
    print(f"Initial C5: {current_c5:.6f} with {n_points} points")

    # Try increasing number of points up to max_points
    for n_points in range(100, max_points + 1, 50):
        if time.time() - start_time > budget_s - 5:
            break
        print(f"Trying with {n_points} points...")
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / h.sum()  # Normalize to satisfy the integral constraint

        # Optimization: minimize max correlation
        # Reformulate to minimize the maximum of the absolute correlations
        # We'll use a convex optimization approach
        # Since the problem is non-convex, we'll use a local optimizer with multiple restarts
        for _ in range(5):  # Multiple restarts to avoid local minima
            # Initial guess
            h_initial = copy.deepcopy(h)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / h_initial.sum()

            # Optimization
            def objective(h):
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / h.sum()
                correlations = signal.correlate(h, 1 - h, mode='full')
                return np.max(correlations)

            result = opt.minimize(
                objective,
                h_initial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'disp': False, 'ftol': 1e-8, 'maxiter': 100}
            )

            # Check if we improved
            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / h.sum()
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    print(f"Found better C5: {current_c5:.6f} with {n_points} points")
            else:
                print("Optimization failed.")

    # Final check with the best solution
    # Try further refinement with more points if time permits
    if time.time() - start_time < budget_s - 5:
        n_points = min(300, max_points + 50)
        print(f"Trying refinement with {n_points} points...")
        h = best_h.copy()
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / h.sum()

        for _ in range(3):
            h_initial = copy.deepcopy(h)
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (n_points / 2) / h_initial.sum()

            def objective(h):
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / h.sum()
                correlations = signal.correlate(h, 1 - h, mode='full')
                return np.max(correlations)

            result = opt.minimize(
                objective,
                h_initial,
                method='L-BFGS-B',
                bounds=[(0, 1) for _ in range(n_points)],
                options={'disp': False, 'ftol': 1e-8, 'maxiter': 100}
            )

            if result.success:
                h = result.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / h.sum()
                current_c5 = evaluate_erdos_solution(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                    print(f"Refined C5: {current_c5:.6f} with {n_points} points")

    # Ensure the best solution is within the time budget
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s:
        # Try one more refinement if time allows
        n_points = min(400, max_points + 100)
        print(f"Final refinement with {n_points} points...")
        h = best_h.copy()
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / h.sum()

        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / h.sum()
            correlations = signal.correlate(h, 1 - h, mode='full')
            return np.max(correlations)

        result = opt.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'disp': False, 'ftol': 1e-8, 'maxiter': 100}
        )

        if result.success:
            h = result.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / h.sum()
            current_c5 = evaluate_erdos_solution(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                print(f"Final C5: {current_c5:.6f} with {n_points} points")

    print(f"Best C5 found: {best_c5:.6f} with {len(best_h)} points")
    return (best_h, best_c5, len(best_h))
```
