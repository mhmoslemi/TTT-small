```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume this is a numpy array of shape (n_points,)

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 100  # Start with 100 points for speed
    dx = 2.0 / n_points
    h = np.copy(initial_h_values[:n_points])
    h = h / np.sum(h) * (n_points / 2.0)  # Normalize h to satisfy the integral constraint

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.copy()
        h = h / np.sum(h) * (n_points / 2.0)
        overlap = np.max(signal.correlate(h, 1 - h, mode="full")) * dx
        return overlap

    # Optimization using numerical optimization
    def optimize_h():
        h = np.copy(initial_h_values[:n_points])
        h = h / np.sum(h) * (n_points / 2.0)

        # Optimization settings
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-6, options={'maxiter': 100})
        return result.x

    # Main loop
    best_h = h
    best_c5 = objective(h)
    elapsed = 0
    while elapsed < budget_s - 1:
        h = optimize_h()
        c5 = objective(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        elapsed = time.time() - start_time

    # Try increasing number of points if time allows
    if elapsed < budget_s - 5:
        n_points = 200
        dx = 2.0 / n_points
        h = np.copy(initial_h_values[:n_points])
        h = h / np.sum(h) * (n_points / 2.0)

        # Optimization with more points
        def objective_large(h_values):
            h = h_values.copy()
            h = h / np.sum(h) * (n_points / 2.0)
            overlap = np.max(signal.correlate(h, 1 - h, mode="full")) * dx
            return overlap

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        h = optimize_h()
        c5 = objective_large(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5

    return best_h, best_c5, n_points
```
