import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions and variables
# evaluate_erdos_solution() is a function to compute C5 for a given h
# initial_h_values is a list of n_points samples for the initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    max_n_points = 200  # Start with 200 points to balance speed and accuracy
    n_points = max_n_points
    dx = 2.0 / n_points
    
    # Initial guess from provided initial_h_values
    h_values = np.copy(initial_h_values)
    h_values = h_values[:n_points]  # Ensure we don't exceed the number of points
    h_values = h_values / np.sum(h_values) * (n_points / 2)  # Normalize to meet the integral constraint
    
    best_c5 = float('inf')
    best_h = np.copy(h_values)
    
    def evaluate(h):
        nonlocal best_c5, best_h
        h = h.reshape(-1)
        h = np.concatenate([h, h])
        corr = np.correlate(h, 1 - h[:n_points], mode="full")
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h[:n_points]
        return c5
    
    # Apply CMA-ES optimizer
    from cma import CMAEvolutionStrategy
    
    # Convert to array of parameters
    params = h_values.copy()
    params = params / np.sum(params) * (n_points / 2)
    
    # Set up the optimizer
    es = CMAEvolutionStrategy(params, 0.1, {
        'popsize': 20,
        'bounds': np.array([[0.0, 1.0]] * n_points),
        'verb_disp': 0,
        'verb_period': 100,
        't_out': budget_s * 2  # Allow some buffer
    })
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        # Run one generation
        solutions = es.ask()
        fitness = np.array([evaluate(solution) for solution in solutions])
        es.tell(solutions, fitness)
        es.recommended_step_size()
        
        # Check if we have a better solution
        current_best = es.result()[0]
        current_c5 = evaluate(current_best)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_best
    
    es.stop()
    
    # Return the best solution found
    return best_h, best_c5, n_points
