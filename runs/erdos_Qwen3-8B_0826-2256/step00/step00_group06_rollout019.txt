```python
import numpy as np
import scipy.optimize as optimize
import time
import sys
import random
import copy
import math

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Load initial solution with n=95

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed
    random.seed(seed)
    np.random.seed(seed)
    
    # Helper function to evaluate the C5 bound
    def evaluate(h):
        return evaluate_erdos_solution(h)
    
    # Helper function to generate a random perturbation
    def perturb(h, magnitude=0.1):
        h_perturbed = copy.deepcopy(h)
        for i in range(len(h_perturbed)):
            if random.random() < magnitude:
                h_perturbed[i] += np.random.uniform(-0.1, 0.1)
                h_perturbed[i] = np.clip(h_perturbed[i], 0.0, 1.0)
        return h_perturbed
    
    # Helper function to generate a random initial solution
    def generate_random_h(n_points):
        h = np.random.uniform(0, 1, n_points)
        total = np.sum(h)
        h = h * (n_points / 2) / total
        return h
    
    # Start with the initial solution
    best_h = initial_h_values
    best_c5 = evaluate(best_h)
    best_n_points = len(best_h)
    
    # Try different numbers of points
    n_points_list = [100, 200, 300, 400]
    
    for n_points in n_points_list:
        start_time = time.time()
        h = generate_random_h(n_points)
        current_c5 = evaluate(h)
        
        # Optimization loop
        for _ in range(1000):
            h = perturb(h, magnitude=0.05)
            current_c5 = evaluate(h)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h
                best_n_points = n_points
                if best_c5 < 0.380876:
                    print(f"New best C5: {best_c5} with n={best_n_points}")
            
            elapsed = time.time() - start_time
            if elapsed > budget_s - 1:
                break
        
        # After trying this n_points, check if we should continue
        if best_c5 < 0.380876:
            print(f"Best C5 found: {best_c5} with n={best_n_points}")
        else:
            print(f"No improvement found for n={n_points}, best remains {best_c5} with n={best_n_points}")
    
    return best_h, best_c5, best_n_points
```
