def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = h[(i + np.arange(n_points)) % n_points]
    overlaps = np.zeros(n_points)
    for k in range(n_points):
        overlaps[k] = np.sum(h * (1 - shifted_h[k, :])) * dx
    return np.max(overlaps)

def objective_function(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))

    overlaps = np.zeros(n_points)
    for k in range(n_points):
        overlaps[k] = np.sum(h * (1 - shifted_h[k, :])) * dx
    return np.max(overlaps)
