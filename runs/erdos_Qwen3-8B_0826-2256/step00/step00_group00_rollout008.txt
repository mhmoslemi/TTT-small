```python
import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 samples
    max_n_points = 200  # Max number of points to try
    best_c5 = float('inf')
    best_h = None
    
    # Try different numbers of points
    for n_points in range(100, max_n_points + 1, 50):
        dx = 2.0 / n_points
        # Initialize h with initial values
        h = np.array(initial_h_values) / (len(initial_h_values) / n_points)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), h)
        h = np.clip(h, 0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Function to evaluate C5
        def evaluate_c5(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5
        
        # Optimization using local search
        def objective(x):
            h = np.clip(x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx
        
        # Use local optimization
        result = opt.minimize(objective, h, method='L-BFGS-B', options={'maxiter': 100})
        
        # Update best solution
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x.copy()
        
        # Check if time is up
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check with the best solution found
    if best_h is not None:
        h_values = np.clip(best_h, 0, 1)
        h_values = h_values * (n_points / 2) / np.sum(h_values)
        c5_bound = evaluate_c5(h_values)
    else:
        h_values = initial_h_values
        c5_bound = evaluate_erdos_solution(h_values)
    
    return (h_values, c5_bound, len(h_values))
```
