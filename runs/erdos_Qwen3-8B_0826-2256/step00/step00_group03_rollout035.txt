```python
import numpy as np
import scipy.optimize as opt
import time
import sys
import random
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assume this initial solution is given

def run(seed=42, budget_s=100, **kwargs):
    # Set the random seed
    np.random.seed(seed)
    random.seed(seed)
    
    # Hyperparameters for optimization
    max_n_points = 100  # Trade-off between accuracy and speed
    num_initial_points = 70
    num_trials = 100
    num_iterations = 100
    num_sweep_points = 50
    
    # Initial guess (from provided initial_h_values)
    h_initial = initial_h_values.copy()
    h_initial = h_initial[:max_n_points]  # Limit to max_n_points
    
    # Ensure the initial solution satisfies the constraint
    h_initial = h_initial / (np.sum(h_initial) / (max_n_points / 2))
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Track time
    start_time = time.time()
    
    # First, randomly sample from the initial solution space
    for _ in range(num_trials):
        # Randomly perturb the initial solution
        h_perturbed = h_initial.copy()
        for i in range(len(h_perturbed)):
            if np.random.rand() < 0.1:
                h_perturbed[i] = np.random.rand()
        h_perturbed = h_perturbed / (np.sum(h_perturbed) / (max_n_points / 2))
        
        # Evaluate C5
        current_c5 = evaluate_erdos_solution(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    # Second, use coordinate descent to improve the solution
    for _ in range(num_iterations):
        # Make a copy to avoid modifying the original
        h_current = best_h.copy()
        
        # Try sweeping through the points
        for i in range(len(h_current)):
            for k in range(1, num_sweep_points + 1):
                # Perturb the point slightly
                h_current[i] = np.clip(h_current[i] + np.random.rand() * 0.1 - 0.05, 0, 1)
                h_current = h_current / (np.sum(h_current) / (max_n_points / 2))
                
                # Evaluate C5
                current_c5 = evaluate_erdos_solution(h_current)
                if current_c5 < best_c5:
                    best_h = h_current
                    best_c5 = current_c5
                else:
                    # Revert the change
                    h_current[i] = np.clip(h_current[i] - np.random.rand() * 0.1 + 0.05, 0, 1)
        
        # Keep the best solution found so far
        if evaluate_erdos_solution(best_h) < best_c5:
            best_c5 = evaluate_erdos_solution(best_h)
    
    # Early exit if time is up
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        print(f"Time limit exceeded, returning best solution found so far")
        return (best_h, best_c5, len(best_h))
    
    # Try some more advanced methods if time allows
    # Use CVXPY to find a better solution
    try:
        # Define the optimization variables and constraints
        n_points = len(best_h)
        h = cp.Variable(n_points)
        
        # Objective: minimize the maximum overlap
        overlap = cp.max(cp.sum(h * (1 - h[cp.IndexRange(1, n_points) + cp.IndexRange(0, n_points - 1)])))
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Solve the optimization problem
        prob = cp.Problem(cp.Minimize(overlap), constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        # Convert to numpy array
        h_opt = np.array(h.value).clip(0, 1)
        h_opt = h_opt / (np.sum(h_opt) / (n_points / 2))
        
        # Evaluate the result
        current_c5 = evaluate_erdos_solution(h_opt)
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
    except:
        pass  # Skip if optimization fails
    
    # Final return
    return (best_h, best_c5, len(best_h))
```
