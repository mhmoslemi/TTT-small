def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Start with a smaller number of points
    dx = 2.0 / n_points
    h = np.array(initial_h_values[:n_points], dtype=float)
    h = h / (dx * n_points) * (n_points / 2)  # Normalize to meet the integral constraint

    # Function to compute C5
    def compute_c5(h_vals):
        h = np.array(h_vals, dtype=float)
        h = np.pad(h, (0, 1), mode='constant')  # Pad to handle full convolution
        h_shifted = np.concatenate([h[1:], [0.0]])
        result = np.correlate(h, h_shifted, mode="full")
        return np.max(result) * dx

    # Objective function to minimize (C5)
    def objective(h_vals):
        return compute_c5(h_vals)

    # Constraints
    def integral_constraint(h_vals):
        return np.sum(h_vals) - n_points / 2

    # Bounds (each h[i] must be in [0, 1])
    bounds = Bounds(0, 1)

    # Linear constraint: sum(h) = n_points / 2
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

    # Optimization
    start_time = time.time()
    result = minimize(objective, h, method='SLSQP', bounds=bounds, constraints=constraint, tol=1e-8)

    # Extract the best solution
    best_h = result.x
    best_c5 = compute_c5(best_h)

    # Refine with more points if time permits
    remaining_time = budget_s - (time.time() - start_time)
