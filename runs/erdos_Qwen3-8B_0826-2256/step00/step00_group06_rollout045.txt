import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
# evaluate_erdos_solution() - evaluates the C5 bound for a given h_values
# initial_h_values - initial construction (n=95 samples)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Define the number of samples
    n_points = 100  # Starting with 100 points (adjust if needed)

    # Function to compute C5
    def compute_c5(h_values):
        dx = 2.0 / n_points
        h = np.array(h_values, dtype=float)
        shifted_h = np.zeros(2 * n_points)
        shifted_h[n_points:] = h
        conv = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(conv) * dx
        return c5

    # Initial guess from initial_h_values
    if hasattr(initial_h_values, '__iter__') and len(initial_h_values) == n_points:
        h_init = initial_h_values.copy()
    else:
        # If no initial solution provided, use uniform distribution
        h_init = np.full(n_points, 0.5)

    # Ensure the initial guess satisfies the constraint
    def objective(x):
        return compute_c5(x)

    def constraint(x):
        return np.sum(x) - n_points / 2

    # Optimization setup
    x = cp.Variable(n_points, nonneg=True, lb=0, ub=1)
    objective_func = cp.Minimize(objective(x))
    constraints = [constraint(x) == 0]
    problem = cp.Problem(objective_func, constraints)

    try:
        result = problem.solve(solver=cp.CVXOPT, verbose=False)
        if result is not None:
            best_h = x.value
            best_c5 = compute_c5(best_h)
            best_n_points = n_points
    except:
        pass

    # Try a few different n_points values
    for n_points in [100, 150, 200]:
        # Initialize h_values with uniform distribution
        h_init = np.full(n_points, 0.5)

        # Define optimization problem
        x = cp.Variable(n_points, nonneg=True, lb=0, ub=1)
        objective_func = cp.Minimize(compute_c5(x))
        constraints = [cp.sum(x) == n_points / 2]
        problem = cp.Problem(objective_func, constraints)

        try:
            result = problem.solve(solver=cp.CVXOPT, verbose=False)
            if result is not None:
                current_h = x.value
                current_c5 = compute_c5(current_h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                    best_n_points = n_points
        except:
            pass

        # Check elapsed time
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    return (best_h, best_c5, best_n_points)
