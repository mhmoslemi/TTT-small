```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted[i, :] = h
    shifted = np.roll(shifted, -i, axis=0)
    overlap = np.zeros(n_points)
    for k in range(n_points):
        overlap[k] = np.sum(h * (1 - shifted[k, :])) * dx
    return np.max(overlap)

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                     0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with initial guess
    n_points = 100
    h_initial = initial_h_values()
    h_initial = h_initial[:n_points]
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    best_h = h_initial
    best_c5 = evaluate_erdos_solution(best_h)
    elapsed = 0
    
    for _ in range(50):
        if elapsed >= budget_s:
            break
        
        # Try different n_points
        for n in [50, 100, 150, 200, 250, 300, 350, 400, 450, 500]:
            if elapsed >= budget_s:
                break
            
            # Initialize h
            h = np.random.rand(n) * 2 - 1
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            
            # Optimization with CVXPY
            x = cp.Variable(n)
            constraints = [0 <= x, x <= 1]
            constraints.append(cp.sum(x) == n / 2)
            
            # Objective: minimize the maximum overlap
            overlap = []
            for k in range(n):
                shifted = np.roll(x, k)
                overlap_k = cp.multiply(x, 1 - shifted)
                overlap.append(overlap_k)
            
            max_overlap = cp.Max(overlap)
            prob = cp.Problem(cp.Minimize(max_overlap), constraints)
            
            try:
                prob.solve(solver='MOSEK', verbose=False)
                if prob.status == 'optimal':
                    h = x.value
                    c5 = evaluate_erdos_solution(h)
                    if c5 < best_c5:
                        best_h = h
                        best_c5 = c5
                        elapsed += time.time() - time.time()
                        print(f"Found better solution with n={n}, C5={best_c5}")
            except:
                pass
            
            elapsed += time.time() - time.time()
    
    return best_h, best_c5, len(best_h)
```
