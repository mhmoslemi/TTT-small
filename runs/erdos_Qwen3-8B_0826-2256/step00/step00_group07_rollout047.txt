```python
import numpy as np
import scipy.optimize as optimize
import time
import copy

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial parameters
    n_points = 100  # Start with 100 points for speed
    max_iterations = 100  # Max iterations for optimization
    dx = 2.0 / n_points
    initial_h = initial_h_values(n_points) if n_points == 100 else np.random.rand(n_points) * 2 - 1
    initial_h = np.clip(initial_h, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)  # Ensure the integral is 1

    best_h = copy.copy(initial_h)
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_time = 0.0

    start_time = time.time()
    
    for i in range(max_iterations):
        # Random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, n_points)
        h_candidate = initial_h + perturbation
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        c5_candidate = evaluate_erdos_solution(h_candidate, dx)
        
        if c5_candidate < best_c5:
            best_h = copy.copy(h_candidate)
            best_c5 = c5_candidate
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1.0:
            break
    
    # Try increasing n_points if time allows
    remaining_time = budget_s - elapsed
    if remaining_time > 1.0:
        n_points = min(200, int(n_points * 2))
        dx = 2.0 / n_points
        initial_h = np.random.rand(n_points) * 2 - 1
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
        for i in range(50):
            perturbation = np.random.uniform(-0.1, 0.1, n_points)
            h_candidate = initial_h + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = evaluate_erdos_solution(h_candidate, dx)
            
            if c5_candidate < best_c5:
                best_h = copy.copy(h_candidate)
                best_c5 = c5_candidate
            
            # Check time budget
            elapsed = time.time() - start_time
            if elapsed >= budget_s - 1.0:
                break
    
    return best_h, best_c5, n_points
```
