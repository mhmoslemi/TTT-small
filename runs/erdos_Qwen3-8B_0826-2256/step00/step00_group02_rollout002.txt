import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import sys
import random
import cvxpy as cp

# Pre-imported functions

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if needed
    max_iterations = 100  # Max number of iterations

    # Initial guess from provided initial_h_values (n=45)
    h_initial = initial_h_values
    h_initial = np.concatenate([h_initial, np.zeros(n_points - len(h_initial))])
    h_initial = h_initial[:n_points]
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    h_initial = h_initial[:n_points]

    def compute_c5(h):
        nonlocal n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlate = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(correlate) * (2.0 / n_points)
        return c5

    def objective_function(h):
        return compute_c5(h)

    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    # Optimization using bounded optimization
    bounds = [(0, 1) for _ in range(n_points)]
    h_opt = cp.Variable(n_points, bounds=bounds)
    obj = cp.Minimize(objective_function(h_opt))
    constraints = [constraint_sum(h_opt) == 0]
    prob = cp.Problem(obj, constraints)
    try:
        result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=60)
        if result is not None:
            best_h = h_opt.value
            best_c5 = compute_c5(best_h)
    except:
        pass

    # If optimization fails, use local search
    if best_h is None:
        best_h = h_initial
        best_c5 = compute_c5(best_h)

    # Local search with simulated annealing
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = best_h.copy()
        idx = np.random.randint(n_points)
        h_new[idx] = np.random.uniform(0, 1)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new

    # Final check on best solution
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = compute_c5(best_h)

    # Output the result
    return (best_h, best_c5, n_points)
