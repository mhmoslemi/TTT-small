import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros(2 * n_points - 1)
    for i in range(n_points):
        shifted_h[i:i + n_points] = h
    ones_minus_h = 1.0 - h
    corr = np.correlate(shifted_h, ones_minus_h, mode='full')
    c5_bound = np.max(corr) * dx
    return c5_bound

def initial_h_values():
    # Initial construction with n=42 samples
    h = np.zeros(42)
    for i in range(42):
        if i % 3 == 0:
            h[i] = 1.0
        elif i % 3 == 1:
            h[i] = 0.5
        else:
            h[i] = 0.0
    return h

def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 42  # Start with 42 samples for efficiency
    
    # Try different n_points
    for n_points in [42, 64, 84, 100, 120, 140, 160, 180, 200, 220, 240]:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Initialize h_values
        h = initial_h_values()
        h = np.resize(h, n_points)
        h = h * (n_points / 2) / np.sum(h)
        
        # Function to evaluate c5
        def evaluate_c5(h_values):
            h = np.array(h_values, dtype=float)
            dx = 2.0 / len(h)
            shifted_h = np.zeros(2 * len(h) - 1)
            for i in range(len(h)):
                shifted_h[i:i + len(h)] = h
            ones_minus_h = 1.0 - h
            corr = np.correlate(shifted_h, ones_minus_h, mode='full')
            return np.max(corr) * dx
        
        # Equality constraint: ∫h(x)dx = 1
        def constraint_func(h_values):
            return np.sum(h_values) * (2.0 / len(h_values)) - 1.0
        
        # Bounds: 0 ≤ h(x) ≤ 1
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Define the optimization problem
        # We use a local optimizer for simplicity and speed
        res = opt.minimize(
            evaluate_c5,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=None,
            options={'ftol': 1e-8, 'maxiter': 500}
        )
        
        if res.success:
            current_c5 = res.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = res.x
                print(f"Found better solution with n_points={n_points}, C5={current_c5}")
        
        # Early exit if budget is exceeded
        if time.time() - start_time >= budget_s:
            break
    
    # Return the best solution found
    return best_h, best_c5, n_points
