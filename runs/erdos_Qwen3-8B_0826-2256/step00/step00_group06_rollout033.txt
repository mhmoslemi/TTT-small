```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Initial guess
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Create initial h_values with n_points samples
    h = np.zeros(n_points)
    h[:(n_points // 2)] = 1.0  # Start with a simple step function
    h = np.repeat(h, 2)  # Double the number of samples for better resolution (optional)
    h = h[:n_points]  # Keep only n_points samples
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h) * (2.0 / n_points)
    h = h * (1.0 / total_integral) * (n_points / 2.0)
    
    # Evaluate initial C5
    c5_bound = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Try increasing number of points
    for n_points in [200, 300, 400, 500]:  # Try different resolutions
        # Create a new h with n_points samples
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5  # Normalize to [0, 1] for better initialization
        h = h * (n_points / 2.0) / np.sum(h)  # Satisfy the integral constraint
        
        # Evaluate C5
        c5_bound = evaluate_erdos_solution(h)
        if c5_bound < best_c5:
            best_h = h.copy()
            best_c5 = c5_bound
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Use convex optimization to find a better solution
    # Reformulate as a convex optimization problem
    h = cp.Variable(n_points)
    constraints = [0 <= h, h <= 1.0, cp.sum(h) == n_points / 2.0]
    obj = cp.Minimize(cp.max(cp.sum(cp.multiply(h, 1 - cp.roll(h, 1, axis=0)) * (2.0 / n_points))))
    
    try:
        prob = cp.Problem(obj, constraints)
        prob.solve(solver='CVXOPT', verbose=False)
        
        if prob.status == 'optimal':
            h_values = np.round(h.value, 5)
            c5_bound = evaluate_erdos_solution(h_values)
            if c5_bound < best_c5:
                best_h = h_values
                best_c5 = c5_bound
    except:
        pass
    
    # Additional refinement using gradient descent
    h = best_h.copy()
    learning_rate = 0.01
    for _ in range(100):
        # Compute gradient and update
        # This is a simplified approach, more sophisticated methods can be used
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                if k != i:
                    grad[i] += (1 - h[k]) * (2.0 / n_points)
        grad -= np.zeros(n_points)
        h -= learning_rate * grad
        h = np.clip(h, 0.0, 1.0)
        h = h * (n_points / 2.0) / np.sum(h)
        c5_bound = evaluate_erdos_solution(h)
        if c5_bound < best_c5:
            best_h = h.copy()
            best_c5 = c5_bound
    
    # Final check for time constraint
    if time.time() - start_time > budget_s - 1:
        pass
    
    return (best_h, best_c5, n_points)
```
