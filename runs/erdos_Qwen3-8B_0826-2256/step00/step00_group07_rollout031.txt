def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define number of points
    n_points = 100  # Start with 100 points for faster optimization
    
    # Initial guess - start with the initial construction and perturb it
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Define constraints
    def constraints(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Define the optimization problem using scipy.optimize
    def optimize_h(h_initial):
        # Define the bounds for each variable
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Define the objective function
        def objective_func(x):
            return objective(x)
        
        # Define the constraint
        def constraint_func(x):
            return constraints(x)
        
        # Set up the optimization problem
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Ensure the solution is valid
        h = result.x
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective(h)
    
    # Run optimization
    start_time = time.time()
    h_opt, c5_bound = optimize_h(h_initial)
    
    # Check if we have time left for further refinement
    elapsed_time = time.time() - start_time
    if elapsed_time < budget_s - 10:

        n_points_new = 200

    return h_opt, c5_bound, n_points
