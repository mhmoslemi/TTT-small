import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assuming initial construction is provided

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    c5_bound = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = c5_bound

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        return evaluate_erdos_solution(h)

    # Function to perform a local optimization
    def optimize_local(h):
        h = h.copy()
        n = len(h)
        dx = 2.0 / n
        x = np.linspace(0, 2, n)
        h = np.clip(h, 0, 1)
        # Local optimization using gradient descent
        for _ in range(100):
            overlap = evaluate_overlap(h)
            if overlap < best_c5:
                best_c5 = overlap
                best_h = h.copy()
            # Small perturbation to avoid getting stuck
            perturb = np.random.normal(0, 0.05, n)
            h += perturb
            h = np.clip(h, 0, 1)
        return h

    # Iterative improvement
    for _ in range(5):
        h = optimize_local(h)
        if time.time() - start_time > budget_s - 1:
            break

    # Try starting point with different n_points
    for n_point in [200, 300, 400]:
        if time.time() - start_time > budget_s - 1:
            break
        if n_point % 2 != 0:
            continue
        h = np.random.rand(n_point) * 2
        h = np.clip(h, 0, 1)
        h = h * (n_point / 2) / np.sum(h)
        current_c5 = evaluate_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

    # Final optimization
    h = best_h.copy()
    n = len(h)
    dx = 2.0 / n
    x = np.linspace(0, 2, n)
    
    # Local search with adaptive step size
    for _ in range(100):
        overlap = evaluate_overlap(h)
        if overlap < best_c5:
            best_c5 = overlap
            best_h = h.copy()
        # Perturb and clip
        perturb = np.random.normal(0, 0.02, n)
        h += perturb
        h = np.clip(h, 0, 1)
        h = h * (n / 2) / np.sum(h)
    
    # Evaluate the final result
    final_c5 = evaluate_overlap(best_h)
    return best_h, final_c5, len(best_h)
