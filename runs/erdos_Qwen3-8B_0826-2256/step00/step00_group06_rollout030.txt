```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize, Bounds
from numpy.random import RandomState

# Pre-imported functions and variables
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    h = initial_h_values[:n_points]
    n_points = len(h)
    dx = 2.0 / n_points
    sum_h = np.sum(h)
    if not np.isclose(sum_h, n_points / 2):
        h = np.array([0.5] * n_points)
    
    def objective(x):
        # x is the h values, shape (n_points,)
        x = np.clip(x, 0, 1)
        sum_h = np.sum(x)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf
        
        h = x
        # Compute the correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Take the maximum of the correlation
        max_corr = np.max(corr)
        # Multiply by dx to get the integral
        c5 = max_corr * dx
        return c5
    
    def evaluate(x):
        return objective(x), np.max(np.correlate(x, 1 - x, mode="full")) * dx
    
    def objective_with_gradient(x):
        x = np.clip(x, 0, 1)
        sum_h = np.sum(x)
        if not np.isclose(sum_h, n_points / 2):
            return np.inf, np.zeros_like(x)
        
        h = x
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        c5 = max_corr * dx
        return c5, np.zeros_like(x)
    
    # Initial guess
    x0 = h.copy()
    
    # Optimization settings
    bounds = Bounds(0, 1)
    method = 'L-BFGS-B'
    
    # Run optimization
    result = minimize(
        objective,
        x0,
        method=method,
        bounds=bounds,
        jac=None,
        options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
    )
    
    # Check time
    elapsed_time = time.time() - start_time
    if elapsed_time >= budget_s - 1:
        return (x0, evaluate(x0)[1], n_points)
    
    # Try more refined optimization
    for _ in range(3):
        result = minimize(
            objective,
            result.x,
            method=method,
            bounds=bounds,
            jac=None,
            options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s - 1:
            break
    
    best_x = result.x
    best_c5 = evaluate(best_x)[1]
    
    # Ensure the sum constraint is satisfied
    best_x = np.clip(best_x, 0, 1)
    sum_h = np.sum(best_x)
    if not np.isclose(sum_h, n_points / 2):
        best_x = np.array([0.5] * n_points)
    
    return (best_x, best_c5, n_points)
```
