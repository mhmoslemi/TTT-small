```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Optimized settings
    max_n_points = 500  # Reduced from 1000 for efficiency
    n_points = min(max_n_points, 500)
    dx = 2.0 / n_points
    h_values = np.random.rand(n_points) * 0.5  # Start with some randomness
    h_values = initial_h_values[:n_points] if len(initial_h_values) > n_points else initial_h_values  # Use initial guess if possible
    
    # Ensure the integral constraint is satisfied
    h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
    
    best_h = h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    best_time = 0
    
    # Optimization using a gradient-based method with constraints
    def objective_func(h):
        nonlocal best_c5
        h = np.clip(h, 0, 1)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5
    
    start_time = time.time()
    
    # Gradient descent optimization with constraints
    h = h_values.copy()
    for _ in range(100):  # Run for a limited number of iterations
        if time.time() - start_time > budget_s - 1:
            break
        # Compute current C5
        c5 = objective_func(h)
        
        # Compute gradients (approximate via finite differences)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            c5_plus = evaluate_erdos_solution(h_plus, dx)
            grad[i] = (c5_plus - c5) / 1e-5
        
        # Adjust h to reduce C5
        h = h - 0.01 * grad
        
        # Apply constraints
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) * (n_points / 2)
    
    # Refinement using CVXPY for better accuracy
    h = cp.Variable(n_points)
    objective = cp.Minimize(cp.max(cp.sum(cp.multiply(h, 1 - cp.roll(h, 1)), axis=0)))
    constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
    
    start_time = time.time()
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - 1)
        if prob.status == 'optimal':
            h_values = h.value
    except:
        pass
    
    h_values = np.clip(h_values, 0, 1)
    h_values = h_values / (np.sum(h_values) * dx) * (n_points / 2)
    
    # Final evaluation
    c5_bound = evaluate_erdos_solution(h_values, dx)
    return h_values, c5_bound, n_points
```
