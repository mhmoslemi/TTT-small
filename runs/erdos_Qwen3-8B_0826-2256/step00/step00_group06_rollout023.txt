import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import cvxpy as cp
import time
import multiprocessing as mp
from functools import partial

# Pre-imported functions
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for initial values
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))

def run(seed=42, budget_s=100, **kwargs):
    n_points = 100  # Start with 100 points for speed
    max_iterations = 100  # Limit iterations to stay within budget
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Convert initial_h_values to 100-point representation
    if len(initial_h_values) != n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    
    # Ensure the initial h satisfies the integral constraint
    initial_h = initial_h_values.copy()
    initial_h_sum = np.sum(initial_h) * (2.0 / n_points)
    if np.abs(initial_h_sum - 1.0) > 1e-6:
        # Rescale to meet the constraint
        initial_h = initial_h * (1.0 / initial_h_sum) * (2.0 / n_points)
    
    # Initial evaluation
    c5_initial = evaluate_erdos_solution(initial_h)
    if c5_initial < best_c5:
        best_c5 = c5_initial
        best_h = initial_h.copy()
    
    # Perform a series of local refinements
    for _ in range(max_iterations):
        # Define the optimization variables
        h = cp.Variable(n_points)
        # Define the constraints
        constraints = [
            cp.sum(h) == n_points / 2,  # Integral constraint
            0 <= h,
            h <= 1
        ]
        
        # Define the objective function: minimize the maximum overlap
        # Approximate the max overlap by computing all overlaps and taking the maximum
        def objective_func(h):
            h_val = h.value
            overlaps = []
            for k in range(-n_points + 1, n_points):
                if k < 0:
                    shifted = np.roll(h_val, k)
                else:
                    shifted = np.roll(h_val, k)
                overlap = np.sum(h_val * (1 - shifted)) * (2.0 / n_points)
                overlaps.append(overlap)
            return np.max(overlaps)
        
        # Use a local optimization method
        result = opt.minimize(objective_func, initial_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        h_opt = result.x
        
        # Ensure the constraint is respected
        if np.abs(np.sum(h_opt) * (2.0 / n_points) - 1.0) > 1e-6:
            h_opt = h_opt * (1.0 / np.sum(h_opt) * (2.0 / n_points))
        
        c5_current = evaluate_erdos_solution(h_opt)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_opt.copy()
        
        # Early stopping if budget is reached
        if time.time() - start_time > budget_s - 1:
            break
    
    # Refine the solution using a grid search over possible shifts
    for shift in range(-n_points + 1, n_points):
        shifted_h = np.roll(best_h, shift)
        # Ensure the shifted_h satisfies the constraint
        if np.abs(np.sum(shifted_h) * (2.0 / n_points) - 1.0) > 1e-6:
            shifted_h = shifted_h * (1.0 / np.sum(shifted_h) * (2.0 / n_points))
        c5_shifted = evaluate_erdos_solution(shifted_h)
        if c5_shifted < best_c5:
            best_c5 = c5_shifted
            best_h = shifted_h.copy()
    
    # Print the final result
    print(f"Best C₅ bound: {best_c5}")
    return best_h, best_c5, n_points
