```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported helper functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted = np.zeros_like(h)
    for k in range(1, n_points):
        shifted[k:] = h[:-k]
    corr = signal.correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def initial_h_values():
    return [0.5] * 45

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with initial guess
    initial_h = initial_h_values()
    n_points = len(initial_h)
    dx = 2.0 / n_points
    sum_h = sum(initial_h)
    if abs(sum_h - n_points / 2) > 1e-6:
        initial_h = [h * (n_points / 2) / sum_h for h in initial_h]
    
    def objective_function(h_values):
        h = np.array(h_values)
        norm = np.sum(h) * dx
        if abs(norm - 1) > 1e-6:
            return float('inf')
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    def objective_function_with_grad(h_values):
        h = np.array(h_values)
        norm = np.sum(h) * dx
        if abs(norm - 1) > 1e-6:
            return float('inf'), np.zeros_like(h)
        h = np.clip(h, 0, 1)
        corr = signal.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr) * dx
        return c5, np.zeros_like(h)
    
    # Try with different numbers of points
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = n_points
    
    for n_points in [45, 50, 60, 70, 80, 90, 100]:
        start_time = time.time()
        if n_points < len(best_h):
            continue
        
        h_initial = np.array(initial_h)
        h_initial = np.repeat(h_initial, n_points // len(initial_h))
        h_initial = h_initial[:n_points]
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Optimize using a local search
        result = optimize.minimize(
            objective_function,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            options={'maxiter': 100, 'ftol': 1e-8}
        )
        
        if result.success:
            current_h = result.x
            current_c5 = evaluate_erdos_solution(current_h)
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5
                best_n_points = n_points
        
        # If we are under time budget, we can also try some random perturbations
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return best_h, final_c5, best_n_points
```
