import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Define the max number of points based on the budget
    max_n_points = 500  # 500 points is a reasonable choice for 100s of computation
    
    # Try starting with the initial solution
    h_initial = initial_h_values[:max_n_points]
    h_initial = h_initial[:max_n_points]
    h_initial = h_initial / (2.0 / max_n_points)  # Normalize to ensure sum(h) * dx = 1
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        shift = np.arange(len(h))
        overlap = np.zeros(len(shift))
        for k in shift:
            overlap[k] = np.dot(h, 1 - np.roll(h, k)) * (2.0 / len(h))
        return np.max(overlap)
    
    # Function to optimize the function using a genetic algorithm
    def optimize_h(initial_guess, n_points, budget_s):
        h = initial_guess.copy()
        h = h / (2.0 / n_points)  # Ensure sum(h) * dx = 1
        best_h = h.copy()
        best_c5 = compute_c5(h)
        
        # Genetic algorithm-like steps
        for _ in range(100):  # Number of iterations
            # Random perturbations
            perturbation = np.random.uniform(-0.2, 0.2, size=len(h))
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)  # Ensure values are in [0, 1]
            h_perturbed = h_perturbed / (2.0 / n_points)  # Ensure sum(h) * dx = 1
            
            # Evaluate
            c5_perturbed = compute_c5(h_perturbed)
            if c5_perturbed < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_perturbed
        
        return best_h, best_c5
    
    # Run the optimization
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    
    # Try different starting points
    for _ in range(5):  # Try multiple starting points for diversity
        # Generate random starting points
        n_points = min(max_n_points, len(best_h))
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand / (2.0 / n_points)  # Ensure sum(h) * dx = 1
        
        # Optimize
        h_optimized, c5_optimized = optimize_h(h_rand, n_points, budget_s)
        if c5_optimized < best_c5:
            best_h = h_optimized
            best_c5 = c5_optimized
        
        # Check time usage
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
    
    return best_h, best_c5, len(best_h)
