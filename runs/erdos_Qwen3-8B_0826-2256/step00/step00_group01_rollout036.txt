```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import OptimizeResult

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Assume initial_h_values.npy is available

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = len(initial_h_values)
    
    # Evaluate the initial solution as a starting point
    h_initial = initial_h_values.copy()
    c5_initial = evaluate_erdos_solution(h_initial)
    best_c5 = c5_initial
    best_h = h_initial
    
    # Try a few random perturbations around the initial solution
    for _ in range(5):
        # Perturb the initial solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = np.clip(h_initial + perturbation, 0, 1)
        # Ensure the integral constraint is satisfied
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = evaluate_erdos_solution(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed
    
    # Try a few random initial solutions with different n_points
    for n_test in [50, 75, 100]:
        h_test = np.random.uniform(0, 1, n_test)
        h_test = h_test * (n_test / 2) / np.sum(h_test)
        c5_test = evaluate_erdos_solution(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test
            n_points = n_test
    
    # Perform a local optimization around the best_h
    def objective(h_flat):
        h = h_flat.reshape(-1)
        h = h * (n_points / 2) / np.sum(h)
        return evaluate_erdos_solution(h)
    
    def constraint(h_flat):
        h = h_flat.reshape(-1)
        return np.sum(h) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    result = minimize(
        objective,
        x0=best_h,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=[{'type': 'eq', 'fun': constraint}],
        options={'ftol': 1e-6, 'maxiter': 100}
    )
    
    best_h = result.x
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Track time and return the best solution
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    # Try a few more random initial guesses
    for _ in range(3):
        h_test = np.random.uniform(0, 1, n_points)
        h_test = h_test * (n_points / 2) / np.sum(h_test)
        c5_test = evaluate_erdos_solution(h_test)
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h_test
    
    # Final check to ensure we're under time budget
    if time.time() - start_time > budget_s - 1:
        return (best_h, best_c5, n_points)
    
    return (best_h, best_c5, n_points)
```
