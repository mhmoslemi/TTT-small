```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h.npy')  # Assume initial construction is pre-loaded

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if needed
    
    # Try different numbers of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 5:
            break
        
        # Create initial guess from initial_h_values or random
        if n_points <= len(initial_h_values):
            h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h_initial = np.random.rand(n_points)
            h_initial = h_initial * (n_points / 2)  # Ensure sum(h) == n_points / 2
        
        # Define variables for optimization
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Objective: minimize the maximum overlap
        # Since we can't directly minimize the maximum, we use a minimax formulation
        # We'll use a Sequential Least Squares Programming (SLSQP) approach for faster evaluation
        def objective_func(h_vals):
            h_vals = np.array(h_vals)
            overlaps = np.correlate(h_vals, 1 - h_vals, mode="full")
            c5 = np.max(overlaps) * (2.0 / len(h_vals))
            return c5
        
        # Use a genetic algorithm for global search
        result = opt.differential_evolution(
            objective_func,
            bounds=[(0, 1) for _ in range(n_points)],
            popsize=15,
            mutation=(0.5, 1),
            recombination='blend',
            strategy='best1bin',
            tol=1e-5,
            maxiter=100,
            seed=seed,
            disp=False
        )
        
        h_vals = result.x
        c5 = objective_func(h_vals)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals
        
        # Early exit if we're under budget and we have a good solution
        if time.time() - start_time > budget_s - 5:
            break
    
    return (best_h, best_c5, len(best_h))
```
