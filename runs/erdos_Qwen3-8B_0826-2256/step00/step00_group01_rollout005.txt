    dx = 2.0 / n_points
    # Objective function: minimize max overlap integral
    def objective(h_values):
        h = h_values
        # Compute all shifted versions of (1 - h)
        shifted_h = np.zeros((n_points, n_points))
        for i in range(n_points):
            shifted_h[i, :] = h[(i + 1):] if i + 1 < n_points else np.zeros(n_points - i - 1)

        overlap_integral = np.zeros(n_points)
        for i in range(n_points):
            overlap_integral[i] = np.dot(h, 1 - shifted_h[i, :]) * dx
        return np.max(overlap_integral)
    
    # Constraints: sum(h) == n_points / 2, and 0 <= h[i] <= 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Optimization problem
    h = cp.Variable(n_points, nonneg=True, integer=False)
    objective_func = objective(h)
    constraints = [constraint(h) == 0, 0 <= h, h <= 1]
    
    # Try a few different numbers of points to find the best
    best_h = h_initial
    best_c5 = objective(best_h)
    best_n_points = n_points
    
    # Try with smaller number of points
    for n in [200, 300, 400]:
        dx = 2.0 / n
        h = cp.Variable(n, nonneg=True, integer=False)
        # Compute all shifted versions of (1 - h)
        shifted_h = np.zeros((n, n))
        for i in range(n):
            shifted_h[i, :] = h[(i + 1):] if i + 1 < n else np.zeros(n - i - 1)
        # Compute the overlap integrals using correlation
        overlap_integral = np.zeros(n)
        for i in range(n):
            overlap_integral[i] = np.dot(h, 1 - shifted_h[i, :]) * dx
        objective_func = cp.Max(overlap_integral)
        constraints = [cp.sum(h) == n / 2, 0 <= h, h <= 1]
        prob = cp.Problem(cp.Minimize(objective_func), constraints)
        start_time = time.time()
        try:
            result = prob.solve(solver=cp.SCS, max_iter=100, time_limit=budget_s - (time.time() - start_time))

            pass
        if time.time() - start_time > budget_s:
