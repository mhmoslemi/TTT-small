```python
import numpy as np
import scipy.optimize as opt
import time
import copy
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set n_points
    n_points = 300  # Adjusted for efficiency and accuracy
    dx = 2.0 / n_points
    num_k = int(2.0 / dx) + 1  # Number of shifts to consider
    
    # Initial guess
    h_initial = initial_h_values.copy()
    h_initial = np.resize(h_initial, (n_points,))  # Ensure correct length
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2)  # Ensure the integral equals 1

    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h_shifted = np.zeros((num_k, n_points))
        for k in range(num_k):
            h_shifted[k, :] = np.roll(h, k)
        overlap = np.zeros(num_k)
        for k in range(num_k):
            overlap[k] = np.sum(h * (1 - h_shifted[k, :])) * dx
        return np.max(overlap)
    
    # Define the optimization problem
    h = cp.Variable(n_points)
    constraints = [
        cp.sum(h) == n_points / 2,
        0 <= h,
        h <= 1
    ]
    
    # Function to compute the maximum overlap
    def objective_function(h_val):
        h_val = np.clip(h_val, 0, 1)
        h_shifted = np.zeros((num_k, n_points))
        for k in range(num_k):
            h_shifted[k, :] = np.roll(h_val, k)
        overlap = np.zeros(num_k)
        for k in range(num_k):
            overlap[k] = np.sum(h_val * (1 - h_shifted[k, :])) * dx
        return np.max(overlap)
    
    # Optimization loop
    best_h = h_initial
    best_c5 = compute_c5(best_h)
    elapsed = 0
    
    # Run for a limited time
    while elapsed < budget_s - 1:
        # Setup optimization problem
        obj = cp.Minimize(objective_function(h))
        prob = cp.Problem(obj, constraints)
        
        try:
            result = prob.solve(solver=cp.SCS, eps=1e-4, verbose=False)
            if result is not None:
                h_opt = np.round(h.value, 5)
                h_opt = np.clip(h_opt, 0, 1)
                current_c5 = compute_c5(h_opt)
                if current_c5 < best_c5:
                    best_h = h_opt
                    best_c5 = current_c5
        except:
            pass
        
        elapsed = time.time() - start_time
    
    # Check if we can improve with a local search
    for _ in range(10):
        h_perturbed = best_h + np.random.normal(0, 0.05, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        current_c5 = compute_c5(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
```
