import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Parameters
    max_n_points = 1000  # Max number of points to consider
    n_points = 45  # Start with the initial n_points
    initial_h = initial_h_values.copy()
    best_c5 = float('inf')
    best_h = initial_h
    start_time = time.time()
    
    # Function to compute C5
    def compute_c5(h):
        n = len(h)
        dx = 2.0 / n
        h = np.array(h)
        h_shifted = np.zeros_like(h)
        for k in range(1, n):
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            if overlap > best_c5:
                best_c5 = overlap
        return best_c5
    
    # Function to compute cost for optimization
    def objective(h):
        return compute_c5(h)
    
    # Function to evaluate and update best solution
    def update_best(h):
        nonlocal best_c5, best_h
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Function to optimize with scipy
    def optimize_with_scipy(h_initial):
        h = cp.Variable(len(h_initial))
        constraints = [0 <= h, h <= 1, cp.sum(h) == len(h_initial) / 2]
        objective = cp.Minimize(compute_c5(h))
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS, max_iters=1000, time_limit=10)
        if result is not None and prob.status == 'optimal':
            return h.value
        else:
            return h_initial
    
    # Function to optimize with genetic algorithm
    def optimize_with_ga(h_initial):
        def fitness(individual):
            return -compute_c5(individual)
        
        bounds = [(0, 1) for _ in range(len(h_initial))]
        result = opt.differential_evolution(fitness, bounds, popsize=15, maxiter=50, tol=1e-4, mutation=(0.5, 1), recombination='blend', seed=seed)
        return result.x
    
    # Function to refine the solution
    def refine_solution(h_initial, n_refinement=5):
        h = h_initial.copy()
        for _ in range(n_refinement):
            h = optimize_with_scipy(h)
            h = np.clip(h, 0, 1)
            h = np.round(h, 5)
            update_best(h)
        return h
    
    # Try with different n_points
    for n_points in range(45, max_n_points, 30):
        h_initial = np.zeros(n_points)
        h_initial[:len(initial_h)] = initial_h
        h_initial = np.random.permutation(h_initial)
        h_initial = np.random.rand(n_points)
        h_initial = np.round(h_initial, 5)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = np.array([h_initial[i] for i in range(n_points)])
        h_initial = np.array([h_initial[i] for i in range(n_points)])
        h_initial = np.round(h_initial, 5)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = np.array([h_initial[i] for i in range(n_points)])
        
        # Optimize
        h = refine_solution(h_initial)
        update_best(h)
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break
    
    # Final update
    update_best(best_h)
    
    return best_h, best_c5, len(best_h)
