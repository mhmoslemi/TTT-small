```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import minimize
from scipy.optimize import Bounds
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    dx = 2.0 / n_points
    h = np.random.uniform(0, 1, n_points)
    h = h * (n_points / 2) / np.sum(h)  # Ensure the integral constraint is satisfied
    
    def evaluate_c5(h):
        h = np.array(h)
        h = np.concatenate((h, h[:n_points//2]))  # Extend for full convolution
        shifted = np.concatenate((np.zeros(n_points), h[:-n_points]))
        result = np.correlate(h, 1 - shifted, mode="full") * dx
        return np.max(result)

    def objective(h):
        return evaluate_c5(h)
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2

    bounds = Bounds(0, 1)
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

    start_time = time.time()
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
    h = result.x
    c5_bound = evaluate_c5(h)
    elapsed = time.time() - start_time

    if elapsed < budget_s:
        # Try with more points if time allows
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=constraint, tol=1e-6)
        h = result.x
        c5_bound = evaluate_c5(h)

    return (h.tolist(), c5_bound, n_points)
```
