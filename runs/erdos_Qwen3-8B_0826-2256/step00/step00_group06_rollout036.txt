# Strategy:
# 1. Start from the initial_h_values as a baseline.
# 2. Use gradient-based optimization with bounds to find a better C₅ value.
# 3. Use a cell structure to approximate the step function and minimize the overlap.
# 4. Perform a fine-tuning step to improve the results further.
# 5. Return the best solution found within time constraints.

def compute_c5(h, n_points):
    dx = 2.0 / n_points
    shifted_h = np.roll(h, 1)
    correlation = np.correlate(h, shifted_h, mode="full")
    c5 = np.max(correlation * dx)
    return c5

        def objective(x):

            sum_h = np.sum(x)
            if np.abs(sum_h - n_points / 2) > 1e-6:
