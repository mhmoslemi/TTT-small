def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 120, 140, 160, 180, 200, 220, 240, 260, 280, 300]
    best_c5 = float('inf')
    best_h = None
    best_n = 0
    start_time = time.time()

    def optimize_h(n_points):
        dx = 2.0 / n_points
        h = (initial_h_values[:n_points] if n_points <= len(initial_h_values) else np.random.uniform(0, 1, n_points))
        h = np.clip(h, 0, 1)
        total = h.sum() * dx
        h = h - (total - 1.0) / n_points
        # Random perturbation
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h - (h.sum() * dx - 1.0) / n_points
        h = np.clip(h, 0, 1)

        def objective(x):
            nonlocal h, dx
            h = x
            total = h.sum() * dx
            h = h - (total - 1.0) / n_points
            h = np.clip(h, 0, 1)
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        # Optimize using L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=None, options={'maxiter': 100})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt - (h_opt.sum() * dx - 1.0) / n_points
        h_opt = np.clip(h_opt, 0, 1)
        c5 = evaluate_erdos_solution(h_opt, n_points)
        return h_opt, c5

    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        print(f"Trying n_points={n_points}")
        h_opt, c5 = optimize_h(n_points)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n = n_points

    return best_h, best_c5, best_n
