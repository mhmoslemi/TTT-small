import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import copy
from scipy import sparse
from scipy.sparse import lil_matrix
from scipy.optimize import Bounds, LinearConstraint
import cvxpy as cp

# Pre-imported functions
initial_h_values = np.random.rand(42) * 2 - 1  # Placeholder for initial H values
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (42 / 2) / np.sum(initial_h_values)  # Normalize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Initial attempt with the provided initial_h_values
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    h = h * (n_points / 2) / np.sum(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
        best_n_points = n_points

    # Attempt with smaller n_points (e.g., 60)
    for n_points in [60, 80, 100, 120, 140, 160, 180, 200]:
        if time.time() - start_time > budget_s - 10:
            break
        # Generate initial guess with uniform distribution
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points
        # Refine with local optimization
        result = optimize.minimize(
            fun=compute_c5,
            x0=h,
            method='L-BFGS-B',
            bounds=Bounds(0, 1),
            options={'ftol': 1e-6, 'gtol': 1e-6, 'maxiter': 100}
        )
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
                best_n_points = n_points
        # Additional refinement
        if time.time() - start_time > budget_s - 10:
            break

    # For higher precision, use a more sophisticated optimization method
    if best_n_points < 200:
        n_points = 200
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points
        result = optimize.minimize(
            fun=compute_c5,
            x0=h,
            method='SLSQP',
            bounds=Bounds(0, 1),
            constraints=[LinearConstraint(np.ones(n_points), lb=n_points/2, ub=n_points/2)],
            options={'ftol': 1e-6, 'maxiter': 100}
        )
        if result.success:
            c5 = result.fun
            if c5 < best_c5:
                best_c5 = c5
                best_h = result.x
                best_n_points = n_points

    # Final check with a small random perturbation
    if time.time() - start_time < budget_s - 10:
        h_perturbed = best_h + np.random.normal(0, 1e-3, best_n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (best_n_points / 2) / np.sum(h_perturbed)
        c5_perturbed = compute_c5(h_perturbed)
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_perturbed
            best_n_points = len(h_perturbed)

    return (best_h, best_c5, best_n_points)

def compute_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    h = np.concatenate((h, h[:2]))  # To handle wrap-around properly
    h_shifted = np.roll(h, 1)
    h_shifted = h_shifted[:n_points + 2]
    h_1_minus_h = 1 - h_shifted[1:]
    conv = signal.correlate(h, h_1_minus_h, mode='full')
    c5 = np.max(conv * dx)
    return c5
