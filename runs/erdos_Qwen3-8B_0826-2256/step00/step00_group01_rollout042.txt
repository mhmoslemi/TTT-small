import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points for efficiency
    max_points = 200  # Max points to try

    # Initial guess
    h = initial_h_values.copy()
    h = h[:n_points]  # Truncate to n_points
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2)  # Ensure sum(h) == n_points / 2

    # Function to compute C5 for a given h
    def compute_c5(h_values):
        h = h_values
        dx = 2.0 / len(h)
        # Compute cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to compute C5 and evaluate for a given h
    def evaluate_solution(h_values):
        nonlocal best_c5, best_h
        h = h_values
        # Ensure sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        return c5

    # Optimization loop
    for n_points in range(100, max_points + 1, 50):  # Try increasing point counts
        # Start with initial guess
        h = initial_h_values.copy()
        h = h[:n_points]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Use L-BFGS-B for optimization
        def objective(x):
            h_values = x
            h_values = np.clip(h_values, 0, 1)
            h_values = h_values * (n_points / 2) / np.sum(h_values)
            return compute_c5(h_values)

        # Bounds: [0, 1] for each value
        bounds = [(0, 1) for _ in range(n_points)]

        # Run optimization
        result = optimize.minimize(
            objective, h, bounds=bounds, method='L-BFGS-B', jac=None, tol=1e-5,
            options={'maxiter': 100, 'ftol': 1e-5, 'gtol': 1e-5}
        )

        # Update best solution
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            c5 = compute_c5(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        # Check time remaining
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

    # Final evaluation
    evaluate_solution(best_h)
    elapsed = time.time() - start_time
    remaining = budget_s - elapsed
    if remaining > 0:
        # Optionally run a final optimization with more points if time remains
        pass

    return best_h, best_c5, len(best_h)
