```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import os

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with the initial construction
    n_points = len(initial_h_values)
    h_values = np.array(initial_h_values, dtype=float)
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        dx = 2.0 / len(h)
        h = np.concatenate((h, h[:2]))  # Extend for full convolution
        result = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(result)
    
    # Function to compute the objective (C5)
    def objective(h):
        return evaluate_c5(h)
    
    # Function to compute the constraints (integral equals 1)
    def constraint(h):
        return np.sum(h) - len(h) / 2  # sum(h) == len(h)/2
    
    # Initial evaluation
    c5_bound = evaluate_c5(h_values)
    best_h_values = h_values.copy()
    best_c5_bound = c5_bound
    
    # Optimization loop
    for i in range(100):  # Run multiple optimization steps
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use L-BFGS-B optimizer
        result = optimize.minimize(
            objective,
            h_values,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(len(h_values))],
            constraints=[{'type': 'eq', 'fun': constraint}]
        )
        
        # Update best solution
        h_values = result.x
        current_c5 = evaluate_c5(h_values)
        if current_c5 < best_c5_bound:
            best_h_values = h_values.copy()
            best_c5_bound = current_c5
        
        # Early stopping if we have a very good solution
        if best_c5_bound <= 0.380876:
            break
        
        # Random small perturbations for diversity
        perturbation = np.random.uniform(-0.05, 0.05, size=len(h_values))
        h_values = np.clip(h_values + perturbation, 0, 1)
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h_values)
    return best_h_values, final_c5, len(best_h_values)
```
