import numpy as np
import scipy.optimize as optimize
import cvxpy as(cp)
import time

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 80
    h = initial_h_values.copy()
    
    # Function to compute C5
    def compute_c5(h_values):
        h_array = np.array(h_values)
        dx = 2.0 / n_points
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        c5 = np.max(corr) * dx
        return c5
    
    # Function to compute overlap integral for all shifts
    def compute_all_overlaps(h_values):
        h_array = np.array(h_values)
        dx = 2.0 / n_points
        corr = np.correlate(h_array, 1 - h_array, mode="full")
        return corr * dx
    
    # Optimization function
    def optimize_h(initial_h, n_points):
        h = cp.Variable(n_points)
        # Objective function: minimize the maximum overlap
        # We minimize the maximum value of all overlaps
        overlaps = cp.sum(cp.multiply(h, 1 - h[cp.reshape(cp.arange(n_points) + cp.reshape(cp.arange(n_points)), (n_points, n_points)) % n_points]))
        max_overlap = cp.Max(overlaps)
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        # Optimization
        prob = cp.Problem(cp.Minimize(max_overlap), constraints)
        result = prob.solve(solver=cp.SCS)
        return h.value
    
    # Evaluate current solution
    current_c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Try optimizing with different initial points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Optimize with current h
        optimized_h = optimize_h(h, n_points)
        optimized_h = np.clip(optimized_h, 0, 1)
        # Compute C5
        c5 = compute_c5(optimized_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = optimized_h.copy()
        # Perturb slightly to avoid local minima
        perturbation = np.random.normal(0, 0.05, n_points)
        h = np.clip(optimized_h + perturbation, 0, 1)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return best_h, final_c5, n_points
