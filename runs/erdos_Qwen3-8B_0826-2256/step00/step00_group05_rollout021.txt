```python
import numpy as np
import scipy.optimize
import scipy.signal
import time
from functools import lru_cache

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Start from initial_h_values
    initial_n_points = len(initial_h_values)
    initial_h = initial_h_values.copy()
    dx = 2.0 / initial_n_points
    corr = scipy.signal.correlate(initial_h, 1 - initial_h, mode="full")
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = initial_h
        best_n_points = initial_n_points

    # Try a few different n_points
    for n_points in [100, 200, 300]:
        if time.time() - start_time > budget_s - 1:
            break

        # Ensure n_points is even
        if n_points % 2 != 0:
            n_points += 1

        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points, endpoint=False)

        # Initial guess as a uniform distribution
        h = np.ones(n_points) / 2

        # Define the objective function
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            corr = scipy.signal.correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        # Define the constraint: sum(h) == n_points / 2
        def constraint(h_vec):
            return np.sum(h_vec) - n_points / 2

        # Setup optimization problem
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint, 'jac': lambda h_vec: np.ones(n_points)}

        # Run optimization
        result = scipy.optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons, tol=1e-6)

        if result.success:
            h_opt = result.x
            corr = scipy.signal.correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt
                best_n_points = n_points

    # Final evaluation
    evaluate_erdos_solution(best_h)
    return best_h, best_c5, best_n_points
```
