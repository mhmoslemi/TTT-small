import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial solutions
    initial_h = initial_h_values
    n_points_initial = len(initial_h)
    
    # Try different number of points to find a good balance between speed and accuracy
    n_points_candidates = [50, 100, 200, 300, 400]
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial
    
    start_time = time.time()
    
    for n_points in n_points_candidates:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess by resampling the initial solution
        h_initial = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_points_initial), initial_h)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial + np.random.normal(0, 0.05, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimization
        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Compute the overlap integral
            dx = 2.0 / n_points
            conv = np.correlate(h, 1 - h, mode="full") * dx
            c5 = np.max(conv)
            return c5
        
        def objective_with_grad(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            dx = 2.0 / n_points
            conv = np.correlate(h, 1 - h, mode="full") * dx
            c5 = np.max(conv)
            # Gradient approximation (finite differences)
            grad = np.zeros_like(h)
            for i in range(len(h)):
                h_plus = h.copy()
                h_plus[i] += 1e-5
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                conv_plus = np.correlate(h_plus, 1 - h_plus, mode="full") * dx
                c5_plus = np.max(conv_plus)
                grad[i] = (c5_plus - c5) / 1e-5
            return c5, grad
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = optimize.minimize(objective, h_initial, method='L-BFGS-B', bounds=bounds, jac=lambda x: objective_with_grad(x)[1], options={'maxiter': 100})
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        # Check if we found a better solution
        conv = np.correlate(h_opt, 1 - h_opt, mode="full") * (2.0 / n_points)
        c5 = np.max(conv)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
            best_n_points = n_points
    
    return (best_h, best_c5, best_n_points)
