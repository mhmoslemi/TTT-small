    def evaluate_c5(h_values):
        h = np.array(h_values)
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        return np.max(conv) * dx

    def optimize_h(current_h, n_points, max_iter=100, learning_rate=0.01):
        h = np.array(current_h)

            # Compute the current C5
            c5 = evaluate_c5(h)
            
            # Compute gradients
            grads = []

                grad_i = 0

                    if i - k >= 0 and i - k < n_points:
                        grad_i += (1 - h[k]) * (1 if h[i] > 0 else 0)
                    if i + k < n_points:
                        grad_i += (1 - h[k]) * (1 if h[i] > 0 else 0)
                grads.append(grad_i)
            
            # Update h with gradient descent
            h = h - learning_rate * np.array(grads)
            
            # Ensure h stays in [0, 1]
            h = np.clip(h, 0, 1)
            
            # Normalize h to maintain the integral constraint
            h = h * (n_points / 2) / np.sum(h)
            
            if np.max(np.abs(grads)) < 1e-6:

    h = optimize_h(h_initial, n_points)

    # Try increasing n_points if time allows
    for n_points in [150, 200, 250]:

        h = np.array(best_h)

            best_h = h.copy()
    
    # Final check with the best h
    best_c5 = evaluate_c5(best_h)
    
    # Return the best solution
    return (best_h.tolist(), best_c5, len(best_h))
