import numpy as np
import scipy.optimize as optimize
import time
import random
import cvxpy as cp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Initial number of points
    max_points = 300  # Maximum number of points we'll try
    max_iter = 100  # Maximum number of iterations
    
    # Initial guess from provided initial_h_values (if available)
    initial_h_values = globals().get('initial_h_values', None)
    if initial_h_values is not None:
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
        if n_points > max_points:
            n_points = max_points
            initial_h = initial_h[:n_points]
    else:
        initial_h = np.zeros(n_points)
        initial_h[0] = 0.5
        initial_h[1] = 0.5
        initial_h[2] = 0.5
        initial_h[3] = 0.5
        initial_h[4] = 0.5
    
    initial_h = initial_h / (n_points / 2)
    initial_h = np.clip(initial_h, 0, 1)
    
    best_c5 = np.inf
    best_h = initial_h.copy()
    best_n = n_points
    
    for n in range(n_points, max_points + 1):
        if time.time() - start_time > budget_s - 0.1:
            break
        
        dx = 2.0 / n
        x = np.linspace(0, 2, n)
        
        # Generate a random perturbation
        perturbation = np.random.rand(n) * 0.1 - 0.05
        h = initial_h.copy()
        h = np.clip(h + perturbation, 0, 1)
        h = h * (n / 2)  # Ensure integral is 1
        h = np.clip(h, 0, n / 2)
        
        # Define objective function
        def objective(h):
            h = h.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n / 2)
            h = np.clip(h, 0, n / 2)
            h = h / (n / 2)
            
            # Compute the overlap integral
            cross_corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(cross_corr) * dx
            return max_corr
        
        # Optimization using scipy.optimize
        result = optimize.minimize(
            objective, 
            h, 
            method='L-BFGS-B', 
            bounds=[(0, 1) for _ in range(n)]
        )
        
        if result.success:
            current_h = result.x
            current_c5 = objective(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_n = n
        
        # Update initial guess for next iteration
        initial_h = result.x
    
    # After optimization, finalize the best solution
    dx = 2.0 / best_n
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (best_n / 2)
    best_h = np.clip(best_h, 0, best_n / 2)
    best_h = best_h / (best_n / 2)
    
    return best_h, best_c5, best_n
