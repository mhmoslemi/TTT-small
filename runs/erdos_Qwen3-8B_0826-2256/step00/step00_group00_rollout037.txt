def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    h_start = initial_h_values.copy()
    n_points = len(h_start)
    
    # Function to compute the overlap integral
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / n_points)
    
    # Function to compute the objective and constraints for optimization
    def objective_func(h):
        return compute_c5(h)
    
    # Perform optimization to minimize the maximum overlap
    # We use a gradient-based method with bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    result = optimize.minimize(
        objective_func,
        h_start,
        method='L-BFGS-B',
        bounds=bounds,
        options={'ftol': 1e-6, 'maxiter': 1000}
    )
    
    best_h = result.x
    best_c5 = compute_c5(best_h)
    
    # Optionally, perform a few more iterations of local search with random perturbations
    for _ in range(5):

        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        current_c5 = compute_c5(perturbed_h)
        if current_c5 < best_c5:
            best_h = perturbed_h
            best_c5 = current_c5
