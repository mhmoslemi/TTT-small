import numpy as np

# Pre-imported functions
def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.concatenate([h[-1:], h[:-1]])
    overlap = np.correlate(h, 1 - shifted_h, mode="full")
    return np.max(overlap) * dx

def initial_h_values():
    return [0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5]

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # Start with 100 points, adjust if needed
    dx = 2.0 / n_points
    initial_h = initial_h_values()[:n_points]
    
    # Objective: minimize the maximum overlap
    # Constraints:
    # 1. h[i] ∈ [0, 1]
    # 2. sum(h) * dx = 1 => sum(h) = n_points / 2
    
    h = cp.Variable(n_points, lb=0, ub=1)
    constraint = cp.sum(h) == n_points / 2
    
    # Objective: minimize max_k (integral of h(x)*(1 - h(x + k)) dx)
    # We approximate this using the correlation approach
    
    def objective_func(h_vals):
        h = np.array(h_vals)
        shifted_h = np.concatenate([h[-1:], h[:-1]])
        overlap = np.correlate(h, 1 - shifted_h, mode="full")
        return np.max(overlap) * dx
    
    # We will perform a gradient descent approach with heuristic adjustments
    # Start with initial guess
    h_vals = np.array(initial_h)
    
    # Optimization parameters
    learning_rate = 0.1
    max_iterations = 1000
    best_c5 = float('inf')
    best_h = initial_h
    
    for _ in range(max_iterations):
        # Compute current C5
        current_c5 = objective_func(h_vals)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_vals.copy()
        
        # Compute gradients using finite differences (approximate)
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_vals.copy()
            h_plus[i] += 1e-5
            c5_plus = objective_func(h_plus)
            
            h_minus = h_vals.copy()
            h_minus[i] -= 1e-5
            c5_minus = objective_func(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * 1e-5)
        
        # Update h_vals
        h_vals -= learning_rate * grad
        
        # Ensure h_vals stays within [0, 1] and satisfies the sum constraint
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals - np.mean(h_vals) + (n_points / 2) / n_points
        
        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break
    
    # Evaluate the best solution found
    final_c5 = objective_func(best_h)
    return best_h, final_c5, n_points
