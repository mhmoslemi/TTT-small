```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # Example initial values (replace with actual path)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start timing
    start_time = time.time()
    
    # Initial parameters
    n_points = 100  # Start with 100 points, can increase if needed
    dx = 2.0 / n_points
    
    # Initial guess based on the provided initial_h_values
    if len(initial_h_values) == n_points:
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0, 1]
    
    # Normalize to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = np.copy(h)
    
    # Optimization parameters
    iterations = 50
    max_time = budget_s - 0.1  # Leave buffer for final evaluation
    
    # Define objective and constraints
    def objective(x):
        h = x.reshape((n_points,))
        c5 = evaluate_erdos_solution(h)
        return c5
    
    def constraints(x):
        h = x.reshape((n_points,))
        return [np.sum(h) - (n_points / 2)]
    
    # Use L-BFGS-B for optimization
    result = optimize.minimize(
        objective,
        h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(n_points)],
        constraints=constraints,
        options={'disp': False, 'maxiter': iterations}
    )
    
    # Update best solution
    if result.success:
        h = result.x
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Try increasing number of points if time allows
    if time.time() - start_time < max_time - 10:
        n_points = 200
        dx = 2.0 / n_points
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0, 1]
        h = h * (n_points / 2) / np.sum(h)
        
        # Optimization with increased n_points
        result = optimize.minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints=constraints,
            options={'disp': False, 'maxiter': iterations}
        )
        
        if result.success:
            h = result.x
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best solution found
    return (best_h, final_c5, len(best_h))
```
