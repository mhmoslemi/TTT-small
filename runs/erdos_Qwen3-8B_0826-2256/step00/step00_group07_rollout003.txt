import numpy as np
import scipy.optimize as optimize
import time
import itertools
import random
import cvxpy as cp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    # Try different numbers of points
    for n_points in [100, 150, 200, 250, 300]:
        if time.time() - start_time > budget_s - 10:
            break
        # Generate initial guess
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = np.random.uniform(0, 1)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
        
        # Define the objective function
        def objective(x):
            nonlocal best_c5
            h_vals = x.reshape(n_points)
            # Rescale to [0, 2]
            dx = 2.0 / n_points
            # Compute the convolution
            conv = np.correlate(h_vals, 1 - h_vals, mode="full")
            # Scale by dx
            conv_scaled = conv * dx
            # Find the maximum
            max_val = np.max(conv_scaled)
            if max_val < best_c5:
                best_c5 = max_val
                best_h = h_vals.copy()
                best_n_points = n_points
            return max_val

        # Use optimization to minimize the maximum overlap
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_n_points = n_points

        # Try some random perturbations
        for _ in range(5):
            if time.time() - start_time > budget_s - 10:
                break
            # Perturb the current best h
            perturbed_h = best_h.copy()
            for i in range(n_points):
                perturbed_h[i] += np.random.uniform(-0.05, 0.05)
                perturbed_h[i] = np.clip(perturbed_h[i], 0, 1)
            # Normalize
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            # Evaluate
            conv = np.correlate(perturbed_h, 1 - perturbed_h, mode="full")
            conv_scaled = conv * (2.0 / n_points)
            current_c5 = np.max(conv_scaled)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = perturbed_h.copy()
                best_n_points = n_points

    # Final check for the best found
    if best_h is not None:
        conv = np.correlate(best_h, 1 - best_h, mode="full")
        conv_scaled = conv * (2.0 / best_n_points)
        best_c5 = np.max(conv_scaled)
    return (best_h, best_c5, best_n_points)
