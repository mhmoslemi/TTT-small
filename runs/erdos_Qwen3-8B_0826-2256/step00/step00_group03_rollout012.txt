```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Placeholder for initial construction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Parametric settings
    max_n_points = 500  # Maximum number of points allowed
    n_points = 500  # Start with 500 points for good resolution
    dx = 2.0 / n_points
    max_iter = 100  # Max iterations for optimization
    tolerance = 1e-5  # Optimization tolerance
    
    # Initial guess
    h = np.copy(initial_h_values)
    if len(h) != n_points:
        h = np.zeros(n_points)
        h[:len(initial_h_values)] = initial_h_values
        h = np.pad(h, (0, n_points - len(h)), 'constant')
    
    # Ensure the integral constraint is satisfied
    total_integral = np.sum(h) * dx
    if np.abs(total_integral - 1.0) > 1e-6:
        # Normalize h
        h = h / (total_integral / 1.0)
    
    # Initialize best solution
    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the optimization objective
    def objective_func(h_vec):
        # Convert to array
        h = h_vec.reshape(-1)
        # Compute the integral constraint
        total_integral = np.sum(h) * dx
        if np.abs(total_integral - 1.0) > 1e-6:
            return np.inf  # Penalize for violating constraint
        # Compute the overlap integral
        overlap = signal.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Define the constraints
    def constraints_func(h_vec):
        h = h_vec.reshape(-1)
        total_integral = np.sum(h) * dx
        if np.abs(total_integral - 1.0) > 1e-6:
            return np.inf  # Penalize for violating constraint
        return np.array([])  # No other constraints
    
    # Define a function to optimize
    def optimize_h(h_start):
        h_vec = h_start.copy()
        for _ in range(max_iter):
            # Define variables
            h = cp.Variable(n_points)
            # Objective: minimize the maximum overlap
            overlap = signal.correlate(h.value, 1 - h.value, mode="full") * dx
            c5 = cp.Max(overlap)
            # Constraint: total integral is 1
            constraint = cp.sum(h) * dx == 1.0
            # Solve the problem
            prob = cp.Problem(cp.Minimize(c5), [constraint])
            try:
                prob.solve(solver=cp.SCS, eps=tolerance, max_iter=100)
                if prob.status == 'optimal':
                    h_vec = h.value
                    current_c5 = evaluate_erdos_solution(h_vec)
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = h_vec.copy()
                    break
            except:
                pass
        return best_h, best_c5
    
    # Run optimization
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Optimize with current h
        best_h, best_c5 = optimize_h(best_h)
        # Try reducing n_points if we are under budget
        if time.time() - start_time < budget_s * 0.8:
            n_points = int(n_points * 0.8)
            dx = 2.0 / n_points
            h = np.zeros(n_points)
            h[:len(best_h)] = best_h[:n_points]
            h = h / (np.sum(h) * dx / 1.0)
        else:
            break
    
    return best_h, best_c5, n_points
```
