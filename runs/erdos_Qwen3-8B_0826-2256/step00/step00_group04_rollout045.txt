import numpy as np
import scipy.optimize as optimize
import copy
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 256  # A reasonable starting point that's less than 1k
    dx = 2.0 / n_points
    initial_h = copy.deepcopy(initial_h_values)
    
    # Ensure the initial h is scaled to meet the integral constraint
    sum_h = np.sum(initial_h)
    scale_factor = (n_points / 2) / sum_h
    initial_h = np.array(initial_h) * scale_factor
    
    def evaluate_c5(h):
        h = h.reshape(-1)
        h = np.pad(h, (0, 1), mode='constant')  # Pad to handle full convolution
        kernel = 1 - h
        conv = np.correlate(h, kernel, mode="full")
        max_overlap = np.max(conv) * dx
        return max_overlap
    
    def objective(h):
        return evaluate_c5(h)
    
    # Optimization with L-BFGS-B
    def optimize_h(h_initial):
        result = optimize.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            jac=None,
            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6}
        )
        return result.x
    
    start_time = time.time()
    best_h = initial_h
    best_c5 = evaluate_c5(best_h)
    
    # Try multiple starting points
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate a random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_initial = initial_h + perturbation
        h_initial = np.clip(h_initial, 0, 1)
        
        # Ensure the integral constraint is met
        sum_h = np.sum(h_initial)
        scale_factor = (n_points / 2) / sum_h
        h_initial = h_initial * scale_factor
        
        h_opt = optimize_h(h_initial)
        current_c5 = evaluate_c5(h_opt)
        
        if current_c5 < best_c5:
            best_h = h_opt
            best_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5, n_points
