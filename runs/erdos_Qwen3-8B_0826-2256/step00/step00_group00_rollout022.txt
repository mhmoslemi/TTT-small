def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros_like(h)
    for k in range(1, n_points):
        h_shifted[k:] = h[:-k]
    overlap = np.correlate(h, 1 - h, mode="full") * dx
    c5_bound = np.max(overlap)
    return c5_bound

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    n_points = 400  # Start with a moderate number of points to balance speed and accuracy
    h = np.copy(initial_h_values[:n_points])
    h = np.concatenate([h, np.zeros(n_points - len(h))])  # Ensure length is n_points

    # Normalize to satisfy the integral constraint
    h = h / np.sum(h) * (n_points / 2)

    # Function to compute C5
    def compute_c5(h_vals):
        h = np.array(h_vals)
        n = len(h)
        dx = 2.0 / n
        h_shifted = np.zeros_like(h)
        for k in range(1, n):
            h_shifted[k:] = h[:-k]
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)

    # Optimization objective: minimize C5
    def objective(h_vals):
        return compute_c5(h_vals)

    # Bounds: h[i] in [0, 1]
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Use L-BFGS-B for faster convergence with bounds
    result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=bounds, tol=1e-8)
