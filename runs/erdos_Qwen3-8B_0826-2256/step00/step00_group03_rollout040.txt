```python
import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Initial construction provided

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # Start with 100 points, can increase if needed

    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        nonlocal n_points
        dx = 2.0 / n_points
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr)
        return max_corr * dx

    # Function to optimize a candidate h
    def optimize_h(h_initial):
        nonlocal n_points
        dx = 2.0 / n_points
        bounds = Bounds(0, 1)
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
        ]
        result = minimize(
            lambda x: evaluate_overlap(x),
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 100, 'ftol': 1e-6}
        )
        if result.success:
            return result.x
        else:
            return h_initial

    # Try starting from the initial construction
    h_initial = initial_h_values[:n_points]
    h_candidate = optimize_h(h_initial)
    current_c5 = evaluate_overlap(h_candidate)
    best_c5 = current_c5
    best_h = h_candidate

    # Try different starting points by random perturbations
    for _ in range(5):
        perturbed_h = h_candidate + 0.1 * np.random.rand(n_points) - 0.05
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        h_candidate = optimize_h(perturbed_h)
        current_c5 = evaluate_overlap(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate

    # Try increasing the number of points if time allows
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        n_points = 150
        dx = 2.0 / n_points
        h_initial = np.random.rand(n_points)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        h_candidate = optimize_h(h_initial)
        current_c5 = evaluate_overlap(h_candidate)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_candidate

    # Ensure the final h stays within [0, 1] and satisfies the integral constraint
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    # Evaluate the final result
    final_c5 = evaluate_overlap(best_h)

    return best_h, final_c5, n_points
```
