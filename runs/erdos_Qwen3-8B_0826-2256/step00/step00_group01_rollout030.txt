def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):
        shifted_h[i, :] = h[(i + n_points - 1): (i + n_points - 1 + n_points) % n_points]
    shifted_1_minus_h = 1 - shifted_h
    correlations = np.zeros(n_points)
    for i in range(n_points):
        corr = np.sum(h * shifted_1_minus_h[i, :]) * dx
        correlations[i] = corr
    return np.max(correlations)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    max_iterations = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial guess
    h = np.copy(initial_h_values[:n_points])
    h = np.clip(h, 0.0, 1.0)
    total_area = np.sum(h) * dx
    h = h * (1.0 / total_area) * (n_points / 2)  # Normalize to satisfy the area constraint
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Convert to a new numpy array
        h = np.clip(h, 0.0, 1.0)
        total_area = np.sum(h) * dx
        h = h * (1.0 / total_area) * (n_points / 2)  # Ensure the area constraint is satisfied
        
        # Evaluate C5
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Perturb the solution
        perturbation = np.random.normal(0, 0.05, n_points)
        h += perturbation
        h = np.clip(h, 0.0, 1.0)
        total_area = np.sum(h) * dx
        h = h * (1.0 / total_area) * (n_points / 2)  # Normalize to satisfy the area constraint
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
