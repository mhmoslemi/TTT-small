import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import convolve

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load('initial_h_values.npy')  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Define number of points and adjust based on budget
    n_points = 100  # Start with 100 points, can be increased if time allows
    if n_points * 2 > 1000:
        n_points = 500
    if n_points * 2 > 2000:
        n_points = 250
    if n_points * 2 > 3000:
        n_points = 200
    if n_points * 2 > 4000:
        n_points = 150
    if n_points * 2 > 5000:
        n_points = 100
    if n_points * 2 > 6000:
        n_points = 80
    if n_points * 2 > 7000:
        n_points = 60
    if n_points * 2 > 8000:
        n_points = 50
    if n_points * 2 > 9000:
        n_points = 40
    if n_points * 2 > 10000:
        n_points = 30
    
    # Start with initial values or random initialization
    h = initial_h_values.copy()
    if len(h) != n_points:
        h = np.random.rand(n_points)
        h = (h * 2.0 - 1.0) * 0.5 + 0.5  # Normalize to [0, 1] range
        # Ensure the integral constraint is satisfied
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    
    # Function to evaluate the C5 value
    def evaluate_C5(h_values):
        dx = 2.0 / len(h_values)
        shifted = np.zeros(len(h_values) * 2 - 1)
        for i in range(len(h_values)):
            shifted[i:i + len(h_values)] += h_values[i] * (1 - h_values)
        overlap = np.max(shifted * dx)
        return overlap
    
    # Optimization using gradient descent with adaptive learning rate
    max_iterations = 1000
    learning_rate = 0.1
    best_h = h.copy()
    best_c5 = evaluate_C5(h)
    
    for _ in range(max_iterations):
        if time.time() - start_time > budget_s:
            break
        
        # Compute current C5
        current_c5 = evaluate_C5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros(len(h))
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-5
            h_minus = h.copy()
            h_minus[i] -= 1e-5
            grad[i] = (evaluate_C5(h_plus) - evaluate_C5(h_minus)) / (2 * 1e-5)
        
        # Update h
        h -= learning_rate * grad
        
        # Enforce bounds and integral constraint
        h = np.clip(h, 0.0, 1.0)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
    
    # Final evaluation
    final_c5 = evaluate_C5(best_h)
    return best_h, final_c5, n_points
