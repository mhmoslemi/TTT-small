def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    conv = signal.correlate(h_values, 1 - h_values, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def initial_h_values():
    return np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

# Strategy: 
# 1. Start with the initial_h_values.
# 2. Use a gradient-based optimization method to find a function h that minimizes the maximum overlap.
# 3. Since the overlap is a non-convex function, we use an evolutionary algorithm with random restarts.
# 4. Use a combination of local and global optimization to prevent getting stuck in a local minimum.
# 5. Use a limited number of samples (n_points = 100) for faster computation.
# 6. Track elapsed time and return the best solution before the budget is exhausted.

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    initial_h = initial_h_values()
    initial_h = initial_h[:n_points]  # Ensure we are using only n_points samples

    # Try multiple random restarts
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    for _ in range(20):  # 20 restarts
        # Perturb the initial guess
        h = initial_h.copy()
        h += np.random.uniform(-0.1, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Rescale to ensure the integral is 1

        # Evaluate
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            print(f"Found better C5: {best_c5}")

    # Local optimization
    for _ in range(5):  # 5 local optimization steps
        # Use a gradient-based method to minimize the maximum overlap
        # In practice, this is a non-convex function, and gradient descent can get stuck in local optima
        # So we use a small step size and try multiple directions
        h = best_h.copy()
        direction = np.random.randn(n_points)
        direction /= np.linalg.norm(direction)
        step = 0.05
        for _ in range(100):  # 100 steps
            new_h = h + step * direction
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            c5 = evaluate_erdos_solution(new_h)
            if c5 < evaluate_erdos_solution(h):

            else:

    if best_c5 < evaluate_erdos_solution(best_h):
        best_h = best_h.copy()
        best_c5 = evaluate_erdos_solution(best_h)
