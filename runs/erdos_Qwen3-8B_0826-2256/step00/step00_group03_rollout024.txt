def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 150, 200, 250, 300, 350, 400]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for n_points in n_points_list:
        dx = 2.0 / n_points
        if n_points < len(initial_h_values):
            initial_h = np.array(initial_h_values[:n_points])
        else:
            initial_h = np.array([0.5] * n_points)

        # Ensure the initial h satisfies the integral constraint
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)

        # Define the function to minimize
        def objective(h):
            h = np.clip(h, 0, 1)
            integral = np.sum(h) * dx
            if np.abs(integral - 1) > 1e-7:
                return float('inf')
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr * dx)
            return c5

        # Use differential evolution for optimization
        result = optimize.differential_evolution(
            objective,
            bounds=[[0, 1] for _ in range(n_points)],
            strategy='best1bin',
            popsize=15,
            maxiter=50,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            polish=True,
            disp=False,
            seed=seed
        )

        h_values = result.x
        c5 = result.fun
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_values

        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s - 1:
            break

    # Final evaluation
    c5_bound = best_c5
    n_points = len(best_h)
    return best_h, c5_bound, n_points
