import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Predefined initial solution (n_points=45)
    n_points_initial = len(initial_h_values)
    h_initial = np.array(initial_h_values)
    
    # Try a few different starting points with varying n_points
    initial_points = [(n_points_initial, h_initial)]
    
    # Try smaller n_points for better performance
    for n_points in [40, 35, 30, 25, 20]:
        initial_points.append((n_points, np.random.rand(n_points)))
    
    best_c5 = float('inf')
    best_h = None
    best_n_points = n_points_initial
    
    start_time = time.time()
    
    for n_points, h0 in initial_points:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Ensure h0 integrates to 1
        h0 = h0 / np.sum(h0) * (n_points / 2)
        
        def objective(h):
            # Objective: minimize the maximum overlap
            # Compute the convolution (correlation)
            corr = correlate(h, 1 - h, mode="full")
            max_overlap = np.max(corr) * (2.0 / n_points)
            return max_overlap
        
        # Bounds for h: [0, 1]
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        # Use scipy.optimize.minimize to find the minimum C5
        result = opt.minimize(
            objective,
            h0,
            method='L-BFGS-B',
            bounds=bounds,
            options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 100}
        )
        
        # Check if the result is better
        if result.success:
            h_candidate = result.x
            c5_candidate = objective(h_candidate)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
    
    # Try an optimization with CVXOPT for more precise results
    if time.time() - start_time < budget_s - 1:
        n_points = best_n_points
        h0 = best_h if best_h is not None else np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        
        # Convert to CVXOPT variables
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h, h <= 1]
        
        # Objective: minimize the maximum overlap
        def objective_cvx(h):
            corr = correlate(h.value, 1 - h.value, mode="full")
            max_overlap = np.max(corr) * (2.0 / n_points)
            return max_overlap
        
        result = cp.Minimize(objective_cvx(h))
        prob = cp.Problem(result, constraints)
        prob.solve(solver=cp.CVXOPT, verbose=False)
        
        if prob.status == 'optimal':
            h_candidate = h.value
            c5_candidate = objective_cvx(h_candidate)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate
                best_n_points = n_points
    
    return best_h, best_c5, best_n_points
