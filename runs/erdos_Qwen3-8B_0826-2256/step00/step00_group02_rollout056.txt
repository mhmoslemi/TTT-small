import numpy as np
import scipy.optimize as opt
import scipy.signal as signal
import time
import cvxpy as cp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial h_values provided
    initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                                 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])
    n_points = len(initial_h_values)
    dx = 2.0 / n_points
    h = initial_h_values.copy()

    # Function to compute C5
    def compute_c5(h_values):
        h = h_values.copy()
        h = np.concatenate([h, h[:n_points//2]])
        corr = signal.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

    # Function to compute C5 and return the value
    def objective(x):
        return compute_c5(x)

    # Function to compute constraints
    def constraints(x):
        return np.array([np.sum(x) - n_points / 2])

    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        x0 = h_initial.copy()
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        result = opt.minimize(objective, x0, method='L-BFGS-B', bounds=bounds, constraints=constraints)
        if result.success:
            return result.x
        else:
            return h_initial

    # Optimization loop
    best_h = h
    best_c5 = compute_c5(best_h)
    remaining_time = budget_s - (time.time() - start_time)

    # Try a few iterations of optimization
    for _ in range(5):
        if remaining_time < 0:
            break
        best_h = optimize_h(best_h)
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        remaining_time = budget_s - (time.time() - start_time)

    return best_h, best_c5, n_points
