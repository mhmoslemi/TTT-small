def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    h_shifted = np.zeros((n_points, n_points))
    for i in range(n_points):
        h_shifted[i, :] = h
    overlap = np.zeros(n_points)
    for k in range(1, n_points):
        overlap[k] = np.dot(h, 1 - h_shifted[k:])
    return np.max(overlap) * dx

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial guess
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Define parameters for optimization
    max_iterations = 100
    refine_steps = 10
    learning_rate = 0.01
    n_refine = 10
    
    for _ in range(max_iterations):
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Refine the solution with gradient descent
        for _ in range(n_refine):
            # Compute objective and gradients
            dx = 2.0 / n_points
            h = np.clip(h, 0, 1)
            h_shifted = np.zeros((n_points, n_points))
            for i in range(n_points):
                h_shifted[i, :] = h
            overlap = np.zeros(n_points)
            for k in range(1, n_points):
                overlap[k] = np.dot(h, 1 - h_shifted[k:])
            c5 = np.max(overlap) * dx
            
            # Compute gradient
            grad = np.zeros(n_points)
            for k in range(1, n_points):
                overlap_k = np.dot(h, 1 - h_shifted[k:])
                if overlap_k == c5:
                    for i in range(n_points):
                        grad[i] += (1 - h_shifted[k, i]) * (overlap_k - 0.5 * c5)
                        grad[i] -= h[i] * (overlap_k - 0.5 * c5)
            
            # Update step
            h -= learning_rate * grad
            h = np.clip(h, 0, 1)
        
        # Early stopping if time is up
        if time.time() - start_time > budget_s:
            break
    
    return (best_h, best_c5, n_points)
