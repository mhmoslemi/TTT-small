```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
from evaluate_erdos_solution import evaluate_erdos_solution
from initial_h_values import initial_h_values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    h_initial = initial_h_values.copy()
    h = h_initial.copy()
    
    # Evaluate initial solution
    c5_initial = evaluate_erdos_solution(h)
    best_c5 = c5_initial
    best_h = h.copy()
    best_n_points = n_points
    
    # Try with different n_points
    for n_points in [70, 100, 140, 180, 220, 260, 300]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Resample h to n_points
        dx = 2.0 / n_points
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h_initial)), h_initial)
        
        # Add small random noise to avoid local minima
        h += np.random.uniform(-0.01, 0.01, size=h.shape)
        h = np.clip(h, 0, 1)
        
        # Normalize to satisfy the integral constraint
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        
        # Evaluate C5
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
    
    # Try local optimization with CVXOPT
    if time.time() - start_time < budget_s - 10:
        n_points = 140
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Define variables
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        
        # Define the objective function to minimize the maximum of correlations
        # We use a convex relaxation by minimizing the maximum correlation
        # This is a simplified approach, aiming for better performance
        obj = cp.Minimize(cp.maximum(cp.sum(cp.multiply(h, 1 - h[np.newaxis, :]) - cp.sum(h) * (1 - h[np.newaxis, :]) / n_points)))
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                h_opt = h.value
                c5_opt = evaluate_erdos_solution(h_opt)
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt
                    best_n_points = n_points
        except:
            pass
    
    # Try with a few more points
    for n_points in [180, 220, 260]:
        if time.time() - start_time > budget_s - 10:
            break
        
        dx = 2.0 / n_points
        x = np.linspace(0, 2, n_points)
        
        # Start from previous best
        h = np.interp(x, np.linspace(0, 2, best_n_points), best_h)
        h += np.random.uniform(-0.01, 0.01, size=h.shape)
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        scale = (n_points / 2) / sum_h
        h *= scale
        
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
