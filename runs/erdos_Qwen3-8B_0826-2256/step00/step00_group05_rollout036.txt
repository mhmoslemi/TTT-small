import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import sys
import random

np.random.seed(42)

# Pre-imported functions
# initial_h_values = ...  # This is a global variable representing an initial construction

def run(seed=42, budget_s=100, **kwargs):
    # Constants
    n_points = 1000  # Use 1000 samples for sufficient accuracy
    dx = 2.0 / n_points
    max_iterations = 50  # Number of iterations for optimization
    population_size = 50  # Genetic algorithm population size
    mutation_rate = 0.1  # Mutation rate
    crossover_rate = 0.8  # Crossover rate
    
    # Helper function to evaluate the overlap integral
    def evaluate_overlap(h):
        # Ensure h is within [0, 1]
        h_clamped = np.clip(h, 0, 1)
        # Compute the correlation
        corr = signal.correlate(h_clamped, 1 - h_clamped, mode="full")
        # Normalize by dx
        c5 = np.max(corr) * dx
        return c5

    # Helper function to calculate the integral of h
    def integral(h):
        return np.sum(h) * dx

    # Fitness function for genetic algorithm
    def fitness(h):
        # Ensure h is within [0, 1]
        h_clamped = np.clip(h, 0, 1)
        # Compute the integral
        if not np.isclose(integral(h_clamped), 1, atol=1e-6):
            return float('inf')  # Penalize for not meeting the integral constraint
        # Evaluate the overlap integral
        return evaluate_overlap(h_clamped)

    # Helper function to generate a random candidate
    def generate_candidate():
        h = np.random.rand(n_points)
        # Scale to meet the integral constraint
        h = h * (1 / (np.sum(h) * dx))
        return h

    # Helper function for mutation
    def mutate(h):
        h_mutated = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                h_mutated[i] = np.random.rand()
        # Scale to meet the integral constraint
        h_mutated = h_mutated * (1 / (np.sum(h_mutated) * dx))
        return h_mutated

    # Helper function for crossover
    def crossover(h1, h2):
        h_crossover = h1.copy()
        for i in range(n_points):
            if np.random.rand() < crossover_rate:
                h_crossover[i] = h2[i]
        # Scale to meet the integral constraint
        h_crossover = h_crossover * (1 / (np.sum(h_crossover) * dx))
        return h_crossover

    # Initialize population
    population = [generate_candidate() for _ in range(population_size)]
    
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    
    for _ in range(max_iterations):
        # Evaluate the fitness of each candidate
        fitness_values = [fitness(h) for h in population]
        
        # Find the best candidate
        best_index = np.argmin(fitness_values)
        best_c5_current = fitness_values[best_index]
        
        # Update the best solution
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = population[best_index]
        
        # If we've reached the time budget, break early
        if time.time() - start_time > budget_s - 1:
            break
        
        # Create a new population
        new_population = []
        
        # Elitism: keep the best candidate
        new_population.append(population[best_index])
        
        # Generate new candidates
        for _ in range(population_size - 1):
            # Select two parents
            parent1, parent2 = random.choices(population, k=2)
            # Crossover
            child = crossover(parent1, parent2)
            # Mutate
            child = mutate(child)
            new_population.append(child)
        
        # Replace the population
        population = new_population
    
    # Final evaluation
    best_c5 = evaluate_overlap(best_h)
    
    return best_h, best_c5, n_points
