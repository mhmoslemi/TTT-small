import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import random
import cvxpy as cp

# Pre-imported functions
# initial_h_values = ...  # Indirectly accessible through initial_h_values global

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set default n_points
    n_points = 100
    if budget_s > 20:
        n_points = 200
    if budget_s > 40:
        n_points = 300
    if budget_s > 60:
        n_points = 400
    if budget_s > 80:
        n_points = 500
    if budget_s > 100:
        n_points = 600
    
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Initial guess
    if hasattr(initial_h_values, '__len__') and len(initial_h_values) == n_points:
        h = np.copy(initial_h_values)
    else:
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute C_5
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        shifted_h = np.roll(h, 1)
        overlap = np.correlate(h, 1 - h, mode="full") * dx
        return np.max(overlap)
    
    # Optimization function with constraints
    def optimize_h(h_initial):
        h = cp.Variable(n_points)
        constraints = [
            cp.sum(h) == n_points / 2,
            0 <= h,
            h <= 1
        ]
        obj = cp.Minimize(cp.Max(signal.correlate(h.value, 1 - h.value, mode="full") * dx))
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status != 'optimal':
                print("Solver did not find optimal solution")
                return h_initial
            return h.value
        except:
            print("Solver failed, using initial guess")
            return h_initial
    
    # Run optimization
    best_h = h
    best_c5 = compute_c5(h)
    
    time_used = time.time() - start_time
    remaining_time = budget_s - time_used
    
    # Perform multiple iterations
    for _ in range(5):
        if time_used + 10 > budget_s:
            break
        h = optimize_h(h)
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_h = h
            best_c5 = current_c5
        time_used = time.time() - start_time
    
    return best_h, best_c5, n_points
