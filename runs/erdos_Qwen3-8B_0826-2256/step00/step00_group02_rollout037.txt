def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    shifted_h = np.zeros((n_points, n_points))
    for i in range(n_points):

        integrals[k] = np.sum(h * (1 - shifted_h[k, :])) * dx
