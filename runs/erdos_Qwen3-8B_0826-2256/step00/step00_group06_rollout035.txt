import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize
import cvxpy as cp
import sys

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h_values.npy")  # placeholder, assuming initial values are loaded

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    n_points = 100  # starting with 100 points
    
    # Try different numbers of points up to 500 to find a good balance between accuracy and speed
    for n_points in [100, 200, 300, 400, 500]:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 10:
            break
        
        # Generate an initial guess based on initial_h_values or random
        if n_points < len(initial_h_values):
            h = np.zeros(n_points)
            idx = np.linspace(0, len(initial_h_values) - 1, n_points, dtype=int)
            h = initial_h_values[idx]
        else:
            h = np.random.rand(n_points)
            h = h * (np.sum(initial_h_values) / np.sum(h))  # scale to match the integral constraint
        
        # Enforce the constraint: sum(h) == n_points / 2 exactly
        def objective(x):
            nonlocal best_c5
            h = x
            # Check if the total integral constraint is satisfied
            total_integral = np.sum(h) * (2.0 / n_points)
            if abs(total_integral - 1.0) > 1e-6:
                return float('inf')  # Penalize for violating the integral constraint
            # Calculate the C5 value
            corr = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            return c5
        
        # Set bounds
        bounds = Bounds(0.0, 1.0)
        
        # Optimize
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={
                'maxiter': 100,
                'ftol': 1e-6,
                'gtol': 1e-6,
                'disp': False
            }
        )
        
        # Update best solution
        if result.success:
            h_opt = result.x
            total_integral = np.sum(h_opt) * (2.0 / n_points)
            if abs(total_integral - 1.0) < 1e-6:
                corr = np.correlate(h_opt, 1 - h_opt, mode="full")
                c5 = np.max(corr) * (2.0 / n_points)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
    
    # Final evaluation
    if best_h is not None:
        total_integral = np.sum(best_h) * (2.0 / len(best_h))
        if abs(total_integral - 1.0) < 1e-6:
            corr = np.correlate(best_h, 1 - best_h, mode="full")
            c5_bound = np.max(corr) * (2.0 / len(best_h))
        else:
            c5_bound = float('inf')
    else:
        c5_bound = float('inf')
    
    return (best_h, c5_bound, len(best_h))
