```python
import numpy as np
import scipy.optimize as optimize
import cvxpy as cp
import time
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(np.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load('initial_h.npy')  # Placeholder for initial construction (n=45 samples)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45
    h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_n = n_points
    
    # Try reducing n_points
    for n_points in [40, 35, 30, 25, 20, 15, 10]:
        h = np.linspace(0, 2, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = h * (2.0 / n_points)  # Normalize to ensure integral is 1
        h = h / np.sum(h) * (n_points / 2)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_points
    
    # Try increasing n_points
    for n_points in [50, 60, 70, 80, 90, 100]:
        h = np.linspace(0, 2, n_points, endpoint=False)
        h = np.random.rand(n_points)
        h = h * (2.0 / n_points)
        h = h / np.sum(h) * (n_points / 2)
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            best_n = n_points
    
    # Use optimization to improve the solution
    def objective(x):
        h = x
        return evaluate_erdos_solution(h)
    
    result = optimize.minimize(
        objective,
        best_h,
        method='L-BFGS-B',
        bounds=[(0, 1) for _ in range(best_n)],
        options={'maxiter': 100, 'ftol': 1e-9}
    )
    best_h = result.x
    best_c5 = evaluate_erdos_solution(best_h)
    
    if best_c5 < evaluate_erdos_solution(best_h):
        best_h = result.x
        best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        return best_h, best_c5, best_n
    
    # Try another initial guess
    h = np.random.rand(best_n)
    h = h * (2.0 / best_n)
    h = h / np.sum(h) * (best_n / 2)
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    return best_h, best_c5, best_n
```
