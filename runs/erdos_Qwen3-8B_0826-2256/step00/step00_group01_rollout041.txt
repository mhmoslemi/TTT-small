```python
import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import lru_cache

# Pre-imported functions
evaluate_erdos_solution = lambda h_values: np.max(signal.correlate(h_values, 1 - h_values, mode="full")) * (2.0 / len(h_values))
initial_h_values = np.load("initial_h_values.npy")  # Load initial solution (n=47 samples)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h = initial_h_values.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    n_points = len(best_h)
    
    # Try different number of points
    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate initial guess
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy the integral constraint
        
        # Define variables and constraints
        x = cp.Variable(n_points)
        constraints = [
            cp.sum(x) == n_points / 2,
            0 <= x,
            x <= 1
        ]
        
        # Function to minimize the maximum overlap
        def objective(x):
            h_values = x.value
            corr = signal.correlate(h_values, 1 - h_values, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        # Use a gradient-based optimization
        for _ in range(20):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Use sequential least squares programming
            prob = cp.Problem(cp.Minimize(0), constraints)
            result = prob.solve(solver=cp.SCS, max_iters=100, verbose=False)
            if result is None:
                break
            
            h = x.value
            current_c5 = objective(x)
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        
        # Evaluate the best solution for this n_points
        if evaluate_erdos_solution(best_h) < best_c5:
            best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, len(best_h)
```
