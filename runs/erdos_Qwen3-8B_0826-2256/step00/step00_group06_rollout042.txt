```python
import numpy as np
import scipy.optimize as opt
import cvxpy as cp
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.load("initial_h.npy")  # Assuming this exists with n_points = 95

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial guess: start from provided initial_h_values and scale to n_points = 128 (faster to optimize)
    n_points = 128
    dx = 2.0 / n_points
    h = np.zeros(n_points)
    h[:len(initial_h_values)] = initial_h_values[:n_points]
    h = h[:n_points]
    h = h / np.sum(h) * (n_points / 2)  # Scale to satisfy integral constraint
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    # Hyperparameters for optimization
    max_iter = 100
    n_samples = 50  # Number of random restarts
    tolerance = 1e-4

    for _ in range(n_samples):
        if time.time() - start_time > budget_s - 1:
            break

        # Use direct search to refine the solution
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            integral = np.sum(h) * dx
            if abs(integral - 1) > tolerance:
                return np.inf
            c5 = evaluate_erdos_solution(h)
            return c5

        result = opt.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        h_candidate = result.x
        c5_candidate = evaluate_erdos_solution(h_candidate)

        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate

    # Use convex optimization for further refinement
    x = cp.Variable(n_points)
    constraints = [0 <= x, x <= 1, cp.sum(x) == n_points / 2]
    objective = cp.Maximize(cp.max(cp.correlate(x, 1 - x, mode="full")) * dx)
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.SCS, eps=1e-4)

    if prob.status == 'optimal':
        best_h = x.value
        best_c5 = evaluate_erdos_solution(best_h)

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5

    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 1:
        best_c5 = np.inf  # Ensure we don't exceed the time budget

    return (best_h, best_c5, n_points)
```
