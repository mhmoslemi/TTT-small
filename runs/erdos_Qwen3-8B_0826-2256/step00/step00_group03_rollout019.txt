def run(seed=42, budget_s=100, **kwargs):
    random.seed(seed)
    start_time = time.time()
    n_points = 200  # Start with 200 points for balance between speed and quality
    h = copy.deepcopy(initial_h_values[:n_points])
    h = h / (n_points / 2.0)  # Ensure the integral constraint is satisfied
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Optimization steps
    for _ in range(15):  # Max 15 iterations
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break

        # First: Gradient descent-style update
        def objective_function(h):
            return evaluate_erdos_solution(h)

        # Use a simple gradient-based approach
        h = h.astype(float)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-4
            h_minus = h.copy()
            h_minus[i] -= 1e-4
            grad[i] = (objective_function(h_plus) - objective_function(h_minus)) / (2 * 1e-4)

        # Update h in the direction of negative gradient
        h -= 0.1 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0)  # Ensure the integral constraint is satisfied

        # Evaluate and keep the best
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Second: Random perturbation to escape local minima
        h = h.astype(float)
        for i in range(len(h)):
            if random.random() < 0.1:
                h[i] += np.random.uniform(-0.1, 0.1)
                h[i] = np.clip(h[i], 0, 1)
        h = h * (n_points / 2.0)  # Ensure the integral constraint is satisfied

        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Check if we should increase the number of points
        if random.random() < 0.5:  # With 50% chance, increase points

            h = h * (n_points / 2.0) / np.mean(h)
            best_h = h.copy()

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    # Return the best result
    return (best_h, final_c5, len(best_h))
