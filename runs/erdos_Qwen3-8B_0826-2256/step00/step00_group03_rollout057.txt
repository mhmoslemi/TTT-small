    for n_points in [100, 200, 300, 400, 500]:
        if time.time() - start_time > budget_s - 10:

        # Generate initial guess with random perturbations
        current_h = np.random.rand(n_points) * 0.5

        def objective(x):
            h = x.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h = np.round(h, 5)

            dx = 2.0 / n_points
            overlap = np.zeros(n_points - 1)
            for k in range(1, n_points):
                h_k = np.roll(h, k)
                overlap[k - 1] = np.sum(h * (1 - h_k)) * dx

        result = optimize.minimize(objective, current_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
