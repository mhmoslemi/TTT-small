import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Use 100 points for balance between speed and accuracy
    dx = 2.0 / n_points
    
    # Initialize with provided initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to symmetric pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Enforce integral constraint
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Objective function: minimize max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Metaheuristic search with adaptive perturbation and local refinement
    for _ in range(50):  # 50 iterations with adaptive perturbation
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb with decreasing magnitude
        perturbation = np.random.uniform(-0.05 * (1 - _ / 50), 0.05 * (1 - _ / 50), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Enforce integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Local refinement with SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 100}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Global exploration with differential evolution
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="differential_evolution",
                options={"maxiter": 10, "popsize": 20}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
