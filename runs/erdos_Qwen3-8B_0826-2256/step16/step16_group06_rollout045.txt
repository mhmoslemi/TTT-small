import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a high-resolution base construction
    n_points = 100
    # Define a base construction that balances h(x) and 1-h(x) over [0, 2]
    h = np.zeros(n_points)
    h[:n_points//2] = 1.0
    h[n_points//2:] = 0.0
    # Smooth the discontinuity to avoid sharp peaks
    h = np.convolve(h, np.ones(5)/5, mode='same')
    h = np.clip(h, 0, 1)
    # Scale to meet the integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.2  # Initial perturbation scale for exploration
    learning_rate = 0.01  # Learning rate for gradient-based methods

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150, "eps": 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # COBYLA for global exploration
        try:
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
