    bounds = Bounds(0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    def differential_evolution_run(h0):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            constraints=constraint,
            x0=h0,
            seed=seed
        )

    def slsqp_refinement(h0):
        try:
            result = minimize(
                objective,
                h0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 200, "eps": 1e-8}
            )

    best_h, best_c5 = differential_evolution_run(h)
