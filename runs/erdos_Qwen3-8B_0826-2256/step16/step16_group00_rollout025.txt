    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)

        return np.sum(h_values) - n_points / 2

    best_c5 = objective(h)
    start_time = time.time()

    # Optimization parameters
    num_trials = 150
    perturbation_scale = 0.25  # Larger initial perturbation for broader exploration
