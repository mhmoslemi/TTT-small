import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency and quality
    
    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.3  # Slightly smaller initial perturbation for better exploration
    learning_rate = 0.02  # Learning rate for gradient-based methods
    initial_temperature = 1.0
    cooling_rate = 0.995

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Simulated Annealing for global exploration
        current_temperature = initial_temperature * (cooling_rate ** trial)
        for iteration in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate a random neighbor
            neighbor = x0 + np.random.uniform(-0.05, 0.05, size=n_points)
            neighbor = np.clip(neighbor, 0, 1)
            neighbor = neighbor / (np.sum(neighbor) * 2.0 / n_points) if np.sum(neighbor) != 0 else x0
            neighbor = np.clip(neighbor, 0, 1)
            
            # Evaluate the cost
            cost_neighbor = objective(neighbor)
            cost_current = objective(x0)
            
            # Acceptance probability
            if cost_neighbor < cost_current or np.random.rand() < np.exp((cost_current - cost_neighbor) / current_temperature):
                x0 = neighbor
        
        # Refine using SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
