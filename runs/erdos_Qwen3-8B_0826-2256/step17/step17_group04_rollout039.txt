import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    n_points = 100  # Start with 100 points for a good balance
    dx = 2.0 / n_points

        h = initial_h_values.copy()
    else:
        # Fallback to a piecewise constant construction

        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 200  # Increased number of trials for better exploration
    
    # Use a combination of perturbation, local refinement, and global search
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Random perturbation with decreasing magnitude
        perturbation = np.random.uniform(-0.04 * (1 - trial / trials), 0.04 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize to satisfy integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Local refinement using SLSQP with tighter tolerances
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 200}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Global exploration using differential evolution with adaptive parameters
        try:
            result_de = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="differential_evolution",
                options={"maxiter": 50, "popsize": 20, "mutation": (0.5, 1.0), "recombination": "blend", "tol": 1e-5}
            )
            if result_de.success:
                current_c5 = result_de.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_de.x
        except:
            pass
    
    return best_h, best_c5, n_points
