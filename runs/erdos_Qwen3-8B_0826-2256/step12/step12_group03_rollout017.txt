import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 300  # Increase the number of trials for better exploration
    perturbation_scale = 0.2  # Initial perturbation scale
    step_size = 0.05  # Initial step size for local search

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * 2.0 / n_points
        if total != 0:
            h = h * (n_points / 2.0) / total
        h = np.clip(h, 0, 1)
        
        # Hill-climbing with local search
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Try small random perturbations
            h_perturbed = h + np.random.uniform(-step_size, step_size, size=n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            total_perturbed = np.sum(h_perturbed) * 2.0 / n_points
            if total_perturbed != 0:
                h_perturbed = h_perturbed * (n_points / 2.0) / total_perturbed
            h_perturbed = np.clip(h_perturbed, 0, 1)
            c5_perturbed = objective(h_perturbed)
            if c5_perturbed < objective(h):
                h = h_perturbed
        
        # Use SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",

            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
        step_size *= 0.95

    return best_h, best_c5, n_points
