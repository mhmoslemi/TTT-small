import numpy as np

import copy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial guess from the provided initial_h_values
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(100)
        h[:50] = 0.5
        h[50:] = 0.5
        h = h / (np.sum(h) * 2.0 / 100) if np.sum(h) != 0 else np.random.rand(100)
    
    h = np.clip(h, 0, 1)
    n_points = 100
    dx = 2.0 / n_points
    
    # Function to calculate overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint function for the integral
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Track the best solution found
    best_solution = (best_h, best_c5)
    
    # Initial exploration with perturbations
    for _ in range(150):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.01, 0.01, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # SLSQP for local refinement
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 100}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_solution = (best_h, best_c5)
        except:
            pass
    
    # Hybrid approach with different starting points
    for _ in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # COBYLA for global exploration
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 100}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
                    best_solution = (best_h, best_c5)
        except:
            pass
    
    # Final refinement with SLSQP
    try:
        result = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
                best_solution = (best_h, best_c5)
    except:
        pass
    
    return best_h, best_c5, n_points
