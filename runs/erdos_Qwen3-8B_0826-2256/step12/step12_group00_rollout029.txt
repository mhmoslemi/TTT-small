import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and effective number of points

    # Initialize with provided initial_h_values (n=100 points)

        h = initial_h_values.copy()
    else:
        # Fallback initial guess: equal 0.5s on [0, 1]
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.5
        h[n_points // 2:] = 0.5

    # Ensure the initial guess is within [0, 1]
    h = np.clip(h, 0, 1)

    # Function to evaluate overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint: integral of h must equal 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)

    start_time = time.time()
    trials = 50  # Reasonable number of trials for exploration
    perturbation_scale = 0.2  # Larger initial perturbation for exploration
    learning_rate = 0.05  # Moderate learning rate for gradient steps

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)

        # Ensure integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Use SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 200}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Use COBYLA for exploration
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 200}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
