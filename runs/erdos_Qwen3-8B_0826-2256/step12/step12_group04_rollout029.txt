import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.sparse import csr_matrix
import copy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for a good balance
    dx = 2.0 / n_points
    max_trial = 150  # Number of trials to explore different initial points
    
    # Use the initial_h_values if provided
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = np.zeros(n_points)
        initial_h[:n_points//2] = 0.5
        initial_h[n_points//2:] = 0.5
        # Ensure the integral is 1
        initial_h = initial_h / (np.sum(initial_h) * dx) if np.sum(initial_h) != 0 else np.random.rand(n_points)
    
    best_h = initial_h
    best_c5 = evaluate_erdos_solution(best_h, dx)
    start_time = time.time()
    
    # Try different initial points to escape local minima
    for trial in range(max_trial):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate a new random initial point
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Optimization with SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        result_slsqp = minimize(
            lambda h_values: evaluate_erdos_solution(h_values, dx),
            x0,
            bounds=Bounds(0, 1),
            constraints=constraint,
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Optimization with COBYLA for global exploration
        result_cobyla = minimize(
            lambda h_values: evaluate_erdos_solution(h_values, dx),
            x0,
            bounds=Bounds(0, 1),
            constraints=constraint,
            method="COBYLA",
            options={"maxiter": 150}
        )
        if result_cobyla.success:
            current_c5 = result_cobyla.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x
    
    return best_h, best_c5, n_points

def evaluate_erdos_solution(h_values, dx):
    h = h_values.reshape(-1)
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5
