import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and resolution
    dx = 2.0 / n_points
    
    # Initialize with a promising starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure the integral constraint is met
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Define the objective to minimize the max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 200  # Max number of trials in the budget
    
    # Run a combination of local and global optimization strategies
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current solution with decreasing magnitude
        perturbation = np.random.uniform(-0.025 * (1 - trial / trials), 0.025 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Adjust the integral to meet the constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h.copy()
        
        # Perform a local optimization with tighter tolerances
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Run a global search with COBYLA to escape local optima
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
