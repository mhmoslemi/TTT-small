import scipy.optimize as optimize
import time

        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 200  # Increase number of trials for better exploration
    
    # Use a hybrid approach with global and local optimization
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution
        perturbation = np.random.normal(0, 0.02, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

        x0 = h

        # Use L-BFGS-B for local refinement with tighter tolerances
        result_lbfgs = optimize.minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="L-BFGS-B",

        )

        result_de = optimize.differential_evolution(
            objective,
            bounds,
            args=(n_points, dx),

            maxiter=50,
            popsize=15,
            strategy="best1bin",
            tol=1e-9
