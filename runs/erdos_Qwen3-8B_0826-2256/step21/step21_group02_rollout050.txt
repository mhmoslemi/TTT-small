import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and sufficient for quality
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Thorough exploration
    perturbation_scale = 0.3
    learning_rate = 0.01
    exploration_rate = 0.3
    adaptive_learning = True
    shuffle_frequency = 50
    mutation_rate = 0.1
    mutation_scale = 0.15
    population_size = 50
    crossover_rate = 0.8
    mutation_rate_genetic = 0.1
    mutation_scale_genetic = 0.2
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Genetic algorithm step
        population = [np.random.rand(n_points) for _ in range(population_size)]
        for p in population:
            p = p / (np.sum(p) * 2.0 / n_points) if np.sum(p) != 0 else p
            p = np.clip(p, 0, 1)
        
        for p in population:
            # Apply mutation
            if np.random.rand() < mutation_rate_genetic:
                mutation = np.random.normal(0, mutation_scale_genetic, size=n_points)
                p = np.clip(p + mutation, 0, 1)
            # Apply crossover with best solution
            if np.random.rand() < crossover_rate:
                idx = np.random.randint(0, len(population))
                crossover_point = np.random.randint(0, n_points)
                p[:crossover_point] = population[idx][:crossover_point]
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_genetic = population[best_idx]
        best_c5_genetic = fitness[best_idx]
        
        # Compare and update best solution
        if best_c5_genetic < best_c5:
            best_c5 = best_c5_genetic
            best_h = best_genetic
        
        # Local refinement with SLSQP
        h = best_h
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # COBYLA for global exploration
        try:
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 300}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Random exploration with learning rate
        if np.random.rand() < exploration_rate:
            if adaptive_learning:
                learning_rate *= 0.98
            h = best_h + np.random.normal(0, learning_rate, size=n_points)
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        
        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.95
        
        # Periodic mutation for diversity
        if trial % shuffle_frequency == 0:
            h = np.random.rand(n_points)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        
        # Mutation to explore new regions
        if np.random.rand() < mutation_rate:
            mutation = np.random.normal(0, mutation_scale, size=n_points)
            h = best_h + mutation
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
