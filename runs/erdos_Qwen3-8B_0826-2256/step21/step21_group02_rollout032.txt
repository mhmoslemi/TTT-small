import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and sufficient for quality
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 100  # Thorough exploration
    perturbation_scale = 0.25
    learning_rate = 0.01
    exploration_rate = 0.3
    adaptive_learning = True
    shuffle_frequency = 50
    mutation_rate = 0.1
    mutation_scale = 0.15
    population_size = 20
    crossover_rate = 0.8
    mutation_rate_genetic = 0.1
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Genetic algorithm-inspired exploration
        population = [best_h + np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points) for _ in range(population_size)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p / (np.sum(p) * 2.0 / n_points) if np.sum(p) != 0 else np.random.rand(n_points) for p in population]
        population = [np.clip(p, 0, 1) for p in population]
        
        # Evaluate population
        fitness = [objective(p) for p in population]
        best_idx = np.argmin(fitness)
        best_pop = population[best_idx]
        best_fitness = fitness[best_idx]
        
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_pop
        
        # Crossover and mutation
        for i in range(population_size):
            if np.random.rand() < crossover_rate:
                j = np.random.randint(population_size)
                child = (population[i] + population[j]) / 2
                child = np.clip(child, 0, 1)
                child = child / (np.sum(child) * 2.0 / n_points) if np.sum(child) != 0 else np.random.rand(n_points)
                child = np.clip(child, 0, 1)
                population[i] = child
            
            if np.random.rand() < mutation_rate_genetic:
                mutation = np.random.normal(0, mutation_scale, size=n_points)
                population[i] = population[i] + mutation
                population[i] = np.clip(population[i], 0, 1)
                population[i] = population[i] / (np.sum(population[i]) * 2.0 / n_points) if np.sum(population[i]) != 0 else np.random.rand(n_points)
                population[i] = np.clip(population[i], 0, 1)
        
        # Local refinement with SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Random exploration with learning rate
        if np.random.rand() < exploration_rate:
            if adaptive_learning:
                learning_rate *= 0.98
            h = best_h + np.random.normal(0, learning_rate, size=n_points)
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        
        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.95
        
        # Periodic mutation for diversity
        if trial % shuffle_frequency == 0:
            h = np.random.rand(n_points)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        
        # Mutation to explore new regions
        if np.random.rand() < mutation_rate:
            mutation = np.random.normal(0, mutation_scale, size=n_points)
            h = best_h + mutation
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
