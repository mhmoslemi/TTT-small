    def objective(h_values):

    learning_rate = 0.01
    exploration_rate = 0.4

    genetic_mutation_rate = 0.1
    crossover_rate = 0.8
    population_size = 20
    
    # Genetic algorithm parameters
    def generate_population(size):
        population = []
        for _ in range(size):
            candidate = np.random.rand(n_points)
            candidate = candidate / (np.sum(candidate) * 2.0 / n_points) if np.sum(candidate) != 0 else candidate
            candidate = np.clip(candidate, 0, 1)
            population.append(candidate)
        return np.array(population)
    
    # Evaluate a population
    def evaluate_population(population):
        return np.array([objective(ind) for ind in population])
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() < crossover_rate:
            child = np.random.choice([parent1, parent2], size=n_points)
            return child
        else:
            return parent1
    
    # Mutation
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < genetic_mutation_rate:
                mutated[i] += np.random.uniform(-0.1, 0.1)
                mutated[i] = np.clip(mutated[i], 0, 1)
        return mutated
    
    # Initialize population
    population = generate_population(population_size)
    best_population = population.copy()
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        fitness = evaluate_population(population)
        best_fitness = np.min(fitness)
        
        # Update best solution
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = population[np.argmin(fitness)]
        
        # Select parents
        parents = population[np.argsort(fitness)]
        
        # Create new population
        new_population = []
        for _ in range(population_size):
            parent1, parent2 = np.random.choice(parents, size=2, replace=False)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_population.append(child)
        
        population = np.array(new_population)
    
    return best_h, best_c5, n_points
