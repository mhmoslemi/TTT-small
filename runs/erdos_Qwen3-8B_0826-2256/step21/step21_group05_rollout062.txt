    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)

    best_c5 = objective(h)

    def differential_evolution_objective(x):
        return objective(x)

    initial_population = np.random.uniform(low=0, high=1, size=(pop_size, n_points))

    result_de = differential_evolution(

    best_h = result_de.x

    for _ in range(100):
