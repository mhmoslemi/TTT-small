import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of points for speed and quality
    
    # Initialize with a balanced, coarse h for exploration
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Parameters for search and optimization
    trials = 200  # Controlled number of trials for balance
    perturbation_scale = 0.5  # Larger perturbation for initial exploration
    learning_rate = 0.01  # Small learning rate for fine-tuning
    exploration_rate = 0.6  # Probability of exploring new regions
    temperature = 1.5  # Higher temperature for more stochastic exploration
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Explore by perturbing current best
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Use SLSQP to minimize the objective under the constraint
        try:
            result_sa = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                method="SLSQP",
                options={"ftol": 1e-12, "maxiter": 250, "eps": 1e-8}
            )
            if result_sa.success:
                current_c5 = result_sa.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_sa.x
        except:
            pass

        # Occasionally restart with random perturbation
        if np.random.rand() < exploration_rate:
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0, 1)

            h = np.clip(h, 0, 1)
            x0 = h
        
        # Reduce perturbation scale for finer adjustments
        perturbation_scale *= 0.96

    return best_h, best_c5, n_points
