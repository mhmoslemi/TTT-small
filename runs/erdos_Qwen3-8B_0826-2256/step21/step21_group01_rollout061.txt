import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and sufficient for quality
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Thorough exploration with fewer trials
    perturbation_scale = 0.3
    learning_rate = 0.01
    exploration_rate = 0.25
    adaptive_learning = True
    n_perturbations = 3
    exploration_steps = 20
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb and clip
        perturbations = np.random.uniform(-perturbation_scale, perturbation_scale, size=(n_perturbations, n_points))
        h_candidates = [best_h + perturbations[i] for i in range(n_perturbations)]
        h_candidates = [np.clip(h_candidate, 0, 1) for h_candidate in h_candidates]
        
        # Maintain integral constraint
        h_candidates = [h_candidate / (np.sum(h_candidate) * 2.0 / n_points) if np.sum(h_candidate) != 0 else np.random.rand(n_points) for h_candidate in h_candidates]
        h_candidates = [np.clip(h_candidate, 0, 1) for h_candidate in h_candidates]
        
        # Evaluate the candidates
        current_c5s = []
        for h_candidate in h_candidates:
            try:
                result = minimize(
                    objective,
                    h_candidate,
                    bounds=bounds,
                    constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                    method="SLSQP",
                    options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
                )
                if result.success:
                    current_c5 = result.fun
                else:
                    current_c5 = np.inf
            except:
                current_c5 = np.inf
            
            current_c5s.append(current_c5)
        
        # Select the best perturbation
        best_idx = np.argmin(current_c5s)
        best_h = h_candidates[best_idx]
        best_c5 = current_c5s[best_idx]
        
        # Random exploration with learning rate
        if np.random.rand() < exploration_rate:
            if adaptive_learning:
                learning_rate *= 0.98
            h = best_h + np.random.normal(0, learning_rate, size=n_points)
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
        
        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.96
    
    return best_h, best_c5, n_points
