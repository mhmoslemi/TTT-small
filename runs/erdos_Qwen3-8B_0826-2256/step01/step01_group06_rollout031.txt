def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [150, 200, 250, 300, 350]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for n_points in n_points_list:
        dx = 2.0 / n_points
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            # Normalize the initial guess
            h = h / total * 1.0

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Normalize h to satisfy the integral constraint
            total = np.sum(h) * dx
            h = h / total * 1.0
            # Evaluate the maximum overlap
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

            recombination=0.8,
            polish=True,
            disp=False
        )

        if elapsed >= budget_s - 1:
