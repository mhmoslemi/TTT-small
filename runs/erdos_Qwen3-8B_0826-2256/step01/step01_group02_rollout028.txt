def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start with a smaller number of points for faster optimization
    n_points = 100
    n_points_max = 200
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        h = np.clip(h, 0, 1)
        total_mass = np.sum(h) * (2.0 / n_points)
        if np.abs(total_mass - 1.0) > 1e-6:
            return np.inf
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Initial guess - start with the initial construction and perturb it
    def initialize_h(n_points):
        h = np.copy(initial_h_values[:n_points])
        h = np.random.uniform(0.95, 1.05, size=n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Optimization function using SLSQP
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(x):
            return evaluate_overlap(x)
        
        def constraint_func(x):
            return np.sum(x) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Clip the result to ensure validity
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, evaluate_overlap(h)
    
    # Run initial optimization
    best_h = initialize_h(n_points)
    best_c5 = evaluate_overlap(best_h)
    start_time = time.time()
    
    # Try multiple starting points with perturbations
    for _ in range(5):
        h_initial = initialize_h(n_points)
        h, c5 = optimize_h(h_initial)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
        if time.time() - start_time > budget_s - 10:
            break

    if time.time() - start_time < budget_s - 10:

                best_h = h

    return best_h, best_c5, n_points
