import scipy.optimize as optimize
import time

from scipy.signal import correlate

evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):

    best_c5 = float('inf')

    # Try different numbers of points
    for n_points in [140, 180, 220, 260, 300]:

        def objective(h_vec):

                best_h = h_opt
