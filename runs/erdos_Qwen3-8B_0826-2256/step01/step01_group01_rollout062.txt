import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds

def run(seed=42, budget_s=100, **kwargs):

    # Start from the initial construction if available
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values
    
    # Interpolate initial_h to n_points if needed
    if len(initial_h) != n_points:
        x_initial = np.linspace(0, 2, len(initial_h))
        x_new = np.linspace(0, 2, n_points)
        initial_h = np.interp(x_new, x_initial, initial_h)
    
    # Ensure sum(h) * dx = 1
    dx = 2.0 / n_points
    h = initial_h / (np.sum(initial_h) * dx)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # sum(h) == n_points / 2
    
    # Setup optimization problem
    bounds = Bounds(0, 1)
    x0 = h
    
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )
    
    # Try different initializations if the optimization didn't converge
    if not result.success:
        print("Optimization did not converge, trying random initialization")
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx)
        h = np.clip(h, 0, 1)
        x0 = h
        result = optimize.minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
    
    # Try a few more random initializations if needed
    if not result.success:

        for _ in range(3):

                method="SLSQP",
