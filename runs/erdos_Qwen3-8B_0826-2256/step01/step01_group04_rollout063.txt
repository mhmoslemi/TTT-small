import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(140) * 2 - 1  # Placeholder, actual initial construction may vary
initial_h_values = np.clip(initial_h_values, 0, 2)
initial_h_values = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (140 / 2)  # Normalize to sum to 140/2
initial_h_values = np.round(initial_h_values).astype(float)
initial_h_values = initial_h_values / np.sum(initial_h_values) * (140 / 2)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Try different numbers of points with careful budgeting
    for n_points in [140, 160, 180, 200, 220, 240]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Initialize h with a structured pattern (sinusoidal) to start near optimal
        h = np.sin(np.linspace(0, np.pi, n_points)) * 0.5 + 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h = np.round(h).astype(float)
        h = h / np.sum(h) * (n_points / 2)
        
        # Optimization using a combination of simulated annealing and gradient descent
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            h = h_vec
            # Compute the max overlap
            cor = correlate(h, 1 - h, mode="full")
            c5 = np.max(cor) * (2.0 / n_points)
            return c5
        
        # Initial optimization with L-BFGS-B
        result = optimize.minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(n_points)])
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.round(h_opt).astype(float)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
        
        # Further refinement with simulated annealing
        def objective_sa(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            h = h_vec
            # Compute the max overlap
            cor = correlate(h, 1 - h, mode="full")
            c5 = np.max(cor) * (2.0 / n_points)
            return c5
        
        # Simulated annealing with temperature schedule
        def simulated_annealing(h_initial, temperature=100, cooling_rate=0.995, iterations=1000):
            h = h_initial.copy()
            current_cost = objective_sa(h)
            best_h = h.copy()
            best_cost = current_cost
            
            for i in range(iterations):
                if temperature < 0.1:
                    break
                # Generate a random perturbation
                perturbation = np.random.normal(0, 0.05, len(h))
                h_perturbed = h + perturbation
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                h_perturbed = np.round(h_perturbed).astype(float)
                h_perturbed = h_perturbed / np.sum(h_perturbed) * (n_points / 2)
                
                # Compute cost
                cost_perturbed = objective_sa(h_perturbed)
                
                # Acceptance probability
                if cost_perturbed < current_cost or np.random.rand() < np.exp((current_cost - cost_perturbed) / temperature):
                    h = h_perturbed
                    current_cost = cost_perturbed
                    if current_cost < best_cost:
                        best_h = h.copy()
                        best_cost = current_cost
                
                temperature *= cooling_rate
            
            return best_h, best_cost
        
        h_opt, c5_opt = simulated_annealing(h_opt)
        
        # Evaluate
        cor = correlate(h_opt, 1 - h_opt, mode="full")
        c5 = np.max(cor) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        print(f"n_points={n_points}, c5={c5}")
    
    # Try local refinement with perturbation and gradient-based optimization
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perturb the best solution
        perturbation = np.random.normal(0, 0.05, len(best_h))
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (len(best_h) / 2) / np.sum(perturbed_h)
        perturbed_h = np.round(perturbed_h).astype(float)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * (len(best_h) / 2)
        
        # Optimization using L-BFGS-B on the perturbed solution
        def objective_perturbed(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (len(best_h) / 2) / np.sum(h_vec)
            h = h_vec
            # Compute the max overlap
            cor = correlate(h, 1 - h, mode="full")
            c5 = np.max(cor) * (2.0 / len(best_h))
            return c5
        
        result = optimize.minimize(objective_perturbed, perturbed_h, method='L-BFGS-B', bounds=[(0, 1) for _ in range(len(best_h))])
        h_opt_perturbed = result.x
        h_opt_perturbed = np.clip(h_opt_perturbed, 0, 1)
        h_opt_perturbed = h_opt_perturbed * (len(best_h) / 2) / np.sum(h_opt_perturbed)
        h_opt_perturbed = np.round(h_opt_perturbed).astype(float)
        h_opt_perturbed = h_opt_perturbed / np.sum(h_opt_perturbed) * (len(best_h) / 2)
        
        # Evaluate
        cor = correlate(h_opt_perturbed, 1 - h_opt_perturbed, mode="full")
        c5_perturbed = np.max(cor) * (2.0 / len(best_h))
        
        if c5_perturbed < best_c5:
            best_c5 = c5_perturbed
            best_h = h_opt_perturbed
    
    return best_h, best_c5, len(best_h)
