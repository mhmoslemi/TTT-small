initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Start with 200 points, since the initial construction has 200 samples
    n_points = 200
    
    # Initial guess - start with the initial construction and perturb it
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Objective function: minimize the maximum overlap
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Constraint: total integral must be 1
    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Optimization using scipy.optimize
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': constraint}]
        result = optimize.minimize(objective, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective(h)
    
    # Start optimization
    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(best_h)
    
    # Try multiple initializations to avoid local optima
    for _ in range(3):
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        try:
            h_candidate, c5_candidate = optimize_h(h_initial)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        except:
            pass
        
        # Check if we are running out of time
        if time.time() - start_time > budget_s - 10:
            break
    
    # Try increasing the number of points if time permits
    if time.time() - start_time < budget_s - 10:
        n_points = 300
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Redefine objective and constraint for new n_points
        def objective_new(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        def constraint_new(h):
            h = np.clip(h, 0, 1)
            return np.sum(h) - n_points / 2
        
        # Optimization for larger n_points
        def optimize_h_new(h_initial):
            bounds = [(0, 1) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': constraint_new}]
            result = optimize.minimize(objective_new, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            h = np.clip(result.x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return h, objective_new(h)
        
        # Run the optimization
        try:
            h_candidate, c5_candidate = optimize_h_new(h_initial)
            if c5_candidate < best_c5:
                best_h = h_candidate
                best_c5 = c5_candidate
        except:
            pass
    
    return best_h, best_c5, n_points
