# Pre-imported functions
evaluate_erdos_solution = lambda h, n_points: max(np.correlate(h, 1 - h, mode="full") * (2.0 / n_points))
initial_h_values = np.array([0.5 for _ in range(100)])  # Example initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # Configuration for the search
    n_points_min, n_points_max = 200, 300
    steps = 5  # Number of sampling points between n_points_min and n_points_max
    n_points_list = np.linspace(n_points_min, n_points_max, steps, dtype=int).tolist()

    # Initial construction post-processing
    def refine_initial_h_values(h_initial, n_points):
        h_refined = np.copy(h_initial)
        for i in range(n_points):
            # Distribute value evenly
            h_refined[i] = 0.5
        return h_refined

    for n_points in n_points_list:
        dx = 2.0 / n_points
        h_initial = refine_initial_h_values(initial_h_values, n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial / np.sum(h_initial) * 1.0
        h_current = np.copy(h_initial)

        def objective(h_flat):
            h_flat = np.clip(h_flat, 0, 1)
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            h = h / total * 1.0
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x.reshape(n_points)) * dx - 1.0}]
        result = minimize(objective, h_current.flatten(), method='L-BFGS-B', bounds=bounds, constraints=constraints, tol=1e-5)

        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            total = np.sum(h_opt) * dx
            h_opt = h_opt / total * 1.0
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return (best_h, best_c5, len(best_h))
