import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, LinearConstraint
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 400  # 400 points provides better accuracy with reasonable performance
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0] * n_points)
    if hasattr(initial_h_values, 'copy'):
        initial_h_values = initial_h_values.copy()
    elif len(initial_h_values) > n_points:
        initial_h_values = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)

    # Ensure sum(h) * dx = 1
    h = initial_h_values[:n_points] if len(initial_h_values) >= n_points else initial_h_values
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # We use a genetic algorithm to explore the space of h functions
    # Define the individual (a vector of h values)
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    # Initialize the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    # Define the objective function
    def evaluate(individual):
        h_values = np.array(individual)
        # Rescale to ensure the total integral is 1
        h_values = h_values / (np.sum(h_values) * dx)
        h_values = np.clip(h_values, 0, 1)
        overlap = np.correlate(h_values, 1 - h_values, mode="full")
        c5 = np.max(overlap) * dx
        return (c5,)

    # Register the evaluation function
    toolbox.register("evaluate", evaluate)

    # Register the crossover and mutation operators
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Start the genetic algorithm
    pop = toolbox.population(n=50)
    stats = tools.Statistics(lambda ind: ind.fitness.values)
    stats.register("avg", np.mean)
    stats.register("std", np.std)
    stats.register("min", np.min)
    stats.register("max", np.max)

    # Track the elapsed time
    start_time = time.time()

    # Run the genetic algorithm
    for gen in range(100):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = tools.selBest(pop + offspring, len(pop))

        # Early stop if time is exceeded
        if time.time() - start_time > budget_s - 5:
            break

    # Get the best individual
    best_ind = tools.selBest(pop, 1)[0]
    best_h = best_ind
    best_c5 = best_ind.fitness.values[0]

    # Fine-tune the best solution with SLSQP to improve the bound
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    x0 = best_h
    result = optimize.minimize(
        objective,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    # Check if optimization succeeded
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    else:
        best_h = best_ind
        best_c5 = best_ind.fitness.values[0]

    return best_h, best_c5, n_points
