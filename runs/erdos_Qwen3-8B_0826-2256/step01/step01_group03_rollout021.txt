import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduce number of points for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Construct an initial search space
    def h_sampler(n_points, seed):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
        h = np.clip(h, 0, 1)
        h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

    # Use simulated annealing for optimization
    def simulate_annealing(n_points, budget_s, seed):
        initial_h = h_sampler(n_points, seed)
        best_h = initial_h
        best_c5 = objective_function(best_h)
        T = 1.0  # Initial temperature
        cooling_rate = 0.995
        iterations = 0
        start_time = time.time()

        while time.time() - start_time < budget_s:
            T = max(T * cooling_rate, 1e-6)
            for _ in range(100):
                # Generate a neighbor
                h = best_h.copy()
                idx = np.random.randint(n_points)
                h[idx] += np.random.choice([-0.05, 0.05])
                h = np.clip(h, 0, 1)
                h = h * (total_integral / np.sum(h))  # Maintain integral constraint
                c5 = objective_function(h)
                delta = c5 - best_c5
                if delta < 0 or np.random.rand() < np.exp(-delta / T):
                    best_h = h
                    best_c5 = c5
            iterations += 1

        return best_h, best_c5

    # Run the optimization
    best_h, best_c5 = simulate_annealing(n_points, budget_s, seed)

    return best_h, best_c5, n_points
