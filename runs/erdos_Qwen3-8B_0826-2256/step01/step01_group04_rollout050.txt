import scipy.optimize as optimize
import time

from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(140) * 2 - 1  # Placeholder, actual initial construction may vary

initial_h_values = np.clip(initial_h_values, 0, 1)

initial_h_values = initial_h_values / np.sum(initial_h_values) * (140 / 2)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Try different numbers of points
    for n_points in [140, 200, 260, 320, 380]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Initialize h with uniform distribution
        h = np.random.rand(n_points) * 2 - 1
        h = np.clip(h, 0, 2)
        h = (h - np.min(h)) / (np.max(h) - np.min(h))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2)  # Normalize to sum to n_points / 2
        h = np.round(h).astype(float)
        h = h / np.sum(h) * (n_points / 2)
        
        # Define objective and constraints for optimization
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            h = h_vec
            cor = correlate(h, 1 - h, mode="full")
            c5 = np.max(cor) * (2.0 / n_points)
            return c5
        
        # Constraints: sum(h) = n_points / 2
        def constraint(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return np.sum(h_vec) - (n_points / 2)
        
        # Optimization using SLSQP with bounds and constraints
        result = optimize.minimize(objective, h, method='SLSQP', bounds=[(0, 1) for _ in range(n_points)], constraints=[{'type': 'eq', 'fun': constraint}])
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.round(h_opt).astype(float)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
        
        # Evaluate

        c5 = np.max(cor) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        print(f"n_points={n_points}, c5={c5}")
    
    # Try local refinement
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perturb the best solution
