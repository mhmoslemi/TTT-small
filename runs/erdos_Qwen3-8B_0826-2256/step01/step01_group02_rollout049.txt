def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 200  # Start with 200 points
    population_size = 50
    generations = 100
    crossover_prob = 0.8
    mutation_prob = 0.1
    tournament_size = 3
    
    # Define the fitness function
    def evaluate(individual):
        h = np.array(individual)
        h = np.clip(h, 0, 1)
        # Normalize to ensure integral constraint
        h_norm = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h_norm, 1 - h_norm, mode="full")
        return np.max(corr * (2.0 / n_points)),  # Minimize the maximum value
    
    # Initialize the genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)
    
    toolbox = base.Toolbox()
    toolbox.register("attr_float", lambda: np.random.uniform(0, 1))
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=tournament_size)
    
    # Initial population
    pop = toolbox.population(n=population_size)

    start_time = time.time()

        offspring = list(map(toolbox.evaluate, offspring))
        pop = tools.selTournament(pop + offspring, len(pop), fit_attr="fitness")
        
        # Check the best solution
        best_ind = min(pop, key=lambda ind: ind.fitness.values[0])
        current_c5 = best_ind.fitness.values[0]
