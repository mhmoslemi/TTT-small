import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduced for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2
    
    # Initial guess using provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial * (total_integral / np.sum(h_initial))
    h_initial = np.clip(h_initial, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Optimizer settings
    def optimize_h_values(h_initial, budget_s):
        best_h = h_initial.copy()
        best_c5 = objective_function(best_h)
        best_h_val = best_h.copy()
        
        start_time = time.time()
        for _ in range(int(budget_s / 0.1)):
            # Generate random perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_candidate = h_initial + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (total_integral / np.sum(h_candidate))
            
            c5_candidate = objective_function(h_candidate)
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                best_h_val = best_h.copy()
        
        return best_h_val, best_c5
    
    # Run optimization
    best_h, best_c5 = optimize_h_values(h_initial, budget_s)
    
    return best_h, best_c5, n_points
