```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([...])  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Try different number of points
    n_points_list = [200, 300, 400]
    
    best_h = None
    best_c5 = float('inf')
    
    # Start with the initial guess
    n_points = len(initial_h_values)
    h = initial_h_values.copy()
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h
    
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on uniform distribution
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the problem with SLSQP
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.minimize(
            objective,
            h,
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint},
            options={'maxiter': 100, 'ftol': 1e-8, 'eps': 1e-8}
        )
        
        if result.success:
            h = result.x
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h
        
        # Final refinement with CVXPY
        if time.time() - start_time < budget_s:
            h = cp.Variable(n_points)
            constraints = [cp.sum(h) == n_points / 2, 0 <= h <= 1]
            
            def cvxpy_objective(h_flat):
                h = h_flat.reshape(n_points)
                corr = np.correlate(h, 1 - h, mode="full")
                return np.max(corr) * (2.0 / n_points)
            
            prob = cp.Problem(cp.Minimize(cvxpy_objective(h)), constraints)
            try:
                prob.solve(solver=cp.SCS, eps=1e-4)
                best_h = h.value
                best_c5 = evaluate_erdos_solution(best_h)
            except:
                pass
    
    return best_h, best_c5, len(best_h)
```
