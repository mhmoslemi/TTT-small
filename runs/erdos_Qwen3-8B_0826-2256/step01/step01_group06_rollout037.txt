$$
C_5 = \max_k \int h(x)(1 - h(x+k)) dx
$$

under the constraint $ \int_0^2 h(x) dx = 1 $.

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [200, 300, 400]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for n_points in n_points_list:
        dx = 2.0 / n_points
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            # Normalize the initial guess
            h = h / total * 1.0

        # Define the objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Normalize h to satisfy the integral constraint
            total = np.sum(h) * dx
            h = h / total * 1.0
            # Evaluate the maximum overlap
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Define constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = []

        # Penalty for violating the integral constraint
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            return total - 1.0

        constraints.append({'type': 'eq', 'fun': constraint_func})

        # Optimization
        result = minimize(
            objective,
            h.flatten(),
            method='SLSQP',
            constraints=constraints,
            bounds=bounds,
            options={'ftol': 1e-6, 'maxiter': 150}
        )

        if result.success:
            h_opt = result.x.reshape(n_points)
