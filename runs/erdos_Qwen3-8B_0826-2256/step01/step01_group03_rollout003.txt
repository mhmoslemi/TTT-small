```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduce number of points for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2
    
    # Start from initial_h_values if provided
    h = initial_h_values.copy()
    h = np.clip(h, 0, 1)
    h = h * (total_integral / np.sum(h))  # Ensure integral constraint
    
    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def simulate_search(n_points, budget_s, seed):
        best_h = h.copy()
        best_c5 = objective_function(best_h)
        start_time = time.time()
        
        # Define a function that generates random perturbations to h
        def perturb_h(h, seed):
            np.random.seed(seed)
            perturbation = np.random.normal(0, 0.05, size=h.shape)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (total_integral / np.sum(h_perturbed))
            return h_perturbed
        
        # Use a population-based approach with parallel evaluation
        pool = Pool(processes=cpu_count())
        population_size = 20
        population = [perturb_h(best_h, seed + i) for i in range(population_size)]
        best_c5 = np.inf
        
        while time.time() - start_time < budget_s:
            # Evaluate the population
            results = pool.map(objective_function, population)
            best_index = np.argmin(results)
            best_c5 = results[best_index]
            best_h = population[best_index]
            
            # Generate new population by perturbing the best solution
            new_population = [perturb_h(best_h, seed + i) for i in range(population_size)]
            population = new_population
        
        pool.close()
        pool.join()
        return best_h, best_c5
    
    # Run the simulation
    best_h, best_c5 = simulate_search(n_points, budget_s, seed)
    
    return best_h, best_c5, n_points
```
