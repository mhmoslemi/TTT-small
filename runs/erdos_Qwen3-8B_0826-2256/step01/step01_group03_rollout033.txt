import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Reduce number of points for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2
    
    # Define initial guess based on provided initial_h_values
    h_initial = np.copy(initial_h_values)
    
    # Objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Constraint: integral of h over [0,2] equals 1
    def integral_constraint(h_values):
        return np.sum(h_values) - total_integral

    # SLSQP optimization
    def optimize_h(n_points, n_iter=50, seed=42):
        np.random.seed(seed)
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
        h = np.clip(h, 0, 1)
        h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        
        # Define optimization variables
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(objective_function(h_var))
        constraints = [cp.sum(h_var) == total_integral, 0 <= h_var, h_var <= 1]
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-6)
        
        if result is None:
            return h
        return h_var.value

    # Run optimization
    best_h = optimize_h(n_points, n_iter=50, seed=seed)
    best_c5 = objective_function(best_h)
    
    # Refine with random restarts within the budget
    start_time = time.time()
    for i in range(5):
        if time.time() - start_time > budget_s:
            break
        h = optimize_h(n_points, n_iter=10, seed=seed + i)
        c5 = objective_function(h)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    
    return best_h, best_c5, n_points
