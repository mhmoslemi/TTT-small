import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(140) * 2 - 1  # Placeholder, actual initial construction may vary
initial_h_values = np.clip(initial_h_values, 0, 2)
initial_h_values = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (140 / 2)  # Normalize to sum to 140/2
initial_h_values = np.round(initial_h_values).astype(float)
initial_h_values = initial_h_values / np.sum(initial_h_values) * (140 / 2)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Try different numbers of points
    for n_points in [140, 180, 220, 260]:
        if time.time() - start_time > budget_s - 10:
            break
        
        # Initialize h with a more structured pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * (n_points / 2) / np.sum(h)
        h = np.round(h).astype(float)
        h = h / np.sum(h) * (n_points / 2)
        
        # Define the objective function
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            h = h_vec
            # Compute the max overlap
            cor = correlate(h, 1 - h, mode="full")
            c5 = np.max(cor) * (2.0 / n_points)
            return c5
        
        # Optimization using differential evolution
        bounds = [(0, 1) for _ in range(n_points)]
        result = optimize.differential_evolution(objective, bounds, strategy='best1bin', maxiter=50, popsize=20, tol=1e-6, mutation=(0.5, 1), recombination=0.8, seed=seed)
        h_opt = result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        h_opt = np.round(h_opt).astype(float)
        h_opt = h_opt / np.sum(h_opt) * (n_points / 2)
        
        # Evaluate
        cor = correlate(h_opt, 1 - h_opt, mode="full")
        c5 = np.max(cor) * (2.0 / n_points)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_opt
        
        print(f"n_points={n_points}, c5={c5}")
    
    # Try local refinement
    for _ in range(10):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perturb the best solution
        perturbation = np.random.normal(0, 0.02, len(best_h))
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (len(best_h) / 2) / np.sum(perturbed_h)
        perturbed_h = np.round(perturbed_h).astype(float)
        perturbed_h = perturbed_h / np.sum(perturbed_h) * (len(best_h) / 2)
        
        cor = correlate(perturbed_h, 1 - perturbed_h, mode="full")
        c5 = np.max(cor) * (2.0 / len(best_h))
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h
    
    return best_h, best_c5, len(best_h)
