    np.random.seed(seed)
    n_points_list = [100, 150, 200]  # Test with smaller n_points to stay within time limits
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for n_points in n_points_list:
        dx = 2.0 / n_points
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            h = h / total * 1.0

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            h = h / total * 1.0
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = []

        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            return total - 1.0

        constraints.append({'type': 'eq', 'fun': constraint_func})

        result = minimize(
            objective,
            h.flatten(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-6, 'maxiter': 100}
        )

            h_opt = result.x.reshape(n_points)
