```python
import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate
import random

# Pre-imported functions
evaluate_erdos_solution = lambda h, n_points: max(np.correlate(h, 1 - h, mode="full") * (2.0 / n_points))
initial_h_values = np.array([0.5 for _ in range(100)])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 150, 200, 250, 300]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # Start with the initial guess
    h = initial_h_values.copy()
    if len(h) > 200:
        h = h[:200]
    h = np.clip(h, 0, 1)
    total = np.sum(h) * (2.0 / len(h))
    h = h / total * 1.0
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * (2.0 / len(h))
    best_c5 = c5
    best_h = h.copy()

    for n_points in n_points_list:
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h) * (2.0 / n_points)
        h = h / total * 1.0

        def objective(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * (2.0 / n_points)
            h = h / total * 1.0
            corr = correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)

        bounds = [(0.0, 1.0) for _ in range(n_points)]

        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * (2.0 / n_points)
            return total - 1.0

        result = minimize(
            objective,
            h.flatten(),
            method='SLSQP',
            bounds=bounds,
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-6, 'maxiter': 100}
        )

        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            total = np.sum(h_opt) * (2.0 / n_points)
            h_opt = h_opt / total * 1.0
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr) * (2.0 / n_points)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()

        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return (best_h, best_c5, len(best_h))
```
