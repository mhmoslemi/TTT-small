def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 100 points for faster optimization
    n_points = 100
    best_c5 = 1.0  # Initialize with a high value
    
    # Initial guess from initial construction
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    def optimize_h(h_initial, n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(x):
            h = np.clip(x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        def constraint_func(x):
            h = np.clip(x, 0, 1)
            return np.sum(h) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective_func(h)
    
    # Initial optimization
    start_time = time.time()
    h_opt, c5_opt = optimize_h(h_initial, n_points)
    best_c5 = c5_opt
    best_h = h_opt
    
    # Try increasing number of points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def optimize_h_new(h_initial, n_points):
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective_func(x):
                h = np.clip(x, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                corr = signal.correlate(h, 1 - h, mode="full")
                return np.max(corr * (2.0 / n_points))
            
            def constraint_func(x):
                h = np.clip(x, 0, 1)
                return np.sum(h) - n_points / 2
            
            cons = [{'type': 'eq', 'fun': constraint_func}]
            result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            
            h = np.clip(result.x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return h, objective_func(h)
        
        h_opt_new, c5_opt_new = optimize_h_new(h_initial, n_points)
        if c5_opt_new < best_c5:
            best_c5 = c5_opt_new
            best_h = h_opt_new
    
    return best_h, best_c5, n_points
