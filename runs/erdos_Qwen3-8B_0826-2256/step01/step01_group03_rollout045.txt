import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from functools import partial
from multiprocessing import Pool, cpu_count

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased number of points for better resolution
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define a function that returns the best solution found so far
    def get_best_solution():
        best_h = np.copy(initial_h_values)
        best_c5 = objective_function(best_h)
        return best_h, best_c5

    # Initialize with the initial solution
    best_h, best_c5 = get_best_solution()

    # Use a particle swarm optimization approach to find better solutions
    def optimize_h(h_initial):
        h = np.copy(h_initial)
        for _ in range(100):  # Number of iterations
            # Perturb h slightly
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h += perturbation
            h = np.clip(h, 0, 1)
            # Scale h to ensure the integral constraint
            h = h * (total_integral / np.sum(h))
            # Evaluate the objective
            c5 = objective_function(h)
            # Update the best solution if this is better
            if c5 < best_c5:
                best_h = np.copy(h)
                best_c5 = c5
        return best_h, best_c5

    # Run the optimization
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Run optimization with a slightly perturbed initial guess
        h_initial = np.copy(best_h) + np.random.normal(0, 0.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (total_integral / np.sum(h_initial))
        optimize_h(h_initial)

    return best_h, best_c5, n_points
