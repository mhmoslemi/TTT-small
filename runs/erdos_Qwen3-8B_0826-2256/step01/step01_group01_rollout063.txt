    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # sum(h) == n_points / 2

        h_start = h.copy()

        h_rand = np.random.rand(n_points)
        h_rand = h_rand / (np.sum(h_rand) * dx)
        h_rand = np.clip(h_rand, 0, 1)
        # Try PCA projection
        pca = PCA(n_components=1)
        h_pca = pca.fit_transform(h_start.reshape(-1, 1))
        h_pca = h_pca.reshape(-1)
        h_pca = h_pca / (np.sum(h_pca) * dx)
        h_pca = np.clip(h_pca, 0, 1)

        h_smooth = np.convolve(h_start, np.ones(5)/5, mode='same')
        h_smooth = h_smooth / (np.sum(h_smooth) * dx)

        candidates = [h_start, h_rand, h_pca, h_smooth]

    result = minimize(
