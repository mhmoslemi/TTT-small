def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 150, 200, 250, 300, 350, 400]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Define functions for evaluating and mutating solutions
    def evaluate(h):
        corr = correlate(h, 1 - h, mode="full")
        return np.max(corr) * (2.0 / len(h))
    
    def normalize(h):
        total = np.sum(h) * (2.0 / len(h))
        return h / total * 1.0
    
    def mutate(h, mutation_rate=0.1):
        h = h.copy()
        for i in range(len(h)):
            if np.random.random() < mutation_rate:
                h[i] = np.clip(np.random.random(), 0, 1)
        return normalize(h)
    
    def crossover(parent1, parent2):
        child = np.zeros(len(parent1))
        for i in range(len(parent1)):
            if np.random.random() < 0.5:
                child[i] = parent1[i]
            else:
                child[i] = parent2[i]
        return normalize(child)

    population_size = 20
    population = [normalize(np.random.random(len(initial_h_values))) for _ in range(population_size)]
