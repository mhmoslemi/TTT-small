import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([...])  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Try different number of points
    n_points_list = [200, 300, 400]
    
    best_h = None
    best_c5 = float('inf')
    best_n_points = 200
    
    # Use initial guess
    n_points = 200
    h_initial = initial_h_values.copy()
    h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # We use a more refined optimization approach with Adam optimizer
    def optimize_h(h_initial, n_points, budget_s):
        h = h_initial.copy()
        h = h / np.sum(h) * (n_points / 2)
        h_flat = h.copy()
        learning_rate = 0.01
        decay = 0.999
        eps = 1e-8
        m = np.zeros_like(h_flat)
        v = np.zeros_like(h_flat)
        t = 0
        
        for _ in range(1000):
            if time.time() - start_time > budget_s:
                break
            
            t += 1
            # Compute the overlap integral
            corr = np.correlate(h, 1 - h, mode="full")
            max_corr = np.max(corr) * (2.0 / n_points)
            
            # Compute gradient
            gradient = np.zeros(n_points)
            for i in range(n_points):
                h_i = h[i]
                for k in range(-n_points + 1, n_points):
                    if 0 <= i + k < n_points:
                        term = (1 - h[i + k]) * (2.0 / n_points)
                        gradient[i] -= term * (2.0 / n_points)
            
            # Update Adam optimizer
            grad = gradient
            m = decay * m + (1 - decay) * grad
            v = decay * v + (1 - decay) * (grad ** 2)
            unbiased_m = m / (1 - decay ** t)
            unbiased_v = v / (1 - decay ** t)
            h_flat -= learning_rate * unbiased_m / (np.sqrt(unbiased_v) + eps)
            h_flat = np.clip(h_flat, 0, 1)
            h_flat = h_flat * (n_points / 2) / np.sum(h_flat)
            
            h = h_flat.copy()
        
        return h
    
    # Try different number of points
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess
        if n_points == 200:
            h = optimize_h(h_initial, n_points, budget_s)
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
            h = optimize_h(h, n_points, budget_s)
        
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points
    
    # Final refinement with CVXPY
    if time.time() - start_time < budget_s:
        n_points = len(best_h)
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h <= 1]
        
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        # Use an approximate objective for CVXPY
        def cvxpy_objective(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        prob = cp.Problem(cp.Minimize(cvxpy_objective(h)), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            best_h = h.value
        except:
            pass
    
    return best_h, best_c5, len(best_h)
