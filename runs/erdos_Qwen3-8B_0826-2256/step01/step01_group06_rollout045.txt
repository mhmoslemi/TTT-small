    n_points = 200  # Use a moderate number of points to keep within the budget
    dx = 2.0 / n_points

    initial_h = np.clip(initial_h, 0, 1)
    total = np.sum(initial_h) * dx
    if abs(total - 1.0) > 1e-6:
        initial_h = initial_h / total * 1.0

    population_size = 50  # Number of individuals in the population
    generations = 100  # Maximum number of generations to run
    mutation_rate = 0.1  # Probability of mutation
    crossover_rate = 0.8  # Probability of crossover
    elitism = 2  # Keep the best individuals for the next generation
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)

    for generation in range(generations):
        # Evaluate fitness of the current population
        population = [initial_h.copy() for _ in range(population_size)]
        for i in range(population_size):
            population[i] = np.clip(population[i], 0, 1)
            total = np.sum(population[i]) * dx
            population[i] = population[i] / total * 1.0

        # Evaluate fitness for each individual
        fitness = []
        for ind in population:
            c5 = evaluate_erdos_solution(ind, n_points)
            fitness.append((c5, ind))

        # Sort by fitness (lower is better)
        fitness.sort()
        fitness = [f[1] for f in fitness]

        # Select the best individuals for the next generation
        next_generation = fitness[:elitism]

        # Create the rest of the population
        while len(next_generation) < population_size:
            # Select two parents
            parent1 = random.choice(fitness)
            parent2 = random.choice(fitness)

                child = np.zeros_like(parent1)

                    if random.random() < 0.5:
                        child[i] = parent1[i]

                        child[i] = parent2[i]

                child = parent1.copy()

            child = child / total_child * 1.0
