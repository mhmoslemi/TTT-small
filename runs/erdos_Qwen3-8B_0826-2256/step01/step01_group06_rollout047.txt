def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [150, 200, 250, 300, 350, 400]  # Test with various n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    # Define initial solution
    def initial_solution(n_points):
        h = np.zeros(n_points)
        # First half filled with 0.5, then some fluctuations
        h[:n_points//2] = 0.5
        for i in range(n_points//2, n_points):
            if i % 3 == 0:
                h[i] = 0.7
            elif i % 3 == 1:
                h[i] = 0.3
            else:
                h[i] = 0.5
        total = np.sum(h) * 2.0 / n_points
        h = h / total * 1.0
        return h

    for n_points in n_points_list:
        dx = 2.0 / n_points
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = initial_solution(n_points)

        # Normalize initial guess to satisfy integral constraint
        total = np.sum(h) * dx
        h = h / total * 1.0

        # Define the objective function
        def objective(h_flat):
            h = h_flat.reshape(n_points)
            # Normalize h to satisfy the integral constraint
            total = np.sum(h) * dx
            h = h / total * 1.0
            # Evaluate the maximum overlap
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Define constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = []

        # Penalty for violating the integral constraint
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            return total - 1.0

        constraints.append({'type': 'eq', 'fun': constraint_func})

        # Optimization
        result = minimize(
            objective,
            h.flatten(),
            method='SLSQP',
            constraints=constraints,
            bounds=bounds,
            options={'ftol': 1e-6, 'maxiter': 100, 'disp': False}
        )

        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            # Normalize to ensure integral constraint
            total = np.sum(h_opt) * dx
            h_opt = h_opt / total * 1.0
            # Evaluate the result
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return (best_h, best_c5, len(best_h))
