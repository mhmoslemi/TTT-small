def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 200, 300, 400]  # Test with n_points=100, 200, 300, 400
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()

    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Initialize h with the initial_h_values, clamped to [0, 1]
        h = np.clip(initial_h_values[:n_points], 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            # Normalize the initial guess
            h = h / total * 1.0
        # Create bounds and initial guess
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        initial_guess = h.copy()

        # Define cost function
        def cost(h_flat):
            h = h_flat.reshape(n_points)
            # Normalize to ensure integral constraint
            total = np.sum(h) * dx
            h = h / total * 1.0
            corr = correlate(h, 1 - h, mode="full")
            return np.max(corr) * dx

        # Define constraints
        constraints = []
        def constraint_func(h_flat):
            h = h_flat.reshape(n_points)
            total = np.sum(h) * dx
            return total - 1.0
        constraints.append({'type': 'eq', 'fun': constraint_func})

        # Run differential evolution
        result = differential_evolution(
            cost,
            bounds,
            constraints=constraints,
            strategy='best1bin',
            popsize=15,
            mutates=0.2,
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            seed=seed,
        )

        if result.success:
            h_opt = result.x.reshape(n_points)
            h_opt = np.clip(h_opt, 0, 1)
            # Normalize to ensure integral constraint
            total = np.sum(h_opt) * dx
            h_opt = h_opt / total * 1.0
            # Evaluate the result
            corr = correlate(h_opt, 1 - h_opt, mode="full")
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt

        # Check if budget is exceeded
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return (best_h, best_c5, len(best_h))
