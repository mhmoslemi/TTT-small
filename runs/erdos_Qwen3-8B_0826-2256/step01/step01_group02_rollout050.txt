def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try multiple configurations with different numbers of points
    best_h = None
    best_c5 = float('inf')
    n_points_list = [100, 150, 200]
    
    for n_points in n_points_list:
        # Initial guess - use the initial construction and perturb it
        h_initial = np.copy(initial_h_values[:n_points])
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define objective function
        def objective(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        # Define constraints
        def constraint(h):
            h = np.clip(h, 0, 1)
            return np.sum(h) - n_points / 2
        
        # Define the optimization problem with SLSQP
        def optimize_h(h_initial):
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective_func(x):
                return objective(x)
            
            def constraint_func(x):
                return constraint(x)
            
            cons = [{'type': 'eq', 'fun': constraint_func}]
            result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
            
            h = np.clip(result.x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            return h, objective_func(h)
        
        start_time = time.time()
        try:
            h, c5 = optimize_h(h_initial)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
            if time.time() - start_time > budget_s - 10:
                break
        except:
            pass
    
    return best_h, best_c5, n_points
