import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.array([...])  # Placeholder for initial values

def run(seed=42, budget_s=100, **kwargs):
    start_time = time.time()
    np.random.seed(seed)
    
    # Try different number of points
    n_points_list = [200, 400, 600]
    
    best_h = None
    best_c5 = float('inf')
    
    # Start with the initial guess
    for n_points in n_points_list:
        if time.time() - start_time > budget_s:
            break
        
        # Generate initial guess based on initial_h_values if available
        if n_points == len(initial_h_values):
            h = initial_h_values.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
            h = h / np.sum(h) * (n_points / 2)
        
        # Optimization using a more sophisticated genetic algorithm
        def evaluate(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        def constraint(h_flat):
            h = h_flat.reshape(n_points)
            return np.sum(h) - (n_points / 2)
        
        # Genetic algorithm parameters
        population_size = 20
        generations = 100
        mutation_rate = 0.1
        crossover_rate = 0.8
        
        # Initial population
        population = np.random.uniform(0, 1, (population_size, n_points))
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        
        best_fitness = float('inf')
        best_h_pop = None
        
        for gen in range(generations):
            if time.time() - start_time > budget_s:
                break
            
            # Evaluate fitness
            fitness = np.array([evaluate(ind) for ind in population])
            best_fitness = np.min(fitness)
            best_h_pop = population[np.argmin(fitness)]
            
            # Select parents
            parents = population[np.argsort(fitness)[:int(population_size / 2)]]
            
            # Crossover
            offspring = []
            for _ in range(population_size - len(parents)):
                parent1, parent2 = np.random.choice(parents, 2, replace=False)
                mask = np.random.rand(n_points) < crossover_rate
                child = np.where(mask, parent1, parent2)
                offspring.append(child)
            
            # Mutate
            for i in range(len(offspring)):
                offspring[i] = np.clip(offspring[i] + np.random.normal(0, 0.05, n_points), 0, 1)
                offspring[i] = offspring[i] * (n_points / 2) / np.sum(offspring[i])
            
            # Replace population
            population = np.vstack((parents, offspring))
            population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        
        h = best_h_pop
        c5 = evaluate_erdos_solution(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Final refinement with CVXPY
    if time.time() - start_time < budget_s:
        n_points = len(best_h)
        h = cp.Variable(n_points)
        constraints = [cp.sum(h) == n_points / 2, 0 <= h <= 1]
        
        def objective_func(h_flat):
            h = h_flat.reshape(n_points)
            corr = np.correlate(h, 1 - h, mode="full")
            return np.max(corr) * (2.0 / n_points)
        
        prob = cp.Problem(cp.Minimize(objective_func(h)), constraints)
        try:
            prob.solve(solver=cp.SCS, eps=1e-4)
            best_h = h.value
        except:
            pass
    
    return best_h, best_c5, len(best_h)
