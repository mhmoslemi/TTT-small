import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Use the initial_h_values if provided, otherwise use a random initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 5  # Number of trials to run for exploration

    # Define linear constraint for integral
    integral_constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

    # Perform multiple optimization trials with different initializations
    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the initial point to explore nearby regions
        h = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Run optimization with multiple methods
        try:
            # First with SLSQP
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=[
                    {"type": "eq", "fun": constraints},
                    integral_constraint
                ],
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

            # Second with COBYLA for more robust exploration
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=[
                    {"type": "eq", "fun": constraints},
                    integral_constraint
                ],
                method="COBYLA",
                options={"maxiter": 100}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            # Handle exceptions gracefully
            pass

    return best_h, best_c5, n_points
