import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300  # Opt for fewer points to save time and memory
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with a piecewise constant function with alternating values
    h_initial = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            h_initial[i] = 1.0
        elif i % 3 == 1:
            h_initial[i] = 0.5
        else:
            h_initial[i] = 0.0

    # Normalize to meet the integral constraint
    h_initial = h_initial * (total_integral / np.sum(h_initial))

    # Define the objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Optimization setup for SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with improved parameters for efficiency and convergence
    result = opt.minimize(
        fun=objective_function,
        x0=h_initial,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300, 'disp': False}
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    # Ensure that the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    # Return the best solution found within the time budget
    return best_h, final_c5, n_points
