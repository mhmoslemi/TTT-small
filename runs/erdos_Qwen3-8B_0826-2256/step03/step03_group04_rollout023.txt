# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Experiment with fewer points to speed up computation
    n_points = 400  # Reduced from 500 for faster evaluation
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with an initial guess based on a periodic pattern
    h = np.linspace(0, 2, n_points)
    # Construct a piecewise-constant function with periodicity to balance the integral
    h_vals = np.zeros(n_points)
    for i in range(n_points):
        # Create a periodic pattern that alternates between high and low
        if i % 2 == 0:
            h_vals[i] = 1.0
        else:
            h_vals[i] = 0.0
    h_vals = h_vals * (total_integral / np.sum(h_vals))  # Normalize

    # Refine the initial pattern with a more structured approach
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Optimization setup for L-BFGS-B
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with a more aggressive initial guess
    result = opt.minimize(
        fun=objective_function,
        x0=h_vals,
        method='L-BFGS-B',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-8, 'gtol': 1e-8, 'maxiter': 500}
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    # Ensure that the solution is valid and meets the constraint
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
