        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx

                h[i] = 0.75

                h[i] = 1.0

        # Apply slight random perturbation for exploration
        h = h + 0.05 * np.random.randn(n_points)

            corr = np.correlate(x, 1 - x, mode="full")

        result = opt.minimize(
