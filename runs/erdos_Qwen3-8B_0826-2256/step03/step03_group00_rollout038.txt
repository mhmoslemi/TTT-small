import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (starting from the provided initial_h_values)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Use a secondary optimization with different starting points
    # Generate a few perturbations of the current best solution
    perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
    for pert in perturbations:
        perturbed_h = best_h + pert
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        # Optimize the perturbed solution
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Use a local search with simulated annealing for refinement
    def simulated_annealing(h_start, temp=1.0, temp_min=1e-6, cooling_rate=0.995, iterations=1000):
        h_current = np.copy(h_start)
        best_h = np.copy(h_current)
        current_c5 = objective_function(h_current)
        best_c5 = current_c5

        for i in range(iterations):
            # Generate a random perturbation
            perturbation = np.random.normal(0, 0.05, size=n_points)
            h_candidate = h_current + perturbation
            h_candidate = np.clip(h_candidate, 0.0, 1.0)
            h_candidate = h_candidate * (total_integral / np.sum(h_candidate))

            candidate_c5 = objective_function(h_candidate)
            delta = candidate_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                h_current = h_candidate
                current_c5 = candidate_c5
                if current_c5 < best_c5:
                    best_h = h_current
                    best_c5 = current_c5
            temp *= cooling_rate
            if temp < temp_min:
                break

        return best_h, best_c5

    # Perform simulated annealing on the best solution
    best_h, best_c5 = simulated_annealing(best_h)
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
