import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define the number of points and initialize the function
    n_points = 100  # Starting with 100 points for balance between speed and quality
    dx = 2.0 / n_points

        h = initial_h_values.copy()
    else:
        # Random initialization
        h = np.random.rand(n_points)
        # Normalize to ensure integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    elapsed_time = 0
    
    # Try different optimization methods and strategies
    # 1. Differential Evolution for global search
    def differential_evolution_run():
        nonlocal best_h, best_c5
        result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            disp=False,
            polish=True,
            polish_bounds=bounds,
            constraints={"type": "eq", "fun": constraints}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # 2. Local optimization with SLSQP to refine the solution
    def local_optimization_run(h_initial):
        nonlocal best_h, best_c5
        result = minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    # 3. Try multiple initializations with perturbations
    def perturb_h(h, magnitude=0.1):
        return np.clip(h + np.random.uniform(-magnitude, magnitude, size=h.shape), 0, 1)
    
    # Perform the optimization process within the time budget
    while elapsed_time < budget_s - 1:
        # Differential evolution for global search
        differential_evolution_run()
        
        # Local optimization to refine the solution
        local_optimization_run(best_h)

        perturbed_h = perturb_h(best_h, magnitude=0.05)
        # Normalize to ensure the integral is 1
        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) if np.sum(perturbed_h) != 0 else perturbed_h
        perturbed_h = np.clip(perturbed_h, 0, 1)
        local_optimization_run(perturbed_h)
        
        # Check time
        elapsed_time = time.time() - start_time
    
    # Return the best solution found
    return best_h, best_c5, n_points
