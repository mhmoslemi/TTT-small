```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from scipy.optimize import Bounds

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use a reasonably large number to balance accuracy and speed
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Enforce the integral constraint

    def objective(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_sum(h_values):
        return np.sum(h_values) - total_integral

    # Use the initial guess provided
    h_initial = initial_h_values[:n_points].copy()
    h_initial = h_initial * (total_integral / np.sum(h_initial))
    h_initial = np.clip(h_initial, 0, 1)

    # Define the optimization problem
    h = cp.Variable(n_points, name='h')
    objective_func = objective(h)
    constraints = [cp.sum(h) == total_integral, 0 <= h, h <= 1]

    def optimize_with_bounds(h_initial):
        # Use SLSQP for optimization
        def objective_func(x):
            return objective(x)

        result = opt.minimize(
            objective_func,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x, result.fun

    def run_with_cvxpy():
        # Use CVXPY for a more structured optimization
        objective_func = cp.Minimize(objective(h))
        prob = cp.Problem(objective_func, constraints)
        result = prob.solve(solver=cp.SCS, eps=1e-5, max_iter=100)
        return h.value, objective_func.value

    # Start with the initial guess
    best_h = h_initial
    best_c5 = objective(best_h)

    # Start timer
    start_time = time.time()

    # First optimization
    try:
        optimized_h, optimized_c5 = optimize_with_bounds(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Run CVXPY optimization
    try:
        optimized_h, optimized_c5 = run_with_cvxpy()
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Perturb and optimize
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        for _ in range(3):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize_with_bounds(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # Final refinement
    if remaining_time > 5:
        perturbed_h = best_h + 0.03 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = optimize_with_bounds(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    return best_h, best_c5, n_points
```
