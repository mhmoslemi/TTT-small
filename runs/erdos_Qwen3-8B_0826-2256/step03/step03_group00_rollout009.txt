import time

from concurrent.futures import ThreadPoolExecutor, as_completed
from deap import base, creator, tools, algorithms

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):

    max_workers = 10
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100

    def fitness_func(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * (2.0 / len(h_values))
        return max_overlap,

    # Define the constraint function
    def constraint_func(h_values):
        return np.sum(h_values) - len(h_values) / 2

    # Set up genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points_options[0])
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", fitness_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("constraint", constraint_func)

    # Define the run for each n_points
    def run_n_points(n_points):
        dx = 2.0 / n_points
        total_integral = n_points / 2

        # Define the population
        pop = toolbox.population(n=50)
        # Evaluate the entire population
        fitness = list(map(toolbox.evaluate, pop))
        for ind, fit in zip(pop, fitness):
            ind.fitness.values = fit
        # CMA-ES for optimization
        result = opt.minimize(
            fun=lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * dx,
            x0=np.array([individual.fitness.values[0] for individual in pop]),
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda h: np.sum(h) - total_integral}

        best_h = result.x
        best_c5 = result.fun
        return best_h, best_c5
