```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use more points for better resolution and tighter bounds
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values: use the provided initial_h_values and normalize
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Perturbations for diversity of initial guesses
    perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
    for pert in perturbations:
        perturbed_h = best_h + pert
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Use a more advanced optimization approach with differential evolution
    def objective_for_de(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Bounds for differential evolution
    bounds_de = [(0.0, 1.0) for _ in range(n_points)]
    result_de = opt.differential_evolution(
        objective_for_de,
        bounds_de,
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-9,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False,
        seed=seed
    )
    de_h = result_de.x
    de_c5 = result_de.fun

    if de_c5 < best_c5:
        best_h = de_h
        best_c5 = de_c5

    # Final evaluation
    final_h = np.clip(best_h, 0.0, 1.0)
    final_h = final_h * (total_integral / np.sum(final_h))
    final_c5 = evaluate_erdos_solution(final_h)

    return final_h, final_c5, n_points
```
