import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define a range of n_points to try, balancing speed and accuracy
    n_points_list = [200, 300, 400]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 200
    
    # Use the initial construction as a starting point
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Function to normalize and clip h_values
    def normalize_and_clip(h_values, n_points):
        dx = 2.0 / n_points
        sum_h = np.sum(h_values) * dx
        h_values = h_values / (sum_h / dx)
        h_values = np.clip(h_values, 0, 1)
        return h_values
    
    # Function to optimize on a given n_points
    def optimize_for_n_points(n_points):
        dx = 2.0 / n_points
        x_initial = np.linspace(0, 2, len(initial_h))
        x_target = np.linspace(0, 2, n_points)
        h_initial = np.interp(x_target, x_initial, initial_h)
        h_initial = normalize_and_clip(h_initial, n_points)
        
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimization function
        def objective(h_values):
            return evaluate_overlap(h_values)
        
        # Constraint function
        def constraint(h_values):
            return np.sum(h_values) - n_points / 2
        
        # Run the optimization
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100, "eps": 1e-8}
        )
        
        # Return the solution and the value
        return result.x, result.fun
    
    # Run the optimization for each n_points in the list
    for n_points in n_points_list:
        start_time = time.time()
        h, c5 = optimize_for_n_points(n_points)
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break  # Stop if budget is exceeded
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
            best_n_points = n_points
    
    # Return the best solution found
    return best_h, best_c5, best_n_points
