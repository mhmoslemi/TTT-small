import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between speed and accuracy
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Define the objective function for optimization
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Generate initial guess based on the provided initial_h_values
    def initial_guess(n_points, seed=42):
        # Use the provided initial_h_values as a starting point
        h_initial = initial_h_values[:n_points].copy()
        h_initial = h_initial * (total_integral / np.sum(h_initial))
        # Apply simple smoothing to promote even distribution
        h_initial = np.convolve(h_initial, np.ones(5) / 5, mode='same')
        h_initial = np.clip(h_initial, 0, 1)
        return h_initial

    # Define the optimization function
    def optimize_with_slsqp(h_initial):
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        # Optimization settings
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start the timer
    start_time = time.time()

    # Run the optimization
    try:
        optimized_h, optimized_c5 = optimize_with_slsqp(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Check remaining time and consider additional perturbations and optimizations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Perturb and optimize a few times
        for _ in range(3):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize_with_slsqp(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # If still not optimal, refine with another round
    if remaining_time > 5:
        perturbed_h = best_h + 0.03 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = optimize_with_slsqp(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    return best_h, best_c5, n_points
