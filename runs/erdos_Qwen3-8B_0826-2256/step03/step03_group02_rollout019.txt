import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # More points for better accuracy, still fast enough
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2
    
    # Define a better initial guess that promotes even distribution
    def initial_guess(n_points, seed=42):
        h = np.random.rand(n_points)
        h = h * (total_integral / np.sum(h))  # Normalize to ensure the integral constraint
        # Apply more sophisticated smoothing
        h = np.convolve(h, np.ones(5) / 5, mode='same')
        h = np.convolve(h, np.ones(5) / 5, mode='same')
        h = np.clip(h, 0, 1)
        return h

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def evaluate_and_optimize(h_initial):
        # Define the objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        # Use L-BFGS-B for optimization with bounds and constraint
        result = opt.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Run the main optimization
    try:
        optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Check remaining time and perform more perturbations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 30:
        # Perturb and optimize a few times
        for _ in range(5):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # If still not optimal, perform a final refinement
    if remaining_time > 10:
        perturbed_h = best_h + 0.03 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    return best_h, best_c5, n_points
