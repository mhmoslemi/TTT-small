```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points for efficiency and performance
    n_points_list = [100, 150, 200]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        if time.time() > budget_s - 1:
            break
        
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2
        
        # Start from the provided initial_h_values, scaled to the correct length
        h = np.copy(initial_h_values[:n_points])
        h = np.clip(h, 0.0, 1.0)
        h = h * (total_integral / np.sum(h))
        
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # First optimization with SLSQP
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        best_h_current = result.x
        best_c5_current = result.fun
        
        # Perturbations to escape local minima
        perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
        for pert in perturbations:
            perturbed_h = best_h_current + pert
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            
            if result_pert.fun < best_c5_current:
                best_h_current = result_pert.x
                best_c5_current = result_pert.fun
        
        # Check if this is the best solution so far
        if best_c5_current < best_c5:
            best_h = best_h_current
            best_c5 = best_c5_current
    
    # Ensure the solution is valid and evaluate
    if best_h is None:
        best_h = np.random.rand(100)
        best_h = np.clip(best_h, 0.0, 1.0)
        best_h = best_h * (100 / 2 / np.sum(best_h))
        best_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, best_c5, len(best_h)
```
