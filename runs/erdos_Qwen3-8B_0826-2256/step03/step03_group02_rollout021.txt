import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between speed and accuracy
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Initial guess based on the provided initial_h_values
    h_initial = initial_h_values.copy()
    h_initial = h_initial * (total_integral / np.sum(h_initial))
    h_initial = np.clip(h_initial, 0, 1)

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def evaluate_and_optimize(h_initial):
        # Define optimization bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral}]
        
        # Define objective function with a small perturbation to avoid flat regions
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx + 1e-8 * np.sum(x**2)  # Small penalty for flat regions
        
        # Run optimization with L-BFGS-B for speed and efficiency
        result = opt.minimize(
            objective,
            h_initial,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = h_initial
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Optimization loop
    for _ in range(5):
        optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
        # Early stopping if we're running out of time
        if time.time() - start_time > budget_s * 0.8:
            break

    # Perturb and optimize a few times to escape local minima
    for _ in range(3):
        perturbed_h = best_h + 0.05 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    # Final refinement
    optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
    if optimized_c5 < best_c5:
        best_h = optimized_h
        best_c5 = optimized_c5

    return best_h, best_c5, n_points
