import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Manually selected for balance between accuracy and speed
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with an initial guess based on a more structured pattern
    h = np.linspace(0, 2, n_points)
    h_vals = np.cos(np.pi * h / 2)**2  # Ellipsoidal-like shape
    h_vals = np.clip(h_vals, 0, 1)  # Ensure within [0,1]
    h_vals = h_vals * (total_integral / np.sum(h_vals))  # Normalize to meet constraint

    # Random perturbation to escape local minima
    perturbation = np.random.normal(0, 0.05, n_points)
    h_vals += perturbation
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (total_integral / np.sum(h_vals))

    # Define the objective function
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Set bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h_vals,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 500}
    )

    # Check result
    if result.success:
        best_h = result.x
    else:
        best_h = h_vals  # Fallback to initial guess

    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
