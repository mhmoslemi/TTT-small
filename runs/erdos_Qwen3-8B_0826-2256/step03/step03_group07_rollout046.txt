    def objective(x):

        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure integral is 1
    def constraint(x):
        return np.sum(x) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    trials = 10  # Number of trials for exploration

    # Perform multiple trials with different initializations and optimization methods
    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        h_perturb = best_h + np.random.uniform(-0.05, 0.05, size=n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb / (np.sum(h_perturb) * dx) if np.sum(h_perturb) != 0 else h_perturb
        h_perturb = np.clip(h_perturb, 0, 1)
        x0 = h_perturb

        # Use SLSQP for optimization
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        try:
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 100}

                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
