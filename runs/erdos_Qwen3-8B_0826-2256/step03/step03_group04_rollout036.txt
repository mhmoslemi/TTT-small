import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from deap import base, creator, tools, algorithms

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Reduced points for faster evaluation
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Constraint: sum(h) == n_points / 2

    # Initialize with a piecewise constant function
    h = np.zeros(n_points)
    h[0:50] = 1.0
    h[50:100] = 0.5
    h[100:150] = 0.0
    h[150:200] = 0.5
    h_vals = h * (total_integral / np.sum(h))  # Normalize to meet constraint

    # Genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("init", lambda: np.random.rand(n_points))
    toolbox.register("eval", lambda ind: evaluate_erdos_solution(ind))
    toolbox.register("mate", tools.cxTwoPoint)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.init)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    pop = toolbox.population(n=50)
    fitnesses = map(toolbox.eval, pop)
    for ind, fit in zip(pop, fitnesses):
        ind.fitness.values = (fit,)
    
    # SLSQP optimization
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Genetic algorithm loop
    best_h = h_vals
    best_c5 = evaluate_erdos_solution(h_vals)
    start_time = time.time()

    for gen in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = map(toolbox.eval, offspring)
        for child, fit in zip(offspring, fits):
            child.fitness.values = (fit,)
        pop = toolbox.select(pop + offspring, len(pop))
        best_ind = tools.selBest(pop, 1)[0]
        best_h = best_ind
        best_c5 = best_ind.fitness.values[0]
        print(f"Gen {gen}: Best C5 = {best_c5}")

    # Refine the best solution with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=best_h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 500}
    )

    best_h = result.x
    best_c5 = result.fun

    # Ensure the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
