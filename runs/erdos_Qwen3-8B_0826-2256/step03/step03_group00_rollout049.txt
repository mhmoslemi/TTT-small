import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start from the provided initial_h_values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Define bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization using SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Generate perturbations for global exploration
    perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
    for pert in perturbations:
        perturbed_h = best_h + pert
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        # Optimize the perturbed solution
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Use a genetic algorithm for further refinement
    def genetic_optimization(population_size=20, generations=20):
        population = [np.random.rand(n_points) for _ in range(population_size)]
        population = [h * (total_integral / np.sum(h)) for h in population]
        population = [np.clip(h, 0.0, 1.0) for h in population]
        best_individual = population[0]
        best_c5_val = objective_function(best_individual)

        for _ in range(generations):
            # Evaluate fitness
            fitness = [objective_function(ind) for ind in population]
            fitness = np.array(fitness)
            best_idx = np.argmin(fitness)
            best_c5_val = fitness[best_idx]
            best_individual = population[best_idx]

            # Selection
            selected = np.random.choice(len(population), size=population_size, p=fitness / fitness.sum())
            selected = population[selected]

            # Crossover and mutation
            new_population = []
            for i in range(population_size):
                parent1 = selected[i % len(selected)]
                parent2 = selected[(i + 1) % len(selected)]
                child = np.zeros(n_points)
                for j in range(n_points):
                    if np.random.rand() < 0.5:
                        child[j] = parent1[j]
                    else:
                        child[j] = parent2[j]
                # Mutation
                for j in range(n_points):
                    if np.random.rand() < 0.05:
                        child[j] = np.clip(child[j] + np.random.normal(0, 0.05), 0.0, 1.0)
                child = np.clip(child, 0.0, 1.0)
                child = child * (total_integral / np.sum(child))
                new_population.append(child)
            population = new_population

        return best_individual, best_c5_val

    # Run genetic optimization
    best_h_genetic, best_c5_genetic = genetic_optimization(population_size=20, generations=20)

    # Compare and choose the best solution
    if best_c5_genetic < best_c5:
        best_h = best_h_genetic
        best_c5 = best_c5_genetic

    # Ensure the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
