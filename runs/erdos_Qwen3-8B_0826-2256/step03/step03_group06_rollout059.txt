    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint
    
    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    def refine_solution(h_values):
        # Refine the solution by slightly perturbing values and optimizing
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        refined_h = np.clip(h_values + perturbation, 0, 1)
        refined_h = refined_h / (np.sum(refined_h) * dx)
        return refined_h
    
    def optimize_solution():
        # Run a global optimization to find a better solution
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100, "eps": 1e-8}
        )
        return result.x, result.fun
    
    # Start with the initial solution
    best_h, best_c5 = optimize_solution()
    
    # Refine the solution iteratively
    for _ in range(3):
        # Perturb and refine
        h = refine_solution(best_h)
        # Run optimization
        result = optimize.minimize(

        )
