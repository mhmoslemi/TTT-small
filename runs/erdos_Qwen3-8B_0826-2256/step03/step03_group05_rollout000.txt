import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200  # Smaller than 1000 points for faster optimization
    dx = 2.0 / n_points

    # Initialize with multiple diverse starting points
    initial_h_values_list = []

    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()

        initial_h = initial_h_values

    x_initial = np.linspace(0, 2, len(initial_h))

    initial_h_clipped = np.clip(initial_h_interpolated, 0, 1)
    initial_h_normalized = initial_h_clipped / (np.sum(initial_h_clipped) * dx)
    initial_h_normalized = np.clip(initial_h_normalized, 0, 1)
    initial_h_values_list.append(initial_h_normalized)

        h = np.clip(h, 0, 1)
        initial_h_values_list.append(h)

    # Evaluate each initial guess
    for initial_h in initial_h_values_list:
        # Define the objective function
        def objective(h_values):
            h = h_values.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5

        # Define the constraint: ∫ h(x) dx = 1
        def constraint(h_values):
            return np.sum(h_values) - n_points / 2

        bounds = [(0, 1) for _ in range(n_points)]

        # Optimize using L-BFGS-B for faster convergence
        result = optimize.minimize(
