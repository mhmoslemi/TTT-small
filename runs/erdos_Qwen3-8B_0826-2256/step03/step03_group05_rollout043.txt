import numpy as np
import scipy.optimize as optimize
import time
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Define grid
    n_points = 200  # Use 200 points for better resolution, still efficient
    dx = 2.0 / n_points

    # Initial guess from initial_h_values
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the constraint: ∫ h(x) dx = 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Best solution so far
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Optimization parameters
    num_runs = 3  # Number of initializations
    perturbation_factor = 0.05  # Factor to control the strength of perturbation
    mutation_rate = 0.1  # Probability of mutation in evolutionary algorithm
    elite_size = 10  # Number of elite solutions to carry forward

    # Main optimization loop
    for _ in range(num_runs):
        if time.time() - start_time > budget_s:
            break
        
        # Generate an initial perturbed guess

        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        # Optimization using a combination of evolutionary and gradient-based methods
        # Evolutionary algorithm
        population = [initial_h + 0.1 * np.random.randn(n_points) for _ in range(elite_size)]
        population = [np.clip(p, 0, 1) for p in population]
        population = [p / (np.sum(p) * dx) for p in population]
        population = [np.clip(p, 0, 1) for p in population]

        for gen in range(100):
            if time.time() - start_time > budget_s:
                break

            fitness = [objective(p) for p in population]
            # Select elites
            elite_indices = np.argsort(fitness)[:elite_size]
            elites = [population[i] for i in elite_indices]
            # Mutate and crossover
            new_population = []
            for _ in range(elite_size):
                if random.random() < mutation_rate:
                    # Mutation

                    mutated = mutated / (np.sum(mutated) * dx)
                    mutated = np.clip(mutated, 0, 1)

                else:
                    # Crossover
                    parent1 = elites[random.randint(0, elite_size - 1)]
                    parent2 = elites[random.randint(0, elite_size - 1)]
                    child = (parent1 + parent2) / 2
                    child = np.clip(child, 0, 1)
                    child = child / (np.sum(child) * dx)
                    child = np.clip(child, 0, 1)

            population = [np.clip(p, 0, 1) for p in population]

        # After evolutionary phase, use gradient descent on best solution
        best_candidate = min(population, key=objective)
        result = optimize.minimize(

            best_candidate,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"ftol": 1e-8, "maxiter": 50}
