import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (starting from the provided initial_h_values)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    # Function to evaluate overlap
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Constraint function
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Try different starting points to avoid local optima
    for _ in range(5):
        # Generate new initial points using random perturbation
        h_perturb = np.random.uniform(0, 1, n_points)
        h_perturb = h_perturb * (total_integral / np.sum(h_perturb))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=h_perturb,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 250}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Use a local search to fine-tune the solution
    for _ in range(3):
        # Small perturbation around the current best solution
        perturbation = np.random.normal(0, 0.05, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Ensure the solution is valid and evaluate
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
