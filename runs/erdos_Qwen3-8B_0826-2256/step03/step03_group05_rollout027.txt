import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points

    # Initial guess from initial_h_values
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Main optimization loop
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

        result = optimize.minimize(
            objective,
            initial_h,
            bounds=bounds,

            options={"ftol": 1e-8, "maxiter": 100}
