import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate
from collections import deque

    initial_h = np.clip(initial_h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        # Compute overlap using correlate
        overlap = correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint_integral(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = initial_h
    best_c5 = objective(best_h)
    start_time = time.time()
    trials = 5  # Number of optimization trials

    # Explore different starting points
    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Random perturbation to escape local optima
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize h to ensure the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Optimization with different methods
        try:
            # First with L-BFGS-B for efficiency
            result_l_bfgs = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_integral},
                method="L-BFGS-B",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_l_bfgs.success:
                current_c5 = result_l_bfgs.fun
                if current_c5 < best_c5:

                    best_h = result_l_bfgs.x

            result_slsqp = minimize(

                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint_integral},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

                    best_c5 = current_c5
                    best_h = result_cobyla.x

            pass
