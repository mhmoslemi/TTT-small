import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Use the initial_h_values if provided, otherwise use a random initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 5  # Number of trials to run for exploration

    # Perform multiple optimization trials with different initializations
    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the initial point to explore nearby regions
        h = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Run optimization with multiple methods
        try:
            # First with SLSQP
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

            # Second with COBYLA for more robust exploration
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="COBYLA",
                options={"maxiter": 100}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            # Handle exceptions gracefully
            pass

    # Try starting with a more structured initial configuration
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        if i % 3 == 0:
            structured_h[i] = 1.0
        elif i % 3 == 1:
            structured_h[i] = 0.5
        else:
            structured_h[i] = 0.0
    structured_h = structured_h / (np.sum(structured_h) * dx) if np.sum(structured_h) != 0 else np.random.rand(n_points)
    structured_h = np.clip(structured_h, 0, 1)

    x0 = structured_h
    try:
        result_slsqp = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass

    return best_h, best_c5, n_points
