import numpy as np
import scipy.optimize as opt
import time

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 400  # Reduce points for faster computation
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start from the initial configuration
    h = np.copy(initial_h_values)
    h = np.resize(h, n_points)
    h = h * (total_integral / np.sum(h))  # Normalize to meet constraint

    # Define hyperparameters
    num_iterations = 100  # Number of iterations to run
    learning_rate = 0.05  # Learning rate for optimization
    penalty_coefficient = 100.0  # Penalty coefficient for constraint violation

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)

    start_time = time.time()

    for iteration in range(num_iterations):
        # Evaluate current solution
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        # Compute the gradient of the objective function
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap_idx = np.argmax(corr)
        max_overlap = corr[max_overlap_idx] * dx
        gradient = np.zeros(n_points)

        # Compute derivative of the overlap integral with respect to h
        # For simplicity, we use finite differences to approximate gradient
        # (this is a heuristic and can be replaced with exact derivative if needed)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus[i] = np.clip(h_plus[i], 0.0, 1.0)
            h_plus = h_plus * (total_integral / np.sum(h_plus))
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            grad_i = (corr_plus[max_overlap_idx] - corr[max_overlap_idx]) * dx / 1e-6
            gradient[i] = grad_i

        # Compute constraint violation
        constraint_violation = np.sum(h) - total_integral

        # Update the function values using gradient descent
        h = h - learning_rate * gradient
        h = np.clip(h, 0.0, 1.0)
        h = h * (total_integral / np.sum(h))

        # Check time budget
        if time.time() - start_time > budget_s - 1:
            break

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
