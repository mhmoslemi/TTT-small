```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # 150 points provides a good balance between resolution and computational feasibility
    
    # Function to evaluate the overlap integral
    def evaluate_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        max_overlap = np.max(corr) * (2.0 / len(h))
        return max_overlap
    
    # Function to compute the integral constraint
    def integral_constraint(h):
        return np.sum(h) - n_points / 2
    
    # Define the initial guess (starting from the provided initial_h_values)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2) / np.sum(h)  # Normalize to satisfy constraint
    
    # Objective function to minimize
    def objective(h_values):
        return evaluate_overlap(h_values)
    
    # Bounds for h_i (0 to 1)
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    # Constraint to enforce integral condition
    cons = {
        'type': 'eq',
        'fun': integral_constraint
    }
    
    # Use SLSQP with tighter tolerance for better optimization
    result = opt.minimize(
        fun=objective,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-10, 'maxiter': 300, 'disp': False}
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Generate a few perturbations of the current best solution
    perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
    for pert in perturbations:
        perturbed_h = best_h + pert
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
        # Optimize the perturbed solution
        result_pert = opt.minimize(
            fun=objective,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun
    
    # Final clipping and normalization to ensure constraints
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation of the solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
```
