import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Use a finer grid than previous attempt to explore more detailed structure
    n_points = 400  # Increased from 200 for better resolution while staying under time budget
    dx = 2.0 / n_points
    
    # Start with a more informed initial guess based on known good structure
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values
    
    # Interpolate to target resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)
    
    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint
    
    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a more efficient optimization strategy with tighter settings
    def optimize_solution():
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100, "eps": 1e-8}
        )
        return result.x, result.fun
    
    # Run the optimization
    start_time = time.time()
    best_h, best_c5 = optimize_solution()
    
    # Early stopping if time is exhausted
    elapsed_time = time.time() - start_time
    if elapsed_time > budget_s - 10:  # Leave 10s buffer
        return best_h, best_c5, n_points
    
    # Try a few perturbations to escape local optima
    for _ in range(3):
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h / (np.sum(perturbed_h) * dx) * (n_points / 2)
        
        # Run a quick optimization on the perturbed initial guess
        result = optimize.minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 50, "eps": 1e-8}
        )
        if result.fun < best_c5:
            best_h, best_c5 = result.x, result.fun
    
    return best_h, best_c5, n_points
