import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=100, **kwargs):
    # Set random seed for reproducibility
    np.random.seed(seed)
    
    # Define grid parameters
    n_points = 200  # Higher resolution for better approximation
    dx = 2.0 / n_points

    # Initial guess from initial_h_values
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    # Interpolate initial_h to match the n_points resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the constraint: ∫ h(x) dx = 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Main optimization loop
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Perform multiple optimization runs with different initializations
    for _ in range(8):
        if time.time() - start_time > budget_s:
            break
        # Create a slightly perturbed initial guess
        initial_h = h + 0.005 * np.random.randn(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        # Optimize using SLSQP
        result = optimize.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    # One final fine-tuning step
    if time.time() - start_time < budget_s - 1:
        initial_h = best_h + 0.005 * np.random.randn(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        result = optimize.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"ftol": 1e-8, "maxiter": 50}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    return best_h, best_c5, n_points
