import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300  # Reduced for faster evaluation and better exploration
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initialize with a hybrid approach: combination of uniform and optimized patterns
    h = np.linspace(0, 2, n_points)
    # Start with a peak in the middle, and a sinusoidal variation
    h_vals = np.sin(np.pi * (h - 1.0) / 1.0) ** 2
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (total_integral / np.sum(h_vals))  # Normalize to meet constraint

    # Add some randomness to escape local optima
    h_vals = np.clip(h_vals + np.random.normal(0, 0.05, n_points), 0, 1)
    h_vals = h_vals * (total_integral / np.sum(h_vals))  # Re-normalize

    # Define the objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Optimization setup for SLSQP with adaptive tolerances and iterations
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with a time budget
    start_time = time.time()
    result = opt.minimize(
        fun=objective_function,
        x0=h_vals,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    elapsed_time = time.time() - start_time

    # If time is remaining, perform a secondary optimization with a different initial guess
    if elapsed_time < budget_s:
        # Perturb the current solution and re-optimize
        h_vals_perturbed = np.clip(h_vals + np.random.normal(0, 0.05, n_points), 0, 1)
        h_vals_perturbed = h_vals_perturbed * (total_integral / np.sum(h_vals_perturbed))
        result = opt.minimize(
            fun=objective_function,
            x0=h_vals_perturbed,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        elapsed_time = time.time() - start_time

    # Extract the best solution
    best_h = result.x
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
