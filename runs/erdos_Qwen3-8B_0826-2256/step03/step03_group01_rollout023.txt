import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use a higher resolution for better accuracy
    n_points = 200
    dx = 2.0 / n_points
    total_integral = n_points / 2.0  # Sum(h) == n_points / 2 exactly

    # Construct a better initial guess based on the existing structure
    def construct_initial_guess(n_points, seed):
        np.random.seed(seed)
        # Create a base structure with alternating high and low values
        base = np.zeros(n_points)
        for i in range(n_points):
            if i % 3 == 0:
                base[i] = 1.0
            elif i % 3 == 1:
                base[i] = 0.5
            else:
                base[i] = 0.0
        # Normalize to ensure the integral constraint
        base = base * (total_integral / np.sum(base))
        # Perturb to avoid local optima
        perturbation = 0.05 * np.random.randn(n_points)
        base = np.clip(base + perturbation, 0, 1)
        return base

    # Define objective function for optimization
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * dx

    # Define an optimization routine using a more robust method
    def optimize_h(initial_guess):
        x = cp.Variable(n_points)
        constraint = cp.sum(x) == total_integral
        # Use a heuristic to minimize the maximum overlap
        # We use a convex relaxation approach here
        # We use a heuristic to convert the max operation into a scalar
        # This is done by finding the max through an auxiliary variable
        max_overlap = cp.Variable()
        constraints = [
            max_overlap >= cp.multiply(cp.reshape(np.correlate(x, 1 - x, mode="full"), (1, -1)), dx)
        ]
        objective = cp.Minimize(max_overlap)
        problem = cp.Problem(objective, constraints + [constraint])
        try:
            result = problem.solve(solver=cp.SCS, eps=1e-6, max_iter=1000)
            optimized_h = x.value
            return optimized_h, max_overlap.value
        except:
            # If the solver fails, return the initial guess
            return initial_guess, objective_function(initial_guess)

    # Start with a good initial guess
    best_h = construct_initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Run the optimization
    try:
        optimized_h, optimized_c5 = optimize_h(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Check remaining time and consider additional perturbations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Perturb and optimize a few times
        for _ in range(5):
            perturbed_h = best_h + 0.02 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize_h(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    return best_h, best_c5, n_points
