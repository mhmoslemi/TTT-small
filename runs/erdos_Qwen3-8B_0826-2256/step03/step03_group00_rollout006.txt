```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient number of points for optimization
    
    # Initial values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (n_points / 2.0 / np.sum(h))  # Normalize to satisfy constraint

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * (2.0 / len(h_values))  # dx = 2.0 / n_points
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2.0  # Ensure integral is 1

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Main optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun
    elapsed_time = time.time() - time.time()

    # Generate perturbations for secondary optimization
    for _ in range(5):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (n_points / 2.0 / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun
        elapsed_time = time.time() - time.time()
        if elapsed_time >= budget_s:
            break

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0 / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
```
