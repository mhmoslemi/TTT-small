import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 400  # Reduced to balance accuracy and speed
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with a more sophisticated initial guess using a piecewise-constant construction
    # Construct an initial h that uses 4 alternating segments of 0 and 1
    segments = 4
    initial_segments = np.zeros(n_points)
    for i in range(segments):
        start = int(i * n_points / segments)
        end = int((i + 1) * n_points / segments)
        initial_segments[start:end] = 1 if i % 2 == 0 else 0
    initial_segments = np.clip(initial_segments, 0, 1)
    initial_segments = initial_segments * (total_integral / np.sum(initial_segments))  # Normalize

    # Define the objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Optimization setup for SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization with more aggressive settings
    result = opt.minimize(
        fun=objective_function,
        x0=initial_segments,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300, 'eps': 1e-8}
    )

    # Check if optimization was successful
    if not result.success:
        return initial_segments, objective_function(initial_segments), n_points

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    # Ensure that the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
