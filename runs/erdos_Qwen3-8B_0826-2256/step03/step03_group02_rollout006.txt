```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points to balance speed and accuracy
    n_points_candidates = [100, 150, 200]
    best_h = initial_h_values
    best_c5 = evaluate_erdos_solution(best_h)
    best_n_points = 100
    best_time = 0

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * (2.0 / len(h_values))  # dx = 2.0 / len(h_values)

    def integral_constraint(h_values):
        return np.sum(h_values) - len(h_values) / 2  # sum(h) == len(h)/2

    def initial_guess(n_points, seed=42):
        h = np.random.rand(n_points)
        h = h * (len(h) / 2) / np.sum(h)  # Normalize to ensure the integral constraint
        # Apply a simple smoothing to promote even distribution
        h = np.convolve(h, np.ones(5) / 5, mode='same')
        h = np.clip(h, 0, 1)
        return h

    def optimize_h(h_initial, n_points):
        # Use SLSQP for optimization
        result = opt.minimize(
            objective_function,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': integral_constraint},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x, result.fun

    # Try different n_points
    for n_points in n_points_candidates:
        start_time = time.time()
        h_initial = initial_guess(n_points, seed)
        optimized_h, optimized_c5 = optimize_h(h_initial, n_points)
        elapsed_time = time.time() - start_time
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
            best_n_points = n_points
            best_time = elapsed_time
        print(f"n_points={n_points}, C5={optimized_c5:.6f}, time={elapsed_time:.2f}s")

    # Use the best n_points and refine with perturbations
    n_points = best_n_points
    dx = 2.0 / n_points
    total_integral = n_points / 2

    # Refine with perturbations
    start_time = time.time()
    for _ in range(3):
        perturbed_h = best_h + 0.05 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = optimize_h(perturbed_h, n_points)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
        elapsed_time = time.time() - start_time
        if elapsed_time > budget_s * 0.8:
            break

    return best_h, best_c5, n_points
```
