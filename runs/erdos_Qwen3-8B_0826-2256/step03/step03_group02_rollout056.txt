import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Slightly larger for better accuracy, still fast enough
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Initial guess based on a uniform distribution and smoothing
    def initial_guess(n_points, seed=42):
        h = np.random.rand(n_points)
        h = h * (total_integral / np.sum(h))  # Normalize to ensure the integral constraint
        # Apply a simple smoothing to promote even distribution
        h = np.convolve(h, np.ones(5) / 5, mode='same')
        h = np.clip(h, 0, 1)
        return h

    # Objective function for optimization
    def objective(x):
        corr = np.correlate(x, 1 - x, mode="full")
        return np.max(corr) * dx

    # Constraint function for the integral
    def integral_constraint(x):
        return np.sum(x) - total_integral

    # Optimization function
    def optimize(h_initial):
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': integral_constraint},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = evaluate_erdos_solution(best_h)

    # Start timer
    start_time = time.time()

    # Run optimization on the initial guess
    optimized_h, optimized_c5 = optimize(best_h)
    if optimized_c5 < best_c5:
        best_h = optimized_h
        best_c5 = optimized_c5

    # Perturb and optimize a few times
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        for _ in range(3):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # Further refinement
    if remaining_time > 5:
        perturbed_h = best_h + 0.03 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = optimize(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    return best_h, best_c5, n_points
