        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

        # Initialize with the provided initial values and perturb if needed
        h = initial_h_values.copy()
        if len(h) != n_points:

            h = np.clip(h, 0, 1)
            h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

        # Objective: minimize the maximum overlap integral
        # Operational constraints: 0 <= h[i] <= 1, sum(h) == total_integral

        # Use a more advanced optimization approach
        # Define the objective as a minimax problem
        # We'll use a convex relaxation approach with a parameter to control the trade-off
        eps = 1e-5

        t = cp.Variable()
        constraints = [cp.sum(x) == total_integral, 0 <= x, x <= 1]

        problem = cp.Problem(cp.Minimize(t), constraints)

        # Convert the result to the h_values
        h_values = x.value
        return h_values, t.value

    # Start best with the initial guess
    best_h = initial_guess(n_points, seed)

    try:
