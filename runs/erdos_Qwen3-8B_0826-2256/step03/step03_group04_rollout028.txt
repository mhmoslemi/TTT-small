import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 300  # Reduced from 500 for faster optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Improved initial guess based on periodic optimization
    initial_guess = np.zeros(n_points)
    for i in range(n_points):
        initial_guess[i] = 0.5 * (1 + np.cos(np.pi * (i - n_points // 2) / (n_points // 2)))
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (total_integral / np.sum(initial_guess))  # Normalize to meet constraint

    # Define the objective function: minimize the maximum overlap integral
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint: sum(h) == n_points / 2
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Optimization setup for SLSQP
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Run optimization
    result = opt.minimize(
        fun=objective_function,
        x0=initial_guess,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )

    # Extract the best solution
    best_h = result.x
    best_c5 = result.fun

    # Ensure that the solution is valid
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))  # Normalize again to ensure constraint

    # Evaluate the final solution
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
