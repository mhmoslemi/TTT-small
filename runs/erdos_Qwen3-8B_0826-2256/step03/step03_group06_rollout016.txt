import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a more refined number of points
    n_points = 400  # Increased for better resolution while staying within budget
    dx = 2.0 / n_points
    
    # Initialize from provided initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values
    
    # Interpolate to match the required resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)
    
    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint
    
    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a combination of optimization strategies with tighter settings
    def optimize_solution():
        # Step 1: Use local optimization to refine initial guess
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"ftol": 1e-8, "maxiter": 50, "eps": 1e-8}
        )
        refined_h = result.x
        
        # Step 2: Use global optimization with a perturbation strategy
        perturbation = 0.05 * np.random.rand(n_points)
        perturbed_h = np.clip(refined_h + perturbation, 0, 1)
        
        result = optimize.minimize(
            objective,
            perturbed_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 50, "eps": 1e-8}
        )
        return result.x, result.fun
    
    # Track time and return best solution within the budget
    start_time = time.time()
    best_h, best_c5 = optimize_solution()
    elapsed_time = time.time() - start_time
    
    # If time remains, perform a few more optimizations
    if elapsed_time < budget_s - 10:
        result = optimize.minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"ftol": 1e-8, "maxiter": 20, "eps": 1e-8}
        )
        best_h = result.x
        best_c5 = result.fun
    
    return best_h, best_c5, n_points
