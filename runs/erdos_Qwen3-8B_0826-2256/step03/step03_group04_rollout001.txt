import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try using smaller n_points for faster optimization
    n_points_list = [300, 400, 500]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0

    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2

        # Start with a structured initial guess based on trigonometric patterns
        h = np.linspace(0, 2, n_points)
        h_vals = np.sin(np.pi * h / 2)**2  # Sine squared for smooth transitions
        h_vals = np.clip(h_vals, 0, 1)  # Ensure [0,1]
        h_vals = h_vals * (total_integral / np.sum(h_vals))  # Normalize

        # Define the objective function: minimize the maximum overlap integral
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap

        # Define the constraint: sum(h) == n_points / 2
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral

        # Optimization setup for SLSQP with tighter tolerances
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}

        # Run optimization
        start_time = time.time()
        result = opt.minimize(
            fun=objective_function,
            x0=h_vals,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        elapsed_time = time.time() - start_time

        # Early stopping if time budget is exceeded
        if elapsed_time >= budget_s:
            break

        # Extract the best solution
        best_h_candidate = result.x
        best_h_candidate = np.clip(best_h_candidate, 0.0, 1.0)
        best_h_candidate = best_h_candidate * (total_integral / np.sum(best_h_candidate))

        # Evaluate the final solution
        final_c5_candidate = evaluate_erdos_solution(best_h_candidate)

        # Update the best solution
        if final_c5_candidate < best_c5:
            best_c5 = final_c5_candidate
            best_h = best_h_candidate
            best_n_points = n_points

    # Return the best solution found
    return best_h, best_c5, best_n_points
