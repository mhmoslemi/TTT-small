import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define grid
    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points
    
    # Initial guess from initial_h_values
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    # Interpolate initial_h to match the n_points resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the max overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the constraint: ∫ h(x) dx = 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Main optimization loop with multiple strategies
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Strategy 1: Use a more sophisticated optimizer with tighter tolerances
    for _ in range(3):
        if time.time() - start_time > budget_s:
            break
        # Create a perturbed initial guess
        initial_h = h + 0.01 * np.random.randn(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        # Optimize using SLSQP with tighter tolerances
        result = optimize.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 150}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    # Strategy 2: Try another initial guess based on a different pattern
    initial_pattern = np.zeros(n_points)
    for i in range(n_points):
        if i % 5 == 0 or i % 5 == 1 or i % 5 == 2:
            initial_pattern[i] = 1.0
    initial_pattern = initial_pattern / (np.sum(initial_pattern) * dx)
    initial_pattern = np.clip(initial_pattern, 0, 1)

    # Optimize using L-BFGS-B with a different initial guess
    result = optimize.minimize(
        objective,
        initial_pattern,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="L-BFGS-B",
        options={"ftol": 1e-10, "maxiter": 100}
    )
    if result.success:
        current_h = result.x
        current_c5 = result.fun
        if current_c5 < best_c5:
            best_h = current_h
            best_c5 = current_c5

    # Strategy 3: Fine-tuning step with a different optimizer
    if time.time() - start_time < budget_s - 1:
        initial_h = best_h + 0.005 * np.random.randn(n_points)
        initial_h = np.clip(initial_h, 0, 1)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        result = optimize.minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"ftol": 1e-10, "maxiter": 50}
        )
        if result.success:
            current_h = result.x
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_h = current_h
                best_c5 = current_c5

    return best_h, best_c5, n_points
