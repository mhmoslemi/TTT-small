import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and quality

    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a combination of differential evolution and local refinement for global and local optimization
    trials = 50  # A moderate number of trials for thorough exploration
    population_size = 30  # Population size for differential evolution
    mutation_factor = 0.5  # Mutation factor for differential evolution
    crossover_probability = 0.9  # Crossover probability
    local_refinement_steps = 10  # Number of local refinement steps per trial

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Generate a diverse initial population for differential evolution
        population = np.random.uniform(low=0, high=1, size=(population_size, n_points))
        population = np.clip(population, 0, 1)
        population = population / (np.sum(population, axis=1) * 2.0 / n_points)[:, np.newaxis]
        population = np.clip(population, 0, 1)

        # Differential evolution for global exploration
        def de_objective(h_values):
            return objective(h_values)

        result_de = differential_evolution(
            de_objective,
            bounds,
            strategy='best1bin',
            popsize=population_size,
            mutation=(mutation_factor, mutation_factor),
            recombination=crossover_probability,
            tol=1e-8,
            polish=True,
            disp=False,
            maxiter=100,
            args=(),
            vectorized=True,
            workers=-1,
            constraints=[LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])]
        )

        # Update best solution from differential evolution result
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x

        # Local refinement with SLSQP
        for _ in range(local_refinement_steps):
            if time.time() - start_time > budget_s - 1:
                break
            x0 = best_h
            try:
                result_slsqp = minimize(
                    objective,
                    x0,
                    bounds=bounds,
                    constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                    method="SLSQP",
                    options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
                )
                if result_slsqp.success:
                    current_c5 = result_slsqp.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result_slsqp.x
            except:
                pass

    return best_h, best_c5, n_points
