    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)

    start_time = time.time()

    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.3  # Slightly smaller initial perturbation for better exploration
    learning_rate = 0.02  # Learning rate for gradient-based methods

    # Simulated Annealing for global exploration
    def simulated_annealing():
        temp = 1.0
        cooling_rate = 0.995
        max_iter = 500
        current_h = np.random.uniform(0, 1, n_points)
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h / (np.sum(current_h) * 2.0 / n_points)
        current_h = np.clip(current_h, 0, 1)
        current_c5 = objective(current_h)

        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate a perturbed candidate solution
            perturbation = np.random.normal(0, perturbation_scale, size=n_points)

                current_c5 = candidate_c5

            temp *= cooling_rate

        return current_h, current_c5

    def local_refinement(h_values):
        try:
            result_slsqp = minimize(
                objective,
                h_values,
                bounds=bounds,

            pass
        return h_values, objective(h_values)

    for trial in range(trials):

        h_refined, c5_refined = local_refinement(h_sa)

        # Update best solution
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
