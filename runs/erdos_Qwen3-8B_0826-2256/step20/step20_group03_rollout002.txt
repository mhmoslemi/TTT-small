import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a multi-phase optimization approach: initial exploration with Simulated Annealing, followed by fine-tuning with SLSQP
    trials = 100  # Number of trials for thorough exploration
    temperature = 1.0  # Initial temperature for Simulated Annealing
    cooling_rate = 0.995  # Cooling rate for temperature reduction

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.25, 0.25, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)

        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Simulated Annealing for global exploration
        current_solution = h.copy()
        current_c5 = objective(current_solution)
        temperature = max(temperature * cooling_rate, 0.01)

        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break

            # Generate a random neighbor
            neighbor = current_solution + np.random.uniform(-0.1, 0.1, size=n_points)

            neighbor = neighbor / (np.sum(neighbor) * 2.0 / n_points) if np.sum(neighbor) != 0 else neighbor
            neighbor = np.clip(neighbor, 0, 1)

            neighbor_c5 = objective(neighbor)
            delta_c5 = neighbor_c5 - current_c5

            # Acceptance probability
            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temperature):
                current_solution = neighbor.copy()
                current_c5 = neighbor_c5

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_solution

        # SLSQP for local refinement with tighter tolerances
        try:
            result_slsqp = minimize(
                objective,
                x0=current_solution,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),

            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

    return best_h, best_c5, n_points
