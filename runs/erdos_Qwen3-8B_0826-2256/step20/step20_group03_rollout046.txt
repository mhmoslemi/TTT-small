import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and quality

    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a multi-phase optimization approach: simulated annealing for global exploration, followed by local refinement with SLSQP
    trials = 150  # Increased number of trials for thorough exploration
    initial_temperature = 1.0
    cooling_rate = 0.995
    perturbation_scale = 0.25

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated annealing step for global exploration
        current_h = best_h.copy()
        current_c5 = best_c5
        temperature = initial_temperature * (cooling_rate ** trial)

        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break

            # Perturb the current solution
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            new_h = current_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            # Ensure the integral constraint is satisfied
            new_h = new_h / (np.sum(new_h) * 2.0 / n_points) if np.sum(new_h) != 0 else np.random.rand(n_points)
            new_h = np.clip(new_h, 0, 1)
            new_c5 = objective(new_h)

            # Acceptance probability
            if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / temperature):
                current_h = new_h
                current_c5 = new_c5

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95

    # Local refinement with SLSQP
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
        )
        if result_slsqp.success:
            if result_slsqp.fun < best_c5:
                best_c5 = result_slsqp.fun
                best_h = result_slsqp.x
    except:
        pass

    return best_h, best_c5, n_points
