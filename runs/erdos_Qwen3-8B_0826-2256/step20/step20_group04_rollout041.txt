import scipy.optimize as opt

from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a hierarchical optimization approach: simulated annealing for global exploration, then local refinement
    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.3  # Initial perturbation for exploration
    learning_rate = 0.02  # Learning rate for gradient-based methods

    # Simulated Annealing Parameters
    temp_start = 1.0
    temp_end = 0.1
    cooling_rate = 0.995
    trials_per_temp = 50

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated Annealing for global exploration
        current_temp = temp_start
        current_h = best_h.copy()
        current_c5 = objective(current_h)

        for _ in range(trials_per_temp):
            if time.time() - start_time > budget_s - 1:
                break

            # Perturb the current solution
            perturbation = np.random.normal(0, perturbation_scale, size=n_points)
            candidate_h = current_h + perturbation
            candidate_h = np.clip(candidate_h, 0, 1)
            
            # Ensure the integral constraint is satisfied
            candidate_h = candidate_h / (np.sum(candidate_h) * 2.0 / n_points) if np.sum(candidate_h) != 0 else np.random.rand(n_points)

            candidate_c5 = objective(candidate_h)
            delta_c5 = candidate_c5 - current_c5

            if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / current_temp):
                current_h = candidate_h
                current_c5 = candidate_c5

        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # Cool down the temperature
        current_temp *= cooling_rate

    # Local refinement with SLSQP
    try:
        result_slsqp = minimize(

            x0=best_h,
            bounds=bounds,
            constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),

            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass

    return best_h, best_c5, n_points
