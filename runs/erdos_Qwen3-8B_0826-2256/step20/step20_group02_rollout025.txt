    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Thorough exploration
    perturbation_scale = 0.3
    learning_rate = 0.01
    exploration_rate = 0.3

        def annealing_step(h_values, temp):
            perturbation = np.random.normal(0, np.sqrt(temp), size=n_points)
            h = h_values + perturbation
            h = np.clip(h, 0, 1)
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)

        temp = 1.0

            if time.time() - start_time > budget_s - 1:

        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

        perturbation_scale *= 0.95
