import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and quality

    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a multi-phase optimization approach: initial exploration with Simulated Annealing, followed by fine-tuning with SLSQP
    trials = 100  # Increased number of trials for thorough exploration
    cooling_rate = 0.995  # Temperature cooling rate for simulated annealing
    initial_temp = 1.0
    learning_rate = 0.02  # Learning rate for gradient-based methods

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated Annealing for global exploration
        temp = initial_temp * (cooling_rate ** trial)
        current_h = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h / (np.sum(current_h) * 2.0 / n_points) if np.sum(current_h) != 0 else np.random.rand(n_points)
        current_h = np.clip(current_h, 0, 1)
        current_c5 = objective(current_h)

        # Acceptance probability
        while temp > 1e-5:
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            next_h = current_h + perturbation
            next_h = np.clip(next_h, 0, 1)

            next_c5 = objective(next_h)
            delta = next_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h = next_h
                current_c5 = next_c5
            temp *= cooling_rate

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h

        # SLSQP for local refinement with tighter tolerances
        try:
            result_slsqp = minimize(

                x0=current_h,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),

            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

    return best_h, best_c5, n_points
