import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for efficiency and quality

    # Use initial_h_values if available for a better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a combination of simulated annealing and gradient-based optimization for a balance
    # between global exploration and local refinement
    trials = 150  # Number of trials for exploration
    cooling_rate = 0.995  # Cooling rate for simulated annealing
    initial_temperature = 10.0  # Initial temperature for simulated annealing

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated Annealing for global exploration
        current_h = best_h.copy()
        temperature = initial_temperature
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate a perturbation
            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            candidate_h = current_h + perturbation
            candidate_h = np.clip(candidate_h, 0, 1)
            # Ensure the integral constraint is satisfied
            candidate_h = candidate_h / (np.sum(candidate_h) * 2.0 / n_points) if np.sum(candidate_h) != 0 else np.random.rand(n_points)
            candidate_h = np.clip(candidate_h, 0, 1)
            # Evaluate the candidate
            candidate_c5 = objective(candidate_h)
            # Accept based on temperature
            if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / temperature):
                current_h = candidate_h
                if candidate_c5 < best_c5:
                    best_c5 = candidate_c5
                    best_h = candidate_h
            temperature *= cooling_rate

        # Gradient-based local refinement
        try:
            result_slsqp = minimize(
                objective,
                x0=current_h,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

    return best_h, best_c5, n_points
