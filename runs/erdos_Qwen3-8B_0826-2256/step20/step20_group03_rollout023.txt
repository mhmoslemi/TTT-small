    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a multi-phase optimization approach: initial exploration with Simulated Annealing, followed by fine-tuning with Nelder-Mead
    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.25  # Slightly smaller initial perturbation for better exploration
    cooling_rate = 0.99  # Temperature cooling rate for Simulated Annealing
    learning_rate = 0.02  # Learning rate for gradient-based methods

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated Annealing for global exploration
        temperature = 1.0
        current_h = best_h.copy()
        current_c5 = best_c5

                break

            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)

            candidate_h = candidate_h / (np.sum(candidate_h) * 2.0 / n_points) if np.sum(candidate_h) != 0 else np.random.rand(n_points)

                current_h = candidate_h
                current_c5 = candidate_c5
            temperature *= cooling_rate

        if current_c5 < best_c5:

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95
