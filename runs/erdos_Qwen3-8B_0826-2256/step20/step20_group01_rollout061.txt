    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Thorough exploration
    perturbation_scale = 0.3
    learning_rate = 0.01
    exploration_rate = 0.3
    adaptive_learning = True

        if time.time() - start_time > budget_s - 1:
            break
        
        # Simulated Annealing for global exploration
        current_h = best_h.copy()
        temperature = 1.0
        for _ in range(100):
            if temperature < 1e-4:
                break
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h_perturbed = np.clip(current_h + perturbation, 0, 1)
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points) if np.sum(h_perturbed) != 0 else np.random.rand(n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            cost_current = objective(current_h)

            temperature *= 0.995
        
        # Local refinement with SLSQP
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

            result = minimize(

                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.95
    
    return best_h, best_c5, n_points
