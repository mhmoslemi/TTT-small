import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and sufficient for quality
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Thorough exploration
    perturbation_scale = 0.3
    learning_rate = 0.01
    diversity = 0.2  # Encourage diversity in perturbations
    
    # Use Simulated Annealing for global exploration
    def simulated_annealing():
        temp = 1.0
        cooling = 0.995
        iterations = 100
        current_h = best_h.copy()
        current_c5 = best_c5
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            perturbation = perturbation * (1 + np.random.uniform(-diversity, diversity, size=n_points))
            new_h = current_h + perturbation
            new_h = np.clip(new_h, 0, 1)
            # Maintain integral constraint
            new_h = new_h / (np.sum(new_h) * 2.0 / n_points) if np.sum(new_h) != 0 else np.random.rand(n_points)
            new_h = np.clip(new_h, 0, 1)
            new_c5 = objective(new_h)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / temp):
                current_h = new_h
                current_c5 = new_c5
            temp *= cooling
        
        return current_h, current_c5

    # Use SLSQP for local refinement
    def local_refinement(h_values):
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                h_values,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                return result.x, result.fun
        except:
            pass
        return h_values, best_c5
    
    # Run simulated annealing
    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        h, c5 = simulated_annealing()
        if c5 < best_c5:
            best_c5 = c5
            best_h = h
    
    # Refine the best solution
    best_h, best_c5 = local_refinement(best_h)
    
    return best_h, best_c5, n_points
