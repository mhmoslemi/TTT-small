import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and sufficient for quality
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Increased for better exploration
    perturbation_scale = 0.3
    learning_rate = 0.01
    diversity = 0.1  # Encourage diversity in perturbations
    random_perturbation = True
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate diverse perturbations
        if random_perturbation:
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        else:
            perturbation = np.zeros(n_points)
            perturbation[::2] = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points//2)
            perturbation[1::2] = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points//2)
        perturbation = perturbation * (1 + np.random.uniform(-diversity, diversity, size=n_points))
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Maintain integral constraint
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Hybrid optimization: SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # COBYLA for global exploration
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 300}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.95
        random_perturbation = not random_perturbation
    
    return best_h, best_c5, n_points
