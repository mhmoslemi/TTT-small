    # Function to evaluate C₅
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Integral constraint
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of optimization trials
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)

        # SLSQP with constraints
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,

            )

                constraints=constraint,
                method="COBYLA",

            )

                if result_cobyla.fun < best_c5:
                    best_c5 = result_cobyla.fun

    )
