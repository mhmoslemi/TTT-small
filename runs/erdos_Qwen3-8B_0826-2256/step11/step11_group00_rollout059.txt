    def genetic_algorithm(population_size=50, generations=50, mutation_rate=0.1, crossover_rate=0.8):
        nonlocal best_h, best_c5

            fitness = np.array([objective(ind) for ind in population])

                    child = parent1.copy()
                offspring.append(child)
