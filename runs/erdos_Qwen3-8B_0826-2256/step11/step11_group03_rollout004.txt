import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Start with initial h values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
    
    # Clip to [0, 1] and normalize to satisfy integral constraint
    h = np.clip(h, 0, 1)
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Parameters for optimization
    trials = 50
    perturbation_range = 0.08
    
    for trial in range(trials):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:
            break
        
        # Perturb the current best solution
        perturbation = np.random.uniform(
            -perturbation_range * (1 - trial / trials),
            perturbation_range * (1 - trial / trials),
            size=n_points
        )
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Normalize to ensure integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Use SLSQP for local refinement
        result = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
