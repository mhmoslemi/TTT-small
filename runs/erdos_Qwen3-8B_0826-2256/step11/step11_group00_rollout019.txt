import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency
    
    # Initial guess from provided initial_h_values
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    elapsed_time = 0
    
    # Use differential evolution for global optimization and refine with local search
    def optimize_h():
        nonlocal best_h, best_c5
        # Use differential evolution for global optimization
        result_de = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-6,
            maxiter=100,
            callback=lambda x: None,
            disp=False
        )
        if result_de.success:
            h_candidate = result_de.x
            # Refine with local optimization
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            result_slsqp = minimize(
                objective,
                h_candidate,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        else:
            # Fallback to local optimization with initial guess
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            result_slsqp = minimize(
                objective,
                h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
    
    while elapsed_time < budget_s - 1:
        optimize_h()
        elapsed_time = time.time() - start_time
    
    return best_h, best_c5, n_points
