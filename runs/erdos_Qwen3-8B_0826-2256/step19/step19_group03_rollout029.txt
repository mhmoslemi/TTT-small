    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Thorough exploration
    perturbation_scale = 0.4
    learning_rate = 0.015
    exploration_rate = 0.35
    elite_fraction = 0.2  # Keep top elite fraction of population for next round

    # Population-based optimization with tournament selection
    population = [np.random.rand(n_points) for _ in range(100)]
    population = [h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else h for h in population]
    population = [np.clip(h, 0, 1) for h in population]
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate population
        population_fitness = [objective(h) for h in population]
        elite_indices = np.argsort(population_fitness)[:int(elite_fraction * len(population))]
        elite_population = [population[i] for i in elite_indices]

        new_population = []
        for _ in range(len(population) - len(elite_population)):
            # Tournament selection
            candidates = np.random.choice(population, size=5, replace=False)
            best_candidate = min(candidates, key=lambda x: objective(x))
            
            # Perturb and clip
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h = best_candidate + perturbation
            h = np.clip(h, 0, 1)
            
            # Maintain integral constraint
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            new_population.append(h)
        
        # Combine elite and new population
        population = elite_population + new_population
        
        # Local refinement with SLSQP
        for h in population:
            # Ensure integral constraint
            h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            
            # SLSQP refinement
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            try:
                result = minimize(
                    objective,
                    h,
                    bounds=bounds,
                    constraints=constraint,
                    method="SLSQP",
                    options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
                )
                if result.success:
                    h = result.x
            except:
                pass
        
        # Update best solution
        current_fitness = [objective(h) for h in population]
        best_idx = np.argmin(current_fitness)
        best_c5 = current_fitness[best_idx]
        best_h = population[best_idx]
    
    return best_h, best_c5, n_points
