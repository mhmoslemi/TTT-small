import scipy.optimize as opt

    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Increased number of trials for thorough exploration
    perturbation_scale = 0.4  # Larger initial perturbation for global exploration
    learning_rate = 0.05  # Learning rate for gradient-based methods
    temperature = 1.0  # For simulated annealing

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.normal(0, perturbation_scale, size=n_points)

        h = np.clip(h, 0, 1)

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Use a hybrid approach with COBYLA for global exploration
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

            )

                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass

        # Simulated annealing for global exploration
        for _ in range(50):
            perturbation = np.random.normal(0, 0.1, size=n_points)

            new_c5 = objective(x0)

                best_h = x0
            temperature *= 0.995

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.9

    return best_h, best_c5, n_points
