import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and high-quality starting point
    
    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 100  # Number of trials for exploration
    perturbation_scale = 0.25  # Initial perturbation scale
    learning_rate = 0.01  # Learning rate for optimization
    cooling_rate = 0.995  # Cooling rate for simulated annealing
    temp = 1.0  # Initial temperature

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Simulated annealing-inspired perturbation
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Ensure integral constraint is satisfied
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * 2.0 / n_points) if np.sum(h_perturbed) != 0 else np.random.rand(n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        x0 = h_perturbed

        # Use SLSQP with adaptive settings for optimization
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.97
        temp *= cooling_rate

    return best_h, best_c5, n_points
