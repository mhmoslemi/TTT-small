import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    else:

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)

    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use a hybrid approach with gradient-based and population-based optimization
    trials = 150  # Moderate number of trials for thorough exploration
    perturbation_scale = 0.3
    learning_rate = 0.02

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # SLSQP for local refinement with tighter tolerances
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),
                method="SLSQP",

            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # Population-based optimization with variable mutation strategy
        try:
            population_size = 50
            elite_size = 10
            mutation_rate = 0.1
            mutation_scale = 0.15

                if time.time() - start_time > budget_s - 1:
                    break

                # Evaluate fitness
                fitness = np.array([objective(ind) for ind in population])

                # Select elites
                elite_indices = np.argsort(fitness)[:elite_size]
                elites = population[elite_indices]

                # Crossover and mutation
                new_population = np.copy(elites)
                for _ in range(population_size - elite_size):
                    parent1 = np.random.choice(elite_indices)
                    parent2 = np.random.choice(elite_indices)

                    new_population = np.vstack((new_population, child))
                population = new_population

            # Update best solution from population
            fitness = np.array([objective(ind) for ind in population])
            best_index = np.argmin(fitness)
            current_c5 = fitness[best_index]
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = population[best_index]
        except:
            pass

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
