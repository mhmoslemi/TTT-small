import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency and quality
    
    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Increased number of trials for thorough exploration
    perturbation_scale = 0.35  # Larger initial perturbation for global exploration
    learning_rate = 0.05  # Learning rate for gradient-based methods
    exploration_rate = 0.6  # Probability of exploratory perturbation
    diversity_threshold = 1e-4  # Diversity threshold for population-based optimization

    population_size = 50
    population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)]
    for i in range(population_size):
        population[i] = population[i] / (np.sum(population[i]) * 2.0 / n_points) if np.sum(population[i]) != 0 else np.random.rand(n_points)
        population[i] = np.clip(population[i], 0, 1)

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Evaluate diversity
        diversity = np.std(population, axis=0).sum()
        if diversity < diversity_threshold:
            break

        # Randomly select a solution from the population
        current_h = population[np.random.randint(0, population_size)]
        
        # Perturb the current solution
        if np.random.random() < exploration_rate:
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            current_h = current_h + perturbation
            current_h = np.clip(current_h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        current_h = current_h / (np.sum(current_h) * 2.0 / n_points) if np.sum(current_h) != 0 else np.random.rand(n_points)
        current_h = np.clip(current_h, 0, 1)

        # Use SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                current_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 150, "eps": 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # COBYLA for global exploration
        try:
            result_cobyla = minimize(
                objective,
                current_h,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass

        # Update population
        if np.random.random() < 0.2:
            population = [np.clip(np.random.uniform(0, 1, n_points), 0, 1) for _ in range(population_size)]
            for i in range(population_size):
                population[i] = population[i] / (np.sum(population[i]) * 2.0 / n_points) if np.sum(population[i]) != 0 else np.random.rand(n_points)
                population[i] = np.clip(population[i], 0, 1)

        # Replace worst solution
        if best_c5 < np.min([objective(sol) for sol in population]):
            population = [best_h.copy() if i == 0 else sol for i, sol in enumerate(population)]

    return best_h, best_c5, n_points
