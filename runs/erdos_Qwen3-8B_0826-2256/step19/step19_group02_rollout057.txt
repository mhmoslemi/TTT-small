import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)

    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    trials = 150  # More thorough exploration
    perturbation_scale = 0.4
    learning_rate = 0.01
    diversity_rate = 0.1  # Encourage diversity in perturbation

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Random perturbation with diversity
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Maintain integral constraint
        integral = np.sum(h)
        if integral != 0:
            h = h * (n_points / 2) / integral
        h = np.clip(h, 0, 1)
        x0 = h

        # Gradient-based local optimization
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="L-BFGS-B",
                options={"ftol": 1e-10, "maxiter": 150, "eps": 1e-8}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

        # Simulated annealing for global exploration
        temp = 0.99
        for it in range(100):
            perturb = np.random.uniform(-0.1, 0.1, size=n_points)
            h_candidate = best_h + perturb
            h_candidate = np.clip(h_candidate, 0, 1)
            integral_candidate = np.sum(h_candidate)
            if integral_candidate != 0:
                h_candidate = h_candidate * (n_points / 2) / integral_candidate
            h_candidate = np.clip(h_candidate, 0, 1)
            candidate_c5 = objective(h_candidate)
            if candidate_c5 < best_c5 or np.random.rand() < np.exp((best_c5 - candidate_c5) / temp):
                best_h = h_candidate
                best_c5 = candidate_c5
            temp *= 0.999

        # Reduce perturbation for finer tuning
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
