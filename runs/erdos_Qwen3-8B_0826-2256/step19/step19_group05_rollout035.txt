import scipy.optimize as opt

from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):

    else:

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip values to ensure they are within [0, 1]
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 200  # Number of trials for exploration
    perturbation_scale = 0.35  # Initial perturbation scale for exploration
    learning_rate = 0.05  # Learning rate for gradient-based methods
    exploration_rate = 0.6  # Probability of exploratory perturbation
    cooling_rate = 0.995  # Cooling rate for simulated annealing

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best solution with probabilistic exploration
        if np.random.random() < exploration_rate:
            perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
            h = best_h + perturbation
            h = np.clip(h, 0, 1)
        else:
            h = best_h.copy()

        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Simulated Annealing for robust optimization
        temp = 1.0
        for _ in range(100):
            if temp < 1e-5:
                break
            perturbation = np.random.normal(0, np.sqrt(temp), size=n_points)
            new_h = x0 + perturbation
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / (np.sum(new_h) * 2.0 / n_points) if np.sum(new_h) != 0 else new_h
            new_c5 = objective(new_h)

            # Acceptance probability
            current_c5 = objective(x0)
            delta = new_c5 - current_c5
            if delta < 0 or np.random.random() < np.exp(-delta / temp):
                x0 = new_h

            temp *= cooling_rate

        # Use SLSQP for refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",

            )

                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
