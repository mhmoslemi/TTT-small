def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)

        for i in range(n_points):
            h[i] = 0.5 if i % 3 == 0 else 0.0 if i % 3 == 1 else 1.0
        # Ensure the integral is 1 by normalizing
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Define the objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)

    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use differential evolution for global search
    def differential_evolution_objective(x):
        x = x.reshape(-1)
        return objective(x)

    # Use a smaller population size and fewer generations to stay within time limit
    result_de = differential_evolution(
        differential_evolution_objective,
        bounds,
        strategy='best1bin',
        popsize=20,
        maxiter=50,
        tol=1e-8,
        mutation=(0.5, 1),
        recombination=0.8,
        disp=False,
        polish=True,
        vectorized=True
    )

    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun

    # Use a local optimizer to refine the solution
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result_slsqp.success and result_slsqp.fun < best_c5:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    except:
        pass

    return best_h, best_c5, n_points
