import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Increased points for better resolution
    
    # Function to evaluate the maximum overlap
    def evaluate_overlap(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        return np.max(corr) * (2.0 / len(h_values))
    
    # Function to generate an initial guess
    def generate_initial_guess(n_points, seed):
        # Start with the provided initial values or random if not provided
        if len(initial_h_values) == n_points:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)  # Ensure integral constraint
        return h
    
    # Optimization function using SLSQP
    def optimize_h(h_initial):
        x = cp.Variable(n_points)
        constraint = cp.sum(x) == n_points / 2
        obj = cp.Minimize(evaluate_overlap(x))
        prob = cp.Problem(obj, [constraint])
        result = prob.solve(solver=cp.SCS, eps=1e-6, max_iter=1000)
        return x.value, evaluate_overlap(x.value)
    
    # Start with the initial guess
    best_h = generate_initial_guess(n_points, seed)
    best_c5 = evaluate_overlap(best_h)
    
    # Start timer
    start_time = time.time()
    
    # Run the first optimization
    try:
        optimized_h, optimized_c5 = optimize_h(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass
    
    # Check remaining time and consider additional perturbations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 15:
        for _ in range(10):
            # Perturb the solution
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            try:
                optimized_h, optimized_c5 = optimize_h(perturbed_h)
                if optimized_c5 < best_c5:
                    best_h = optimized_h
                    best_c5 = optimized_c5
            except:
                pass
    
    return best_h, best_c5, n_points
