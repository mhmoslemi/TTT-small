import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Use a slightly smaller n_points than the previous attempt for faster optimization
    n_points = 120
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Refine initial guess using the provided initial_h_values with smoothing and normalization
    def initial_guess(n_points, seed=42):
        if len(initial_h_values) == n_points:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
            h = np.clip(h, 0, 1)
            h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        # Apply smoothing to reduce noise and improve quality
        h = np.convolve(h, np.ones(5) / 5, mode='same')
        h = np.clip(h, 0, 1)
        h = h * (total_integral / np.sum(h))  # Re-normalize
        return h

    # Define the objective function that minimizes the maximum overlap
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Evaluate the solution and optimize it
    def evaluate_and_optimize(h_initial):
        x = cp.Variable(n_points)
        constraint = cp.sum(x) == total_integral
        # Use a scalar optimization by minimizing the max overlap
        def objective(x_eval):
            corr = np.correlate(x_eval, 1 - x_eval, mode="full")
            return np.max(corr) * dx
        # Use SLSQP for optimization with tighter tolerances
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 200, 'disp': False}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start the timer
    start_time = time.time()

    # Run the optimization
    try:
        optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Remaining time allows for additional perturbations and optimizations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Perturb and optimize a few times with adaptive perturbation
        for _ in range(10):
            perturbed_h = best_h + 0.03 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    return best_h, best_c5, n_points
