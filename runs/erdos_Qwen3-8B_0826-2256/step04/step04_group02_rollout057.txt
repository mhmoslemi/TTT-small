import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between speed and accuracy
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    def objective_function(h_values):
        # Evaluate the maximum overlap integral
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def construct_initial_guess(n_points, seed):
        # Create a smooth initial guess with alternating high/low regions
        # Inspired by known optimal solutions for the Erdos problem
        h = np.zeros(n_points)
        for i in range(n_points):
            if i % 5 == 0 or i % 5 == 4:
                h[i] = 1.0
            elif i % 5 == 1 or i % 5 == 3:
                h[i] = 0.5
            else:
                h[i] = 0.0
        # Normalize to meet the integral constraint
        h = h * (total_integral / np.sum(h))
        # Add small random perturbation to break symmetry
        h += 0.05 * np.random.randn(n_points)
        h = np.clip(h, 0, 1)
        return h

    def evaluate_and_optimize(h_initial):
        # Define the objective function
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        # Use SLSQP for optimization with bounds and constraints
        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 150}
        )
        return result.x, result.fun

    # Start best with the initial guess
    best_h = construct_initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Run optimization
    try:
        optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Check remaining time and consider additional perturbations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        # Perturb and optimize a few times with different initial guesses
        for _ in range(3):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # If still not optimal, refine with another round
    if remaining_time > 10:
        perturbed_h = best_h + 0.03 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5

    return best_h, best_c5, n_points
