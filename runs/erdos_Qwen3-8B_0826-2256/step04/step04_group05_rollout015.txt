import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Slightly more points for better resolution
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (starting from the provided initial_h_values)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Function to perform optimization with a given set of parameters
    def optimize_h(x0):
        result = opt.minimize(
            fun=objective_function,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        return result.x, result.fun

    # Initial optimization
    best_h, best_c5 = optimize_h(h)

    # Perform a series of perturbations and optimizations
    for _ in range(10):
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        perturbed_h, perturbed_c5 = optimize_h(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h, best_c5 = perturbed_h, perturbed_c5

    # Fine-tuning with smaller perturbations
    for _ in range(10):
        perturbation = np.random.normal(0, 0.01, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        perturbed_h, perturbed_c5 = optimize_h(perturbed_h)
        if perturbed_c5 < best_c5:
            best_h, best_c5 = perturbed_h, perturbed_c5

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
