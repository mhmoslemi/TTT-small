import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # 100 points is efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with initial values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    # Define objective function
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define constraint function
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Track time
    start_time = time.time()

    # Initial optimization with SLSQP, but with tighter ftol and fewer iterations
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=[(0.0, 1.0) for _ in range(n_points)],
        constraints={'type': 'eq', 'fun': constraint_func},
        options={'ftol': 1e-8, 'maxiter': 50}
    )
    best_h = result.x
    best_c5 = result.fun

    # Use a more sophisticated gradient-based method with fewer iterations but less tolerance
    for _ in range(10):
        if time.time() - start_time >= budget_s:
            break
        result = opt.minimize(
            fun=objective_function,
            x0=best_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Generate a few perturbations and optimize
    for _ in range(5):
        if time.time() - start_time >= budget_s:
            break
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Random restart to escape local optima
    for _ in range(3):
        if time.time() - start_time >= budget_s:
            break
        random_h = np.random.rand(n_points)
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))
        result = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='L-BFGS-B',
            bounds=[(0.0, 1.0) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': constraint_func},
            options={'ftol': 1e-8, 'maxiter': 50}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
