import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from deap import base, creator, tools, algorithms

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better accuracy while keeping computation efficient
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values (using the provided initial_h_values)
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Genetic Algorithm for Global Search
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    def eval_func(ind):
        h_values = np.array(ind)
        fitness = objective_function(h_values)
        return (fitness,)

    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0.0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Initial population
    pop = toolbox.population(n=50)
    # Evaluate initial population
    fitnesses = list(map(toolbox.evaluate, pop))
    for ind, fit in zip(pop, fitnesses):
        ind.fitness.values = fit

    # Start genetic algorithm
    start_time = time.time()
    for gen in range(100):
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        fits = list(map(toolbox.evaluate, offspring))
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        pop = toolbox.select(pop + offspring, len(pop))
        if time.time() - start_time > budget_s - 5:
            break

    # Extract the best individual from the genetic algorithm
    best_h = min(pop, key=lambda ind: ind.fitness.values[0])
    best_h_values = np.array(best_h)
    best_h_values = np.clip(best_h_values, 0.0, 1.0)
    best_h_values = best_h_values * (total_integral / np.sum(best_h_values))
    best_c5 = evaluate_erdos_solution(best_h_values)

    # Local refinement with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=best_h_values,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 200}
    )
    best_h_values = result.x
    best_h_values = np.clip(best_h_values, 0.0, 1.0)
    best_h_values = best_h_values * (total_integral / np.sum(best_h_values))
    final_c5 = evaluate_erdos_solution(best_h_values)

    return best_h_values, final_c5, n_points
