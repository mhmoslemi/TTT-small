import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Balance between precision and computational cost
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def initial_guess(n_points, seed=42):
        # Start with the provided initial_values if available; else random
        if len(initial_h_values) == n_points:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0, 1]
            h = np.clip(h, 0, 1)
            h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

    def evaluate_and_optimize(h_initial):
        # Define the optimization problem using CVXPY
        x = cp.Variable(n_points)
        constraint = cp.sum(x) == total_integral
        # Objective: minimize the maximum overlap
        corr = np.correlate(x.value, 1 - x.value, mode="full")
        max_overlap = np.max(corr) * dx
        objective = cp.Minimize(max_overlap)

        # Use SLSQP with constraints
        result = opt.minimize(
            lambda x: objective.value,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 200}
        )
        return result.x, result.fun

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Run initial optimization
    try:
        optimized_h, optimized_c5 = evaluate_and_optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Remaining budget for perturbations and multiple optimizations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 15:
        # Perturb and optimize
        for _ in range(10):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = evaluate_and_optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    # Output the best solution
    return best_h, best_c5, n_points
