import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Increased number of points for better resolution
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initial values from the provided initial_h_values, scaled and normalized
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Initial optimization with SLSQP
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Perturb and optimize with multiple steps to avoid local minima
    for _ in range(10):
        perturbation = np.random.normal(0, 0.02, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Final fine-tuning with smaller perturbations
    for _ in range(10):
        perturbation = np.random.normal(0, 0.01, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Final evaluation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))

    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
