    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)

    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2

    # Optimization configuration
    options = {'maxiter': 100, 'disp': False, 'ftol': 1e-8}

    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)

    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points
    best_time = time.time()
    best_c5_history = [best_c5]

    # Strategies to improve the exploration
    strategies = [
        # Try increasing n_points
        lambda n_points, current_h: {
            'action': 'increase',
            'new_n_points': n_points + 10,
            'new_h': np.zeros(n_points + 10)
        },
        # Try decreasing n_points
        lambda n_points, current_h: {
            'action': 'decrease',
            'new_n_points': max(n_points - 10, 10),
            'new_h': np.zeros(n_points - 10)
        },
        # Try random perturbations
        lambda n_points, current_h: {
            'action': 'perturb',
            'new_n_points': n_points,
            'new_h': current_h + np.random.normal(0, 0.05, n_points)
        },
        # Try random initialization
        lambda n_points, current_h: {
            'action': 'random',
            'new_n_points': n_points,
            'new_h': np.random.rand(n_points) * (n_points / 2) / np.sum(np.random.rand(n_points))
        }
    ]

    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        # Try different strategies
        for strategy in strategies:

                if strategy_result['action'] == 'increase':

                    best_c5_history.append(new_c5)
                    print(f"Improved C5: {best_c5}")
                
                # Early stopping if improvement is negligible
                if len(best_c5_history) > 1 and best_c5_history[-1] - best_c5_history[-2] < 1e-6:
                    break
            except:
                continue

    return best_h, best_c5, best_n_points
