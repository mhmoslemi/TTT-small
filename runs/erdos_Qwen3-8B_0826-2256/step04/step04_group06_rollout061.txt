```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with a structured function (e.g., alternating values)
    h = np.zeros(n_points)
    h[:n_points//2] = 0.5
    h[n_points//2:] = 0.5
    # Ensure the integral is 1
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for h values
    bounds = Bounds(0, 1)
    
    # Initial best solution
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Try different optimization methods with adaptive parameters
    trials = 50
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Introduce randomness to escape local optima
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Use differential evolution for global optimization
        result_de = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            maxiter=50,
            popsize=15,
            tol=1e-6,
            mutation=(0.5, 1),
            recombination=0.8,
            disp=False
        )
        
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
        
        # Use SLSQP for local refinement
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
```
