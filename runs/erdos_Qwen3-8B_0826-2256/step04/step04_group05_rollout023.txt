```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points for efficiency and accuracy balance
    n_points_list = [100, 150, 200]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2
        
        # Use the initial_h_values and scale it
        h_initial = np.copy(initial_h_values[:n_points])
        h_initial = np.clip(h_initial, 0.0, 1.0)
        h_initial = h_initial * (total_integral / np.sum(h_initial))
        
        # Function to evaluate the objective
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Function to enforce the integral constraint
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        # Bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Initial optimization with SLSQP
        result = opt.minimize(
            fun=objective_function,
            x0=h_initial,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 300}
        )
        
        current_h = result.x
        current_c5 = result.fun
        
        # Perturb and optimize
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if result_pert.fun < current_c5:
                current_h = result_pert.x
                current_c5 = result_pert.fun
        
        # Fine-tuning
        for _ in range(5):
            perturbation = np.random.normal(0, 0.01, size=n_points)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result_pert.fun < current_c5:
                current_h = result_pert.x
                current_c5 = result_pert.fun
        
        # Final evaluation
        current_h = np.clip(current_h, 0.0, 1.0)
        current_h = current_h * (total_integral / np.sum(current_h))
        final_c5 = evaluate_erdos_solution(current_h)
        
        # Track best solution
        if final_c5 < best_c5:
            best_h = current_h
            best_c5 = final_c5
    
    # Return the best solution found
    return best_h, best_c5, len(best_h)
```
