import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Try different n_points values to optimize in a time-efficient way
    # Start with a moderate size to balance exploration and computation
    n_points = 100  # Initial trial
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start from initial_h_values and adjust it
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    best_h = np.copy(h)
    best_c5 = evaluate_erdos_solution(best_h)

    # Use a combination of approaches: optimization and direct search

    # Function to evaluate the objective
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to enforce the integral constraint
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    # Bounds and constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    # Use an advanced optimization method with tighter tolerances
    result = opt.minimize(
        fun=objective_function,
        x0=h,
        method='SLSQP',
        bounds=bounds,
        constraints=cons,
        options={'ftol': 1e-9, 'maxiter': 300}
    )
    best_h = result.x
    best_c5 = result.fun

    # Perform a more aggressive perturbation and optimization
    for _ in range(5):
        # Add random perturbation with a larger scale
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Fine-tuning: smaller perturbations for local refinement
    for _ in range(5):
        perturbation = np.random.normal(0, 0.01, size=n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_pert.fun < best_c5:
            best_h = result_pert.x
            best_c5 = result_pert.fun

    # Check performance with larger n_points if time allows
    # This serves as a fallback if we have remaining budget
    if budget_s > 50:
        n_points = 150
        dx = 2.0 / n_points
        total_integral = n_points / 2
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (total_integral / np.sum(h))
        # Re-define the objective function
        def objective_function_large(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap

        # Define the constraint function
        def constraint_func_large(h_values):
            return np.sum(h_values) - total_integral

        bounds_large = [(0.0, 1.0) for _ in range(n_points)]
        cons_large = {'type': 'eq', 'fun': constraint_func_large}

        result_large = opt.minimize(
            fun=objective_function_large,
            x0=h,
            method='SLSQP',
            bounds=bounds_large,
            constraints=cons_large,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        if result_large.fun < best_c5:
            best_h = result_large.x
            best_c5 = result_large.fun

    # Final evaluation and normalization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
