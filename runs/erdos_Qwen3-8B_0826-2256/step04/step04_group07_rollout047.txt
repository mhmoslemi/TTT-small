import numpy as np
import scipy.optimize as opt
import time
import multiprocessing as mp
from functools import partial
from scipy.optimize import minimize
from scipy.optimize import Bounds, LinearConstraint
from scipy.stats import norm
from scipy.interpolate import interp1d
from scipy.ndimage import convolve

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    best_h = h_values
    best_c5 = evaluate_erdos_solution(h_values)
    best_n_points = initial_n_points
    best_time = time.time()
    best_c5_history = [best_c5]
    
    # Helper function to evaluate C5 for a given h
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)
    
    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Optimization configuration
    options = {'maxiter': 100, 'disp': False, 'ftol': 1e-8}
    
    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)
    
    def perturb_h(h, perturbation_factor=0.05):
        perturbed = h + np.random.normal(0, perturbation_factor, len(h))
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (len(h) / 2) / np.sum(perturbed)
        return perturbed
    
    def refine_h(h, n_points, max_iter=50):
        for _ in range(max_iter):
            # Local optimization
            bounds = [(0.0, 1.0) for _ in range(n_points)]
            cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
            result = minimize(lambda x: objective(x, n_points), h, method='L-BFGS-B', bounds=bounds, constraints=cons, options=options)
            h = result.x
            if np.max(np.abs(result.fun - objective(h, n_points))) < 1e-7:
                break
        return h
    
    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points
    best_c5_history = [best_c5]
    
    # Adaptive parameter sweep with local refinement
    start_time = time.time()
    while time.time() - start_time < budget_s - 5:
        # Try increasing n_points
        new_n_points = best_n_points + 10
        if new_n_points > 1000:
            break
        
        # Initialize with previous best values
        new_h = np.zeros(new_n_points)
        new_h[:best_n_points] = best_h
        # Fill the rest with random values and normalize
        new_h[best_n_points:] = np.random.rand(new_n_points - best_n_points)
        new_h = new_h * (new_n_points / 2) / np.sum(new_h)
        
        # Refine the new_h
        new_h = refine_h(new_h, new_n_points)
        new_c5 = evaluate_c5(new_h, new_n_points)
        
        # Check if the new solution is better
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
            best_n_points = new_n_points
            best_c5_history.append(new_c5)
            print(f"Improved C5: {best_c5}")
        else:
            # Try local refinement by perturbing current solution
            perturbed_h = perturb_h(best_h)
            perturbed_h = refine_h(perturbed_h, best_n_points)
            perturbed_c5 = evaluate_c5(perturbed_h, best_n_points)
            
            if perturbed_c5 < best_c5:
                best_h, best_c5 = perturbed_h, perturbed_c5
                best_c5_history.append(perturbed_c5)
                print(f"Improved C5: {best_c5}")
            else:
                # Check if improvement is too small to continue
                if len(best_c5_history) > 1 and best_c5_history[-1] - best_c5_history[-2] < 1e-5:
                    break
    
    return best_h, best_c5, best_n_points
