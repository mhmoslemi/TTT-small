```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Experiment with different number of points
    n_points_list = [150, 200, 250]
    best_h = None
    best_c5 = float('inf')
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # Sum(h) == n_points / 2
        
        # Initial guess based on existing values or random
        if len(initial_h_values) == n_points:
            h_initial = initial_h_values.copy()
        else:
            h_initial = np.random.rand(n_points)
            h_initial = (h_initial * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
            h_initial = np.clip(h_initial, 0, 1)
            h_initial = h_initial * (total_integral / np.sum(h_initial))  # Ensure integral constraint
        
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Define the optimization problem using CVXPY
        x = cp.Variable(n_points)
        constraint = cp.sum(x) == total_integral
        objective = cp.Minimize(cp.Max(np.correlate(x, 1 - x, mode="full")) * dx)
        
        # Solve the optimization problem
        try:
            problem = cp.Problem(objective, [constraint])
            problem.solve(solver='SLSQP', eps=1e-8, max_iter=200)
            if problem.status == 'optimal':
                h_optimized = x.value
                c5_bound = objective.value
                if c5_bound < best_c5:
                    best_c5 = c5_bound
                    best_h = h_optimized
        except:
            pass
        
        # Optional: Perturb and optimize further if time remains
        start_time = time.time()
        remaining_time = budget_s - (time.time() - start_time)
        if remaining_time > 10:
            for _ in range(5):
                perturbed_h = best_h + 0.05 * np.random.randn(n_points)
                perturbed_h = np.clip(perturbed_h, 0, 1)
                perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
                try:
                    problem = cp.Problem(cp.Minimize(cp.Max(np.correlate(perturbed_h, 1 - perturbed_h, mode="full")) * dx), [cp.sum(perturbed_h) == total_integral])
                    problem.solve(solver='SLSQP', eps=1e-8, max_iter=200)
                    if problem.status == 'optimal':
                        h_optimized = perturbed_h
                        c5_bound = objective_function(h_optimized)
                        if c5_bound < best_c5:
                            best_c5 = c5_bound
                            best_h = h_optimized
                except:
                    pass
    
    return best_h, best_c5, n_points
```
