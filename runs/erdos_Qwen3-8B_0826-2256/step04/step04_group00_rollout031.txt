import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values for efficiency and accuracy
    # Start with 100 points, then try 200 if time permits
    n_points = 100
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2
    
    # Start with initial values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint
    
    best_h = h
    best_c5 = evaluate_erdos_solution(h)
    best_n_points = n_points
    
    # Track time and run optimization
    start_time = time.time()
    
    # Try different n_points if time permits
    for n_points in [100, 200]:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2
        
        # Initialize h for current n_points
        h = np.random.rand(n_points)
        h = np.clip(h, 0.0, 1.0)
        h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint
        
        best_h_current = h
        best_c5_current = evaluate_erdos_solution(h)
        
        # Define objective function
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Define constraint function
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        # Define bounds and constraints
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Run optimization
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 200}
        )
        
        if result.fun < best_c5_current:
            best_c5_current = result.fun
            best_h_current = result.x

        perturbations = [np.random.normal(0, 0.05, size=n_points) for _ in range(5)]
        for pert in perturbations:

            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 200}
            )
            if result_pert.fun < best_c5_current:
                best_c5_current = result_pert.fun
                best_h_current = result_pert.x
        
        # Random restart to escape local optima
        random_h = np.random.rand(n_points)
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))
        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='SLSQP',
            bounds=bounds,

            best_h = best_h_current
            best_n_points = n_points
        
        # Check if time is up

            break
    
    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
