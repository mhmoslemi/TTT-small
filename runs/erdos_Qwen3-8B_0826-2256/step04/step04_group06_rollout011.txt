import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a higher number of points for better precision
    n_points = 200
    dx = 2.0 / n_points

    # Use the initial_h_values if provided, otherwise use a more informed initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initialize h with a more structured approach based on frequency
        h = np.zeros(n_points)
        h[:n_points//3] = 1.0
        h[n_points//3:2*n_points//3] = 0.0
        h[2*n_points//3:] = 1.0
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    
    # Initial best solution
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    elapsed = 0

    # Use a global optimization technique to explore the solution space
    # Define bounds for the differential evolution
    bounds_de = [(0, 1) for _ in range(n_points)]
    
    # Define the objective function for differential evolution
    def objective_de(h_values):
        return objective(h_values)

    # Run differential evolution for global exploration
    result_de = differential_evolution(
        objective_de,
        bounds_de,
        strategy='best1bin',
        maxiter=100,
        popsize=20,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        seed=seed
    )
    
    # Update best solution from differential evolution
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun
        print(f"Differential Evolution: Best C5 = {best_c5}")
    
    # Use local optimization to refine the solution
    # Run SLSQP for local refinement starting from the best solution found
    result_slsqp = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )
    
    # Update best solution if SLSQP improves it
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x
        print(f"SLSQP: Best C5 = {best_c5}")
    
    # Perform additional refinement with COBYLA
    result_cobyla = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraints},
        method="COBYLA",
        options={"maxiter": 100}
    )
    
    # Update best solution if COBYLA improves it
    if result_cobyla.success:
        best_c5 = result_cobyla.fun
        best_h = result_cobyla.x
        print(f"COBYLA: Best C5 = {best_c5}")
    
    # Ensure we do not exceed the budget
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        print(f"Time limit reached. Final best C5 = {best_c5}")
        return best_h, best_c5, n_points
    
    return best_h, best_c5, n_points
