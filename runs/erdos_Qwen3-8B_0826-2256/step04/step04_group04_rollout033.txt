```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Start with a higher resolution for better quality, but keep it under 1000 points
    n_points = 150
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Initial guess using the provided initial_h_values or random
    def initial_guess(n_points, seed=42):
        if len(initial_h_values) == n_points:
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

    # Define the objective function to minimize the maximum overlap
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Optimization function using SLSQP
    def optimize_with_slsqp(h_initial):
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        result = opt.minimize(
            fun=objective,
            x0=h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 300}
        )
        return result.x, result.fun

    # Start with an initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Run the main optimization
    try:
        optimized_h, optimized_c5 = optimize_with_slsqp(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # Use remaining time for more optimization attempts
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        # Try multiple perturbations and refine
        for _ in range(5):
            perturbed_h = best_h + 0.05 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize_with_slsqp(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    return best_h, best_c5, n_points
```
