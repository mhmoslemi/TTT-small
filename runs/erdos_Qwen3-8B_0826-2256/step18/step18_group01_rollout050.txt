    def objective(h_values):

        c5 = np.max(overlap) * (2.0 / n_points)

    def differential_evolution_objective(h_values):
        return objective(h_values)

    de_bounds = [(0, 1) for _ in range(n_points)]
    de_constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

    result_de = differential_evolution(
