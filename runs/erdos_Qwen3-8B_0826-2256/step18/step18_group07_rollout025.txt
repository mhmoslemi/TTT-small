    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    best_c5 = objective(h)
    start_time = time.time()
    
    # Use differential evolution with constraints for global exploration
    def differential_evolution_with_constraints():
        nonlocal best_h, best_c5
        try:
            # Define the constraint
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            
            # Define the bounds and objective function
            result = differential_evolution(
                objective,
                bounds,
                constraints=constraint,
                strategy='best1bin',
                popsize=15,
                mutation=(0.5, 1),
                recombination=0.8,
                tol=1e-10,
                maxiter=500,
                disp=False
            )
            
            # Update best solution if improvement is found
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass

    # Run differential evolution with constraints
    trials = 1  # Single trial to avoid overuse of time budget

        differential_evolution_with_constraints()

    if time.time() - start_time < budget_s - 1:

        result = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints=constraint,
            method="SLSQP",
            options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
        )
        if result.success:
            best_c5 = result.fun
            best_h = result.x
