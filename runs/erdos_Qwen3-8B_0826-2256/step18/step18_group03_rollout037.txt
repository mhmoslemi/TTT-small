        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Differential Evolution with constraints for global search
    if time.time() - start_time < budget_s - 1:
        # Define the objective and constraints for differential evolution
        def objective_de(h_values):
            return objective(h_values)
        
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        
        # Set up bounds and constraints
        de_bounds = [(0, 1) for _ in range(n_points)]
        
        # Run differential evolution
        try:
            result = differential_evolution(
                objective_de,
                bounds=de_bounds,
                constraints=constraint,
                strategy='best1bin',
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                maxiter=100,
                disp=False
            )
            best_h = result.x
            best_c5 = result.fun
        except:
            pass
    
    # Local refinement with SLSQP
    if time.time() - start_time < budget_s - 1:
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 250, "eps": 1e-8}
            )
            if result.success:
                best_h = result.x
                best_c5 = result.fun
        except:
            pass
    
    return best_h, best_c5, n_points
