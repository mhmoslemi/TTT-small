import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and high-quality starting point
    
    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    trials = 150  # Balanced exploration and refinement
    perturbation_scale = 0.15  # Initial perturbation scale for diversity
    learning_rate = 0.01  # Learning rate for gradient-based methods
    population_size = 30  # Size of population for evolutionary search
    mutation_rate = 0.1  # Rate of mutation in evolutionary algorithm

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Evolutionary search for better solutions
        population = [np.clip(h + np.random.uniform(-0.05, 0.05, size=n_points), 0, 1) for _ in range(population_size)]
        population = [h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else h for h in population]
        population = [np.clip(h, 0, 1) for h in population]

        # Evaluate all in population
        fitness = [objective(h) for h in population]
        best_index = np.argmin(fitness)
        best_c5 = fitness[best_index]
        best_h = population[best_index]

        # Generate new population with mutation and selection
        new_population = []
        for h in population:
            # Mutate
            mutated = h + np.random.uniform(-mutation_rate, mutation_rate, size=n_points)
            mutated = np.clip(mutated, 0, 1)
            mutated = mutated / (np.sum(mutated) * 2.0 / n_points) if np.sum(mutated) != 0 else mutated
            mutated = np.clip(mutated, 0, 1)
            new_population.append(mutated)
        
        # Select top performers
        population = [np.clip(h, 0, 1) for h in population + new_population]
        population = [h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else h for h in population]
        population = [np.clip(h, 0, 1) for h in population]
        population = [h for h in population if np.sum(h) * 2.0 / n_points > 0.95 and np.sum(h) * 2.0 / n_points < 1.05]
        population = population[:population_size]
        
        # Use COBYLA for fine-tuning
        if len(population) > 0:
            h = population[0]
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            try:
                result = minimize(
                    objective,
                    h,
                    bounds=bounds,
                    constraints=constraint,
                    method="COBYLA",
                    options={"maxiter": 200}
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.97
    
    return best_h, best_c5, n_points
