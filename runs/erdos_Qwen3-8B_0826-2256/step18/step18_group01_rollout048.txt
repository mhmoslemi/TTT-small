    def objective(h_values):

        c5 = np.max(overlap) * (2.0 / n_points)

    start_time = time.time()
    
    # Use differential evolution for global exploration
    def differential_evolution_objective(h_values):
        return objective(h_values)

    def integral_constraint_diff_ev(h_values):
        return np.sum(h_values) - n_points / 2

    bounds_diff_ev = [(0, 1) for _ in range(n_points)]
