import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, differential_evolution, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and high-quality starting point
    
    # Use initial_h_values if available for better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Function to evaluate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Initial steps with differential evolution for exploration
    def differential_evolution_objective(h_values):
        return objective(h_values)

    # Define constraints for differential evolution
    def integral_constraint_de(h_values):
        return np.sum(h_values) - n_points / 2

    de_bounds = np.array([[0, 1]] * n_points)
    de_result = differential_evolution(
        differential_evolution_objective,
        bounds=de_bounds,
        constraints=[{"type": "eq", "fun": integral_constraint_de}],
        strategy='best1bin',
        popsize=15,
        maxiter=50,
        tol=1e-10,
        mutation=(0.5, 1),
        recombination=0.8,
        polish=True,
        disp=False
    )
    
    best_h = de_result.x
    best_c5 = de_result.fun
    
    # Refinement stage using local optimization
    constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Use SLSQP for local refinement
        try:
            result_slsqp = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-10, "maxiter": 200, "eps": 1e-8}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Use COBYLA for further exploration
        try:
            result_cobyla = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 250}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass
    
    return best_h, best_c5, n_points
