    h = np.clip(h, 0, 1)

        return np.max(overlap) * dx

    bounds = Bounds(0, 1)
    best_h = h

    # Use differential evolution for global search, then refine with SLSQP

    de_result = differential_evolution(

        maxiter=50,
        tol=1e-5,
