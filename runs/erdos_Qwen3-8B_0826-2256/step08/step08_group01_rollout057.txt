    dx = 2.0 / n_points

        h[1::2] = 0.0  # Set the other points to 0.0
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of trials to explore different solutions
    perturbation_scale = 0.5  # Initial perturbation scale

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Use L-BFGS-B for local optimization
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="L-BFGS-B",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        if result.success:
            current_c5 = result.fun

                best_c5 = current_c5
                best_h = result.x

        perturbation_scale *= 0.95
