import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Start with a simple alternating pattern
        h = np.zeros(100)
        h[::2] = 0.5
        h[1::2] = 0.5
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * 2.0 / len(h)
    
    def constraint(h_values):
        return np.sum(h_values) - len(h_values) / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Perform a structured search with perturbation and local optimization
    n_points = len(h)
    trials = 100  # Number of trials
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else h
        
        # Use SLSQP for local refinement
        try:
            result = minimize(
                evaluate_overlap,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Use COBYLA to explore new regions
        try:
            result = minimize(
                evaluate_overlap,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
    
    return best_h, best_c5, n_points
