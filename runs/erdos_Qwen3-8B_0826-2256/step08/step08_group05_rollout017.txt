import numpy as np
import scipy.optimize as optimize
import time

from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 100 points for a balance between speed and quality
    n_points = 100
    dx = 2.0 / n_points

    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initialize with a simple alternating pattern
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a combination of optimization strategies for robustness
    # 1. Use differential evolution for global search
    # 2. Use SLSQP for fine-tuning
    # 3. Use a greedy local search for improvement
    
    # First, perform differential evolution for global search
    def objective_de(x):
        return evaluate_overlap(x)
    
    result_de = differential_evolution(
        objective_de,
        bounds,
        strategy='best1bin',
        popsize=15,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.9,
        maxiter=100,
        disp=False,
        polish=True,
        x0=h
    )
    
    best_h = result_de.x
    best_c5 = result_de.fun
    
    # Use SLSQP for fine-tuning
    result_slsqp = minimize(
        evaluate_overlap,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 150}
    )
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    # Use a local greedy descent to further improve
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the solution
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h_perturbed = best_h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) if np.sum(h_perturbed) != 0 else best_h
        
        # Evaluate the perturbed solution
        current_c5 = evaluate_overlap(h_perturbed)
        if current_c5 < best_c5:
            best_h = h_perturbed
            best_c5 = current_c5
    
    return best_h, best_c5, n_points
