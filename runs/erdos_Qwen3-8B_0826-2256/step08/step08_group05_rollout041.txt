import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()

        # Initialize with a simple pattern that satisfies the integral constraint
        h = np.zeros(100)
        h[::2] = 0.5
        h[1::2] = 0.5
        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * 2.0 / 100) if np.sum(h) != 0 else np.random.rand(100)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * 2.0 / len(h)
    
    def constraint(h_values):
        return np.sum(h_values) - 50  # 50 = 100 / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a combination of global and local search methods
    # 1. Differential evolution for global search
    try:
        result_de = differential_evolution(
            evaluate_overlap,
            bounds,
            strategy='best1bin',
            maxiter=50,
            popsize=20,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            constraints=[{"type": "eq", "fun": constraint}]
        )
        if result_de.success:
            best_h = result_de.x
            best_c5 = result_de.fun
    except:
        pass
    
    # 2. SLSQP for refinement
    try:
        result_slsqp = minimize(
            evaluate_overlap,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        if result_slsqp.success:
            best_h = result_slsqp.x
            best_c5 = result_slsqp.fun
    except:
        pass
    
    # 3. COBYLA for exploration
    try:
        result_cobyla = minimize(
            evaluate_overlap,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="COBYLA",
            options={"maxiter": 150}
        )
        if result_cobyla.success:
            best_h = result_cobyla.x
            best_c5 = result_cobyla.fun
    except:
        pass
    
    # 4. Nelder-Mead for further refinement
    try:
        result_nelder = minimize(
            evaluate_overlap,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="Nelder-Mead",
            options={"maxiter": 150}
        )
        if result_nelder.success:
            best_h = result_nelder.x
            best_c5 = result_nelder.fun
    except:
        pass
    
    # Final check to ensure the integral constraint is satisfied
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) * 2.0 / len(best_h)) if np.sum(best_h) != 0 else best_h
    
    return best_h, best_c5, len(best_h)
