import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Balance between detail and computational speed
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = np.zeros((n_points, 2))
    bounds[:, 0] = 0
    bounds[:, 1] = 1
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Initial global exploration using a perturbation-based approach
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Use SLSQP for local refinement
        result_slsqp = optimize.minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150, "disp": False}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    # Local refinement with random restarts
    for _ in range(30):
        if time.time() - start_time > budget_s - 1:
            break
        perturbation = np.random.uniform(-0.05, 0.05, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        result_cobyla = optimize.minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="COBYLA",
            options={"maxiter": 150, "disp": False}
        )
        if result_cobyla.success:
            current_c5 = result_cobyla.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_cobyla.x
    
    return best_h, best_c5, n_points
