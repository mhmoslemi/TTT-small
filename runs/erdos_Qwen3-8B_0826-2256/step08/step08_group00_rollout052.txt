import scipy.optimize as optimize

from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = 100
    dx = 2.0 / n_points

    else:

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Maximum number of trials
    
    # Use a combination of local and global methods with a focus on diversity in perturbations
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution with a random direction
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral remains 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Run a robust optimization with the SLSQP method for local refinement
        result_slsqp = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        
        # Update best solution if improvement is found
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
        
        # Use the Nelder-Mead simplex method for global exploration
        result_nelder = minimize(

            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"maxiter": 150, "ftol": 1e-9}
        )
        
        # Update best solution if improvement is found
        if result_nelder.success:
            current_c5 = result_nelder.fun
            if current_c5 < best_c5:

                best_h = result_nelder.x
    
    return best_h, best_c5, n_points
