import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Initialize with the provided initial_h_values or a reasonable structure
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Perform a guided search with perturbation and multiple trials
    for trial in range(150):  # Increased number of trials for better exploration
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.2 * (1 - trial / 150), 0.2 * (1 - trial / 150), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Use a combination of methods with adaptive strategy
        try:
            # Switch to SLSQP for local refinement
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

            # Use COBYLA for robustness
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x

        except:
            # Handle exceptions gracefully
            pass

    # Optionally, perform a final differential evolution search
    if time.time() - start_time < budget_s - 1:
        result_de = differential_evolution(
            objective,
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={"type": "eq", "fun": constraints},
            maxiter=50,
            popsize=15,
            mutation=(0.5, 1),
            recombination="blend",
            tol=1e-6
        )
        if result_de.success:
            best_h = result_de.x
            best_c5 = result_de.fun

    return best_h, best_c5, n_points
