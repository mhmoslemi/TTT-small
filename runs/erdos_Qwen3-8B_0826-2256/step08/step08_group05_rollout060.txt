def run(seed=42, budget_s=100, **kwargs):

    # Start with 200 points to balance speed and quality
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial_h_values if available

        h = initial_h_values.copy()
    else:
        # Initialize with a simple alternating pattern
        h = np.zeros(n_points)

        h[1::2] = 0.5
        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use differential evolution for global optimization
    bounds_array = np.array([(0, 1) for _ in range(n_points)])
    
    # Custom differential evolution with early stopping
    def objective_function(x):
        return evaluate_overlap(x)
    
    # Initial population
    initial_population = np.random.uniform(0, 1, (10, n_points))
    initial_population = initial_population / np.sum(initial_population, axis=1, keepdims=True) * (n_points / 2)
    initial_population = np.clip(initial_population, 0, 1)
    
    # Setup differential evolution
    result = differential_evolution(
        objective_function,
        bounds_array,
        popsize=10,
        tol=1e-6,
        mutation=(0.5, 1),
        recombination=0.8,
        strategy='best1bin',
        maxiter=100,
        polish=True,
        disp=False,
        vectorized=True
    )
    
    best_h = result.x
    best_c5 = result.fun
    
    # Early stopping if time is exceeded
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    
    # Refinement with SLSQP
    result_slsqp = minimize(
        evaluate_overlap,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 150}
    )
    
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun
    
    return best_h, best_c5, n_points
