import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 100 points for a balance of speed and accuracy
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided starting point if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to a simple alternating pattern
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        # Reshape and compute the overlap
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        # Enforce the integral constraint
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Run a more structured optimization with diverse strategies
    # Adaptive splitting based on time remaining
    time_remaining = budget_s - (time.time() - start_time)
    num_tries = int(time_remaining / 1.5)
    if num_tries < 1:
        num_tries = 1
    
    # Use a different perturbation strategy based on trial number
    for trial in range(num_tries):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Apply a noise-based perturbation to escape local optima
        perturbation = np.random.normal(0, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Adjust to maintain the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        x0 = h
        
        # Run a combination of optimization methods for robustness
        try:
            # SLSQP for precise optimization under constraints
            result_slsqp = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        try:
            # COBYLA for robust exploration in constrained spaces
            result_cobyla = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass
        
        try:
            # Nelder-Mead for global search in high-dimensional spaces
            result_nelder = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="Nelder-Mead",
                options={"maxiter": 150}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x
        except:
            pass
    
    return best_h, best_c5, n_points
