import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 100 points for a balance of speed and quality
    n_points = 100
    dx = 2.0 / n_points
    
    # Use the initial construction if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Start with a simple alternating pattern
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
        # Ensure integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a different starting point to avoid local optima
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a diverse perturbation
        perturbation = np.random.uniform(-0.5, 0.5, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize to meet the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        x0 = h
        
        # Use SLSQP with tighter tolerances
        try:
            result = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150, "disp": False}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Use a random restart to escape local optima
        if np.random.rand() < 0.3:
            h = np.random.uniform(0, 1, n_points)
            h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
    
    return best_h, best_c5, n_points
