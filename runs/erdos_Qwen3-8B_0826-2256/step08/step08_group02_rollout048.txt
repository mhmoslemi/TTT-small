import numpy as np
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution
from scipy.optimize import basinhopping

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for balance between speed and quality
    dx = 2.0 / n_points
    initial_h_values = np.zeros(n_points)
    initial_h_values[:n_points//2] = 0.5
    initial_h_values[n_points//2:] = 0.5
    # Ensure integral is 1
    initial_h_values = initial_h_values / (np.sum(initial_h_values) * dx) if np.sum(initial_h_values) != 0 else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)
    best_h = initial_h_values.copy()
    best_c5 = calculate_c5(best_h, dx)
    start_time = time.time()
    # Use basinhopping for global optimization
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    result = basinhopping(
        objective,
        best_h,
        niter=100,  # Number of basin hopping iterations
        T=1.0,      # Initial temperature
        stepsize=0.5,  # Step size for generating trial points
        minimizer_kwargs={"method": "L-BFGS-B", "bounds": bounds, "constraints": {"type": "eq", "fun": constraints}},
        interval=50,  # Interval between successive attempts to find the global minimum
        niter_success=5,  # Number of successive success iterations after which T is reduced
        take_walks=True,  # Generate more trial points
        polish=True,  # Refine the solution with L-BFGS-B
        disp=False
    )
    best_h = result.x
    best_c5 = calculate_c5(best_h, dx)
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) * dx) if np.sum(best_h) != 0 else np.random.rand(n_points)
    best_h = np.clip(best_h, 0, 1)
    best_c5 = calculate_c5(best_h, dx)
    # Check if we have exceeded the budget
    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
    return best_h, best_c5, n_points

def calculate_c5(h, dx):
    overlap = np.correlate(h, 1 - h, mode="full")
    c5 = np.max(overlap) * dx
    return c5
