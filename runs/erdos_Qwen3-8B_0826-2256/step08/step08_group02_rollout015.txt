import numpy as np
import time
import scipy.optimize as optimize
import random

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points for efficiency

    # Initialize with a known construction if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initial guess: a simple alternating pattern
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure integral constraint
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)

    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)

    def compute_c5(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * (2.0 / n_points)

    # Constraint function to ensure integral is 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]

    best_c5 = compute_c5(h)
    best_h = h.copy()

    start_time = time.time()
    total_trials = 150
    cooling_factor = 0.9

    for trial in range(total_trials):
        if time.time() - start_time > budget_s - 1:
            break

        # Perturb the current best with decreasing magnitude
        perturbation = np.random.uniform(-0.1 * (1 - trial / total_trials), 0.1 * (1 - trial / total_trials), n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Local search with SLSQP
        result = optimize.minimize(
            compute_c5,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": integral_constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )

        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x

    return best_h, best_c5, n_points
