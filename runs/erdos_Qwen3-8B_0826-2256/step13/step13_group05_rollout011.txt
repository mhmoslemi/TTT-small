            elif i % 3 == 1:
                h[i] = 0.5
            else:
                h[i] = 0.0
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)

    # Objective function: minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to maintain integral equal to 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    for trial in range(trials):

        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.1 * (1 - trial / trials), 0.1 * (1 - trial / trials), size=n_points)

        h = np.clip(h, 0, 1)
        # Re-normalize to maintain the integral condition
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])

                x0,

                constraints=constraint,

                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:

            result_de = minimize(
                objective,

                if current_c5 < best_c5:
