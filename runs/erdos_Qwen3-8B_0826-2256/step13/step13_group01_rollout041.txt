import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Optimized for speed and quality
    
    # Initialize with a better starting point if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    # Define the objective and constraint
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    best_c5 = min(best_c5, 0.381081)  # Start with the prior best as an upper bound
    start_time = time.time()
    trials = 200  # Increased exploration for better global search
    perturbation_scale = 0.25  # Larger initial scale for global movement
    cooling_rate = 0.995  # Simulated annealing-style cooling

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        h = h * (n_points / 2) / np.sum(h) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Use SLSQP for local refinement
        constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # Use COBYLA for global search
        try:
            result_cobyla = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass

        # Simulated annealing cooling
        perturbation_scale *= cooling_rate
    
    return best_h, best_c5, n_points
