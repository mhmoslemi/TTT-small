    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5

    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    perturbation_scale = 0.2  # Larger initial perturbation for exploration
    learning_rate = 0.01  # Learning rate for gradient-based methods
    num_threads = 10  # Number of threads for parallel computation

    for trial in range(trials):

        # Generate perturbed solutions in parallel
        perturbations = [np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points) for _ in range(num_threads)]
        solutions = [best_h + perturbation for perturbation in perturbations]
        solutions = [np.clip(sol, 0, 1) for sol in solutions]

        solutions = [sol / (np.sum(sol) * 2.0 / n_points) if np.sum(sol) != 0 else np.random.rand(n_points) for sol in solutions]
