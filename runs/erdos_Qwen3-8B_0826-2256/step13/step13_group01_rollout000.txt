import scipy.optimize as opt
import time
import random
from scipy.optimize import Bounds, minimize
from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency and quality
    
    # Initialize with the provided initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    
    # Function to compute the overlap integral
    def compute_c5(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * (2.0 / n_points)
    
    # Constraint to maintain the integral of h(x) over [0, 2] as 1
    def integral_constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = compute_c5(h)
    start_time = time.time()
    trials = 300  # Increase number of trials for better exploration
    perturbation_scale = 0.4  # Start with a larger perturbation scale
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-perturbation_scale, perturbation_scale, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is approximately met
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Use L-BFGS-B for local refinement
        try:
            constraint = LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2])
            result = minimize(
                compute_c5,
                x0,
                bounds=bounds,
                constraints=constraint,
                method="L-BFGS-B",
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Use differential evolution for global exploration
        try:
            result = minimize(
                compute_c5,
                x0,
                bounds=bounds,
                constraints=LinearConstraint(np.ones(n_points), [n_points / 2], [n_points / 2]),

                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Reduce perturbation scale for finer tuning
        perturbation_scale *= 0.98
    
    return best_h, best_c5, n_points
