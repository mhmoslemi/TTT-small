    def evaluate_c5(h, n_points):

        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    def objective(x, n_points):
        return evaluate_c5(x, n_points)

        return np.sum(x) - n_points / 2
    
    # Optimization configuration
    options = {'maxiter': 100, 'disp': False}

    def optimize_h(n_points, initial_h):
        bounds = Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = opt.minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
