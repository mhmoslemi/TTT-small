import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import Bounds, differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use 200 points for better resolution
    dx = 2.0 / n_points

    # Use the initial_h_values if provided, otherwise use a more refined initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Use a more structured initialization based on periodic patterns
        h = np.zeros(n_points)
        # Create a "checkerboard" pattern with 20 segments
        segments = 20
        for i in range(n_points):
            segment_idx = i % segments
            if segment_idx < segments // 2:
                h[i] = 0.5
            else:
                h[i] = 0.0
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Use differential evolution for global optimization
    def differential_evolution_run():
        result = differential_evolution(
            objective,
            bounds,
            constraints={"type": "eq", "fun": constraints},
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            polish=True,
            disp=False
        )
        if result.success:
            return result.x, result.fun
        else:
            return h, best_c5

    # Run differential evolution
    if time.time() - start_time < budget_s - 1:
        best_h, best_c5 = differential_evolution_run()

    # Local refinement with SLSQP
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass

    return best_h, best_c5, n_points
