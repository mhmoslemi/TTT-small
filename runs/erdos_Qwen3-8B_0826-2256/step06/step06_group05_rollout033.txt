import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize.optimize import OptimizeResult

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points to balance speed and quality
    dx = 2.0 / n_points

    # Initialize with the provided initial construction
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to maintain the integral equal to 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h.copy()
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of optimization trials
    perturbation_scale = 0.15  # Initial perturbation scale

    # Function to perturb a solution
    def perturb_solution(solution, scale):
        perturbation = np.random.uniform(-scale, scale, size=len(solution))
        return np.clip(solution + perturbation, 0, 1)

    # Function to refine a solution with optimization
    def refine_solution(solution):
        result = minimize(
            objective,
            solution,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        return result.x if result.success else solution

    # Main optimization loop
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        h = perturb_solution(best_h, perturbation_scale)
        h = np.clip(h, 0, 1)
        # Enforce the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        # Refine the perturbed solution
        h = refine_solution(h)
        # Evaluate the refined solution
        current_c5 = objective(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h

        # Reduce perturbation scale over time
        perturbation_scale *= 0.95

    return best_h, best_c5, n_points
