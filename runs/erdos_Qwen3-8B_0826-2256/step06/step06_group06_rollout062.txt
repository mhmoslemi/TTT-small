import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import Bounds
from scipy.optimize import minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial construction from initial_h_values
    initial_n_points = len(initial_h_values)
    h_values = initial_h_values.copy()
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Optimization configuration
    options = {'maxiter': 100, 'disp': False}
    
    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h):
        bounds = Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        result = minimize(lambda x: evaluate_c5(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)
    
    # Function to perform a local search with perturbations
    def local_search(h, n_points):
        best_h = h.copy()
        best_c5 = evaluate_c5(best_h, n_points)
        
        for _ in range(10):
            perturbed_h = best_h + np.random.normal(0, 0.01, n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (n_points / 2) / np.sum(perturbed_h)
            current_c5 = evaluate_c5(perturbed_h, n_points)
            
            if current_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = current_c5
        
        return best_h, best_c5
    
    # Function to perform a grid search over n_points with multiple initializations
    def grid_search(n_start, n_end, step, initial_h):
        best_h = initial_h.copy()
        best_c5 = evaluate_c5(best_h, len(best_h))
        
        for n_points in range(n_start, n_end + 1, step):
            # Initialize with previous best values
            new_h = np.zeros(n_points)
            new_h[:len(best_h)] = best_h
            # Fill the rest with random values and normalize
            new_h[len(best_h):] = np.random.rand(n_points - len(best_h))
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Local search to refine
            refined_h, refined_c5 = local_search(new_h, n_points)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
        
        return best_h, best_c5, len(best_h)
    
    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points
    
    start_time = time.time()
    
    # Perform grid search over n_points
    n_start = initial_n_points
    n_end = min(initial_n_points + 200, 1000)
    step = 10
    best_h, best_c5, best_n_points = grid_search(n_start, n_end, step, best_h)
    
    # If time remains, perform perturbative local search
    if time.time() - start_time < budget_s - 5:
        best_h, best_c5 = local_search(best_h, best_n_points)
    
    return best_h, best_c5, best_n_points
