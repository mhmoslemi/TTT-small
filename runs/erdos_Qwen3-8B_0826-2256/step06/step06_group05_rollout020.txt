import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)

        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 50  # Number of trials for DE
    max_iter = 40  # Maximum iterations for DE
    pop_size = 15  # Population size for DE

    # Use differential evolution for global optimization
    def de_objective(x):
        return objective(x)

    # Define bounds for differential evolution
    de_bounds = [(0, 1) for _ in range(n_points)]

    # Run differential evolution
    try:
        result = differential_evolution(
            de_objective,
            de_bounds,
            strategy='best1bin',
            popsize=pop_size,
            mutation=(0.5, 1),
            recombination=0.8,
            tol=1e-8,
            maxiter=max_iter,
            disp=False,
            polish=True,
            callback=None,
            vectorized=True
        )
        # Update best_h and best_c5 if DE succeeds
        best_h = result.x
        best_c5 = result.fun
    except:
        # Fallback to local optimization if DE fails
        pass

    # Perform local refinement with SLSQP
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    except:
        pass

    # Perform local refinement with Nelder-Mead
    try:
        result_nelder = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="Nelder-Mead",

            current_c5 = result_nelder.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_nelder.x
    except:
        pass

    if time.time() - start_time > budget_s - 1:
        return best_h, best_c5, n_points
