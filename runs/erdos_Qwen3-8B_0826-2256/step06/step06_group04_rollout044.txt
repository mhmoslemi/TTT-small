    h = np.clip(h, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    start_time = time.time()

    population_size = 50  # Genetic algorithm population size
    mutation_rate = 0.1  # Mutation rate
    cooldown = 10  # Reduce perturbation magnitude over time

    population = np.random.rand(population_size, n_points)
    population = np.clip(population, 0, 1)

    population = np.clip(population, 0, 1)

    fitness = np.array([objective(ind) for ind in population])

    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]

    for trial in range(trials):

    h = best_h

            if current_c5 < best_c5:

                best_h = result_slsqp.x

        pass

    return best_h, best_c5, n_points
