import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different n_points values for efficiency and performance
    n_points_list = [100, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 100
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2
        
        # Start with a better initial construction
        if n_points == 100:
            h = np.zeros(n_points)
            for i in range(n_points):
                h[i] = 1.0 if i < n_points // 2 else 0.0
            h = h * (total_integral / np.sum(h))
        else:
            # For larger n_points, use a more structured initial guess
            h = np.zeros(n_points)
            for i in range(n_points):
                if i % 3 == 0:
                    h[i] = 1.0
                elif i % 3 == 1:
                    h[i] = 0.5
                else:
                    h[i] = 0.0
            h = h * (total_integral / np.sum(h))
        
        # Function to compute the overlap integral
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Constraint function
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        start_time = time.time()
        elapsed_time = 0
        
        # Initial optimization
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        current_c5 = result.fun
        current_h = result.x
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h
            best_n_points = n_points
        
        # Generate diverse perturbations for exploration
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbation = np.clip(perturbation, -0.1, 0.1)
            perturbed_h = current_h + perturbation
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result_pert.fun < best_c5:
                best_c5 = result_pert.fun
                best_h = result_pert.x
                best_n_points = n_points
        
        # Random restart
        noise = np.random.normal(0, 0.05, size=n_points)
        noise = np.clip(noise, -0.1, 0.1)
        random_h = current_h + noise
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))
        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_random.fun < best_c5:
            best_c5 = result_random.fun
            best_h = result_random.x
            best_n_points = n_points
        
        elapsed_time = time.time() - start_time
        if elapsed_time >= budget_s:
            break
    
    # Final validation and clipping
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, best_n_points
