import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initialize with a piecewise constant function
    h = np.zeros(n_points)
    for i in range(n_points):
        if i < n_points // 2:
            h[i] = 1.0
        else:
            h[i] = 0.0
    h = h * (total_integral / np.sum(h))

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # Use a particle swarm optimization with local refinement
    # Initialize multiple particles with structured perturbations
    num_particles = 5
    particles = []
    for _ in range(num_particles):
        perturbation = np.random.normal(0, 0.1, size=n_points)
        perturbation = np.clip(perturbation, -0.2, 0.2)
        new_h = best_h + perturbation
        new_h = np.clip(new_h, 0.0, 1.0)
        new_h = new_h * (total_integral / np.sum(new_h))
        particles.append(new_h)

    while time.time() - start_time < budget_s:
        # Evaluate all particles
        particle_values = []
        for p in particles:
            corr = np.correlate(p, 1 - p, mode="full")
            max_overlap = np.max(corr) * dx
            particle_values.append(max_overlap)

        # Find the best particle
        best_idx = np.argmin(particle_values)
        best_particle = particles[best_idx]
        best_c5 = particle_values[best_idx]

        # Update best_h
        if best_c5 < evaluate_erdos_solution(best_h):
            best_h = best_particle.copy()
            best_c5 = evaluate_erdos_solution(best_h)

        # Generate new particles from the best particle with perturbations
        for _ in range(num_particles):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbation = np.clip(perturbation, -0.1, 0.1)
            new_h = best_h + perturbation
            new_h = np.clip(new_h, 0.0, 1.0)
            new_h = new_h * (total_integral / np.sum(new_h))
            particles.append(new_h)

        # Keep only the top num_particles particles to avoid explosion
        if len(particles) > num_particles:
            particle_values = []
            for p in particles:
                corr = np.correlate(p, 1 - p, mode="full")
                max_overlap = np.max(corr) * dx
                particle_values.append(max_overlap)
            # Sort by value and keep top num_particles
            sorted_indices = np.argsort(particle_values)
            particles = [particles[i] for i in sorted_indices[:num_particles]]
            particle_values = [particle_values[i] for i in sorted_indices[:num_particles]]

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
