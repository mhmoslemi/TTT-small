def run(seed=42, budget_s=100, **kwargs):

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Function to calculate the overlap integral
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Control the number of trials within the budget

    # Use more sophisticated optimization strategies
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution
        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Use SLSQP for local refinement
        try:
            result_slsqp = minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # Use Nelder-Mead for global exploration
        try:
            result_nelder = minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="Nelder-Mead",
                options={"maxiter": 150}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x
        except:
            pass

    return best_h, best_c5, n_points
