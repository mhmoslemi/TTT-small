        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Run a single optimization with more sophisticated methods
    result_slsqp = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 300}
    )
    if result_slsqp.success:
        best_h = result_slsqp.x
        best_c5 = result_slsqp.fun

    # Run a global search with differential evolution
    result_de = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="differential_evolution",
        options={"tol": 1e-5, "popsize": 20, "mutation": (0.5, 1.0), "maxiter": 100}
    )
    if result_de.success:
        best_h = result_de.x
        best_c5 = result_de.fun

    # Run a local refinement with Nelder-Mead
    result_nelder = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="Nelder-Mead",
        options={"maxiter": 150}
    )
    if result_nelder.success:
        best_c5 = result_nelder.fun
        best_h = result_nelder.x

    return best_h, best_c5, n_points
