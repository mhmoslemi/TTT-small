        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)

    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    # Function to generate a random perturbation
    def generate_perturbation(h):
        perturbation = np.random.uniform(-0.1, 0.1, size=len(h))
        return np.clip(h + perturbation, 0, 1)

    # Function to perform a local optimization with a given initial guess
    def optimize_with_method(initial_guess, method):
        try:
            result = minimize(
                objective,
                initial_guess,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method=method,
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result.success:
                return result.fun, result.x
            else:
                return np.inf, initial_guess
        except:
            return np.inf, initial_guess

    # Main optimization loop
    for trial in range(trials):

            break

        h = generate_perturbation(best_h)

        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

        methods = ["SLSQP", "Nelder-Mead", "COBYLA"]

            current_c5, current_h = optimize_with_method(h, method)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

        # Adaptive cooling for perturbation magnitude
