        h = h_values.reshape(-1)

    def perturb(h, magnitude=0.15):
        perturbation = np.random.normal(0, magnitude, size=h.shape)
        h_perturbed = h + perturbation
        h_perturbed = np.clip(h_perturbed, 0, 1)
        return h_perturbed
    
    # Use a combination of strategies for global and local optimization
    # 1. Local refinement with SLSQP
    def local_refinement(h):
        result = minimize(
            evaluate_overlap,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        if result.success:
            return result.x
        return h
    
    # 2. Global exploration with random restarts and gradient-based methods

        h_refined = local_refinement(h)
