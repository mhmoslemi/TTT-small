import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try with different n_points to find the best balance between accuracy and speed
    n_points_list = [100, 200, 300]
    best_c5 = float('inf')
    best_h = np.zeros(100)
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2
        
        # Initialize with a structured candidate to avoid local minima
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 1.0 if i < n_points // 2 else 0.0
        h = h * (total_integral / np.sum(h))
        
        best_h_current = h.copy()
        best_c5_current = evaluate_erdos_solution(h)
        
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        start_time = time.time()
        
        # Use a more sophisticated optimization with adaptive perturbation and more efficient sampling
        while time.time() - start_time < budget_s:
            # Run SLSQP optimization
            result = opt.minimize(
                fun=objective_function,
                x0=best_h_current,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result.fun < best_c5_current:
                best_c5_current = result.fun
                best_h_current = result.x
            
            # Generate diverse perturbations with structured noise
            perturbations = []
            for _ in range(3):
                perturbation = np.random.normal(0, 0.05, size=n_points)
                perturbation = np.clip(perturbation, -0.1, 0.1)
                perturbations.append(perturbation)
            
            for pert in perturbations:
                perturbed_h = best_h_current + pert
                perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
                perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
                result_pert = opt.minimize(
                    fun=objective_function,
                    x0=perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if result_pert.fun < best_c5_current:
                    best_c5_current = result_pert.fun
                    best_h_current = result_pert.x
            
            # Random restart with structured noise
            noise = np.random.normal(0, 0.05, size=n_points)
            noise = np.clip(noise, -0.1, 0.1)
            random_h = best_h_current + noise
            random_h = np.clip(random_h, 0.0, 1.0)
            random_h = random_h * (total_integral / np.sum(random_h))
            result_random = opt.minimize(
                fun=objective_function,
                x0=random_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result_random.fun < best_c5_current:
                best_c5_current = result_random.fun
                best_h_current = result_random.x
        
        # Final validation
        best_h_current = np.clip(best_h_current, 0.0, 1.0)
        best_h_current = best_h_current * (total_integral / np.sum(best_h_current))
        final_c5 = evaluate_erdos_solution(best_h_current)
        
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = best_h_current
    
    return best_h, best_c5, n_points_list[0]  # Returning n_points of the first iteration for consistency
