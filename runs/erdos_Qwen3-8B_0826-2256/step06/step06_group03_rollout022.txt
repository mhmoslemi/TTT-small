import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with initial values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    # Define objective function
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define constraint function
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # More effective initialization: start with a structured initial guess
    initial_guess = np.zeros(n_points)
    for i in range(n_points):
        initial_guess[i] = 1.0 if i < n_points // 2 else 0.0
    initial_guess = initial_guess * (total_integral / np.sum(initial_guess))
    h = np.copy(initial_guess)

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    # Use a combination of optimization methods for better convergence
    # and introduce structured perturbations to avoid local minima
    while time.time() - start_time < budget_s:
        # Local optimization with adaptive perturbation
        # Introduce some structured perturbation to avoid local minima
        perturbation = np.zeros_like(h)
        for i in range(n_points):
            perturbation[i] = np.random.normal(0, 0.05)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        perturbed_h = h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))

        result = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )

        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Try a different perturbation strategy
        perturbation = np.random.normal(0, 0.02, size=n_points)
        perturbation = np.clip(perturbation, -0.05, 0.05)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))

        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )

        if result_pert.fun < best_c5:
            best_c5 = result_pert.fun
            best_h = result_pert.x

        # Random restart with structured noise
        noise = np.random.normal(0, 0.05, size=n_points)
        noise = np.clip(noise, -0.1, 0.1)
        random_h = best_h + noise
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))

        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )

        if result_random.fun < best_c5:
            best_c5 = result_random.fun
            best_h = result_random.x

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
