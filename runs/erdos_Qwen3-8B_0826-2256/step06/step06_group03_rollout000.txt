```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with initial values
    h = np.copy(initial_h_values)
    h = np.clip(h, 0.0, 1.0)
    h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    # Define the objective function for minimization
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Define the constraint function for the integral
    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # Initialize with a structured pattern: alternating blocks
    structured_h = np.zeros(n_points)
    for i in range(n_points):
        structured_h[i] = 1.0 if i % 5 < 2 else 0.0
    structured_h = structured_h * (total_integral / np.sum(structured_h))
    h = np.copy(structured_h)

    # Start with the best_h as the structured pattern
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()

    # Main optimization loop
    while time.time() - start_time < budget_s:
        # Run local optimization
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Apply noise to diversify the search
        noise = np.random.normal(0, 0.05, size=n_points)
        noise = np.clip(noise, -0.1, 0.1)
        perturbed_h = best_h + noise
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_pert.fun < best_c5:
            best_c5 = result_pert.fun
            best_h = result_pert.x

        # Random restart with structured noise
        structured_noise = np.random.choice([-0.1, 0.0, 0.1], size=n_points)
        structured_noise = np.clip(structured_noise, -0.1, 0.1)
        random_h = best_h + structured_noise
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))
        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_random.fun < best_c5:
            best_c5 = result_random.fun
            best_h = result_random.x

    # Final validation and normalization
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
```
