import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 200 points for better resolution while keeping computation manageable
    n_points = 200
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Initialize with a simple alternating pattern
        h = np.zeros(n_points)
        h[::2] = 0.5
        h[1::2] = 0.5
        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Define the overlap integral
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    # Define the constraint function
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Initialize DEAP for genetic algorithm
    creator.create("FitnessMin", base.Fitness, weights=(1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
    
    def eval_func(individual):
        individual = np.clip(individual, 0, 1)
        fitness = evaluate_overlap(individual)
        return (fitness,)
    
    def mut_gaussian(individual, indpb=0.05):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] += np.random.normal(0, 0.1)
                individual[i] = np.clip(individual[i], 0, 1)
        return individual,
    
    def cx_uniform(ind1, ind2):
        ind1 = np.clip(ind1, 0, 1)
        ind2 = np.clip(ind2, 0, 1)
        size = len(ind1)
        ind = np.empty(size)
        for i in range(size):
            if np.random.random() < 0.5:
                ind[i] = ind1[i]
            else:
                ind[i] = ind2[i]
        return ind, ind
    
    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", eval_func)
    toolbox.register("mate", cx_uniform)
    toolbox.register("mutate", mut_gaussian)
    toolbox.register("select", tools.selTournament, tournsize=3)
    
    # Initialize best solution
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Run genetic algorithm
    pop = toolbox.population(n=50)
    for gen in range(100):
        if time.time() - start_time > budget_s - 1:
            break
        # Evaluate all individuals in the population
        fitnesses = toolbox.map(toolbox.evaluate, pop)
        for ind, fit in zip(pop, fitnesses):
            ind.fitness.values = fit
        
        # Select the best individuals
        offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
        
        # Evaluate the offspring
        offspring = toolbox.map(toolbox.evaluate, offspring)
        
        # Replace the population
        pop = toolbox.select(pop + offspring, len(pop))
        
        # Update best solution
        for ind in pop:
            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.copy(ind)
    
    # Refine the best solution with local optimization
    def objective_function(h_values):
        return evaluate_overlap(h_values)
    
    # Add constraint to the objective function
    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Use SLSQP for local refinement
    try:
        result = minimize(
            objective_function,
            best_h,
            bounds=Bounds(0, 1),
            constraints={"type": "eq", "fun": constraint_func},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result.success:
            best_h = result.x
            best_c5 = result.fun
    except:
        pass
    
    return best_h, best_c5, n_points
