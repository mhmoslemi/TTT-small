import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for a balance between speed and quality
    dx = 2.0 / n_points

    # Initialize with the provided initial construction
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Use differential evolution for global optimization
    def differential_evolution_run(x0):
        result = differential_evolution(
            objective,
            bounds,
            strategy='best1bin',
            popsize=15,
            maxiter=100,
            tol=1e-8,
            constraints=[{"type": "eq", "fun": constraint}],
            seed=seed,
            disp=False
        )
        if result.success:
            return result.x, result.fun
        return x0, objective(x0)

    # Run differential evolution with multiple initializations
    for _ in range(5):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        # Run differential evolution
        h_new, c5_new = differential_evolution_run(h)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new
    
    return best_h, best_c5, n_points
