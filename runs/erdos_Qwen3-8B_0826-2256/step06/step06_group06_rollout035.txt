    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)
    
    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2
    
    # Function to compute the correlation in an efficient way
    def fast_correlation(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Helper function to optimize h for given n_points and initial guess
    def optimize_h(n_points, initial_h, tolerance=1e-4, max_iter=100):
        bounds = Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, tol=tolerance, options={'maxiter': max_iter})
        return result.x, evaluate_c5(result.x, n_points)
    
    # Function to compute the correlation over all possible shifts
    def compute_all_correlations(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = correlate(h, 1 - h, mode="full")
        return cross_corr * dx
    
    # Function to perform simulated annealing with better exploration
    def simulated_annealing(h_initial, n_points, temp_start=1.0, temp_end=0.1, steps=1000, tolerance=1e-4):
        h = h_initial.copy()
        current_c5 = evaluate_c5(h, n_points)
        best_h = h.copy()
        best_c5 = current_c5
        temp = temp_start
        
        for step in range(steps):
            if temp < temp_end:
                break
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = np.clip(h + perturbation, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            perturbed_c5 = evaluate_c5(h_perturbed, n_points)
            
            if perturbed_c5 < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = perturbed_c5
                temp *= 0.95

                energy_diff = current_c5 - perturbed_c5
                if np.random.rand() < np.exp(-energy_diff / temp):
                    h = h_perturbed.copy()
                    current_c5 = perturbed_c5
                temp *= 0.98
        
        return best_h, best_c5

            fitness = np.array([evaluate_c5(h, n_points) for h in h_pop])

            parents_idx = np.argsort(fitness)[:population_size // 2]

            offspring = np.zeros((population_size // 2, n_points))
            for i in range(population_size // 2):
                p1, p2 = np.random.choice(parents_idx, 2, replace=False)
                crossover_point = np.random.randint(1, n_points - 1)
                offspring[i, :] = np.concatenate([parents[p1, :crossover_point], parents[p2, crossover_point:]])

            for i in range(population_size // 2):
                if np.random.rand() < mutation_rate:
                    mutation = np.random.normal(0, 0.05, n_points)
                    offspring[i, :] = np.clip(offspring[i, :] + mutation, 0, 1)
                    offspring[i, :] = offspring[i, :] * (n_points / 2) / np.sum(offspring[i, :])
            
            # Combine population
            h_pop = np.vstack([parents, offspring])
            h_pop = np.clip(h_pop, 0, 1)
            h_pop = h_pop * (n_points / 2) / np.sum(h_pop, axis=1)[:, np.newaxis]
            
            # Update best
            for i in range(population_size):
                if fitness[i] < best_c5:
                    best_h = h_pop[i, :]
                    best_c5 = fitness[i]
        
        return best_h, best_c5
    
    # Function to perform gradient-based optimization with enhanced exploration
    def gradient_based_optimization(h_initial, n_points, tolerance=1e-4, max_iter=100):
        bounds = Bounds(0.0, 1.0)
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
