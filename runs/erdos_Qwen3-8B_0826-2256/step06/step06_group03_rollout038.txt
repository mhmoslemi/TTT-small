import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Slightly larger for better resolution while still efficient
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Start with a refined initial construction
    def construct_initial_h():
        # Construct a piecewise constant function with 4 segments
        segments = np.linspace(0, 2, 5)
        h = np.zeros(n_points)
        for i in range(n_points):
            x = segments[0] + i * dx
            if segments[0] <= x < segments[1]:
                h[i] = 1.0
            elif segments[1] <= x < segments[2]:
                h[i] = 0.5
            elif segments[2] <= x < segments[3]:
                h[i] = 0.25
            elif segments[3] <= x <= segments[4]:
                h[i] = 0.1
        return h * (total_integral / np.sum(h))

    h = construct_initial_h()
    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # Use a more sophisticated optimization with adaptive perturbation
    # Introduce a local search with stochastic gradient descent and simulated annealing logic
    for _ in range(100):  # Limit the number of iterations to stay within budget
        # Local optimization with stochastic gradient descent
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Introduce simulated annealing-like perturbations
        perturbation = np.random.normal(0, 0.05, size=n_points)
        perturbation = np.clip(perturbation, -0.1, 0.1)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        result_pert = opt.minimize(
            fun=objective_function,
            x0=perturbed_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_pert.fun < best_c5:
            best_c5 = result_pert.fun
            best_h = result_pert.x

        # Random restart with structured noise
        noise = np.random.normal(0, 0.05, size=n_points)
        noise = np.clip(noise, -0.1, 0.1)
        random_h = best_h + noise
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (total_integral / np.sum(random_h))
        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_random.fun < best_c5:
            best_c5 = result_random.fun
            best_h = result_random.x

        # Early stopping if time is up
        if time.time() - start_time >= budget_s:
            break

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
