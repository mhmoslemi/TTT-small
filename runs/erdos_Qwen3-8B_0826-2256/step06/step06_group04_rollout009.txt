import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for speed and quality balance
    n_points = 100
    dx = 2.0 / n_points
    
    # Initial guess - use initial_h_values if provided, otherwise use a more informed initialization
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Use a frequency-based initialization: random intervals of 0.5
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.5
        h[n_points // 2:] = 0.5
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    # Objective function: minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        # Compute the full cross-correlation
        overlap = np.correlate(h, 1 - h, mode="full")
        # Find the maximum overlap value
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for each h[i]
    bounds = Bounds(0, 1)
    
    # Best solution found so far
    best_h = h
    best_c5 = objective(h)
    best_n_points = n_points
    
    # Early exit if budget is exhausted
    start_time = time.time()
    while time.time() - start_time < budget_s:
        # Try different optimization methods with different initializations
        # 1. Differential evolution: global optimization
        try:
            # Define bounds and constraints
            def de_objective(h_values):
                return objective(h_values)
            def de_constraint(h_values):
                return constraint(h_values)
            
            # Set bounds and constraints for differential evolution
            de_bounds = [(0, 1) for _ in range(n_points)]
            de_result = differential_evolution(
                de_objective,
                bounds=de_bounds,
                constraints=[{"type": "eq", "fun": de_constraint}],
                strategy="best1bin",
                maxiter=50,
                popsize=15,
                tol=1e-6,
                mutation=(0.5, 1),
                recombination=0.8,
                polish=True,
                disp=False
            )

                current_h = de_result.x
                current_c5 = de_result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = current_h
                    best_n_points = n_points
        except:
            pass
        
        # 2. Local refinement with SLSQP
        try:
            result_slsqp = minimize(
                objective,
                best_h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # 3. Try increasing n_points for better resolution
        if n_points < 200 and time.time() - start_time < budget_s - 5:
            n_points += 10
            dx = 2.0 / n_points
            # Resample current solution to the new n_points
            new_h = np.zeros(n_points)

            new_h = new_h / (np.sum(new_h) * dx) if np.sum(new_h) != 0 else np.random.rand(n_points)
            new_h = np.clip(new_h, 0, 1)
            best_h = new_h
            best_n_points = n_points
            # Recompute best_c5
            best_c5 = objective(best_h)
    
    return best_h, best_c5, best_n_points
