import numpy as np
import scipy.optimize as optimize
import time
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import minres

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Use the initial_h_values if provided
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Start with a structured initial guess
        n_points = 100
        h = np.zeros(n_points)
        h[:n_points//3] = 0.8
        h[n_points//3:2*n_points//3] = 0.2
        h[2*n_points//3:] = 0.8
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / n_points)
        return c5
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # More trials for better exploration
    perturbation_factor = 0.3  # Initial perturbation factor
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Create a new perturbed solution
        h = best_h + np.random.uniform(-perturbation_factor, perturbation_factor, size=n_points)
        h = np.clip(h, 0, 1)
        # Renormalize to satisfy the integral constraint
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Use a combination of methods for better convergence
        try:
            # Use SLSQP for refinement
            result = minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-9, "maxiter": 100}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Reduce perturbation size over time
        perturbation_factor *= 0.95
    
    return best_h, best_c5, n_points
