import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # High-resolution but manageable
    dx = 2.0 / n_points

    # Initialize with a structured approach
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # More trials for exploration
    cooldown = 0.8  # Reduce perturbation magnitude over time
    perturbation_factor = 0.2

    # Perform multiple optimization trials with different initializations
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-perturbation_factor * (1 - trial / trials), perturbation_factor * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Use a variety of optimization methods for robustness
        try:
            # Run optimization with SLSQP for local refinement
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x

            # Run optimization with Nelder-Mead for global exploration
            result_nelder = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="Nelder-Mead",
                options={"maxiter": 150}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x

            result_slsqp2 = minimize(
                objective,

                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp2.success:
                current_c5 = result_slsqp2.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp2.x

        except:
            # Handle exceptions gracefully
            pass

    return best_h, best_c5, n_points
