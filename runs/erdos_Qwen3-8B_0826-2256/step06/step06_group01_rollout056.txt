import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.sparse import csr_matrix

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Initialize with the provided initial_h_values
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to a baseline that shapes the function
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Ensure the function is within [0, 1]
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 150  # Trials to iterate over
    cooldown = 10  # Reduce perturbation magnitude over time

    # Define objective for optimization
    def optimize_with_method(x0, method):
        try:
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method=method,
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result.success:
                return result.x, result.fun
        except:
            pass
        return None, float('inf')

    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.1 * (1 - trial / trials), 0.1 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Try different optimization methods in sequence
        methods = ["SLSQP", "COBYLA", "L-BFGS-B", "Nelder-Mead"]
        for method in methods:
            current_h, current_c5 = optimize_with_method(x0, method)
            if current_h is not None and current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h

    return best_h, best_c5, n_points
