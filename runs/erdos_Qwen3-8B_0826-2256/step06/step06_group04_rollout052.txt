import scipy.optimize as optimize
import time

        # Adjust to satisfy the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = optimize.Bounds(0, 1)
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Parameters for optimization
    trials = 100
    cooldown = 10
    
    # Use a more efficient optimization method with metrics tracking
    def optimize_with_method(h_initial, method, max_iter=100):
        result = optimize.minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method=method,
            options={"ftol": 1e-8, "maxiter": max_iter}
        )
        if result.success:
            return result.x, result.fun
        return h_initial, best_c5
    
    # Run a series of optimizations with varying methods and perturbations
    for trial in range(trials):

        perturbation = np.random.uniform(-0.1 * (1 - trial / trials), 0.1 * (1 - trial / trials), size=n_points)

        h_slsqp, c5_slsqp = optimize_with_method(x0, "SLSQP", max_iter=50)
