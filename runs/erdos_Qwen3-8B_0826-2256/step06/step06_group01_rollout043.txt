import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for balance between speed and quality
    dx = 2.0 / n_points
    
    # Initialize with a reasonable structure based on initial_h_values or random
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * n_points / 2
    
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = correlate(h, 1 - h, mode='full')
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial best solution
    best_h = h
    best_c5 = objective(h)
    
    start_time = time.time()
    
    # Use differential evolution for global optimization
    result = differential_evolution(
        objective,
        bounds,
        constraints={"type": "eq", "fun": constraint},
        strategy='best1bin',
        popsize=20,
        mutation=(0.5, 1),
        recombination=0.8,
        tol=1e-10,
        maxiter=50,
        polish=True,
        disp=False,
        seed=seed
    )
    
    # Update best solution if optimization was successful
    if result.success:
        best_h = result.x
        best_c5 = result.fun
    
    # Perform local refinement with SLSQP
    try:
        result_slsqp = minimize(
            objective,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result_slsqp.success:
            best_c5 = result_slsqp.fun
            best_h = result_slsqp.x
    except:
        pass
    
    # Ensure the solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / (np.sum(best_h) * dx) * n_points / 2
    
    return best_h, best_c5, n_points
