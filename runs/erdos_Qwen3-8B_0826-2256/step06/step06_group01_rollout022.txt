def run(seed=42, budget_s=100, **kwargs):

        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Ensure the function is within [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function: minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to enforce the integral constraint
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 150
    tolerance = 1e-6

    # Multi-phase optimization: explore then refine
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.2 * (1 - trial / trials), 0.2 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        try:
            # Refinement with SLSQP
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": tolerance, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:

                    best_h = result_slsqp.x

            result_lbfgs = minimize(

                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="L-BFGS-B",
                options={"ftol": tolerance, "maxiter": 150}
            )
            if result_lbfgs.success:
                current_c5 = result_lbfgs.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
