import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for a balance between speed and quality
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with a structured approach using the initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points // 2] = 0.5
        h[n_points // 2:] = 0.5
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap using correlate
        overlap = correlate(h, 1 - h, mode="full")
        # Normalize by dx to get the integral over the interval
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Use a combination of parameter sweeping and random restarts
    num_restarts = 10
    num_hyperparameters = 50
    hyperparameter_values = np.linspace(0.1, 0.9, num_hyperparameters)
    
    for restart in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Randomly adjust the initial h to avoid local optima
        h = best_h + np.random.uniform(-0.1, 0.1, size=n_points)
        h = np.clip(h, 0, 1)
        # Normalize to satisfy the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Try multiple hyperparameters for the perturbation magnitude
        for perturbation in hyperparameter_values:
            # Perturb the solution
            h_perturbed = h + perturbation * np.random.randn(n_points)
            h_perturbed = np.clip(h_perturbed, 0, 1)
            # Normalize the perturbed solution
            h_perturbed = h_perturbed / (np.sum(h_perturbed) * dx) if np.sum(h_perturbed) != 0 else h_perturbed
            
            # Run optimization using SLSQP
            result = minimize(
                objective,
                h_perturbed,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
    
    return best_h, best_c5, n_points
