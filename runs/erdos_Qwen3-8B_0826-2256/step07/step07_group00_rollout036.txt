    def optimize_with_slsqp(h_initial):
        result = minimize(
            objective,
            h_initial,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 150}
        )
        if result.success:
            return result.x, result.fun
        return h_initial, float('inf')

    for trial in range(trials):
