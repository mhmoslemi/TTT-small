import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for efficiency
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with the provided initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to a regular pattern: half ones, half zeros
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Ensure integral constraint is met
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 300  # Number of trials for optimization
    min_perturbation = 0.01  # Minimum perturbation magnitude
    
    # Define a function to generate random perturbations
    def random_perturbation(h, magnitude, seed=None):
        np.random.seed(seed)
        perturbation = np.random.uniform(-magnitude, magnitude, size=len(h))
        return np.clip(h + perturbation, 0, 1)
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Randomly perturb the current best solution
        perturbation_magnitude = 0.05 * (1 - trial / trials)
        h = random_perturbation(best_h, perturbation_magnitude, seed=trial)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Run optimization with SLSQP for local refinement
        result_slsqp = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
