def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = 100

    else:

        h[:mid] = 0.5
        h[mid:] = 0.5
        # Normalize to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 150  # Number of trials for exploration
    perturbation_magnitude = 0.15
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current solution
        perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize to meet the integral constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Run SLSQP for local refinement
        try:
            result_slsqp = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # Run Nelder-Mead for global exploration
        try:
            result_nelder = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="Nelder-Mead",
                options={"maxiter": 150}
            )

                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5

        except:
            pass
    
    return best_h, best_c5, n_points
