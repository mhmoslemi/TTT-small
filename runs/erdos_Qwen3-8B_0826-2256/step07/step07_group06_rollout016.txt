import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Starting with 100 points for balance between speed and quality
    dx = 2.0 / n_points

    # Use the initial_h_values for a better starting point
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback initialization: equal distribution
        h = np.full(n_points, 0.5)

    h = np.clip(h, 0, 1)
    # Ensure the integral is 1
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)

    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 150  # Number of trials
    max_iter_per_trial = 100  # Maximum iterations per trial
    cooling_rate = 0.995  # Cooling rate for perturbation magnitude

    # Execute the optimization
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.1 * (1 - trial / trials), 0.1 * (1 - trial / trials), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)

        # Run optimization with SLSQP for local refinement
        try:
            result_slsqp = minimize(
                objective,
                x0=h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": max_iter_per_trial}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass

        # Run optimization with Nelder-Mead for global exploration
        try:
            result_nelder = minimize(
                objective,
                x0=h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="Nelder-Mead",
                options={"maxiter": max_iter_per_trial}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x
        except:
            pass

        # Run optimization with COBYLA for robustness
        try:
            result_cobyla = minimize(
                objective,
                x0=h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": max_iter_per_trial}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass

    return best_h, best_c5, n_points
