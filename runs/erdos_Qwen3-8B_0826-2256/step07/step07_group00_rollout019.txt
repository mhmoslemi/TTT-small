import scipy.optimize as optimize

        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Constraint to ensure the integral is 1
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Multi-start and adaptive cooling strategy
    n_trials = 50  # Number of initial trials
    cooling_rate = 0.995  # Rate at which perturbation magnitude decreases
    perturbation_max = 0.4  # Initial maximum perturbation size
    
    for trial in range(n_trials):
        if time.time() - start_time > budget_s - 1:
            break

        perturbation = np.random.uniform(-perturbation_max, perturbation_max, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)

        result = optimize.minimize(

            h,

                best_c5 = current_c5
                best_h = result.x
        
        # Reduce perturbation magnitude for next trial
