import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # More samples for better resolution, still efficient for optimization

    # Initialize with a structured candidate
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 if i < n_points // 2 else 0.0
    h = h * (n_points / 2.0 / np.sum(h))

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * (2.0 / n_points)
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - n_points / 2.0

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    start_time = time.time()

    # Adaptive optimization with exploration and exploitation
    while time.time() - start_time < budget_s:
        # Run SLSQP optimization
        result = opt.minimize(
            fun=objective_function,
            x0=best_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result.fun < best_c5:
            best_c5 = result.fun
            best_h = result.x

        # Generate diverse perturbations
        perturbations = []
        for _ in range(5):
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbation = np.clip(perturbation, -0.1, 0.1)
            perturbations.append(perturbation)

        for pert in perturbations:
            perturbed_h = best_h + pert
            perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
            perturbed_h = perturbed_h * (n_points / 2.0 / np.sum(perturbed_h))
            result_pert = opt.minimize(
                fun=objective_function,
                x0=perturbed_h,
                method='SLSQP',
                bounds=bounds,
                constraints=cons,
                options={'ftol': 1e-9, 'maxiter': 100}
            )
            if result_pert.fun < best_c5:
                best_c5 = result_pert.fun
                best_h = result_pert.x

        # Random restart with structured noise
        noise = np.random.normal(0, 0.05, size=n_points)
        noise = np.clip(noise, -0.1, 0.1)
        random_h = best_h + noise
        random_h = np.clip(random_h, 0.0, 1.0)
        random_h = random_h * (n_points / 2.0 / np.sum(random_h))
        result_random = opt.minimize(
            fun=objective_function,
            x0=random_h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        if result_random.fun < best_c5:
            best_c5 = result_random.fun
            best_h = result_random.x

        # Reduce perturbation size for fine-tuning
        if time.time() - start_time > budget_s * 0.8:
            perturbations = []
            for _ in range(5):
                perturbation = np.random.normal(0, 0.02, size=n_points)
                perturbation = np.clip(perturbation, -0.05, 0.05)
                perturbations.append(perturbation)

            for pert in perturbations:
                perturbed_h = best_h + pert
                perturbed_h = np.clip(perturbed_h, 0.0, 1.0)
                perturbed_h = perturbed_h * (n_points / 2.0 / np.sum(perturbed_h))
                result_pert = opt.minimize(
                    fun=objective_function,
                    x0=perturbed_h,
                    method='SLSQP',
                    bounds=bounds,
                    constraints=cons,
                    options={'ftol': 1e-9, 'maxiter': 100}
                )
                if result_pert.fun < best_c5:
                    best_c5 = result_pert.fun
                    best_h = result_pert.x

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (n_points / 2.0 / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
