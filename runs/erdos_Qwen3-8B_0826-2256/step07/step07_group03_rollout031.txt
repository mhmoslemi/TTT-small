        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)

        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a combination of methods to explore the solution space effectively
    # Start with differential evolution for global search
    def objective_function(h_values):
        return evaluate_overlap(h_values)
    
    # Define bounds for differential evolution
    bounds_de = [(0, 1) for _ in range(n_points)]
    
    # Run differential evolution
    try:
        result_de = differential_evolution(objective_function, bounds_de, 
                                          strategy='best2bin', 
                                          popsize=15, 
                                          tol=1e-5, 
                                          mutation=(0.5, 1), 
                                          recombination=0.8, 
                                          maxiter=100,

            current_c5 = result_de.fun

                best_h = result_de.x

        pass

    try:

            current_c5 = result_slsqp.fun

    except:
        pass

    if time.time() - start_time > budget_s - 1:
