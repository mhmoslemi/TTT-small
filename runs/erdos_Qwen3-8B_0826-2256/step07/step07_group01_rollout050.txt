def run(seed=42, budget_s=100, **kwargs):

    dx = 2.0 / n_points

    else:

        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        # Ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    
    # Clip to [0, 1] to ensure constraints are met
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    # Bounds for all points
    bounds = [(0, 1) for _ in range(n_points)]

    best_h = h
    best_c5 = objective(h)

    start_time = time.time()
    trials = 50  # Use differential evolution for global optimization
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        # Use differential evolution for global exploration
        result_de = differential_evolution(
            objective,
            bounds,
            constraints={"type": "eq", "fun": constraint},
            strategy='best1bin',
            maxiter=100,
            popsize=15,
            mutation=(0.5, 1),
            recombination=0.9,
            tol=1e-6,
            seed=seed + trial
        )
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x

    # Final refinement using SLSQP for local optimization
    result_slsqp = minimize(
        objective,
        best_h,
        bounds=bounds,
        constraints={"type": "eq", "fun": constraint},
        method="SLSQP",
        options={"ftol": 1e-9, "maxiter": 150}
    )
    if result_slsqp.success:
        best_c5 = result_slsqp.fun
        best_h = result_slsqp.x

    return best_h, best_c5, n_points
