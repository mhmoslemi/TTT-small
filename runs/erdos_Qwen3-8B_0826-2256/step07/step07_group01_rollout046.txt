import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy import sparse

from scipy.optimize import LinearConstraint

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Start with 100 points for efficiency and quality balance
    
    # Initialize h using the provided initial construction or a random pattern
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 0.5
        h[n_points//2:] = 0.5
        h = h / (np.sum(h) * 2.0 / n_points) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    best_h = h
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    start_time = time.time()
    
    # Repeatable, adaptive optimization steps for better exploration
    for _ in range(50):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the best solution with adaptive magnitude
        perturbation = np.random.uniform(-0.1 * (1 - _ / 50), 0.1 * (1 - _ / 50), size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Normalize integral to ensure the constraint
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h_vec = h.copy()
        
        # Objective function to minimize the maximum overlap
        def objective(h_values):
            h = h_values.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        # Constraint: integral of h(x) over [0, 2] must be 1
        def constraint(h_values):
            return np.sum(h_values) * dx - 1
        
        # Define bounds and constraints
        bounds = Bounds(0, 1)
        constraint_func = LinearConstraint(np.ones(n_points), [1], [1], keep_feasible=True)
        
        # Use a combination of global and local optimization methods
        try:
            # Local refinement with SLSQP method
            result_slsqp = minimize(
                objective,
                h_vec,
                method="SLSQP",
                bounds=bounds,
                constraints=constraint_func,
                options={"ftol": 1e-9, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        
            # Global exploration with COBYLA method
            result_cobyla = minimize(
                objective,
                h_vec,
                method="COBYLA",
                bounds=bounds,
                constraints=constraint_func,
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        
        except:
            # Gracefully handle exceptions
            pass
    
    return best_h, best_c5, n_points
