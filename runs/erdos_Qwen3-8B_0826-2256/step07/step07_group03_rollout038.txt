```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 150 points for better resolution while keeping computation manageable
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize with a pattern that has periodic structure
    h = np.zeros(n_points)
    h[::3] = 1.0
    h[1::3] = 0.5
    h[2::3] = 0.0
    
    # Adjust to satisfy the integral constraint
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a more extensive optimization with diverse methods
    trials = 80
    cooldown = 10
    perturbation_scale = 0.2
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb current best with a decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale * (1 - trial / trials), 
                                        perturbation_scale * (1 - trial / trials), 
                                        size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        x0 = h
        
        # Perform constrained optimization using SLSQP
        try:
            result = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result.x
        except:
            pass
        
        # Perform a more global search using differential evolution
        try:
            result_de = optimize.differential_evolution(
                evaluate_overlap,
                bounds=np.array([[0, 1] for _ in range(n_points)]),
                constraints={"type": "eq", "fun": constraint},
                maxiter=50,
                popsize=15,
                mutation=(0.5, 1),
                recombination="exponential",
                disp=False
            )
            if result_de.success:
                current_c5 = result_de.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_de.x
        except:
            pass
    
    return best_h, best_c5, n_points
```
