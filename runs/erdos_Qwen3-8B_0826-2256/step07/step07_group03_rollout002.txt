        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)

    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a more efficient approach with simulated annealing and local optimization
    # Start with a coarser grid to initialize
    n_points_initial = 50
    dx_initial = 2.0 / n_points_initial
    h_initial = np.zeros(n_points_initial)
    h_initial[::2] = 0.5
    h_initial[1::2] = 0.5
    h_initial = h_initial / (np.sum(h_initial) * dx_initial) if np.sum(h_initial) != 0 else np.random.rand(n_points_initial)

    # Use simulated annealing for global search
    def simulated_annealing(h_initial, T=1.0, cooling_rate=0.995, max_iter=1000):
        h = h_initial.copy()
        current_c5 = evaluate_overlap(h)
        best_h = h.copy()
        best_c5 = current_c5

            if time.time() - start_time > budget_s - 1:

            perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
            h_perturbed = h + perturbation

            delta = c5_perturbed - current_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / T):
                h = h_perturbed
                current_c5 = c5_perturbed
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
            T *= cooling_rate
        return best_h, best_c5

        else:
            return h_initial, best_c5
