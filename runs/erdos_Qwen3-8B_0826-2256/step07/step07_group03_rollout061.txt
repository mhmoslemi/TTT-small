import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a larger number of points to allow for more detailed control
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with a simple alternating pattern
    h = np.zeros(n_points)
    h[::2] = 0.5
    h[1::2] = 0.5
    # Adjust to satisfy the integral constraint
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Try multiple optimization strategies in parallel
    strategies = [
        ('SLSQP', minimize, {"ftol": 1e-8, "maxiter": 150}),
        ('COBYLA', minimize, {"maxiter": 150}),
        ('Nelder-Mead', minimize, {"maxiter": 150}),
        ('Differential Evolution', differential_evolution, {"tol": 1e-5, "popsize": 15, "mutation": 0.8, "recombination": 0.8})
    ]
    
    # Use a different starting point to avoid local optima
    for strategy_name, optimizer, options in strategies:
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb with decreasing magnitude
        perturbation = np.random.uniform(-0.15, 0.15, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        x0 = h
        
        # If using differential evolution, we need to pass bounds and constraint separately
        if strategy_name == 'Differential Evolution':
            # Define the objective function separately for DE
            def objective_func(h_values):
                return evaluate_overlap(h_values)
            # Define the constraint for DE
            def constraint_func(h_values):
                return constraint(h_values)
            
            # Create bounds for differential evolution
            de_bounds = [(0, 1) for _ in range(n_points)]
            
            # Run differential evolution
            try:
                result = optimizer(
                    objective_func,
                    bounds=de_bounds,
                    constraints=[{"type": "eq", "fun": constraint_func}],
                    options=options
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
        else:
            # For other strategies, use the minimize function directly
            try:
                result = optimizer(
                    evaluate_overlap,
                    x0,
                    bounds=bounds,
                    constraints={"type": "eq", "fun": constraint},
                    options=options
                )
                if result.success:
                    current_c5 = result.fun
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h = result.x
            except:
                pass
    
    return best_h, best_c5, n_points
