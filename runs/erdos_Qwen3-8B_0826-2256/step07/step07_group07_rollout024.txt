import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a moderate number of points for efficiency
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with the provided initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to a regular pattern: half ones, half zeros
        h = np.zeros(n_points)
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Ensure the integral constraint is met
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 150  # Number of trials for optimization
    min_perturbation = 0.01  # Minimum perturbation magnitude
    use_slsqp = True
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Generate a new random perturbation
        perturbation = np.random.uniform(-0.1, 0.1, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Run optimization with SLSQP for local refinement
        if use_slsqp:
            result_slsqp = optimize.minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        else:
            # Run optimization with COBYLA for a different approach
            result_cobyla = optimize.minimize(
                objective,
                h,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
    
    return best_h, best_c5, n_points
