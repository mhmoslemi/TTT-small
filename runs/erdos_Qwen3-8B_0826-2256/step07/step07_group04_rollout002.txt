import numpy as np

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Efficient and suitable for optimization
    dx = 2.0 / n_points
    total_integral = n_points / 2  # sum(h) == n_points / 2

    # Initialize with a structured candidate
    h = np.zeros(n_points)
    for i in range(n_points):
        h[i] = 1.0 if i < n_points // 2 else 0.0
    h = h * (total_integral / np.sum(h))

    best_h = h
    best_c5 = evaluate_erdos_solution(h)

    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    def constraint_func(h_values):
        return np.sum(h_values) - total_integral

    bounds = [(0.0, 1.0) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': constraint_func}

    def evolve_population(population, budget, n_points):
        population_size = len(population)
        new_population = []
        for _ in range(population_size):
            # Random perturbation
            perturbation = np.random.normal(0, 0.05, size=n_points)
            perturbation = np.clip(perturbation, -0.1, 0.1)
            candidate = population[np.random.randint(population_size)] + perturbation
            candidate = np.clip(candidate, 0.0, 1.0)
            candidate = candidate * (total_integral / np.sum(candidate))
            new_population.append(candidate)
        return new_population

    start_time = time.time()
    population_size = 10
    population = [best_h.copy() for _ in range(population_size)]

    while time.time() - start_time < budget_s:
        # Evaluate the population
        evaluated = [(h, evaluate_erdos_solution(h)) for h in population]
        # Select the best individual
        best_c5 = min([c for _, c in evaluated])
        best_h = [h for h, c in evaluated if c == best_c5][0]

        # Evolve population
        population = evolve_population(population, budget_s - (time.time() - start_time), n_points)

    # Final validation
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = best_h * (total_integral / np.sum(best_h))
    final_c5 = evaluate_erdos_solution(best_h)

    return best_h, final_c5, n_points
