import numpy as np
import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        h = np.zeros(100)
        h[:50] = 1.0
        h[50:] = 0.0
        # Normalize to satisfy the integral constraint
        h = h / (np.sum(h) * 2.0 / 100) if np.sum(h) != 0 else np.random.rand(100)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * (2.0 / len(h))
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - len(h_values) / 2
    
    bounds = [(0, 1) for _ in range(len(h))]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 100  # Number of trials for optimization
    perturbation_decay = 0.95  # Decay factor for perturbation magnitude
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb the current best solution with decreasing magnitude
        perturbation = np.random.uniform(-0.05 * (perturbation_decay ** trial), 
                                        0.05 * (perturbation_decay ** trial), 
                                        size=len(h))
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * 2.0 / len(h)) if np.sum(h) != 0 else np.random.rand(len(h))
        h = np.clip(h, 0, 1)
        
        # Run optimization with SLSQP for local refinement
        result_slsqp = optimize.minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150}
        )
        
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, len(best_h)
