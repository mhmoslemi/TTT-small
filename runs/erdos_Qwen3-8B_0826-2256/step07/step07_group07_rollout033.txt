import numpy as np
import scipy.optimize as optimize
import time
import copy

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to staggered pattern
        h = np.zeros(n_points)
        h[::2] = 1.0
        h = h / (np.sum(h) * dx)
        h = np.clip(h, 0, 1)
    
    def evaluate(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = evaluate(h)
    start_time = time.time()
    
    # Use a combination of global and local search
    for trial in range(200):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Construct a new candidate by random perturbation
        perturb = np.random.rand(n_points) * 0.2
        h_candidate = np.clip(h + perturb - 0.1, 0, 1)
        h_candidate = h_candidate / (np.sum(h_candidate) * dx)
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Simple evaluation
        current_c5 = evaluate(h_candidate)
        if current_c5 < best_c5:
            best_h = h_candidate
            best_c5 = current_c5
        
        # Local refinement with SLSQP
        result_slsqp = optimize.minimize(
            evaluate,
            best_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
