```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with 150 points for better resolution while keeping computation manageable
    n_points = 150
    dx = 2.0 / n_points
    
    # Initialize with a pattern that has periodic structure
    h = np.zeros(n_points)
    h[::3] = 1.0
    h[1::3] = 0.5
    h[2::3] = 0.0
    
    # Adjust to satisfy the integral constraint
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    def evaluate_overlap(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        return np.max(overlap) * dx
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = evaluate_overlap(h)
    start_time = time.time()
    
    # Use a more extensive optimization with diverse strategies
    trials = 80
    cooldown = 10
    perturbation_scale = 0.2
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb current best with a decreasing magnitude
        perturbation = np.random.uniform(-perturbation_scale * (1 - trial / trials), 
                                        perturbation_scale * (1 - trial / trials), 
                                        size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        x0 = h
        
        # Use a combination of optimization methods for robustness and diversity
        # 1. SLSQP with tight tolerances for precision
        try:
            result_slsqp = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 150}
            )
            if result_slsqp.success:
                current_c5 = result_slsqp.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_slsqp.x
        except:
            pass
        
        # 2. COBYLA for robust exploration of the search space
        try:
            result_cobyla = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="COBYLA",
                options={"maxiter": 150}
            )
            if result_cobyla.success:
                current_c5 = result_cobyla.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_cobyla.x
        except:
            pass
        
        # 3. Nelder-Mead for global search with fewer constraints
        try:
            result_nelder = minimize(
                evaluate_overlap,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraint},
                method="Nelder-Mead",
                options={"maxiter": 150}
            )
            if result_nelder.success:
                current_c5 = result_nelder.fun
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = result_nelder.x
        except:
            pass
    
    return best_h, best_c5, n_points
```
