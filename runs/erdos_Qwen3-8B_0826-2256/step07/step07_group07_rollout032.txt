import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with the initial construction if available
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
    else:
        # Fallback to a regular pattern for initialization
        h = np.zeros(100)
        h[:50] = 1.0
        h = h / (np.sum(h) * 2.0 / 100)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    n_points = len(h)
    dx = 2.0 / n_points
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Constraint to ensure the integral is 1
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Parameters for the optimization process
    trials = 100
    perturbation_factors = np.linspace(0.5, 0.01, trials)
    
    for trial in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        perturbation = np.random.uniform(-perturbation_factors[trial], 
                                        perturbation_factors[trial], 
                                        size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        
        # Ensure the integral constraint is satisfied
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Use SLSQP for local refinement
        result_slsqp = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 150, "disp": False}
        )
        
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x
    
    return best_h, best_c5, n_points
