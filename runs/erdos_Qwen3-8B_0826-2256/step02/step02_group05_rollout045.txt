```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define a range of n_points values to try
    n_points_list = [80, 100, 120, 140, 160]
    
    best_h = None
    best_c5 = float('inf')
    
    # Try each n_points value
    for n_points in n_points_list:
        dx = 2.0 / n_points
        total_integral = n_points / 2  # sum(h) == n_points / 2
        
        # Start with initial_h_values, then reshape to n_points
        h = np.copy(initial_h_values[:n_points])
        h = np.clip(h, 0.0, 1.0)
        h = h * (total_integral / np.sum(h))  # Normalize to satisfy constraint
        
        # Define the objective function: minimize the maximum overlap integral
        def objective_function(h_values):
            corr = np.correlate(h_values, 1 - h_values, mode="full")
            max_overlap = np.max(corr) * dx
            return max_overlap
        
        # Define the constraint: sum(h) == n_points / 2
        def constraint_func(h_values):
            return np.sum(h_values) - total_integral
        
        # Optimization setup for SLSQP
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_func}
        
        # Track time
        start_time = time.time()
        result = opt.minimize(
            fun=objective_function,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 500, 'disp': False}
        )
        elapsed_time = time.time() - start_time
        
        # Check if we are within the budget
        if elapsed_time >= budget_s:
            break
        
        # Extract the best solution
        best_h_current = result.x
        best_h_current = np.clip(best_h_current, 0.0, 1.0)
        best_h_current = best_h_current * (total_integral / np.sum(best_h_current))  # Normalize again to ensure constraint
        
        # Evaluate the final solution
        final_c5 = evaluate_erdos_solution(best_h_current)
        
        # Update best solution
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = best_h_current
    
    return best_h, best_c5, len(best_h)
```
