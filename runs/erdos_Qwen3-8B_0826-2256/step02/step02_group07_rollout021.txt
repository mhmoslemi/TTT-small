def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Define a function to evaluate the C5 value for a given h array
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / len(h)))
    
    # Define a function to optimize h using SLSQP for a given number of points
    def optimize_h(n_points, h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(x):
            return evaluate_c5(x)
        
        def constraint_func(x):
            h = np.clip(x, 0, 1)
            return np.sum(h) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-7)
        
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, evaluate_c5(h)
    
    # Start with the initial construction and a small number of points
    n_points = 150
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    start_time = time.time()
    
    # First optimization with small n_points
    try:
        h, c5 = optimize_h(n_points, h_initial)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    except:
        pass
    
    # If time allows, try increasing n_points for better results
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        try:
            h, c5 = optimize_h(n_points, h_initial)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
        except:
            pass
    
    return best_h, best_c5, n_points
