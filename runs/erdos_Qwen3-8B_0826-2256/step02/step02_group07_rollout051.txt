def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a medium number of points
    n_points = 150
    best_c5 = float('inf')
    best_h = np.zeros(n_points)
    
    # Function to evaluate the C5 value
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return float('inf')
        h = h * (n_points / 2) / total
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Function to enforce the constraint
    def constraint(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        return total - n_points / 2
    
    # Function to perform optimization with SLSQP
    def optimize_h(initial_h):
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function
        def objective_func(x):
            return evaluate_c5(x)
        
        # Constraint function
        def constraint_func(x):
            return constraint(x)
        
        # Set up the optimization problem
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, initial_h, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Ensure valid solution
        h = np.clip(result.x, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h, float('inf')
        h = h * (n_points / 2) / total
        return h, evaluate_c5(h)
    
    # Initial guess: start from the initial construction and perturb it
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Run initial optimization
    h, c5 = optimize_h(h_initial)
    best_c5 = min(best_c5, c5)
    best_h = h
    
    # Try increasing n_points for better results
    if time.time() < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define new evaluate and constraint functions
        def evaluate_c5_new(h):
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            if total == 0:
                return float('inf')
            h = h * (n_points / 2) / total
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        def constraint_new(h):
            h = np.clip(h, 0, 1)
            total = np.sum(h)
            return total - n_points / 2
        
        def optimize_h_new(initial_h):
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Objective function
            def objective_func_new(x):
                return evaluate_c5_new(x)
            
            # Constraint function
            def constraint_func_new(x):
                return constraint_new(x)
            
            # Set up the optimization problem
            cons_new = [{'type': 'eq', 'fun': constraint_func_new}]
            result_new = optimize.minimize(objective_func_new, initial_h, bounds=bounds, constraints=cons_new, method='SLSQP', tol=1e-6)
            
            # Ensure valid solution
            h_new = np.clip(result_new.x, 0, 1)
            total = np.sum(h_new)
            if total == 0:
                return h_new, float('inf')
            h_new = h_new * (n_points / 2) / total
            return h_new, evaluate_c5_new(h_new)
        
        # Run optimization for larger n_points
        h_new, c5_new = optimize_h_new(h_initial)
        best_c5 = min(best_c5, c5_new)
        best_h = h_new
    
    return best_h, best_c5, n_points
