from multiprocessing import Pool, cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Initial number of points for performance
    dx = 2.0 / n_points
    initial_h_values = np.array([0.0]*n_points)  # A known good initial solution is available

    # If initial_h_values is given, use it as a base
    if hasattr(initial_h_values, 'copy'):
        h = initial_h_values.copy()
