def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points_list = [100, 150, 200]
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Start with the initial_h_values, then interpolate and normalize
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        h = np.clip(h, 0, 1)
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            h = h / total * 1.0
        
        def objective(h_flat):
            h_vals = h_flat.reshape(n_points)
            total = np.sum(h_vals) * dx
            h_vals = h_vals / total * 1.0
            corr = correlate(h_vals, 1 - h_vals, mode='full')
            c5 = np.max(corr) * dx
            return c5
        
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        
        constraints = []
        
        def constraint_func(h_flat):
            h_vals = h_flat.reshape(n_points)
            total = np.sum(h_vals) * dx
            return total - 1.0
        
        constraints.append({'type': 'eq', 'fun': constraint_func})
        
        for _ in range(3):
            h_initial = h.copy()
            h_initial += np.random.normal(0, 0.05, n_points)
            h_initial = np.clip(h_initial, 0, 1)
            total = np.sum(h_initial) * dx
            h_initial = h_initial / total * 1.0
            
            result = minimize(
                objective,
                h_initial.flatten(),
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                options={'ftol': 1e-6, 'maxiter': 100}
            )
            
            if result.success:
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0, 1)
                total = np.sum(h_opt) * dx
                h_opt = h_opt / total * 1.0
                corr = correlate(h_opt, 1 - h_opt, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt
    
    elapsed = time.time() - start_time
    if elapsed >= budget_s - 1:
        return (best_h, best_c5, len(best_h))
    
    return (best_h, best_c5, len(best_h))
