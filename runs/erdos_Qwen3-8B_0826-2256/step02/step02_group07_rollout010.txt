import numpy as np
import scipy.optimize as optimize
import scipy.signal as signal
import time
import cvxpy as cp
from functools import partial

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(signal.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))

# Initial construction
initial_h_values = np.array([0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5,
                             0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5, 0.5])

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    n_points = 150
    best_h = initial_h_values[:n_points].copy()
    best_h = np.random.uniform(0.95, 1.05, size=n_points)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    best_c5 = evaluate_erdos_solution(best_h)
    
    def optimize_h(h_initial, n_points):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(x):
            h = np.clip(x, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        def constraint_func(x):
            h = np.clip(x, 0, 1)
            return np.sum(h) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective_func(h)
    
    start_time = time.time()
    
    # Initial optimization
    h, c5 = optimize_h(best_h, n_points)
    if c5 < best_c5:
        best_h = h
        best_c5 = c5
    
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        def optimize_h_new(h_initial, n_points):
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective_func_new(x):
                h = np.clip(x, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                corr = signal.correlate(h, 1 - h, mode="full")
                return np.max(corr * (2.0 / n_points))
            
            def constraint_func_new(x):
                h = np.clip(x, 0, 1)
                return np.sum(h) - n_points / 2
            
            cons_new = [{'type': 'eq', 'fun': constraint_func_new}]
            result_new = optimize.minimize(objective_func_new, h_initial, bounds=bounds, constraints=cons_new, method='SLSQP', tol=1e-6)
            h_new = np.clip(result_new.x, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return h_new, objective_func_new(h_new)
        
        h_new, c5_new = optimize_h_new(h_initial, n_points)
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    return best_h, best_c5, n_points
