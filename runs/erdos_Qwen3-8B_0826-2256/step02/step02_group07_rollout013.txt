def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a medium number of points
    n_points = 150
    
    # Initial guess: start from the initial construction and perturb it
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Function to evaluate C5
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Function to enforce the constraint
    def constraint(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Optimization using SLSQP
    def optimize_h(h_initial, n_points, evaluate_func, constraint_func):
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function
        def objective_func(x):
            return evaluate_func(x)
        
        # Constraint function
        def constraint_func(x):
            return constraint_func(x)
        
        # Set up the optimization problem
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Ensure the solution is valid
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective_func(h)
    
    # Start with initial optimization
    start_time = time.time()
    best_h = h_initial
    best_c5 = evaluate_c5(best_h)
    
    # Initial optimization
    try:
        h, c5 = optimize_h(h_initial, n_points, evaluate_c5, constraint)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    except:
        pass
    
    # If time allows, try increasing n_points for better results
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define new evaluation function
        def evaluate_c5_new(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        # Define new constraint function
        def constraint_new(h):
            h = np.clip(h, 0, 1)
            return np.sum(h) - n_points / 2
        
        # Optimization for larger n_points
        try:
            h_new, c5_new = optimize_h(h_initial, n_points, evaluate_c5_new, constraint_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except:
            pass
    
    # If time allows, try perturbing the best solution with random noise
    if time.time() - start_time < budget_s - 10:
        perturbation = np.random.normal(0, 0.05, size=len(best_h))
        perturbed_h = np.clip(best_h + perturbation, 0, 1)
        perturbed_h = perturbed_h * (len(perturbed_h) / 2) / np.sum(perturbed_h)
        c5_perturbed = evaluate_c5(perturbed_h)
        if c5_perturbed < best_c5:
            best_h = perturbed_h
            best_c5 = c5_perturbed
    
    return best_h, best_c5, n_points
