def run(seed=42, budget_s=100, **kwargs):

    def generate_initial_config(n_points):
        h = np.zeros(n_points)

        num_ones = int(np.ceil(n_points / 2))
        indices = np.random.choice(n_points, size=num_ones, replace=False)
        h[indices] = 1.0
        # Fill the rest with random values and normalize
        h[~np.isin(np.arange(n_points), indices)] = np.random.rand(n_points - num_ones)
        h = h * (n_points / 2) / np.sum(h)
        return h

    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    # Generate initial population of different start points
