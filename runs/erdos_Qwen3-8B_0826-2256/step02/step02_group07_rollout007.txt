def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Initial n_points
    n_points = 100
    best_h = np.zeros(n_points)
    best_c5 = 1.0
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        h = np.clip(h, 0, 1)
        total = np.sum(h)
        if total == 0:
            return 1.0
        h = h * (n_points / 2) / total
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Function to optimize h with constraints
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function to minimize
        def objective_func(x):
            return evaluate_c5(x)
        
        # Constraint function
        def constraint_func(x):
            total = np.sum(x)
            return total - n_points / 2
        
        # Set up the optimization problem
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Ensure the solution is valid
        h = np.clip(result.x, 0, 1)
        total = np.sum(h)
        if total == 0:
            return h, 1.0
        h = h * (n_points / 2) / total
        return h, evaluate_c5(h)
