import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize import differential_evolution

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for balance between performance and precision
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # If initial_h_values is given, use it as a base
    def initial_h_values_func():
        return np.array([0.0]*n_points)
    
    # Generate initial guess based on the initial_h_values
    initial_h_values = initial_h_values_func()
    h = initial_h_values.copy()
    
    # Normalize the initial guess to satisfy the integral constraint
    h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values
        # Compute the correlation
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral is 1
    
    bounds = Bounds(0, 1)
    
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    # Use differential evolution for global optimization
    def differential_evolution_run(h_initial):
        result = differential_evolution(
            objective,
            bounds,
            args=(),
            strategy='best1bin',
            popsize=15,
            tol=1e-8,
            mutation=(0.5, 1),
            recombination=0.8,
            seed=seed,
            maxiter=100,
            disp=False
        )
        return result.x
    
    # Try multiple random initializations
    for _ in range(5):  # Try 5 different initializations
        if time.time() - start_time > budget_s - 1:
            break
        # Generate new random initial guess
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / (np.sum(h_initial) * dx) if np.sum(h_initial) != 0 else np.random.rand(n_points)
        h_initial = np.clip(h_initial, 0, 1)
        
        # Run differential evolution
        result = differential_evolution_run(h_initial)
        result_c5 = objective(result)
        
        if result_c5 < best_c5:
            best_c5 = result_c5
            best_h = result
    
    return best_h, best_c5, n_points
