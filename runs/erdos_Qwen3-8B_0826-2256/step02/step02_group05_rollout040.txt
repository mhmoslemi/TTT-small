import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp
from deap import base, creator, tools, algorithms

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    n_points = 100  # Start with 100 samples for efficiency

    # Optimizer parameters
    max_iterations = 100
    time_limit = budget_s

    # Precompute dx
    dx = 2.0 / n_points
    total_integral = n_points / 2

    # Function to evaluate C5
    def evaluate(h_values):
        return evaluate_erdos_solution(h_values)

    # Normalize function
    def normalize(h_values):
        total = np.sum(h_values)
        return h_values * (total_integral / total) if total != 0 else h_values

    # Setup for SLSQP optimization
    def slsqp_optimization(h_start):
        h = np.copy(h_start)
        h = normalize(h)
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        def objective(h_values):
            return evaluate(h_values)
        def constraint(h_values):
            return np.sum(h_values) - total_integral
        cons = {'type': 'eq', 'fun': constraint}
        result = opt.minimize(
            fun=objective,
            x0=h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            options={'ftol': 1e-9, 'maxiter': 100}
        )
        return result.x

    # Setup for Genetic Algorithm
    def ga_optimization(population_size=50, generations=50, cxpb=0.5, mutpb=0.1):
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", list, fitness=creator.FitnessMin)

       toolbox = base.Toolbox()
        toolbox.register("attr_float", np.random.uniform, 0.0, 1.0)
        toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)

        def eval_func(individual):
            h_values = normalize(individual)
            return (evaluate(h_values),)

        def cx_uniform(ind1, ind2):
            ind1, ind2 = list(ind1), list(ind2)
            for i in range(len(ind1)):
                if np.random.rand() < 0.5:
                    ind1[i], ind2[i] = ind2[i], ind1[i]
            return ind1, ind2

        def mut_gaussian(individual):
            for i in range(len(individual)):
                individual[i] = np.clip(np.random.normal(individual[i], 0.1), 0.0, 1.0)
            return individual,

        toolbox.register("evaluate", eval_func)
        toolbox.register("mate", cx_uniform)
        toolbox.register("mutate", mut_gaussian)
        toolbox.register("select", tools.selTournament, tournsize=3)

        population = toolbox.population(n=population_size)
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        pe = tools.ParetoFront()
        logbook = tools.Logbook()
        logbook.header = ["gen", "nevals"] + stats.fields

        for gen in range(generations):
            offspring = algorithms.varAnd(population, toolbox, cxpb, mutpb)
            fits = toolbox.map(toolbox.evaluate, offspring)
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            population = tools.selBest(population + offspring, population_size)
            record = stats.compile(population)
            logbook.record(gen=gen, nevals=len(offspring), **record)
            pe.update(population)
            if gen % 10 == 0:
                print(f"Gen {gen}: Best = {record['min']}")
            # Early stop if no new min found for some generations
            if gen > 10 and record["min"] == pe.get_all()[0][0]:
                print("Early stopping: no improvement.")
                break

        best_ind = pe.best()
        return best_ind

    # Begin the optimization loop
    start_time = time.time()
    for iteration in range(max_iterations):

        current_h = slsqp_optimization(np.copy(initial_h_values))
        current_c5 = evaluate(current_h)

            best_c5 = current_c5
            best_h = current_h
        # Try GA optimization
        ga_pop = ga_optimization(population_size=50, generations=30, cxpb=0.8, mutpb=0.2)
        ga_h = np.array(ga_pop)
        ga_h = normalize(ga_h)
        ga_c5 = evaluate(ga_h)
        if ga_c5 < best_c5:
            best_c5 = ga_c5
            best_h = ga_h
        # Check remaining time
        elapsed_time = time.time() - start_time
        if elapsed_time >= time_limit - 1:
            break

    # Ensure we return a valid solution
    best_h = np.clip(best_h, 0.0, 1.0)
    best_h = normalize(best_h)
    final_c5 = evaluate(best_h)
    return best_h, final_c5, n_points
