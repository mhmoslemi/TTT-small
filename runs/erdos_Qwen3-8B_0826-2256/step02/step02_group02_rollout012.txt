import numpy as np
import scipy.optimize as opt
import time
import multiprocessing as mp
from functools import partial
from scipy.optimize import minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    initial_h_values = np.load("initial_h_values.npy")  # Assuming this is available
    
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5
    
    # Genetic algorithm setup
    def setup_ga(n_points):
        creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
        creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)
        
        toolbox = base.Toolbox()
        toolbox.register("evaluate", lambda ind: evaluate_c5(ind, n_points))
        toolbox.register("individual", tools.initRepeat, creator.Individual, lambda: np.random.uniform(0, 1, n_points))
        toolbox.register("population", tools.initRepeat, list, toolbox.individual)
        toolbox.register("mate", tools.cxTwoPoint)
        toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.05)
        toolbox.register("select", tools.selTournament, tournsize=3)
        
        return toolbox
    
    # Local optimization with SLSQP
    def optimize_h_local(n_points, initial_h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}]
        result = minimize(lambda x: evaluate_c5(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options={'maxiter': 100, 'disp': False})
        return result.x, evaluate_c5(result.x, n_points)
    
    # Main optimization loop
    best_h = initial_h_values
    best_c5 = evaluate_c5(best_h, len(best_h))
    n_points = len(best_h)
    start_time = time.time()
    
    # Initialize genetic algorithm
    toolbox = setup_ga(n_points)
    population = toolbox.population(n=50)
    
    # Evaluate initial population
    fitnesses = list(map(toolbox.evaluate, population))
    for ind, fit in zip(population, fitnesses):
        ind.fitness.values = (fit,)
    
    # Main genetic algorithm loop
    gen = 0
    while time.time() - start_time < budget_s - 1:
        offspring = algorithms.varAnd(population, toolbox, cxpb=0.5, mutpb=0.1)
        fits = map(toolbox.evaluate, offspring)
        offspring = algorithms.varOr(offspring, toolbox, lambda ind: ind.fitness.values)
        for ind, fit in zip(offspring, fits):
            ind.fitness.values = (fit,)
        
        population = toolbox.select(population + offspring, len(population))
        gen += 1
        
        # Check for best solution
        best_individual = min(population, key=lambda ind: ind.fitness.values[0])
        if best_individual.fitness.values[0] < best_c5:
            best_h = best_individual
            best_c5 = best_individual.fitness.values[0]
            print(f"Improved C5: {best_c5}")
    
    # Refine the best solution using local optimization
    best_h, best_c5 = optimize_h_local(n_points, best_h)
    
    return best_h, best_c5, n_points
