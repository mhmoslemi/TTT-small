import numpy as np
import scipy.optimize as optimize
import time
import random
from scipy.optimize import minimize
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h, n_points: max(np.correlate(h, 1 - h, mode="full") * (2.0 / n_points))
initial_h_values = np.array([0.5]*100)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    best_h_values = None
    best_c5_value = float('inf')

    # Define the range of n_points to try
    n_points_list = [100, 150, 200, 250]  # focus on smaller, more efficient search
    
    for n_points in n_points_list:
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        
        # Initialize with initial_h_values interpolated for better performance
        if n_points < len(initial_h_values):
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        else:
            h = np.zeros(n_points)
        
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            h = h / total * 1.0
        
        # Define the objective function
        def objective(h_flat):
            h_values = h_flat.reshape(n_points)
            # Normalize to ensure integral constraint
            total = np.sum(h_values) * dx
            h_values = h_values / total * 1.0
            # Evaluate the maximum overlap
            corr = correlate(h_values, 1 - h_values, mode="full")
            c5 = np.max(corr) * dx
            return c5

        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = [
            {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
        ]
        
        # Run optimization from multiple starting points to avoid local optima
        for _ in range(3):  # Run multi-start optimization
            h_initial = h.copy()
            # Add some random noise for variation
            h_initial += np.random.normal(0, 0.05, n_points)
            h_initial = np.clip(h_initial, 0, 1)
            # Ensure integral constraint again
            total = np.sum(h_initial) * dx
            h_initial = h_initial / total * 1.0

            result = minimize(
                objective,
                h_initial.flatten(),
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                options={'ftol': 1e-6, 'maxiter': 100}
            )

            if result.success:
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0, 1)

                total = np.sum(h_opt) * dx

                    best_c5_value = c5
                    best_h_values = h_opt

        # Check time budget

        if elapsed >= budget_s - 1:
            break

    return (best_h_values, best_c5_value, len(best_h_values))
