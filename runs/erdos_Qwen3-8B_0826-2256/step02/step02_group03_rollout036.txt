import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from deap import base, creator, tools, algorithms
import multiprocessing as mp

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points

    # Use initial_h_values if it exists
    if hasattr(initial_h_values, 'copy'):
        h_initial = initial_h_values.copy()
    else:
        h_initial = np.random.rand(n_points)

    h_initial = h_initial / (np.sum(h_initial) * dx) if np.sum(h_initial) != 0 else np.random.rand(n_points)
    h_initial = np.clip(h_initial, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)

    # GA setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

   toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.uniform, 0, 1)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    toolbox.register("evaluate", objective)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Define a function to run GA in parallel
    def ga_run():
        pop = toolbox.population(n=50)
        stats = tools.Statistics(lambda ind: ind.fitness.values)
        stats.register("avg", np.mean)
        stats.register("std", np.std)
        stats.register("min", np.min)
        stats.register("max", np.max)

        algorithms.eaSimple(pop, toolbox, cxpb=0.5, mutpb=0.1, ngen=20, stats=stats, halloffame=None, verbose=False)
        best_ind = tools.selBest(pop, 1)[0]
        return best_ind.fitness.values[0], best_ind

    # Run multiple GA trials in parallel
    start_time = time.time()
    best_c5 = float('inf')
    best_h = h_initial

    with mp.Pool(processes=10) as pool:
        results = []
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            # Start a new GA trial
            result = pool.apply_async(ga_run)
            results.append(result)

        for result in results:
            if time.time() - start_time > budget_s - 1:
                break
            c5, h = result.get()
            if c5 < best_c5:

    def refine_solution(h_values):

        return np.sum(h_values) - n_points / 2

    x0 = best_h
    result = minimize(
        refine_solution,
        x0,
        bounds=bounds,
        constraints={"type": "eq", "fun": refine_constraint},
        method="SLSQP",
        options={"ftol": 1e-8, "maxiter": 100}
    )

    if result.success:
        best_c5 = result.fun
        best_h = result.x

    return best_h, best_c5, n_points
