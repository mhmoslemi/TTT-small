import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.optimize._differentiable_functions import ScalarFunction

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    n_points = 200
    dx = 2.0 / n_points
    
    # Start with initial_h_values if available
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values
    
    # Interpolate initial_h to the target resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)
    
    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)
    
    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    # Define the constraints
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint
    
    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimizer settings
    options = {
        'ftol': 1e-8,
        'maxiter': 100,
        'disp': False
    }
    
    # Define a more robust optimization algorithm
    def optimization_step(initial_guess):
        result = minimize(
            objective,
            initial_guess,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options=options
        )
        return result.x, result.fun
    
    # Run a series of optimizations with diverse starting points
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    
    for _ in range(10):  # Perform multiple iterations
        if time.time() - start_time > budget_s:
            break
        
        # Generate a diverse initial guess
        initial_guess = np.random.rand(n_points)
        initial_guess = initial_guess / (np.sum(initial_guess) * dx)
        initial_guess = np.clip(initial_guess, 0, 1)
        
        h_opt, c5_opt = optimization_step(initial_guess)
        
        # Update the best solution
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt
    
    # Final evaluation
    return best_h, best_c5, n_points
