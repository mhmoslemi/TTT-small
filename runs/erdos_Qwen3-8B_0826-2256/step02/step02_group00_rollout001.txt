import scipy.optimize as optimize
import time

def run(seed=42, budget_s=100, **kwargs):

    n_points = 200  # Increase resolution to 200 points for better results
    dx = 2.0 / n_points

    # Ensure initial_h_values are available
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    # Interpolate initial_h to match the n_points resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Run multiple simulated annealing iterations with different initializations
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    n_restarts = 5

    for _ in range(n_restarts):

            break

        initial_guess = h + 0.05 * np.random.randn(n_points)

        # Use L-BFGS-B for faster optimization
        result = optimize.minimize(
            objective,
            initial_guess,
            bounds=bounds,
