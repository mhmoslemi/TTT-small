import numpy as np
import scipy.optimize as opt
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):

    h_values = initial_h_values.copy()

    # Function to evaluate C5 for a given h
    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(cross_corr) * dx
        return c5

    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)

    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2

    # Optimization configuration
    options = {'maxiter': 200, 'disp': False}

    # Optimization
    def optimize_h(n_points, initial_h):
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, n_points)}]
        result = minimize(lambda x: objective(x, n_points), initial_h, method='SLSQP', bounds=bounds, constraints=cons, options=options)
        return result.x, evaluate_c5(result.x, n_points)

    # Initial optimization
    best_h, best_c5 = optimize_h(initial_n_points, h_values)
    best_n_points = initial_n_points

    # Refinement loop
    start_time = time.time()
    while time.time() - start_time < budget_s - 1:
        # Try increasing n_points
        new_n_points = best_n_points * 2
        # Initialize with previous best values
        new_h = np.zeros(new_n_points)
        new_h[:best_n_points] = best_h
        # Fill the rest with random values and normalize
        new_h[best_n_points:] = np.random.rand(new_n_points - best_n_points)
        new_h = new_h * (new_n_points / 2) / np.sum(new_h)
        
        new_h, new_c5 = optimize_h(new_n_points, new_h)
        
        # Check if the new solution is better
        if new_c5 < best_c5:
            best_h, best_c5 = new_h, new_c5
            best_n_points = new_n_points
            print(f"Improved C5: {best_c5}")
        else:
            # Try local refinement by perturbing current solution
            perturbed_h = best_h + np.random.normal(0, 0.02, best_n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (best_n_points / 2) / np.sum(perturbed_h)
            perturbed_h, perturbed_c5 = optimize_h(best_n_points, perturbed_h)
            
            if perturbed_c5 < best_c5:
                best_h, best_c5 = perturbed_h, perturbed_c5
                print(f"Improved C5: {best_c5}")

                # Try further refinement with different perturbation scale
                perturbed_h = best_h + np.random.normal(0, 0.01, best_n_points)

                perturbed_h, perturbed_c5 = optimize_h(best_n_points, perturbed_h)
                
                if perturbed_c5 < best_c5:
                    best_h, best_c5 = perturbed_h, perturbed_c5
                    print(f"Improved C5: {best_c5}")
                else:
                    # No improvement, break
                    break
