```python
import numpy as np
import scipy.optimize as opt
import time
import cvxpy as cp

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(100)  # Placeholder for actual initial values

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 150  # Increased for better accuracy while keeping it under 1k
    dx = 2.0 / n_points
    total_integral = n_points / 2  # Sum(h) == n_points / 2

    # Function to evaluate the maximum overlap
    def objective_function(h_values):
        corr = np.correlate(h_values, 1 - h_values, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap

    # Function to generate an initial guess
    def initial_guess(n_points, seed=42):
        h = np.random.rand(n_points)
        h = (h * 2 - 1) * 0.5 + 0.5  # Normalize to [0,1]
        h = np.clip(h, 0, 1)
        h = h * (total_integral / np.sum(h))  # Ensure integral constraint
        return h

    # Function to optimize using SLSQP with bounds and constraints
    def optimize(h_initial):
        def objective(x):
            corr = np.correlate(x, 1 - x, mode="full")
            return np.max(corr) * dx

        result = opt.minimize(
            objective,
            h_initial,
            method='SLSQP',
            bounds=[(0, 1) for _ in range(n_points)],
            constraints={'type': 'eq', 'fun': lambda x: np.sum(x) - total_integral},
            options={'ftol': 1e-10, 'maxiter': 100}
        )
        return result.x, result.fun

    # Function to perturb and improve a solution
    def perturb_and_improve(h_current):
        perturbed_h = h_current + 0.05 * np.random.randn(n_points)
        perturbed_h = np.clip(perturbed_h, 0, 1)
        perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
        optimized_h, optimized_c5 = optimize(perturbed_h)
        return optimized_h, optimized_c5

    # Start with the initial guess
    best_h = initial_guess(n_points, seed)
    best_c5 = objective_function(best_h)

    # Start timer
    start_time = time.time()

    # Optimize the initial guess
    try:
        optimized_h, optimized_c5 = optimize(best_h)
        if optimized_c5 < best_c5:
            best_h = optimized_h
            best_c5 = optimized_c5
    except:
        pass

    # If time remains, perform multiple perturbations
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 20:
        for _ in range(5):
            perturbed_h = best_h + 0.1 * np.random.randn(n_points)
            perturbed_h = np.clip(perturbed_h, 0, 1)
            perturbed_h = perturbed_h * (total_integral / np.sum(perturbed_h))
            optimized_h, optimized_c5 = optimize(perturbed_h)
            if optimized_c5 < best_c5:
                best_h = optimized_h
                best_c5 = optimized_c5

    return best_h, best_c5, n_points
```
