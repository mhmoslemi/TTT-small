import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from joblib import Parallel, delayed
from multiprocessing import cpu_count

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 500  # Adjusted to balance performance and exploration
    dx = 2.0 / n_points
    
    # Initial construction
    initial_h_values = np.load('initial_h_values.npy') if hasattr(initial_h_values, 'copy') else np.random.rand(n_points)
    initial_h_values = initial_h_values.copy()

    initial_h_values = np.clip(initial_h_values, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral is 1

    bounds = Bounds(0, 1)
    
    def evaluate_candidate(h_values):
        h = h_values.copy()
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        return result.fun, result.x
    
    # Parallel evaluation
    num_cores = cpu_count()
    num_samples = min(10, num_cores)  # Number of parallel evaluations

    # Evaluate in parallel
    results = Parallel(n_jobs=num_cores)(
