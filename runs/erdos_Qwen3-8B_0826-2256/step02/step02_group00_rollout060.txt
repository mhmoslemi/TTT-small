import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use more points for better resolution
    dx = 2.0 / n_points

    # Ensure the initial_h_values are available
    if hasattr(initial_h_values, 'copy'):
        initial_h = initial_h_values.copy()
    else:
        initial_h = initial_h_values

    # Interpolate initial_h to match the n_points resolution
    x_initial = np.linspace(0, 2, len(initial_h))
    x_target = np.linspace(0, 2, n_points)
    h = np.interp(x_target, x_initial, initial_h)

    # Normalize to ensure the integral constraint
    h = h / (np.sum(h) * dx)
    h = np.clip(h, 0, 1)

    # Define the objective function to minimize the maximum overlap
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    # Define the constraints
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral constraint

    # Define bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Use a hybrid approach combining SLSQP with local optimization
    def optimize_solution():
        # Initial guess
        initial_h = np.random.rand(n_points)
        initial_h = initial_h / (np.sum(initial_h) * dx)
        initial_h = np.clip(initial_h, 0, 1)

        # Run SLSQP for global exploration
        result = minimize(
            objective,
            initial_h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        h_opt = result.x
        c5_opt = result.fun

        # Refine with local optimization
        result = minimize(
            objective,
            h_opt,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="L-BFGS-B",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        return result.x, result.fun

    # Run multiple optimization iterations
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()

    for _ in range(5):
        if time.time() - start_time > budget_s:
            break
        h_opt, c5_opt = optimize_solution()
        if c5_opt < best_c5:
            best_h = h_opt
            best_c5 = c5_opt

    # Final evaluation
    return best_h, best_c5, n_points
