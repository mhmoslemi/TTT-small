import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try a range of n_points values for better performance and exploration
    n_points_candidates = [100, 150, 200]
    best_c5 = float('inf')
    best_h = np.array([])
    best_n_points = 100
    
    for n_points in n_points_candidates:
        dx = 2.0 / n_points
        
        # Use the initial_h_values if provided as a starting point
        if hasattr(initial_h_values, 'copy'):
            h = initial_h_values.copy()
        else:
            h = np.random.rand(n_points)
        
        # Ensure sum(h) * dx = 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        def objective(h_values):
            h = h_values.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        def constraints(h_values):
            return np.sum(h_values) - n_points / 2  # Ensure the integral is 1
        
        bounds = Bounds(0, 1)
        best_h_current = h
        best_c5_current = objective(h)
        start_time = time.time()
        
        # Run multiple trials with different initializations within the budget
        for _ in range(5):  # Try 5 different initializations
            if time.time() - start_time > budget_s - 1:
                break
            # Use a new random initialization
            h = np.random.rand(n_points)
            h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            x0 = h
            
            # Use SLSQP with tighter settings
            result = minimize(
                objective,
                x0,
                bounds=bounds,
                constraints={"type": "eq", "fun": constraints},
                method="SLSQP",
                options={"ftol": 1e-8, "maxiter": 100}
            )
            
            if result.success:
                current_c5 = result.fun
                if current_c5 < best_c5_current:
                    best_c5_current = current_c5
                    best_h_current = result.x
        
        # Update the best solution found so far
        if best_c5_current < best_c5:
            best_c5 = best_c5_current
            best_h = best_h_current
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
