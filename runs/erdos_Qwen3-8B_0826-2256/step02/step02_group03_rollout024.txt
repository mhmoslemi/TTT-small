import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Use the provided initial values if available
    initial_h_values = np.array([])
    if hasattr(initial_h_values, 'copy'):
        initial_h_values = initial_h_values.copy()
    else:
        initial_h_values = np.random.rand(200)
    
    n_points = 200
    dx = 2.0 / n_points
    # Ensure the sum of the initial guess satisfies the constraint
    initial_h_values = initial_h_values / (np.sum(initial_h_values) * dx) if np.sum(initial_h_values) != 0 else np.random.rand(n_points)
    initial_h_values = np.clip(initial_h_values, 0, 1)

    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5
    
    def constraints(h_values):
        return np.sum(h_values) - n_points / 2  # Ensure the integral is 1

    bounds = Bounds(0, 1)
    
    def evaluate(individual):
        return objective(individual),  # Return the single objective value

    # Define the DEAP fitness and individual
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", np.ndarray, fitness=creator.FitnessMin)

    # Setup the toolbox
    toolbox = base.Toolbox()
    toolbox.register("attr_float", np.random.rand, n_points)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attr_float, n=n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)
    toolbox.register("evaluate", evaluate)
    toolbox.register("mate", tools.cxBlend, alpha=0.5)
    toolbox.register("mutate", tools.mutGaussian, mu=0, sigma=0.1, indpb=0.1)
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Initial population
    pop = toolbox.population(n=50)

    # Define a simple genetic algorithm for global exploration
    def genetic_search():
        start_time = time.time()
        for gen in range(100):  # Limit number of generations to keep within budget
            if time.time() - start_time > budget_s - 1:
                break
            offspring = algorithms.varAnd(pop, toolbox, cxpb=0.5, mutpb=0.1)
            fits = [toolbox.evaluate(ind) for ind in offspring]
            for fit, ind in zip(fits, offspring):
                ind.fitness.values = fit
            pop = toolbox.select(pop + offspring, len(pop))
            # Replace the population with the best individuals
            pop = tools.selBest(pop, len(pop))
        # Return the best individual from the genetic search
        best_ind = tools.selBest(pop, 1)[0]
        return best_ind
    
    # Run the genetic search
    best_h = genetic_search()

    # Refine the best solution using SLSQP for fine-tuning
    def refine_solution(h_values):
        h = h_values.reshape(-1)
        # Enforce the constraint using SLSQP
        result = minimize(
            objective,
            h,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraints},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 50}
        )
        return result.x if result.success else h

    best_h = refine_solution(best_h)
    best_c5 = objective(best_h)

    # Additional refinement by trying multiple restarts
    for _ in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        refined_h = refine_solution(h)
        refined_c5 = objective(refined_h)
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = refined_h

    return best_h, best_c5, n_points
