from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(140) * 2 - 1  # Placeholder, actual initial construction may vary
initial_h_values = np.clip(initial_h_values, 0, 2)
initial_h_values = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))

initial_h_values = initial_h_values / np.sum(initial_h_values) * (140 / 2)

def run(seed=42, budget_s=100, **kwargs):

    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    best_h_list = [best_h]
    
    # Define the objective to minimize the maximum overlap
    def objective(h):
        cor = correlate(h, 1 - h, mode='full')
        return np.max(cor) * (2.0 / n_points)
    
    # Constraint: integral of h is 1
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Optimizer setup
    x = cp.Variable(n_points)
    objective_func = cp.Minimize(objective(x))
    constraints = [constraint(x) == 0, 0 <= x, x <= 1]
    
    for _ in range(10):
        if time.time() - time.time() > budget_s - 10:
            break
        if _ == 0:
            result = cp.Problem(objective_func, constraints).solve(solver='MOSEK', verbose=False)
