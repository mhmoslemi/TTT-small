def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a medium number of points
    n_points = 150
    # We'll try different n_points and select the best solution
    
    # Function to evaluate a candidate solution
    def evaluate(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Function to optimize a solution for a given n_points
    def optimize_solution(n_points, initial_guess):
        bounds = [(0, 1) for _ in range(n_points)]
        
        def objective_func(x):
            return evaluate(x)
        
        def constraint_func(x):
            x = np.clip(x, 0, 1)
            return np.sum(x) - n_points / 2
        
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, initial_guess, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, evaluate(h)
    
    # Initial guess: start from the initial construction and perturb it
    initial_guess = np.copy(initial_h_values[:n_points])
    initial_guess = np.random.uniform(0.95, 1.05, size=n_points)
    initial_guess = np.clip(initial_guess, 0, 1)
    initial_guess = initial_guess * (n_points / 2) / np.sum(initial_guess)
    
    # Start optimization
    best_h = initial_guess
    best_c5 = evaluate(best_h)
    best_n_points = n_points
    
    # Try with different n_points and select the best solution
    for n_points in [150, 200, 250]:
        if time.time() > budget_s - 5:
            break
        
        # Generate a new initial guess for this n_points
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Optimize
        try:
            h, c5 = optimize_solution(n_points, h_initial)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
                best_n_points = n_points
        except:
            pass
    
    return best_h, best_c5, best_n_points
