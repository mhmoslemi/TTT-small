def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    # Define the range of n_points to try
    n_points_list = [100, 150, 200, 250]  # Focus on smaller, more efficient search
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    best_h_values = None
    best_c5_value = float('inf')

    for n_points in n_points_list:
        dx = 2.0 / n_points
        # Initialize with the initial_h_values, then interpolate and normalize for better performance
        if n_points < len(initial_h_values):
            h = initial_h_values[:n_points]
        else:
            h = np.zeros(n_points)
        # Interpolate and adjust the initial guess for better optimization
        h = np.clip(h, 0, 1)
        # Ensure the integral constraint is satisfied
        total = np.sum(h) * dx
        if abs(total - 1.0) > 1e-6:
            h = h / total * 1.0

        # Define the objective function
        def objective(h_flat):
            h_values = h_flat.reshape(n_points)
            # Normalize to ensure integral constraint
            total = np.sum(h_values) * dx
            h_values = h_values / total * 1.0
            # Evaluate the maximum overlap
            corr = correlate(h_values, 1 - h_values, mode="full")
            c5 = np.max(corr) * dx
            return c5

        # Define constraints and bounds
        bounds = [(0.0, 1.0) for _ in range(n_points)]
        constraints = []

        # Penalty function to maintain the integral constraint
        def constraint_func(h_flat):
            h_values = h_flat.reshape(n_points)
            total = np.sum(h_values) * dx
            return total - 1.0

        constraints.append({'type': 'eq', 'fun': constraint_func})

        # Define a regularizer to avoid extreme variance in h
        def regularizer(h_flat):
            h_values = h_flat.reshape(n_points)
            variance = np.var(h_values)
            return variance

        # Run optimization from multiple initial points to avoid local optima
        for _ in range(3):  # Run multi-start optimization
            h_initial = h.copy()
            # Add some random noise for variation
            h_initial += np.random.normal(0, 0.05, n_points)
            h_initial = np.clip(h_initial, 0, 1)
            # Ensure integral constraint again
            total = np.sum(h_initial) * dx
            h_initial = h_initial / total * 1.0

            result = minimize(
                objective,
                h_initial.flatten(),
                method='SLSQP',
                constraints=constraints,
                bounds=bounds,
                options={'ftol': 1e-6, 'maxiter': 100}
            )

            if result.success:
                h_opt = result.x.reshape(n_points)
                h_opt = np.clip(h_opt, 0, 1)
                # Normalize to ensure the integral constraint
                total = np.sum(h_opt) * dx
                h_opt = h_opt / total * 1.0

                corr = correlate(h_opt, 1 - h_opt, mode="full")
                c5 = np.max(corr) * dx
                if c5 < best_c5_value:
                    best_c5_value = c5
                    best_h_values = h_opt

        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 1:
            break

    return (best_h_values, best_c5_value, len(best_h_values))
