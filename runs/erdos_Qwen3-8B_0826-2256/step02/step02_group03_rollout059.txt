import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy.signal import correlate

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    n_points = 100  # Use 100 points for performance and sufficient resolution
    
    # Precompute shifts for correlation
    shifts = np.arange(-n_points + 1, n_points)
    
    # Initial guess based on the given initial_h_values
    initial_h_values = np.load('initial_h_values.npy')  # Assuming this is precomputed
    
    # Define the objective function
    def objective(h_values):
        h = h_values.reshape(-1)
        # Compute the overlap integral for all shifts
        overlaps = np.zeros_like(shifts)
        for i, shift in enumerate(shifts):
            shifted_h = np.roll(h, shift)
            overlap = np.sum(h * (1 - shifted_h)) * (2.0 / n_points)
            overlaps[i] = overlap
        # Return the maximum overlap
        return np.max(overlaps)
    
    # Define the constraint function
    def constraint(h_values):
        return np.sum(h_values) - n_points / 2  # Ensures the integral equals 1
    
    bounds = Bounds(0, 1)
    best_h = initial_h_values.copy()
    best_c5 = objective(best_h)
    start_time = time.time()
    
    # Run multiple trials with different initializations
    for _ in range(5):  # Try 5 different initializations
        if time.time() - start_time > budget_s - 1:
            break
        # Use a new random initialization
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * (2.0 / n_points)) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h
        
        # Use SLSQP with tighter settings
        result = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-8, "maxiter": 100}
        )
        
        if result.success:
            current_c5 = result.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result.x
    
    return best_h, best_c5, n_points
