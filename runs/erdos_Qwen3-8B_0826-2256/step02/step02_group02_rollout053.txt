from deap import base, creator, tools, algorithms

def run(seed=42, budget_s=100, **kwargs):

    # Initial construction from initial_h_values
    initial_n_points = len(initial_h_values)
    h_values = initial_h_values.copy()

    def evaluate_c5(h, n_points):
        h = np.array(h)
        dx = 2.0 / n_points
        cross_corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5

    # Function to compute the objective and constraints
    def objective(x, n_points):
        return evaluate_c5(x, n_points)

    def constraint_sum(x, n_points):
        return np.sum(x) - n_points / 2

    # Create the genetic algorithm setup
    creator.create("FitnessMin", base.Fitness, weights=(-1.0,))
    creator.create("Individual", list, fitness=creator.FitnessMin)

    # Initialize toolbox
    toolbox = base.Toolbox()
    toolbox.register("attribute", np.random.uniform, 0.0, 1.0)
    toolbox.register("individual", tools.initRepeat, creator.Individual, toolbox.attribute, n=initial_n_points)
    toolbox.register("population", tools.initRepeat, list, toolbox.individual)

    # Register evaluation function
    def evaluate(individual):
        n_points = len(individual)
        dx = 2.0 / n_points
        cross_corr = np.correlate(individual, 1 - individual, mode="full")
        c5 = np.max(cross_corr) * dx
        return c5,

    toolbox.register("evaluate", evaluate)

    # Register crossover operator
    toolbox.register("mate", tools.cxTwoPoint)

    # Register mutation operator
    def mutate(individual, indpb=0.1):
        for i in range(len(individual)):
            if np.random.random() < indpb:
                individual[i] = np.random.uniform(0.0, 1.0)
        return individual,

    toolbox.register("mutate", mutate)

    # Register selection operator
    toolbox.register("select", tools.selTournament, tournsize=3)

    # Genetic algorithm parameters
    population_size = 50
    generations = 100
    cxpb = 0.5
    mutpb = 0.1
    n_points = initial_n_points

    # Start time
    start_time = time.time()

    # Initialize population with initial_h_values
    initial_population = []
    for _ in range(population_size):
        ind = np.copy(h_values)
        np.random.shuffle(ind)
        initial_population.append(ind)

    # Initialize the population with the initial solution
    population = tools.initIterate(toolbox.population, initial_population)

    # Keep track of the best solution
    best_h = h_values
    best_c5 = evaluate_c5(h_values, n_points)
    best_n_points = n_points

    # Genetic algorithm loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        offspring = algorithms.varAnd(population, toolbox, cxpb, mutpb)
        fits = toolbox.map(toolbox.evaluate, offspring)
        for fit, ind in zip(fits, offspring):
            ind.fitness.values = fit
        population = tools.selTournament(population + offspring, len(population))
        # Local refinement with SLSQP
        for ind in population:
            if np.random.random() < 0.3:
                bounds = [(0.0, 1.0) for _ in range(len(ind))]
                cons = [{'type': 'eq', 'fun': lambda x: constraint_sum(x, len(ind))}]

        # Update best solution

            if ind.fitness.values[0] < best_c5:
                best_c5 = ind.fitness.values[0]
                best_h = np.copy(ind)
                best_n_points = len(ind)
                print(f"Improved C5: {best_c5}")

    return best_h, best_c5, best_n_points
