```python
import numpy as np
import scipy.optimize as optimize
import time
import cvxpy as cp
from scipy.signal import correlate

# Pre-imported functions
evaluate_erdos_solution = lambda h: np.max(np.correlate(h, 1 - h, mode="full")) * (2.0 / len(h))
initial_h_values = np.random.rand(140) * 2 - 1  # Placeholder, actual initial construction may vary
initial_h_values = np.clip(initial_h_values, 0, 2)
initial_h_values = (initial_h_values - np.min(initial_h_values)) / (np.max(initial_h_values) - np.min(initial_h_values))
initial_h_values = np.clip(initial_h_values, 0, 1)
initial_h_values = initial_h_values * (140 / 2)  # Normalize to sum to 140/2
initial_h_values = np.round(initial_h_values).astype(float)
initial_h_values = initial_h_values / np.sum(initial_h_values) * (140 / 2)

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_c5 = float('inf')
    best_h = initial_h_values.copy()
    n_points = len(best_h)
    
    # Define the objective function
    def objective(x):
        h = x
        # Compute the max overlap
        cor = correlate(h, 1 - h, mode="full")
        c5 = np.max(cor) * (2.0 / n_points)
        return c5
    
    # Define the constraint for the integral
    def integral_constraint(x):
        return np.sum(x) - n_points / 2
    
    # Optimization parameters
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': integral_constraint}]
    
    # Initial optimization with SLSQP
    result = optimize.minimize(objective, best_h, method='SLSQP', bounds=bounds, constraints=constraints)
    best_h = result.x
    cor = correlate(best_h, 1 - best_h, mode="full")
    best_c5 = np.max(cor) * (2.0 / n_points)
    
    # Refinement with differential evolution
    for _ in range(3):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Create a new population with random perturbations
        population = np.random.rand(10, n_points)
        population = np.clip(population, 0, 1)
        population = population * (n_points / 2) / np.sum(population, axis=1, keepdims=True)
        
        # Evaluate the population
        population_c5 = np.zeros(10)
        for i in range(10):
            cor = correlate(population[i], 1 - population[i], mode="full")
            population_c5[i] = np.max(cor) * (2.0 / n_points)
        
        # Select the best individual from the population
        best_index = np.argmin(population_c5)
        best_h = population[best_index]
        best_c5 = population_c5[best_index]
    
    # Final refinement with local optimization
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.01, n_points)
        perturbed_h = best_h + perturbation
        perturbed_h = np.clip(perturbed_h, 0, 1)
        
        # Optimize the perturbed solution
        result = optimize.minimize(objective, perturbed_h, method='SLSQP', bounds=bounds, constraints=constraints)
        perturbed_h = result.x
        
        # Compute the C5 value for the perturbed solution
        cor = correlate(perturbed_h, 1 - perturbed_h, mode="full")
        c5 = np.max(cor) * (2.0 / n_points)
        
        # Update best solution if improvement is found
        if c5 < best_c5:
            best_c5 = c5
            best_h = perturbed_h
    
    return best_h, best_c5, n_points
```
