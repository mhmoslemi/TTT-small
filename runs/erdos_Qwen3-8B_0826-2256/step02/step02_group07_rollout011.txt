def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Start with a medium number of points
    n_points = 180
    h_initial = np.copy(initial_h_values[:n_points])
    h_initial = np.random.uniform(0.95, 1.05, size=n_points)
    h_initial = np.clip(h_initial, 0, 1)
    h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
    
    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = signal.correlate(h, 1 - h, mode="full")
        return np.max(corr * (2.0 / n_points))
    
    # Define the constraint
    def constraints(h):
        h = np.clip(h, 0, 1)
        return np.sum(h) - n_points / 2
    
    # Optimize using SLSQP
    def optimize_h(h_initial):
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Objective function
        def objective_func(x):
            return objective(x)
        
        # Constraint function
        def constraint_func(x):
            return constraints(x)
        
        # Setup optimization
        cons = [{'type': 'eq', 'fun': constraint_func}]
        result = optimize.minimize(objective_func, h_initial, bounds=bounds, constraints=cons, method='SLSQP', tol=1e-6)
        
        # Ensure valid solution
        h = np.clip(result.x, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h, objective_func(h)
    
    # Start with initial optimization
    start_time = time.time()
    best_h = h_initial
    best_c5 = objective(best_h)
    
    # Run optimization
    try:
        h, c5 = optimize_h(h_initial)
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
    except:
        pass
    
    # Try increasing n_points if time allows
    if time.time() - start_time < budget_s - 10:
        n_points = 200
        h_initial = np.random.uniform(0.95, 1.05, size=n_points)
        h_initial = np.clip(h_initial, 0, 1)
        h_initial = h_initial * (n_points / 2) / np.sum(h_initial)
        
        # Define new objective and constraint
        def objective_new(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = signal.correlate(h, 1 - h, mode="full")
            return np.max(corr * (2.0 / n_points))
        
        def constraints_new(h):
            h = np.clip(h, 0, 1)
            return np.sum(h) - n_points / 2
        
        def optimize_h_new(h_initial):
            bounds = [(0, 1) for _ in range(n_points)]
            
            def objective_func_new(x):
                return objective_new(x)
            
            def constraint_func_new(x):
                return constraints_new(x)
            
            cons_new = [{'type': 'eq', 'fun': constraint_func_new}]
            result_new = optimize.minimize(objective_func_new, h_initial, bounds=bounds, constraints=cons_new, method='SLSQP', tol=1e-6)
            
            h_new = np.clip(result_new.x, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            return h_new, objective_func_new(h_new)
        
        try:
            h_new, c5_new = optimize_h_new(h_initial)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
        except:
            pass
    
    return best_h, best_c5, n_points
