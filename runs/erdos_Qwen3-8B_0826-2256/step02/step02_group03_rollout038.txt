    # Function to compute the overlap integral
    def compute_overlap(h):
        overlap = correlate(h, 1 - h, mode='full')
        return np.max(overlap) * dx
    
    # Function to compute the objective
    def objective(h_values):
        return compute_overlap(h_values)
    
    # Constraint: sum(h) * dx = 1
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
    
    # Bounds for h_values
    bounds = Bounds(0, 1)

    h = h_initial

    best_h = h.copy()
    best_c5 = compute_overlap(h)

        result = minimize(
