```python
import numpy as np
import scipy.optimize as optimize
import time
from scipy.optimize import Bounds, minimize
from scipy import linalg
from scipy.sparse import csr_matrix
from scipy.sparse.linalg import minres

def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)
    
    # Try different numbers of points to find a balance between speed and accuracy
    n_points_candidates = [100, 200, 300]
    best_c5 = float('inf')
    best_h = None
    best_n_points = 0
    
    for n_points in n_points_candidates:
        dx = 2.0 / n_points
        
        # Initialize h with a simple alternating pattern
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 if i % 3 == 0 else 0.0
        # Adjust to ensure the integral is 1
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        
        # Objective function to minimize the maximum overlap
        def objective(h_values):
            h = h_values.reshape(-1)
            overlap = np.correlate(h, 1 - h, mode="full")
            c5 = np.max(overlap) * dx
            return c5
        
        # Constraint to ensure the integral is 1
        def constraints(h_values):
            return np.sum(h_values) - n_points / 2
        
        bounds = Bounds(0, 1)
        start_time = time.time()
        trials = 150  # Increase number of trials for better exploration
        
        best_h_trial = h.copy()
        best_c5_trial = objective(h)
        
        for trial in range(trials):
            if time.time() - start_time > budget_s - 1:
                break
            # Perturb the current best solution with decreasing magnitude
            perturbation = np.random.uniform(-0.02 * (1 - trial / trials), 0.02 * (1 - trial / trials), size=n_points)
            h = best_h_trial + perturbation
            h = np.clip(h, 0, 1)
            # Ensure the integral is 1
            h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            x0 = h
            
            # Run a combination of local and global optimization methods
            try:
                # Local refinement using SLSQP with tighter tolerances
                result_slsqp = minimize(
                    objective,
                    x0,
                    bounds=bounds,
                    constraints={"type": "eq", "fun": constraints},
                    method="SLSQP",
                    options={"ftol": 1e-9, "maxiter": 150}
                )
                if result_slsqp.success:
                    current_c5 = result_slsqp.fun
                    if current_c5 < best_c5_trial:
                        best_c5_trial = current_c5
                        best_h_trial = result_slsqp.x
                
                # Use COBYLA for global exploration
                result_cobyla = minimize(
                    objective,
                    x0,
                    bounds=bounds,
                    constraints={"type": "eq", "fun": constraints},
                    method="COBYLA",
                    options={"maxiter": 150}
                )
                if result_cobyla.success:
                    current_c5 = result_cobyla.fun
                    if current_c5 < best_c5_trial:
                        best_c5_trial = current_c5
                        best_h_trial = result_cobyla.x
            except:
                # Gracefully handle exceptions
                pass
        
        if best_c5_trial < best_c5:
            best_c5 = best_c5_trial
            best_h = best_h_trial
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
```
