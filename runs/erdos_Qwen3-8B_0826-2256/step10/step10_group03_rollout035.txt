def run(seed=42, budget_s=100, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points

    # Use initial_h_values if available

        h = initial_h_values.copy()
    else:
        h = np.random.rand(n_points)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
    
    def objective(h_values):
        h = h_values.reshape(-1)
        overlap = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(overlap) * dx
        return c5

    def constraint(h_values):
        return np.sum(h_values) - n_points / 2

    bounds = Bounds(0, 1)
    best_h = h
    best_c5 = objective(h)
    start_time = time.time()
    trials = 50  # Reduce number of trials to improve performance

    for _ in range(trials):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Perturb current best with decreasing magnitude
        perturbation = np.random.uniform(-0.02, 0.02, size=n_points)
        h = best_h + perturbation
        h = np.clip(h, 0, 1)
        h = h / (np.sum(h) * dx) if np.sum(h) != 0 else np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        x0 = h

        # Local optimization with SLSQP
        result_slsqp = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="SLSQP",
            options={"ftol": 1e-9, "maxiter": 100}
        )
        if result_slsqp.success:
            current_c5 = result_slsqp.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_slsqp.x

        # Global exploration with Differential Evolution
        result_de = minimize(
            objective,
            x0,
            bounds=bounds,
            constraints={"type": "eq", "fun": constraint},
            method="L-BFGS-B",
            options={"maxiter": 50}
        )
        if result_de.success:
            current_c5 = result_de.fun
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = result_de.x
    
    return best_h, best_c5, n_points
