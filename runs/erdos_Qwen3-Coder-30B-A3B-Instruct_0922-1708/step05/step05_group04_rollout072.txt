import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Convert initial h to Fourier coefficients
    h_ff = fft(h_init)
    fourier_coeffs = np.concatenate([np.real(h_ff[:n_fourier]), np.imag(h_ff[:n_fourier])])
    
    # Objective function
    def objective(coeffs):
        # Reconstruct h from Fourier coefficients
        coeffs_real = coeffs[:n_fourier]
        coeffs_imag = coeffs[n_fourier:n_fourier*2]
        h_ff = coeffs_real + 1j * coeffs_imag
        h_ff = np.concatenate([h_ff, np.conj(h_ff[-2:0:-1])])
        h = np.real(ifft(h_ff)).real
        
        # Clip to feasible range
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        return c5
    
    # Constraint function for sum
    def constraint_sum(coeffs):
        # Reconstruct h from Fourier coefficients
        coeffs_real = coeffs[:n_fourier]
        coeffs_imag = coeffs[n_fourier:n_fourier*2]
        h_ff = coeffs_real + 1j * coeffs_imag
        h_ff = np.concatenate([h_ff, np.conj(h_ff[-2:0:-1])])
        h = np.real(ifft(h_ff)).real
        
        # Clip to feasible range
        h = np.clip(h, 0, 1)
        
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return np.sum(h) - n_points / 2
    
    # Gradient computation
    def compute_gradient(coeffs, eps=1e-5):
        grad = np.zeros_like(coeffs)
        base_val = objective(coeffs)
        
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            val_plus = objective(coeffs_plus)
            grad[i] = (val_plus - base_val) / eps
            
        return grad
    
    # Optimize
    bounds = [(-10, 10)] * (n_fourier * 2)  # Reasonable bounds for coefficients
    
    # Initial guess
    x0 = fourier_coeffs.copy()
    
    # Clip bounds to reasonable range
    bounds = [(-5, 5)] * len(x0)
    
    # Flatten bounds for scipy
    bounds_start = [b[0] for b in bounds]
    bounds_end = [b[1] for b in bounds]
    
    # L-BFGS-B optimization
    best_c5 = float('inf')
    best_coeffs = x0.copy()
    stagnation_count = 0
    
    # Warm start with initial solution if available
    if 'initial_h_values' in globals():
        test_c5 = objective(x0)
        if test_c5 < best_c5:
            best_c5 = test_c5
            best_coeffs = x0.copy()
    
    for iteration in range(500):
        if time.time() - start_time > budget_s - 5:

        # Compute gradient
        try:
            grad = compute_gradient(x0)

            # Fallback: random search step
            x0 += np.random.normal(0, 0.01, len(x0))
            continue
        
        # Update using gradient descent
        learning_rate = 0.01
        x0 -= learning_rate * grad
        
        # Apply bounds (simple clipping)
        for i in range(len(x0)):
            x0[i] = max(bounds_start[i], min(bounds_end[i], x0[i]))
        
        # Evaluate current solution
        try:
            current_c5 = objective(x0)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_coeffs = x0.copy()
                stagnation_count = 0
            else:
                stagnation_count += 1
        except:
            # If evaluation fails, continue
            continue
        
        # Early termination
        if stagnation_count > 100:
            break
    
    # Convert best coefficients back to h
    coeffs_real = best_coeffs[:n_fourier]
    coeffs_imag = best_coeffs[n_fourier:n_fourier*2]
    h_ff = coeffs_real + 1j * coeffs_imag
    h_ff = np.concatenate([h_ff, np.conj(h_ff[-2:0:-1])])
    h_opt = np.real(ifft(h_ff)).real
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    final_c5 = best_c5
    
    # Final validation
    final_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    final_c5_valid = np.max(final_corr) * dx
    
    return (h_opt.tolist(), final_c5_valid, n_points)
