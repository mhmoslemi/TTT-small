import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize randomly with constraint
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # CS parameters
    alpha = 0.25  # Discovery probability
    epsilon = 1e-5  # Small perturbation
    max_iter = 1000
    stagnation_limit = 200
    
    # Initialize best
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Cuckoo population
    population_size = 20
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_in_pop = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(best_in_pop)
    if best_in_pop[best_idx] < best_value:
        best_h = population[best_idx].copy()
        best_value = best_in_pop[best_idx]
    
    # Track stagnation
    last_improvement = 0
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution via Lévy flight
        new_h = best_h.copy()
        # Select random element to perturb
        idx = np.random.randint(n_points)
        # Add Lévy perturbation
        sigma = 1.0
        step = np.random.normal(0, sigma)
        new_h[idx] += step * epsilon
        # Repair solution
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(new_h, dx)
        
        # Accept or reject according to CS rules
        if new_value < best_value:
            best_h = new_h.copy()
            best_value = new_value
            last_improvement = iteration
            stagnation_count = 0
        else:
            # Probabilistic acceptance
            if np.random.rand() < alpha:
                population[np.random.randint(population_size)] = new_h
                best_in_pop[np.random.randint(population_size)] = new_value
                # Update global best if needed
                if new_value < best_value:
                    best_h = new_h.copy()
                    best_value = new_value
                    last_improvement = iteration
                    stagnation_count = 0

        # Local search on top few solutions
        if iteration % 30 == 0:
            top_k = min(5, len(population))
            indices = np.argsort(best_in_pop)[:top_k]
            for idx in indices:
                h = population[idx].copy()
                current_eval = best_in_pop[idx]
                for _ in range(50):
                    if time.time() - start_time > budget_s - 10:
                        break
                    eps = 1e-6
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - current_eval) / eps

                    h = h - 0.001 * grad
                    h = repair_h(h, n_points)
                    corr = correlate(h, 1 - h, mode='full')
                    current_eval = np.max(corr) * dx
                    if current_eval < best_in_pop[idx]:
                        population[idx] = h.copy()
                        best_in_pop[idx] = current_eval
                        
                        if current_eval < best_value:
                            best_h = h.copy()
                            best_value = current_eval
                            last_improvement = iteration
                            stagnation_count = 0

        # Update stagnation counter
        if iteration - last_improvement > 0:
            stagnation_count += 1
        if stagnation_count > stagnation_limit:
            break
    
    # Final gradient descent refinement
    best_h_refined = best_h.copy()
    best_c5_refined = best_value
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps

        best_h_refined = best_h_refined - 0.0005 * grad
        best_h_refined = repair_h(best_h_refined, n_points)
        corr = correlate(best_h_refined, 1 - best_h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
    
    return (best_h_refined.tolist(), best_c5_refined, n_points)
