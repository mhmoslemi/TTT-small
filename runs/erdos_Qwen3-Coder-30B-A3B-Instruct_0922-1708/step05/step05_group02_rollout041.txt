import numpy as np
import time
from scipy.signal import correlate
from scipy.stats import levy

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_perturb(h, scale=0.5, alpha=1.5):
    """Apply Lévy flight perturbation."""
    n_points = len(h)
    perturbation = levy.rvs(scale=scale, size=n_points, random_state=None)
    new_h = h + perturbation
    return new_h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population
    population_size = 50
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track fitness
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    max_iter = 1000
    levy_scale = 0.5
    levy_alpha = 1.5
    learning_rate = 0.001
    top_candidates = 10
    patience = 100
    
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new candidates with Lévy flight
        new_population = []
        for h in population:
            new_h = levy_perturb(h, scale=levy_scale, alpha=levy_alpha)
            new_h = repair_h(new_h, n_points)
            new_population.append(new_h)
        
        # Evaluate new candidates
        new_fitness = [evaluate_h(h, dx) for h in new_population]
        
        # Combine and select top candidates
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Select top candidates for gradient descent
        sort_idx = np.argsort(combined_fitness)
        selected_indices = sort_idx[:top_candidates]
        
        # Apply gradient descent to selected candidates
        for idx in selected_indices:
            h = combined_population[idx].copy()
            current_fitness = combined_fitness[idx]
            
            # Compute gradient
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_fitness) / eps
            
            # Update and repair
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            
            # Re-evaluate
            corr = correlate(h, 1 - h, mode='full')
            new_fitness_val = np.max(corr) * dx
            
            if new_fitness_val < current_fitness:
                combined_population[idx] = h
                combined_fitness[idx] = new_fitness_val
                
        # Update population and fitness
        population = combined_population
        fitness = combined_fitness
        
        # Update global best
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_fitness:
            best_h = population[best_idx].copy()
            best_fitness = fitness[best_idx]
            last_improvement = iteration
            
        # Early stopping
        if iteration - last_improvement > patience:
            break
    
    # Final refinement with gradient descent on best solution
    best_h_refined = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')

            grad[j] = (c5_plus - best_fitness) / eps
            
        best_h_refined = best_h_refined - learning_rate * grad
        best_h_refined = repair_h(best_h_refined, n_points)
        corr = correlate(best_h_refined, 1 - best_h_refined, mode='full')
        best_fitness = np.max(corr) * dx
        
    return (best_h_refined.tolist(), best_fitness, n_points)
