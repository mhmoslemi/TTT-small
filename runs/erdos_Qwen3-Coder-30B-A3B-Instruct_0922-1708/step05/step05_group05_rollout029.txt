import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize with the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: rectangular block
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    h = h * (n_points / 2) / np.sum(h)
    
    # Project h to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    # Objective function: compute C5 bound
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Constraint function: sum should be 1
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Gradient computation via finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_obj = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project h_plus back to feasible set
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            obj_plus = objective(h_plus)
            grad[i] = (obj_plus - base_obj) / eps
        return grad
    
    # Project point onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main optimization loop with projected gradient descent
    alpha = 0.01
    best_h = h.copy()
    best_c5 = objective(h)
    num_iter = 0
    
    while num_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Add small noise to escape local minima
        if num_iter % 50 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = project(h)
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Compute candidate point
        h_candidate = h - alpha * grad
        h_candidate = project(h_candidate)
        
        # Evaluate candidate
        c5_candidate = objective(h_candidate)
        
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            
        # Accept or reject
        if c5_candidate < objective(h):
            h = h_candidate
            alpha *= 1.1  # Increase step size if successful
        else:
            alpha *= 0.5  # Decrease step size if failed
            
        # Ensure step size doesn't become too small
        alpha = max(alpha, 1e-6)
        
        num_iter += 1
    
    # Final check of feasibility and bounds
    final_h = project(best_h)
    final_c5 = objective(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
