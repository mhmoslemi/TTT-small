import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        correlation = np.correlate(h_parent, 1 - h_parent, mode='full')
        c5_parent = np.max(correlation) * dx
        if c5_parent < best_c5:
            best_c5 = c5_parent
            best_h = h_parent.copy()
    
    # Scale pyramid: [10, 20, 40, 80, 100]
    scales = [10, 20, 40, 80, 100]
    
    h_prev = None
    for scale in scales:
        n = scale
        dx_scale = 2.0 / n
        if h_prev is None:
            h = np.ones(n) * 0.5
        else:
            # Upsample h_prev to current scale with linear interpolation
            x_prev = np.linspace(0, 2, len(h_prev))
            x_new = np.linspace(0, 2, n)
            h = np.interp(x_new, x_prev, h_prev)
        
        # Smoothing parameters
        lambda_ = 0.1
        
        # Solve relaxed problem using CVXPY
        h_var = Variable(n)
        # Objective: minimize max of lags within 1/4 of the domain
        max_lag = n // 4
        warm = []
        for k in range(1, max_lag + 1):
            # Approximate cross-correlation as linear sum
            corr_term = 0
            for i in range(n - k):
                corr_term += h_var[i] * (1 - h_var[i + k])
            warm.append(corr_term)
        
        # Soft constraint handling
        constraints = [h_var >= 0, h_var <= 1]
        constraints += [sum(h_var) == n / 2]
        
        # Add soft penalty for constraints
        penalty = 0
        for i in range(n):
            penalty += -lambda_ * (h_var[i] * (1 - h_var[i]))
        
        objective = Minimize(sum(warm) / len(warm) + penalty)
        prob = Problem(objective, constraints)
        
        # Solve with ECOS
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status == 'optimal':
                h_solution = np.array(h_var.value).reshape(-1)
                # Clip and normalize
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (n / 2) / np.sum(h_solution)
                h = h_solution
        except:
            pass
        
        # Refinement step with gradient descent
        def compute_c5_and_grad(h, eps=1e-4):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx_scale
            grad = np.zeros_like(h)
            for i in range(n):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx_scale
                grad[i] = (c5_plus - c5) / eps
            return c5, grad
        
        # Refine for 50 steps
        for _ in range(50):
            c5, grad = compute_c5_and_grad(h)
            h = h - 0.001 * grad
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            # Update best if improved
            corr = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(corr) * dx_scale
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        h_prev = h.copy()
    
    # Final local optimization on best_h if exists
    if best_h is not None:
        h = best_h.copy()
        def compute_c5_and_grad_final(h, eps=1e-4):
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            return c5, grad
        
        # Gradient descent for 100 steps
        for _ in range(100):
            c5, grad = compute_c5_and_grad_final(h)
            h = h - 0.0005 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = np.correlate(h, 1 - h, mode='full')
            c5_new = np.max(corr) * dx
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    if best_h is None:
        if 'initial_h_values' in globals():
            h = h_parent
        else:
            h = np.ones(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
        best_h = h
    
    return (best_h.tolist(), best_c5, n_points)
