import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_generations = 1500
    mu = 0.5
    sigma = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Create population around initial solution
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Create random population
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for fitness evaluation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Scale to meet sum constraint
        h = h * (n_points / 2) / h_sum
        return h
    
    # Evolutionary strategy main loop
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    
    generation = 0
    while generation < max_generations and time.time() - start_time < budget_s - 10:
        
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_fitness(p) for p in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Selection: keep top 50% of population
        selected_indices = np.argsort(fitnesses)[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent = selected_population[np.random.randint(0, len(selected_population))]
            # Gaussian mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mu * noise
            child = repair(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = np.argsort(fitnesses)[-offspring_size:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Update mutation parameters
        if generation % 100 == 0:
            mu *= 0.9
            if mu < 0.01:
                mu = 0.01
        if mu < 0.02:
            sigma = mu / 5
        else:
            mu = 0.5
            sigma = 0.1
        
        # Restart strategy if stagnation
        if stagnation_count > 200:
            # Restart by reinitializing with better solution if available
            if best_individual is not None:
                for i in range(population_size // 4):
                    population[i] = best_individual + np.random.normal(0, 0.05, n_points)
                    population[i] = repair(population[i])
            stagnation_count = 0
            
        generation += 1
    
    # Final refinement using gradient descent
    if best_individual is not None:
        h_refined = best_individual.copy()
        
        for _ in range(300):
            # Estimate gradient
            grad = np.zeros(n_points)
            eps = 1e-4
            base_fitness = evaluate_fitness(h_refined)
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                fitness_plus = evaluate_fitness(h_plus)
                grad[i] = (fitness_plus - base_fitness) / eps
            
            # Update
            h_refined -= 0.001 * grad
            h_refined = repair(h_refined)
            
            # Check for convergence
            current_fitness = evaluate_fitness(h_refined)
            if current_fitness >= base_fitness:
                break
                
        final_fitness = evaluate_fitness(h_refined)
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_individual = h_refined.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
