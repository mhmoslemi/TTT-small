import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Penalty parameters
    penalty_weight = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    learning_rate = 1e-3
    learning_rate_decay = 0.5
    plateau_patience = 200
    perturbation_fraction = 0.01
    max_iterations = 5000
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization
    iteration = 0
    no_improvement_count = 0
    current_penalty = penalty_weight
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlation and find maximum overlap
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_val = corr[max_corr_idx]
        c5 = max_corr_val * dx
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive penalty
        sum_violation = np.abs(np.sum(h) - n_points / 2)
        bound_violation = np.sum(np.maximum(0, -h)) + np.sum(np.maximum(0, h - 1))
        total_violation = sum_violation + bound_violation
        
        if total_violation > 1e-8:
            current_penalty = max(current_penalty * penalty_update_factor, penalty_min)
        else:
            current_penalty = max(current_penalty / penalty_update_factor, penalty_min)
            
        # Compute gradient
        grad = np.zeros(n_points)
        
        # Gradient of main objective
        # Find lag k that gives max overlap
        k = max_corr_idx - (n_points - 1)
        
        # Construct lazy gradient for the identified correlation
        for i in range(n_points):
            # Derivative of h[i] * (1 - h[i + k]) for k < 0 is -h[i+k]
            # For k > 0 it's -h[i-k]  
            # For k = 0, it's 1 - 2 * h[i]
            if k > 0:
                if i + k < n_points:
                    grad[i] -= h[i + k]
            elif k < 0:
                if i - k < n_points:
                    grad[i] -= h[i - k]
            else:
                grad[i] -= 1 - 2 * h[i]
        
        # Gradient of penalty
        penalty_grad = np.zeros_like(h)
        penalty_grad += 2 * current_penalty * (np.sum(h) - n_points / 2) / n_points
        penalty_grad += 2 * current_penalty * np.maximum(0, -h)
        penalty_grad -= 2 * current_penalty * np.maximum(0, h - 1)
        
        # Combine gradients
        full_grad = grad + penalty_grad
        
        # Normalize gradient for step size scaling
        grad_norm = np.linalg.norm(full_grad)
        if grad_norm > 0:
            full_grad = full_grad / grad_norm
        
        # Line search
        step_size = learning_rate
        h_new = h - step_size * full_grad
        
        # Apply feasibility constraints and rescale
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum - n_points / 2) > 1e-8:
            h_new *= (n_points / 2) / h_sum
            
        # Check if the new point improves the objective
        h_check = h_new.copy()
        corr_check = correlate(h_check, 1 - h_check, mode='full')
        c5_check = np.max(corr_check) * dx

        # If it's better, accept and continue
        if c5_check < c5:
            h = h_new.copy()
        else:
            # If not, try to decrease learning rate or perturb
            if no_improvement_count >= plateau_patience:
                learning_rate *= learning_rate_decay
                # Apply small random perturbations
                perturb = np.random.normal(0, perturbation_fraction, n_points)
                h += perturb
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                
    # Final validation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
