import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with either initial_h_values or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    alpha = 0.25  # Discovery rate
    epsilon = 1e-5  # Small perturbation
    max_iter = 1000
    no_improvement_limit = 200
    
    # Track best solution
    no_improvement_count = 0
    best_global = best_h.copy()
    best_global_c5 = best_c5
    
    # Main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Generate new solution via random walk (cuckoo)
        new_h = best_h.copy()
        # Perturb about 20% of elements
        num_perturb = max(1, int(0.2 * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            # Add small random perturbation
            new_h[idx] += np.random.normal(0, 0.01)
        
        # Repair solution to satisfy constraints
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept new solution if better, or accept probabilistically
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            if new_c5 < best_global_c5:
                best_global = new_h.copy()
                best_global_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            # With probability, accept worse solution
            r = np.random.rand()
            if r < alpha:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count += 1
        
        # Local search refinement (gradient descent on top 20% solutions)
        if iter_count % 50 == 0:
            # Apply a simple local search
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                eps = epsilon
                for j in range(n_points):
                    h_plus = best_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / eps
                
                # Update using gradient
                best_h = best_h - 0.0001 * grad
                best_h = repair_h(best_h, n_points)
                corr = correlate(best_h, 1 - best_h, mode='full')
                best_c5 = np.max(corr) * dx
                
                if best_c5 < best_global_c5:
                    best_global = best_h.copy()
                    best_global_c5 = best_c5
                    no_improvement_count = 0
        
        # Early stopping if no improvement
        if no_improvement_count > no_improvement_limit:
            break
    
    # Final intensive refinement on best solution
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        if best_c5 < best_global_c5:
            best_global = best_h.copy()
            best_global_c5 = best_c5
    
    return (best_global.tolist(), best_global_c5, n_points)
