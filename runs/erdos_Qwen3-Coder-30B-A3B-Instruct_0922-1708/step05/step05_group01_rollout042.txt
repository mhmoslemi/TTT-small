import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize z (binary decisions)
    z = (h > 0.5).astype(int)
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 0.001
    mu = 0.1
    mu_factor = 1.2
    perturb_frac = 0.01
    best_h = h.copy()
    best_c5 = float('inf')
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Initial evaluation
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current max lag
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        # Compute gradient w.r.t. h for the max lag
        # For max_lag k, the gradient for h[i] is (1 - h[i-k] - h[i])
        # But we need to compute partial derivatives correctly
        # Note: Using simple approximation (discrete derivative)
        grad = np.zeros(n_points)
        eps = 1e-6
        
        # We'll compute it via finite differences along one dimension
        # This is approximated using direct computation of the correlation
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        
        # Clip h_new to feasible range
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z based on new h
        z_new = (h_new > 0.5).astype(int)
        
        # Simple penalty term for intermediate values of h
        penalty = mu * np.sum(np.abs(h_new - z_new))
        
        # Recompute actual C5 with adjusted h
        c5_new = compute_c5(h_new) + penalty
        
        # Accept if improvement
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            z = z_new
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Occasionally perturb to escape local minimum
            if np.random.rand() < perturb_frac:
                h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                c5 = compute_c5(h)
                z = (h > 0.5).astype(int)
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
        
        # Increase penalty parameter
        mu *= mu_factor
    
    return (best_h.tolist(), best_c5, n_points)
