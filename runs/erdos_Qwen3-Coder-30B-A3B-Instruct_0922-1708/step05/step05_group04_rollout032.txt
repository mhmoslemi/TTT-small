import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 5000
    initial_mu = 0.5
    mu_decay = 0.999
    mutation_threshold = 100
    restart_threshold = 200
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Add small perturbations around initial solution
        for _ in range(population_size):
            h = h_init + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for constraint repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Scale to meet sum constraint
        h = h * (n_points / 2) / h_sum
        return h
    
    # Fitness evaluation
    def evaluate_fitness(h):
        # Compute all correlations using np.correlate
        corr = np.correlate(h, 1 - h, mode='full')
        # Return maximum correlation (scaled by dx)
        return np.max(corr) * dx
    
    # Evolution Strategy main loop
    current_mu = initial_mu
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate fitness of all individuals
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best individual
        min_fitness_idx = np.argmin(fitnesses)
        min_fitness = fitnesses[min_fitness_idx]
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        selected_indices = sorted_indices[:population_size // 2]
        
        # Create offspring
        offspring = []
        sigma = current_mu / 5.0
        
        for _ in range(offspring_size):
            # Select parent from top 50%
            parent_idx = np.random.choice(selected_indices)
            parent = population[parent_idx]
            
            # Gaussian mutation
            mutation = np.random.normal(0, sigma, n_points)
            child = parent + current_mu * mutation
            
            # Repair child
            child = repair(child)
            
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[population_size // 2:]
        for i, (worst_idx, child) in enumerate(zip(worst_indices, offspring)):
            population[worst_idx] = child
        
        # Update mutation parameters
        current_mu *= mu_decay
        
        # Restart mechanism
        if iteration % mutation_threshold == 0:
            current_mu *= 0.9
            
        if stagnation_count > restart_threshold:
            # Reinitialize population with better diversity
            for i in range(population_size):
                if np.random.random() < 0.3:  # 30% chance to reset
                    h = np.random.uniform(0, 1, n_points)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
                    population[i] = h
            current_mu = initial_mu
            stagnation_count = 0
        
        iteration += 1
    
    # Final refinement with gradient descent
    h_refined = best_individual.copy()
    
    # Wrap objective to use in scipy.optimize
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h):
        # Numerical gradient estimation
        eps = 1e-5
        grad = np.zeros_like(h)
        base_value = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            grad[i] = (objective(h_plus) - base_value) / eps
        return grad
    
    # Gradient descent for refinement
    max_refinement_steps = 100
    lr = 0.001
    previous_loss = objective(h_refined)
    
    for _ in range(max_refinement_steps):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = gradient(h_refined)
        h_refined -= lr * grad
        h_refined = repair(h_refined)
        
        new_loss = objective(h_refined)
        if new_loss >= previous_loss:
            break
        previous_loss = new_loss
    
    # Final verification
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
