import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current maximum overlap
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        
        # Compute gradient (approximate)
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute gradient for the maximum overlap term
        max_k = np.argmax(corr)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus) * dx
            
            grad[i] = (max_plus - max_overlap) / eps
        
        # Update h with gradient descent
        h = h - learning_rate * grad
        
        # Project onto feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbations
        if iteration % perturbation_freq == 0:
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            perturbation_magnitude *= 0.99  # Decay perturbation
            
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
