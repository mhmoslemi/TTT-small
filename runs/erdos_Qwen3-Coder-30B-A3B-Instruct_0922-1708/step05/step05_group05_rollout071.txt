import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_region_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    max_c5_improvement = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_bound
    
    def compute_gradient(h, k):
        """Compute gradient for overlap at lag k"""
        grad = np.zeros_like(h)
        # Gradient w.r.t. h[i]
        for i in range(n_points):
            # d/dh[i] (h[i] * (1 - h[(i+k)%n])) 
            grad[i] = (1 - h[(i - k) % n_points]) + (1 - h[(i + k) % n_points])
        return grad / n_points
    
    def project_to_feasible(h):
        """Ensure h is in [0,1] and sum = n/2"""
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    iteration = 0
    success_count = 0
    fail_count = 0
    
    while time.time() - start_time < budget_s - 10 and iteration < max_iterations:
        # Compute all overlaps and find dominant lag
        all_corr = []
        for k in range(2 * n_points - 1):
            corr = np.correlate(h, 1 - np.roll(h, k), mode='full')[n_points-1:]
            all_corr.append(np.max(corr) * dx)
        
        max_k = np.argmax(all_corr)
        max_value = all_corr[max_k]
        
        # Compute gradient at dominant lag
        grad = compute_gradient(h, max_k)
        
        # Compute proposed step
        step = -trust_region_radius * grad
        
        # Project step to feasible region
        h_new = h + step
        h_new = project_to_feasible(h_new)
        
        # Evaluate new C5
        c5_new = compute_c5(h_new)
        
        # Accept or reject based on improvement
        if c5_new < c5_bound - max_c5_improvement:
            h = h_new
            c5_bound = c5_new
            success_count += 1
            fail_count = 0
            trust_region_radius = min(0.1, trust_region_radius * 1.2)
        else:
            fail_count += 1
            trust_region_radius = max(min_trust_radius, trust_region_radius * 0.8)
        
        # Apply simulated annealing perturbations
        if iteration % perturb_freq == 0 and iteration > 0:
            # Cooling temperature
            temperature *= cooling_rate
            
            # Perturb with SA probability
            for i in range(n_points):
                if np.random.rand() < 0.3:  # 30% chance to perturb
                    # Perturbation based on temperature
                    perturbation = temperature * (1.0 - h[i]) * h[i]
                    delta = np.random.uniform(-perturbation, perturbation)
                    
                    h[i] += delta
                    h[i] = np.clip(h[i], 0, 1)
            
            # Renormalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate if improvement
            c5_new_perturb = compute_c5(h)
            if c5_new_perturb < c5_bound:
                c5_bound = c5_new_perturb
                best_h = h.copy()
                best_c5 = c5_bound
            else:
                # Restore from best if needed
                h = best_h.copy()
                c5_bound = best_c5

        iteration += 1
        
        # Update best solution if necessary
        if c5_bound < best_c5:
            best_h = h.copy()
            best_c5 = c5_bound
    
    # Final intensive optimization
    h_opt = best_h.copy()
    c5_final = best_c5
    for _ in range(500):
        # Compute gradient  
        all_corr = []
        for k in range(2 * n_points - 1):
            corr = np.correlate(h_opt, 1 - np.roll(h_opt, k), mode='full')[n_points-1:]
            all_corr.append(np.max(corr) * dx)
        
        max_k = np.argmax(all_corr)
        
        # Gradients
        grad = compute_gradient(h_opt, max_k)
        
        # Update with optimization step
        step = -0.001 * grad
        h_new = h_opt + step
        h_new = project_to_feasible(h_new)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < c5_final:
            h_opt = h_new
            c5_final = c5_new
        else:
            break
    
    # Final check
    final_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_opt.tolist(), final_c5, n_points)
