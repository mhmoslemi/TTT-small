import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_time = budget_s - 10  # Leave 10s for cleanup
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    temp_start = 10.0
    temp_min = 1e-8
    start_time = time.time()
    
    iteration = 0
    
    # Temperature schedule: T = T0 / ln(1 + iteration)
    def get_temperature(iteration):
        return temp_start / np.log(1 + iteration)
    
    # Feasibility projection
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main Simulated Annealing loop
    while time.time() - start_time < max_time:
        iteration += 1
        temp = get_temperature(iteration)
        
        if temp < temp_min:
            break
            
        # Generate a candidate by perturbing current solution
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        h_candidate = project_h(h_candidate)
        
        # Evaluate
        c5_candidate = compute_c5(h_candidate)
        c5_current = compute_c5(h)
        
        # Accept or reject
        delta_c5 = c5_candidate - c5_current
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / temp):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
    
    # Gradient-based refinement on best solution
    h_refined = best_h.copy()
    lr = 0.001
    eps = 1e-4
    
    for _ in range(50):
        if time.time() - start_time > max_time:
            break
            
        grad = np.zeros_like(h_refined)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_refined = h_refined - lr * grad
        h_refined = project_h(h_refined)
        
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refined.copy()
    
    # Verify final solution
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
