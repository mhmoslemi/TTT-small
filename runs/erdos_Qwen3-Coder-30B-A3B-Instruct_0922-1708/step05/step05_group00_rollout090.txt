import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h 
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse grid optimization (n=50)
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    # Smooth max approximation parameter
    beta = 10.0
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    # Initialize Adam moments
    m = np.zeros(n_coarse)
    v = np.zeros(n_coarse)
    
    best_coarse_h = h_coarse.copy()
    best_c5_coarse = float('inf')
    
    # Optimization loop
    for t in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation with 1-h
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        c5 = np.max(correlation)
        
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_coarse_h = h_coarse.copy()
        
        # Compute gradient using finite differences
        grad = np.zeros(n_coarse)
        eps_fd = 1e-5
        
        for i in range(n_coarse):
            h_plus = h_coarse.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            # Renormalize
            h_plus = h_plus * (n_coarse / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - c5) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        h_coarse = h_coarse - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Upsample to fine grid
    n_fine = 200
    dx_fine = 2.0 / n_fine
    
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), best_coarse_h)
    
    # Initialize Adam moments for fine grid
    m = np.zeros(n_fine)
    v = np.zeros(n_fine)
    
    # Fine grid optimization
    best_fine_h = h_fine.copy()
    best_c5_fine = float('inf')
    
    for t in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation with 1-h
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_fine
        c5 = np.max(correlation)
        
        if c5 < best_c5_fine:
            best_c5_fine = c5
            best_fine_h = h_fine.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros(n_fine)
        eps_fd = 1e-5
        
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            # Renormalize
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - c5) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(t+1))
        v_hat = v / (1 - beta2**(t+1))
        h_fine = h_fine - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
        
        # Early stopping check
        if t > 500 and abs(best_c5_fine - c5) < 1e-8:
            break
    
    final_h = best_fine_h
    final_c5 = best_c5_fine
    
    # Ensure constraints
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_fine / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx_fine
    final_c5 = np.max(correlation)
    
    return (final_h.tolist(), final_c5, n_fine)
