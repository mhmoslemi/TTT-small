import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    # Normalize to sum = n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize best solution
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
    
    # Binary search for better C5
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    # Sample shifts for constraint evaluation
    k_samples = np.linspace(-2, 2, 50)
    
    def compute_overlap(h, k):
        # This function computes the overlap integral for shift k
        # Using the definition from the problem
        # But because we have sampled values, we approximate this 
        # with the discrete version of the correlation
        # h: [n_points] array, k: shift in [0, 2] (normalized to steps)
        k_index = int((k + 2) / (4.0 / n_points))  # k in [-2,2] mapped to [0, n_points]
        k_index = max(0, min(n_points-1, k_index))
        
        if k_index == 0:
            # Full overlap
            result = np.sum(h * h) * dx
        else:
            # Correlation shift
            h1 = h[:n_points-k_index]
            h2 = h[k_index:]
            result = np.sum(h1 * h2) * dx
        
        return result
    
    def constraint_violation(h, mu):
        # Check all k samples
        violations = []
        for k in k_samples:
            overlap = compute_overlap(h, k)
            if overlap < mu:
                violations.append(mu - overlap)
        return np.sum(violations) if violations else 0.0
    
    # Optimization parameters
    iter_count = 0
    
    # Local search with gradient descent
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        # Try gradient-based improvement on current best
        h_current = best_h.copy()
        
        # Simple gradient ascent with small step
        for step in range(100):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Approximate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            base_val = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                # Ensure it sums to n_points/2 and is in [0,1]
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                val_plus = np.max(corr_plus) * dx
                grad[i] = (val_plus - base_val) / eps
                
            # Update with negative gradient
            h_current -= 0.001 * grad
            # Repair to feasible region
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            base_val = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
        
        new_c5 = np.max(np.correlate(h_current, 1 - h_current, mode='full')) * dx
        if new_c5 < best_c5:
            best_h = h_current.copy()
            best_c5 = new_c5
            
        iter_count += 1
    
    # Return best solution
    return (best_h.tolist(), best_c5, n_points)
