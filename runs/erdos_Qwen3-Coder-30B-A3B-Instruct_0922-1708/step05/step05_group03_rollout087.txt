import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Scales: 10, 20, 40, 80, 100 points
    scales = [10, 20, 40, 80, 100]
    start_time = time.time()
    
    # Multi-scale optimization
    for i, scale in enumerate(scales):
        if time.time() - start_time > budget_s - 100:
            break
            
        # Create solution vector
        if i == 0:
            h_coarse = np.ones(scale) * 0.5
        else:
            # Interpolate from previous coarse scale
            x_coarse = np.linspace(0, 2, len(h_coarse))
            x_fine = np.linspace(0, 2, scale)
            h_coarse = np.interp(x_fine, x_coarse, h_coarse)
        
        # Convex relaxation approach for coarse solution
        h_cvx = Variable(scale)
        constraints = [0 <= h_cvx, h_cvx <= 1]
        soft_constraints = []
        mu = 0.1  # relaxation parameter
        
        # Use a subset of lags for convex relaxation (approximation)
        lags = list(range(1, scale//4))
        if lags:
            total_lag_sum = 0
            for k in lags:
                # approximate objective with average over lags
                cross_corr = 0
                for i in range(scale-k):
                    cross_corr += h_cvx[i] * (1 - h_cvx[i+k])
                total_lag_sum += cross_corr / (scale - k)
            relaxed_obj = total_lag_sum / len(lags)
            obj = Minimize(relaxed_obj)
        else:
            # fallback objective
            obj = Minimize(sum(h_cvx) * (1 - sum(h_cvx)) / scale)
            
        # Add penalty for constraint violations  
        for j in range(scale):
            soft_constraints.append(mu * (h_cvx[j] - 1)**2)
            soft_constraints.append(mu * (-h_cvx[j])**2)
        obj += sum(soft_constraints)
        
        constraints.append(sum(h_cvx) == scale/2)
        
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=500)
            if prob.status == "optimal":
                h_new = np.array(h_cvx.value).flatten()
            else:
                h_new = h_coarse
        except:
            h_new = h_coarse
            
        # Ensure feasibility of new vector
        h_new = np.clip(h_new, 0, 1)
        sum_h = np.sum(h_new)
        if abs(sum_h - scale/2) > 1e-6:
            h_new = h_new * (scale / 2) / sum_h
            
        # Downsample to target resolution if needed
        if scale == n_points:
            h = h_new
        else:
            x_fine = np.linspace(0, 2, scale)
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_fine, h_new)
        
        # Refine with gradient descent on this resolution
        # Reduced steps for efficiency
        steps = 20
        lr = 0.001
        for _ in range(steps):
            if time.time() - start_time > budget_s - 100:
                break
            grad = np.zeros_like(h)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
                
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        h_coarse = h.copy()
    
    # Final refinement and validation
    if time.time() - start_time < 50:
        # Manual gradient descent refinement
        steps = 20
        lr = 0.001
        for _ in range(steps):
            grad = np.zeros_like(best_h)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
                
            best_h = best_h - lr * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
