import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    min_alpha = 1e-6
    max_alpha = 1.0
    decay_rate = 0.95
    perturb_freq = 50
    perturb_magnitude = 0.05
    
    # Load initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            # Project constraints
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - base_c5) / eps
        
        # Update step
        h_new = h - alpha * grad
        # Project into feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject step
        if c5_new < base_c5:
            h = h_new.copy()
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
            alpha = min(alpha * 1.1, max_alpha)  # Increase step size
        else:
            alpha = max(alpha * decay_rate, min_alpha)  # Decrease step size
            
        # Periodically add perturbation to escape local minima
        if i % perturb_freq == 0 and i > 0:
            noise = np.random.normal(0, perturb_magnitude, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            if c5_pert < best_c5:
                best_h = h_perturbed.copy()
                best_c5 = c5_pert
                h = h_perturbed.copy()
    
    # Final sanity check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
