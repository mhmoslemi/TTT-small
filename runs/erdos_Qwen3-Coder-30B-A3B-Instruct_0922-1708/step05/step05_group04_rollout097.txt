import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    max_iter = 500
    learning_rate = 0.01
    eps = 1e-4
    stagnation_limit = 100
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Convert initial to Fourier coeffs (real FFT)
    h_ref = h_init.copy()
    # Compute initial C5
    corr = np.correlate(h_ref, 1 - h_ref, mode='full')
    c5_ref = np.max(corr) * dx
    
    # Fourier space representation
    # We use real FFT: coefficients are [a0, a1, b1, a2, b2, ...]
    # For n_points, we need n_fourier pairs of coefficients plus a0
    
    # Shift h to center around 0
    h_shifted = h_ref - 0.5
    h_fft = np.fft.rfft(h_shifted)
    # Initial coefficients: real part from FFT, imaginary = 0
    coeffs = np.zeros(2 * n_fourier)
    if len(h_fft) > 0:
        coeffs[0] = np.real(h_fft[0])
        for i in range(1, min(n_fourier, len(h_fft))):
            if 2*i < len(h_fft):
                coeffs[2*i-1] = np.real(h_fft[i])
                coeffs[2*i] = np.imag(h_fft[i])
    
    # Gradient function with reinitialization
    def compute_gradient(coeffs, h_ref):
        h_orig = reconstruct_h(coeffs, n_points, n_fourier)
        h = np.clip(h_orig, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        base_fitness = np.max(corr) * dx
        
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = reconstruct_h(coeffs_plus, n_points, n_fourier)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - base_fitness) / eps
            
        return grad
    
    # Reconstruct h from coefficients
    def reconstruct_h(coeffs, n_points, n_fourier):
        # Build complex FFT coefficients
        fft_coeffs = np.zeros(n_points // 2 + 1, dtype=complex)
        fft_coeffs[0] = coeffs[0]
        for i in range(1, min(n_fourier, n_points//2 + 1)):
            if 2*i < len(coeffs):
                fft_coeffs[i] = coeffs[2*i-1] + 1j*coeffs[2*i]
        # Inverse FFT to get signal
        h = np.real(ifft(fft_coeffs, n=n_points))
        return h + 0.5  # unshift
    
    # Optimization
    best_c5 = c5_ref
    best_coeffs = coeffs.copy()
    stagnation_count = 0
    
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Compute gradient
        grad = compute_gradient(coeffs, h_ref)
        # Update
        coeffs -= learning_rate * grad
        # Ensure the values don't explode
        coeffs = np.clip(coeffs, -1, 1)
        
        # Reconstruct h and compute new C5
        h = reconstruct_h(coeffs, n_points, n_fourier)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_coeffs = coeffs.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= stagnation_limit:
            # Reinitialize with perturbed coefficients
            coeffs = best_coeffs.copy() + np.random.normal(0,0.01,len(coeffs))
            stagnation_count = 0
            
        current_iter += 1
    
    # Final refinement using local search
    h_final = reconstruct_h(best_coeffs, n_points, n_fourier)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Final local search on h
    for _ in range(100):
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_final -= 0.001 * grad
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        corr_new = np.correlate(h_final, 1 - h_final, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    # Final validation and cleanup
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(corr_final) * dx
    final_h = h_final.copy()
    
    return (final_h.tolist(), c5_final, n_points)
