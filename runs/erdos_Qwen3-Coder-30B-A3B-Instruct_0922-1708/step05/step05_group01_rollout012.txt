import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial overlap
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Optimization loop
    max_iter = 6000
    alpha = 0.01
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Find the maximum overlap lag
        max_lag_idx = np.argmax(corr)
        max_val = corr[max_lag_idx]
        
        # Compute gradient-like updates
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points/2) > 1e-8:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        h = h_new
        
        # Compute new overlap
        corr_new = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adaptive learning rate
        if iteration % 200 == 0 and iteration > 0:
            alpha *= 0.5
            
        # Perturbation to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.001, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
        # Early termination if no improvement for long time
        if stagnation_count > 200:
            break
    
    # Final check
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
