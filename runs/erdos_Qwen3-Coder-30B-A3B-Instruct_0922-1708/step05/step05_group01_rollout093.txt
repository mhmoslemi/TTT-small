import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize z (binary indicator)
    z = (h > 0.5).astype(float)
    
    # Augmented Lagrangian parameters
    mu = 0.1
    max_iter = 5000
    lr = 0.001
    penalty_factor = 1.2
    perturb_fraction = 0.01
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find current maximum lag
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        max_val = corr[max_lag_idx]
        
        # Compute gradient with respect to current max lag
        # Effectively this is the gradient of (h * (1-h))[k] for the identified lag
        grad = np.zeros(n_points)
        
        # Compute overlap region: h[i] * (1 - h[i + max_lag])
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] += (1 - 2 * h[i])  # d/dh[i] (h[i] * (1 - h[j])) = 1 - 2*h[i]
            else:
                grad[i] += 0
                
        # Remove contribution from the indices that don't contribute if max_lag is out of bounds
        # Adjust gradient based on allowed index ranges
        # Apply additional penalty gradient for z constraint
        penalty_grad = 2 * mu * (h - z)
        # We can also use the gradient of z with respect to h to keep track of that
        # For now, we use a simpler approach - we just approximate all indices
        
        # Gradient step
        h_new = h - lr * (grad + penalty_grad)
        
        # Repair bounds and normalization
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            if h_sum > target_sum:
                sorted_indices = np.argsort(h_new)[::-1]
                excess = h_sum - target_sum
                for idx in sorted_indices:
                    if excess <= 0:
                        break
                    reduce_amount = min(excess, h_new[idx])
                    h_new[idx] -= reduce_amount
                    excess -= reduce_amount
            else:
                sorted_indices = np.argsort(h_new)
                deficit = target_sum - h_sum
                for idx in sorted_indices:
                    if deficit <= 0:
                        break
                    increase_amount = min(deficit, 1 - h_new[idx])
                    h_new[idx] += increase_amount
                    deficit -= increase_amount
        
        # Rescale to exact sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z
        z_new = (h_new > 0.5).astype(float)
        
        # Update h and z
        h = h_new.copy()
        z = z_new.copy()
        
        # Compute C5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update penalty
        mu *= penalty_factor
    
    # Final local refinement
    best_h = best_h.copy()
    for i in range(20):
        if time.time() - start_time > budget_s - 5:
            break
        corr = correlate(best_h, 1 - best_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(best_h) - 1)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_lag
            if 0 <= j < n_points:
                grad[i] += (1 - 2 * best_h[i])
        
        # Gradient step with small learning rate
        best_h = best_h - 0.0005 * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        corr = correlate(best_h, 1 - best_h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5

    return (best_h.tolist(), best_c5, n_points)
