import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Main loop with hybrid strategy
    for iter_count in range(1000):
        if time.time() - start_time > budget_s - 10:

        # Gradient step
        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        
        # Update using gradient
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        
        # Evaluate new value
        corr = correlate(best_h, 1 - best_h, mode='full')
        new_value = np.max(corr) * dx
        
        # Accept improvement
        if new_value < best_value:
            best_value = new_value
        else:

            if iter_count % 5 == 0:
                perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)

                if new_value < best_value:

        # Occasionally reinitialize to prevent stagnation
        if iter_count % 50 == 0 and iter_count > 0:

            h_restart = np.random.rand(n_points)
