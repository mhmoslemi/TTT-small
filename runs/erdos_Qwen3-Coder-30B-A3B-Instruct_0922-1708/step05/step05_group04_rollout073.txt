import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize with initial_h_values or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients
    def h_to_fourier(h):
        # Use real FFT for symmetric real signal
        f = fft(h)
        return np.real(f[:n_fourier]), np.imag(f[:n_fourier])
    
    def fourier_to_h(a, b):
        # Reconstruct h from Fourier coefficients
        f = np.zeros(n_points, dtype=complex)
        f[0] = a[0] + 0j
        if n_fourier > 1:
            for k in range(1, min(n_fourier, n_points//2 + 1)):
                f[k] = complex(a[k], b[k])
                if k < n_points//2:
                    f[n_points - k] = complex(a[k], -b[k])
        h = np.real(ifft(f))
        return np.clip(h, 0, 1)
    
    # Objective function for optimization
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient computation with finite differences
    def compute_gradient(a, b, eps=1e-4):
        h_base = fourier_to_h(a, b)
        base_c5 = compute_c5(h_base)
        grad_a = np.zeros_like(a)
        grad_b = np.zeros_like(b)
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = fourier_to_h(a_plus, b)
            c5_plus = compute_c5(h_plus)
            grad_a[i] = (c5_plus - base_c5) / eps
        
        for i in range(len(b)):
            b_plus = b.copy()
            b_plus[i] += eps
            h_plus = fourier_to_h(a, b_plus)
            c5_plus = compute_c5(h_plus)
            grad_b[i] = (c5_plus - base_c5) / eps
            
        return grad_a, grad_b
    
    # Initial Fourier coefficients
    a_init, b_init = h_to_fourier(h_init)
    current_a, current_b = a_init.copy(), b_init.copy()
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization loop
    iterations = 0
    stagnation = 0
    max_stagnation = 100
    learning_rate = 0.01
    
    while time.time() - start_time < budget_s - 5 and stagnation < max_stagnation:
        # Convert to h
        h = fourier_to_h(current_a, current_b)
        
        # Compute objective and gradient
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stagnation = 0
        else:
            stagnation += 1
        
        # Compute gradient
        grad_a, grad_b = compute_gradient(current_a, current_b)
        
        # Update parameters
        current_a -= learning_rate * grad_a
        current_b -= learning_rate * grad_b
        
        # Clip to prevent oscillations
        current_a = np.clip(current_a, -10, 10)
        current_b = np.clip(current_b, -10, 10)
        
        iterations += 1
    
    # Final refinement using local search
    h_refined = fourier_to_h(current_a, current_b)
    c5_refined = compute_c5(h_refined)
    
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
