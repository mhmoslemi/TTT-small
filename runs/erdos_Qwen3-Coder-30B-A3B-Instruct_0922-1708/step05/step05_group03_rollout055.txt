import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Population-based approach with genetic algorithm elements
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [h.copy()]
        for _ in range(population_size - 1):
            # Add variation around initial solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Initialize with random feasible solutions
        population = []
        for _ in range(population_size):
            individual = np.random.rand(n_points)
            individual = individual * (n_points / 2) / np.sum(individual)
            population.append(individual)
    
    generation = 0
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Find best so far
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_idx = fitness.index(min_fitness)
            best_h = population[best_idx].copy()
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        
        for i in range(0, population_size, 2):
            if i + 1 < population_size and np.random.rand() < crossover_rate:
                # Blend crossover
                parent1 = selected[i]
                parent2 = selected[i+1]
                alpha = np.random.uniform(0.7, 0.95)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = selected[i].copy()
                child2 = selected[min(i+1, population_size-1)].copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.03)
            
            # Repair to constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        
        # Local refinement occasionally
        if generation % refinement_interval == 0:
            # Refine top individuals
            top_indices = np.argsort(fitness)[:population_size//10]
            for idx in top_indices:
                individual = population[idx].copy()
                # Simple local search with finite differences
                for _ in range(30):
                    grad = np.zeros_like(individual)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = individual.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    individual = individual - 0.001 * grad
                    individual = np.clip(individual, 0, 1)
                    individual = individual * (n_points / 2) / np.sum(individual)
                    corr = np.correlate(individual, 1 - individual, mode='full')
                    fitness[idx] = np.max(corr) * dx
                
                population[idx] = individual.copy()
    
    # Final gradient refinement
    for _ in range(30):
        # Try several candidates
        candidates = [best_h.copy()]
        for _ in range(4):
            candidate = best_h + np.random.normal(0, 0.02, n_points)
            candidate = np.clip(candidate, 0, 1)
            candidate = candidate * (n_points / 2) / np.sum(candidate)
            candidates.append(candidate)
        
        for candidate in candidates:
            # Gradient descent
            for step in range(30):
                grad = np.zeros_like(candidate)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = candidate.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - best_c5) / eps
                
                candidate = candidate - 0.0005 * grad
                candidate = np.clip(candidate, 0, 1)
                candidate = candidate * (n_points / 2) / np.sum(candidate)
                
                corr = np.correlate(candidate, 1 - candidate, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = candidate.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
