import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_max = np.max(base_corr)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            grad[j] = (max_plus - base_max) / eps
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Re-normalize
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Recompute C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Periodic perturbations
        if i % perturb_freq == 0 and i > 0:
            noise = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            # Re-evaluate after perturbation
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                
    return (best_h.tolist(), best_c5, n_points)
