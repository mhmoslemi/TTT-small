import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start approach
    initial_solutions = []
    
    # Use provided solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    else:
        initial_solutions.append(np.ones(n_points) * 0.5)
    
    # Add random initializations
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    for h_start in initial_solutions:
        h = h_start.copy()
        
        # Convex relaxation step to get better starting point
        try:
            # Create a rough relaxation - we want to minimize the overlap, so try
            # to solve a QP where we bound the correlation terms
            h_relax = cp.Variable(n_points)
            # While we could find a better CVX relaxation, just do feasibility
            constraints = [
                h_relax >= 0,
                h_relax <= 1,
                cp.sum(h_relax) == n_points / 2
            ]
            # For simplicity in this framework, we skip the full convexification
            # and proceed with gradient-based method from the current h
            
        except Exception:
            pass
        
        # Refinement via gradient based method with bounds
        best_local_h = h.copy()
        best_local_c5 = float('inf')
        
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(correlation)
        
        def gradient(h_vec):
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c_plus = objective(h_plus)
                grad[i] = (c_plus - objective(h_vec)) / eps
            return grad
        
        # L-BFGS-B optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Normalize initial h to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        max_iter = 2000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current c5
            c5 = objective(h)
            if c5 < best_local_c5:
                best_local_c5 = c5
                best_local_h = h.copy()
            
            # Compute gradient
            grad = gradient(h)
            
            # Update using gradient descent with adaptive step size
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update 
            h = h_new
            
            # Adaptive learning rate
            if i % 100 == 0:
                lr = max(1e-6, lr * 0.9)
        
        if best_local_c5 < best_c5:
            best_c5 = best_local_c5
            best_h = best_local_h.copy()
    
    if best_h is None:
        # Fallback to simplest approach
        h_default = np.ones(n_points) * 0.5
        h_default = h_default * (n_points / 2) / np.sum(h_default)
        correlation = np.correlate(h_default, 1 - h_default, mode='full') * dx
        best_c5 = np.max(correlation)
        best_h = h_default.copy()
    
    # Final validation and correction  
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
