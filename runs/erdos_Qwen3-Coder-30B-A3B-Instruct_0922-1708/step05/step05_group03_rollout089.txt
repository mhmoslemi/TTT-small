import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Scale sequence
    scales = [10, 20, 40, 80, 100]
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent or fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-scale optimization
    for scale_idx, n in enumerate(scales):
        if time.time() - start_time > budget_s - 100:
            break
            
        # Create current resolution array
        current_h = np.zeros(n)
        if scale_idx == 0:
            # Initialize coarse resolution
            current_h[:] = 0.5
        else:
            # Interpolate from previous scale
            prev_h = h[:scales[scale_idx-1]]
            x_orig = np.linspace(0, 2, scales[scale_idx-1])
            x_target = np.linspace(0, 2, n)
            current_h = np.interp(x_target, x_orig, prev_h)
        
        # Formulate convex relaxation
        def solve_convex(current_h, n):
            # Continuous relaxation variables
            h_vars = cp.Variable(n)
            
            # Objective: approximate max over lags
            # We consider lags from -n//4 to n//4 for approximation
            lag_range = min(n//4, n//2)
            avg_overlap = 0
            for k in range(1, lag_range+1):
                # Convex approximation of the integral for correlation term
                # Using the fact that we're looking at ∫ h(x)(1-h(x+k)) dx
                # Approximating as sum of terms over the discrete samples
                # We focus on small lag range
                corr = 0
                for i in range(n-k):
                    corr += h_vars[i] * (1 - h_vars[i+k])
                avg_overlap += corr
            avg_overlap /= (lag_range) if lag_range > 0 else 1
            
            # Constraints
            constraints = [
                h_vars >= 0,
                h_vars <= 1,
                cp.sum(h_vars) == n/2
            ]
            
            # Solve
            prob = cp.Problem(cp.Minimize(avg_overlap), constraints)
            try:
                prob.solve(solver=cp.ECOS, verbose=False)
                return h_vars.value
            except:
                return current_h
        
        # Solve convex relaxation
        if n <= 40:  # Only solve convex for smaller problems
            relaxed_h = solve_convex(current_h, n)
            if relaxed_h is not None:
                current_h = relaxed_h
                
        # Apply hard constraints (clipping and normalization)
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n / 2) / np.sum(current_h)
        
        # If this is fine scale, perform refinement
        if n == n_points:
            h = current_h
            
            # Local refinement with gradient descent
            for _ in range(50):
                if time.time() - start_time > budget_s - 50:
                    break
                
                # Compute gradient using finite differences
                grad = np.zeros(n)
                eps = 1e-4
                base_corr = np.correlate(h, 1 - h, mode='full')
                base_c5 = np.max(base_corr) * dx
                
                for i in range(n):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - base_c5) / eps
                
                # Update
                h = h - 0.001 * grad
                h = np.clip(h, 0, 1)
                h = h * (n / 2) / np.sum(h)
                
                # Evaluate new value
                corr = np.correlate(h, 1 - h, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
        else:
            # Progressively build up solution
            h = current_h
    
    # Final refinement with full solution
    if n_points > 100:
        h = h[:100]
    else:
        h = h[:n_points]

    # Final local optimization
    for _ in range(100):
        if time.time() - start_time > budget_s - 50:
            break
        
        grad = np.zeros(n_points)
        eps = 1e-4
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - base_c5) / eps
            
        h = h - 0.0005 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
