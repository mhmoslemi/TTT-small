import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    adaptive_lr_decay = 0.95
    
    # Initialize with initial construction or block construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-6
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_max = np.max(base_corr)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            grad[i] = (max_plus - base_max) / eps
        
        # Gradient descent step
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
        
        # Periodic perturbations to escape local optima
        if iteration % perturb_freq == 0:
            perturb_magnitude_t = perturb_magnitude / (1.0 + iteration / 100.0)
            perturbation = np.random.normal(0, perturb_magnitude_t, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
            # Adaptive learning rate adjustment
            learning_rate *= adaptive_lr_decay
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
