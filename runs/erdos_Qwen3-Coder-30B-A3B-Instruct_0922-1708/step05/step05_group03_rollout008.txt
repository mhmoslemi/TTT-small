import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()

    generations = 0
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [h.copy()]
        for _ in range(population_size - 1):
            # Add perturbed version of initial solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        population = []
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    # Evaluation function
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Tournament selection
    def tournament_selection(population, fitnesses, tournament_size=5):
        selected_idx = np.random.choice(len(population), tournament_size)
        selected_fitness = [fitnesses[i] for i in selected_idx]
        return population[selected_idx[np.argmin(selected_fitness)]]
    
    # Crossover operator
    def crossover(parent1, parent2, alpha=0.8):
        child1 = alpha * parent1 + (1 - alpha) * parent2
        child2 = (1 - alpha) * parent1 + alpha * parent2
        return child1, child2
    
    # Mutation operator
    def mutate(individual, mutation_rate=0.1, sigma=0.03):
        mask = np.random.random(len(individual)) < mutation_rate
        noise = np.random.normal(0, sigma, len(individual))
        individual += mask * noise
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        return individual
    
    # Main loop
    while time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Create new population
        new_population = []
        num_parents = population_size // 2
        
        # Elitism - keep best two
        sorted_indices = np.argsort(fitnesses)[:2]
        new_population.extend([population[i].copy() for i in sorted_indices])
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        generations += 1

        if generations % 20 == 0:
            # Apply local refinement to top individuals
            sorted_indices = np.argsort(fitnesses)[:10]
            for i in sorted_indices:
                h_ref = population[i].copy()
                
                # Gradient-based refinement
                for _ in range(30):

                    eps = 1e-4

                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitnesses[i]) / eps
                    
                    h_ref = h_ref - 0.001 * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                
                # Evaluate refined result
                corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                new_c5 = np.max(corr_new) * dx
                
                if new_c5 < fitnesses[i]:
                    population[i] = h_ref.copy()
                    fitnesses[i] = new_c5
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_ref.copy()

    candidates = []

    for _ in range(3):
        if 'initial_h_values' in globals():
            h_candidate = best_h.copy() + np.random.normal(0, 0.02, n_points)
        else:
            h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidates.append(h_candidate)
    
    # Gradient refinement
    for i, h_ref in enumerate(candidates):
        for _ in range(30):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
