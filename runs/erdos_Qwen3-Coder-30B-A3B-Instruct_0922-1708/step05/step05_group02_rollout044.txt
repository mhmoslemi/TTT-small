import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate Lévy distributed random numbers
    # Using the characteristic function approach: for alpha=1.5,
    # we can simulate it as a scaled stable distribution
    # A simpler approach for practical implementation:
    # sample from a heavy-tailed distribution
    return np.random.standard_cauchy(size) * scale

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add 49 more random initial candidates
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Evaluate initial candidates
    current_values = [evaluate_h(cand, dx) for cand in candidates]
    
    # Find best candidate
    best_idx = np.argmin(current_values)
    best_h = candidates[best_idx].copy()
    best_value = current_values[best_idx]
    
    # Parameters
    max_iter = 1000
    top_k = 10
    alpha = 1.5
    scale = 0.5
    lr = 0.001
    perturb_rate = 0.1
    patience = 100
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        iter_count += 1
        
        # Generate new candidates with Lévy flights
        new_candidates = []
        
        for cand in candidates:
            # Perturb with Lévy flight
            perturb = levy_flight(n_points, alpha, scale)
            new_cand = cand + perturb_rate * perturb
            new_cand = repair_h(new_cand, n_points)
            new_candidates.append(new_cand)
        
        # Evaluate new candidates
        new_values = [evaluate_h(cand, dx) for cand in new_candidates]
        
        # Update candidates with some gradient descent
        top_indices = np.argsort(new_values)[:top_k]
        
        for idx in top_indices:
            h = new_candidates[idx].copy()
            current_c5 = new_values[idx]
            
            # Gradient descent step
            for _ in range(5):
                if (time.time() - start_time) > budget_s - 5:
                    break
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                h = h - lr * grad
                h = repair_h(h, n_points)
                corr = correlate(h, 1 - h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < new_values[idx]:
                    new_candidates[idx] = h.copy()
                    new_values[idx] = current_c5
        
        # Add more candidates by re-initializing bad ones
        # Replace worst candidates with new ones
        worst_indices = np.argsort(current_values)[-20:]
        for idx in worst_indices:
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            new_candidates.append(h)
            new_values.append(evaluate_h(h, dx))
        
        # Update candidates
        candidates = new_candidates
        current_values = new_values
        
        # Update best
        best_idx = np.argmin(current_values)
        if current_values[best_idx] < best_value:
            best_h = candidates[best_idx].copy()
            best_value = current_values[best_idx]
            last_improvement = iter_count
        
        # Early stopping condition
        if iter_count - last_improvement > patience:
            break
    
    # Final gradient descent refinement
    best_h_original = best_h.copy()
    best_value_original = best_value
    
    for _ in range(1000):
        if (time.time() - start_time) > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
    
    if best_value < best_value_original:
        pass  # Keep refined solution
    else:
        best_h = best_h_original
        best_value = best_value_original
    
    return (best_h.tolist(), best_value, n_points)
