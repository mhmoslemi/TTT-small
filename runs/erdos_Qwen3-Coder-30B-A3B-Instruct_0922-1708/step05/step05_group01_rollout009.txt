import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Scale to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    perturbation_fraction = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlation lags 
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        k_star = np.argmax(corr)
        
        # Compute gradient components
        grad = np.zeros(n_points)
        # The gradient for the maximum correlation
        # We compute it by looking at the indices where max lag occurs
        n_corr = len(corr)
        lag = k_star - (n_points - 1)
        
        # We apply chain rule: d/dh_i of h_i * (1 - h_{i+lag}) where lag > 0
        # This is -h_{i+lag} for i = 0 to n_points - lag - 1
        # And 1 - h_{i-lag} for i = lag to n_points - 1
        
        if lag > 0:
            # h[i] * (1 - h[i+lag]) contributes -h[i+lag] to d/dh[i]
            # and (1 - h[i+lag]) to d/dh[i+lag]
            for i in range(n_points - lag):
                grad[i] -= h[i + lag] * dx
                # Also, the contribution to the derivative of (1-h[i+lag]) wrt h[i]
                grad[i + lag] -= (1 - h[i + lag]) * dx
        else:
            # lag < 0, same logic but modified
            lag_abs = -lag
            for i in range(n_points - lag_abs):
                grad[i] -= h[i + lag_abs] * dx
                grad[i + lag_abs] -= (1 - h[i + lag_abs]) * dx
        
        # Because we have max over lags, we only consider the gradient 
        # for the specific lag that maximizes
        # Actually, re-evaluate that. Our approach is to directly compute
        # the gradient for the maximum correlation term
        
        # Compute gradient for correlation k_star that is max
        grad = np.zeros(n_points)
        
        # We want gradient with respect to correlation
        # d/dh_i [max_k \sum_j h[j]*h[j+k]], we pick k*
        
        # Instead, we use the Helmholtz decomposition concept used in 
        # probability theory gradients for convolutions:
        # The derivative of correlation at lag k* can be expressed
        # as a convolution of the gradients
        
        # Correct method:
        # For a correlation sum at lag k*:  sum_j h[j] * h[j+k*]
        # Gradient wrt h[i]: (when j=i -> 1 - h[i+k*] and when j=i+k* -> h[i])
        # But we consider max over all k*, so only the density of lag k* matters
        
        # We actually compute the Hessian properly...
        # But for simplicity of line search, we do the trick:
        
        # The key insight is that it's about the maximum k* value
        # and its relative contribution to the maximum correlation
        
        # Reconsider gradient setup with:
        # correlation k* between h(x) and 1-h(x+k*)
        # For index i of h: contribution is if matching h[j] h[j+k*] 
        # where j = i or j = i-k* (correlated)
        
        # Instead we do finite diff gradient to reduce complexity:
        # computational糊涂. Real model gradient is from the max lags
        
        # Working with Lag k* from where C5 is maximum:
        if lag >= 0:
            for i in range(n_points - lag):
                grad[i] += 1 - h[i + lag] if i + lag < n_points else 0
                grad[i + lag] -= h[i] if i < n_points - lag else 0
        else:
            lag_abs = -lag
            for i in range(n_points - lag_abs):
                grad[i] -= h[i + lag_abs] if i + lag_abs < n_points else 0
                grad[i + lag_abs] += 1 - h[i] if i < n_points - lag_abs else 0
        
        grad = grad * dx
        
        # Line search
        # Track current value
        current_c5 = c5
        while True:
            # Compute next h
            h_new = h - alpha * grad
            # Project to feasible set (0,1)
            h_new = np.clip(h_new, 0, 1)
            # Rescale sum to n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < current_c5:
                h = h_new
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                # Accept step
                break
            else:
                # Reduce alpha
                alpha *= line_search_factor
                if alpha < 1e-10:
                    break
            
        # Occasionally add random perturbation to escape local minima
        if iteration % 100 == 0:
            noise = np.random.normal(0, perturbation_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
