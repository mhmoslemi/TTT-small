import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    max_time = time.time() + budget_s - 10

    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraint ∫h(x)dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute the initial C5 value
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Maintain step history for trust region
    step_history = []
    
    for iteration in range(max_iterations):
        if time.time() > max_time:
            break
            
        # Compute gradient estimate for dominant correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        current_max_k = max_k
        
        # Compute approximate gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Compute candidate step using trust region
        step = -trust_radius * grad
        
        # Project step to maintain constraints
        h_new = h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept new solution if improvement or within trust region
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            
            # Expand trust region
            trust_radius = min(trust_radius * 1.2, 0.1)
        else:
            # Shrink trust region
            trust_radius = max(trust_radius * 0.8, min_trust_radius)
            
        # Apply simulated annealing perturbations periodically
        if iteration % perturbation_freq == 0 and iteration > 0:
            # Perturb with SA-type effect
            for i in range(n_points):
                if np.random.rand() < 0.1:  # Perturbation probability
                    delta = temperature * (1 - h[i]) * h[i] * (np.random.rand() - 0.5)
                    h[i] += delta
            # Re-normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr_new = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr_new) * dx
            
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Temperature decay for SA
        temperature *= cooling_rate
        
        # Occasionally refine the solution with more accurate gradient
        if iteration % 50 == 0 and iteration > 0:
            # Compute refinement using optimizer
            def obj(h_vals):
                h_refine = np.clip(h_vals, 0, 1)
                h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                corr_ref = np.correlate(h_refine, 1 - h_refine, mode='full')
                return np.max(corr_ref) * dx
                
            def constraint_sum(h_vals):
                return np.sum(h_vals) - n_points / 2
                
            cons = {'type': 'eq', 'fun': constraint_sum}
            bounds = [(0, 1) for _ in range(n_points)]
            
            try:
                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                              options={'maxiter': 50, 'ftol': 1e-6, 'gtol': 1e-6})
                if not res.failed:
                    h = res.x
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    corr_ref = np.correlate(h, 1 - h, mode='full')
                    current_c5 = np.max(corr_ref) * dx
            except:
                pass  # Skip refinement if error occurs

    # Final intensive refinement step
    def fine_obj(h_vals):
        h_refine = np.clip(h_vals, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        corr_ref = np.correlate(h_refine, 1 - h_refine, mode='full')
        return np.max(corr_ref) * dx

    def fine_constraint(h_vals):
        return np.sum(h_vals) - n_points / 2

    cons = {'type': 'eq', 'fun': fine_constraint}
    bounds = [(0, 1) for _ in range(n_points)]

    try:
        res = minimize(fine_obj, best_h, method='SLSQP', bounds=bounds, constraints=cons, 
                      options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-9})
        if not res.failed:
            best_h = res.x
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            corr_final = np.correlate(best_h, 1 - best_h, mode='full')
            best_c5 = np.max(corr_final) * dx
    except:
        pass  # Use previous best if refinement fails

    return (best_h.tolist(), best_c5, n_points)
