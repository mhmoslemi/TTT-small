import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_lr = 1e-3
    lr_decay = 0.999
    min_lr = 1e-6
    plateau_patience = 200
    perturb_frac = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    lr = initial_lr
    patience = 0
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # If plateaued, reduce learning rate or perturb
        if patience > plateau_patience:
            if lr > min_lr:
                lr *= lr_decay
                patience = 0
            else:
                # Add small random perturbation
                noise = np.random.normal(0, perturb_frac, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                patience = 0
        
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_val = corr[max_lag_idx]
        
        # Compute gradient
        grad = np.zeros(n_points)
        
        # Reconstruct the gradient based on the maximum lag
        # The correspondence between lag and derivative is tricky.
        # Use first-order finite difference approximation to get gradient.
        eps = 1e-6
        h_plus = h + eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        c5_plus = np.max(corr_plus) * dx
        
        grad = (c5_plus - c5) / eps * np.ones(n_points)
        
        # Perform gradient step
        h = h - lr * grad
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        last_c5 = c5
    
    # Final validation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
