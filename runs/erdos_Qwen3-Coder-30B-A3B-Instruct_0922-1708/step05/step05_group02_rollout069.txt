import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(n_points, scale=0.5, alpha=1.5):
    """Generate a Lévy flight perturbation."""
    # Generate Lévy distributed random variables
    # Simple implementation using the inverse transform method for Cauchy distribution
    # as a proxy for heavy-tailed distribution
    return scale * np.random.standard_cauchy(n_points)

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Generate initial population
    population_size = 50
    population = []
    population_scores = []
    
    # Add the initial solution
    population.append(h_init.copy())
    population_scores.append(evaluate_h(h_init, dx))
    
    # Add random initial solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        population_scores.append(evaluate_h(h, dx))
    
    # Keep track of best solution
    best_idx = np.argmin(population_scores)
    best_h = population[best_idx].copy()
    best_score = population_scores[best_idx]
    
    # Parameters
    max_iter = 1000
    levy_scale = 0.5
    levy_alpha = 1.5
    learning_rate = 0.001
    perturb_rate = 0.1
    top_candidates = 10
    
    # Main optimization loop
    no_improvement_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new candidates via Lévy flights
        new_population = []
        new_scores = []
        
        for h in population:
            # Generate Lévy perturbation
            perturbation = levy_flight(n_points, scale=levy_scale, alpha=levy_alpha)
            
            # Apply perturbation and repair
            new_h = h + perturbation
            new_h = repair_h(new_h, n_points)
            
            new_population.append(new_h)
            new_scores.append(evaluate_h(new_h, dx))
        
        # Add some random perturbations to maintain diversity
        for _ in range(population_size // 4):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            new_population.append(h)
            new_scores.append(evaluate_h(h, dx))
        
        # Combine and sort by fitness
        all_pop = population + new_population
        all_scores = population_scores + new_scores
        
        # Sort by scores
        sorted_indices = np.argsort(all_scores)
        population = [all_pop[i] for i in sorted_indices[:population_size]]
        population_scores = [all_scores[i] for i in sorted_indices[:population_size]]
        
        # Update best solution
        current_best_idx = np.argmin(population_scores)
        if population_scores[current_best_idx] < best_score:
            best_h = population[current_best_idx].copy()
            best_score = population_scores[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Local gradient descent on top candidates
        if iteration % 5 == 0:
            top_indices = np.argsort(population_scores)[:top_candidates]

                h = population[idx].copy()

                # Gradient descent step
                for _ in range(50):

                        break
                    eps = 1e-6
                    grad = np.zeros(n_points)

                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - current_score) / eps
                    
                    h = h - learning_rate * grad

                    corr = correlate(h, 1 - h, mode='full')
                    current_score = np.max(corr) * dx
                    
                    if current_score < population_scores[idx]:
                        population[idx] = h.copy()
                        population_scores[idx] = current_score
                        if current_score < best_score:
                            best_h = h.copy()
                            best_score = current_score
                            no_improvement_count = 0
        
        # Early stopping if no improvement
        if no_improvement_count > 100:
            break
    
    # Final intensive gradient descent on the best solution
    h_final = best_h.copy()
    score_final = best_score
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - score_final) / eps
        
        h_final = h_final - 0.0005 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        score_final = np.max(corr) * dx
    
    return (h_final.tolist(), score_final, n_points)
