import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    swarm_size = 50
    inertia_weight = 0.7
    phi_p = 1.5
    phi_g = 1.5
    local_search_freq = 30
    rejuvenation_rate = 0.15
    max_ls_iter = 200
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        for _ in range(swarm_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        for _ in range(swarm_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal bests and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * swarm_size
    
    # Compute initial fitness for all particles
    for i in range(swarm_size):
        correlation = correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(correlation) * dx
        pbest_fitness[i] = fitness
        if i == 0 or fitness < pbest_fitness[0]:
            gbest_idx = i
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Optimization variables
    current_iter = 0
    stagnation_count = 0
    
    # Main loop
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Update particles
        for p_idx in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[p_idx] = (inertia_weight * velocities[p_idx] + 
                                phi_p * r1 * (pbest_particles[p_idx] - particles[p_idx]) +
                                phi_g * r2 * (gbest_particle - particles[p_idx]))
            
            # Position update
            particles[p_idx] += velocities[p_idx]
            
            # Repair
            particles[p_idx] = repair(particles[p_idx])
            
            # Fitness computation
            correlation = correlate(particles[p_idx], 1 - particles[p_idx], mode='full')
            fitness = np.max(correlation) * dx
            
            # Update pbest
            if fitness < pbest_fitness[p_idx]:
                pbest_fitness[p_idx] = fitness
                pbest_particles[p_idx] = particles[p_idx].copy()
                
                # Update gbest
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[p_idx].copy()
                    stagnation_count = 0
        
        # Stagnation check
        stagnation_count += 1
        if stagnation_count > 200:
            # Inject random particles
            for _ in range(int(swarm_size * rejuvenation_rate)):
                idx = np.random.randint(0, swarm_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.05, 0.05, n_points)
            stagnation_count = 0
        
        # Local search
        if current_iter % local_search_freq == 0 and current_iter > 0:
            # Local search on top 20% of particles
            sorted_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[:swarm_size//5]
            for idx in sorted_indices:
                h_local = pbest_particles[idx].copy()
                # Local refinement with L-BFGS
                def obj(h):
                    # Ensure h is valid
                    h = np.clip(h, 0, 1)
                    h_sum = np.sum(h)
                    if abs(h_sum - n_points / 2) > 1e-10:
                        h = h * (n_points / 2) / h_sum
                    correlation = correlate(h, 1 - h, mode='full')
                    return -np.max(correlation) * dx
                
                def grad(h):
                    correlation = correlate(h, 1 - h, mode='full')
                    max_lag = np.argmax(correlation)
                    # Compute gradient w.r.t. h_i for recent lags only
                    g = np.zeros_like(h)
                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += 1e-6
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus_sum = np.sum(h_plus)
                        if abs(h_plus_sum - n_points / 2) > 1e-10:
                            h_plus = h_plus * (n_points / 2) / h_plus_sum
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        max_plus = np.argmax(corr_plus)
                        g[i] = (corr_plus[max_plus] - correlation[max_lag]) / 1e-6
                    return -g * dx
                
                # Gradient-based local search
                h_opt = h_local.copy()
                for _ in range(50):
                    grad_val = grad(h_opt)
                    h_opt -= 0.0001 * grad_val
                    h_opt = repair(h_opt)
                    
                # Verify final result
                correlation = correlate(h_opt, 1 - h_opt, mode='full')
                fitness = np.max(correlation) * dx
                if fitness < pbest_fitness[idx]:
                    pbest_fitness[idx] = fitness
                    pbest_particles[idx] = h_opt.copy()
                    
                    if fitness < gbest_fitness:
                        gbest_fitness = fitness
                        gbest_particle = h_opt.copy()
                        stagnation_count = 0
        
        current_iter += 1
    
    # Final refinement
    final_candidates = [pbest_particles[i] for i in range(swarm_size) if pbest_fitness[i] < gbest_fitness * 1.05]
    final_candidates = final_candidates[:3]
    
    best_h = gbest_particle.copy()
    best_c5 = gbest_fitness
    
    def gradient_descent_refinement(h_start):
        h_ref = h_start.copy()
        for _ in range(100):
            correlation = correlate(h_ref, 1 - h_ref, mode='full')
            max_lag = np.argmax(correlation)
            fitness = correlation[max_lag] * dx
            
            # Compute gradient
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                max_plus = np.argmax(corr_plus)
                grad[i] = (corr_plus[max_plus] - correlation[max_lag]) / eps
            
            h_ref -= 0.0005 * grad
            h_ref = repair(h_ref)
            
            new_corr = correlate(h_ref, 1 - h_ref, mode='full')
            new_fitness = new_corr[np.argmax(new_corr)] * dx
            
            if new_fitness >= fitness:
                break
            else:
                fitness = new_fitness
                
        return h_ref, new_fitness
    
    for h_ref in final_candidates:
        h_opt, c5_opt = gradient_descent_refinement(h_ref)
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt
    
    return (best_h.tolist(), best_c5, n_points)
