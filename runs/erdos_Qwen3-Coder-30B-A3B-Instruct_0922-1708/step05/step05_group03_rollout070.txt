import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO parameters
    swarm_size = 50
    inertia_weight = 0.7
    cognitive_param = 2.0
    social_param = 2.0
    
    # Initialize swarm
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as basis
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        for _ in range(swarm_size - 1):
            # Add variation around initial solution
            perturbed = h.copy() + np.random.normal(0, 0.02, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            particles.append(perturbed)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        # Initialize swarm with random feasible solutions
        for _ in range(swarm_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            particles.append(particle)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal bests
    pbest = [p.copy() for p in particles]
    pbest_c5 = [float('inf')] * swarm_size
    
    # Global best tracking
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    iteration = 0
    
    # Main PSO loop
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Update personal and global bests
        for i in range(swarm_size):
            # Compute current particle fitness
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = particles[i].copy()
                
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                
        # Keep the best solution found so far
        if gbest_c5 < best_c5:
            best_c5 = gbest_c5
            best_h = pbest[gbest_idx].copy()
        
        # Update velocities and positions for all particles
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            cognitive_param * r1 * (pbest[i] - particles[i]) + 
                            social_param * r2 * (pbest[gbest_idx] - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair solution to maintain constraints
            # Clip to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Adjust to maintain sum constraint
            sum_h = np.sum(particles[i])
            if abs(sum_h - n_points/2) > 1e-8:
                # Rescale to meet sum constraint
                particles[i] = particles[i] * (n_points / 2) / sum_h
                
        # Local search occasionally
        if iteration % 20 == 0:
            # Do local refinement on top 10% particles
            top_indices = np.argsort(pbest_c5)[:swarm_size//5]
            for idx in top_indices:
                # Gradient descent refinement
                h_refine = pbest[idx].copy()
                lr = 0.001
                
                for _ in range(50):
                    # Estimate gradient
                    grad = np.zeros_like(h_refine)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_refine.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - pbest_c5[idx]) / eps
                    
                    # Update
                    h_refine = h_refine - lr * grad
                    h_refine = np.clip(h_refine, 0, 1)
                    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                    
                    # Re-evaluate
                    corr_new = np.correlate(h_refine, 1 - h_refine, mode='full')
                    new_c5 = np.max(corr_new) * dx
                    
                    if new_c5 < pbest_c5[idx]:
                        pbest_c5[idx] = new_c5
                        pbest[idx] = h_refine.copy()
                    
                    if new_c5 < gbest_c5:
                        gbest_c5 = new_c5
                        gbest_idx = idx
    
    # Final refinement on best found solutions
    final_candidates = []
    for _ in range(5):
        if 'initial_h_values' in globals():
            h_candidate = best_h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        final_candidates.append(h_candidate)
    
    # Add top from swarm
    top_indices = np.argsort(pbest_c5)[:min(2, len(pbest_c5))]
    for i in top_indices:
        final_candidates.append(pbest[i].copy())
        
    # Gradient refinement of candidates
    for h_ref in final_candidates:
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
                
            h_ref = h_ref - 0.0003 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
