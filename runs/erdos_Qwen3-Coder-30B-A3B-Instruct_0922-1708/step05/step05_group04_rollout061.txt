import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
            
    # Helper function to repair solutions
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) < 1e-10:
            return h
        diff = current_sum - n_points / 2
        if diff > 0:
            # Remove from largest
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            # Add to smallest
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluation function
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Initial fitness evaluation
    fitnesses = [evaluate(p) for p in population]
    best_idx = np.argmin(fitnesses)
    best_individual = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Differential Evolution main loop
    current_iter = 0
    stagnation_count = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = x1 + mutation_factor * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to maintain constraint
            mutant = repair(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_prob or j == j_rand:
                    trial[j] = mutant[j]
            
            # Repair again
            trial = repair(trial)
            
            # Evaluation and selection
            trial_fitness = evaluate(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
                fitnesses[i] = trial_fitness
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_individual = trial.copy()
                    stagnation_count = 0
            else:
                new_population.append(population[i])
        
        population = new_population
        current_iter += 1
        
        # Adaptive mutation
        if current_iter % 50 == 0:
            mutation_factor *= 0.99
            
        # Stagnation detection
        stagnation_count += 1
        if stagnation_count > 200:
            # Reinitialize population with new random individuals
            tmp_pop = []
            for _ in range(population_size):
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                tmp_pop.append(h)
            population = tmp_pop
            stagnation_count = 0
    
    # Final gradient-based refinement of best solution
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def grad(h):
        # Approximate gradient using finite differences
        eps = 1e-5
        g = np.zeros_like(h)
        base_val = obj(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            g[i] = (obj(h_plus) - base_val) / eps
        return g
    
    # Gradient descent
    refined_h = best_individual.copy()
    for _ in range(100):
        g = grad(refined_h)
        refined_h -= 0.001 * g
        refined_h = repair(refined_h)
    
    final_corr = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    return (refined_h.tolist(), final_fitness, n_points)
