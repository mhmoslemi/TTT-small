import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize with a simple construction
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients (real FFT of symmetric extension)
    def h_to_fourier(h):
        # Extend to make it even symmetric
        h_extended = np.concatenate([h, h[::-1]])
        # Compute real FFT
        fft_coeffs = fft(h_extended)
        # Take only first half (real coefficients)
        real_coeffs = fft_coeffs[:n_fourier].real
        imag_coeffs = fft_coeffs[:n_fourier].imag
        return np.concatenate([real_coeffs, imag_coeffs])
    
    def fourier_to_h(coeffs):
        # Separate real and imag parts
        real_coeffs = coeffs[:n_fourier]
        imag_coeffs = coeffs[n_fourier:]
        
        # Reconstruct fft with real input
        # Note: Real FFT for even symmetric signal
        new_coeffs = np.zeros(2*n_points)
        new_coeffs[:n_fourier] = real_coeffs + 1j * imag_coeffs
        new_coeffs[1:n_fourier] = real_coeffs[1:][::-1] - 1j * imag_coeffs[1:][::-1]
        
        # Inverse FFT
        h_full = ifft(new_coeffs).real
        return h_full[:n_points]
    
    # Initialize with h_init
    init_coeffs = h_to_fourier(h_init)
    
    # Optimization function minimizing C5
    def objective(coeffs):
        h = fourier_to_h(coeffs)
        # Clip values to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    def gradient(coeffs):
        eps = 1e-6
        grad = np.zeros_like(coeffs)
        
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            f_plus = objective(coeffs_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[i] -= eps
            f_minus = objective(coeffs_minus)
            
            grad[i] = (f_plus - f_minus) / (2 * eps)
        return grad
    
    # Optimization with L-BFGS-B
    opt_result = minimize(
        objective,
        init_coeffs,
        method='L-BFGS-B',
        jac=gradient,
        bounds=[(-10, 10)] * len(init_coeffs),
        options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
    )
    
    # Final refinement
    best_coeffs = opt_result.x
    h_final = fourier_to_h(best_coeffs)
    
    # Apply repair to ensure constraints
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # Evaluate final result
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Validate that we respect all constraints
    assert np.isclose(np.sum(h_final) * dx, 1.0, atol=1e-6), "Constraint violated"
    assert np.all(h_final >= 0) and np.all(h_final <= 1), "Bounds violated"
    
    return (h_final.tolist(), final_c5, n_points)
