```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given h."""
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(correlation)

def project_to_feasible(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and h[i] ∈ [0,1]."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.ones_like(h) * (n_points / 2) / n_points
    return h

def smooth_max(x, beta=10.0):
    """Computes a smooth approximation to max(x)."""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        # Better initial point using structured sequence
        h = np.linspace(0.1, 0.9, n_points)
        # Slightly randomize to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
    
    # Normalize
    h = project_to_feasible(h, n_points)
    
    # Gradient descent with smoothing for better convergence
    dx = 2.0 / n_points
    lr = 0.01
    beta = 10.0  # Smooth max parameter
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Use two-stage approach with coarse grid first to escape local minima
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    h_coarse = project_to_feasible(h_coarse, n_coarse)
    
    # Coarse optimization with smoothing
    for iter_coarse in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current
        c5_current = evaluate_c5(h_coarse, dx_coarse)
        # Smooth approximation to the objective to allow gradient descent
        # We do not directly optimize the true maximum since it's non-differentiable
        corr_vals = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        smooth_c5 = smooth_max(corr_vals, beta)
        
        # Compute gradient (finite differences)
        grad = np.zeros_like(h_coarse)
        eps = 1e-5
        for i in range(n_coarse):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_coarse)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            smooth_plus = smooth_max(corr_plus, beta)
            grad[i] = (smooth_plus - smooth_c5) / eps
        
        # Update
        h_coarse -= lr * grad
        h_coarse = project_to_feasible(h_coarse, n_coarse)
        
        # Evaluate final coarse result
        c5_check = evaluate_c5(h_coarse, dx_coarse)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
            best_h = project_to_feasible(best_h, n_points)
    
    # Transfer to fine grid
    if n_points > n_coarse:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
        h = project_to_feasible(h, n_points)
    
    # Fine optimization
    for iter_fine in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = evaluate_c5(h, dx)
        
        # Smooth approximation to objective
        corr_vals = np.correlate(h, 1 - h, mode='full') * dx
        smooth_c5 = smooth_max(corr_vals, beta)
        
        # Compute grad via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            smooth_plus = smooth_max(corr_plus, beta)
            grad[i] = (smooth_plus - smooth_c5) / eps
        
        # Update with learning rate scheduling
        lr = max(1e-6, 0.01 * 0.99**(iter_fine / 500))
        h -= lr * grad
        h = project_to_feasible(h, n_points)
        
        # Track best
        c5_check = evaluate_c5(h, dx)
        if c5_check < best_c5:
            best_c5 = c5_check
            best_h = h.copy()
    
    # Final validation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
```
