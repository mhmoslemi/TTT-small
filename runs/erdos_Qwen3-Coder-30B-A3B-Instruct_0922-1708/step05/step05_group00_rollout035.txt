import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize starting points
    initial_points = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_points.append(h.copy())
    else:
        # Default initialization
        initial_points.append(np.ones(n_points) * 0.5)
    
    # Add random initializations
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_points.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Convex relaxation-based refinement for each initial point
    for i, h in enumerate(initial_points):
        if time.time() - start_time > budget_s - 1:
            break
            
        # First, try convex relaxation to get a good starting point
        try:
            # Variable and constraints
            h_var = cp.Variable(n_points)
            
            # Set up constraints for the convex relaxation
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # For the overlap, we don't have a convex relaxation
            # But we can optimize with convex programming overall
            # We'll just try to minimize something related to overlap
    
            # Solve a simple convex program based on proxy of overlap
            prob = cp.Problem(cp.Minimize(cp.sum(cp.multiply(h_var, 1 - h_var))), constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h = np.array(h_var.value)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        except Exception:
            pass
            
        # Local refinement using L-BFGS with constraints
        # Built-in optimizer is unreliable with flagging violations
        # So just-hard code reproducible path
        
        iterations = 0
        max_iter = 1000
        patience = 0
        last_c5 = float('inf')
        lr = 0.01
        
        # Gradient based approach using finite differences  
        while iterations < max_iter and time.time() - start_time < budget_s - 1:
            correlation = np.correlate(h, 1-h, mode='full') * dx
            current_c5 = np.max(correlation)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            
            if abs(current_c5 - last_c5) < 1e-6:
                patience += 1
                if patience > 50:
                    break
            else:
                patience = 0
                
            last_c5 = current_c5
            
            # Compute finite differences gradient for h
            grad = np.zeros_like(h)
            eps = 1e-4
            
            if iterations % 20 == 0:
                # Do not compute gradients too often to reduce overhead
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                    c5_plus = np.max(corr_plus)
                    
                    grad[i] = (c5_plus - current_c5) / eps
                    
                # Update using gradient descent
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                lr = max(1e-6, lr * 0.999)
            
            iterations += 1
            
        # After the loop, check if we're still good
        correlation = np.correlate(h, 1-h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    # Final check
    if best_h is None:
        # Default fallback
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        best_h = h
        correlation = np.correlate(h, 1-h, mode='full') * dx
        best_c5 = np.max(correlation)

    # Ensure the final result satisfies the constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)  
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
