import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points_list = [10, 20, 40, 80, 100]
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (100 / 2) / np.sum(h_parent)
    else:
        h_parent = None
    
    h_current = None
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 100:
            break
            
        dx = 2.0 / n_points
        
        # Initialize or upsample
        if h_current is None:
            h_current = np.ones(n_points) * 0.5
        else:
            # Linear upsampling
            x_old = np.linspace(0, 2, len(h_current))
            x_new = np.linspace(0, 2, n_points)
            h_current = np.interp(x_new, x_old, h_current)
        
        # Convex relaxation
        # Variables
        h = cp.Variable(n_points)
        
        # Objective: approximate max by average over central lags
        # Use lags ±1 to ±(n/4) for relaxation
        lag_min, lag_max = 1, min(n_points//4, n_points-1)
        lags = list(range(lag_min, lag_max+1))
        
        # Soft constraints
        lambda_reg = 0.1  # regularization weight
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Approximate objective with sum over lags
        approx_obj = 0
        for k in lags:
            # Compute correlation using CVXPY
            corr_left = cp.sum(cp.multiply(h[:n_points-k], 1 - h[k:]))
            approx_obj += corr_left
        
        approx_obj = approx_obj / len(lags) * dx
        
        # Objective
        objective = cp.Minimize(approx_obj)
        
        # Solver
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_vals = h.value
            else:
                # Fallback
                h_vals = np.clip(h_current, 0, 1)
                h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        except:
            h_vals = np.clip(h_current, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        
        # Project solution
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        h_current = h_vals.copy()
        
        # Evaluate full overlap
        corr_full = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
    
    # Local gradient refinement
    if time.time() - start_time > budget_s - 50:
        pass
    else:
        h_ref = best_h.copy()
        dx = 2.0 / 100
        lr = 0.001
        eps = 1e-4
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 20:
                break
            grad = np.zeros_like(h_ref)
            for j in range(100):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (100 / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - lr * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (100 / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_ref.copy()
    
    # Final validation
    if best_h is None:
        if h_parent is not None:
            best_h = h_parent
        else:
            best_h = np.ones(100) * 0.5
            best_h = best_h * (100 / 2) / np.sum(best_h) 
    
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * (2.0 / 100)
    
    return (best_h.tolist(), final_c5, 100)
