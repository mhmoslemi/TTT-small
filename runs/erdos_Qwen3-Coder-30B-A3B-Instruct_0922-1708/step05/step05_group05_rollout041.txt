import numpy as np
from scipy.stats import halton
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    num_generations = 600
    crossover_rate = 0.8
    initial_mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population using Halton sequence
    def initialize_population():
        population = []
        # Use Halton sequence for diversity
        halton_seq = halton.Halton(1)
        points = halton_seq.random(pop_size * n_points).reshape(pop_size, n_points)
        
        for i in range(pop_size):
            # Adjust points to [0,1] range, but with specific pattern
            h = points[i] * 0.8 + 0.1  # make it look like a step function
            # Normalize sum
            h = h * (n_points / 2) / np.sum(h)
            # Clip to bounds
            h = np.clip(h, 0, 1)
            population.append(h)
        
        # Add initial construction if available
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population[0] = h_init.copy()
            
        return population
    
    # Evaluate fitness
    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Tournament selection
    def tournament_selection(population, fitnesses):
        selected_idx = np.random.choice(len(population), tournament_size)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return population[best_idx]
    
    # Single-point crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        crossover_point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    # Mutation with adaptive rate
    def mutate(h, mutation_rate):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Gaussian perturbation
                delta = np.random.normal(0, 0.05 * (1 - h[i]) if h[i] < 0.5 else 0.05 * h[i])
                h_mut[i] += delta
        # Repair
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    # Initialize
    population = initialize_population()
    fitnesses = [evaluate_fitness(h) for h in population]
    
    # Track best
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # GA main loop
    mutation_rate = initial_mutation_rate
    for generation in range(num_generations):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism
        new_population = sorted_population[:elite_size]
        new_fitnesses = sorted_fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
            new_fitnesses.extend([evaluate_fitness(child1), evaluate_fitness(child2)])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        fitnesses = new_fitnesses[:pop_size]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_h = population[current_best_idx].copy()
            best_fitness = fitnesses[current_best_idx]
        
        # Mutation rate decay
        if generation % 10 == 0 and generation > 0:
            mutation_rate *= 0.99
    
    # Final refinement with local search on best solution
    local_best_h = best_h.copy()
    local_best_fitness = best_fitness
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        # Gradient estimation with finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        base_corr = np.correlate(local_best_h, 1 - local_best_h, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        improved = False
        for j in range(n_points):
            h_plus = local_best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[j] = (fitness_plus - base_fitness) / eps
            
            if fitness_plus < base_fitness:
                improved = True
                
        if not improved:
            break
            
        # Update with step size
        step_size = 0.005
        local_best_h -= step_size * grad
        local_best_h = np.clip(local_best_h, 0, 1)
        local_best_h = local_best_h * (n_points / 2) / np.sum(local_best_h)
        
        corr = np.correlate(local_best_h, 1 - local_best_h, mode='full')
        new_fitness = np.max(corr) * dx
        
        if new_fitness < local_best_fitness:
            local_best_fitness = new_fitness
        else:
            break
    
    # Final evaluation
    corr_final = np.correlate(local_best_h, 1 - local_best_h, mode='full')
    final_fitness = np.max(corr_final) * dx
    
    return (local_best_h.tolist(), final_fitness, n_points)
