import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Start with a simple gradient descent approach
    # Will use "golden section" search + local refinement
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project_to_feasible(h_vals):
        # Clip to [0,1] and rescale to sum to n_points/2
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    # Golden section search for step size
    def golden_section_search(h_current, grad, max_step=0.1, tol=1e-6):
        phi = (1 + np.sqrt(5)) / 2
        rho = 1 / phi
        
        a, b = 0.0, max_step
        h1 = a + (b - a) * (1 - rho)
        h2 = a + (b - a) * rho
        
        # Evaluate at h1 and h2
        new_h1 = project_to_feasible(h_current - h1 * grad)
        new_h2 = project_to_feasible(h_current - h2 * grad)
        
        f1 = compute_c5(new_h1)
        f2 = compute_c5(new_h2)
        
        while abs(b - a) > tol:
            if f1 < f2:
                b = h2
                h2 = h1
                f2 = f1
                h1 = a + (b - a) * (1 - rho)
                new_h1 = project_to_feasible(h_current - h1 * grad)
                f1 = compute_c5(new_h1)
            else:
                a = h1
                h1 = h2
                f1 = f2
                h2 = a + (b - a) * rho
                new_h2 = project_to_feasible(h_current - h2 * grad)
                f2 = compute_c5(new_h2)
        
        step = (a + b) / 2
        return step
    
    # Local search gradient descent approach
    max_iter = 3000
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    for i in range(max_iter):
        # Compute finite difference approximation of gradient
        grad = np.zeros_like(current_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - current_c5) / eps
        
        # Now optimize step size
        step = golden_section_search(current_h, grad, 0.05)
        
        # Update
        new_h = project_to_feasible(current_h - step * grad)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        if time.time() - start_time > budget_s - 10:
            break
            
        # Occasionally restart with random pertubation if no improvement
        if i % 100 == 0 and i > 0:
            if abs(current_c5 - best_c5) < 1e-6:
                # Random perturbation
                noise = np.random.normal(0, 1e-2, n_points)
                current_h = project_to_feasible(current_h + noise)
                current_c5 = compute_c5(current_h)
                
    # Final refinement - use a few more gradient steps with adaptive learning rate
    current_h = best_h.copy()
    current_c5 = best_c5
    lr = 0.001
    
    for i in range(200):
        # Compute finite difference gradient with better precision
        grad = np.zeros_like(current_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - current_c5) / eps
            
        # Update with adaptive learning rate
        new_h = project_to_feasible(current_h - lr * grad)
        new_c5 = compute_c5(new_h)
        
        if new_c5 < current_c5:
            current_h = new_h
            current_c5 = new_c5
            lr *= 1.05  # Gradually increase learning rate if we're doing well
        else:
            lr *= 0.9  # Reduce learning rate if there's no improvement
            
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        if time.time() - start_time > budget_s - 5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
