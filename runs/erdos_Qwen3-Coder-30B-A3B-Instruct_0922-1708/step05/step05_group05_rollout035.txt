import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find minimal correlation index
        min_idx = np.argmin(corr)
        k_min = min_idx - (len(corr) - 1) // 2
        
        # Compute gradient for the minimal correlation
        grad = np.zeros(n_points)
        
        # Compute gradient using finite differences
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h with gradient ascent
        h = h + lr * grad
        
        # Project h to feasible region [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbations to escape local minima
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration / 100.0)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
