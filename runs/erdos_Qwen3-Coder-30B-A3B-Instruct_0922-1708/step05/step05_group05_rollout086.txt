import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Create a simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to meet constraint sum(h) * dx = 1/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    # Parameters
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    max_iter = 1000
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlation and identify the minimal overlap
        corr = correlate(h, 1 - h, mode='full')
        min_k = np.argmin(corr)
        min_value = corr[min_k]
        
        # Compute gradient with respect to the minimal overlap
        # For correlation h * (1-h), the gradient d/dh[i] of corr[k] = h[i]*h[i-k] is:
        # 0 if k is not in the range where convolution happens
        # 1 if k = 0 (i.e., self-correlation, which is constant, so ignore)
        # For min_k where correlation is minimal, we calculate the derivative differently
        # Assume that we are at position k in the full correlation vector,
        # Compute the derivatives from the constituent values
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Compute the contribution to correlation[k] = sum h[j] * (1 - h[j-k])
            # So derivative wrt h[i] involves terms where h[j] is present with j = i + k
            # and also where h[j] appears with j = i - k - but must be careful of borders
            # But since we only care about the minimal point, we can do finite differences
            # Let's compute the direct gradient for the k that gives the minimum
            
            # Start a relaxed derivative estimate by finite differences
            eps = 1e-5
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Re-normalize to sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            min_val_plus = corr_plus[min_k]
            
            gradient[i] = (min_val_plus - min_value) / eps
        
        # Non-negativity and Lipschitz constraints
        # Update h using gradient ascent w.r.t. the minimum overlap (we want to minimize it)
        h = h - learning_rate * gradient
        
        # Project onto feasible set [0,1] and enforce sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Apply random perturbations periodically to escape local minima
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration / 100)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            # If we're stuck or making little progress, reduce learning rate
            if iteration > 100 and best_c5 > 0.381:
                learning_rate *= 0.9
            elif best_c5 < 0.38085:
                # If we're improving quickly, increase learning rate up to 0.01
                learning_rate = min(learning_rate * 1.1, 0.01)
                
    return (best_h.tolist(), best_c5, n_points)
