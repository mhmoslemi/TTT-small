from scipy.optimize import minimize
import time

    start_time = time.time()

    population_size = 200
    offspring_size = 200
    max_iter = 3000
    mutation_strength = 0.5
    mutation_variance = 0.1
    elite_fraction = 0.5
    stagnation_threshold = 200

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Create population around best solution with some perturbation
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Create random solutions
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Fitness evaluation for a single individual
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    best_solution = None
    best_fitness = float('inf')
    stagnation_counter = 0
    current_mutation_strength = mutation_strength
    current_mutation_variance = mutation_variance
    
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate fitness for all population members
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:

    # Final refinement with gradient-based method
    if best_solution is None:

    best_c5 = best_fitness
    
    for _ in range(100):
        corr = np.correlate(best_refined, 1 - best_refined, mode='full')
        fitness = np.max(corr) * dx

        eps = 1e-4
        for i in range(n_points):
            h_plus = best_refined.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)

        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:

        else:
            fitness = fitness_new
    
    # Final verification
    corr_final = np.correlate(best_refined, 1 - best_refined, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (best_refined.tolist(), min(c5_final, best_c5), n_points)
