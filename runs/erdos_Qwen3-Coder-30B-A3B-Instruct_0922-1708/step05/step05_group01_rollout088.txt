import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
    
    # Optimization parameters
    max_iter = 6000
    alpha = 0.01
    decay_rate = 0.5
    decay_interval = 200
    eval_freq = 50
    stagnation_limit = 200
    perturb_freq = 100
    perturb_fraction = 0.01
    
    last_improvement = 0
    stagnation_count = 0
    
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Periodic re-evaluation
        if i % eval_freq == 0:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = i
                stagnation_count = 0
            else:
                stagnation_count += 1
                
        # Check for stagnation
        if stagnation_count > stagnation_limit:
            # Perturb solution
            noise = np.random.normal(0, perturb_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            
        # Adaptive learning rate
        if i % decay_interval == 0 and i > 0:
            alpha *= decay_rate
            
        # Compute gradients via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_c5 = float('inf')
        
        # Get base correlation and value
        corr_base = correlate(h, 1 - h, mode='full')
        base_c5 = np.max(corr_base) * dx
        
        # Compute gradient approximation
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_c5) / eps
            
        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally random perturbation
        if i % perturb_freq == 0 and i > 0:
            noise = np.random.normal(0, perturb_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
