import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth max approximation parameter
    beta = 10.0
    
    # Coarse optimization
    coarse_points = 50
    dx_coarse = 2.0 / coarse_points
    h_coarse = np.interp(np.linspace(0, 2, coarse_points), 
                         np.linspace(0, 2, n_points), h)
    
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = float('inf')
    
    # Adam optimizer parameters
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(h_coarse)
    v = np.zeros_like(h_coarse)
    epsilon = 1e-8
    
    # Adam optimization for coarse grid
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and smooth max
        correlation = correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        smooth_max = (1.0 / beta) * np.log(np.sum(np.exp(beta * correlation)))
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h_coarse)
        eps = 1e-5
        
        for i in range(coarse_points):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (coarse_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            smooth_max_plus = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr_plus)))
            grad[i] = (smooth_max_plus - smooth_max) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(step + 1))
        v_hat = v / (1 - beta2**(step + 1))
        
        h_coarse_new = h_coarse - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (coarse_points / 2) / np.sum(h_coarse_new)
        
        h_coarse = h_coarse_new
        
        # Check if better
        corr = correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        c5 = np.max(corr)
        if c5 < best_c5_coarse:
            best_c5_coarse = c5
            best_h_coarse = h_coarse.copy()
    
    # Transfer to fine grid
    n_points = 200
    dx = 2.0 / n_points
    h = np.interp(np.linspace(0, 2, n_points), 
                  np.linspace(0, 2, coarse_points), best_h_coarse)
    
    # Fine optimization with Adam
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Reset Adam
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and smooth max
        correlation = correlate(h, 1 - h, mode='full') * dx
        smooth_max = (1.0 / beta) * np.log(np.sum(np.exp(beta * correlation)))
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full') * dx
            smooth_max_plus = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr_plus)))
            grad[i] = (smooth_max_plus - smooth_max) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(step + 1))
        v_hat = v / (1 - beta2**(step + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check if better
        corr = correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final check
    final_corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    
    # Ensure constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
