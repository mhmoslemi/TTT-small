import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 1000
    no_improvement_iter = 0
    patience = 200
    
    # Main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new solution by perturbing current best
        new_h = best_h.copy()
        # Randomly select a subset of indices to perturb
        num_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            new_h[i] += np.random.normal(0, 0.01)
        
        # Repair and rescale
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improvement_iter = 0
        else:
            no_improvement_iter += 1
        
        # Local search: gradient descent on top performing solutions
        if iter_count % 20 == 0:
            # Small gradient descent pass
            h_local = best_h.copy()
            for _ in range(10):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / epsilon
                h_local = h_local - 0.0005 * grad
                h_local = repair_h(h_local, n_points)
                corr = correlate(h_local, 1 - h_local, mode='full')
                c5_local = np.max(corr) * dx
                if c5_local < best_c5:
                    best_h = h_local.copy()
                    best_c5 = c5_local
                    no_improvement_iter = 0
                    
        # Early stopping if no improvement
        if no_improvement_iter >= patience:
            break
    
    # Final refinement
    final_h = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / epsilon
        final_h = final_h - 0.0001 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (final_h.tolist(), best_c5, n_points)
