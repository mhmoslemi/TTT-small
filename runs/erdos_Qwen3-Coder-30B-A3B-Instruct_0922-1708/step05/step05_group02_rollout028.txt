import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_h = repair_h(current_h, n_points)
    current_c5 = evaluate_h(current_h, dx)
    
    # Simulated Annealing parameters
    T = 0.1 * current_c5 if current_c5 > 0 else 0.1
    alpha = 0.99
    max_iter = 10000
    patience = 100
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        iter_count += 1
        
        # Generate neighbor by perturbing a subset of elements
        neighbor_h = current_h.copy()
        num_perturb = max(1, int(0.1 * n_points))  # Perturb about 10% of elements
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            # Perturb by a small amount
            delta = np.random.uniform(-0.05, 0.05)
            neighbor_h[i] += delta
        
        # Repair neighbor
        neighbor_h = repair_h(neighbor_h, n_points)
        
        # Evaluate neighbor
        neighbor_c5 = evaluate_h(neighbor_h, dx)
        
        # Accept or reject
        delta = neighbor_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling schedule
        T *= alpha
        
        # Adaptive cooling if no improvement
        if no_improvement_count > patience:
            T *= 0.95
            no_improvement_count = 0
    
    # Final gradient descent refinement
    h_ref = best_h.copy()
    c5_ref = best_c5
    
    for _ in range(1000):
        if (time.time() - start_time) > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_ref) / eps
            
        h_ref = h_ref - 0.0005 * grad
        h_ref = repair_h(h_ref, n_points)
        corr = correlate(h_ref, 1 - h_ref, mode='full')
        c5_ref = np.max(corr) * dx
        
        if c5_ref < best_c5:
            best_h = h_ref.copy()
            best_c5 = c5_ref
    
    return (best_h.tolist(), best_c5, n_points)
