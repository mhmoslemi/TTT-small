import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    mutation_decay = 0.99
    
    # Initialize Halton sequence for better exploration
    sampler = qmc.Halton(d=n_points, seed=seed)
    samples = sampler.random(n=population_size)
    
    # Convert to step function values between 0 and 1
    population = []
    for i in range(population_size):
        # Transform to [0,1] using inverse CDF of uniform
        # Taking the first n_points elements and scaling
        h = np.interp(samples[i], [0, 1], [0, 1])
        h = np.clip(h, 0, 1)
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # If we have initial construction, use it
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population[0] = h_init.copy()
    
    start_time = time.time()
    
    best_fitness = float('inf')
    best_individual = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitness_scores.append(fitness)
            
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = h.copy()
        
        # Tournament selection
        new_population = []
        
        # Elitism: keep the best individuals
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parents = []
            for _ in range(2):
                tournament = np.random.choice(population_size, tournament_size)
                winner = tournament[np.argmin([fitness_scores[i] for i in tournament])]
                parents.append(population[winner].copy())
            
            # Crossover
            if np.random.random() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parents[0][:crossover_point], parents[1][crossover_point:]])
                child2 = np.concatenate([parents[1][:crossover_point], parents[0][crossover_point:]])
            else:
                child1 = parents[0].copy()
                child2 = parents[1].copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.random() < mutation_rate:
                    # Gaussian mutation
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise
                    # Repair constraints
                    child = np.clip(child, 0, 1)
                    # Re-normalize
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
        
        # Keep only the population size
        population = new_population[:population_size]
        
        # Adjust mutation rate
        if gen % 10 == 0:
            mutation_rate *= mutation_decay
    
    # Final refinement using local search on best individual
    if best_individual is not None:
        h_refined = best_individual.copy()
        
        # Gradient-based refinement (simple finite differences)
        for iter_ref in range(500):
            grad = np.zeros(n_points)
            eps = 1e-5
            
            base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            base_fitness = np.max(base_corr) * dx
            
            improved = False
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                
                grad[j] = (fitness_plus - base_fitness) / eps
                
                if fitness_plus < base_fitness:
                    improved = True
            
            if not improved:
                break
                
            h_refined -= 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            new_fitness = np.max(corr) * dx
            
            if new_fitness < base_fitness:
                base_fitness = new_fitness
            else:
                break
        
        # Recompute final evaluation
        final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(final_corr) * dx
        
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_individual = h_refined.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
