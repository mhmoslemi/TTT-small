import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_factor = 1.2
    penalty_min = 1e-4
    penalty_initial = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize penalty
    penalty_weight = penalty_initial
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function to compute overlap integral
    def compute_c5(h_vals):
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Helper function to compute gradient
    def compute_gradient(h_vals, k_max):
        # Create gradient for the maximum correlation lag
        grad = np.zeros_like(h_vals)
        
        # For the k_max lag, compute the gradient
        n = len(h_vals)
        if k_max >= 0:
            # h[i] * (1 - h[i + k_max]) term contributes to gradient
            for i in range(n - k_max):
                grad[i] += 1 - h_vals[i + k_max]
                if i + k_max < n:
                    grad[i + k_max] -= h_vals[i]
        else:
            # h[i] * (1 - h[i - k_max]) term contributes to gradient
            k_abs = -k_max
            for i in range(n - k_abs):
                grad[i] += 1 - h_vals[i - k_abs]
                if i - k_abs < n:
                    grad[i - k_abs] -= h_vals[i]
        
        return grad * dx
    
    # Helper function for feasibility repair
    def repair_feasibility(h_vals):
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to sum to n/2
        h_sum = np.sum(h_vals)
        if abs(h_sum - n_points/2) > 1e-8:
            h_vals = h_vals * (n_points / 2) / h_sum
        return h_vals
    
    # Initialize current solution
    h = repair_feasibility(h)
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization loop
    last_improvement = 0
    current_lr = learning_rate
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute all correlations to find the maximum
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        k_max = max_lag_idx - (n_points - 1)  # map from full to k range
        
        # Compute gradient
        gradient = compute_gradient(h, k_max)
        
        # Compute penalty gradient
        penalty_grad = np.zeros_like(h)
        violated_lb = np.maximum(0, -h)
        violated_ub = np.maximum(0, h - 1)
        violated_sum = np.maximum(0, np.sum(h) - n_points/2)
        penalty_grad += 2 * penalty_weight * violated_lb
        penalty_grad += 2 * penalty_weight * violated_ub
        penalty_grad += 2 * penalty_weight * violated_sum * np.ones_like(h) / n_points
        
        # Combine gradients
        total_grad = gradient + penalty_grad
        
        # Perform line search for step size (simple forward difference)
        best_step_size = 0.0
        best_c5_val = c5
        
        step_sizes = [1e-5, 1e-4, 1e-3, 5e-3, 1e-2]
        for step_size in step_sizes:
            h_test = h - step_size * total_grad
            h_test = repair_feasibility(h_test)
            test_c5 = compute_c5(h_test)
            if test_c5 < best_c5_val:
                best_c5_val = test_c5
                best_step_size = step_size
        
        # Update h
        h = h - best_step_size * total_grad
        h = repair_feasibility(h)
        
        # Recompute c5
        c5 = compute_c5(h)
        
        # Update penalty weight
        constraint_violations = np.sum(np.maximum(0, -h)) + \
                               np.sum(np.maximum(0, h - 1)) + \
                               np.maximum(0, np.sum(h) - n_points/2)
        
        if constraint_violations > 1e-6:
            penalty_weight = penalty_factor * penalty_weight
        else:
            penalty_weight = max(penalty_min, penalty_weight / penalty_factor)
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early stopping and perturbation
        if iteration - last_improvement > plateau_patience:
            current_lr *= learning_rate_decay
            if current_lr < 1e-6:
                break
            # Add some random perturbations to avoid local minima
            perturbation = np.random.normal(0, perturbation_fraction * 0.1, n_points)
            h = np.clip(h + perturbation, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
