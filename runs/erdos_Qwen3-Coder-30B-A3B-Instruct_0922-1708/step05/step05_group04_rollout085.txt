import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 2000
    F = 0.8
    CR = 0.9
    mutation_decay = 0.99
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        diff = h_sum - n_points / 2
        if abs(diff) < 1e-10:
            return h
        if diff > 0:  # Excess, remove from largest
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:  # Deficit, add to smallest
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial population
    fitnesses = [evaluate_fitness(p) for p in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # DE main loop
    current_iter = 0
    stagnation_count = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        
        for i in range(population_size):
            # Select three different individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair
            mutant = repair(mutant)
            
            # Crossover
            trial = np.zeros(n_points)
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Repair trial
            trial = repair(trial)
            
            # Selection
            trial_fitness = evaluate_fitness(trial)
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
                fitnesses[i] = trial_fitness
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Adapt mutation factor
        if current_iter % 50 == 0:
            F *= mutation_decay
            
        # Stagnation control
        if stagnation_count > 100:
            # Reinitialize some particles
            for _ in range(population_size // 10):
                idx = np.random.randint(0, population_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                population[idx] = h
            stagnation_count = 0
            
        current_iter += 1
    
    # Final gradient-based refinement on best solution
    def gradient_descent_refinement(h_start, max_iter=100):
        h = h_start.copy()
        for _ in range(max_iter):
            fitness = evaluate_fitness(h)
            
            # Estimate gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair(h_plus)
                fitness_plus = evaluate_fitness(h_plus)
                grad[j] = (fitness_plus - fitness) / eps
            
            # Update step
            h -= 0.001 * grad
            h = repair(h)
            
            # Early stopping if no improvement
            new_fitness = evaluate_fitness(h)
            if new_fitness >= fitness:
                break
        return h
    
    # Refine the best solution found
    refined_h = gradient_descent_refinement(best_h)
    final_fitness = evaluate_fitness(refined_h)
    
    if final_fitness < best_fitness:
        best_h = refined_h
        best_fitness = final_fitness
    
    return (best_h.tolist(), best_fitness, n_points)
