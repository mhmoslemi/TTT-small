import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    dx_start = 2.0 / n_start
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
        
    # Normalize coarse solution
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Refine coarse to fine
    x_coarse = np.linspace(0, 2, n_start)
    x_fine = np.linspace(0, 2, n_points)
    h = np.interp(x_fine, x_coarse, h_coarse)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Optimization variables for fine grid
    h_var = cp.Variable(n_points)
    C5 = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Use a quadratic penalty approach to handle the C5 constraint
    # Define a quadratic penalty on overlap constraint
    penalty = 1e-4
    objective = cp.Minimize(C5 + penalty * cp.sum_squares(h_var - h))
    
    # Solve the QP relaxation to get good initial guess
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=ECOS, max_iter=1000, abstol=1e-6, reltol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_solution = h_var.value
            if h_solution is not None:
                h = np.array(h_solution)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
    except Exception:
        pass  # Use current h if optimization fails
    
    # Gradient descent refinement
    best_h = h.copy()
    best_c5 = float('inf')
    last_c5 = float('inf')
    patience = 0
    max_patience = 300
    lr = 1e-3
    
    # Compute correlation only once for efficiency
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    iterations = 0
    max_iter = 10000
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        if abs(current_c5 - last_c5) < 1e-6:
            lr = max(1e-6, lr * 0.95)
        
        last_c5 = current_c5
        iterations += 1
        
    # Final validation and cleanup
    final_c5 = compute_c5(best_h)
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
