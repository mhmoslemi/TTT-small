import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize parameters
    alpha = 1e-3
    penalty_mu = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Track best solution
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Main optimization loop
    iteration = 0
    no_improvement_count = 0
    last_improvement = 0
    
    while (time.time() - start_time) < budget_s - 5 and iteration < 5000:
        iteration += 1
        
        # Compute correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient with respect to h for the maximum lag
        grad = np.zeros(n_points)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Gradient computation
        if max_lag > 0:
            for i in range(n_points - max_lag):
                # Derivative of h[i] * (1 - h[i + max_lag]) w.r.t. h[i]
                grad[i] += 1 - h[i + max_lag]
        elif max_lag < 0:
            for i in range(-max_lag, n_points):
                # Derivative of h[i] * (1 - h[i + max_lag]) w.r.t. h[i]
                grad[i] += 1 - h[i + max_lag]
        else:  # max_lag == 0
            for i in range(n_points):
                grad[i] += 1 - 2 * h[i]
        
        # Add penalties for constraint violations
        penalty_grad = np.zeros(n_points)
        sum_violation = np.sum(h) - n_points / 2
        
        for i in range(n_points):
            # Gradient for bound constraints
            if h[i] < 0:
                penalty_grad[i] += -2 * penalty_mu * h[i]
            elif h[i] > 1:
                penalty_grad[i] += -2 * penalty_mu * (h[i] - 1)
            
            # Gradient for sum constraint
            penalty_grad[i] += 2 * penalty_mu * sum_violation
        
        # Compute total gradient
        total_grad = grad * dx + penalty_grad
        
        # Line search for step size
        step_size = alpha
        h_new = h - step_size * total_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new objective
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Adjust step size if needed
        while c5_new >= c5 and step_size > 1e-10:
            step_size *= 0.5
            h_new = h - step_size * total_grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
        
        # Update h
        h = h_new.copy()
        
        # Update penalty parameter
        sum_violation = np.sum(h) - n_points / 2
        constraint_violations = np.sum(np.maximum(0, -h)) + np.maximum(0, sum_violation)**2
        if constraint_violations > 1e-6:
            penalty_mu = penalty_mu * penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_min)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Rejuvenation if no improvement for plateau_patience iterations
        if no_improvement_count >= plateau_patience:
            no_improvement_count = 0
            alpha *= learning_rate_decay
            # Apply random perturbations
            perturb_mask = np.random.rand(n_points) < perturbation_fraction
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
