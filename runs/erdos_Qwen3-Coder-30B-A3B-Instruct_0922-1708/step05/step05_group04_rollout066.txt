import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    tolerance = 1e-5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Normalize sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Binary search for optimal C5 bound
    low, high = 0.0, 0.5
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper functions
    def compute_overlap(h_vec):
        # Compute full convolution using numpy
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h_vec):
        # Clip and renormalize
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    # Binary search iterations
    for _ in range(20):
        if time.time() - start_time > budget_s - 10:
            break
            
        mu = (low + high) / 2
        c5_bound = compute_overlap(h)
        
        # Check if we can improve further
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
            
        # Try to improve with gradient descent
        h_opt = h.copy()
        
        for iter_count in range(100):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute current correlation
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            current_c5 = np.max(corr) * dx
            
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - current_c5) / eps
            
            # Update using gradient descent
            h_opt -= 0.001 * grad
            h_opt = repair_h(h_opt)
            
            # Early stopping if no improvement
            new_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 >= current_c5:
                break
                
        h = h_opt.copy()
        c5_bound = compute_overlap(h)
        
        if c5_bound < mu:
            high = mu
        else:
            low = mu
            
    # Final refinement
    final_h = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        corr = np.correlate(final_h, 1 - final_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = final_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - current_c5) / eps
        
        final_h -= 0.001 * grad
        final_h = repair_h(final_h)
        
        # Early stopping
        new_corr = np.correlate(final_h, 1 - final_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        if new_c5 >= current_c5:
            break
    
    final_corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
