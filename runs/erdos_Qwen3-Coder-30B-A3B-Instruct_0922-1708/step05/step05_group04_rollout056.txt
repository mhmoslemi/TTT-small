        # Add some random perturbations

            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)

    best_h = None
    best_c5 = float('inf')
    
    # Evolution strategy main loop
    generation = 0
    no_improvement_count = 0
    mutation_strength = mutation_strength_initial
    
    while generation < max_generations and time.time() - start_time < budget_s - 10:
        # Evaluate fitness for population
        fitness_values = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * dx
            fitness_values.append(fitness)
        
        # Track best solution
        current_best_idx = np.argmin(fitness_values)
        current_best_fitness = fitness_values[current_best_idx]
        if current_best_fitness < best_c5:
            best_c5 = current_best_fitness
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Selection: Elite 50%
        selected_indices = np.argsort(fitness_values)[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            if len(selected_population) > 0:
                parent = selected_population[np.random.randint(0, len(selected_population))]
                # Gaussian mutation
                mutation = np.random.normal(0, mutation_strength, n_points)
                child = parent + mutation
                # Repair
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                child = child * (n_points / 2) / child_sum
                offspring.append(child)
            else:
                # Fallback to random
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                offspring.append(h)

        worst_indices = np.argsort(fitness_values)[population_size // 2:]

            no_improvement_count = 0

    # Final refinement with gradient descent
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

        eps = 1e-4
        grad = np.zeros(n_points)
        base_fitness = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            fitness_plus = compute_c5(h_plus)
            grad[i] = (fitness_plus - base_fitness) / eps
        return grad

        h_refine_sum = np.sum(h_refine)
        h_refine = h_refine * (n_points / 2) / h_refine_sum
        if compute_c5(h_refine) >= compute_c5(best_h):
            break
    
    # Final computation
    best_h = h_refine.copy()
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
