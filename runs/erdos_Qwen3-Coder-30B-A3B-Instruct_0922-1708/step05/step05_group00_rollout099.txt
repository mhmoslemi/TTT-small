import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        # Create a smooth, balanced starting function
        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x) 
        h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse grid optimization (n=50)
    n_coarse = 50
    x_coarse = np.linspace(0, 2, n_coarse)
    x_fine = np.linspace(0, 2, n_points)
    
    h_coarse = np.interp(x_coarse, x_fine, h)
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Smooth approximation to max
    def smooth_max(x, beta=100):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimizer with Adam
    def optimize_smooth(h_start, n_steps=2000):
        h_current = h_start.copy()
        beta = 100
        lr = 0.01
        momentum = 0.9
        velocity = np.zeros_like(h_current)
        
        for i in range(n_steps):
            # Evaluate function and its gradient (finite differences)
            correlation = np.correlate(h_current, 1 - h_current, mode='full') * (2.0/n_points)
            c5_val = np.max(correlation)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h[:] = h_current[:]
                
            # Use gradient update with Adam
            grad = np.zeros_like(h_current)
            eps = 1e-5
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_coarse / 2) / np.sum(h_plus)
                c_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0/n_points))
                grad[j] = (c_plus - c5_val) / eps
            
            # Adam update
            velocity = momentum * velocity + (1 - momentum) * grad
            h_current = h_current - lr * velocity
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_coarse / 2) / np.sum(h_current)
            
            # Anneal learning rate
            lr = max(1e-6, lr * 0.9995)
            
            if time.time() - start_time > budget_s - 1:
                break
                
        return h_current
    
    h_optimized = optimize_smooth(h_coarse)
    
    # Upsample to fine grid
    h_fine = np.interp(x_fine, x_coarse, h_optimized)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Final fine-tune
    def finalize(h_fine_start, n_steps=2000):
        h_current = h_fine_start.copy()
        beta = 100
        lr = 0.005
        momentum = 0.95
        velocity = np.zeros_like(h_current)
        
        for i in range(n_steps):
            # Gradient evaluation
            correlation = np.correlate(h_current, 1 - h_current, mode='full') * (2.0/n_points)
            c5_val = np.max(correlation)
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h[:] = h_current[:]
            
            # Gradient
            grad = np.zeros_like(h_current)
            eps = 1e-4
            for j in range(len(h_current)):
                h_plus = h_current.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0/n_points))
                grad[j] = (c_plus - c5_val) / eps
            
            # Adam with momentum
            velocity = momentum * velocity + (1 - momentum) * grad
            h_current = h_current - lr * velocity
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
            lr = max(1e-6, lr * 0.9997)
            
            if time.time() - start_time > budget_s - 1:
                break
                
        return h_current
    
    final_h = finalize(h_fine)
    
    # Final computation
    correlation = np.correlate(final_h, 1 - final_h, mode='full') * (2.0/n_points)
    final_c5 = np.max(correlation)
    
    return (final_h.tolist(), final_c5, n_points)
