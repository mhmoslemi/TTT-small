import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 6000
    dx = 2.0 / n_points
    alpha = 0.01
    decay_rate = 0.5
    stagnation_tolerance = 200
    perturbation_freq = 100
    adjustment_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize correlation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    iter_since_improvement = 0
    last_alpha = alpha
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Adaptive learning rate
        if iteration % 200 == 0 and iteration > 0:
            alpha *= decay_rate
            last_alpha = alpha
            
        # Perturbation
        if iteration % perturbation_freq == 0 and iteration > 0:
            # Randomly perturb 1% of values
            mask = np.random.random(n_points) < 0.01
            h[mask] += np.random.normal(0, alpha * 0.1, np.sum(mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Compute new correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        if iter_since_improvement > stagnation_tolerance:
            # Reset if stuck
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                if len(h) != n_points:
                    x_orig = np.linspace(0, 2, len(h))
                    x_target = np.linspace(0, 2, n_points)
                    h = np.interp(x_target, x_orig, h)
                h = h * (n_points / 2) / np.sum(h)
            else:
                h = np.ones(n_points) * 0.5
            iter_since_improvement = 0
            # Reset learning rate
            alpha = last_alpha
            
        # Adjust h to reduce maximum overlap
        if iteration % adjustment_freq == 0:
            # Find the maximum overlap lag and its contribution regions
            max_lag_idx = np.argmax(corr)
            
            # Find regions contributing to max overlap
            # The maximum overlap is at lag: max_lag_idx - (n_points - 1)
            lag = max_lag_idx - (n_points - 1)
            
            # Calculate gradient approximation for each position
            grad = np.zeros(n_points)
            eps = 1e-5
            
            for i in range(n_points):
                # Perturb h[i] up and down
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1.0
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h.copy()
                h_minus[i] -= eps
                if h_minus[i] < 0:
                    h_minus[i] = 0.0
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                
                corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
                
            # Apply the update
            h_new = h - alpha * grad
            
            # Repair feasibility
            h_new = np.clip(h_new, 0, 1)
            h_sum = np.sum(h_new)
            target_sum = n_points / 2
            
            if abs(h_sum - target_sum) > 1e-8:
                # Rescale proportionally
                h_new = h_new * (target_sum / h_sum)
            
            h = h_new
            
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
