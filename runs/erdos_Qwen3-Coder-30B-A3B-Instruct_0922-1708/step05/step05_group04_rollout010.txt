import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 1000
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover probability
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    def evaluate(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    fitness_values = [evaluate(p) for p in population]
    
    # Track best solution
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx].copy()
    best_fitness = fitness_values[best_idx]
    
    # Main DE loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Adjust mutation factor
        if iteration % 50 == 0 and iteration > 0:
            F *= 0.99
            
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutate
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to maintain constraint
            h_sum = np.sum(mutant)
            if abs(h_sum - n_points / 2) > 1e-10:
                diff = h_sum - n_points / 2
                if diff > 0:
                    # Remove excess
                    indices = np.argsort(mutant)[::-1][:int(diff/dx)]
                    for idx in indices:
                        if mutant[idx] > 0 and diff > 0:
                            dec = min(mutant[idx], diff)
                            mutant[idx] -= dec
                            diff -= dec
                else:
                    # Add deficit
                    indices = np.argsort(mutant)[:int(-diff/dx)]
                    for idx in indices:
                        if mutant[idx] < 1 and diff < 0:
                            inc = min(1 - mutant[idx], -diff)
                            mutant[idx] += inc
                            diff += inc
                # Rescale to exact sum
                mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Evaluate trial
            trial_fitness = evaluate(trial)
            
            # Selection
            if trial_fitness < fitness_values[i]:
                new_population.append(trial)
                fitness_values[i] = trial_fitness
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update global best
        current_best_idx = np.argmin(fitness_values)
        if fitness_values[current_best_idx] < best_fitness:
            best_fitness = fitness_values[current_best_idx]
            best_h = population[current_best_idx].copy()
    
    # Final local refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Simple gradient descent as local refinement
    h_refined = best_h.copy()
    for _ in range(200):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        current_max = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            # Repair
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-10:
                diff = h_plus_sum - n_points / 2
                if diff > 0:
                    indices = np.argsort(h_plus)[::-1][:int(diff/dx)]
                    for idx in indices:
                        if h_plus[idx] > 0 and diff > 0:
                            dec = min(h_plus[idx], diff)
                            h_plus[idx] -= dec
                            diff -= dec
                else:
                    indices = np.argsort(h_plus)[:int(-diff/dx)]
                    for idx in indices:
                        if h_plus[idx] < 1 and diff < 0:
                            inc = min(1 - h_plus[idx], -diff)
                            h_plus[idx] += inc
                            diff += inc
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - current_max) / eps
        
        h_refined -= 0.0005 * grad
        # Repair
        h_refined = np.clip(h_refined, 0, 1)
        h_refined_sum = np.sum(h_refined)
        if abs(h_refined_sum - n_points / 2) > 1e-10:
            diff = h_refined_sum - n_points / 2
            if diff > 0:
                indices = np.argsort(h_refined)[::-1][:int(diff/dx)]
                for idx in indices:
                    if h_refined[idx] > 0 and diff > 0:
                        dec = min(h_refined[idx], diff)
                        h_refined[idx] -= dec
                        diff -= dec
            else:
                indices = np.argsort(h_refined)[:int(-diff/dx)]
                for idx in indices:
                    if h_refined[idx] < 1 and diff < 0:
                        inc = min(1 - h_refined[idx], -diff)
                        h_refined[idx] += inc
                        diff += inc
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
    
    # Final evaluation
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_fitness, n_points)
