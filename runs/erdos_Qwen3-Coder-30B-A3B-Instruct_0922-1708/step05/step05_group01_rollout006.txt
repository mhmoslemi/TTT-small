import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations
        corr_full = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximizing lag
        max_lag_idx = np.argmax(corr_full)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        
        # For the max lag, compute gradient contribution
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] += (1 - h[i + max_lag]) * dx
                grad[i + max_lag] += -h[i] * dx
        else:
            for i in range(-max_lag, n_points):
                grad[i] += (1 - h[i + max_lag]) * dx
                grad[i + max_lag] += -h[i] * dx
        
        # Line search
        step_size = alpha
        while step_size > 1e-10:
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new
                break
            else:
                step_size *= line_search_factor
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping based on gradient magnitude
        grad_norm = np.linalg.norm(grad)
        if grad_norm < grad_threshold:
            break
    
    # Final evaluation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
