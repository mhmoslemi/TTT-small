import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import dct, idct
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert initial to Fourier coefficients (DCT-II)
    h_init_dct = dct(h_init, type=2, norm='ortho')[:n_fourier]
    h_init_dct = h_init_dct.copy()  # Ensure it's writable
    current_coeffs = h_init_dct.copy()
    
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
    
    # Gradient-based optimization
    def compute_c5_and_grad(coeffs):
        # Convert coefficients to function values
        h = idct(coeffs, type=2, norm='ortho')
        h = h[:n_points]
        
        # Ensure boundedness
        h = np.clip(h, 0, 1)
        
        # Rescale
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient
        grad = np.zeros_like(coeffs)
        eps = 1e-5
        base_c5 = c5
        
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = idct(coeffs_plus, type=2, norm='ortho')
            h_plus = h_plus[:n_points]
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - base_c5) / eps
        
        return c5, grad
    
    # Optimization parameters
    max_iter = 500
    learning_rate = 0.01
    patience = 100
    patience_counter = 0
    
    current_coeffs = h_init_dct.copy()
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        c5, grad = compute_c5_and_grad(current_coeffs)
        
        # Update using gradient descent
        current_coeffs -= learning_rate * grad
        
        # Projection back to reasonable range
        current_coeffs = np.clip(current_coeffs, -10, 10)
        
        # Evaluate solution
        h = idct(current_coeffs, type=2, norm='ortho')
        h = h[:n_points]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5_candidate = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
    
    # Final refinement
    current_h = best_h.copy()
    current_c5 = best_c5
    
    # Evaluate local gradient and refine
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        correlation = np.correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(correlation) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - c5) / eps
            
        # Update
        current_h -= 0.001 * grad
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        new_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
        if new_c5 >= c5:
            break
        else:
            c5 = new_c5
            
    final_c5 = np.max(np.correlate(current_h, 1 - current_h, mode='full')) * dx
    
    return (current_h.tolist(), final_c5, n_points)
