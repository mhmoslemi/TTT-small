import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Learning rate and parameters
    alpha = 1e-3
    line_search_factor = 0.5
    min_alpha = 1e-8
    stagnation_threshold = 1e-6
    max_iter = 5000
    
    def compute_c5(h_vec):
        # Compute correlation for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Max over all lags
        c5 = np.max(corr) * dx
        return c5
    
    def compute_gradient(h_vec, k_star):
        # Compute gradient for lag k*
        grad = np.zeros_like(h_vec)
        
        # Determine overlap indices
        if k_star >= 0:
            # h[i] * (1 - h[i + k_star])
            for i in range(len(h_vec) - k_star):
                grad[i] += (1 - h_vec[i + k_star]) * dx
                grad[i + k_star] -= h_vec[i] * dx
        else:
            # h[i - k_star] * (1 - h[i])
            k_abs = -k_star
            for i in range(k_abs, len(h_vec)):
                grad[i] += (1 - h_vec[i - k_abs]) * dx
                grad[i - k_abs] -= h_vec[i] * dx
        
        return grad
    
    def project_h(h_vec):
        # Clip values to [0, 1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to sum to n_points/2
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Initial evaluation
    c5 = compute_c5(h)
    best_c5 = c5
    best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Find the lag that gives the maximum overlap
        k_star = np.argmax(corr) - (len(h) - 1)
        max_val = corr[np.argmax(corr)]
        
        # Compute gradient
        grad = compute_gradient(h, k_star)
        
        # Line search for step size
        step_alpha = alpha
        prev_c5 = c5
        
        while step_alpha >= min_alpha:
            h_new = h - step_alpha * grad
            h_new = project_h(h_new)
            c5_new = compute_c5(h_new)
            
            if c5_new < c5:
                h = h_new.copy()
                c5 = c5_new
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h.copy()
                alpha = min(1.0, alpha * 1.1)  # Adapt step size
                break
            else:
                step_alpha *= line_search_factor
                if step_alpha < min_alpha:
                    # If no improvement even with small steps, reduce alpha and try again
                    alpha = max(alpha * 0.5, min_alpha)
                    break
        
        # Stagnation check
        if abs(prev_c5 - c5) < stagnation_threshold:
            alpha = max(alpha * 0.5, min_alpha)
    
    return (best_h.tolist(), best_c5, n_points)
