import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with available initial_h_values if present
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Random initialization
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    # Evaluate initial solution
    current_c5 = evaluate_h(best_h, dx)
    best_c5 = current_c5
    
    # Annealing parameters
    initial_temp = 0.1 * current_c5 if current_c5 > 0 else 0.01
    temperature = initial_temp
    cooling_rate = 0.99
    min_temp = 1e-8
    
    # Parameters
    max_iter = 10000
    perturb_rate = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Perturb h
        perturb_count = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, size=perturb_count, replace=False)
        new_h = best_h.copy()
        
        for idx in indices:
            # Add Gaussian noise
            new_h[idx] += np.random.normal(0, 0.05)
        
        # Repair to constraints
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            current_c5 = new_c5
            best_h = new_h.copy()
            if new_c5 < best_c5:
                best_c5 = new_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        
        # Cool down
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
            
        # Adaptive restart
        if stagnation_count > max_stagnation:
            # Restart with a new random solution
            new_h = np.random.rand(n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            current_c5 = evaluate_h(new_h, dx)
            best_h = new_h.copy()
            if current_c5 < best_c5:
                best_c5 = current_c5
            stagnation_count = 0
        
        # Adjust perturbation rate
        if iteration % 100 == 0 and perturb_rate > 0.01:
            perturb_rate *= 0.95

    # Final refinement using gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 2:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h.tolist(), best_c5, n_points)
