import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Starting resolution
    n_start = 50
    n_points = 200  # Final resolution
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse grid optimization with Adam
    def compute_correlation(h_vec):
        return np.correlate(h_vec, 1 - h_vec, mode='full') * dx

    def compute_gradient(h_vec, eps=1e-4):
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            c5_plus = np.max(compute_correlation(h_plus))
            grad[i] = (c5_plus - np.max(compute_correlation(h_vec))) / eps
        return grad

    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    lr = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    
    # Coarse grid optimization loop
    iterations = 0
    max_iter_coarse = 2000
    
    while iterations < max_iter_coarse and time.time() - start_time < budget_s - 1:
        # Compute current value and gradient
        current_c5 = np.max(compute_correlation(h))
        grad = compute_gradient(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if abs(current_c5 - best_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iterations + 1))
        v_hat = v / (1 - beta2**(iterations + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h = h_new
        iterations += 1
    
    # Transfer to fine grid
    if len(best_h) != n_points:
        x_coarse = np.linspace(0, 2, n_start)
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, best_h)
    else:
        h_fine = best_h.copy()
        
    # Normalize
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization with Adam
    h = h_fine
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    lr = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    
    # Fine grid optimization loop
    iterations = 0
    max_iter_fine = 2000
    
    while iterations < max_iter_fine and time.time() - start_time < budget_s - 1:
        # Compute current value and gradient
        current_c5 = np.max(compute_correlation(h))
        grad = compute_gradient(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if abs(current_c5 - best_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iterations + 1))
        v_hat = v / (1 - beta2**(iterations + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        
        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        iterations += 1
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
