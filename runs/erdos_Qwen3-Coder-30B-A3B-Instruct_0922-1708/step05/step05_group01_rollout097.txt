import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    learning_rate = 1e-3
    penalty_weight = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    iteration = 0
    last_improvement = 0
    current_penalty = penalty_weight
    current_lr = learning_rate
    
    while (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        c5 = np.max(corr) * dx
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute gradient w.r.t. h for the maximum correlation
        max_k = max_corr_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Construct gradient: for each position compute partial derivative of max correlation
        # This implementation approximates the gradient by finite differences
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Compute penalty gradient
        penalty_grad = np.zeros(n_points)
        violation_sum = np.sum(h) - n_points / 2
        if violation_sum > 0:
            penalty_grad += 2 * current_penalty * violation_sum
        else:
            # If below the target sum, gradients w.r.t. h elements are 0 for sum constraint
            pass  # No gradient contribution for sum constraint when under the limit
        
        # Clip h and adjust for bounds
        for i in range(n_points):
            if h[i] < 0:
                penalty_grad[i] -= 2 * current_penalty * h[i]
            elif h[i] > 1:
                penalty_grad[i] -= 2 * current_penalty * (h[i] - 1)
        
        # Combine gradients
        total_grad = grad + penalty_grad
        
        # Update h using line search or fixed step
        h_new = h - current_lr * total_grad
        
        # Apply feasibility repair
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum - n_points / 2) > 1e-8:
            h_new = h_new * (n_points / 2) / h_sum
        
        h = h_new
        
        # Update penalty weight based on constraint violations
        violation_sum = np.sum(h) - n_points / 2
        if abs(violation_sum) > 1e-4:
            current_penalty *= penalty_update_factor
        else:
            current_penalty = max(current_penalty / penalty_update_factor, penalty_min)
            
        # Check for stagnation and apply learning rate decay
        if iteration - last_improvement > plateau_patience:
            current_lr *= learning_rate_decay
            # Perturb solution
            perturb_indices = np.random.choice(n_points, size=int(n_points * perturbation_fraction), replace=False)
            for idx in perturb_indices:
                h[idx] += (np.random.rand() - 0.5) * 0.01
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
