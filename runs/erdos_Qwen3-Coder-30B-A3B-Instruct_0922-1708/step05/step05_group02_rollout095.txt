import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
        if len(h_current) != n_points:
            x_orig = np.linspace(0, 2, len(h_current))
            x_target = np.linspace(0, 2, n_points)
            h_current = np.interp(x_target, x_orig, h_current)
    else:
        h_current = np.random.rand(n_points)
    
    # Scale to meet constraint
    h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = evaluate_h(h_current, dx)
    
    # Gradient descent with adaptive perturbations
    learning_rate = 0.001
    perturb_rate = 0.1
    max_iter = 10000
    stagnation_threshold = 100
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h_current, dx)
        
        for i in range(n_points):
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update with gradient
        h_new = h_current - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Adaptive perturbation
        if np.random.rand() < perturb_rate:
            # Randomly perturb some elements
            indices = np.random.choice(n_points, size=int(n_points * 0.05), replace=False)
            for i in indices:
                h_new[i] += np.random.normal(0, 0.01)
            h_new = repair_h(h_new, n_points)
            
        new_c5 = evaluate_h(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        h_current = h_new
        
        # Early stopping if no improvement for a while
        if stagnation_count > stagnation_threshold:
            # Add more randomness to escape
            h_current = np.random.rand(n_points)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            stagnation_count = 0
    
    # Final refinement with gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(best_h, dx)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
