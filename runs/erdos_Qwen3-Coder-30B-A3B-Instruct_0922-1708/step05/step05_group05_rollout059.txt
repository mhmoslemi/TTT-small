import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default: block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute C5
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Function to compute gradient (finite difference approximation)
    def compute_gradient(h_vec, eps=1e-5):
        base_c5 = compute_c5(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = compute_c5(h)
    learning_rate = 0.01
    max_iter = 1000
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with finite differences
        grad = compute_gradient(h)
        
        # Apply gradient step
        h_new = h - learning_rate * grad
        
        # Clip to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        # Accept step if improvement
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            best_h = h.copy()
            # Adapt learning rate
            learning_rate = min(0.05, learning_rate * 1.05)
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.99
        
        # Apply random perturbations occasionally to escape local minima
        if iteration % 50 == 0:
            noise_magnitude = 0.1 / (1.0 + iteration / 100.0)
            noise = np.random.normal(0, noise_magnitude, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            perturbed_c5 = compute_c5(h_perturbed)
            if perturbed_c5 < best_c5:
                h = h_perturbed.copy()
                best_c5 = perturbed_c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
