import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the C5 value for a given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if abs(sum_h) > 1e-10:
        h = h * (n_points / 2.0) / sum_h
    else:
        h = np.ones_like(h) * (n_points / 2.0) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.9):
    """Blend crossover with alpha parameter"""
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            gamma = (1.0 + 2.0 * alpha) * np.random.rand() - alpha
            child1[i] = (1 - gamma) * parent1[i] + gamma * parent2[i]
            child2[i] = (1 - gamma) * parent2[i] + gamma * parent1[i]
    return child1, child2

def mutate(individual, mutation_rate=0.1, mutation_strength=0.05):
    """Apply Gaussian mutation to individual"""
    mutant = individual.copy()
    for i in range(len(mutant)):
        if np.random.rand() < mutation_rate:
            mutant[i] += np.random.normal(0, mutation_strength)
    return mutant

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        # Interpolate to target resolution if needed
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_target = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_target, x_orig, h_parent)
        # Scale to correct integral
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        
        # Create population with variations of parent and random solutions
        for _ in range(population_size // 2):
            h_var = h_parent + np.random.normal(0, 0.05, n_points)
            population.append(project_to_feasible(h_var, n_points))
        
        for _ in range(population_size // 2):
            h_random = np.random.rand(n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            population.append(h_random)
    else:
        # Purely random population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Main GA loop
    start_time = time.time()
    
    best_fitness = float('inf')
    best_individual = None
    generation = 0
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Evaluate fitness
        fitnesses = [compute_fitness(ind, dx) for ind in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_individual = population[current_best_idx].copy()
        
        # Create next generation
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Selection, crossover, mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < 0.8:  # Crossover rate
                child1, child2 = blend_crossover(parent1, parent2, alpha=0.9)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate=0.1, mutation_strength=0.05)
            child2 = mutate(child2, mutation_rate=0.1, mutation_strength=0.05)
            
            new_population.append(project_to_feasible(child1, n_points))
            if len(new_population) < population_size:
                new_population.append(project_to_feasible(child2, n_points))
        
        population = new_population
        
        # Occasionally apply gradient refinement to top individuals
        if generation % 50 == 0 and best_individual is not None:
            # Do gradient-based refinement on top individuals
            top_count = min(10, len(population) // 10)
            top_indices = np.argsort(fitnesses)[:top_count]
            
            for idx in top_indices:
                h_ref = population[idx].copy()
                old_fitness = fitnesses[idx]
                lr = 0.001
                
                for _ in range(50):
                    # Compute gradient numerically
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        c5_plus = compute_fitness(h_plus, dx)
                        grad[j] = (c5_plus - old_fitness) / eps
                    
                    # Update
                    h_ref = h_ref - lr * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    # Re-evaluate
                    new_fitness = compute_fitness(h_ref, dx)
                    
                    if new_fitness < old_fitness:
                        old_fitness = new_fitness
                        population[idx] = h_ref.copy()
                        
                        if old_fitness < best_fitness:
                            best_fitness = old_fitness
                            best_individual = population[idx].copy()
    
    # Final refinement of best solution
    if best_individual is not None:
        h_final = best_individual.copy()
        lr = 0.001
        
        for _ in range(100):
            grad = np.zeros_like(h_final)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_final.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = compute_fitness(h_plus, dx)
                grad[j] = (c5_plus - best_fitness) / eps
                
            h_final = h_final - lr * grad
            h_final = project_to_feasible(h_final, n_points)
            
            new_fitness = compute_fitness(h_final, dx)
            if new_fitness < best_fitness:
                best_fitness = new_fitness
                best_individual = h_final.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
