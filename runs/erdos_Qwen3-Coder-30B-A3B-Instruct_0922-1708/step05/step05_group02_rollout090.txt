import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate random numbers from Lévy distribution
    # This is a simplified version for 1D case
    sigma = scale * np.random.normal(0, 1, size)
    # For Lévy flight, we use a simple approach
    # Sample from a distribution that approximates Lévy
    return np.random.laplace(0, scale, size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add more candidates
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Evaluate initial candidates
    candidate_values = [evaluate_h(c, dx) for c in candidates]
    
    # Track the best solution
    best_idx = np.argmin(candidate_values)
    best_h = candidates[best_idx].copy()
    best_value = candidate_values[best_idx]
    
    # Parameters
    max_iter = 1000
    levy_scale = 0.5
    levy_alpha = 1.5
    learning_rate = 0.001
    top_k = 10
    perturbation_rate = 0.1
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        iter_count += 1
        
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new candidates via Lévy flight
        new_candidates = []
        for h in candidates:
            # Apply Levy flight perturbation
            perturb = levy_flight(n_points, levy_alpha, levy_scale)
            new_h = h + perturb_rate * perturb
            
            # Repair to constraints
            new_h = repair_h(new_h, n_points)
            new_candidates.append(new_h)
            
        # Update using gradient descent on top candidates
        indices = np.argsort(candidate_values)[:top_k]
        for idx in indices:
            h = candidates[idx].copy()
            current_value = candidate_values[idx]
            
            # Estimate gradient using finite differences
            eps = 1e-4
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
            
            # Update
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            corr = correlate(h, 1 - h, mode='full')
            new_value = np.max(corr) * dx
            
            if new_value < candidate_values[idx]:
                candidates[idx] = h.copy()
                candidate_values[idx] = new_value
            
            # Keep track of best solution
            if new_value < best_value:

                best_value = new_value
                last_improvement = iter_count

        candidates = [candidates[i] for i in indices]
        candidate_values = [candidate_values[i] for i in indices]
        
        # Final update of best value
        new_best_idx = np.argmin(candidate_values)
        if candidate_values[new_best_idx] < best_value:
            best_h = candidates[new_best_idx].copy()
            best_value = candidate_values[new_best_idx]
            last_improvement = iter_count
            
        # Stop if no improvement in the last 100 iterations
        if iter_count - last_improvement > 100:
            break
    
    # Final refinement
    h_refined = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        h_refined = h_refined - 0.0005 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_value = np.max(corr) * dx
    
    return (h_refined.tolist(), best_value, n_points)
