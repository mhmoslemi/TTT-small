import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parent validation
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        parent_h = np.clip(parent_h, 0, 1)
        parent_h = parent_h * (100 / 2) / np.sum(parent_h)
    else:
        parent_h = None

    n_points_list = [10, 20, 40, 80, 100]
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Scale pyramid optimization
    for i, n_points in enumerate(n_points_list):
        if time.time() - start_time > budget_s - 50:
            break
            
        dx = 2.0 / n_points
        h = np.zeros(n_points)
        
        # Initialize
        if i == 0:
            h = np.ones(n_points) * 0.5
        else:
            # Up-sample from previous
            prev_h = h_prev
            x_prev = np.linspace(0, 2, len(prev_h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_prev, prev_h)
            # Normalize
            h = h * (n_points / 2) / np.sum(h)
        
        # Convex relaxation
        # Soft constraints
        h_var = Variable(n_points)
        lambda_soft = 0.1
        
        # Objective: approximated max via average of key lags
        # Only consider middle lags to make problem smaller
        max_lag = n_points // 4
        lags_to_consider = list(range(-max_lag, 0)) + list(range(1, max_lag+1))
        total_lag = 2 * max_lag
        
        # Build lagged version of h_var
        obj_terms = []
        for k in lags_to_consider:
            if k < 0:
                # h[i] * (1 - h[i - k]) for i = 0 to n+k
                corr_val = 0
                for i in range(n_points + k):
                    corr_val += h_var[i] * (1 - h_var[i - k])
                obj_terms.append(corr_val)
            else:
                # h[i] * (1 - h[i + k]) for i = 0 to n-k
                corr_val = 0
                for i in range(n_points - k):
                    corr_val += h_var[i] * (1 - h_var[i + k])
                obj_terms.append(corr_val)
        
        # Objective: average lags
        obj_avg = sum(obj_terms) / len(obj_terms)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
        
        prob = Problem(Minimize(obj_avg), constraints)
        
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_solved = h_var.value
            else:
                # Fallback to previous if convex solver fails
                h_solved = h.copy()
        except:
            h_solved = h.copy()
        
        # Project to feasibility
        h_solved = np.clip(h_solved, 0, 1)
        h_solved = h_solved * (n_points / 2) / np.sum(h_solved)
        
        # Local gradient refinement with finite differences
        if time.time() - start_time > budget_s - 50:
            break
            
        def compute_corr(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
            
        def finite_diff_grad(h_orig, eps=1e-4):
            grad = np.zeros_like(h_orig)
            base_c5 = compute_corr(h_orig)
            for i in range(len(h_orig)):
                h_plus = h_orig.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_corr(h_plus)
                grad[i] = (c5_plus - base_c5) / eps
            return grad
            
        # Gradient descent
        h_ref = h_solved.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 50:
                break
            grad = finite_diff_grad(h_ref)
            h_ref -= 0.001 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
        c5_ref = compute_corr(h_ref)
        
        if c5_ref < best_c5:
            best_c5 = c5_ref
            best_h = h_ref.copy()
            
        h_prev = h_ref.copy()
        
        # Early exit if below target
        if best_c5 <= 0.3807:
            break
            
    # Final refinement on best found
    if best_h is None:
        if parent_h is not None:
            best_h = parent_h
        else:
            best_h = np.ones(100) * 0.5
            best_h = best_h * (100 / 2) / np.sum(best_h)
    
    final_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * (2.0 / 100)
    
    # Final validation
    if not np.all((best_h >= 0) & (best_h <= 1)):
        best_h = np.clip(best_h, 0, 1)
    total_sum = np.sum(best_h) * (2.0 / 100)
    if abs(total_sum - 1.0) > 1e-6:
        best_h = best_h * (100 / 2) / np.sum(best_h)
        
    assert np.all((best_h >= 0) & (best_h <= 1))
    assert abs(np.sum(best_h) * (2.0 / 100) - 1.0) < 1e-6
    
    return (best_h.tolist(), final_c5, 100)
