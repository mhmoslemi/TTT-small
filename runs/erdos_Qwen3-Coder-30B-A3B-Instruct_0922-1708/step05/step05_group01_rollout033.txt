import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper to compute C5
    def compute_c5(h_vec):
        # Compute all correlations
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Max correlation (scaled)
        c5 = np.max(corr) * dx
        return c5
    
    # Gradients - analytic approach for max overlap
    def compute_gradient(h_vec, max_lag_idx):
        # Compute correlation for max lag
        # Specialized for full mode
        n = len(h_vec)
        grad = np.zeros(n)
        
        # Determine positions in correlation
        # Full mode: 2*n-1 elements
        # Center: n-1 (corresponds to lag 0)
        # max_lag_idx from 0 to 2*n-2
        
        # For a given lag k, compute contribution of each element
        # When lag is k, we look at:
        # sum over i: h[i] * (1 - h[i+k])
        # Contribution to h[i]: (1 - h[i+k])  if i+k < n && i < n
        # Contribution to h[i+k]: -h[i]          if i+k < n && i < n
        
        # Force this over the entire array using equivalent indexing
        # For indexing below,
        # The max lag index directly tells us where the max occurs
        effective_lag = max_lag_idx - (n - 1)  # lag = 0 is at index n-1
        
        # PID for np.correlate interpretation
        # We create a sliding window for the kernel
        if effective_lag >= 0:
            # h[i] * (1 - h[i + lag])
            # For valid indices where i+lag < n
            for i in range(n - effective_lag):
                grad[i] += 1 - h_vec[i + effective_lag]
                if i + effective_lag < n:
                    grad[i + effective_lag] -= h_vec[i]
        else:
            # h[i] * (1 - h[i + lag]), with lag < 0
            # h[i] * (1 - h[i - |lag|])
            for i in range(-effective_lag, n):
                grad[i] += 1 - h_vec[i + effective_lag]
                if i + effective_lag < n:
                    grad[i + effective_lag] -= h_vec[i]
        
        return grad
    
    # Line search
    def line_search(h0, grad, learning_rate=1e-3):
        # Find best alpha in direction of grad
        # Initialize c5 from current h0
        current_c5 = compute_c5(h0)
        # New h = h0 - alpha * grad
        # We try a decreasing sequence of alphas
        alphas = [learning_rate * (0.5 ** i) for i in range(10)]
        for alpha in alphas:
            h_new = h0 - alpha * grad
            # Project h_new to feasible set
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = compute_c5(h_new)
            if c5_new < current_c5:
                return h_new, c5_new, alpha
        return h0, current_c5, 0
    
    # Optimization
    best_h = h.copy()
    best_c5 = compute_c5(h)
    current_h = h.copy()
    current_c5 = best_c5
    
    # Main loop
    max_iter = 5000
    learning_rate = 1e-3
    tolerance = 1e-6
    stagnation_counter = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlations and identify max
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag_val = corr[max_lag_idx]

        # Compute gradient
        gradient = compute_gradient(current_h, max_lag_idx)
        
        # Line search
        h_new, c5_new, alpha = line_search(current_h, gradient, learning_rate)
        
        # Accept if better
        if c5_new < current_c5:
            current_h = h_new
            current_c5 = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = current_h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            if stagnation_counter > max_stagnation:
                # Small random perturbation to escape
                perturb = np.random.normal(0, 0.01, n_points)
                current_h = current_h + perturb
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                current_c5 = compute_c5(current_h)
                stagnation_counter = 0

    return (best_h.tolist(), best_c5, n_points)
