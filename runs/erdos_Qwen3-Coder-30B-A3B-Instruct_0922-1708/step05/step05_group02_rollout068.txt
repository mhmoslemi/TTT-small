import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(alpha, size):
    """Generate Lévy flight steps."""
    sigma = np.power(np.random.gamma(1 + alpha) * np.sin(np.pi * alpha / 2) / 
                     (np.power(np.random.gamma(1 + alpha), 1 / alpha) * np.power(np.pi, 1 / alpha)), 
                     1 / alpha)
    return np.random.standard_normal(size) * sigma

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add random initial candidates
    for _ in range(49):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Initialize fitness values
    fitness = [evaluate_h(c, dx) for c in candidates]
    
    # Find best solution
    best_idx = np.argmin(fitness)
    best_h = candidates[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Hyperparameters
    alpha = 1.5
    perturb_rate = 0.1
    learning_rate = 0.001
    max_iter = 1000
    top_k = 10
    
    # Main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new candidates via Lévy flights
        new_candidates = []
        new_fitness = []
        
        for i in range(len(candidates)):
            # Generate Lévy perturbation
            perturbation = levy_flight(alpha, n_points) * perturb_rate
            new_h = candidates[i] + perturbation
            new_h = repair_h(new_h, n_points)
            new_candidates.append(new_h)
            new_fitness.append(evaluate_h(new_h, dx))
            
        # Combine old and new candidates
        all_candidates = candidates + new_candidates
        all_fitness = fitness + new_fitness
        
        # Select best candidates
        indices = np.argsort(all_fitness)
        selected_indices = indices[:len(candidates)]
        
        candidates = [all_candidates[i] for i in selected_indices]
        fitness = [all_fitness[i] for i in selected_indices]
        
        # Gradient descent on top candidates
        top_indices = np.argsort(fitness)[:top_k]
        for idx in top_indices:
            h = candidates[idx].copy()
            current_c5 = fitness[idx]
            
            # Compute gradient
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_c5) / eps
            
            # Update
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
            # Replace if better
            if current_c5 < fitness[idx]:
                candidates[idx] = h.copy()
                fitness[idx] = current_c5
                
                # Update global best if improved
                if current_c5 < best_value:
                    best_h = h.copy()
                    best_value = current_c5
            
        # Update global best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:
            best_h = candidates[current_best_idx].copy()
            best_value = fitness[current_best_idx]
            
    # Final intense gradient descent
    for _ in range(2000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
    
    return (best_h.tolist(), best_value, n_points)
