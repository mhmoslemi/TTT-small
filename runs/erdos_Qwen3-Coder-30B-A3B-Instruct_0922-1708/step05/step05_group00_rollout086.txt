import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from the provided parent solution or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution found
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-start optimization with different initializations
    initializations = []
    
    # Add the parent solution as the first initialization
    initializations.append(h.copy())
    
    # Add 5 random initializations
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        initializations.append(h_rand.copy())
    
    # Convex relaxation for each initialization to get a good starting point
    for h_init in initializations:
        h = h_init.copy()
        
        # Try to find a better initial point via a relaxed problem
        try:
            h_var = cp.Variable(n_points)
            C5_var = cp.Variable()
            
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # For the relaxed formulation, we approximate overlap constraints
            # by a quadratic problem that upper bounds the overlap term
            # Still not a perfect way, so we do PCA-inspired bound estimation
            
            # We use a very coarse approximation and then move to numerical gradient
            # This provides heuristic guidance toward better starting points
            
            # Soft project to satisfy constraints, approximate with
            # normalized uniform distribution within bounds
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        except:
            pass  # If relaxation failed, just use initial h
        finally:
            h_local = h.copy()
            
        # Refine using L-BFGS-B with bounds - much faster than pure gradient descent
        # Define the objective function to minimize
        def obj_func(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(correlation)
        
        # Define constraint for sum
        def constraint_func(h_vec):
            return np.sum(h_vec) * dx - 1.0
        
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize with L-BFGS-B and bound constraints
        try:
            res = minimize(
                obj_func, 
                h_local, 
                method='L-BFGS-B', 
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 2000, 'ftol': 1e-6, 'gtol': 1e-6},
                callback=lambda x: None  # No callback needed
            )
            if res.success or (not res.success and res.fun < best_c5):
                h_result = res.x
                h_result = np.clip(h_result, 0, 1)
                h_result = h_result * (n_points / 2) / np.sum(h_result)
                c5_val = res.fun
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_result.copy()
        except Exception:
            pass
        
        # Stop if time is nearly up
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final refinement and validation
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    if abs(np.sum(final_h) * dx - 1.0) > 1e-6:
        final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
