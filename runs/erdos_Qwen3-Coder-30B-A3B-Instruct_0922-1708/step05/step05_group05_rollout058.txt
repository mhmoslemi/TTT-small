import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.01
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize h using the provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Create a simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlation and get the maximum overlap
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Find the index where maximum overlap occurs
        max_idx = np.argmax(corr)
        
        # Compute gradient of maximum overlap with respect to h
        # We need to compute partial derivatives for each h[i]
        grad = np.zeros_like(h)
        eps = 1e-6
        
        # Use finite differences to approximate gradient
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_max = np.max(base_corr)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure h_plus remains feasible
            h_plus = np.clip(h_plus, 0, 1)
            # Re-normalize to sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            grad[i] = (max_plus - base_max) / eps
            
        # Apply gradient descent update
        h = h - learning_rate * grad
        
        # Project h onto the feasible region [0,1]
        h = np.clip(h, 0, 1)
        
        # Re-normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if this is the best solution
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Periodically apply perturbations
        if iteration % perturbation_freq == 0:
            # Apply random perturbations to escape local minima
            perturbation = np.random.uniform(-perturbation_magnitude, perturbation_magnitude, n_points)
            h = h + perturbation
            
            # Clip to [0,1]
            h = np.clip(h, 0, 1)
            
            # Re-normalize
            h = h * (n_points / 2) / np.sum(h)
            
            # Update if better
            corr_pert = np.correlate(h, 1 - h, mode='full')
            pert_c5 = np.max(corr_pert) * dx
            
            if pert_c5 < best_c5:
                best_h = h.copy()
                best_c5 = pert_c5
                
    return (best_h.tolist(), best_c5, n_points)
