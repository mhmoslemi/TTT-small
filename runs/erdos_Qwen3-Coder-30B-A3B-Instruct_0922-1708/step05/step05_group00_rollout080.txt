import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    
    # Create initial solution using parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse-to-fine approach with smaller initial grid for escape from local minima
    coarse_points = 50
    coarse_dx = 2.0 / coarse_points
    
    # Initialize coarse grid with proper normalization
    coarse_h = np.ones(coarse_points) * 0.5
    coarse_h = coarse_h * (coarse_points / 2) / np.sum(coarse_h)
    coarse_h = np.clip(coarse_h, 0, 1)
    
    # Optimize coarse grid
    last_c5 = float('inf')
    patience = 0
    coarse_max_iter = 2000
    coarse_lr = 0.01
    
    for coarse_iter in range(coarse_max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        correlation = np.correlate(coarse_h, 1 - coarse_h, mode='full') * coarse_dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Upscale coarse solution to fine grid
            x_coarse = np.linspace(0, 2, coarse_points)
            x_fine = np.linspace(0, 2, n_points)
            fine_h = np.interp(x_fine, x_coarse, coarse_h)
            fine_h = np.clip(fine_h, 0, 1)
            fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
            best_h = fine_h.copy()

        # Compute finite difference gradient
        grad = np.zeros_like(coarse_h)
        eps = 1e-4
        
        for i in range(coarse_points):
            h_plus = coarse_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (coarse_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * coarse_dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update coarse solution
        coarse_h_new = coarse_h - coarse_lr * grad
        coarse_h_new = np.clip(coarse_h_new, 0, 1)
        coarse_h_new = coarse_h_new * (coarse_points / 2) / np.sum(coarse_h_new)
        
        coarse_h = coarse_h_new
        
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 100:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
    
    # Fine-grid refinement with better optimization
    h = best_h.copy()

    fine_lr = 0.002
    fine_max_iter = 5000
    fine_patience = 0
    fine_best_c5 = float('inf')
    
    for fine_iter in range(fine_max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < fine_best_c5:
            fine_best_c5 = current_c5
            best_h = h.copy()
            
        if abs(current_c5 - last_c5) < 1e-7:
            fine_patience += 1
            if fine_patience > 200:
                break
        else:
            fine_patience = 0
            
        last_c5 = current_c5
        
        # Gradient step
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        h_new = h - fine_lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        fine_lr = max(1e-6, fine_lr * 0.9995)
    
    # Final validation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_correlation)
    
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
