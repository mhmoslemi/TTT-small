import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, n_points):
    dx = 2.0 / n_points
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():

        # Create diverse initial population

            if i == 0:
                h = h_parent.copy()
            else:
                # Add small random noise
                noise = np.random.normal(0, 0.03, n_points)
                h = h_parent + noise
                h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # All random initial population
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # GA parameters
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    
    start_time = time.time()
    
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, n_points) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Generate rest through selection, crossover, mutation
        while len(new_population) < population_size:
            # Tournament selection
            def tournament_selection():
                k = tournament_size
                selected_indices = np.random.choice(population_size, k, replace=False)
                best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
                return population[best_idx].copy()
            
            # Crossover
            if np.random.rand() < crossover_rate:
                parent1 = tournament_selection()
                parent2 = tournament_selection()
                
                # Blend crossover
                alpha = 0.8
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                # Clone
                child1 = tournament_selection()
                child2 = tournament_selection()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                noise_magnitude = 0.03 + 0.02 * np.random.rand()
                noise = np.random.normal(0, noise_magnitude, n_points)
                child1 = child1 + noise
                child1 = project_to_feasible(child1, n_points)
                
            if np.random.rand() < mutation_rate:
                noise_magnitude = 0.03 + 0.02 * np.random.rand()
                noise = np.random.normal(0, noise_magnitude, n_points)
                child2 = child2 + noise
                child2 = project_to_feasible(child2, n_points)

            if len(new_population) < population_size:
                new_population.append(child2)

                    if compute_fitness(h_ref, n_points) < compute_fitness(h, n_points):

                if compute_fitness(h, n_points) < compute_fitness(population[top_count + i], n_points):
                    population[top_count + i] = h.copy()
    
    # Final refinement on best solution found

        h_ref = best_h.copy()
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = compute_fitness(h_plus, n_points)
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            c5_candidate = compute_fitness(h_ref, n_points)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_ref.copy()
    
    # Final evaluation
    final_fitness = compute_fitness(best_h, n_points)
    
    return (best_h.tolist(), final_fitness, n_points)
