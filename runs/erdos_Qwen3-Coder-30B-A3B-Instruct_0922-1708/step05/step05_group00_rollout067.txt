import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Ensure proper normalization
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth max approximation parameter
    beta = 10.0
    
    # Gradient descent with Adam-like updates
    def compute_smooth_max_gradient(h, beta):
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        # Smooth max approximation
        exp_corr = np.exp(beta * corr)
        smooth_max = np.log(np.sum(exp_corr)) / beta
        
        # Compute gradient of smooth max
        grad = np.zeros_like(h)
        for i in range(n_points):
            #Finite difference approximation of gradient
            eps = 1e-5
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_points)
            exp_corr_plus = np.exp(beta * corr_plus)
            smooth_max_plus = np.log(np.sum(exp_corr_plus)) / beta
            
            grad[i] = (smooth_max_plus - smooth_max) / eps
            
        return grad
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coarse grid optimization (n=50)
    n_coarse = 50
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    # Normalize coarse grid
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Adam-like gradient descent for coarse
    lr = 0.01
    m = np.zeros(n_coarse)
    v = np.zeros(n_coarse)
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    
    for i in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_smooth_max_gradient(h_coarse, beta)
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h_coarse_new = h_coarse - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        h_coarse = h_coarse_new
        
        # Evaluate
        corr = correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_coarse)
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h_coarse = h_coarse.copy()
    
    # Transfer to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient for smooth max
        grad = compute_smooth_max_gradient(h, beta)
        
        # Adam updates
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full') * (2.0 / n_points)
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full') * (2.0 / n_points)
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), final_c5, n_points)
