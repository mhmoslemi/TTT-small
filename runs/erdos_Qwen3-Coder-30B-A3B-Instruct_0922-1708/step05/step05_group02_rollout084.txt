import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the provided initial construction or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Cuckoo Search parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 1000
    min_improvement = 1e-8
    stagnation_limit = 200
    
    # Initialize population
    population = [h_init.copy()]
    fitness = [evaluate_h(h_init, dx)]
    
    # Add random initial solutions
    for _ in range(19):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_h(h, dx))
    
    # Track the best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    stagnation_count = 0
    last_improvement = 0
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution through random walk
        new_h = best_h.copy()
        # Perturb a random subset
        num_perturb = max(1, int(0.1 * n_points))  # 10% of elements
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for idx in indices:
            new_h[idx] += np.random.normal(0, 0.01)
        
        # Repair to constraints
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(new_h, dx)
        
        # Accept or reject based on probability
        if new_value < best_value:
            # Always accept improvement
            population.append(new_h)
            fitness.append(new_value)
            if new_value < best_value:
                best_h = new_h.copy()
                best_value = new_value
                last_improvement = iteration
        else:
            # Probabilistic acceptance
            r = np.random.rand()
            if r < alpha:
                population.append(new_h)
                fitness.append(new_value)
        
        # Local search on the best solution
        if iteration % 10 == 0 and iteration > 0:
            h_current = best_h.copy()
            current_value = best_value
            
            # Gradient descent
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_current.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / epsilon
                
                # Update
                h_current = h_current - 0.001 * grad
                h_current = repair_h(h_current, n_points)
                
                corr = correlate(h_current, 1 - h_current, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_h = h_current.copy()
                    best_value = current_value
                    last_improvement = iteration
        
        # Remove worst solutions to keep population size manageable
        if len(population) > 20:
            worst_idx = np.argmax(fitness)
            population.pop(worst_idx)
            fitness.pop(worst_idx)
            
        # Check for stagnation
        if iteration - last_improvement > stagnation_limit:
            break
    
    # Final refinement with gradient descent
    h_final = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / epsilon
        
        h_final = h_final - 0.0001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        best_value = np.max(corr) * dx
    
    return (h_final.tolist(), best_value, n_points)
