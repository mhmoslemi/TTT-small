import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturb_freq = 50
    perturb_mag_decay = 100
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1-h, mode='full')) * dx
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlations and find minimum
        corr = correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find index where correlation is minimum
        min_idx = np.argmin(corr)
        
        # Gradient computation for the minimum correlation
        grad = np.zeros(n_points)
        
        # For correlation at index min_idx, find the corresponding k
        # Correlation indices go from -(n-1) to (n-1) 
        k = min_idx - (n_points - 1)
        
        # Compute gradient step based on the minimum correlation term
        # Gradient contribution from correlation with lag k
        for i in range(n_points):
            if 0 <= i - k < n_points:
                grad[i] += (1-h[i-k]) * dx
            if 0 <= i + k < n_points:
                grad[i] += (1-h[i+k]) * dx
            
        # Apply gradient step
        h_new = h - learning_rate * grad
        
        # Project to feasible region: [0,1] and maintain sum constraint
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1-h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept if better
        if c5_new < c5:
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        else:
            # Occasionally accept worse solutions to escape local optima
            if np.random.rand() < 0.05:
                h = h_new.copy()
        
        # Apply small random perturbations periodically
        if iteration % perturb_freq == 0:
            perturb_mag = 0.1 / (1.0 + iteration / perturb_mag_decay)
            noise = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
    # Final evaluation
    final_corr = correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
