import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.001
    max_evals = 5000
    eval_count = 0
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h_vec
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h_vec)[::-1][:int(diff/dx)]
            for idx in indices:
                if h_vec[idx] > 0 and diff > 0:
                    dec = min(h_vec[idx], diff)
                    h_vec[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h_vec)[:int(-diff/dx)]
            for idx in indices:
                if h_vec[idx] < 1 and diff < 0:
                    inc = min(1 - h_vec[idx], -diff)
                    h_vec[idx] += inc
                    diff += inc
        return h_vec * (n_points / 2) / np.sum(h_vec)
    
    # Gradient computation
    def compute_gradient(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_corr_idx = np.argmax(corr)
        max_lag = max_corr_idx - (len(h_vec) - 1)
        
        grad = np.zeros_like(h_vec)
        eps = 1e-6
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
        return grad

    # Main optimization loop
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update
        h = h - lr * grad
        h = repair_h(h)
        
        eval_count += 1
        
        # Periodic random perturbations
        if it % 50 == 0 and it > 0:
            for _ in range(5):  # Perturb a few points
                i = np.random.randint(0, n_points)
                h[i] = np.random.uniform(0, 1)
                h = repair_h(h)
    
    # Final refinement with L-BFGS
    def objective(h_vec):
        return compute_c5(h_vec)
    
    def constraint(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    # Local refinement with scipy minimize
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = {'type': 'eq', 'fun': constraint}
    
    # Try several random restarts
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        try:
            res = minimize(objective, h_rand, method='L-BFGS-B', bounds=bounds, 
                          constraints=constraints, options={'maxiter': 100, 'ftol': 1e-9})
            if res.success:
                c5_new = objective(res.x)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = res.x.copy()
        except:
            pass
    
    # Final validation
    best_h = repair_h(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
