import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Block construction as fallback
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = 1/2
    h = h * (0.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.max(signal.correlate(h, 1-h, mode='full')) * dx
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Correlation computation
        corr = signal.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        min_k = np.argmax(corr)
        
        # Gradient based on the worst correlation
        # For correlation at min_k, we need to update h to decrease that value
        # The gradient of the correlation with respect to h_j is:
        # d/dh_j [sum_i h_i * (1-h_{i+k})] = 1 - h_{j+k} - h_{j-k} (or similar)
        # But because of the correlation indexing, we look at the structure
        # We compute gradient for each h_j:
        grad = np.zeros(n_points)
        
        # For the correlation at index min_k, 
        # in the full convolution, that corresponds to:
        # sum_{i=0}^{n-1} h[i] * (1 - h[i + min_k - (n-1)])
        
        # But we simplify: we peek at the gradient to decrease the max value
        # Since we're trying to minimize max correlation, we update h towards this
        # We treat min_k as spatial offset and compute derivative:
        # This is an approximation: we just use sparsity in correlation:
        # Correlation at k contributes h[i] * (1-h[i+k])
        # so gradient of this wrt h[i] is -(1-h[i+k]) = h[i+k]-1
        # gradient of this wrt h[i+k] is -h[i]
        
        # For approximate gradient:
        # Using finite difference approach:
        # Let's start simple:
        # Locate min_k (which is offset: from 0 to 2*n-2)
        # For a given kernel shift we want to compute good update
    
        # More accurate gradient estimation using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        h_plus = h.copy()
        
        # For gradient, we do finite differences for each component
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            
            # Re-normalize to sum = 1/2
            h_plus = h_plus * (0.5 / np.sum(h_plus))
            
            corr_plus = signal.correlate(h_plus, 1-h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update step
        h = h - learning_rate * grad
        
        # Project to constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - 0.5) > 1e-6:
            h = h * (0.5 / h_sum)
        
        # Check improvement
        corr = signal.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Perturbations for escape from local optima
        if iteration % perturb_freq == 0 and iteration > 0:
            perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (0.5 / h_sum)
    
    # Final projection
    best_h = np.clip(best_h, 0, 1)
    best_h_sum = np.sum(best_h)
    final_h = best_h * (0.5 / best_h_sum)
    
    final_corr = signal.correlate(final_h, 1-final_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
