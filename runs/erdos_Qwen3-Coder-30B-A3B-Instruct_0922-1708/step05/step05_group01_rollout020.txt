import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Penalty parameter
    penalty_mu = 0.1
    
    # Optimization loop
    last_improvement = 0
    plateau_count = 0
    current_lr = learning_rate
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution if needed
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            plateau_count = 0
        else:
            plateau_count += 1
        
        # Check for plateau
        if plateau_count >= plateau_patience:
            if current_lr > 1e-6:
                current_lr *= learning_rate_decay
                plateau_count = 0
            else:
                # Random perturbation
                perturb = np.random.normal(0, perturbation_fraction * 0.1, n_points)
                h = h + perturb
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                plateau_count = 0
        
        # Compute gradient
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)
        
        # Gradient computation (simplified approach)
        grad = np.zeros(n_points)
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h_extended = np.concatenate([1 - h, np.zeros(n_points - 1)])
        
        # For positive k (shift right)
        if k >= 0:
            # We are measuring h[i] * (1 - h[i + k])
            # Gradient w.r.t. h[i]
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])
            # Gradient w.r.t. h[i + k] 
            for i in range(n_points - k):
                grad[i + k] -= h[i]
        
        # For negative k (shift left)
        else:
            # We are measuring h[i - k] * (1 - h[i]) 
            # Which is equivalent to measuring h[i] * (1 - h[i - k])
            for i in range(-k, n_points):
                grad[i] += (1 - h[i + k])
            # Gradient w.r.t. h[i + k]
            for i in range(-k, n_points):
                grad[i + k] -= h[i]
        
        # Add penalty gradient:
        # Penalty for upper bound (h[i] - 1)^2
        penalty_grad_upper = 2 * penalty_mu * np.clip(h - 1, 0, 1) 
        # Penalty for lower bound (-h[i])^2 
        penalty_grad_lower = 2 * penalty_mu * np.clip(h, 0, 1)
        # Penalty for total sum excess
        sum_violation = np.sum(h) - n_points / 2
        penalty_grad_sum = 2 * penalty_mu * sum_violation
        
        # Final gradient
        grad = grad * dx + penalty_grad_upper + penalty_grad_lower + penalty_grad_sum
        
        # Update h with learning rate decay
        h = h - current_lr * grad
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            # Rescale to correct sum
            h = h * (n_points / 2) / h_sum
            
        # Update penalty
        violations = np.sum(np.clip(h, 0, 1) - h) + np.sum(np.clip(h, 0, 1) - h) + abs(h_sum - n_points / 2)
        if violations > 1e-6:
            penalty_mu = penalty_factor * penalty_mu
        else:
            penalty_mu = max(penalty_mu / penalty_factor, penalty_min)
    
    # Final computation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
