import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    swarm_size = 100
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 30
    rejuvenation_rate = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        for _ in range(swarm_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        for _ in range(swarm_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * swarm_size
    
    # Compute initial fitness for all particles
    for i in range(swarm_size):
        correlation = correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(correlation) * dx
        pbest_fitness[i] = fitness
        if i == 0 or fitness < pbest_fitness[0]:
            gbest_idx = i
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # PSO main loop
    current_iter = 0
    stagnation_count = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Update particles
        for p_idx in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[p_idx] = (inertia_weight * velocities[p_idx] + 
                                phi_p * r1 * (pbest_particles[p_idx] - particles[p_idx]) +
                                phi_g * r2 * (gbest_particle - particles[p_idx]))
            
            # Position update
            particles[p_idx] += velocities[p_idx]
            
            # Repair
            particles[p_idx] = repair(particles[p_idx])
            
            # Fitness computation
            correlation = correlate(particles[p_idx], 1 - particles[p_idx], mode='full')
            fitness = np.max(correlation) * dx
            
            # Update pbest
            if fitness < pbest_fitness[p_idx]:
                pbest_fitness[p_idx] = fitness
                pbest_particles[p_idx] = particles[p_idx].copy()
                
                # Update gbest
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[p_idx].copy()
                    stagnation_count = 0
        
        stagnation_count += 1
        if stagnation_count > 100:
            # Inject random particles
            for _ in range(int(swarm_size * rejuvenation_rate)):
                idx = np.random.randint(0, swarm_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
        
        current_iter += 1
    
    # Final L-BFGS refinement
    def objective(h):
        h = repair(h)
        correlation = correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def gradient(h):
        h = repair(h)
        correlation = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(correlation)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - np.max(correlation) * dx) / eps
        return -grad
    
    # Refine the best solution with L-BFGS-B
    refined_h = gbest_particle.copy()
    
    # Try multiple local optimizations to avoid getting stuck in a local minimum
    for _ in range(3):
        if time.time() - start_time > budget_s - 30:
            break
        try:
            result = minimize(objective, refined_h, method='L-BFGS-B', 
                            bounds=[(0, 1)] * n_points, 
                            jac=gradient, options={'maxiter': 50})
            if result.success:
                refined_h = result.x
        except:
            pass
    
    # Final repair and integrity check
    refined_h = repair(refined_h)
    
    # Final C₅ computation
    correlation = correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Verify constraints
    constraint_violation = abs(np.sum(refined_h) - n_points / 2)
    if constraint_violation > 1e-10:
        refined_h = repair(refined_h)
        correlation = correlate(refined_h, 1 - refined_h, mode='full')
        final_c5 = np.max(correlation) * dx
    
    return (refined_h.tolist(), final_c5, n_points)
