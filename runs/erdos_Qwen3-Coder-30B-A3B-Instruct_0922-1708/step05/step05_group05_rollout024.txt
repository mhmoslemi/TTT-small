import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    alpha_init = 0.01
    alpha_min = 1e-5
    temp_init = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    alpha = alpha_init
    temp = temp_init
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient (approximate)
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        # For computing gradient in direction of max_k,
        # we get approximation: gradient is approximately
        # (1 - h_{i - k}) + (1 - h_{i + k})
        # But we'll use numerical differentiation for stable approach
        
        # Enhanced gradient estimation using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_max = np.max(base_corr)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            grad[i] = (max_plus - base_max) / eps
        
        # Take gradient step
        step = -alpha * grad
        h_new = h + step
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.8, alpha_min)
        
        # Simulated Annealing perturbation
        if iteration % perturb_freq == 0:
            perturb_mask = np.random.random(n_points) < 0.2
            if np.any(perturb_mask):
                perturb_mag = temp * (1 - h[perturb_mask]) * h[perturb_mask]
                delta = np.random.random(np.sum(perturb_mask)) * 2 * perturb_mag - perturb_mag
                h[perturb_mask] += delta
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            
            temp *= cooling_rate
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Estimate gradient
        eps = 1e-6
        grad = np.zeros(n_points)
        base_corr = np.correlate(best_h, 1 - best_h, mode='full')
        base_max = np.max(base_corr)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            grad[i] = (max_plus - base_max) / eps
        
        step = -0.001 * grad
        h_new = best_h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
