import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_generations = 2000
    mutation_strength = 0.5
    sigma_init = 0.1
    elite_ratio = 0.5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Create initial population around the parent
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Initialize with random vectors
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Main evolution loop
    generation = 0
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    mutation_strength = 0.5
    sigma = sigma_init
    
    while generation < max_generations and time.time() - start_time < budget_s - 10:
        # Evaluate fitness for all population members
        fitnesses = []
        for individual in population:
            corr = np.correlate(individual, 1 - individual, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection - keep top elite_ratio
        elite_size = int(population_size * elite_ratio)
        sorted_indices = np.argsort(fitnesses)[:elite_size]
        elite = [population[i] for i in sorted_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            # Select parent from elite
            parent = np.random.choice(elite)
            
            # Gaussian mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mutation_strength * noise
            
            # Repair - clip to [0,1] and adjust to correct sum
            child = np.clip(child, 0, 1)
            h_sum = np.sum(child)
            if abs(h_sum - n_points / 2) > 1e-10:
                child = child * (n_points / 2) / h_sum
            
            offspring.append(child)
        
        # Replace worst in population with best offspring
        worst_indices = np.argsort(fitnesses)[-len(offspring):]
        for i, offspring_idx in enumerate(np.argsort([fitnesses[j] for j in worst_indices])[:len(offspring)]):
            population[worst_indices[offspring_idx]] = offspring[i]
        
        # Adjust mutation parameters
        mutation_strength *= 0.999
        sigma = mutation_strength / 5.0
        
        # Restart if stagnation
        if stagnation_count > 200:
            mutation_strength = 0.5
            sigma = 0.1
            for i in range(population_size):
                population[i] = np.random.uniform(0, 1, n_points)
                h_sum = np.sum(population[i])
                population[i] = population[i] * (n_points / 2) / h_sum
        
        generation += 1
    
    # Final gradient descent refinement on best candidate
    best_h = best_individual.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h):
        eps = 1e-4
        grad = np.zeros_like(h)
        base_fitness = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_sum
            fitness_plus = objective(h_plus)
            grad[i] = (fitness_plus - base_fitness) / eps
        return grad
    
    # Perform gradient descent
    h_refined = best_h.copy()
    for _ in range(100):
        grad = gradient(h_refined)
        h_refined -= 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_sum = np.sum(h_refined)
        h_refined = h_refined * (n_points / 2) / h_sum
        if np.allclose(h_refined, best_h, atol=1e-6):
            break
        best_h = h_refined.copy()
    
    # Final fitness evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
