import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    T0 = 10.0
    max_iter = 1000000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Feasibility projection
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Annealing
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    
    # Annealing parameters
    current_temp = T0
    iterations = 0
    
    while time.time() - start_time < budget_s - 10 and iterations < max_iter:
        iterations += 1
        
        # Perturb
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        h_candidate = project(h_candidate)
        
        # Compute
        c5_candidate = compute_c5(h_candidate)
        c5_current = compute_c5(h)
        
        # Accept or reject
        delta = c5_candidate - c5_current
        
        if delta < 0 or np.random.rand() < np.exp(-delta / current_temp):
            h = h_candidate
            c5_current = c5_candidate
            
        # Update best
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            
        # Update temperature
        current_temp = T0 / np.log(1 + iterations)
    
    # Local refinement with gradient descent
    def refine_solution(h_ref):
        for _ in range(50):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project(h_plus)
                c5_plus = compute_c5(h_plus)
                
                grad[j] = (c5_plus - compute_c5(h_ref)) / eps
            
            h_ref = h_ref - 0.001 * grad
            h_ref = project(h_ref)
            
            # Early stopping if no improvement
            if compute_c5(h_ref) >= compute_c5(h_ref + 0.001 * grad):
                break
                
        return h_ref
    
    refined_h = refine_solution(best_h)
    final_c5 = compute_c5(refined_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = refined_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
