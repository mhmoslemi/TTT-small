import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 1000
    mutation_factor = 0.8
    crossover_prob = 0.9
    adaptive_mutation = True
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.full(n_points, 0.5)
    
    # Normalize to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population around best solution
    population.append(h_init.copy())
    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial population
    fitness = [compute_c5(p) for p in population]
    best_idx = np.argmin(fitness)
    gbest = population[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    # Differential Evolution
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        if adaptive_mutation:
            current_mutation = mutation_factor * (0.99 ** (generation // 50))
        else:
            current_mutation = mutation_factor
            
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + current_mutation * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Repair
            mutant = repair_h(mutant)
            
            # Crossover (binomial)
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Repair again
            trial = repair_h(trial)
            
            # Evaluate
            trial_fitness = compute_c5(trial)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                fitness[i] = trial_fitness
                if trial_fitness < gbest_fitness:
                    gbest = trial.copy()
                    gbest_fitness = trial_fitness
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Final refinement with local optimization
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # Negative because minimize
    
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Gradient-based refinement on best solution
    h_refined = gbest.copy()
    
    # Simple gradient descent
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_corr = np.max(corr) * dx
        
        # Estimate gradient numerically
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - max_corr) / eps
        
        h_refined -= 0.001 * grad
        h_refined = repair_h(h_refined)
        
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_fitness = np.max(corr_new) * dx
        
        if new_fitness >= max_corr:
            break
    
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_fitness, n_points)
