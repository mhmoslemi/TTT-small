import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_value = evaluate_h(best_h, dx)
    
    # Counter for no improvement
    no_improvement_count = 0
    max_no_improvement = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 10 and no_improvement_count < max_no_improvement:
        iteration += 1
        # Generate new candidates by random mutation
        candidates = []
        for _ in range(100):
            # Create a perturbation
            candidate = best_h.copy()
            # Flip a random number of elements
            num_flip = np.random.randint(1, n_points // 10)
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            for idx in flip_indices:
                candidate[idx] = 1 - candidate[idx]
            # Repair
            candidate = repair_h(candidate, n_points)
            candidates.append(candidate)
        
        # Evaluate all candidates
        for candidate in candidates:
            candidate_value = evaluate_h(candidate, dx)
            if candidate_value < best_value:
                best_h = candidate.copy()
                best_value = candidate_value
                no_improvement_count = 0
                break  # Early exit on improvement
        else:
            no_improvement_count += 1
    
    # Fine-tune with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
        
    return (best_h.tolist(), best_value, n_points)
