import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Population-based search using a hybrid approach
    population_size = 100
    generations = 0
    max_generations = 5000
    
    # Create initial population
    population = []
    if 'initial_h_values' in globals():
        # Use initial solution as basis
        population.append(h.copy())
        for _ in range(population_size - 1):
            # Add variation around initial solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Initialize with random feasible solutions
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    # Evaluation function
    def evaluate(h_vec):
        # Normalize
        h_norm = h_vec * (n_points / 2) / np.sum(h_vec)
        # Compute overlap
        correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
        return np.max(correlation) * dx
    
    # Mutation function
    def mutate(x, mutation_rate=0.1):
        x_mut = x.copy()
        for i in range(len(x_mut)):
            if np.random.rand() < mutation_rate:
                x_mut[i] += np.random.normal(0, 0.03)
        return np.clip(x_mut, 0, 1) * (n_points / 2) / np.sum(x_mut)
    
    # Crossover function (blend)
    def crossover(x1, x2, alpha=0.8):
        x_child = alpha * x1 + (1 - alpha) * x2
        return np.clip(x_child, 0, 1) * (n_points / 2) / np.sum(x_child)
    
    # Main genetic algorithm loop
    while time.time() - start_time < budget_s - 10 and generations < max_generations:
        generations += 1
        
        # Evaluate population
        fitness = [evaluate(p) for p in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection (tournament)
        selected = []
        tournament_size = 5
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create new population
        new_population = []
        
        # Elitism: keep best
        best_idx = np.argmin(fitness)
        new_population.append(population[best_idx].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Select parents
            p1_idx, p2_idx = np.random.choice(population_size, 2, replace=False)
            p1, p2 = selected[p1_idx], selected[p2_idx]
            
            # Crossover
            if np.random.rand() < 0.8:
                child = crossover(p1, p2, 0.8)
            else:
                child = p1.copy()
            
            # Mutation
            child = mutate(child, 0.1)
            
            new_population.append(child)
        
        population = new_population[:population_size]
        
        # Occasionally perform local search
        if generations % 50 == 0:
            # Do local refinement on top individuals
            top_indices = np.argsort(fitness)[:population_size//10]
            for idx in top_indices:
                # Gradient descent refinement using finite differences
                h_ref = population[idx].copy()
                for _ in range(20):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        c5_plus = evaluate(h_plus)
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    # Update with gradient
                    h_ref = h_ref - 0.001 * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                    
                    # Re-evaluate
                    new_fitness = evaluate(h_ref)
                    if new_fitness < fitness[idx]:
                        fitness[idx] = new_fitness
                        population[idx] = h_ref.copy()
    
    # Fine-tune the best solution found
    for _ in range(200):
        grad = np.zeros_like(best_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        if np.linalg.norm(grad) > 1e-10:
            best_h = best_h - 0.0005 * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            
            # Re-evaluate
            corr = np.correlate(best_h, 1 - best_h, mode='full')
            new_c5 = np.max(corr) * dx
            
            if new_c5 < best_c5:
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
