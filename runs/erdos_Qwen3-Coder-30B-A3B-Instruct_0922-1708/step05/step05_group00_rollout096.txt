        # Compute the exact C5 value for full overlap integral
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx

    best_h_refined = best_h_refined * (n_points / 2) / np.sum(best_h_refined)
    
    return (best_h_refined.tolist(), final_c5, n_points)
