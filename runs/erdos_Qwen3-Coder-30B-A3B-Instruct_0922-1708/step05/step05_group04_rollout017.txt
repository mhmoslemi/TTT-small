import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    epsilon = 1e-5
    max_iterations = 20
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Sample shifts for constraint checking
    k_samples = np.linspace(-2, 2, 50)
    
    def compute_overlap(h_vec):
        # Compute overlap integral using correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def compute_constraints(h_vec, mu):
        # Check if h satisfies the constraint for given mu
        # Compute overlaps for specified shifts
        for k in k_samples:
            # Extract relevant part of correlation
            shift_idx = int((k + 2) / (4.0 / n_points))  # Adjust for convolution indexing
            if shift_idx >= 0 and shift_idx < len(h_vec) * 2 - 1:
                # Since we're computing h(x)(1-h(x+k)), we approximate:
                # For k>0: h(x)*h(x-k) term by flipping h and shifting
                # For k<0: h(x)*h(x+k) term
                # Here we compute low-accuracy overlap with a simple approach
                if k >= 0:
                    # Compute ∫ h(x)(1-h(x+k)) dx ≈ ∑ h[i]*(1-h[i+k]) dx
                    overlap = 0
                    for i in range(len(h_vec)):
                        j = i + int(k/dx)
                        if j < len(h_vec):
                            overlap += h_vec[i] * (1 - h_vec[j])
                    overlap *= dx
                else:
                    # k < 0
                    overlap = 0
                    for i in range(len(h_vec)):
                        j = i + int(k/dx)
                        if j >= 0:
                            overlap += h_vec[i] * (1 - h_vec[j])
                    overlap *= dx
                if overlap < mu:
                    return False
        return True
    
    def repair_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_sum = np.sum(h_vec)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_vec = h_vec * (n_points / 2) / h_sum
        return h_vec
    
    # Binary search for optimal C5
    low = 0.0
    high = 0.5
    best_h = h.copy()
    best_C5 = compute_overlap(h)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        mu = (low + high) / 2
        
        # Attempt to find h that satisfies constraint mu
        # Use a simple approach: iterate to improve h while maintaining sum=1
        
        # Start optimization with repair
        h_current = repair_h(h.copy())
        
        # Gradient step refinement for constraint satisfaction
        for _ in range(20):
            # Compute gradient for the local constraint
            grad = np.zeros_like(h_current)
            eps = 1e-4
            base_overlap = compute_overlap(h_current)
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus)
                overlap_plus = compute_overlap(h_plus)
                grad[i] = (overlap_plus - base_overlap) / eps
            
            # Update using gradient
            h_current = h_current - 0.01 * grad
            h_current = repair_h(h_current)
            
            # Early termination if constraint is satisfied
            if compute_overlap(h_current) <= mu:
                break
            
        # Check if we improved
        current_overlap = compute_overlap(h_current)
        if current_overlap < best_C5:
            best_C5 = current_overlap
            best_h = h_current.copy()
        
        # Refinement of mu based on feasibility check
        feasible = True  # Assume feasible first
        # Actually compute minimum overlap among moments to debug stopping
        for k in k_samples[:10]:  # Check sparsely for speed
            # Simpler version
            overlap = 0
            for i in range(len(h_current)):
                j = i + int(k/dx)
                if 0 <= j < len(h_current):
                    overlap += h_current[i] * (1 - h_current[j])
            overlap *= dx
            if overlap < mu * 0.9:  # Very loose due to discretization
                feasible = False
                break
        
        if feasible:
            low = mu
        else:
            high = mu
            
        if abs(high - low) < epsilon:
            break
    
    # Final refinement of best solution
    h_final = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        base_overlap = np.max(corr) * dx
        
        grad = np.zeros_like(h_final)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            overlap_plus = np.max(corr_plus) * dx
            grad[i] = (overlap_plus - base_overlap) / eps
        
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final)
        
        new_corr = np.correlate(h_final, 1 - h_final, mode='full')
        new_overlap = np.max(new_corr) * dx
        if new_overlap >= base_overlap:
            break
    
    final_corr = np.correlate(h_final, 1 - h_final, mode='full')
    final_C5 = np.max(final_corr) * dx
    
    return (h_final.tolist(), final_C5, n_points)
