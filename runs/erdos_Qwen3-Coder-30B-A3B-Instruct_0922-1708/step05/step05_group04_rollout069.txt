import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
        
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Binary search for the optimal C5
    def compute_overlap(h, k):
        corr = np.correlate(h, 1 - h, mode='full')
        return corr[len(corr)//2 + int(k/dx)]
    
    def compute_c5(h):
        max_corr = -float('inf')
        for k in np.linspace(-2, 2, 50):
            corr_val = compute_overlap(h, k)
            max_corr = max(max_corr, corr_val * dx)
        return max_corr
    
    # Gradient-based refinement
    def evaluate_fitness(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return compute_c5(h)
    
    def gradient_step(h, lr=0.001):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        current_c5 = compute_c5(h)
        eps = 1e-4
        
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Start with the initial construction
    h_current = h_init.copy()
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    
    # Gradient ascent iterations
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        h_new = gradient_step(h_current)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_current = h_new
    
    # Final refinement
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        h_new = gradient_step(h_current, lr=0.0005)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        h_current = h_new
    
    # Ensure constraints are still met
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
