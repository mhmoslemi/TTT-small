from scipy.optimize import minimize
import time

    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mutation_strength = 0.5
    mutation_variance = 0.1
    elite_ratio = 0.5
    restart_threshold = 200
    decay_interval = 100
    max_no_improvement = 200

    population = []
    
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Create population with small perturbations around initial
        population.append(h_init)
        for _ in range(population_size - 1):
            perturbation = np.random.normal(0, 0.05, n_points)
            h_perturbed = h_init + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            population.append(h_perturbed)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    fitness_values = []
    for h in population:
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        fitness_values.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx].copy()
    best_fitness = fitness_values[best_idx]
    
    # Track stagnation
    no_improvement_count = 0
    last_best_fitness = best_fitness
    
    # Main evolution loop
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Selection: keep top elite_ratio of population
        elite_count = int(population_size * elite_ratio)
        sorted_indices = np.argsort(fitness_values)[:elite_count]
        elite_population = [population[i] for i in sorted_indices]
        elite_fitness = [fitness_values[i] for i in sorted_indices]
        
        # Generate offspring via Gaussian mutation
        offspring = []
        current_mutation_strength = mutation_strength * (0.999 ** current_iter)
        current_mutation_variance = current_mutation_strength / 5.0
        
        for i in range(offspring_size):
            parent = np.random.choice(elite_population)
            noise = np.random.normal(0, current_mutation_variance, n_points)
            child = parent + current_mutation_strength * noise
            child = np.clip(child, 0, 1)
            
            # Repair constraints
            h_sum = np.sum(child)
            if abs(h_sum - n_points / 2) > 1e-10:
                child = child * (n_points / 2) / h_sum
                
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = []
        for child in offspring:
            correlation = np.correlate(child, 1 - child, mode='full')
            fitness = np.max(correlation) * dx
            offspring_fitness.append(fitness)
        
        # Replacement: keep best from combined pop
        all_candidates = elite_population + offspring
        all_fitness = elite_fitness + offspring_fitness

        combined_indices = np.argsort(all_fitness)

        fitness_values = new_fitness

            no_improvement_count = 0

            current_mutation_strength = 0.5
            current_mutation_variance = 0.1

        if current_iter % decay_interval == 0:
            current_mutation_strength *= 0.9
        
        current_iter += 1
    
    # Final gradient-based refinement
    def refine_step(h_in):
        h_ref = h_in.copy()
        for _ in range(100):
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx

            grad = np.zeros(n_points)

                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')

                grad[i] = (fitness_plus - fitness) / eps

            step_size = 0.001
            h_new = h_ref - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            if abs(h_new_sum - n_points / 2) > 1e-10:
                h_new = h_new * (n_points / 2) / h_new_sum

            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            h_ref = h_new
            fitness = fitness_new
            
        return h_ref
    
    # Refine best solution
    refined_h = refine_step(best_h)
    final_corr = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (refined_h.tolist(), final_c5, n_points)
