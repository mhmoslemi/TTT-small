import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points

    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5

    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Smooth approximation to max function
    def smooth_max(x, beta=10):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Compute correlation and its gradient
    def compute_c5_and_grad(h_val):
        correlation = np.correlate(h_val, 1 - h_val, mode='full') * dx
        c5 = np.max(correlation)
        return c5, correlation
    
    # Finite difference gradient computation
    def finite_diff_grad(h_val, eps=1e-5):
        grad = np.zeros_like(h_val)
        current_c5, _ = compute_c5_and_grad(h_val)
        
        for i in range(len(h_val)):
            h_plus = h_val.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus, _ = compute_c5_and_grad(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
            
        return grad
    
    # Adam optimizer step
    def adam_step(h_val, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h_val - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        return h_new, m, v
    
    # Track Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 1
    
    # Gradient descent loop
    iterations = 0
    max_iter = 10000
    patience = 0
    lr = 0.01
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        current_c5, _ = compute_c5_and_grad(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - best_c5) < 1e-7:
            patience += 1
            if patience > 500:
                break
        else:
            patience = 0

        # Compute gradient
        grad = finite_diff_grad(h)
        
        # Perform Adam step
        h, m, v = adam_step(h, grad, m, v, t, lr=lr, beta1=0.9, beta2=0.999)
        t += 1
        
        # Projection to [0,1] and normalize sum
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decrease learning rate slowly
        lr = max(1e-6, lr * 0.9995)
        
        iterations += 1

    # Final check
    final_c5, _ = compute_c5_and_grad(best_h)
    
    # Ensure sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
