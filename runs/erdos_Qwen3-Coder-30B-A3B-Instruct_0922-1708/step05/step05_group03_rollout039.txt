import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Scale sequence for multi-resolution optimization
    scales = [10, 20, 40, 80, 100]
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Parent fallback
    if 'initial_h_values' in globals():
        parent_h = np.array(initial_h_values)
        parent_h = np.clip(parent_h, 0, 1)
        parent_h = parent_h * (100 / 2) / np.sum(parent_h)
    else:
        parent_h = np.ones(100) * 0.5
    
    # Multi-scale optimization
    for scale_idx, n_points in enumerate(scales):
        if time.time() - start_time > budget_s - 100:
            break
            
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5
        
        # Warm start from previous scale if available
        if scale_idx > 0:
            prev_n = scales[scale_idx - 1]
            if prev_n < n_points:
                x_orig = np.linspace(0, 2, prev_n)
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h_prev)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Convex relaxation optimization
        if n_points <= 80:
            # Use CVXPY for convex optimization
            h_var = Variable(n_points)
            objective = Minimize(0)  # Placeholder, will compute later
            
            # Soft constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                sum(h_var) == n_points / 2
            ]
            
            # For simplicity, we just do a basic refinement
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
            
        # Simple local update with gradient
        if time.time() - start_time > budget_s - 100:
            break
            
        for _ in range(20):
            if time.time() - start_time > budget_s - 50:
                break
                
            # Non-convex local improvement
            h_new = h.copy()
            eps = 1e-4
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
            
            h_new = h - 0.001 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate
            corr = np.correlate(h_new, 1-h_new, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            h = h_new.copy()
        
        h_prev = h.copy()
    
    # Final local refinement for n_points=100
    if best_h is None:
        best_h = parent_h.copy()
        best_c5 = float('inf')
        
    h = best_h.copy()
    dx = 2.0 / 100
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(100):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (100 / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        h = h - 0.0005 * grad
        h = np.clip(h, 0, 1)
        h = h * (100 / 2) / np.sum(h)
        
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final validation
    corr = np.correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, 100)
