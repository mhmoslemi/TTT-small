import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    initial_alpha = 0.01
    alpha_decay = 0.5
    alpha_growth = 1.2
    perturb_freq = 50
    perturb_magnitude = 0.05
    perturb_decay = 0.95
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial solution
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    start_time = time.time()
    alpha = initial_alpha
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Compute candidate step
        h_new = h - alpha * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum) > 1e-10:
            h_new = h_new * (n_points / 2) / h_new_sum
            
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject step
        if new_c5 < base_c5:
            h = h_new
            best_c5 = new_c5
            if new_c5 < np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx:
                best_h = h.copy()
            alpha *= alpha_growth
        else:
            alpha *= alpha_decay
            
        # Perturb periodically to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h_perturbed = h + perturb
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed_sum = np.sum(h_perturbed)
            if abs(h_perturbed_sum) > 1e-10:
                h_perturbed = h_perturbed * (n_points / 2) / h_perturbed_sum
            
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            if c5_pert < best_c5:
                h = h_perturbed
                best_c5 = c5_pert
                best_h = h.copy()
                
            perturb_magnitude *= perturb_decay
    
    # Final validation and return
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
