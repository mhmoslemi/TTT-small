import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize solution
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    n_nests = 15
    max_iter = 1000
    alpha = 0.25
    epsilon = 1e-5
    po = 0.25  # probability of discovering a better nest
    local_search_freq = 50
    max_stagnation = 200
    
    # Initialize nests
    nests = [h_init.copy()]
    for _ in range(n_nests - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        nests.append(h)
    
    best_nest = best_h.copy()
    best_nest_value = best_c5
    
    stagnation_counter = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solutions (Levy flight)
        for i in range(n_nests):
            if np.random.rand() < po:
                # Levy flight step (simplified)
                step_size = np.random.normal(0, 0.5, n_points)
                new_h = nests[i] + alpha * step_size
                
                # Repair and normalize
                new_h = repair_h(new_h, n_points)
                new_c5 = evaluate_h(new_h, dx)
                
                # Accept or reject
                if new_c5 < best_nest_value:
                    best_nest = new_h.copy()
                    best_nest_value = new_c5
                    
                if new_c5 < evaluate_h(nests[i], dx):
                    nests[i] = new_h.copy()
        
        # Find best nest
        current_values = [evaluate_h(nest, dx) for nest in nests]
        best_nest_idx = np.argmin(current_values)
        if current_values[best_nest_idx] < best_nest_value:
            best_nest = nests[best_nest_idx].copy()
            best_nest_value = current_values[best_nest_idx]
        
        # Local search every few iterations
        if iteration % local_search_freq == 0:
            # Gradient descent on best nest
            h = best_nest.copy()
            current_c5 = best_nest_value
            
            for _ in range(20):  # Limited gradient steps
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / epsilon
                
                h = h - 0.0001 * grad
                h = repair_h(h, n_points)
                corr = correlate(h, 1 - h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_nest_value:
                    best_nest = h.copy()
                    best_nest_value = current_c5
        
        # Check for stagnation
        if best_nest_value < best_c5:
            best_h = best_nest.copy()
            best_c5 = best_nest_value
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        if stagnation_counter > max_stagnation:
            # Restart with new random solutions
            stagnation_counter = 0
            for i in range(n_nests):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                nests[i] = h
    
    # Final gradient descent on the best solution
    final_h = best_h.copy()
    final_c5 = best_c5
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - final_c5) / epsilon
            
        final_h = final_h - 0.0001 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
