import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Use higher resolution for better precision
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Use a low-discrepancy-like distribution to spread values
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Parameter for smooth maximum approximation
    beta = 10.0
    
    # Helper to compute smooth maximum
    def smooth_max(x, beta):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Gradient descent with Adam-style optimizer
    def adam_step(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        return h_new, m, v
    
    # Compute current correlation
    def compute_correlation(h):
        return np.correlate(h, 1 - h, mode='full') * dx
    
    # Compute surrogate C5 using smooth maximum
    def surrogate_c5(h, beta):
        corr = compute_correlation(h)
        return smooth_max(corr, beta)
    
    # Compute exact C5 (for reporting)
    def exact_c5(h):
        corr = compute_correlation(h)
        return np.max(corr)
    
    # Initialize optimizer state
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    t = 0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    max_iter = 5000
    patience = 0
    patience_limit = 1000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        t += 1
        # Compute current surrogate c5
        current_c5 = surrogate_c5(h, beta)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = surrogate_c5(h_plus, beta)
            grad[j] = (c5_plus - current_c5) / eps
            
        # Update with Adam
        h_new, m, v = adam_step(h, grad, m, v, t, lr=0.01)
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Early stopping
        if i > 0 and i % 100 == 0:
            curr_c5 = exact_c5(best_h)
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = h.copy()
            patience += 1
            if patience > patience_limit:
                break
    
    # Final evaluation
    final_c5 = exact_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
