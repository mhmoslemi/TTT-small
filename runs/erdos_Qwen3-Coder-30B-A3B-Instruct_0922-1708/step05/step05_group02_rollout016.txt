import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate a Lévy flight step."""
    # Sample from a Lévy distribution
    # For simplicity, using a simplified approach
    # A common approximation for Levy flight in 1D is to sample from a power-law distributed step size
    # This is an approximation; a full implementation would sample from the actual Levy distribution
    steps = np.random.standard_cauchy(size) * scale
    return steps

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    num_initial = 50
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
        
    # Add more initial candidates
    for _ in range(num_initial - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Evaluate all candidates
    values = [evaluate_h(c, dx) for c in candidates]
    best_idx = np.argmin(values)
    best_h = candidates[best_idx].copy()
    best_value = values[best_idx]
    
    max_iter = 1000
    top_candidates = 10
    alpha = 1.5
    scale = 0.5
    learning_rate = 0.001
    perturb_rate = 0.1
    patience = 100
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Lévy flight perturbations
        new_candidates = []
        for h in candidates:
            # Apply Lévy flight
            perturbation = levy_flight(n_points, alpha, scale)
            h_perturbed = h + perturbation * perturb_rate
            h_perturbed = repair_h(h_perturbed, n_points)
            new_candidates.append(h_perturbed)
            
        # Gradient descent on top candidates
        top_indices = np.argsort(values)[:top_candidates]
        for idx in top_indices:
            h = candidates[idx].copy()
            current_c5 = values[idx]
            
            # Estimate gradient using finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_c5) / eps
            
            # Update with gradient
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            
            new_candidates.append(h)
            
        # Add new random candidates
        for _ in range(10):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            new_candidates.append(h)
            
        # Evaluate new candidates
        new_values = [evaluate_h(c, dx) for c in new_candidates]
        
        # Update candidates and keep best
        candidates = new_candidates
        values = new_values
        
        # Check for better solutions
        new_best_idx = np.argmin(values)
        if values[new_best_idx] < best_value:
            best_h = candidates[new_best_idx].copy()
            best_value = values[new_best_idx]
            last_improvement = iteration
            
        # Check for patience
        if iteration - last_improvement > patience:
            # Restart with a few better candidates and mutation
            top_idx = np.argsort(values)[:10]
            candidates = [candidates[i] for i in top_idx]
            values = [values[i] for i in top_idx]
            
            # Mutate the best solution
            mutated = best_h + np.random.normal(0, 0.01, n_points)
            mutated = repair_h(mutated, n_points)
            candidates.append(mutated)
            values.append(evaluate_h(mutated, dx))
            last_improvement = iteration
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
    
    return (best_h.tolist(), best_value, n_points)
