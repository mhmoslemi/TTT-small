import numpy as np
from scipy.optimize import minimize
import time

import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_points = 200  # Final resolution
    dx = 2.0 / n_points
    dx_coarse = 2.0 / n_start
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
        
    # Normalize coarse grid to meet constraint
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Optimization parameters
    max_iter_coarse = 2000
    max_iter_fine = 2000
    patience_limit = 200
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adam optimizer state
    m = np.zeros(n_start)
    v = np.zeros(n_start)
    
    # For tracking best solution
    best_h_coarse = h_coarse.copy()
    best_c5 = float('inf')
    
    # Coarse optimization using Adam
    last_c5 = float('inf')
    patience = 0
    
    iter_count = 0
    for it in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h_coarse.copy()
            
        # Termination criterion
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_coarse)
        eps = 1e-4
        
        for i in range(n_start):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(it + 1))
        v_hat = v / (1 - beta2**(it + 1))
        h_coarse = h_coarse - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection step
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
        
        iter_count += 1
    
    # Refine with fine grid using interpolated solution
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)

    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization with Adam
    # Reinitialize Adam optimizer
    m_fine = np.zeros(n_points)
    v_fine = np.zeros(n_points)
    
    # Reset optimization state
    best_h_fine = h_fine.copy()
    best_c5_fine = best_c5
    
    last_c5_fine = float('inf')
    patience_fine = 0
    
    iter_count_fine = 0

        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation_fine = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5_fine = np.max(correlation_fine)
        
        if current_c5_fine < best_c5_fine:
            best_c5_fine = current_c5_fine
            best_h_fine = h_fine.copy()
            
        # Convergence check
        if abs(current_c5_fine - last_c5_fine) < 1e-6:
            patience_fine += 1
            if patience_fine > patience_limit:
                break
        else:
            patience_fine = 0
            
        last_c5_fine = current_c5_fine
        
        # Compute finite difference gradient
        grad_fine = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad_fine[i] = (c5_plus - current_c5_fine) / eps
        
        # Adam update
        m_fine = beta1 * m_fine + (1 - beta1) * grad_fine
        v_fine = beta2 * v_fine + (1 - beta2) * grad_fine**2
        m_hat = m_fine / (1 - beta1**(it + 1))
        v_hat = v_fine / (1 - beta2**(it + 1))
        h_fine = h_fine - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection step
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        iter_count_fine += 1
    
    # Final solution selection
    final_h = best_h_fine if best_c5_fine < best_c5 else best_h_coarse
    final_c5 = min(best_c5_fine, best_c5)
    
    # Ensure sum constraint is met
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
