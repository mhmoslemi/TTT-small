import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 2000
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Fitness evaluation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # DE mutation and crossover
    def de_step(h_idx):
        # Select three distinct individuals
        indices = list(range(population_size))
        indices.remove(h_idx)
        a, b, c = np.random.choice(indices, 3, replace=False)
        v = population[a] + F * (population[b] - population[c])
        v = np.clip(v, 0, 1)
        v = repair(v)
        
        # Binomial crossover
        u = np.zeros(n_points)
        jrand = np.random.randint(0, n_points)
        for j in range(n_points):
            if np.random.rand() < CR or j == jrand:
                u[j] = v[j]
            else:
                u[j] = population[h_idx][j]
        u = repair(u)
        
        return u
    
    # Evaluate all individuals and keep track of best
    fitnesses = [evaluate_fitness(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # DE main loop
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        if iteration % 50 == 0:
            F = max(0.1, F * 0.99)
        new_population = []
        for i in range(population_size):
            u = de_step(i)
            u_fitness = evaluate_fitness(u)
            if u_fitness < fitnesses[i]:
                new_population.append(u)
                if u_fitness < best_fitness:
                    best_fitness = u_fitness
                    best_h = u.copy()
            else:
                new_population.append(population[i])
        population = new_population
        fitnesses = [evaluate_fitness(h) for h in population]
        iteration += 1
    
    # Final refinement with gradient descent
    h_ref = best_h.copy()
    for _ in range(100):
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_ref -= 0.001 * grad
        h_ref = repair(h_ref)
        
        corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    # Final evaluation
    final_corr = np.correlate(h_ref, 1 - h_ref, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_ref.tolist(), final_c5, n_points)
