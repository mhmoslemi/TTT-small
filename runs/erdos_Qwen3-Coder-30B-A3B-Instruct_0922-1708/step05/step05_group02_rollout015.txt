import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight steps."""
    # Sample from symmetric stable distribution with parameter alpha
    # For simplicity, using a truncated Cauchy distribution as approximation
    return scale * np.random.standard_cauchy(size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    candidates = []
    num_initial = 50
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add more random initial candidates
    for _ in range(num_initial - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Current best solution
    best_solution = candidates[0].copy()
    best_value = evaluate_h(best_solution, dx)
    
    # Parameters
    max_iter = 1000
    top_candidates = 10
    levy_scale = 0.5
    levy_alpha = 1.5
    learning_rate = 0.001
    perturb_rate = 0.1
    stagnation_limit = 100
    
    # Main loop
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all candidates
        values = [evaluate_h(c, dx) for c in candidates]
        
        # Update best
        min_idx = np.argmin(values)
        if values[min_idx] < best_value:
            best_solution = candidates[min_idx].copy()
            best_value = values[min_idx]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination on stagnation
        if stagnation_count > stagnation_limit:
            break
            
        # Select top candidates for gradient update
        top_indices = np.argsort(values)[:top_candidates]
        top_candidates_list = [candidates[i] for i in top_indices]
        
        # Gradient descent on top candidates
        new_candidates = []
        for c in top_candidates_list:
            h = c.copy()
            grad = np.zeros(n_points)
            
            # Finite differences for gradient
            eps = 1e-6
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - evaluate_h(h, dx)) / eps
            
            # Gradient step
            h = h - learning_rate * grad
            h = repair_h(h, n_points)
            new_candidates.append(h)
        
        # Generate Lévy perturbed candidates
        perturbed_candidates = []
        for c in candidates:
            # Lévy flight perturbation
            perturbation = levy_flight(n_points, levy_alpha, levy_scale)
            
            # Apply perturbation with some probability
            mask = np.random.rand(n_points) < perturb_rate
            perturbed = c + perturbation * mask
            perturbed = repair_h(perturbed, n_points)
            perturbed_candidates.append(perturbed)
        
        # Shuffle and select next generation
        candidates = new_candidates + perturbed_candidates
        candidates = candidates[:num_initial]
        
    # Final gradient refinement
    best_h = best_solution.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
    
    return (best_h.tolist(), best_value, n_points)
