import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize randomly
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Start with best solution found
    h_current = h_init.copy()
    h_current = repair_h(h_current, n_points)
    c5_current = evaluate_h(h_current, dx)
    
    # Set up simulated annealing parameters
    T = c5_current * 0.1  # Initial temperature
    alpha = 0.99
    max_iter = 10000
    perturb_rate = 0.10  # Percent of points to perturb
    stagnation_threshold = 100
    stagnation_count = 0
    best_h = h_current.copy()
    best_c5 = c5_current
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the solution
        h_new = h_current.copy()
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            # Add small random perturbation
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            
        # Repair and rescale
        h_new = repair_h(h_new, n_points)
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < c5_current:
            h_current = h_new.copy()
            c5_current = c5_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
            stagnation_count = 0
        else:
            # Accept with probability based on temperature
            delta_e = c5_new - c5_current
            accept_prob = np.exp(-delta_e / T)
            if np.random.rand() < accept_prob:
                h_current = h_new.copy()
                c5_current = c5_new
            stagnation_count += 1
            
        # Cooling schedule
        T *= alpha
        if stagnation_count > stagnation_threshold:
            # Reset temperature if no improvement for a while
            T = max(T * 0.9, 1e-8)
            stagnation_count = 0
            
    # Final gradient descent refinement
    h_refined = best_h.copy()
    c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5_refined) / eps
            
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr) * dx
        
        if c5_refined < c5_refined:
            best_h = h_refined.copy()
            best_c5 = c5_refined
            
    return (best_h.tolist(), best_c5, n_points)
