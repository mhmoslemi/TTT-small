import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_region_init = 0.01
    trust_region_min = 1e-5
    temperature_init = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    max_gradient_steps = 500
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use block construction as default
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation and C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Trust region and temperature
    alpha = trust_region_init
    T = temperature_init
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient by finding the dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_val = corr[max_k]
        
        # Compute gradient for current maximum
        grad = np.zeros(n_points)
        # The gradient is derived from the overlap computation
        # For the lag k that maximizes, we need to compute \partial/ \partial h_i of overlap_k 
        # overlap_k = sum h_i * (1 - h_{i+k})
        # So, \partial overlap_k / \partial h_i = (1 - h_{i+k}) if k such that i is the first term
        # But note: we need to work with the actual indices that contribute.
        # The full correlation means i from 0 to n-1 and we need to compute for all lags
        # The major lag max_k... 
        # Let's think more carefully.
        # Config is h, 1-h, corr[i] = sum h[i+j] * (1 - h[j])  with j mod n
        # Let's compute numerical gradient instead for robustness
        
        # Use finite differences for robustness
        eps = 1e-5
        grad = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_c5) / eps
        
        # Create a step
        step = -alpha * grad
        
        # Project to feasible region
        h_candidate = h + step
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate new solution
        corr_new = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < current_c5:
            h = h_candidate.copy()
            current_c5 = new_c5
            alpha = min(alpha * 1.2, 1.0)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            alpha = max(alpha * 0.8, trust_region_min)
        
        # Apply simulated annealing occasionally
        if iteration % perturbation_freq == 0 and T > 1e-6:
            h_pert = h.copy()
            for j in range(n_points):
                if np.random.rand() < 0.1:  # perturb 10% of components
                    delta = (np.random.rand() - 0.5) * T * h_pert[j] * (1 - h_pert[j])
                    h_pert[j] += delta
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            pert_c5 = np.max(corr_pert) * dx
            
            if pert_c5 < current_c5:
                h = h_pert.copy()
                current_c5 = pert_c5
                if pert_c5 < best_c5:
                    best_c5 = pert_c5
                    best_h = h.copy()
            
            T *= cooling_rate
    
    # Final refinement
    h_refined = best_h.copy()
    for _ in range(max_gradient_steps):
        eps = 1e-5
        grad = np.zeros(n_points)
        base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        improved = False
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - base_c5) / eps
            
            if c5_plus < base_c5:
                improved = True
                
        if not improved:
            break
            
        h_refined -= 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_c5 = np.max(corr_final) * dx
        
        if new_c5 < base_c5:
            base_c5 = new_c5
        else:
            break
    
    # Final evaluation
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
