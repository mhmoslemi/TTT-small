import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import GLPK, ECOS
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up discretization and parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Smooth max approximation parameter
    beta = 10.0
    
    def smooth_max(x, beta):
        # Smooth approximation to max
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    def compute_overlap(h_vals):
        # Returns all overlap values
        corr = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return corr
    
    def objective(h_vals):
        Corr = compute_overlap(h_vals)
        return smooth_max(Corr, beta)
    
    def gradient(h_vals):
        # Compute gradient of smooth max approximation
        Corr = compute_overlap(h_vals)
        exp_vals = np.exp(beta * Corr)
        sum_exp = np.sum(exp_vals)
        weights = exp_vals / sum_exp
        # Local gradient contribution from the correlation
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            # d/dh_i of sum_j h_j * (1 - h_{j+i})
            local_grad = np.zeros_like(h_vals)
            for j in range(len(h_vals)):
                k = (j + i) % len(h_vals)
                if k < len(h_vals):
                    local_grad[j] += 1 - h_vals[k]
                if j < len(h_vals):
                    local_grad[k] += -h_vals[j]
            grad[i] = np.sum(weights * local_grad)
        return grad
    
    # Adaptive SGD with constraints
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    patience_limit = 200
    lr = 0.005
    
    # Use bfgs with feasible projection to minimize smooth max
    for iteration in range(3000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        corr = compute_overlap(h)
        current_c5 = np.max(corr)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            patience = 0
        else:
            patience += 1
            if patience >= patience_limit:
                break
                
        # Compute gradient using finite diff
        eps = 1e-4
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = compute_overlap(h_plus)
            c5_plus = np.max(corr_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Adapt learning rate
        lr = max(1e-6, lr * 0.9995)
        h = h_new
        
    # Final evaluation
    final_corr = compute_overlap(best_h)
    final_c5 = np.max(final_corr)
    
    # Ensure sum constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
