import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Generate multiple initializations
    initial_h_values = [h_init.copy()]
    
    # Add 5 random initializations
    for _ in range(5):
        h_random = np.random.uniform(0, 1, n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        initial_h_values.append(h_random)
    
    best_h = None
    best_c5 = float('inf')
    
    for i, h_start in enumerate(initial_h_values):
        # Ensure sum constraint and bounds
        h = np.clip(h_start, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Convex relaxation attempt with ECOS
        try:
            h_var = cp.Variable(n_points)
            # For initial relaxation we don't solve the actual objective but
            # get an approximate convex upper bound via a simpler quadratic program
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2,
            ]
            # This is just a feasibility problem to help with initialization
            prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h = np.array(h_var.value)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
        
        # Now refine using L-BFGS-B
        def c5_objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(corr)
        
        def c5_grad(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = correlate(h_vec, 1 - h_vec, mode='full') * dx
            max_idx = np.argmax(corr)

            # Gradient approximation - finite diff for simplicity
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full') * dx
                grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
            return grad
        
        # Optimize with hard constraint enforcement
        def constraint_func(x):
            return np.sum(x) * dx - 1.0  # should be 0

        # L-BFGS-B optimization
        bounds = [(0, 1) for _ in range(n_points)]
        options = {'maxiter': 2000, 'gtol': 1e-6, 'ftol': 1e-6}
        
        try:
            result = minimize(
                c5_objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=c5_grad,
                options=options,
                callback=None
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_opt = c5_objective(h_opt)
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
        except Exception as e:
            continue
            
        if time.time() - start_time > budget_s - 1:
            break
    
    if best_h is None:
        # Fallback to initial solution
        h = h_init.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode='full') * dx
        best_c5 = np.max(corr)
        best_h = h
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    # Ensure sum constraint 
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
