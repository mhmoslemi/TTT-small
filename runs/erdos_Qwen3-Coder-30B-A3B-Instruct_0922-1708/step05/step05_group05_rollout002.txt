import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Default construction: piecewise constant
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
    
    # Normalize to satisfy sum constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Objective function to minimize
    def obj(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient approximation using finite differences
    def grad(h, eps=1e-5):
        grad_arr = np.zeros_like(h)
        base_val = obj(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad_arr[i] = (obj(h_plus) - base_val) / eps
        return grad_arr
    
    # Projection onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Trust region method with adaptive step size
    max_iter = 10000
    alpha = 0.01  # Initial step size
    best_h = h_init.copy()
    best_c5 = obj(best_h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        g = grad(best_h)
        
        # Compute candidate step
        h_candidate = best_h - alpha * g
        h_candidate = project(h_candidate)
        
        # Evaluate candidate
        c5_candidate = obj(h_candidate)
        
        # Accept if improvement
        if c5_candidate < best_c5:
            best_h = h_candidate.copy()
            best_c5 = c5_candidate
            # Increase step size
            alpha = min(alpha * 1.1, 0.1)
        else:
            # Decrease step size
            alpha *= 0.5
            
            # perturb occasionally to escape local minima
            if iteration % 50 == 0 and alpha < 0.001:
                noise = np.random.normal(0, 0.01, n_points)
                h_perturbed = best_h + noise
                h_perturbed = project(h_perturbed)
                c5_perturbed = obj(h_perturbed)
                if c5_perturbed < best_c5:
                    best_h = h_perturbed.copy()
                    best_c5 = c5_perturbed
        
        # Occasionally do random exploration
        if iteration % 200 == 0:
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            c5_rand = obj(h_rand)
            if c5_rand < best_c5:
                best_h = h_rand.copy()
                best_c5 = c5_rand
    
    # Final product
    return (best_h.tolist(), best_c5, n_points)
