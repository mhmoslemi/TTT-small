import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def compute_fitness(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / len(h))
    return c5

def create_initial_population(n_points, swarm_size, initial_h_values=None):
    """Create initial population with diversity"""
    population = []
    
    if initial_h_values is not None:
        h_parent = np.array(initial_h_values)
        h_parent = project_to_feasible(h_parent, n_points)
        
        # 50% variants from parent with noise
        for _ in range(swarm_size // 2):
            noise = np.random.normal(0, 0.05, n_points)
            h = h_parent + noise
            h = project_to_feasible(h, n_points)
            population.append(h)
            
        # 50% random feasible solutions
        for _ in range(swarm_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # All random
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
            
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent using tournament selection"""
    selected_idx = np.random.randint(0, len(population))
    for _ in range(tournament_size - 1):
        competitor_idx = np.random.randint(0, len(population))
        if fitnesses[competitor_idx] < fitnesses[selected_idx]:
            selected_idx = competitor_idx
    return selected_idx

def blend_crossover(parent1, parent2, alpha=0.8):
    """Blend crossover with alpha parameter"""
    offspring1 = parent1.copy()
    offspring2 = parent2.copy()
    
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            lb = min(parent1[i], parent2[i])
            ub = max(parent1[i], parent2[i])
            range_val = ub - lb
            offset1 = alpha * range_val * (np.random.rand() - 0.5)
            offset2 = alpha * range_val * (np.random.rand() - 0.5)
            offspring1[i] = parent1[i] + offset1
            offspring2[i] = parent2[i] + offset2
            
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.1, max_mutation=0.05):
    """Add Gaussian noise to individual"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, max_mutation)
    return mutated

def gradient_descent_refinement(h, max_iter=50, learning_rate=0.001):
    """Locally refine solution using gradient descent"""
    h = h.copy()
    
    for _ in range(max_iter):
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = compute_fitness(h_plus)
            grad[j] = (c5_plus - compute_fitness(h)) / eps
            
        h = h - learning_rate * grad
        h = project_to_feasible(h, len(h))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    swarm_size = 100
    max_generations = 1000
    refinement_interval = 50
    start_time = time.time()
    
    # Initialize population
    population = create_initial_population(n_points, swarm_size)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(ind) for ind in population]
    
    best_individual = population[np.argmin(fitnesses)]
    best_fitness = min(fitnesses)
    
    # Main evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        while len(new_population) < swarm_size:
            # Selection
            idx1 = tournament_selection(population, fitnesses)
            idx2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = blend_crossover(population[idx1], population[idx2])
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
            
        # Keep population size to swarm_size
        population = new_population[:swarm_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(ind) for ind in population]
        
        # Update best individual
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_fitness:
            best_fitness = fitnesses[min_idx]
            best_individual = population[min_idx].copy()
        
        # Periodic refinement
        if generation % refinement_interval == 0 and generation > 0:
            # Refine top 10%
            top_indices = np.argsort(fitnesses)[:swarm_size//10]
            for idx in top_indices:
                refined = gradient_descent_refinement(population[idx], max_iter=10)
                refined_fitness = compute_fitness(refined)
                if refined_fitness < fitnesses[idx]:
                    population[idx] = refined
                    fitnesses[idx] = refined_fitness
                    if refined_fitness < best_fitness:
                        best_fitness = refined_fitness
                        best_individual = refined.copy()
    
    # Final refinement
    final_refinement = gradient_descent_refinement(best_individual, max_iter=100)
    final_fitness = compute_fitness(final_refinement)
    
    if final_fitness < best_fitness:
        best_individual = final_refinement
        best_fitness = final_fitness
    
    return (best_individual.tolist(), best_fitness, n_points)
