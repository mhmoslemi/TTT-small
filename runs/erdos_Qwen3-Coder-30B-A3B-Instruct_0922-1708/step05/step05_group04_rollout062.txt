import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 500
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Compute initial fitness for all population
    fitness_values = []
    for h in population:
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        fitness_values.append(fitness)
    
    # Find best individual
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx].copy()
    best_fitness = fitness_values[best_idx]
    
    # Main DE loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Adjust mutation factor over time
        current_mutation = mutation_factor * (0.99 ** (iter_count // 50))
        
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            h1, h2, h3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = h1 + current_mutation * (h2 - h3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to ensure sum = n_points/2
            mutant_sum = np.sum(mutant)
            if abs(mutant_sum - n_points / 2) > 1e-10:
                diff = mutant_sum - n_points / 2
                if diff > 0:
                    # Remove excess proportionally
                    for j in range(n_points):
                        dec = min(mutant[j], diff * mutant[j] / mutant_sum)
                        mutant[j] -= dec
                        diff -= dec
                        if diff <= 0:
                            break
                else:
                    # Add deficit proportionally
                    for j in range(n_points):
                        inc = min(1 - mutant[j], -diff * mutant[j] / mutant_sum)
                        mutant[j] += inc
                        diff += inc
                        if diff >= 0:
                            break
                mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob or j == 0:
                    trial[j] = mutant[j]
            
            # Evaluate
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(correlation) * dx
            
            # Selection
            if trial_fitness < fitness_values[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness_values[i])
        
        population = new_population
        fitness_values = new_fitness
        
        # Update global best
        min_idx = np.argmin(fitness_values)
        if fitness_values[min_idx] < best_fitness:
            best_fitness = fitness_values[min_idx]
            best_h = population[min_idx].copy()
    
    # Final local refinement
    h_refined = best_h.copy()
    
    # Simple gradient-like refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            # Repair
            h_plus = np.clip(h_plus, 0, 1)
            h_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_refined = h_refined - 0.001 * grad
        # Repair
        h_refined = np.clip(h_refined, 0, 1)
        h_sum = np.sum(h_refined)
        h_refined = h_refined * (n_points / 2) / h_sum
    
    # Final evaluation
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
