import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        fitnesses.append(fitness)
    
    # Find best individual
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # DE main loop
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        new_fitnesses = []
        
        for i in range(population_size):
            # Selection of three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            h1, h2, h3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = h1 + mutation_factor * (h2 - h3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to maintain sum constraint
            mutant_sum = np.sum(mutant)
            if abs(mutant_sum - n_points / 2) > 1e-10:
                diff = mutant_sum - n_points / 2
                if diff > 0:
                    # Remove excess from largest elements
                    indices = np.argsort(mutant)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        dec = min(mutant[idx], diff)
                        mutant[idx] -= dec
                        diff -= dec
                else:
                    # Add deficit to smallest elements
                    indices = np.argsort(mutant)
                    for idx in indices:
                        if diff >= 0:
                            break
                        inc = min(1 - mutant[idx], -diff)
                        mutant[idx] += inc
                        diff += inc
                mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Evaluate trial vector
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(correlation) * dx
            
            # Selection
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
                new_fitnesses.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitnesses.append(fitnesses[i])
            
            # Update global best
            if trial_fitness < best_fitness:
                best_fitness = trial_fitness
                best_h = trial.copy()
        
        population = new_population
        fitnesses = new_fitnesses
        
        # Update mutation factor
        if current_iter % 50 == 0 and current_iter > 0:
            mutation_factor *= 0.99
        
        current_iter += 1
    
    # Final refinement
    h_ref = best_h.copy()
    for _ in range(100):
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_ref -= 0.001 * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref_sum = np.sum(h_ref)
        if abs(h_ref_sum - n_points / 2) > 1e-10:
            h_ref = h_ref * (n_points / 2) / h_ref_sum
    
    corr_final = np.correlate(h_ref, 1 - h_ref, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (h_ref.tolist(), c5_final, n_points)
