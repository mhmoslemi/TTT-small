import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    max_iter = 500
    eta = 0.01  # Learning rate
    eps = 1e-4  # Gradient estimation step
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create initial guess
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial h
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Compute initial Fourier coefficients
    h_fft = fft(h_init)
    f_coeffs = h_fft[:n_fourier]
    
    # Function to reconstruct h from Fourier coefficients
    def reconstruct_h(coeffs):
        # Extend to full spectrum
        full_fft = np.zeros(n_points, dtype=complex)
        full_fft[:n_fourier] = coeffs
        full_fft[-n_fourier+1:] = np.conj(coeffs[-1:0:-1])
        # Reconstruct time domain
        h = ifft(full_fft).real
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to correct sum
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Objective function
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    best_h = h_init.copy()
    best_c5 = compute_c5(best_h)
    
    # Optimization loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Reconstruct h
        h = reconstruct_h(f_coeffs)
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_fourier)
        for i in range(n_fourier):
            delta = np.zeros(n_fourier)
            delta[i] = eps
            h_plus = reconstruct_h(f_coeffs + delta)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update coefficients
        new_coeffs = f_coeffs - eta * grad
        
        # Clip coefficients to prevent blowup
        new_coeffs = np.clip(new_coeffs, -10, 10)
        
        f_coeffs = new_coeffs
        
        # Check if current solution is better and update
        h_new = reconstruct_h(f_coeffs)
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
