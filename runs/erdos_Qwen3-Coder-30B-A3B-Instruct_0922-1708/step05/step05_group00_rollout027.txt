import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach
    n_start = 50  # Initial coarse grid
    n_points = 200  # Final fine grid
    dx_start = 2.0 / n_start
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to constraint
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Adam optimizer
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse grid optimization
    steps = 0
    patience = 0
    max_patience = 200
    
    while steps < 2000 and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx_start
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
            
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps_fd = 1e-4
        
        for i in range(n_start):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_start
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps_fd
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(steps + 1))
        v_hat = v / (1 - beta2**(steps + 1))
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_start / 2) / np.sum(h)
        
        steps += 1
    
    # Transfer to fine grid and refine
    if len(best_h) != n_points:
        # Interpolate from coarse to fine grid
        x_coarse = np.linspace(0, 2, n_start)
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, best_h)
    else:
        h_fine = best_h
        
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine grid optimization
    m = np.zeros_like(h_fine)
    v = np.zeros_like(h_fine)
    lr = 0.01
    
    best_h = h_fine.copy()
    best_c5 = float('inf')
    patience = 0
    
    for steps in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
            
        # Compute finite difference gradient
        grad = np.zeros_like(h_fine)
        eps_fd = 1e-4
        
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps_fd
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(steps + 1))
        v_hat = v / (1 - beta2**(steps + 1))
        
        h_fine = h_fine - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Validate constraints
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_correlation)
    
    return (best_h.tolist(), final_c5, n_points)
