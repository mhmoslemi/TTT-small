import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute C5 fitness for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h * (n_points/2)"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    else:
        h = np.ones_like(h) * (n_points / 2.0) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual via tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.1, std=0.03):
    """Gaussian mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    individual += mask * np.random.normal(0, std, len(individual))
    return individual

def gradient_descent_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine solution using gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        # Estimate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_fitness = compute_fitness(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            fitness_plus = compute_fitness(h_plus, dx)
            grad[i] = (fitness_plus - base_fitness) / eps
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        # Re-evaluate
        new_fitness = compute_fitness(h, dx)
        if new_fitness >= base_fitness:
            break
        base_fitness = new_fitness
    return h

            h = project_to_feasible(h, n_points)
            population.append(h)
        # Add random individuals
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Initialize entirely with random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(h, dx) for h in population]
    
    # Track best
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Selection, crossover, mutation
        new_population = []

            # Selection
            parent1 = tournament_selection(population, fitnesses)

            child1 = mutate(child1)

    # Final verification

    return (best_h.tolist(), final_fitness, n_points)
