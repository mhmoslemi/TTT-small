import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Penalty parameter
    penalty_mu = 0.1
    
    # Variables for plateau detection
    last_improvement = 0
    current_plateau = 0
    current_lr = learning_rate
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            current_plateau = 0
        else:
            current_plateau += 1
            
        # Check for stagnation and adjust learning rate
        if current_plateau >= plateau_patience:
            if current_lr > 1e-6:
                current_lr *= learning_rate_decay
            current_plateau = 0
            
            # Apply random perturbation
            perturb_indices = np.random.choice(n_points, int(n_points * perturbation_fraction), replace=False)
            h[perturb_indices] += np.random.normal(0, 0.01, len(perturb_indices))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Early restart
            if iteration - last_improvement > plateau_patience * 2:
                if 'initial_h_values' in globals():
                    h = np.array(initial_h_values)
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    h = np.ones(n_points) * 0.5
                    h = h * (n_points / 2) / np.sum(h)
                current_lr = learning_rate
                current_plateau = 0
                last_improvement = iteration
        
        # Compute gradient
        k_max = np.argmax(corr_full)
        n_corr = len(corr_full)
        k = k_max - (n_corr - 1) // 2  # Convert from full to lag index
        
        # Gradient from correlation
        grad = np.zeros(n_points)
        if k >= 0:
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])
        else:
            k = -k
            for i in range(n_points - k):
                grad[i + k] += (1 - h[i])
        
        # Apply penalty gradient
        penalty_grad = np.zeros(n_points)
        
        # Constraint: h[i] >= 0
        penalty_grad += 2 * penalty_mu * np.clip(-h, 0, np.inf)
        
        # Constraint: h[i] <= 1
        penalty_grad += 2 * penalty_mu * np.clip(h - 1, 0, np.inf)
        
        # Constraint: sum(h) = n/2
        total_violation = np.sum(h) - n_points / 2
        penalty_grad += 2 * penalty_mu * total_violation
        
        # Combine gradients
        total_grad = grad * dx + penalty_grad
        
        # Update h
        h = h - current_lr * total_grad
        
        # Repair feasibility
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            h = h * (target_sum / h_sum)
        
        # Adapt penalty
        if (np.any(h < 0) or np.any(h > 1) or abs(h_sum - target_sum) > 1e-4):
            penalty_mu *= penalty_factor
        else:
            penalty_mu = max(penalty_mu / penalty_factor, penalty_min)
    
    # Final re-evaluation to make sure it's the correct value
    corr_full = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_full) * dx
    
    return (best_h.tolist(), final_c5, n_points)
