import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary variables (ideal states)
    z = (h > 0.5).astype(float)
    
    # Penalty parameter
    mu = 0.1
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Main optimization loop
    max_iter = 5000
    lr = 0.001
    mu_factor = 1.2
    perturb_frac = 0.01
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        iter_count += 1
        
        # Compute overlap at current h
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find index of maximum overlap
        max_idx = np.argmax(corr)
        # max_lag = max_idx - (len(h) - 1)
        
        # Compute and apply gradient ascent step for the identified lag
        # Gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # For that specific lag (hardcoded for now or can find from correlation)
        actual_lag = max_idx - (len(h) - 1)
        
        # approximating gradient for the max overlap lag
        # This is approximately: d/dh_i [h ⋆ (1-h)]_{actual_lag}
        # The result is: -2 * 1_{i+actual_lag} + 1_{i} = -2 * h[i+actual_lag] + h[i]
        # Actually, the correct math:
        # If the convolution at lag k has value h*(1-h)[k] = sum over i of h[i] * (1 - h[i+k])
        
        # Simple sample-wise method: for computation efficiency
        # Compute actual gradient contribution for each h[i]
        for i in range(len(h)):
            # Compute derivative of max_k h[i] * (1 - h[i+k]) where max_k is at max_idx
            # Approximate by finite difference
            delta_h = np.zeros_like(h)
            delta_h[i] = eps
            h_plus = h + delta_h
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update h
        h = h + lr * grad
        h = np.clip(h, 0, 1)
        
        # Repair: ensure sum = n_points/2
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            if h_sum > target_sum:
                # Reduce largest elements
                sorted_indices = np.argsort(h)[::-1]
                excess = h_sum - target_sum
                for idx in sorted_indices:
                    if excess <= 0:
                        break
                    reduce_amount = min(excess, h[idx])
                    h[idx] -= reduce_amount
                    excess -= reduce_amount
            else:
                # Increase smallest elements
                sorted_indices = np.argsort(h)
                deficit = target_sum - h_sum
                for idx in sorted_indices:
                    if deficit <= 0:
                        break
                    increase_amount = min(deficit, 1 - h[idx])
                    h[idx] += increase_amount
                    deficit -= increase_amount
            h = h * (n_points / 2) / np.sum(h)

        # Update binary decision variables
        z = (h > 0.5).astype(float)
        
        # Update penalty and jitter
        if iter_count % 100 == 0:
            # Apply penalty for non-binary solution
            mu *= mu_factor
            
        # Apply small amount of random noise occasionally
        if iter_count % 300 == 0:
            noise = np.random.normal(0, perturb_frac, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
