import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Scale to correct sum
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
        
    best_c5 = evaluate_h(best_h, dx)
    
    # Main optimization loop
    max_iter = 10000
    learning_rate = 0.001
    perturb_rate = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient estimation
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        # Update with gradient
        new_h = best_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Random perturbation to escape local minima
        if np.random.rand() < perturb_rate or stagnation_count > max_stagnation:
            perturb_idx = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
            for idx in perturb_idx:
                delta = np.random.normal(0, 0.05)
                best_h[idx] += delta
            best_h = repair_h(best_h, n_points)
            best_c5 = evaluate_h(best_h, dx)
            stagnation_count = 0
            
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
