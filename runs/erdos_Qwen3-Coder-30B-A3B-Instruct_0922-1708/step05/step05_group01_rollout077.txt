import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    lr = initial_learning_rate
    last_improvement = 0
    current_patience = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            current_patience = 0
        else:
            current_patience += 1
        
        # Exit if plateaued
        if current_patience > plateau_patience:
            if lr > min_learning_rate:
                lr *= learning_rate_decay
                current_patience = 0
            else:
                break
        
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_val = corr[max_lag_idx]
        
        # Compute gradient
        # Correlation: (h ⋆ (1-h))[k] = sum h[i] * (1-h[i+k])
        # For lag k, grad wrt h[i] = (1 - h[i+k]) if i+k < n_points and 0 otherwise
        # Also grad wrt h[i+k] = -h[i] if i+k < n_points and 0 otherwise
        # But we want the full gradient from all terms that contribute to the maximum
        # We only compute gradient for the specific lag that achieves max
        
        # The gradient w.r.t. h[i] of (h ⋆ (1-h))[k] is:
        # 1) if i+k < n_points: (1 - h[i+k]) 
        # 2) if i >= k: -h[i-k]
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            if i + max_lag_idx - (n_points - 1) < n_points:  # ensure valid index
                # We need to compute the appropriate indices 
                # in the full convolution (we have offset due to mode='full')
                # k can range from 0 to 2*n_points - 2
                # So we have 0-indexed indices in the convolution: actual shift is
                # k - (n_points - 1)
                actual_shift = max_lag_idx - (n_points - 1)
                if actual_shift >= 0:
                    # h[i] contributes to (1-h[i+actual_shift]) term 
                    # So gradient wrt h[i] = -1 if shifted index valid
                    if i + actual_shift < n_points:
                        gradient[i] -= 1.0
                else:
                    # h[i] contributes to h[i - actual_shift] term in 1-h
                    # written as h[i+(-actual_shift)] in 1-h, we look at elements for 1-h
                    # i - actual_shift = i - (-shift) = i + shift  
                    # So need i + shift < n_points (element exists)
                    shift = -actual_shift  # becomes positive
                    if i + shift < n_points:
                        gradient[i] += 1.0
            if i - (max_lag_idx - (n_points - 1)) >= 0:
                # Backwards contribution from h[i-shift] term? 
                # Direct derivative of (h ⋆ (1-h))[k]: 
                # 1) For term h[i] * (1-h[i+k]) where k is fixed lag
                #    d/dh[i] = 1 - h[i+k] if i+k < n_points
                # 2) For term h[i+k] * (1-h[i]) where k is fixed lag  
                #    d/dh[i] = -h[i+k] if i+k < n_points and i >= k
      
                # First form  
                if i >= (max_lag_idx - (n_points - 1)) and i + (max_lag_idx - (n_points - 1)) < n_points:
                    # This also needs double checking.
                    pass
            
        # Gradient trick: we will derive a simpler expression
        # Let Conv_k = sum_{i=0}^{n-1} h[i] * (1-h[i+k])  
        # The gradient to the maximum one
        # For each i in h, the variables 1-h are such that
        # the derivative wrt h[i] comes from the shift k = max_lag_idx:
        # term h[i] * (1-h[i+k])  
        # dx = 1.0 (since we're in terms of h, and we're looking for derivative
        # w.r.t. h[i], we have local computation on indices)
        # d(Conv[k])/dh[i] are:
        #  * -1 if k = i+lag
        #  * +1 if k = i-lag 
        # Actually, let’s do it differently
        
        # Reset gradient
        gradient = np.zeros(n_points)
        
        # The actual derivative of correlation wrt h[i] for particular max lag k
        # is the sum over all i using indices
        # (h ⋆ (1 - h))[k] = sum (h[i] * (1 - h[i-k])) for all valid i
        # d(h ⋆ (1 - h))[k] / dh[i] =  
        # if i - k >= 0 and i - k < n: (1 - h[i-k])
        # if k = i and i < n: -h[i] 
        # etc.
        # This is delicate. Just write out calculation:
        
        # Due to mode='full', lags 0..2n-2 correspond to shifts -n+1..n-1.
        # Lag k (0..2n-2) actually corresponds to shift k+1-n.
        shift = max_lag_idx + 1 - n_points
        
        # Clarify:  corr[k] = sum_{i=0}^{n-1} h[i] * 1-h[i-k] 
        # where if i-k < 0: then h[i-k] = h[0] with k-i (or similar logic)
        # Therefore CONTRIBUTION ORDER: 
        # element h[i] of the first sequence multiplies 1-h[i-k] of the second
        # If we're concerned about a specific lag, 
        # contribution to derivative of h[i] from h[i+shift] term =>
        # positive contribution in 1 term, so -h[i] for those correct indices  
        
        # The derivative of (h ⋆ (1 - h))[k] wrt h[i]  
        # is sum over k' of derivative at k' wrt h[i]
        
        # We trace the derivative of corr_k = sum_{i=0}^{n-1} h[i] * (1 - h[i-k]) 
        # Derivative = 1 if i-k >= 0 and i-k < n, and 0 else.
        # i-k >= 0 -> i >= k. i-k < n -> i < n+k. 
        # So i ranges from max(k,0) to min(n+k, n-1). And the derivative is:
        #  1-h[i-k]    for i where i >=k 
        # -h[i+k]      for i where i+k < n  
        # But we're indexing h[i-k] and h[i+k], so it's messy.
        # Actually, to get the gradient correctly using standard derivation rules,
        # just ZIP[ h  , 1-h ] convolution gradients:
        # Let the expanded version where the second sequence is reversed:
        # D_k = sum_{i} h[i] * (1 - h[i-k]) = (h * (1-h)_{reverse})[k]
        # Then:
        # dD_k/dh[i] = (1-h[i-k]) * I_{i-k >= 0 and i-k < n} for the first element
        #              -h[i+k] * I_{i+k < n} for the second element  
        # Simplifying:
        gradient = np.zeros_like(h)
        
        # formatted indices for max_lag_idx
        # index = i where h[i] influences the maximum term corr[max_lag_idx]
        # That is h[i] is part of cross product h[i]*(1-h[i-k]) where k is determined
        # from max_lag_idx. If lag k is max_lag_idx among all lags, 
        # we take derivative of that.
        # This is easy way: the derivative thatloses us maximum k:
        # determine all i where term h[i] is involved in corr[max_lag_idx].
        
        # Correlation flexibly maps i to h[i] and h[i-k] for lag k=lagidx-n+1
        # so i from 0..n-1, and for any i, the term with 1-h[i-k] contributes
        # if i-k >= 0 and i-k < n. So the derivative contributions are:
        # if i+k < N: -h[i]
        # if i-k >= 0: (1 - h[i-k])
        # Let’s compute it manually:
        # The contribution of gradient for each position is: 
        #  how many terms contribute a derivative
        #  (frac) -h[i] += 1 when INDEX = i+k in h 
        #  (frac) +1 when INDEX = i-k in h contributes part  
        # When i-th element contributes to lag k:
        # The lag idx = k = i + shift (since single lag) where shift = k - (n-1) = lags_idx - n+1
        # This implies that for the maximum lag, we rewrite everything
        # shift=0: z[i] = (1-h[i])  contribution = -h[i] 
        # shift = -1: h[i-1] * (1 - h[i]) gives d/dh[i] = -1 * (1-h[i])  
        # shift = +1: h[i] * (1 - h[i+1]) gives d/dh[i] = 1 * (1 -h[i+1])  
        # etc.
        # So for specifying actual h[i] contribution, cause we use corr[n-1] = k=0,
        # we have diff lags. This parity confusion. Let’s step by step.
        
        # Start cleaner:
        # Hi is -> 1-h_j entries, so i corresponds to j, convolution index is i-k to match
        # under full mode:
        # corr k = sum_{i = 0}^{n-1}  h[i] * (1-h[i-k]) = (h * reverse 1-h)[k]  
        # So it’s transposed.
        # The derivative of summation is 
        # K = {i: 0 <= i, i <= n-1} 
        # for each i: h[i] contributes to corr[k] if i-k >= 0 and i-k < n
        # This means for i from {k} to {n-1}, and k also from 0 to 2*n-2, 
        # For fixed i: if k >= 0 and k <= (n-1), h[i] contributes to k-th element  
        # But for the specific maximum element corr[max_lag_idx]
        # The relative shift = max_lag_idx - (n-1)
        # So for fixing current point like h[i], it is in term (h ⋆ (1-h))[k] 
        # when k = i + shift. 
        # But that is h[i] * (1 - h[i-shift]) in full mode.
        # So the term k corresponds to index k = i + (max_lag_idx - (n-1))
        # Thus for h[i], effect on maximum correaltion:
        # if max_lag_idx - (n-1) < 0, so k = i + neg = i - abs  
        # then h[i] contributes in the form h[i]* (1 - h[i-shift])  
        # So argument of 1-h is i-shift (i + abs) = i + |k| = i + shift =
        # So the k = i + shift, i runs from 0..n-1.
        # Let’s carefully redefine derivatives:
        # Let shift be such that shift = max_lag_idx - (n-1) 
        # Lag index max_lag_idx corresponds to shift=shift  
        # then 
        # (h ⋆ (1-h))[max_lag_idx] = sum_{i=0}^{n-1} h[i] * (1-h[i-shift]) 
        # if shift==0 then k=0, it's simple right from i. i-k bound case. 
        # Let's do separately for h[i] influence
        # j = i, k = j - i, so the identifying derivative wrt h[i]:
        # k = max_lag_idx = i + shift 
        # i+k <= n-1, i + shift + k <= n-1 As k is same h[i], 
        # Actually we just make better solitary calculation.
        
        # Every h[i] contributes to exactly one lag:
        # Which is the lag k = (i + shift) resp. i= k-shift = (i+shift) -shift = i  
        # No, i goes from i to n-1, and the maximization goes over all.
        # The derivative of (h ⋆ (1-h))[k] wrt h[i] is:
        #  if 0 <= i-shift < n: (1 - h[i-shift])
        #  if 0 <= i+shift < n: -h[i+shift]  
        # combined for full lag term i depends:  
        
        # The easy case: 
        # corr[k] = sum(i) h[i] * (1 - h[i-k]) if k is given shift  
        # We're under the condition in which max_lag_idx is maximum
        # So: max_k (h_k) and let this be at k, where h[i] term influence...
        # gradient(k) = d/dh[i] ( h[i] * (1-h[i-k]) ) =  (1-h[i-k]) * I{i-k < n} * I{i >= 0} 
        # + d/dh[i] ( h[i+k] * (1-h[i]) ) = -h[i+k] * I{i+k < n} * I{i >= 0} 
        # (where I is indicator function that i+k < n)
        # This is correct derivative if we reduced indices correctly.
        # h[i] * (1-h[i-k]) contributes to corr_{k}  
        # derivative wrt h[i] = 1 - h[i-k]   
        # h[i+k] * (1-h[i]) contributes to shifted case, reverse index k:
        # derivative wrt h[i] = - h[i+k] 
        
        # Compute cumulative gradients  
        gradient = np.zeros(n_points)
        
        # index current lag k on correlation array - the maximum
        shift = max_lag_idx - (n_points - 1)
        
        # Following: w.r.t. h[i], we have:
        # from i to i+shift term derivative is (1 - h[i-shift])
        # from i to i-shift term derivative is - (h[i+shift])  
        # but we're on an edge!
        
        # Simplified derivation:
        # Use position error: compute twelve rows.
        for i in range(n_points):
            if 0 <= i - shift < n_points:
                gradient[i] += 1 - h[i - shift]  # from shifting by -shift  
            if 0 <= i + shift < n_points:
                gradient[i] -= h[i + shift]  # From 1-h[i+k] where k=-shift 
    
        # Normalize gradient by dx = (2/100) = 0.02 
        gradient = gradient * 2 / n_points
        
        # Standard gradient descent step
        h = h - lr * gradient
        
        # Repair feasibility constraints:
        h = np.clip(h, 0, 1)  # Ensure values in [0,1]
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum  # Rescale to exact sum
        
    return (best_h.tolist(), best_c5, n_points)
