import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import dct, idct
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    max_iter = 500
    lr = 0.01
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a reasonable initial guess
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        
    # Convert initial h to Fourier coefficients
    def h_to_fourier(h):
        # Ensure the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Use DCT to get cosine coefficients
        return dct(h, type=2, norm='ortho')[:n_fourier]
    
    def fourier_to_h(coeffs):
        # Extend to full length
        full_coeffs = np.zeros(n_points)
        full_coeffs[:n_fourier] = coeffs
        # Inverse DCT to get h values
        h = idct(full_coeffs, type=2, norm='ortho')
        # Clip to [0,1] and rescale sum
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to evaluate C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Gradient computation with finite differences
    def compute_gradient(coeffs, eps=1e-4):
        grad = np.zeros_like(coeffs)
        base_h = fourier_to_h(coeffs)
        base_c5 = compute_c5(base_h)
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = fourier_to_h(coeffs_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Initialize optimization
    current_coeffs = h_to_fourier(h_init)
    best_c5 = compute_c5(fourier_to_h(current_coeffs))
    best_coeffs = current_coeffs.copy()
    
    # Optimization loop
    iteration = 0
    stagnation = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        # Compute gradient
        grad = compute_gradient(current_coeffs)
        
        # Update
        current_coeffs -= lr * grad
        
        # Clip coefficients to prevent explosion
        current_coeffs = np.clip(current_coeffs, -10, 10)
        
        # Convert back to h and compute C5
        h = fourier_to_h(current_coeffs)
        c5 = compute_c5(h)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_coeffs = current_coeffs.copy()
            stagnation = 0
        else:
            stagnation += 1
            
        # Terminate if stagnation
        if stagnation > 100:
            break
            
        iteration += 1
    
    # Final result
    final_h = fourier_to_h(best_coeffs)
    
    return (final_h.tolist(), best_c5, n_points)
