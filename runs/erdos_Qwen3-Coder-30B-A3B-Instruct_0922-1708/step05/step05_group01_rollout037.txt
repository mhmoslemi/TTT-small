import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Current best
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Optimization settings
    max_iter = 6000
    alpha = 0.01
    decay_every = 200
    eval_freq = 50
    stagnation_limit = 200
    perturb_freq = 100
    perturb_fraction = 0.01
    
    # Track improvement
    last_improvement = 0
    current_c5 = best_c5
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Adaptive learning rate
        if iteration > 0 and iteration % decay_every == 0:
            alpha *= 0.5
            
        # Evaluation
        if iteration % eval_freq == 0:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < current_c5:
                current_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
                if c5 < best_c5:
                    best_c5 = c5
        
        # Stagnation check
        if iteration - last_improvement > stagnation_limit:
            # Reinitialize with a better random start
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            continue
            
        # Perturbation
        if iteration % perturb_freq == 0:
            # Add Gaussian noise to some elements
            noise_indices = np.random.choice(n_points, 
                                           size=int(n_points * perturb_fraction), 
                                           replace=False)
            noise = np.random.normal(0, 0.01, len(noise_indices))
            h[noise_indices] += noise
            # Clamp and rescale
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            continue
            
        # Gradient-based update (simplified)
        # Compute gradient approximation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find where the maximum occurs
        max_lag = np.argmax(corr)
        
        # Calculate gradient using finite difference
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Apply gradient descent
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Final check
        if iteration == max_iter - 1:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
