import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Parent construction fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        # Evaluate
        corr = np.correlate(h_parent, 1 - h_parent, mode='full')
        c5_parent = np.max(corr) * dx
        if c5_parent < best_c5:
            best_c5 = c5_parent
            best_h = h_parent.copy()
    
    start_time = time.time()
    
    # Scale pyramid
    scales = [10, 20, 40, 80, 100]
    h_prev = None
    
    for scale in scales:
        if time.time() - start_time > budget_s - 100:
            break
            
        # Generate coarse h
        if scale == 10:
            h = np.ones(10) * 0.5
        else:
            # Interpolate from previous
            x_prev = np.linspace(0, 2, len(h_prev))
            x_new = np.linspace(0, 2, scale)
            h = np.interp(x_new, x_prev, h_prev)
        
        # Convex relaxation optimization
        h_var = Variable(scale)
        # Objective: approximate max correlation over a window
        # We'll use a smoothed average
        lags = range(1, min(scale // 4, 50))  # truncate to prevent huge computation
        total_corr = 0
        num_lags = 0
        
        # For each lag, add simplified correlation approximation
        # Approximation: average of sequential product over lags
        for k in lags:
            if k >= scale: continue
            corr_term = 0
            for i in range(scale - k):
                corr_term += h_var[i] * (1 - h_var[i+k])
            total_corr += corr_term
            num_lags += 1
            
        if num_lags == 0: num_lags = 1
            
        # Minimize average cross-term
        obj = Minimize(total_corr / num_lags)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == scale / 2)
        
        # Add soft constraints to keep within bounds
        # Penalty for deviations from bounds and sum
        penalty_weight = 0.1
        constraints.append(penalty_weight * (sum(h_var) - scale/2)**2 <= 1e-6)
        
        # Solve with ECOS
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=500)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h_opt = np.array(h_var.value).flatten()
                h = h_opt
            else:
                # Fallback to previous if problem fails
                h = h_prev.copy() if h_prev is not None else np.ones(scale) * 0.5
        except:
            # Fallback to previous or random
            h = h_prev.copy() if h_prev is not None else np.ones(scale) * 0.5
        
        h = np.clip(h, 0, 1)
        h = h * (scale / 2) / np.sum(h)
        h_prev = h.copy()
        
        # Downsample if needed, and flatten to 100 points for final use
        if scale == 100:
            h_final = h
        elif scale < 100:
            x_orig = np.linspace(0, 2, scale)
            x_target = np.linspace(0, 2, 100)
            h_final = np.interp(x_target, x_orig, h)
        else:
            continue
            
        # Final check and evaluate
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (n_points / 2) / np.sum(h_final)
        
        # Compute actual C5 with full correlation
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
    
    # Local refinement
    if best_h is not None:
        # Gradient descent to fine-tune, using gradient from finite differences
        h_ref = best_h.copy()
        step = 50
        lr = 0.001
        for _ in range(step):
            if time.time() - start_time > budget_s - 20:
                break
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
                
            h_ref = h_ref - lr * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5_ref = np.max(corr) * dx
            
            if c5_ref < best_c5:
                best_c5 = c5_ref
                best_h = h_ref.copy()
    
    # Ensure final validation
    if best_h is None:
        # Fallback to naive construction
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
