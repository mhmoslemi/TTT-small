import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 3000
    mutation_strength = 0.5
    sigma_initial = 0.1
    stagnation_limit = 200
    restart_factor = 0.9
    decay_rate = 0.999
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Eval function
    def evaluate_h(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Main evolution loop
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    current_mu = mutation_strength
    current_sigma = sigma_initial
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate fitness
        fitnesses = [evaluate_h(h) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Selection: top 50%
        selected_indices = np.argsort(fitnesses)[:population_size//2]
        selected = [population[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent = selected[np.random.randint(0, len(selected))]
            # Mutation
            noise = np.random.normal(0, current_sigma, n_points)
            child = parent + current_mu * noise
            offspring.append(child)
        
        # Repair offspring
        offspring = [repair(h) for h in offspring]
        
        # Replace worst in population with best offspring
        worst_indices = np.argsort(fitnesses)[population_size//2:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Update mutation parameters
        current_mu *= decay_rate
        current_sigma = current_mu / 5.0
        
        # Restart if needed
        if stagnation_count > stagnation_limit:
            current_mu = mutation_strength * restart_factor
            for i in range(len(population)):
                r = np.random.rand() * 0.1
                population[i] += np.random.normal(0, 0.02, n_points) * r
            stagnation_count = 0
        
        current_iter += 1
    
    # Final refinement with gradient descent
    if best_individual is not None:
        h_ref = best_individual.copy()
        for _ in range(100):
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad
            h_ref = repair(h_ref)
            
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        
        corr_final = np.correlate(h_ref, 1 - h_ref, mode='full')
        final_fitness = np.max(corr_final) * dx
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_individual = h_ref.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
