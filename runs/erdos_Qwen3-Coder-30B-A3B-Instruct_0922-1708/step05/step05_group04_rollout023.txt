import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert initial to Fourier coefficients
    h_fft = fft(h_init)
    coeffs = np.concatenate([np.real(h_fft[:n_fourier]), np.imag(h_fft[:n_fourier])])
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.01
    eps = 1e-4
    stagnation_limit = 100
    best_c5 = float('inf')
    best_h = None
    stagnation_count = 0
    
    def to_h(coeffs):
        real_coeffs = coeffs[:n_fourier]
        imag_coeffs = coeffs[n_fourier:2*n_fourier]
        h_fft = np.zeros(n_points, dtype=complex)
        h_fft[:n_fourier] = real_coeffs + 1j * imag_coeffs
        h_fft[n_points-n_fourier:] = real_coeffs[-2:0:-1] - 1j * imag_coeffs[-2:0:-1]
        h = np.real(ifft(h_fft))
        # Ensure it sums to n_points/2 and is in [0,1]
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Convert coefficients to h
        h = to_h(coeffs)
        
        # Compute current C5
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > stagnation_limit:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = to_h(coeffs_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Update coefficients
        coeffs = coeffs - learning_rate * grad
        
        # Clip coefficients to prevent wild oscillations
        coeffs = np.clip(coeffs, -10, 10)
    
    # Final refinement
    if best_h is not None:
        h_ref = best_h.copy()
        for _ in range(100):
            c5 = compute_c5(h_ref)
            grad = np.zeros(n_points)
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus_sum = np.sum(h_plus)
                if h_plus_sum > 0:
                    h_plus = h_plus * (n_points / 2) / h_plus_sum
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / eps
                
            h_ref = h_ref - 0.001 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref_sum = np.sum(h_ref)
            if h_ref_sum > 0:
                h_ref = h_ref * (n_points / 2) / h_ref_sum
                
            c5_new = compute_c5(h_ref)
            if c5_new >= c5:
                break
            c5 = c5_new
    
    if best_h is None:
        best_h = h_ref if 'h_ref' in locals() else h_init
    
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
