import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def mutate_h(h, n_points, mutation_rate=0.15):
    """Generate a mutated version of h."""
    h_mut = h.copy()
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        # Flip or randomly adjust
        if np.random.random() < 0.5:
            h_mut[i] = 1 - h_mut[i]
        else:
            h_mut[i] = np.random.random()
    return repair_h(h_mut, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        current_best = h_init.copy()
    else:
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
        current_best = h_init.copy()
    
    best_value = evaluate_h(current_best, dx)
    
    # Parameters
    n_mutations = 100
    max_stall = 100
    stall_count = 0
    best_stall = best_value
    
    # Main loop
    iterations = 0
    while time.time() - start_time < budget_s - 10 and stall_count < max_stall:
        iterations += 1
        # Generate mutations
        candidates = []
        for _ in range(n_mutations):
            mutated = mutate_h(current_best, n_points)
            candidates.append(mutated)
        
        # Evaluate candidates
        values = []
        for cand in candidates:
            val = evaluate_h(cand, dx)
            values.append(val)
        
        # Find best among candidates
        min_idx = np.argmin(values)
        if values[min_idx] < best_value:
            current_best = candidates[min_idx].copy()
            best_value = values[min_idx]
            stall_count = 0
            best_stall = best_value
        else:
            stall_count += 1
    
    # Local refinement
    best_h = current_best.copy()
    best_c5 = best_value
    
    # Gradient descent on best
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        # Update step
        step_size = 0.001
        best_h = best_h - step_size * grad
        best_h = repair_h(best_h, n_points)
        
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
