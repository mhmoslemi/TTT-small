import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_value = evaluate_h(best_h, dx)

    # Use a mix of gradient descent and random perturbation
    max_iter = 2000
    perturbation_rate = 0.05
    learning_rate = 0.001
    stagnation_threshold = 100
    
    # Track improvement
    last_improvement = 0
    last_best_value = best_value
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Periodically apply random perturbation to escape local minima
        if iter_count % 20 == 0:
            # Randomly perturb a fraction of the elements
            num_perturb = int(n_points * perturbation_rate)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            # Add Gaussian noise to selected indices
            for i in indices:
                best_h[i] += np.random.normal(0, 0.001)
            
            # Repair solution
            best_h = repair_h(best_h, n_points)
            # Recompute value
            current_value = evaluate_h(best_h, dx)
            if current_value < best_value:
                best_value = current_value
                last_improvement = iter_count
            last_best_value = current_value

        # Gradient descent step

        grad = np.zeros(n_points)

            h_plus[j] += eps

            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps

        best_h = best_h - learning_rate * grad
