import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 6000
    alpha = 0.01
    decay_rate = 0.5
    decay_interval = 200
    adjustment_freq = 50
    stagnation_limit = 200
    perturb_freq = 100
    perturb_fraction = 0.01
    
    # Track progress
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Decay learning rate
        if iteration % decay_interval == 0:
            alpha *= decay_rate
            
        # Periodic adjustments
        if iteration % adjustment_freq == 0:
            corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
            # Compute gradient approximation
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_sum = np.sum(h_plus)
                h_plus = h_plus * (n_points / 2) / h_sum
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update h
            h = h - alpha * grad
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
        
        # Periodic perturbations to escape local optima
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, perturb_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            
        # Check for improvement
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Check for stagnation
        if iteration - last_improvement > stagnation_limit:
            # Random restart
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    # Ensure sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
