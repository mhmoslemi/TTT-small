import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise create uniform
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth max approximation parameter
    beta = 10.0
    
    def smooth_max(arr, beta):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def compute_overlap(h_values):
        # Compute correlation of h with (1-h) 
        corr = np.correlate(h_values, 1 - h_values, mode='full') * dx
        return corr
    
    def objective_smooth(h_values):
        overlap = compute_overlap(h_values)
        return smooth_max(overlap, beta)
    
    def gradient_smooth(h_values):
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h_values)
        base_obj = objective_smooth(h_values)
        for i in range(len(h_values)):
            h_plus = h_values.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (objective_smooth(h_plus) - base_obj) / eps
        return grad
    
    # Optimize using Adam-like approach with projection
    h = h.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    best_c5 = float('inf')
    best_h = h.copy()
    
    iterations = 0
    patience = 0
    max_patience = 500
    
    while time.time() - start_time < budget_s - 1:
        # Compute current overlap matrix and optimal value
        corr = compute_overlap(h)
        current_c5 = np.max(corr)
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Compute gradient
        grad = gradient_smooth(h)
        
        # Update with Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iterations+1))
        v_hat = v / (1 - beta2**(iterations+1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region  
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iterations % 100 == 0:
            lr *= 0.995
            
        iterations += 1
    
    # Final check on best solution
    corr_final = compute_overlap(best_h)
    final_c5 = np.max(corr_final)
    
    return (best_h.tolist(), final_c5, n_points)
