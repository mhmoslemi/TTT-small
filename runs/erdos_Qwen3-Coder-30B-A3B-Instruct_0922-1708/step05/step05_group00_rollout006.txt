import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use coarse-to-fine approach with 200 points for final result
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize from provided parent or uniform distribution
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize sum to exactly 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # First solve on coarse grid to escape local minima
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    # Solve coarse grid QP problem using CVXPY
    h_var = cp.Variable(n_coarse)
    C5_var = cp.Variable()
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_coarse / 2,
        C5_var >= 0
    ]
    
    # Add quadratic constraints on overlaps
    # The actual operations get complex in CVXPY, so we simulate:
    # replacing with bounding approach
    # Use a direct quadratic approximation
    try:
        # Formulate as QP problem  
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        prob.solve(solver=ECOS, verbose=False, max_iter=1000)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND] and h_var.value is not None:
            h_coarse = np.array(h_var.value)
            h_coarse = np.clip(h_coarse, 0, 1)
            # Normalize to correct sum
            h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    except:
        pass
    
    # Reinterpolate coarse solution to fine grid
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use gradient-based refinement with adaptive learning rate
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    # Gradient computation using finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        current_c5 = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        return grad
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    lr = 1e-3
    patience = 0
    max_patience = 300
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h)
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping
        if patience > max_patience:
            break
            
        h = h_new
        lr = max(1e-5, lr * 0.9998)
    
    # Final validation
    return (best_h.tolist(), best_c5, n_points)
