import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import Variable, Minimize, Problem, norm

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0.1, 0.9, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute overlap
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Helper function to repair h
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Rescale to meet constraint
        h = h * (n_points / 2) / h_sum
        return h
    
    # Optimize with gradient-based method
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Use gradient-based local search
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-5
        base_c5 = compute_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update
        h -= 0.001 * grad
        h = repair_h(h)
        
        # Check for improvement
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
