import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    # Ensure sum is correct
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients
    def h_to_fourier(h):
        # Use real FFT (conjugate symmetric for real signals)
        H = fft(h)
        # Take only first half (to match n_fourier coefficients)
        return H[:n_fourier]
    
    def fourier_to_h(coeffs):
        # Reconstruct signal from Fourier coefficients
        # Make sure it's real-valued (conjugate symmetric)
        H = np.zeros(n_points, dtype=complex)
        H[:n_fourier] = coeffs
        if n_fourier < n_points // 2 + 1:
            H[n_points - n_fourier:] = np.conj(coeffs[-2:0:-1])
        elif n_fourier == n_points // 2 + 1:
            H[n_points - n_fourier] = np.conj(coeffs[1])
        # FFT with real input should be symmetric:
        # Inverse FFT to get real signal in time domain
        h = ifft(H).real
        return h
    
    def compute_c5(h):
        # Correlation computation
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Small modifications to handle boundary conditions
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            # Rescale to target sum
            scale_factor = (n_points / 2) / h_sum
            h = h * scale_factor
        return h
    
    # Initial check
    h_current = h_init.copy()
    h_current = repair_h(h_current)
    c5_current = compute_c5(h_current)
    c5_best = c5_current
    
    # Optimization loop
    max_iter = 1000
    iter_no_improve = 0
    stagnation_limit = 100
    
    # Use Fourier coefficients as optimization variables
    coeffs = h_to_fourier(h_current)
    # Optimize using L-BFGS-B
    def opt_func(coeffs_flat):
        # Convert to complex form for reconstruction
        coeffs_complex = coeffs_flat[:len(coeffs)] + 1j * coeffs_flat[len(coeffs):]
        # Reconstruct h
        h_reconstructed = fourier_to_h(coeffs_complex)
        h_reconstructed = repair_h(h_reconstructed)
        # Compute c5
        c5_val = compute_c5(h_reconstructed)
        return c5_val
    
    def opt_jac(coeffs_flat):
        eps = 1e-6
        grad_real = np.zeros_like(coeffs_flat)
        for i in range(len(coeffs_flat)):
            coeffs_plus = coeffs_flat.copy()
            coeffs_plus[i] += eps
            c5_plus = opt_func(coeffs_plus)
            coeffs_minus = coeffs_flat.copy()
            coeffs_minus[i] -= eps
            c5_minus = opt_func(coeffs_minus)
            grad_real[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad_real
    
    # Optimization using scipy hints
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        # Create objective function and jacobian
        coeffs_flat = np.concatenate([coeffs.real, coeffs.imag])
        
        # Standard optimization approach to update coeffs
        try:
            # Approximate gradient in step domain
            coeffs_new = coeffs.copy()
            h_test = fourier_to_h(coeffs_new)
            h_test = repair_h(h_test)
            c5_base = compute_c5(h_test)
            
            # Compute gradient
            grad = np.zeros_like(coeffs_flat)
            eps = 1e-6
            
            for i in range(len(coeffs_flat)):
                coeffs_plus = coeffs_flat.copy()
                coeffs_plus[i] += eps
                coeffs_new_plus = coeffs_plus[:len(coeffs)] + 1j * coeffs_plus[len(coeffs):]
                h_plus = fourier_to_h(coeffs_new_plus)
                h_plus = repair_h(h_plus)
                c5_plus = compute_c5(h_plus)
                
                coeffs_minus = coeffs_flat.copy()
                coeffs_minus[i] -= eps
                coeffs_new_minus = coeffs_minus[:len(coeffs)] + 1j * coeffs_minus[len(coeffs):]
                h_minus = fourier_to_h(coeffs_new_minus)
                h_minus = repair_h(h_minus)
                c5_minus = compute_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update
            eta = 0.01
            coeffs_flat_new = coeffs_flat - eta * grad
            coeffs_new = coeffs_flat_new[:len(coeffs)] + 1j * coeffs_flat_new[len(coeffs):]
            
            # Clip coefficients to avoid saturation
            coeffs_new = coeffs_new * 0.99
            
            coeffs = coeffs_new
            h_updated = fourier_to_h(coeffs)
            h_updated = repair_h(h_updated)
            c5_updated = compute_c5(h_updated)
            
            if c5_updated < c5_best:
                c5_best = c5_updated
                h_current = h_updated.copy()
                iter_no_improve = 0
            else:
                iter_no_improve += 1
            
        except Exception as e:
            pass  # Do minimal and safe improvement steps
        
        iter_count += 1
        
        if iter_no_improve >= stagnation_limit:
            break
    
    # Final refinement with local search
    h_final = h_current.copy()
    h_final = repair_h(h_final)
    c5_final = compute_c5(h_final)
    
    # Gradient based local refinement
    last_best = c5_final
    for iter_refine in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        h_ref = h_final.copy()
        h_grad = np.zeros(n_points)
        
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            h_minus = h_ref.copy()
            h_minus[i] -= eps
            h_minus = repair_h(h_minus)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            c5_minus = np.max(corr_minus) * dx
            
            h_grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        h_new = h_ref - 0.001 * h_grad
        h_new = repair_h(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < last_best:
            h_final = h_new.copy()
            last_best = c5_new
        else:
            break
    
    c5_out = compute_c5(h_final)
    return (h_final.tolist(), c5_out, n_points)
