import numpy as np
from scipy.optimize import minimize
import time

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def compute_fitness(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * (2.0 / len(h))
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        # Create variations around parent
        for _ in range(population_size // 2):
            noise = np.random.normal(0, 0.05, n_points)
            h_var = h_parent + noise
            h_var = project_to_feasible(h_var, n_points)
            population.append(h_var)
        # Add random individuals
        for _ in range(population_size // 2):
            h_rand = np.random.rand(n_points)
            h_rand = project_to_feasible(h_rand, n_points)
            population.append(h_rand)
    else:
        # Initialize with random feasible individuals
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    # GA parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    
    start_time = time.time()
    
    for generation in range(10000):  # arbitrary large number, will break on time
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute fitness for all individuals
        fitnesses = [compute_fitness(ind) for ind in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection with tournament
        new_population = []
        for _ in range(population_size):
            # Tournament selection
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            new_population.append(population[winner_idx].copy())
        
        # Crossover and mutation
        for i in range(0, population_size, 2):
            if i + 1 < population_size and np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                parent1 = new_population[i]
                parent2 = new_population[i+1]
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                child1 = project_to_feasible(child1, n_points)
                child2 = project_to_feasible(child2, n_points)
                new_population[i] = child1
                new_population[i+1] = child2
        
        # Mutation
        for i in range(population_size):
            if np.random.rand() < mutation_rate:
                mutation_magnitude = np.random.uniform(0.01, 0.05)
                noise = np.random.normal(0, mutation_magnitude, n_points)
                new_population[i] = new_population[i] + noise
                new_population[i] = project_to_feasible(new_population[i], n_points)
                
        population = new_population
        
        # Periodic refinement of top solutions
        if generation % refinement_interval == 0:
            # Sort by fitness
            sorted_indices = np.argsort(fitnesses)
            # Refine top 10%
            for i in range(population_size // 10):
                idx = sorted_indices[i]
                h_ref = population[idx]
                lr = 0.001
                for _ in range(50):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        c5_plus = compute_fitness(h_plus)
                        grad[j] = (c5_plus - fitnesses[idx]) / eps
                    h_ref = h_ref - lr * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    c5_new = compute_fitness(h_ref)
                    if c5_new < fitnesses[idx]:
                        fitnesses[idx] = c5_new
                        population[idx] = h_ref.copy()
                        
    # Final refinement of best solution
    if best_h is not None:
        h_ref = best_h.copy()
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = compute_fitness(h_plus)
                grad[j] = (c5_plus - best_c5) / eps
            h_ref = h_ref - 0.0005 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            c5_new = compute_fitness(h_ref)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_ref.copy()
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
