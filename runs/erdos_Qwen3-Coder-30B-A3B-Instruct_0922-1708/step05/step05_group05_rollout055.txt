import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlation and max value
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Compute gradient of max overlap w.r.t. h
        max_idx = np.argmax(corr)
        # Approximate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Renormalize
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - c5) / eps
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < c5:
            h = h_new
            c5 = new_c5
            
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Apply random perturbations occasionally
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, perturb_magnitude, n_points)
            h_perturbed = h + noise
            # Project onto feasible region
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            perturbed_c5 = compute_c5(h_perturbed, dx)
            
            if perturbed_c5 < c5:
                h = h_perturbed
                c5 = perturbed_c5
                
        # Adapt learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.95
    
    # Final check
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
