import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 0.001
    mu = 0.1
    mu_factor = 1.2
    perturb_fraction = 0.01

    else:

    z = (h > 0.5).astype(int)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient for the maximum lag
        max_lag_idx = np.argmax(corr)
        k_star = max_lag_idx - (len(h) - 1)
        
        # Create shifted version for gradient computation
        shifted_h = np.roll(h, -k_star) if k_star > 0 else np.roll(h, k_star)
        gradient = (1 - shifted_h - h) * (2 / n_points)
        
        # Gradient descent step
        h_new = h - learning_rate * gradient
        
        # Repair step
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            if h_sum > target_sum:
                # Reduce largest elements
                sorted_indices = np.argsort(h_new)[::-1]
                excess = h_sum - target_sum
                for idx in sorted_indices:
                    if excess <= 0:
                        break
                    reduce_amount = min(excess, h_new[idx])
                    h_new[idx] -= reduce_amount
                    excess -= reduce_amount
            else:
                # Increase smallest elements
                sorted_indices = np.argsort(h_new)
                deficit = target_sum - h_sum
                for idx in sorted_indices:
                    if deficit <= 0:
                        break
                    increase_amount = min(deficit, 1 - h_new[idx])
                    h_new[idx] += increase_amount
                    deficit -= increase_amount
        
        # Rescale
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update binary variables based on new continuous values
        z_new = (h_new > 0.5).astype(int)
        
        # Apply penalty and update
        penalty = mu * np.sum(np.abs(h_new - z_new))
        # Increment mu
        mu *= mu_factor
        
        h = h_new
        z = z_new
        
        # Random perturbation occasionally
        if iteration % 100 == 0 and np.random.random() < perturb_fraction:
            perturb = np.random.normal(0, 0.01, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
