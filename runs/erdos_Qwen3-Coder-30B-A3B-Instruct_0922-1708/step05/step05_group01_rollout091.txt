import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 0.001
    mu = 0.1
    mu_factor = 1.2
    perturbation_fraction = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        y = h_init * (n_points / 2) / np.sum(h_init)
    else:
        y = np.ones(n_points) * 0.5
    
    # Binary decisions
    z = (y > 0.5).astype(float)
    
    # Compute initial C5
    corr = correlate(y, 1 - y, mode='full')
    c5 = np.max(corr) * dx
    
    best_y = y.copy()
    best_c5 = c5
    
    # Iteration loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the maximum lag
        corr = correlate(y, 1 - y, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros_like(y)
        shift = max_lag
        if shift >= 0:
            grad[shift:] = -(1 - y[:-shift] - y[shift:])
        else:
            grad[:shift] = -(1 - y[-shift:] - y[:shift])
        grad = 2 * grad * dx / n_points
        
        # Apply gradient step
        y = y - learning_rate * grad
        
        # Update binary decisions
        z = (y > 0.5).astype(float)
        
        # Repair feasibility (clip and rescale)
        y = np.clip(y, 0, 1)
        y_sum = np.sum(y)
        if abs(y_sum - n_points/2) > 1e-8:
            y = y * (n_points / 2) / y_sum
            
        # Apply penalty
        penalty = mu * np.sum(np.abs(y - z))
        
        # Update current C5
        corr = correlate(y, 1 - y, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_y = y.copy()
        
        # Update penalty parameter
        if iteration % 100 == 0:
            mu *= mu_factor
        
        # Random perturbation occasionally
        if iteration % 500 == 0 and iteration > 0:
            perturb = np.random.rand(n_points) * perturbation_fraction
            y = np.clip(y + perturb - np.mean(perturb), 0, 1)
            y = y * (n_points / 2) / np.sum(y)
            
    return (best_y.tolist(), best_c5, n_points)
