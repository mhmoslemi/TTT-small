import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the C5 bound for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h onto feasible region: clip to [0,1] and rescale to integral=1"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover with parameter alpha"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.1, noise_magnitude=0.03):
    """Apply Gaussian noise mutation"""
    mask = np.random.random(len(individual)) < mutation_rate
    noise = np.random.normal(0, noise_magnitude, len(individual))
    individual = individual + mask * noise
    return individual

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine solution using gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        # Compute gradient numerically
        grad = np.zeros_like(h)
        eps = 1e-5
        base_fitness = compute_fitness(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            fitness_plus = compute_fitness(h_plus, dx)
            grad[i] = (fitness_plus - base_fitness) / eps
        
        # Update with gradient
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        
        # Recompute to update base fitness
        base_fitness = compute_fitness(h, dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        if len(h_start) != n_points:
            x_orig = np.linspace(0, 2, len(h_start))
            x_target = np.linspace(0, 2, n_points)
            h_start = np.interp(x_target, x_orig, h_start)
        h_start = h_start * (n_points / 2) / np.sum(h_start)
    else:
        # Uniform start
        h_start = np.ones(n_points) * 0.5

    # GA parameters
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    refinement_interval = 50
    
    # Initialize population
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            # Perturb parent solution
            h = h_start + np.random.normal(0, 0.05, n_points)
        else:
            # Random starting solution
            h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
        population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(ind, dx) for ind in population]
    
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    start_time = time.time()
    
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Selection, crossover, mutation
        new_population = []
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.random() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Evaluate new population
        fitnesses = [compute_fitness(ind, dx) for ind in population]
        
        # Update best solution
        current_best_fitness = min(fitnesses)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[np.argmin(fitnesses)].copy()
        
        # Periodic gradient refinement
        if generation % refinement_interval == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for idx in top_indices:
                refined_individual = gradient_refinement(
                    population[idx], dx, max_iter=30, lr=0.001
                )
                refined_fitness = compute_fitness(refined_individual, dx)
                if refined_fitness < fitnesses[idx]:
                    population[idx] = refined_individual
                    fitnesses[idx] = refined_fitness
                    if refined_fitness < best_fitness:
                        best_fitness = refined_fitness
                        best_individual = refined_individual.copy()
    
    # Final gradient refinement on best solution
    best_individual = gradient_refinement(best_individual, dx, max_iter=100, lr=0.0005)
    final_fitness = compute_fitness(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
