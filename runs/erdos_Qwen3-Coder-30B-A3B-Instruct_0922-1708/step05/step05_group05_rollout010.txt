import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturb_freq = 50
    perturb_mag = 0.1
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1-h, mode='full')) * dx
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation and find minimal overlap
        corr = correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Find index of minimum correlation value 
        min_idx = np.argmin(corr)
        min_corr = corr[min_idx]
        
        # Compute gradient
        grad = np.zeros(n_points)
        for j in range(n_points):
            # Forward difference
            if j + min_idx - (n_points - 1) >= 0:
                grad[j] += 1.0
            # Backward difference  
            if j - (min_idx - (n_points - 1)) >= 0:
                grad[j] += 1.0
                
        # Take gradient step - for maximization of minimum
        h = h + learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Normalize to the constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasional perturbations to escape local minima
        if i % perturb_freq == 0:
            noise = np.random.normal(0, perturb_mag * (1.0 / (1.0 + i/100)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_corr = correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
