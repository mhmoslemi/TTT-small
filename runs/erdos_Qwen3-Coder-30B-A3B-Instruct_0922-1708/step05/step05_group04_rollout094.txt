import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization with correct sum
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Sample shifts for constraints
    k_samples = np.linspace(-2, 2, 50)
    
    # Function to compute C5 from h
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def repair_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_vals = h_vals * (n_points / 2) / h_sum
        return h_vals
    
    # Optimize using gradient descent on the overlap integral
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradient descent for refinement
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient numerically
        eps = 1e-5
        grad = np.zeros(n_points)
        base_C5 = compute_C5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            
            C5_plus = compute_C5(h_plus)
            grad[i] = (C5_plus - base_C5) / eps
        
        # Update h
        h = h - 0.001 * grad
        h = repair_h(h)
        
        # Update best if improved
        current_C5 = compute_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    # Final verification
    best_h = repair_h(best_h)
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
