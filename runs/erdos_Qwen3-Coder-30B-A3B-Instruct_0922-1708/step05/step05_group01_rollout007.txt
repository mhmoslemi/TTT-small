import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 1e-3
    line_search_factor = 0.5
    max_iter = 5000
    grad_threshold = 1e-6
    perturbation_fraction = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the lag k* corresponding to maximum correlation
        k_star = np.argmax(corr_full)
        max_corr = corr_full[k_star]
        
        # Compute gradient
        # For correlation at lag k*, we need to compute contributions
        # from overlapping positions
        gradient = np.zeros(n_points)
        
        # Determine the actual lag (negative for right shift, positive for left shift)
        lag = k_star - (n_points - 1)
        
        # We'll compute the gradient for the specific max lag
        # The gradient at position i for a correlation with lag is:
        #    d/dh_i [ sum_j h[j] * (1 - h[j + lag]) ] = 1 - h[j+lag] if j = i, 
        #                                               -h[i] if j + lag = i 
        # But need to handle the constraint properly
        
        # Direct analytical gradient computation approach
        # For each point i, we compute the influence of changing h[i] on max correlation
        
        for i in range(n_points):
            # At position i, how h[i] affects the correlations
            # We only consider those positions where h[i] contributes to correlation
            # with h[i + lag]: (i + lag) must be within bounds and in [0, n_points)
            if i + lag >= 0 and i + lag < n_points:
                # Overlap contribution: h[i] * (1 - h[i + lag])
                # Gradients:
                #    ∂/∂h[i] = (1 - h[i + lag])
                #    ∂/∂h[i + lag] = -h[i]
                if i + lag < n_points:
                    # Marginally increase h[i], reduces the correlation if the term is active
                    if lag >= 0:
                        gradient[i] = (1 - h[i + lag])
                    else:
                        gradient[i] = (1 - h[i + lag])  # add correct term from both segments
                        
        # Simplification: most significant e.g. by easy direct derivative
        temp_h = h.copy()
        
        # Create appropriate correlation corresponding to the max-valued lag
        # Correlation convolution: with h and (1 - h) that gives max
        if k_star >= n_points - 1:
            forward_lag = k_star - (n_points - 1)
            # This is the maximum at forward_lag (positive lag)
            # h[j] * (1 - h[j + forward_lag]) over overlapping regions
            # Gradient contribution of i-th element:
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = i + forward_lag
                if j >= 0 and j < n_points:
                    gradient[i] = 1 - h[j]
        else:
            backward_lag = -(n_points - 1 - k_star)
            # backward lag case
            gradient = np.zeros(n_points)
            for i in range(n_points):
                j = i + backward_lag
                if j >= 0 and j < n_points:
                    gradient[i] = 1 - h[j]
        
        # Simple gradient descent step
        new_h = h - alpha * gradient
        
        # Project new_h onto feasible region
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to enforce sum(h) = n_points / 2 exactly
        h_sum = np.sum(new_h)
        target_sum = n_points / 2
        new_h = new_h * target_sum / h_sum
        
        h = new_h
        
        # Line search for adaptive step size
        temp_c5 = float('inf')

            corr_temp = correlate(h, 1 - h, mode='full')
            temp_c5 = np.max(corr_temp) * dx
            if temp_c5 < c5:
                break
            alpha *= line_search_factor
            new_h = h - alpha * gradient
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * target_sum / np.sum(new_h)
            h = new_h
            
        if abs(temp_c5 - c5) < 1e-8:
            alpha = alpha * 0.9  # Slow down if no progress
            
        # Random perturbation occasional
        if iteration % 100 == 0:
            noise = np.random.normal(scale=perturbation_fraction, size=n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * target_sum / np.sum(h)
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
