import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    initial_alpha = 0.01
    min_alpha = 1e-6
    max_alpha = 1.0
    alpha_decay = 0.5
    alpha_growth = 1.2
    perturb_freq = 50
    perturb_magnitude = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block-like
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points/2
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    
    # Evaluate initial solution
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    alpha = initial_alpha
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Re-normalize to sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update step
        new_h = h - alpha * grad
        
        # Project onto feasible region
        # First, ensure bounds 0 <= h <= 1
        new_h = np.clip(new_h, 0, 1)
        # Then, project to sum = n_points/2
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Check improvement
        new_c5 = evaluate(new_h)
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            # Increase step size if successful
            alpha = min(alpha * alpha_growth, max_alpha)
        else:
            # Decrease step size if failed
            alpha = max(alpha * alpha_decay, min_alpha)
        
        # Periodic perturbations to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturb_magnitude, n_points)
            # Ensure it stays within [0, 1] and sum constraint
            new_h = h + noise
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            new_c5 = evaluate(new_h)
            if new_c5 < best_c5:
                h = new_h.copy()
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
