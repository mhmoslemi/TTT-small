import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 10.0
    cooling_rate = 1.001
    start_time = time.time()
    
    # Local gradient descent parameters
    lr = 0.001
    max_grad_steps = 50
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Annealing temperature
        T = T0 / np.log(1 + iteration)
        
        # Generate candidate by perturbation
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate's C5
        corr_cand = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr_cand) * dx
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
    
    # Refinement with gradient descent
    h_ref = best_h.copy()
    
    for _ in range(max_grad_steps):
        if time.time() - start_time > budget_s - 15:
            break
            
        grad = np.zeros_like(h_ref)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_ref.copy()
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
