import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Simple initialization
        h_init = np.ones(n_points) * 0.5
    
    # Fourier representation: real coefficients a_k, b_k
    # Only optimize half the coefficients for real signal (symmetry)
    # Represent h as sum of cosines and sines
    # But better approach: we'll optimize the time domain signal directly
    
    # Strategy: optimize by projecting to Fourier space, then back
    def h_to_fourier(h):
        # Compute discrete fourier transform
        H = fft(h)
        return H.real, H.imag
    
    def fourier_to_h(a, b):
        # Reconstruct h from real and imaginary parts of FFT
        # Assume real signal
        H = a + 1j * b
        h = ifft(H).real
        return h
    
    # Map h vector to Fourier coefficients
    def parametrize_h(h):
        # Return a list of Fourier coefficients for a real time signal
        # FFT returns n complex numbers
        H = fft(h)
        # Keep only the first n_fouerier coefficients (for optimization)
        return np.concatenate([H.real[:n_fourier], H.imag[:n_fourier]])
    
    # Project back to h vector
    def project_h(params):
        # params are real + imaginary parts
        a = params[:n_fourier]
        b = params[n_fourier:2*n_fourier]
        
        # Construct complex FFT coefficients but ensure real signal
        # For real signal, FFT[0] is real; FFT[n-1] is conjugate of FFT[1]
        # We may need to handle symmetry carefully
        H = np.zeros(n_points, dtype=complex)
        H[:n_fourier] = a + 1j * b
        # For even n, the last terms depend on symmetry
        # Simple approach: Mirror to satisfy real signal property
        if n_points % 2 == 0:
            H[n_points//2] = a[n_points//2]
        # Symmetric case for fft
        for i in range(1, n_points//2):
            H[n_points-i] = np.conj(H[i])
        
        # Now inverse FFT
        h = ifft(H).real
        # Normalize to fix DC term and scale appropriately
        return h
    
    # Repair to valid h
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            # Scale to exact sum
            h = h * (n_points / 2) / h_sum
        return h
    
    # Evaluate C5
    def compute_c5(h):
        # Use numpy.correlate for overlap
        cor = np.correlate(h, 1-h, mode='full')
        return np.max(cor) * dx
    
    # Objective for L-BFGS
    def objective(params):
        h = project_h(params)
        h_repaired = repair_h(h)
        c5 = compute_c5(h_repaired)
        return c5
    
    # Gradient computation via finite differences
    def grad_objective(params, epsilon=1e-4):
        grad = np.zeros_like(params)
        base_c5 = objective(params)
        
        for i in range(len(params)):
            params_plus = params.copy()
            params_plus[i] += epsilon
            c5_plus = objective(params_plus)
            grad[i] = (c5_plus - base_c5) / epsilon
        return grad
    
    # Initialize from h_init or use random
    h_current = h_init.copy()
    h_current = repair_h(h_current)
    
    # Convert to fourier parameters
    init_params = parametrize_h(h_current)
    
    # Now do optimization using L-BFGS
    # Only optimize up to n_fourier terms
    # Use minimize with L-BFGS-B for decent speed and convergence properties
    max_iter = 500
    stagnation_count = 0
    best_c5 = float('inf')
    best_params = init_params.copy()
    
    current_params = init_params.copy()
    current_c5 = compute_c5(repair_h(project_h(current_params)))
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_params = current_params.copy()
    
    for iter_num in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        try:
            # Compute gradient
            g = grad_objective(current_params)
            # Compute update
            step_size = 0.01
            new_params = current_params - step_size * g
            
            # Project to new h
            h_temp = project_h(new_params)
            h_repaired = repair_h(h_temp)
            new_c5 = compute_c5(h_repaired)
            
            if new_c5 < current_c5:
                current_params = new_params.copy()
                current_c5 = new_c5
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_params = current_params.copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
            else:
                stagnation_count += 1
            
            if stagnation_count > 100:
                # Restart with new random parameters based on current solution
                noise_amp = 0.01
                current_params = best_params + np.random.normal(0, noise_amp, len(best_params))
                stagnation_count = 0
                
        except Exception as e:
            # Handle any issue during optimization steps
            break
    
    # Final repair and validation
    final_h = project_h(best_params)
    final_h = repair_h(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
