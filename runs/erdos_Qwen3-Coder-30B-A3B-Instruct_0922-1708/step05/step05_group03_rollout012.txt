import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize with parent if available, otherwise use a uniform distribution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-scale approach with convex relaxation
    scales = [10, 20, 40, 80, 100]
    current_h = h_init.copy()
    
    for scale in scales:
        if time.time() - start_time > budget_s - 50:
            break
            
        n_current = scale
        dx_current = 2.0 / n_current
        
        # Resize current solution to new resolution (linear interpolation)
        x_old = np.linspace(0, 2, len(current_h))
        x_new = np.linspace(0, 2, n_current)
        current_h = np.interp(x_new, x_old, current_h)
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_current / 2) / np.sum(current_h)
        
        # Define convex minimization problem
        h_var = Variable(n_current)
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_current / 2]
        
        # Simplified convex approximation (to speed up iterations)
        # Approximate the max by average over lags
        weights = np.zeros(n_current)
        for k in range(1, n_current//4 + 1):
            # Correlation between h and (1-h) over lag k
            weights[0] += 1
            weights[-k] += 1
        weights = weights / np.sum(weights)
        
        # Minimize weighted average of constraints over possible lags
        # Use a proxy objective: sum over lags with weights
        # Actually construct C5 through correlation
        def compute_overlap(h_vec):
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            max_corr = np.max(corr)
            return max_corr * dx_current
        
        # Simple gradient-like approach with projection
        h_best = current_h.copy()
        c5_best = compute_overlap(h_best)
        
        # Gradient descent with few steps
        steps = min(100, max(10, 1000 - len(scales)*100))
        for _ in range(steps):
            if time.time() - start_time > budget_s - 50:
                break
                
            # Compute approximate gradient using finite differences
            grad = np.zeros(n_current)
            eps = 1e-4
            
            for i in range(n_current):
                h_plus = h_best.copy()
                h_plus[i] += eps
                
                if h_plus[i] > 1:
                    h_plus[i] = 1.0
                elif h_plus[i] < 0:
                    h_plus[i] = 0.0
                    
                # Adjust for sum constraint
                sum_h = np.sum(h_plus)
                h_plus = h_plus * (n_current / 2) / sum_h
                
                c5_plus = compute_overlap(h_plus)
                
                grad[i] = (c5_plus - c5_best) / eps
            
            # Update
            h_new = h_best - 0.001 * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_current / 2) / np.sum(h_new)
            
            # Compute new overlap
            c5_new = compute_overlap(h_new)
            
            if c5_new < c5_best:
                h_best = h_new.copy()
                c5_best = c5_new
                
        # Use refined solution if better
        if c5_best < best_c5:
            best_c5 = c5_best
            best_h = h_best.copy()
            
        current_h = h_best.copy()
    
    # Local refinement with a few more steps
    h_refined = best_h.copy()
    
    for step in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        h_refined = h_refined - 0.0005 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refined.copy()
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
