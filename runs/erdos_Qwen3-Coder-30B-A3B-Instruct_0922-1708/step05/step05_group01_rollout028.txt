import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturbation_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Current best
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    lr = initial_learning_rate
    last_improvement = 0
    current_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Stop if no improvement for a while
        if iteration - last_improvement > plateau_patience:
            if lr > min_learning_rate:
                lr *= learning_rate_decay
                # Add small random perturbation
                noise = np.random.normal(0, perturbation_fraction * lr, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                break
        
        # Find lag with maximum correlation
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # Construct gradient
        gradient = np.zeros(n_points)
        
        # The gradient for the maximum overlap depends on which portion of the convolution
        # is involved. For two sequences A and B, the gradient of their convolution w.r.t. A is B_rev
        # Here, we differentiate h * (1 - h) over h, so we compute:
        # d/dh[ sum_i h[i] * (1 - h[i + max_lag]) ] = 1 - 2*h[i] - (1-h[i+max_lag])*(delta_{i,i+max_lag})
        # So gradient at i = 1 - 2*h[i] for the main term
        # and possibly additional terms from the convolution
        
        # For exact gradient computation, we compute the direction of steepest ascent
        # of the overlap for lag max_lag
        
        # Recover the relevant segments
        if max_lag >= 0:
            h_left = h[:n_points - max_lag]
            h_right = h[max_lag:]
            gradient[:n_points - max_lag] += (1 - h_right)
            gradient[max_lag:] -= h_left
        else:
            h_left = h[-max_lag:]
            h_right = h[:n_points + max_lag]
            gradient[-max_lag:] += (1 - h_right)
            gradient[:n_points + max_lag] -= h_left
        
        # Apply learning rate
        h_new = h - lr * gradient
        
        # Repair
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            # Rescale to target sum
            h_new = h_new * target_sum / h_sum
            
        h = h_new
        
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
