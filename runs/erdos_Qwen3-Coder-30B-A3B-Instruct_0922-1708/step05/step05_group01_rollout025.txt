import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    lr = 1e-3
    lr_decay = 0.999
    min_lr = 1e-6
    plateau_patience = 200
    perturb_frac = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    last_improvement = 0
    current_lr = lr
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find maximum lag
        max_lag_idx = np.argmax(corr)
        max_val = corr[max_lag_idx]
        
        # Compute gradient w.r.t. h for the maximum lag
        # This is the gradient of (h * (1-h))[max_lag_idx]
        # For convolution, we compute the derivative of max term
        
        # Get the indices for the correlation overlap window
        # lag k corresponds to correlation at index k
        # for array size n, max_lag_idx from 0 to 2n-2
        lag = max_lag_idx - (n_points - 1)
        
        # Initialize gradient
        grad = np.zeros(n_points)
        
        # Construct gradient contribution from the correlation term
        # We compute ∂/∂h_k [∑_l h[l] (1-h[l+lag])] 
        # = (1 - h[l]) - h[l] = 1 - 2*h[l] if l is occupied
        
        # For shorthand, compute manually the gradient for max_lag_idx
        # Since correlation is [0:(2n-1)], we map back to lags:
        # lag_k = k - (n-1)
        # So we have convolution of h against (1-h) with lag.
        
        # The gradient is a sum of delta functions indicating overlap region
        # Let's do this more directly.
        
        # Create mask for where the convolution contributes to maximum
        # That is: for h[i] * (1-h[i+lag])
        # The overlap contribution is immediate
        if lag >= 0:
            # Base h segment (size n-lag) 
            # (1-h) segment starts at lag (size n-lag)
            for i in range(n_points - lag):
                grad[i] += 1 - 2 * h[i + lag]
        else:
            # Lag is negative, so h shift is to the left
            # (1-h) starts at position -lag where h shifts left
            lag_abs = -lag
            for i in range(n_points - lag_abs):
                grad[i + lag_abs] += 1 - 2 * h[i]
        
        # Update h using gradient descent
        h = h - current_lr * grad
        
        # Clip to [0,1] range
        h = np.clip(h, 0, 1)
        
        # Rescale to fixed sum = n/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate again
        corr = correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iter_num
            current_lr = lr  # reset learning rate on improvement
        else:
            # Check for plateau
            if iter_num - last_improvement > plateau_patience:
                current_lr *= lr_decay
                if current_lr < min_lr:
                    # Perturb slightly
                    perturb = np.random.normal(0, perturb_frac, n_points)
                    h = np.clip(h + perturb, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    # Reset patience
                    last_improvement = iter_num
                else:
                    # Skip too small learning rates
                    pass
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr) * dx
    
    return (best_h.tolist(), c5_final, n_points)
