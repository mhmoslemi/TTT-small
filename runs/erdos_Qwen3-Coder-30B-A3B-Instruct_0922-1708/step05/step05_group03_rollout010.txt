import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize based on parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Simple initial uniform construction
        h = np.full(n_points, 1.0)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define scale sequence
    scales = [10, 20, 40, 80, 100]
    
    start_time = time.time()
    
    # Multi-scale optimization
    for i_scale, n in enumerate(scales):
        if time.time() - start_time > budget_s - 100:
            break
            
        # Upsample previous solution if available
        if i_scale == 0:
            h_scaled = np.full(n, 1.0)
            h_scaled = h_scaled * (n / 2) / np.sum(h_scaled)
        else:
            # Interpolate to new resolution
            x_old = np.linspace(0, 2, len(h_prev))
            x_new = np.linspace(0, 2, n)
            h_scaled = np.interp(x_new, x_old, h_prev)
            h_scaled = h_scaled * (n / 2) / np.sum(h_scaled)
            
        # Try convex relaxation at this resolution
        try:
            # Define optimization variables
            h_var = cp.Variable(n)
            
            # Define objective with relaxed constraints
            # Use average of relevant lags for convexity
            lag_range = min(20, n//4)
            agg_corr = 0
            for k in range(1, lag_range+1):
                corr = cp.sum(cp.multiply(h_var[:n-k], 1 - h_var[k:]))
                agg_corr += corr
            avg_corr = agg_corr / (2*lag_range - 1) if 2*lag_range - 1 > 0 else 0
            
            # Add regularization to maintain sparsity
            obj = cp.Minimize(avg_corr - 0.05 * cp.sum(h_var) + 0.05 * cp.sum(cp.multiply(h_var, h_var)))
            
            # Define constraints
            cons = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n/2
            ]
            
            # Solve using ECOS
            prob = cp.Problem(obj, cons)
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=500)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_opt = h_var.value
                h_scaled = np.clip(h_opt, 0, 1)
                h_scaled = h_scaled * (n / 2) / np.sum(h_scaled)
        except:
            pass
        
        # Local refinement with gradient descent
        if n < 100 or i_scale == len(scales)-1:
            h_refined = h_scaled.copy()
            lr = 0.001
            eps = 1e-4
            
            for _ in range(30):
                if time.time() - start_time > budget_s - 50:
                    break
                grad = np.zeros_like(h_refined)
                
                for j in range(n):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * (2.0 / n)
                    
                    grad[j] = (c5_plus - best_c5) / eps
                    
                h_refined = h_refined - lr * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n / 2) / np.sum(h_refined)
                
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5 = np.max(corr) * (2.0 / n)
                
                if c5 < best_c5:
                    best_c5 = c5
                    if n == 100:
                        best_h = h_refined.copy()
            
            # Also bisect the current solution to get better
            if i_scale == len(scales)-1:
                h_bisected = h_scaled.copy()
                # Sample at 10 points then interpolate
                h_bisected = h_bisected[::10] if len(h_bisected) > 10 else h_bisected
                # Interpolate back to n_points
                h_bisected_full = np.interp(
                    np.linspace(0, 2, n_points),
                    np.linspace(0, 2, len(h_bisected)),
                    h_bisected
                )
                h_bisected_full = h_bisected_full * (n_points / 2) / np.sum(h_bisected_full)
                corr_refined = np.correlate(h_bisected_full, 1 - h_bisected_full, mode='full')
                c5_refined = np.max(corr_refined) * dx
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_bisected_full.copy()
        
        h_prev = h_scaled.copy()
    
    # Final refinement
    if time.time() - start_time < budget_s - 50:
        h_final = best_h.copy()
        lr = 0.0005
        eps = 1e-4
        
        for _ in range(30):
            if time.time() - start_time > budget_s - 10:
                break
            grad = np.zeros_like(h_final)
            
            for j in range(n_points):
                h_plus = h_final.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
                
            h_final = h_final - lr * grad
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            corr = np.correlate(h_final, 1 - h_final, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_final.copy()
                
    # Final check with full evaluation
    if time.time() - start_time < budget_s - 10:
        corr_final = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr_final) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
    
    # Ensure constraints
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
