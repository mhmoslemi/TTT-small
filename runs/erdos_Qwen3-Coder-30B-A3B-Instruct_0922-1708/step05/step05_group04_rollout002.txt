import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 100
        h_init = np.random.uniform(0, 1, n_points)
    
    # Ensure correct sum
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 1000
    swarm_size = 100
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    rejuvenation_rate = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        for _ in range(swarm_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        for _ in range(swarm_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * swarm_size
    gbest_idx = 0
    gbest_fitness = float('inf')
    
    # Compute initial fitness
    for i in range(swarm_size):
        corr = np.correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        if fitness < gbest_fitness:
            gbest_fitness = fitness
            gbest_idx = i
    
    gbest_particle = particles[gbest_idx].copy()
    
    # Repair function to keep h in feasible region
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # PSO iteration
    current_iter = 0
    stagnation_count = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 5:
        for p_idx in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[p_idx] = (inertia_weight * velocities[p_idx] + 
                                phi_p * r1 * (pbest_particles[p_idx] - particles[p_idx]) +
                                phi_g * r2 * (gbest_particle - particles[p_idx]))
            
            # Update position
            particles[p_idx] += velocities[p_idx]
            particles[p_idx] = repair(particles[p_idx])
            
            # Evaluate
            corr = np.correlate(particles[p_idx], 1 - particles[p_idx], mode='full')
            fitness = np.max(corr) * dx
            
            # Update personal best
            if fitness < pbest_fitness[p_idx]:
                pbest_fitness[p_idx] = fitness
                pbest_particles[p_idx] = particles[p_idx].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[p_idx].copy()
                    stagnation_count = 0
        
        # Stagnation mechanism
        stagnation_count += 1
        if stagnation_count > 50:
            for _ in range(int(swarm_size * rejuvenation_rate)):
                idx = np.random.randint(0, swarm_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.05, 0.05, n_points)
            stagnation_count = 0
        
        current_iter += 1
    
    # Local optimization refinement
    h_ref = gbest_particle.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += 1e-6
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr)) / 1e-6
        return -grad * dx
    
    # L-BFGS-B local search
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    # Try several local searches with different start points
    best_h = h_ref.copy()
    best_c5 = gbest_fitness
    
    for _ in range(5):
        if time.time() - start_time > budget_s - 5:
            break
        # Create a random perturbation
        h_pert = h_ref.copy()
        for i in range(n_points):
            if np.random.rand() < 0.1:
                h_pert[i] += np.random.normal(0, 0.01)
        h_pert = repair(h_pert)
        
        # Use L-BFGS-B
        res = minimize(objective, h_pert, method='L-BFGS-B', bounds=bounds, 
                       jac=gradient, options={'maxiter': 100, 'gtol': 1e-6})
        
        if res.success:
            h_opt = repair(res.x)
        else:
            h_opt = h_pert
            
        corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_final = np.max(corr_final) * dx
        if c5_final < best_c5:
            best_c5 = c5_final
            best_h = h_opt.copy()
    
    return (best_h.tolist(), best_c5, n_points)
