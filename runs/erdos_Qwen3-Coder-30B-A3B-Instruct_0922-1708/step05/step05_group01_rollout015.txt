import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    lr = 0.001
    mu_init = 0.1
    mu_factor = 1.2
    perturb_fraction = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        y = h_init * (n_points / 2) / np.sum(h_init)
    else:
        y = np.ones(n_points) * 0.5
    
    z = (y > 0.5).astype(int)
    mu = mu_init
    
    # Track best solution
    best_y = y.copy()
    corr = correlate(y, 1 - y, mode='full')
    best_c5 = np.max(corr) * dx
    best_iter = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(y, 1 - y, mode='full')
        c5 = np.max(corr) * dx
        k_max = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_y = y.copy()
            best_iter = iteration
        
        # Compute gradient for max lag
        grad = np.zeros(n_points)
        # For the specific lag k_max
        lag = k_max - (n_points - 1)
        if lag >= 0:
            # Forward lag
            y_lagged = np.concatenate([y[lag:], np.zeros(lag)])
        else:
            # Backward lag
            lag = -lag
            y_lagged = np.concatenate([np.zeros(lag), y[:-lag]])
        
        # Gradient is roughly (1 - y_shifted - y) for the overlap
        # but more accurately we just take it as the derivative of overlap integral
        # For an overlap at lag k, (1-h(x))(h(x+k)) -> gradient wil be (1-h(x+k)) - h(x)
        # But we need to compute d/dh_i of the max of overlaps
        # Approximate using one specific max lag derivative:
        # The overlap at k = 0,1,2,... is integral h(x)(1 - h(x+k))
        # At k = k_max:
        # The contributions to overlap sum come from y[i] * (1 - y[i+k])
        # So gradient for h_i = y[i] is -1 * (1 - y[i+k]) + 0 + 0 + ...
        # So gradient_i = - (1 - y[i+k])
        # We're looking at max of this, so we should just compute gradients at the edges
        # Due to computational difficulty, we will compute an actual numerical derivative
        # Better approach: compute gradient at k_max lag
        forward = np.roll(y, -k_max + (n_points - 1))
        backward = np.roll(y, k_max - (n_points - 1))
        gradient = -forward - backward + 1
        
        # Apply gradient descent update
        y = y - lr * gradient
        
        # Update z
        z = (y > 0.5).astype(int)
        
        # Repair step (ensuring y in [0,1] and sum = n_points/2)
        y = np.clip(y, 0, 1)
        y_sum = np.sum(y)
        if abs(y_sum - n_points/2) > 1e-6:
            # Adjust total sum
            adjustment = (n_points/2 - y_sum) / n_points
            y = y + adjustment
            
        # Rescale
        y = y * (n_points / 2) / np.sum(y)
        
        # Apply penalty relaxation
        mu = mu * mu_factor
        
        # Random perturbation occasionally
        if np.random.rand() < perturb_fraction:
            perturb = np.random.normal(0, 0.01, n_points)
            y = y + perturb
            y = np.clip(y, 0, 1)
            y = y * (n_points / 2) / np.sum(y)
    
    # Final refinement
    corr = correlate(best_y, 1 - best_y, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_y.tolist(), final_c5, n_points)
