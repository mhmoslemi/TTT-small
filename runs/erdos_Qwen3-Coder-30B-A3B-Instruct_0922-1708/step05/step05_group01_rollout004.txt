import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturbation_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    current_lr = initial_learning_rate
    last_improvement = 0
    prev_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Check for plateau
        if abs(c5 - prev_c5) < 1e-10:
            if iteration - last_improvement > plateau_patience:
                # Reduce learning rate
                current_lr *= learning_rate_decay
                current_lr = max(current_lr, min_learning_rate)
                if current_lr <= min_learning_rate:
                    # Add small random perturbation
                    perturbation = np.random.normal(0, perturbation_fraction * 0.1, n_points)
                    h += perturbation
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        else:
            prev_c5 = c5
            
        # Find the index of maximum correlation
        max_idx = np.argmax(corr)
        # The maximum lag will be halfway through the correlation result
        max_lag = max_idx - (len(corr) - 1) // 2
        
        # Construct the gradient
        gradient = np.zeros(n_points)
        # The gradient of the overlap with respect to h for this lag is:
        # It is a convolution (1-h) with an indicator function
        # but since we don't want to convolve twice, we compute it directly:
        # For a specific lag k, this is the derivative of (h ⋆ (1-h))[k],
        # with respect to h[i]. 
        # The derivative of (h[i] * (1-h[i+k])) with respect to h[i] is (1-h[i+k]) - h[i]
        # But more carefully:
        # (h[i] * (1-h[i+k])) - (h[i+k] * (1-h[i]))
        # Reconstructed properly:
        # Given correlation (h ⋆ (1-h)) [k], for the term h[i] in the original,
        # the contribution is h[i] * (1-h[i+k]).
        # The partial derivative is (1-h[i+k]) for the first term  
        # But we also have h[i+k] * (1-h[i]) from the second term.
        # So the gradient of the kth term is:
        # For positions i to which the correlation contributes:
        # 
        # (1 - h[i+k]) + (1 - h[i]) where we change the sign based on whether it is positive or not
        # Actually, let's recall the correct way:
        # (h ⋆ (1-h))_k = sum_{i} h[i] * (1-h[i+k])
        # 
        # The derivative w.r.t. h[j] is:
        # sum_{i} d/dh[j] (h[i] * (1 - h[i+k])) 
        # if i == j: coefficient is (1 - h[j+k])
        # if i == j-k: coefficient is 1 - h[j]
        # else: 0
        # 
        # So derivative of overlap at lag k w.r.t. h[j] is:
        # if j+k < n: (1 - h[j+k])
        # if j-k >= 0: (1 - h[j-k])  
        # if both: sum of the two contributions.
        
        # This is bit messy to compute directly in full generality, let's just be direct:
        # We'll just compute the gradient as the main contribution to the result.
        # When correcting for contribution of interaction at lag k:
        # At each position i, (h[i] * (1 - h[i+k])) - (h[i+k] * (1-h[i]))
        # The gradient w.r.t. h[j] is:
        # For j = i: (1 - h[i+k])
        # For j = i+k: (1 - h[i])
        
        # We compute gradient for lag max_lag
        lag_pos = max_lag + (n_points - 1) // 2  # Locations 
        # But average-assuming a full convolution:
        # (h ⋆ (1-h))_k = sum_{i=0}^{n-1} h[i] * (1-h[i+k]) ...  
        # We assume symmetric support (we probably shouldn't).  
        # Since conv is computed with full mode, the middle is k = 0.
        # We have to be careful where this index points in h.
        # The index at which the convolution gets to measure the correlation.
        # The index in full mode = k + (n-1) // 2 = 0 + (n-1)//2 ... well this is wrong.
        # Full mode extends the input symmetrically ⇒ sizes increase.
        # Let's recompute with proper indices.
        # The positions h[0:n] in conjunction with 1-h[0:n] (for fixed k),
        # where one overlaps with the other from i to i+k.
        # Since we are storing the histogram values, we do it straightforwardly:
        
        # Reconstruct the convolution result to choose the maximum lag correctly
        # For overlap lag k, the term is ∑ h[i] * (1 - h[i+k])
        # ∂C₅/∂h[j] = (sum over i of d/dh[j] (h[i] * (1 - h[i+k]))) = 
        # when i = j: (1 - h[j+k]) 
        # when i = j-k: (1 - h[j])
        # Both of these correspond to gradient contributions.
        # We compute the convolution gradient for maximum lag.
        if max_lag >= 0:
            # The rest of the contribution we get from indices from which to compute gradients.
            # Let's build it manually for correctness:
            for i in range(n_points):
                j = i + max_lag
                if j < n_points:
                    gradient[i] += (1 - h[j])
                j = i - max_lag
                if j >= 0:
                    gradient[i] += (1 - h[j])
        else:
            for i in range(n_points):
                j = i + max_lag
                if j >= 0:
                    gradient[i] += (1 - h[j])
                j = i - max_lag
                if j < n_points:
                    gradient[i] += (1 - h[j])
        
        # Apply gradient step
        h = h - current_lr * gradient
        
        # Repair step
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
