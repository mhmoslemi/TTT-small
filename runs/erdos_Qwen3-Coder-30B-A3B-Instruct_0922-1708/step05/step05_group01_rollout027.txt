import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Scale to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    lr = initial_learning_rate
    plateau_count = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
        
        # Compute gradient for this lag
        grad = np.zeros_like(h)
        
        # Construct backward shift for (1-h) part
        # Lag goes from 0 to 2*n_points - 2
        # We need to shift h and 1-h
        # For lag k, we have h[i] * (1-h[i+k]) for i = 0 to n-1-k
        # Gradient from h[i] and 1-h[i+k] terms
        if max_lag < n_points:
            # Forward lag
            k = max_lag
            # Contribution from h[i] * (1 - h[i+k])
            # Gradient of h[i] * (1 - h[i+k]) w.r.t h[i] is (1 - h[i+k])
            # Gradient of h[i] * (1 - h[i+k]) w.r.t h[i+k] is -h[i]
            # In the full correlation, we check where i+k is valid
            for i in range(n_points - k):
                grad[i] += (1 - h[i+k])
                if i+k < n_points:
                    grad[i+k] -= h[i]
        else:
            # Backward lag: need to reverse the logic
            k = max_lag - (n_points - 1)
            for i in range(n_points - k):
                grad[i+k] += (1 - h[i])
                if i < n_points:
                    grad[i] -= h[i+k]
        
        # Apply gradient step
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum - n_points/2) > 1e-8:
            h_new = h_new * (n_points / 2) / h_sum
        
        h = h_new
        
        # Evaluate
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            plateau_count = 0
        else:
            plateau_count += 1
            if plateau_count > plateau_patience:
                # Reduce learning rate
                lr = max(lr * learning_rate_decay, min_learning_rate)
                plateau_count = 0
                # Optional: perturb
                if np.random.rand() < 0.1:
                    perturb = np.random.normal(0, perturb_fraction, n_points)
                    h = np.clip(h + perturb, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
