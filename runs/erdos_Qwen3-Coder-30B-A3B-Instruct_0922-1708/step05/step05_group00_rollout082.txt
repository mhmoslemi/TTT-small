import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points

    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse to fine optimization approach
    n_coarse = min(50, n_points // 2)
    dx_coarse = 2.0 / n_coarse
    
    # Initialize coarse solution
    if 'initial_h_values' in globals() and len(initial_h_values) >= n_coarse:
        h_coarse = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_coarse))
        x_new = np.linspace(0, 2, n_coarse)
        h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_coarse) * 0.5
        
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1) 
    
    # Solve coarse problem using quadratic programming
    try:
        h_coarse_var = cp.Variable(n_coarse)
        C5_coarse = cp.Variable()
        
        constraints = [
            h_coarse_var >= 0,
            h_coarse_var <= 1,
            cp.sum(h_coarse_var) == n_coarse / 2,
            C5_coarse >= 0
        ]
        
        # For quadratic approach, try to approximate the correlation
        # This is a simplified convex program to explore regions of 
        # variable behavior that might be difficult to solve exactly
        try:
            prob = cp.Problem(cp.Minimize(C5_coarse), constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_coarse_solution = h_coarse_var.value
                if h_coarse_solution is not None:
                    h_coarse = np.array(h_coarse_solution)
                    h_coarse = np.clip(h_coarse, 0, 1)
                    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        except Exception:
            pass
    except Exception:
        pass
    
    # Refine solution to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), 
                       np.linspace(0, 2, n_coarse), h_coarse)
    h = h_fine.copy()
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Gradient-based refinement
    max_iter = 5000
    lr = 0.01
    patience = 0
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5

        # Compute finite difference gradients
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(corr_plus)
            
            grad[j] = (c5_plus - current_c5) / eps
        
        # Update with gradient
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        lr = max(1e-6, lr * 0.999)
        
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    # Ensure sum constraint exactly
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
