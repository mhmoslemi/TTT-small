import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with a good construction or random feasible solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    patience = 0
    max_patience = 100
    mutation_rate = 0.15
    
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Generate multiple mutations
        num_mutations = 100
        for _ in range(num_mutations):
            # Create a new candidate by perturbing the best solution
            candidate = best_h.copy()
            
            # Flip some elements
            num_flip = int(mutation_rate * n_points)
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            for idx in flip_indices:
                candidate[idx] = 1 - candidate[idx]
            
            # Repair feasibility
            candidate = repair_h(candidate, n_points)
            
            # Evaluate
            c5 = evaluate_h(candidate, dx)
            
            # Update best
            if c5 < best_c5:
                best_h = candidate.copy()
                best_c5 = c5
                patience = 0
            else:
                patience += 1
                
        # Reduce mutation rate if stuck
        if patience > 30:
            mutation_rate = min(0.5, mutation_rate * 1.1)
            patience = 0
            
        # Early termination if no improvement
        if patience > max_patience:
            break
    
    # Final gradient-based refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
