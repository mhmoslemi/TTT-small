import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial correlation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        
    # Optimization parameters
    max_iter = 6000
    lr = 0.01
    stagnation_limit = 200
    last_improvement = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 2:
            break
            
        # Compute current correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            if h_plus[i] < 0:
                h_plus[i] = 0
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update h
        h_new = h - lr * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new.copy()
        
        # Adjust learning rate
        if iteration % 200 == 0 and iteration > 0:
            lr *= 0.5
            
        # Early termination if no improvement for long
        if iteration - last_improvement > stagnation_limit:
            break
            
        # Random perturbation occasionally
        if iteration % 100 == 0 and iteration > 0:
            perturb_mask = np.random.random(n_points) < 0.01
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
