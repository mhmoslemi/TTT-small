import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points_list = [10, 20, 40, 80, 100]
    dx_list = [2.0 / n for n in n_points_list]
    
    best_c5 = float('inf')
    best_h = None
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (100 / 2) / np.sum(h_parent)
        corr_parent = np.correlate(h_parent, 1 - h_parent, mode='full')
        c5_parent = np.max(corr_parent) * dx_list[-1]
        best_c5 = min(best_c5, c5_parent)
        if c5_parent == best_c5:
            best_h = h_parent.copy()
    
    start_time = time.time()
    
    # Multi-scale optimization
    h_prev = None
    
    for scale_idx, (n_points, dx) in enumerate(zip(n_points_list, dx_list)):
        if time.time() - start_time > budget_s - 50:
            break
            
        # Initialize h at current scale
        if scale_idx == 0:
            h = np.ones(n_points) * 0.5
        else:
            # Upsample from previous scale
            h_prev_interp = np.interp(np.linspace(0, 2, n_points), 
                                    np.linspace(0, 2, len(h_prev)), h_prev)
            h = h_prev_interp
            
        # Convex relaxation: approximate with soft constraints
        # We'll use CVXPY to solve a relaxed version with log-sum penalties
        
        try:
            h_var = cp.Variable(n_points)
            constraints = []
            
            # Sum constraint
            constraints.append(cp.sum(h_var) == n_points / 2)
            
            # Bound constraints
            constraints.append(h_var >= 0)
            constraints.append(h_var <= 1)
            
            # Soft constraint on overlap terms
            # Sum over main lags
            lag_range = min(5, n_points//4)
            avg_overlap = 0
            overlap_terms = []
            
            for k in range(1, lag_range):
                left = h_var[:-k] if k > 0 else h_var
                right = h_var[k:] if k > 0 else h_var
                term = cp.sum(cp.multiply(left, 1 - right))
                overlap_terms.append(term)
                
            avg_overlap = cp.sum(overlap_terms) / len(overlap_terms)
            objective = cp.Minimize(avg_overlap)
            
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed = h_var.value
                h = h_relaxed
            else:
                # fallback to interpolation approach
                h = np.clip(h, 0, 1)
                
        except:
            # Fallback: just ensure bounds and normalization
            h = np.clip(h, 0, 1)
            
        # Finalize h for current scale to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Local refinement with gradient ascent via scipy
        def obj_func(h_flat):
            h_arr = h_flat.reshape(-1)
            corr = np.correlate(h_arr, 1 - h_arr, mode='full')
            return -np.max(corr) * dx
            
        def grad_func(h_flat):
            h_arr = h_flat.reshape(-1)
            grad = np.zeros_like(h_arr)
            eps = 1e-4
            for i in range(len(h_arr)):
                h_plus = h_arr.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                h_minus = h_arr.copy()
                h_minus[i] -= eps
                h_minus = np.clip(h_minus, 0, 1)
                h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
                c5_minus = np.max(corr_minus) * dx
                
                grad[i] = (c5_plus - c5_minus) / (2*eps)
                
            return -grad.flatten()
            
        # Perform local optimization if not too small
        if time.time() - start_time < budget_s - 50 and n_points > 10:
            # Refine with scipy minimize using L-BFGS-B
            h_flat = h.copy()
            res = minimize(obj_func, h_flat, method='L-BFGS-B', 
                          jac=grad_func, bounds=[(0,1)]*len(h_flat),
                          options={'maxiter': 50})
            h = res.x.reshape(-1)
            
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Test current solution
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        h_prev = h
        
    # Final verification with full correlation
    if best_h is not None:
        final_corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(final_corr) * dx_list[-1]
        if final_c5 < best_c5:
            best_c5 = final_c5
            
    if best_h is None:
        # Fallback to uniform as last resort
        h_uniform = np.ones(100) * 0.5
        h_uniform = h_uniform * (100 / 2) / np.sum(h_uniform)
        corr = np.correlate(h_uniform, 1 - h_uniform, mode='full')
        best_c5 = np.max(corr) * dx_list[-1]
        best_h = h_uniform
        
    return (best_h.tolist(), best_c5, 100)
