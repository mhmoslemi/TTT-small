import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_h = h_init.copy()
    else:
        current_h = np.random.uniform(0, 1, n_points)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
    
    best_h = current_h.copy()
    best_c5 = float('inf')
    
    # Function to compute C5
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Gradient approximation
    def estimate_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Estimate gradient and update
        grad = estimate_gradient(current_h)
        current_h -= 0.001 * grad
        current_h = repair(current_h)
        
        # Evaluate current solution
        c5 = compute_c5(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
        
        # Occasionally perturb to avoid local optima
        if i % 50 == 0:
            current_h = repair(current_h + np.random.normal(0, 0.01, n_points))
            
    # Final local optimization
    def objective(h):
        return compute_c5(h)
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Use scipy's minimize with L-BFGS-B for final refinement
    try:
        result = minimize(objective, best_h, method='L-BFGS-B', 
                         bounds=[(0, 1) for _ in range(n_points)],
                         options={'maxiter': 100, 'ftol': 1e-8})
        if result.success:
            refined_h = result.x
            refined_h = repair(refined_h)
            refined_c5 = compute_c5(refined_h)
            if refined_c5 < best_c5:
                best_h = refined_h
                best_c5 = refined_c5
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
