import numpy as np
import time

def evaluate_overlap(h):
    """Evaluate the overlap integral C5 for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum
    c5 = np.max(corr) * dx
    return c5

def compute_gradient(h, k_star):
    """Compute the gradient of C5 with respect to h for a given lag k_star."""
    n = len(h)
    grad = np.zeros(n)
    
    # Compute the contribution to gradient for each position
    # For lag k_star, we consider positions where both h[i] and h[i+k_star] are active
    # This corresponds to the correlation computation at lag k_star
    if k_star >= 0:
        # For positive lag
        # The gradient is (1 - h[i+k_star]) - h[i] for valid indices
        for i in range(n - k_star):
            grad[i] += 1 - h[i + k_star]
            grad[i + k_star] -= h[i]
    else:
        # For negative lag, reverse indices
        k_abs = -k_star
        for i in range(n - k_abs):
            grad[i] += 1 - h[i - k_abs]
            grad[i - k_abs] -= h[i]
    
    return grad

def project_h(h, target_sum=100.0/2.0):
    """Project h onto feasible region: [0,1] and sum=h_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to target sum
    current_sum = np.sum(h)
    if abs(current_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def line_search(h, grad, alpha_init=1e-3, reduction_factor=0.5, max_iter=20):
    """Perform line search to find good step size."""
    alpha = alpha_init
    h_current = h.copy()
    for _ in range(max_iter):
        h_new = h_current - alpha * grad
        h_new = project_h(h_new)
        c5_new = evaluate_overlap(h_new)
        c5_current = evaluate_overlap(h_current)
        # If improvement, accept; otherwise reduce alpha
        if c5_new < c5_current:
            return alpha, h_new
        else:
            alpha *= reduction_factor
    return alpha, h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h
    if 'initial_h_values' in globals():

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = evaluate_overlap(h)

    # Optimization loop
    max_iter = 5000
    alpha = 1e-3
    stagnation_count = 0
    stagnation_threshold = 100

    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate all correlations to find the maximum
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr) - (n_points - 1)  # Convert index to lag
        c5_current = np.max(corr) * dx

        # Compute gradient
        grad = compute_gradient(h, k_star)
        
        # Line search for step size
        alpha, h_new = line_search(h, grad, alpha=alpha)
        
        # Update h
        h = h_new
        
        # Evaluate new overlap
        c5_new = evaluate_overlap(h)
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Adaptive step size
