import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturbation_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    current_lr = initial_learning_rate
    last_improvement = 0
    plateau_count = 0
    
    for i in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
            plateau_count = 0
        else:
            plateau_count += 1
            
        if plateau_count > plateau_patience:
            # Reduce learning rate
            current_lr = max(current_lr * learning_rate_decay, min_learning_rate)
            plateau_count = 0
            
            # Add small random perturbation
            perturb = np.random.normal(0, perturbation_fraction, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            continue
            
        # Find the lag that gives the maximum correlation
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient for the maximum lag
        # The gradient w.r.t. C5 at lag max_lag_idx
        # dx * (1 - 2 * h) with appropriate shifting
        
        # Construct indicator arrays for convolution overlap at max_lag_idx
        # For mode='full', the correlation at lag k is a convolution of h with (1-h)
        # The lag index k represents a shift of k - (n_points - 1)
        # So for a valid alignment, consider:
        # h at i and (1-h) at i + (k - (n_points - 1))
        # Overlap indices: i = 0 to min(n_points-1, n_points-1 + (k - (n_points-1))) and such
        # Actually more directly, due to convolution mode:
        # For full mode, the output length is 2*n_points - 1, so max_lag_idx = 0 to 2*n_points - 2
        
        # Current math: in full mode, correlation at index k represents the overlap shift of (k - (n_points-1))
        lag_shift = max_lag_idx - (n_points - 1)
        
        # If the lag shift is positive, it means (1-h) is shifted to the right,
        # which aligns old_index with new_index = old_index + lag_shift
        # So we compute contribution of h[i] with (1-h)[i + lag_shift] for i in valid range
        # This means: compute gradient adapting to what the actual coefficients are
        
        grad = np.zeros(n_points)
        if lag_shift >= 0:
            # h[i] * (1 - h[i+lag_shift]) contributes to max correlation for i where i + lag_shift < n_points
            # and h defines overlap from left to right
            for i in range(n_points - lag_shift):
                if i + lag_shift < n_points:
                    # Contribution of h[i] to the gradient
                    # But also we look at advancing the peak to favor smaller h for that spot
                    # For h[i], the term contributing to peak is (h[i]*1h[i+lag_shift]) 
                    # This requires computing the gradient as derivative of a product
                    # d/dh[i]((h[i])(1 - h[i+lag_shift])) = 1 - h[i+lag_shift] - h[i]*0 = 1 - h[i+lag_shift]
                    # But additionally we have the symmetric term (h[i+lag_shift]) * 1 (1 - h[i]) 
                    # = 1 - h[i], so the total gradient is:
                    # Due to convolution structure, we must compute from all contributions: 
                    # from (1-h) * h shifted, we get grad for h[i] = 1 - h[i + lag_shift]
                    # and from h * (1-h) shifted, we get:
                    # from (1-h)[i] * h[i + lag_shift] we get grad for h[i + lag_shift] = - h[i]
                    # So compute components appropriately
                    
                    # Because the actual value to differentiate is:
                    # sum_i h[i] * (1 - h[i + lag_shift]) over valid overlaps
                    # For h[i] we get contribution (1 - h[i + lag_shift]) to gradient
                    # For h[i + lag_shift] we get gradient component (- h[i])
                    grad[i] += 1 - h[i + lag_shift]
                    if i + lag_shift < n_points:
                        grad[i + lag_shift] -= h[i]
        else:
            # Negative lag shift: (1-h) shifted to the left, so h patches to left
            # h[i + (-lag_shift)] * (1 - h[i]) contributes
            for i in range(-lag_shift, n_points):
                if i - (-lag_shift) < n_points:
                    if i - (-lag_shift) >= 0:
                        grad[i] += 1 - h[i - (-lag_shift)]
                        grad[i - (-lag_shift)] -= h[i]
        
        # Scale gradient by 2/n * dx = 2/n * 2/n
        grad = grad * 2*dx / n_points
        
        # Perform gradient descent step
        h = h - current_lr * grad
        
        # Repair step
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
