import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    plateau_patience = 200
    learning_rate = 1e-3
    learning_rate_decay = 0.5
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    penalty_initial = 0.1
    perturbation_fraction = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Penalty parameter
    penalty_mu = penalty_initial
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Track plateau
    last_improvement = 0
    current_lr = learning_rate
    
    # Main optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute penalty
        penalty = 0
        h_violation = np.sum(np.maximum(0, -h)) + np.maximum(0, np.sum(h) - n_points/2)
        penalty += penalty_mu * h_violation**2
        
        # Compute gradients
        k_max = np.argmax(corr)
        # The correlation is overfull, so we map it back
        # to the appropriate correlation index in the valid range
        if k_max < n_points:
            k = k_max
        else:
            k = k_max - n_points + 1
        
        # Create gradient for the specific maximum correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h[i + k]) * dx
            if i - k >= 0:
                grad[i] += (1 - h[i - k]) * dx
        
        # Gradient penalization (same as in gradient computation for constraint)
        penalty_grad = np.zeros(n_points)
        penalty_grad += 2 * penalty_mu * np.maximum(0, -h)  # for h[i] <= 0
        penalty_grad += 2 * penalty_mu * (np.sum(h) - n_points/2)  # for sum constraint
        
        # Combined gradient
        gradient = grad + penalty_grad
        
        # Line search for optimal step
        alpha = current_lr
        for _ in range(10):  # Simplified line search
            h_new = h - alpha * gradient
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5 + 1e-8:
                break
            alpha *= 0.5
        
        # Update h
        h = h - alpha * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update penalty parameter
        violation = np.sum(np.maximum(0, -h)) + np.maximum(0, np.sum(h) - n_points/2)
        if violation > 0:
            penalty_mu *= penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_min)
            
        # Check for stagnation
        if iteration - last_improvement > plateau_patience:
            # Perturb
            perturbation = np.random.normal(0, perturbation_fraction, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Reduce learning rate
            current_lr *= learning_rate_decay
            last_improvement = iteration
    
    # Final verification
    corr_best = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
