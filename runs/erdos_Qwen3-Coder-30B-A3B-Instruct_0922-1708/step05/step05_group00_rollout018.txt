import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Normalize the initial h to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse grid optimization
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    # Smooth max implementation
    def smooth_max(x, beta=10):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    def compute_overlap(h_vec):
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        return conv * dx_coarse
    
    def gradient_smooth_max(h_vec, beta=10):
        # This computes gradient of smooth max of overlap values
        overlap = compute_overlap(h_vec)
        exp_terms = np.exp(beta * overlap)
        sum_exp = np.sum(exp_terms)
        weights = exp_terms / sum_exp
        
        # Gradient of smooth max
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Compute gradient of overlap[i] w.r.t to h_vec
            g = np.zeros_like(h_vec)
            for j in range(len(h_vec)):
                if j == i:
                    g[j] = 1 - h_vec[(i + j) % len(h_vec)]  # h'(j) 
                elif j == (i + 1) % len(h_vec):
                    g[j] = -h_vec[i]  # (1-h)'(i)
                elif i == (j + 1) % len(h_vec):
                    g[j] = -h_vec[j]  # (1-h)'(j)
                else:
                    g[j] = 0
            grad[i] = np.sum(g * weights)
        return grad * dx_coarse * beta
    
    # Adam optimizer stepping
    def adam_optimize(h_start, max_iter=2000, lr=0.01, beta1=0.9, beta2=0.999):
        h = h_start.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        best_h = h.copy()
        best_c5 = float('inf')
        
        for t in range(1, max_iter + 1):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient
            try:
                grad = gradient_smooth_max(h)
            except:
                break
            
            # Update Adam
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            h_update = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            h = h - h_update
            
            # Project
            h = np.clip(h, 0, 1)
            h = h * (n_coarse / 2) / np.sum(h)
            
            # Evaluate
            overlap = compute_overlap(h)
            current_c5 = np.max(overlap)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
        return best_h, best_c5
    
    # Optimize coarse
    h_coarse_opt, c5_coarse = adam_optimize(h_coarse, 2000, 0.01, 0.9, 0.999)
    
    # Transfer to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse_opt)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine grid optimization with finer step size and more iterations
    def gradient_smooth_max_fine(h_vec, beta=10):
        overlap = compute_overlap(h_vec)
        exp_terms = np.exp(beta * overlap)
        sum_exp = np.sum(exp_terms)
        weights = exp_terms / sum_exp
        
        # Derivative of smooth max with respect to h[i]
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Compute gradients of overlaps that depend on h[i]
            # Overlap at position k will have contribution to h[i] from terms involving h[i-k] and h[i+k]
            # This is a bit tricky as the convolution kernel involves shifted positions.
            g = np.zeros_like(h_vec)
            for j in range(len(h_vec)):
                if (j - i) % len(h_vec) == 0:  # When h[i] * (1 - h[i])
                    g[j] = 1 - h_vec[j] if i == j else 0
                elif (j - i) % len(h_vec) == 1:  # When h[i] * (1 - h[i+1])
                    g[j] = -h_vec[i]
                elif (i - j) % len(h_vec) == 1:  # When h[i-1] * (1 - h[i])
                    g[j] = -h_vec[j]
                else:
                    g[j] = 0  
            grad[i] = np.sum(g * weights)
        return grad * dx_coarse * beta
    
    # Fine optimization
    h_fine_opt, c5_fine = adam_optimize(h_fine, 2000, 0.005, 0.9, 0.999)
    
    # Final evaluation
    final_overlap = np.correlate(h_fine_opt, 1 - h_fine_opt, mode='full') * (2.0 / n_points)
    final_c5 = np.max(final_overlap)
    
    # Final checks
    assert abs(np.sum(h_fine_opt) * (2.0 / n_points) - 1.0) < 1e-6, "Sum constraint violated"
    assert np.all(h_fine_opt >= 0) and np.all(h_fine_opt <= 1), "Bounds violated"
    
    return (h_fine_opt.tolist(), final_c5, n_points)
