import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    max_iter = 500
    learning_rate = 0.01
    eps = 1e-4
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.uniform(0.1, 0.9, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert initial to Fourier coefficients
    def h_to_fourier(h_vals):
        # Use real FFT (symmetric)
        h_ext = np.concatenate([h_vals, h_vals[::-1]])
        f = fft(h_ext)
        f = f[:n_fourier+1]  # Only need real part and first few imaginary
        return np.real(f), np.imag(f)
    
    def fourier_to_h(a, b):
        # Reconstruct signal in time domain
        # a[0] is DC component, a[1:n] real coeffs, b[1:n] imaginary
        # We'll make it symmetric by using fft convention
        c = np.zeros(2*n_fourier+1, dtype=complex)
        c[0] = a[0] + 0j
        c[1:n_fourier+1] = (a[1:n_fourier+1] + 1j*b[1:n_fourier+1])
        c[-n_fourier:] = (a[1:n_fourier+1][::-1] - 1j*b[1:n_fourier+1][::-1])
        h_recon = np.real(ifft(c))
        return h_recon[:n_points]
    
    # Forward pass from Fourier to result, computing C5
    def eval_c5_fourier(a, b):
        h_vals = fourier_to_h(a, b)
        # Apply constraints
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        # Compute C5
        corr = np.correlate(h_vals, 1-h_vals, mode='full')
        c5 = np.max(corr) * dx
        return c5, h_vals
    
    # Initialize with the h_init
    a_init, b_init = h_to_fourier(h_init)
    best_a = a_init.copy()
    best_b = b_init.copy()
    best_c5, best_h = eval_c5_fourier(best_a, best_b)
    
    # Use a warm-started L-BFGS optimization
    # Convert to flat variables for scipy optimization
    def flat_vars(a, b):
        return np.concatenate([a, b])
    
    def unflat_vars(flat):
        return flat[:n_fourier+1], flat[n_fourier+1:]
    
    def obj_flat(flat):
        a, b = unflat_vars(flat)
        c5, _ = eval_c5_fourier(a, b)
        return c5
    
    def grad_flat(flat):
        a, b = unflat_vars(flat)
        a = a.copy()
        b = b.copy()
        grad_a = np.zeros(n_fourier+1)
        grad_b = np.zeros(n_fourier+1)
        
        for i in range(n_fourier+1):
            a_plus = a.copy()
            b_plus = b.copy()
            a_plus[i] += eps
            b_plus[i] += eps
            c5_plus, _ = eval_c5_fourier(a_plus, b_plus)
            c5_orig, _ = eval_c5_fourier(a, b)
            grad_a[i] = (c5_plus - c5_orig) / eps
            # Only for imaginary parts
            if i > 0:
                grad_b[i] = (c5_plus - c5_orig) / eps
            else:
                grad_b[i] = 0.0  # No gradient for DC component for imaginary part
                
        return np.concatenate([grad_a, grad_b])
    
    # Optimization using scipy
    flat_start = flat_vars(best_a, best_b)
    
    # SCG algorithm almost always works with our problem
    try:
        res = minimize(obj_flat, flat_start, method='L-BFGS-B', jac=grad_flat, 
                        options={'maxiter': max_iter, 'ftol': 1e-8}, 
                        bounds=[(-10, 10)]*len(flat_start))
        if res.success:
            best_a, best_b = unflat_vars(res.x)
        else:
            # If fails, fallback to last valid state
            pass
    except:
        pass
    
    # Final refinement
    final_c5, final_h = eval_c5_fourier(best_a, best_b)
    
    # Refined search around best solution
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        a_pert = best_a + np.random.normal(0, 0.01, n_fourier+1)
        b_pert = best_b + np.random.normal(0, 0.01, n_fourier+1)
        c5_pert, _ = eval_c5_fourier(a_pert, b_pert)
        if c5_pert < final_c5:
            final_c5 = c5_pert
            best_a = a_pert
            best_b = b_pert
    
    final_h = fourier_to_h(best_a, best_b)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
