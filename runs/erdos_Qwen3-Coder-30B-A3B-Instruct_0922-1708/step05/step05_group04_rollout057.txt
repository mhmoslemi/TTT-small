import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 5000
    mutation_strength = 0.5
    diversity_threshold = 200
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)

    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)

    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Main evolution loop
    best_fitness = float('inf')
    best_h = None
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            fitness = compute_c5(h)
            fitnesses.append(fitness)
            
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection: top 50%
        sorted_indices = np.argsort(fitnesses)[:population_size // 2]
        selected = [population[i] for i in sorted_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            if len(selected) > 0:
                parent = selected[np.random.randint(len(selected))]
                # Mutation with adaptive variance
                sigma = mutation_strength / 5.0
                noise = np.random.normal(0, sigma, n_points)
                child = parent + mutation_strength * noise
                child = repair_h(child)
                offspring.append(child)
            else:
                h = np.random.uniform(0, 1, n_points)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                offspring.append(h)
        
        # Replace worst in population
        population.extend(offspring)
        # Sort by fitness and truncate
        fitness_pairs = [(f, i) for i, (f, h) in enumerate(zip(fitnesses, population))]
        fitness_pairs.sort()
        population = [population[i] for f, i in fitness_pairs[:population_size]]
        
        # Mutation decay
        mutation_strength *= 0.999
        if mutation_strength < 0.01:
            mutation_strength = 0.01
            
        # Restart if stagnation
        if stagnation_count > diversity_threshold:
            mutation_strength = 0.5
            stagnation_count = 0
            # Perturb some individuals
            for i in range(0, population_size, 4):
                if i < population_size:
                    population[i] += np.random.normal(0, 0.02, n_points)
                    population[i] = repair_h(population[i])

    # Final refinement with gradient descent
    def refine_h(h):
        h_ref = h.copy()
        for _ in range(100):
            f_current = compute_c5(h_ref)
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus)
                f_plus = compute_c5(h_plus)
                grad[i] = (f_plus - f_current) / eps
            h_ref -= 0.001 * grad
            h_ref = repair_h(h_ref)
            f_new = compute_c5(h_ref)
            if f_new >= f_current:
                break
        return h_ref

    if best_h is not None:
        refined_h = refine_h(best_h)
        final_c5 = compute_c5(refined_h)
        if final_c5 < best_fitness:
            best_fitness = final_c5
            best_h = refined_h.copy()

    return (best_h.tolist(), best_fitness, n_points)
