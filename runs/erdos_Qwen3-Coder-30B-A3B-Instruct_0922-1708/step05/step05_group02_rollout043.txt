import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Sample from Lévy distribution for perturbations
    # Using the symmetric stable distribution with alpha parameter
    # For simplicity, we sample from a Laplace distribution as a proxy
    return np.random.laplace(0, scale, size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with a feasible solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial population
    population_size = 50
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Find best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Main optimization loop
    max_iter = 1000
    max_no_improvement = 100
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new candidates using Lévy flights
        new_candidates = []
        for h in population:
            # Apply Lévy flight perturbation to current candidate
            pert = levy_flight(n_points, alpha=1.5, scale=0.5)
            new_h = h + pert
            
            # Repair new candidate to satisfy constraints
            new_h = repair_h(new_h, n_points)
            new_candidates.append(new_h)
            
        # Evaluate new candidates
        new_fitness = [evaluate_h(h, dx) for h in new_candidates]
        
        # Update population with better candidates
        for i in range(len(population)):
            if new_fitness[i] < fitness[i]:
                population[i] = new_candidates[i].copy()
                fitness[i] = new_fitness[i]
                
        # Update best solution if found
        current_best_idx = np.argmin(fitness)
        current_best_value = fitness[current_best_idx]
        if current_best_value < best_value:
            best_h = population[current_best_idx].copy()
            best_value = current_best_value
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Occasionally refine top individuals using gradient descent
        if iteration % 20 == 0:
            # Select top 10 candidates
            top_indices = np.argsort(fitness)[:min(10, len(fitness))]
            for idx in top_indices:
                h = population[idx].copy()
                current_c5 = fitness[idx]
                # Simple gradient approximation with small steps
                for _ in range(50):
                    if time.time() - start_time > budget_s - 10:
                        break
                    eps = 1e-6
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - current_c5) / eps
                    
                    # Update using gradient
                    h = h - 0.001 * grad
                    h = repair_h(h, n_points)
                    corr = correlate(h, 1 - h, mode='full')
                    current_c5 = np.max(corr) * dx
                    
                    if current_c5 < fitness[idx]:
                        population[idx] = h.copy()
                        fitness[idx] = current_c5
                        if current_c5 < best_value:
                            best_h = h.copy()
                            best_value = current_c5
                            
        # Early stopping if no improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Final refinement using a more aggressive gradient descent
    final_h = best_h.copy()
    final_value = best_value
    
    for _ in range(300):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - final_value) / eps
            
        final_h = final_h - 0.0005 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')
        final_value = np.max(corr) * dx
    
    return (final_h.tolist(), final_value, n_points)
