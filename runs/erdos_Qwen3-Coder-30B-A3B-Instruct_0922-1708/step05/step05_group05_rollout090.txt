import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_count = 5
    initial_mutation_rate = 0.1
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        initial_h = h_init.copy()
    else:
        # Use Halton sequence initialization
        sampler = qmc.Halton(d=n_points, seed=seed)
        samples = sampler.random(n=population_size)
        initial_h = np.zeros(n_points)
        for i in range(n_points):
            initial_h[i] = samples[0, i] * 2.0  # Scale to [0, 2]
        # Normalize
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Create initial population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_fitness = float('inf')
    best_h = None
    
    # Precompute some constants
    half_len = n_points - 1  # Length of full correlation
    
    start_time = time.time()
    
    # Evolutionary loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute fitness for all individuals
        fitnesses = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
            
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = h.copy()
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, size=tournament_size)
            winner_idx = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
        
        # Create new population with elitism
        new_population = []
        
        # Elitism: keep the best individuals
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select parents
            parent1 = selected[np.random.randint(0, population_size)]
            parent2 = selected[np.random.randint(0, population_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single point crossover
                point = np.random.randint(1, n_points)
                offspring1 = np.concatenate([parent1[:point], parent2[point:]])
                offspring2 = np.concatenate([parent2[:point], parent1[point:]])
            else:
                offspring1 = parent1.copy()
                offspring2 = parent2.copy()
            
            # Mutation
            mut_rate = initial_mutation_rate * (0.99 ** (gen // 10))
            for i in range(n_points):
                if np.random.rand() < mut_rate:
                    offspring1[i] += np.random.normal(0, 0.01)
                if np.random.rand() < mut_rate:
                    offspring2[i] += np.random.normal(0, 0.01)
            
            # Repair constraints
            offspring1 = np.clip(offspring1, 0, 1)
            offspring2 = np.clip(offspring2, 0, 1)
            
            offspring1 = offspring1 * (n_points / 2) / np.sum(offspring1)
            offspring2 = offspring2 * (n_points / 2) / np.sum(offspring2)
            
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
        
        # Update population
        population = new_population[:population_size]
    
    # Final refinement with gradient descent on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        for _ in range(500):
            # Estimate gradient via finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            base_fitness = np.max(base_corr) * dx
            
            improved = False
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                
                grad[j] = (fitness_plus - base_fitness) / eps
                
                if fitness_plus < base_fitness:
                    improved = True
            
            if not improved:
                break
                
            # Update
            h_refined -= 0.001 * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Recompute final fitness
        corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(corr_final) * dx
        
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_fitness, n_points)
