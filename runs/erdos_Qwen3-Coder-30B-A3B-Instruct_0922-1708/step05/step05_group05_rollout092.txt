import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        initial_population = [h_init.copy()]
    else:
        # Generate initial population with Halton sequences
        sampler = qmc.Halton(d=n_points, seed=seed)
        points = sampler.random(n=population_size-1)
        initial_population = []
        for i in range(population_size-1):
            h = points[i] * 2  # Scale to [0,2]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            initial_population.append(h.copy())
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        initial_population.append(h_init.copy())
    
    # Create population
    population = initial_population
    fitness_history = []
    
    start_time = time.time()
    
    # Helper functions
    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair_constraints(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-6:
            diff = current_sum - n_points / 2
            if diff > 0:
                sorted_indices = np.argsort(h)[::-1]
                for idx in sorted_indices:
                    if diff <= 0:
                        break
                    available = h[idx]
                    reduce = min(diff, available)
                    h[idx] -= reduce
                    diff -= reduce
            else:
                sorted_indices = np.argsort(h)
                for idx in sorted_indices:
                    if diff >= 0:
                        break
                    available = 1.0 - h[idx]
                    increase = min(-diff, available)
                    h[idx] += increase
                    diff += increase
            h = h * (n_points / 2) / np.sum(h)
        return h
    
    def tournament_selection(population, fitnesses, k):
        indices = np.random.choice(len(population), size=k)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx]
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, rate):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                h[i] += np.random.normal(0, 0.05)
        return repair_constraints(h)
    
    # Main GA loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute fitness
        fitnesses = [compute_fitness(h) for h in population]
        
        # Track best
        best_idx = np.argmin(fitnesses)
        best_fitness = fitnesses[best_idx]
        fitness_history.append(best_fitness)
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        # Adapt mutation rate
        if generation % 10 == 0 and generation > 0:
            mutation_rate *= 0.99
    
    # Final evaluation
    final_fitnesses = [compute_fitness(h) for h in population]
    best_idx = np.argmin(final_fitnesses)
    best_h = population[best_idx].copy()
    final_fitness = final_fitnesses[best_idx]
    
    return (best_h.tolist(), final_fitness, n_points)
