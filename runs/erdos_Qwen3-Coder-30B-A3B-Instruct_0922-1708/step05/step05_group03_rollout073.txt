import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def evaluate(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Differential Evolution parameters
    popsize = 50
    mutation_factor = 0.8
    crossover_rate = 0.7
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(popsize):
        if 'initial_h_values' in globals():
            new_h = h + np.random.normal(0, 0.05, n_points)
        else:
            new_h = np.random.rand(n_points)
        new_h = project(new_h)
        population.append(new_h)
    
    # Main evolution loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness of population
        fitness = [evaluate(ind) for ind in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Create new population
        new_population = []
        
        for i in range(popsize):
            # Select three random individuals
            candidates = list(range(popsize))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = project(mutant)
            
            # Crossover
            trial = np.where(np.random.rand(n_points) < crossover_rate, mutant, population[i])
            trial = project(trial)
            
            # Selection
            trial_fitness = evaluate(trial)
            if trial_fitness < fitness[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
    
    # Local refinement
    def local_refinement(h, max_iter=100):
        h = h.copy()
        for _ in range(max_iter):
            # Gradient estimation
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = project(h_plus)
                c5_plus = evaluate(h_plus)
                grad[j] = (c5_plus - evaluate(h)) / eps
            
            # Update with gradient descent
            h = h - 0.005 * grad
            h = project(h)
            
            if evaluate(h) < best_c5:
                best_c5 = evaluate(h)
                best_h = h.copy()
        
        return h
    
    # Apply local refinement to top solutions
    fitness = [evaluate(ind) for ind in population]
    top_indices = np.argsort(fitness)[:popsize//5]
    
    for i in top_indices:
        refined = local_refinement(population[i], 50)
        if evaluate(refined) < best_c5:
            best_c5 = evaluate(refined)
            best_h = refined.copy()
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
