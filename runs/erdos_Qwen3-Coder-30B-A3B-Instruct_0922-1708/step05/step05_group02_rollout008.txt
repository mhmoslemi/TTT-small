import numpy as np
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    dx = 2.0 / n_points
    start_time = time.time()

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    # Cuckoo Search parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 1000
    min_improvements = 200
    
    best_value = evaluate_h(best_h, dx)
    no_improvement_count = 0
    step_size = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution via random walk (Levy flight)
        new_h = best_h.copy()
        
        # Perturb a subset of elements
        n_perturbed = max(1, int(n_points * 0.1))  # Perturb 10% of elements
        indices = np.random.choice(n_points, n_perturbed, replace=False)
        
        # Apply perturbation with step size
        for i in indices:
            new_h[i] += np.random.normal(0, step_size)
            
        # Repair the new solution
        new_h = repair_h(new_h, n_points)
        
        # Evaluate the new solution
        new_value = evaluate_h(new_h, dx)
        
        # Accept or reject the new solution
        if new_value < best_value:
            best_h = new_h.copy()
            best_value = new_value
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Accept with probability based on negative log of improvement
            if np.random.rand() < np.exp(-alpha * (new_value - best_value)):
                best_h = new_h.copy()
                best_value = new_value
                no_improvement_count = 0
                
        # Decrease step size progressively to be more fine-tuned
        step_size *= 0.995
        
        # Adaptive disintegration rate to prevent stagnation
        if no_improvement_count > min_improvements:
            step_size *= 1.1
            no_improvement_count = 0

    # Final refinement with gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:

        grad = np.zeros(n_points)
