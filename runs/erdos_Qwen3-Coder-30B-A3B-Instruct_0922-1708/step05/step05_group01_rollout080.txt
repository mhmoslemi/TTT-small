import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or randomly
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimizer parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    min_alpha = 1e-8
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find maximum lag and compute gradient
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Create the shifted version
        if max_lag > 0:
            shifted = np.concatenate([np.zeros(max_lag), h[:-max_lag]])
        elif max_lag < 0:
            shifted = np.concatenate([h[-max_lag:], np.zeros(-max_lag)])
        else:
            shifted = h.copy()
        
        # Compute gradient: (2/n) * (1 - shifted - h)
        grad = (2.0 / n_points) * (1 - shifted - h)
        
        # Adaptive line search
        new_h = h - alpha * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Check if this improves the objective
        new_corr = correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        # If not, reduce step size
        while new_c5 >= c5 and alpha > min_alpha:
            alpha *= line_search_factor
            new_h = h - alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_corr = correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(new_corr) * dx
            
        # Update parameters
        h = new_h
        if np.linalg.norm(grad) < grad_threshold:
            alpha *= 0.95
        if new_c5 < c5:
            # Gradually increase learning rate if we improve
            alpha = min(alpha * 1.05, 1e-1)
        else:
            # Reduce learning rate if no improvement
            alpha *= 0.95
        alpha = max(alpha, min_alpha)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
