import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 100
        # Use structured initialization to distribute values well
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store the best solution found
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper function to compute C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)
    
    # Step 1: Coarse grid optimization
    coarse_points = 50
    coarse_dx = 2.0 / coarse_points
    coarse_h = np.interp(np.linspace(0, 2, coarse_points), 
                         np.linspace(0, 2, n_points), h)
    
    # Fallback to uniform if necessary
    if np.sum(coarse_h) == 0:
        coarse_h = np.ones(coarse_points) * 0.5
    else:
        coarse_h = coarse_h * (coarse_points / 2) / np.sum(coarse_h)
    coarse_h = np.clip(coarse_h, 0, 1)
    
    # Optimize coarse version using finite differences
    # Use Adam-inspired update with adaptive step
    max_iter_coarse = 2000
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    m = np.zeros_like(coarse_h)
    v = np.zeros_like(coarse_h)
    
    for step in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(coarse_h)
        eps_fd = 1e-5
        
        for i in range(coarse_points):
            h_plus = coarse_h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (coarse_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - compute_c5(coarse_h)) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(step + 1))
        v_hat = v / (1 - beta2**(step + 1))
        
        # Update parameters
        coarse_h = coarse_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        coarse_h = np.clip(coarse_h, 0, 1)
        
        # Scale to sum constraint
        coarse_h = coarse_h * (coarse_points / 2) / np.sum(coarse_h)
        
        # Evaluate and update best
        c5_current = compute_c5(coarse_h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = np.interp(np.linspace(0, 2, n_points), 
                              np.linspace(0, 2, coarse_points), 
                              coarse_h)
            
            if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
                best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Step 2: Transfer solution to fine grid
    h = best_h.copy()
    if len(h) != n_points:
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
    
    # Refine fine grid
    max_iter_fine = 2000
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    best_c5 = float('inf')
    
    for step in range(max_iter_fine):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - compute_c5(h)) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(step + 1))
        v_hat = v / (1 - beta2**(step + 1))
        
        # Update parameters
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        
        # Scale to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate and update best
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    
    final_c5 = compute_c5(best_h)
    
    # Ensure sum constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
