import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided initial construction if available
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        n_coarse = len(h_coarse)
        if n_coarse != 100:
            # Interpolate to 100 points
            x_orig = np.linspace(0, 2, n_coarse)
            x_new = np.linspace(0, 2, 100)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        # Start with uniform distribution
        h_coarse = np.ones(100) * 0.5
        
    # Normalize sum to n_points / 2 = 50
    h_coarse = h_coarse * 50 / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # First, try a combination of LP with refinement
    current_h = h_coarse.copy()
    
    # Set up cvxpy optimization for relaxation
    try:
        # Use CVXPY with linear program relaxation
        h_var = cp.Variable(n_points)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == 50,  # sum h = 1/dx = 100/2 = 50
            C5_var >= 0
        ]
        
        # Try to minimize the maximum constraint
        # We'll add an approximation of correlation constraint using auxiliary variables
        
        # For the relaxation approach: 
        # We won't try to exactly model h*x in a convex way
        # Instead we'll reach out to the gradient based optimization but 
        # we'll seed with this better initial solution to improve convergence
        
        C5 = 1.0  # Initial guess, not a variable in LP
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        try:
            prob.solve(solver=ECOS, max_iters=200)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_solution = h_var.value
                if h_solution is not None:
                    current_h = np.array(h_solution)
                    current_h = np.clip(current_h, 0, 1)
                    current_h = current_h * 50 / np.sum(current_h)
        except Exception:
            pass
    except Exception:
        pass  # Fall back to other methods
    
    # Refinement loop starting from near-optimal solution
    best_h = current_h.copy()
    best_c5 = float('inf')
    
    iterations = 0
    max_iter = 5000
    lr = 1e-3
    patience = 0
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(current_h, 1 - current_h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            patience = 0  # reset patience on improvement
        else:
            patience += 1
            if patience > 100:
                break  # early stopping

        # Compute gradient using finite differences
        grad = np.zeros_like(current_h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * 50 / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Apply gradient step
        current_h_new = current_h - lr * grad
        current_h_new = np.clip(current_h_new, 0, 1)
        current_h_new = current_h_new * 50 / np.sum(current_h_new)
        
        current_h = current_h_new
        
        # Adaptive learning rate
        lr = max(1e-6, lr * 0.999)
        
        iterations += 1
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Final normalization check
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * 50 / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
