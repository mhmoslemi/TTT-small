import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Default initialization
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients (real FFT)
    h_fft = fft(h_init)
    # Take only the first n_fourier coefficients for optimization
    fourier_coeffs = h_fft[:n_fourier]
    
    # Function to convert Fourier coefficients back to h
    def fourier_to_h(coeffs):
        # Pad or truncate coefficients
        full_coeffs = np.zeros(n_points, dtype=complex)
        full_coeffs[:n_fourier] = coeffs
        # Ensure symmetry for real signal
        full_coeffs[n_points//2:] = np.conj(full_coeffs[n_points//2:0:-1])
        h = np.real(ifft(full_coeffs))
        # Clip to bounds
        h = np.clip(h, 0, 1)
        # Rescale
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to compute C5
    def compute_c5(h):
        # Correlation in full mode
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient computation using finite differences
    def compute_gradient(coeffs, eps=1e-5):
        grad = np.zeros_like(coeffs, dtype=complex)
        c_orig = coeffs.copy()
        for i in range(len(coeffs)):
            coeffs[i] += eps
            h_plus = fourier_to_h(coeffs)
            f_plus = compute_c5(h_plus)
            
            coeffs[i] -= 2 * eps
            h_minus = fourier_to_h(coeffs)
            f_minus = compute_c5(h_minus)
            
            grad[i] = (f_plus - f_minus) / (2 * eps)
            coeffs[i] = c_orig[i]  # Restore
        return grad
    
    # Optimization
    best_coeffs = fourier_coeffs.copy()
    best_c5 = compute_c5(fourier_to_h(best_coeffs))
    
    # Gradient descent
    learning_rate = 0.01
    max_iter = 500
    stagnation_count = 0
    patience = 100
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        grad = compute_gradient(best_coeffs)
        
        # Update
        best_coeffs -= learning_rate * grad
        
        # Project to feasible region (clip if necessary)
        h_test = fourier_to_h(best_coeffs)
        test_c5 = compute_c5(h_test)
        
        if test_c5 < best_c5:
            best_c5 = test_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > patience:
            # Early stopping if no improvement
            break
    
    # Final reconstruction
    final_h = fourier_to_h(best_coeffs)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
