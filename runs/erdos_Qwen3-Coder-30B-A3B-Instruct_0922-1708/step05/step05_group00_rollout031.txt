import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    patience_limit = 300
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    # Normalize coarse solution
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Coarse grid optimization
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = float('inf')
    patience = 0
    last_c5 = float('inf')
    
    # Fine grid initialization
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Optimization loop for fine grid
    best_h = h_fine.copy()
    best_c5 = float('inf')
    iterations = 0
    lr = 1e-3
    last_c5 = float('inf')
    patience = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update
        h_new = h_fine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_fine = h_new
        lr = max(1e-6, lr * 0.9995)
        iterations += 1
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint exactly
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
