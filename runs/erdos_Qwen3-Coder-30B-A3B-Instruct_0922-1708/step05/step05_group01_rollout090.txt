import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 0.001
    mu = 0.1
    mu_factor = 1.2
    perturb_fraction = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary indicator z[i] = 1 if h[i] > 0.5 else 0
    z = (h > 0.5).astype(float)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the lag that maximizes overlap
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        # Compute gradient based on the max lag
        # For lag k, gradient is (1 - h_shifted - h)
        # We compute this as a convolution with an indicator vector
        grad = np.zeros(n_points)
        if max_lag >= 0:
            # h_i contributes to (1-h_{i+k}) - h_i for h_{i+k} in valid range
            for i in range(n_points - max_lag):
                grad[i] -= (1 - h[i + max_lag] - h[i]) * dx
            # Add derivative of penalty term
            grad += mu * (h - z)
        else:
            # max_lag < 0
            for i in range(-max_lag, n_points):
                grad[i] -= (1 - h[i + max_lag] - h[i]) * dx
            grad += mu * (h - z)
        
        # Update h
        h = h - learning_rate * grad
        
        # Repair feasibility: keep h in [0,1] and sum to n/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            if h_sum > target_sum:
                # reduce largest elements
                sorted_indices = np.argsort(h)[::-1]
                excess = h_sum - target_sum
                for idx in sorted_indices:
                    if excess <= 0:
                        break
                    reduce_amount = min(excess, h[idx])
                    h[idx] -= reduce_amount
                    excess -= reduce_amount
            else:
                # increase smallest elements
                sorted_indices = np.argsort(h)
                deficit = target_sum - h_sum
                for idx in sorted_indices:
                    if deficit <= 0:
                        break
                    increase_amount = min(deficit, 1 - h[idx])
                    h[idx] += increase_amount
                    deficit -= increase_amount
            
            # Rescale
            h = h * (n_points / 2) / np.sum(h)
            h = np.clip(h, 0, 1)
        
        # Update z (binary indicator)
        z = (h > 0.5).astype(float)
        
        # Update penalty parameter
        mu *= mu_factor
        
        # Add random perturbation occasionally
        if iteration % 100 == 0 and iteration > 0:
            perturb = np.random.randn(n_points) * perturb_fraction
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
