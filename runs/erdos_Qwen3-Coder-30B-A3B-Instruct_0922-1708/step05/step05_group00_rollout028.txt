import numpy as np
from scipy.optimize import minimize
import time

import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        # Ensure it has 100 points by interpolation if necessary
        if n_points != 100:
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 100)
            h = np.interp(x_new, x_orig, h)
            n_points = 100
    else:
        # Random initialization with some structure
        n_points = 100
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    
    # Function to compute C5
    def compute_c5(h_values):
        h = np.array(h_values)
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    # Function to compute gradient using finite differences
    def compute_gradient(h_values, eps=1e-4):
        grad = np.zeros_like(h_values)
        base_c5 = compute_c5(h_values)
        
        for i in range(len(h_values)):
            h_plus = h_values.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        return grad
    
    # Optimization loop
    max_iter = 3000
    last_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 and gradient
        current_c5 = compute_c5(h)
        grad = compute_gradient(h)
        
        # Update using Adam
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** (iteration + 1))
        v_hat = v / (1 - beta2 ** (iteration + 1))
        
        # Update parameters
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Check for improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping
        if patience > 200:
            break
            
        # Update learning rate
        if iteration > 100:
            lr = max(0.001, lr * 0.999)
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
