import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the maximum overlap integral for a given h."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region with constraints h[i] in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2.0) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover between two parents."""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.03):
    """Apply Gaussian mutation to the individual."""
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + mask * np.random.normal(0, sigma, len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        # Use the initial solution as a base
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
        
        # Add variations around the base solution
        for _ in range(population_size // 2):
            h = h_base + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        
        # Add random solutions
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Initialize completely random
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Compute initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = min(fitnesses)
    
    start_time = time.time()
    
    generation = 0
    stagnation_counter = 0
    prev_best_c5 = best_c5 + 1.0
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Stop if stagnated
        if stagnation_counter > 100:
            break
            
        # Selection, crossover, and mutation
        new_population = [population[0]]  # Elitism
        
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < 0.8:  # Crossover rate
                child1, child2 = crossover(parent1, parent2, alpha=0.8)
                child1 = project_to_feasible(mutate(child1), n_points)
                child2 = project_to_feasible(mutate(child2), n_points)
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Just mutate
                child1 = project_to_feasible(mutate(parent1), n_points)
                new_population.append(child1)
        
        population = new_population
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Periodic gradient refinement
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                
                # Gradient-based refinement
                for _ in range(30):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        
                        c5_plus = compute_fitness(h_plus, dx)
                        grad[j] = (c5_plus - fitnesses[idx]) / eps
                    
                    # Update with gradient
                    h_ref = h_ref - 0.0005 * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    # Re-evaluate
                    fitnesses[idx] = compute_fitness(h_ref, dx)
                    population[idx] = h_ref.copy()
                    
                    # Update best if improved
                    if fitnesses[idx] < best_c5:
                        best_c5 = fitnesses[idx]
                        best_h = h_ref.copy()
    
    # Final refinement
    for _ in range(10):
        h_ref = best_h.copy()
        for _ in range(50):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                
                c5_plus = compute_fitness(h_plus, dx)
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            new_c5 = compute_fitness(h_ref, dx)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
