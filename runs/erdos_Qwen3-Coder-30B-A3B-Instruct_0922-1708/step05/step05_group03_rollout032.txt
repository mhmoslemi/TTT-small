import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Population-based evolutionary search with local refinement
    population_size = 100
    generations = 0
    max_generations = 2000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        # Use initial solution as basis
        population.append(h.copy())
        for _ in range(population_size - 1):
            # Add variation around initial solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Initialize population with random feasible solutions
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    # Main evolutionary loop
    while time.time() - start_time < budget_s - 10 and generations < max_generations:
        generations += 1
        
        # Evaluate fitness for all individuals in population
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Update best solution found
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Select parents (tournament selection)
        selected_parents = []
        tournament_size = 5
        for _ in range(population_size // 2):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected_parents.append(population[winner_idx])
        
        # Generate new population
        new_population = []
        
        # Elitism: keep best 20%
        elite_count = population_size // 5
        elite_indices = np.argsort(fitness)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected_parents, 2, replace=False)
            
            # Blend crossover (arithmetic)
            alpha = np.random.uniform(0.7, 0.95)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            mutation_rate = 0.1
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    # Add small Gaussian noise
                    child1[i] += np.random.normal(0, 0.02)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.02)
            
            # Repair solutions
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population
        
        # Periodic local refinement (every 50 generations)
        if generations % 50 == 0:
            # Local refinement on top 10% of population
            top_indices = np.argsort(fitness)[:population_size // 10]
            for idx in top_indices:
                # Simple gradient ascent refinement with finite differences
                h_refine = population[idx].copy()
                lr = 0.001
                
                for _ in range(20):
                    # Estimate gradient using finite differences
                    grad = np.zeros_like(h_refine)
                    eps = 1e-5
                    
                    for j in range(n_points):
                        h_plus = h_refine.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    # Update with gradient descent (but maximization is done by minimizing negative)
                    h_refine = h_refine - lr * grad
                    h_refine = np.clip(h_refine, 0, 1)
                    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                    
                    # Recompute fitness
                    corr_new = np.correlate(h_refine, 1 - h_refine, mode='full')
                    new_c5 = np.max(corr_new) * dx
                    
                    if new_c5 < fitness[idx]:
                        fitness[idx] = new_c5
                        population[idx] = h_refine.copy()
                        
                        # Update global best if needed
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h_refine.copy()
    
    # Final refinement pass on best individuals
    top_individuals = np.argsort(fitness)[:5]
    for idx in top_individuals:
        h_refine = population[idx].copy()
        
        # Gradient-based local optimization
        for _ in range(50):
            grad = np.zeros_like(h_refine)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h_refine.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - fitness[idx]) / eps
            
            h_refine = h_refine - 0.001 * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            
            corr = np.correlate(h_refine, 1 - h_refine, mode='full')
            new_c5 = np.max(corr) * dx
            
            if new_c5 < fitness[idx]:
                fitness[idx] = new_c5
                population[idx] = h_refine.copy()
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_refine.copy()
    
    # Final validation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
