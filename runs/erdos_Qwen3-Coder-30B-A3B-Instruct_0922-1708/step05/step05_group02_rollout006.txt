import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population with 20 solutions + initial
    population = [h_init.copy()]
    for _ in range(19):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate all solutions
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Best solution tracking
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 1000
    stagnation_limit = 200
    stagnation_count = 0
    last_improvement = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new solution by Levy flight
        new_h = best_h.copy()
        
        # Perturb a random subset of elements
        num_perturbed = max(1, int(n_points * alpha))
        indices = np.random.choice(n_points, num_perturbed, replace=False)
        
        # Apply perturbation using random walk
        for idx in indices:
            # Gaussian perturbation
            step = np.random.normal(0, 0.1)
            new_h[idx] += step
            
        # Repair solution
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(new_h, dx)
        
        # Accept or reject based on probability
        if new_value < best_value:
            # Accept immediately
            population.append(new_h)
            fitness.append(new_value)
            if new_value < best_value:
                best_h = new_h.copy()
                best_value = new_value
                last_improvement = iter_count
        else:
            # Accept with probability based on difference
            diff = new_value - best_value
            prob = np.exp(-diff * 10)  # 10 is a scaling factor
            if np.random.rand() < prob:
                population.append(new_h)
                fitness.append(new_value)
        
        # Remove worst solution to maintain size
        if len(population) > 20:
            worst_idx = np.argmax(fitness)
            population.pop(worst_idx)
            fitness.pop(worst_idx)
        
        # Local search on best solution
        if iter_count % 30 == 0:
            h_refined = best_h.copy()
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_value) / epsilon
                
                h_refined = h_refined - 0.001 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                new_value = np.max(corr) * dx
                
                if new_value < best_value:
                    best_h = h_refined.copy()
                    best_value = new_value
                    last_improvement = iter_count
        
        # Adaptive alpha based on progress
        if iter_count - last_improvement > 50:
            alpha = min(0.5, alpha * 1.1)
        else:
            alpha = max(0.05, alpha * 0.99)
        
        # Check for stagnation
        if iter_count - last_improvement > stagnation_limit:
            stagnation_count += 1
            if stagnation_count > 10:
                # Reset with new random solutions
                population = []
                for _ in range(20):
                    h = np.random.rand(n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    population.append(h)
                fitness = [evaluate_h(h, dx) for h in population]
                best_idx = np.argmin(fitness)
                best_h = population[best_idx].copy()
                best_value = fitness[best_idx]
                last_improvement = iter_count
                stagnation_count = 0
        else:
            stagnation_count = 0
    
    # Final refinement of best solution
    h_refined = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / epsilon
        
        h_refined = h_refined - 0.0005 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_value = np.max(corr) * dx
    
    return (h_refined.tolist(), best_value, n_points)
