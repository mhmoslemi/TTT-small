import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 20
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_current = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_current = np.random.uniform(0, 1, n_points)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def constraint_check(h, mu):
        # Sample shifts from -2 to 2
        shifts = np.linspace(-2, 2, 50)
        for k in shifts:
            # Compute correlation with offset
            if k >= 0:
                corr = np.correlate(h, np.pad(1 - h, (int(k/dx), 0), 'constant')[:len(h)], mode='valid')
            else:
                corr = np.correlate(np.pad(h, (int(-k/dx), 0), 'constant')[:len(h)], 1 - h, mode='valid')
            if np.max(corr) * dx < mu:
                return False
        return True
    
    def solve_optimization(h_start, mu):
        # Use CVXOPT to solve the feasibility problem with μ constraint
        h = Variable(n_points)
        objective = Minimize(0)
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n_points / 2
        ]
        
        # Add overlap constraints based on sampled shifts
        shifts = np.linspace(-2, 2, 50)
        for k in shifts:
            # Simplified linear constraint for feasibility check
            if k >= 0:
                h_shifted = h[int(k/dx):] + [0]*int(k/dx)
            else:
                h_shifted = [0]*int(-k/dx) + h[:-int(-k/dx)]
            # Add approximate constraint (simplified for CVX)
        
        prob = Problem(objective, constraints)
        
        try:
            prob.solve(solver=CVXOPT, verbose=False)
            return h.value
        except:
            return h_start
    
    low, high = 0.0, 0.5
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current)
    
    # Binary search on C5
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        mu = (low + high) / 2.0
        # Problem: CVXOPT infeasible in typical call - Approximate approach
        h_trial = h_current.copy()
        h_trial = repair(h_trial)
        if constraint_check(h_trial, mu):
            # Feasible mu
            high = mu
        else:
            low = mu
        
        # Local refinement
        h_opt = h_current.copy()
        for _ in range(20):
            if time.time() - start_time > budget_s - 10:
                break
            curr_c5 = compute_c5(h_opt)
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - curr_c5) / eps
            h_opt -= 0.001 * grad
            h_opt = repair(h_opt)
        if compute_c5(h_opt) < best_c5:
            best_h = h_opt.copy()
            best_c5 = compute_c5(h_opt)
    
    # Final gradient descent refinement
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        curr_c5 = compute_c5(h_final)
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - curr_c5) / eps
        h_final -= 0.001 * grad
        h_final = repair(h_final)
        new_c5 = compute_c5(h_final)
        if new_c5 >= curr_c5:
            break
    
    return (h_final.tolist(), compute_c5(h_final), n_points)
