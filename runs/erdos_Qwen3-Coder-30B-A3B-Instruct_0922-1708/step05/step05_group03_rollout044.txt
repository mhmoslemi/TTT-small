import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T0 = 10.0
    cooling_rate = 0.9999
    min_T = 1e-6
    
    # Core SA loop
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Perturb solution
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute correlation and C5
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        # Compute current C5
        correlation_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation_current) * dx
        
        # Accept or reject based on SA criterion
        delta = c5_candidate - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate.copy()
            c5_current = c5_candidate
            
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Update temperature
        T = max(min_T, T * cooling_rate)
    
    # Local refinement using gradient descent
    h_ref = best_h.copy()
    lr = 0.001
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h_ref)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        # Update with gradient descent
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        # Compute new C5 and update best if improved
        corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_ref.copy()
    
    # Final evaluation of best solution
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
