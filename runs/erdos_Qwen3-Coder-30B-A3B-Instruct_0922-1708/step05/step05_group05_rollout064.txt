import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize from initial_h_values or use block construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the index of minimum correlation
        min_idx = np.argmin(corr)
        # The actual shift is min_idx - (2*n_points - 1) // 2
        k_shift = min_idx - (2*n_points - 1) // 2
        
        # Compute gradient w.r.t. h
        grad = np.zeros(n_points)
        
        # Gradient for correlation at k_shift
        # d/dh_i [h * (1-h_{i+k_shift})] = 1 - 2*h_i if i is valid, 
        # and the contribution comes from h_{i+k_shift} = h_{i+k_shift}
        # and the cross term contribution from h_{i-k_shift} if it exists
        
        # We look at the correlation contribution differently
        # correlation[i] = sum_j h[j] * (1-h[j+i]) for i from -n+1 to n-1
        # For a fixed shift i, we're interested in the gradients:
        # d/dh[k] (h[j] * (1-h[j+i])) = -h[j+i] if k = j, -h[j-i] if k = j+i, 0 otherwise
        # but the actual correlation at shift is just a scalar sum, 
        # so we approximate gradient of the maximum term.
        
        # We compute: since we're maximizing min of correlation,
        # focus on the place where min occurs
        grad = np.zeros_like(h)
        
        # Compute the full convolution gradient for minimal term
        # Using the fact that the minimal correlation is at k_shift
        # We find positions that contribute to term h_{k_shift} (when j = k_shift).
        # Note that correlation[i] = sum_{j=0}^{n-1} h[j] * (1 - h[j+i-n+1])
        # Which means correlation[k_shift + (n-1)] = sum_{j=0}^{n-1} h[j] * (1 - h[j + (k_shift - n+1)])
        # Simplified: For the max correlation term, we compute the gradient as 
        # of a single fixed term in the correlation that we minimizwe
        
        # For both left and right sides and try to compute influence of each h[i] 
        # on that specific element of correlation 
        # i = 0..n-1 we can only influence correlation index k_shift through
        # term h[i] * (1 - h[i+k_shift])
        # and if we also have h[i] or something similar that interacts with h[j] in a sum
        # but that requires walking the actual convolution 
        
        # Simple way of computing gradient:
        # Consider correlation[k_shift] = sum_j h[j] (1 - h[j - k_shift])
        # Gradient d/dh_i [sum_j h[j] (1 - h[j - k_shift])] = 
        #    if j=i: (1 - h[i - k_shift]) else: (0 if no cross term) 
        # Wait no. Better approach: the derivative of the sum w.r.t. h_i 
        # is (1-h[i-k_shift]) when i-k_shift>=0 and (1-h[i+k_shift]) when i+k_shift<n
        # actually for correlation[k_shift] = sum_{j=0}^{n-1} h[j] * (1 - h[j+k_shift])
        # derivative w.r.t. h[i] = (1 - h[i+k_shift]) (if i+k_shift < n) and 0 otherwise
        
        # Easy idea: let's do proper forward gradient in the variable space
        # In practice, we can use approx derivatives easily but since we still go to max,
        # let's reconsider the gradient computation.
        
        # Let's preserve an alternative, direct gradient estimate
        # We estimate the gradient of the min loss at k_shift (directly)  
        
        for i in range(n_points):
            grad_i = 0.0
            # For all h[j] in the sum correlation[k_shift] = sum_j h[j] * (1 - h[j + k_shift + n - 1])
            # but the actual correlation looks like:
            # correlation at shift k = sum_{j=0}^{n-1} h[j] * (1 - h[j+k]) 
            # Since we have indices going from full convolution, we map back:
            # full convolution gives indices from -(n-1) to (n-1) 
            # Real shift k is from -n+1 to n-1  
            # correlation(k) corresponds to correlation index k + n - 1 in full
            # but (k_shift+n-1)th index in full convolution is h[j] * (1 - h[j-k_shift])  
            # So: at h[i] we account: i = j-k_shift means j = i + k_shift.
            # The impact here assumed that for correlation index k_shift (relative),
            # we're looking at h[j] * (1 - h[j + k_shift]) where j = i + k_shift
            # so that term is affected by h[i]
            # => d/dh[i] = (1 - h[i + k_shift])
            # But some of those h[i + k_shift] may be outside the valid range
            
            # Just do the forward finite diff
            # Forward finite diff approach: estimate gradient directly
            
        # To get a cleaner gradient, we compute the exact gradient of the minimal term
            
        # The contribution of h[i] to correlation[k_shift] = (via the linear term in h[j]) = (1 - h[j+k_shift])
        # when j = i  -- that gives the first term in correlation = h[i] * (1 - h[i + k_shift])
        # So d/dh[i] = (1 - h[i + k_shift]) taking index modulo to ensure it's in range
        # Though this is not perfect but simplest idea for integral gradient updates
    
        # Manually compute the gradient:
        # correlation at shift k is sum_{j=0}^{n-1} h[j] * (1 - h[j+k]) for k=-n+1,...,n-1
        # let's always relate to the fullcross-correlation results to make mapping easier
        
        # Fix: set up full indices as in literature, convolution = full, correlation over -n+1 to n-1
        # actual convolution of 1-H with H: corresponds to indices from -(n-1) to (n-1) 
        # but shift-related is  from -n+1 to n-1, so use proper mapping
        
        # So full_corr_idx = k + (n-1) (goes from 0 to 2(n-1))
        full_corr_idx = k_shift + (n_points - 1)
        
        # For term in this index: correlation[k_shift] = sum_j h[j] (1 - h[j-k_shift])
        # d/dh[i] = (1 - h[i-k_shift]) 
        # From correlation derivative: 
        # at j=i, it's 1 - h[i - k_shift] (etc)
        # if i-k_shift is out of bounds, it gets 0 contribution to that correlation term
        
        # Handle cases when k_shift is positive or negative this compares the real shift to 
        # the offset indices. Just calculate in a direct loop:
        for i in range(n_points):
            j = i - k_shift
            if 0 <= j < n_points:
                grad[i] += (1.0 - h[j]) * dx
                
        # Asymptotic behavior:
        # But simpler: we just scale by dx from finite diff computation:  
        
        # Project into feasible region and update
        h = h - learning_rate * grad
        
        # Clip to bounds and rescale  
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h) 
    
        # Periodically apply perturbation
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration/100)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final validation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
