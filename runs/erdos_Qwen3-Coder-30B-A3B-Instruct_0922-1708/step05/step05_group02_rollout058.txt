import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with either initial_h_values or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        best_h = np.random.rand(n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    best_c5 = evaluate_h(best_h, dx)
    improvement_count = 0
    last_improvement = 0
    
    # Cuckoo Search parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iterations = 1000
    abandon_rate = 0.25
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution via Lévy flight
        new_h = best_h.copy()
        # Randomly choose indices to perturb
        n_perturb = max(1, int(n_points * abandon_rate))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Apply small random perturbations
        for i in indices:
            new_h[i] += np.random.normal(0, 0.01)
        
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            improvement_count += 1
            last_improvement = iteration
        elif np.random.rand() < alpha:
            # Probabilistically accept worse solution
            best_h = new_h.copy()
            best_c5 = new_c5
            improvement_count += 1
            last_improvement = iteration
            
        # Local search with gradient descent on top solutions
        if iteration % 20 == 0 and improvement_count > 0:
            # Refine the best solution with gradient descent
            h_refined = best_h.copy()
            c5_refined = best_c5
            
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - c5_refined) / epsilon
                    
                h_refined = h_refined - 0.001 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                c5_refined = np.max(corr) * dx
                
                if c5_refined < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = c5_refined
                    improvement_count += 1
                    last_improvement = iteration

        # Adaptive step size
        if iteration - last_improvement > 100 and alpha < 0.5:
            alpha *= 1.05  # Increase discovery rate
        elif iteration - last_improvement > 100 and alpha > 0.05:
            alpha *= 0.95  # Reduce discovery rate
            
    return (best_h.tolist(), best_c5, n_points)
