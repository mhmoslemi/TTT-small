import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with 50 points for coarse optimization
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize coarse solution
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse optimization
    best_c5 = float('inf')
    best_h_coarse = h.copy()
    
    # Try solving with CVXPY for a better starting point
    try:
        h_var = cp.Variable(n_start)
        C5_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_start / 2,
        ]
        
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_solution = h_var.value
            if h_solution is not None:
                h = np.array(h_solution)
                h = np.clip(h, 0, 1)
                h = h * (n_start / 2) / np.sum(h)
    except:
        pass
    
    # Gradient based approach for coarse
    max_iter_coarse = 5000
    lr_coarse = 0.01
    patience_coarse = 0
    
    for iter_coarse in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 2:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full') * (2.0 / n_start)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h.copy()
            
        if abs(current_c5 - best_c5) < 1e-8:
            patience_coarse += 1
            if patience_coarse > 100:
                break
        else:
            patience_coarse = 0
            
        # Compute gradient using finite difference
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_start):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
                
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_start)
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update
        h_new = h - lr_coarse * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h = h_new
        lr_coarse = max(1e-6, lr_coarse * 0.999)
        
    # Transfer to fine grid
    if len(best_h_coarse) < n_points:
        x_coarse = np.linspace(0, 2, len(best_h_coarse))
        x_fine = np.linspace(0, 2, n_points)
        h = np.interp(x_fine, x_coarse, best_h_coarse)
        # Re-normalize
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
    else:
        h = best_h_coarse[:n_points]
        
    # Refine with fine grid
    best_h = h.copy()
    best_c5 = float('inf')
    patience_fine = 0
    
    # Adapting learning rate curve with time
    max_iter_fine = 30000
    lr_fine = 0.005
    
    for iter_fine in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - best_c5) < 1e-7:
            patience_fine += 1
            if patience_fine > 300:
                break
        else:
            patience_fine = 0
            
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with clip and re-normalize
        h_new = h - lr_fine * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        lr_fine = max(1e-6, lr_fine * 0.9995)
        
    # Final normalization
    h = best_h.copy()
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Final validation
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint
    assert abs(np.sum(h) * dx - 1.0) < 1e-6, "Sum constraint violated"
    assert all(0 <= x <= 1 for x in h), "Value bounds violated"
    
    return (h.tolist(), final_c5, n_points)
