```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with multiple strategies
    initial_h_list = []
    
    # 1. Use provided parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_h_list.append(h.copy())
    
    # 2. Add random initializations
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_h_list.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Try each initialization
    for h in initial_h_list:
        # Convex relaxation approach to get better starting point
        try:
            h_cvx = cp.Variable(n_points)
            constraints = [
                h_cvx >= 0,
                h_cvx <= 1,
                cp.sum(h_cvx) == n_points / 2
            ]
            
            # Do a convex relaxation step to improve the starting point
            obj = cp.Minimize(cp.sum(cp.multiply(h_cvx, 1 - h_cvx)))
            prob = cp.Problem(obj, constraints)
            prob.solve(solver=ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_init = np.array(h_cvx.value)
                h_init = np.clip(h_init, 0, 1)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                h = h_init
        except Exception:
            pass

        # Refinement with optimizer
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(corr)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        # Use L-BFGS-B for refinement with constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        result = minimize(
            objective,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-6, 'gtol': 1e-6},
            callback=lambda x: None
        )
        
        if result.success:
            refined_h = result.x
            refined_h = np.clip(refined_h, 0, 1)
            refined_h = refined_h * (n_points / 2) / np.sum(refined_h)
            c5_val = objective(refined_h)
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = refined_h.copy()
        
        if time.time() - start_time > budget_s - 1:
            break

    # Final validation
    if best_h is None:
        # Fallback to uniform distribution
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        best_h = h

    # Compute final C5
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    # Ensure constraints
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
```
