import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS
from scipy.ndimage import gaussian_filter1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with parent if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_start:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_start)
            h_init = np.interp(x_new, x_orig, h_init)
    else:
        h_init = np.ones(n_start) * 0.5
    
    # Normalize initial vector
    h_init = h_init * (n_start / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # Coarse grid optimization
    h_coarse = h_init.copy()
    best_c5_coarse = float('inf')
    
    # Define quadratic program for coarse resolution
    h_vars = cp.Variable(n_start)
    C5 = cp.Variable()
    
    constraints = [
        h_vars >= 0,
        h_vars <= 1,
        cp.sum(h_vars) == n_start / 2,
        C5 >= 0
    ]
    
    # Generate all overlap terms for coarse grid
    # Compute cross-correlation terms with 1-h for constraint relaxation
    total_corr = 0
    corr_terms = []
    for k in range(n_start):
        if k == 0:  # identity term
            term = cp.sum(cp.multiply(h_vars, 1 - h_vars))
            corr_terms.append(term)
            total_corr += term
        else:
            # shifted correlation terms
            shifted_idx = np.arange(n_start) - k
            valid_idx = np.logical_and(shifted_idx >= 0, shifted_idx < n_start)
            valid_shift = shifted_idx[valid_idx]
            updated_idx = np.arange(n_start)[valid_idx]
            
            # Calculate h[i] * (1 - h[i+k]) in a batched way
            if len(updated_idx) > 0:
                term = cp.sum(cp.multiply(h_vars[updated_idx], 1 - h_vars[valid_shift]))
                total_corr += term
                corr_terms.append(term)
    
    # Add additional approximate inequality constraints to constrain C5
    # For each shift k, we want max_i h[i] * (1 - h[i+k]) <= C5 * dx
    # This is handles via convex relaxations but we place bounds
    
    prob = cp.Problem(cp.Minimize(C5), constraints)
    
    try:
        prob.solve(solver=ECOS, max_iter=500, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_sol = h_vars.value
            if h_sol is not None:
                h_coarse = np.array(h_sol)
                h_coarse = np.clip(h_coarse, 0, 1)
                h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
                # Evaluate
                correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_start)
                c5_coarse = np.max(correlation)
                best_c5_coarse = c5_coarse
    except Exception:
        pass
    
    # Now refine
    # Use upsampling of coarse
    h_coarse_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), h_coarse)
    h_fine = h_coarse_upsampled.copy()  # basis for fine resolution
        
    # Normalize sum constraint
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Optimization parameters
    max_iter = 20000
    patience = 0
    last_c5 = float('inf')
    best_h = h_fine.copy()
    best_c5 = float('inf')
    
    lr = 0.01
    
    # Objective function for gradient descent
    def obj_and_grad(h_vec):
        h_vec = h_vec.copy()
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        c5 = np.max(correlation)
        
        # Gradient via finite difference with eps
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            grad[i] = (c5_plus - c5) / eps
        
        return c5, grad
    
    # Simulated annealing style approach to escape local minima
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        c5, grad = obj_and_grad(h_fine)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_fine.copy()
            
        if abs(c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
            
        last_c5 = c5
        
        h_fine = h_fine - lr * grad
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        lr = max(1e-6, lr * 0.9994)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure the normalization constraint is satisfied
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Return result
    return (best_h.tolist(), final_c5, n_points)
