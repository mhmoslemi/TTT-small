import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 2000
    mutation_strength = 0.5
    sigma_init = 0.1
    adaptive_factor = 0.999
    restart_interval = 100
    stagnation_limit = 200
    elite_ratio = 0.5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Evaluate initial population
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    fitness_values = [compute_c5(h) for h in population]
    best_idx = np.argmin(fitness_values)
    best_h = population[best_idx].copy()
    best_c5 = fitness_values[best_idx]
    
    # Evolve
    current_iter = 0
    stagnation_count = 0
    mu = mutation_strength
    sigma = sigma_init
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Selection: Elite
        elite_count = int(population_size * elite_ratio)
        sorted_indices = np.argsort(fitness_values)[:elite_count]
        elite_population = [population[i] for i in sorted_indices]
        elite_fitness = [fitness_values[i] for i in sorted_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent = elite_population[np.random.randint(0, len(elite_population))]
            # Mutation
            delta = np.random.normal(0, sigma, n_points)
            child = parent + mu * delta
            # Repair
            child = np.clip(child, 0, 1)
            h_sum = np.sum(child)
            if abs(h_sum - n_points / 2) > 1e-10:
                child = child * (n_points / 2) / h_sum
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = [compute_c5(h) for h in offspring]
        
        # Replacement
        all_h = elite_population + offspring
        all_fitness = elite_fitness + offspring_fitness
        sorted_indices = np.argsort(all_fitness)[:population_size]
        population = [all_h[i] for i in sorted_indices]
        fitness_values = [all_fitness[i] for i in sorted_indices]
        
        # Update best
        new_best_idx = np.argmin(fitness_values)
        if fitness_values[new_best_idx] < best_c5:
            best_c5 = fitness_values[new_best_idx]
            best_h = population[new_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Decay mutation
        mu *= adaptive_factor
        sigma = mu / 5
        
        # Restart if stagnated
        if stagnation_count > stagnation_limit:
            if 'initial_h_values' in globals():
                h_init = np.array(initial_h_values)
                if len(h_init) != n_points:
                    x_orig = np.linspace(0, 2, len(h_init))
                    x_target = np.linspace(0, 2, n_points)
                    h_init = np.interp(x_target, x_orig, h_init)
                h_init = h_init * (n_points / 2) / np.sum(h_init)
                population = [h_init]
                for _ in range(population_size - 1):
                    h = h_init + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
                    population.append(h)
            else:
                for i in range(population_size):
                    h = np.random.uniform(0, 1, n_points)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
                    population[i] = h
            mu = mutation_strength
            sigma = sigma_init
            stagnation_count = 0
        
        # Restart every 100 iterations
        if current_iter % restart_interval == 0 and current_iter > 0:
            mu *= 0.9
        
        current_iter += 1
    
    # Final refinement
    def refine_solution(h):
        h_ref = h.copy()
        for _ in range(100):
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_sum = np.sum(h_plus)
                if abs(h_sum - n_points / 2) > 1e-10:
                    h_plus = h_plus * (n_points / 2) / h_sum
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_sum = np.sum(h_ref)
            if abs(h_sum - n_points / 2) > 1e-10:
                h_ref = h_ref * (n_points / 2) / h_sum
            
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        return h_ref, fitness_new
    
    h_refined, c5_refined = refine_solution(best_h)
    if c5_refined < best_c5:
        best_h = h_refined
        best_c5 = c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
