import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.8):
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        gamma = (1 + 2*alpha) * np.random.random() - alpha
        child1[i] = (1 - gamma) * parent1[i] + gamma * parent2[i]
        child2[i] = (1 - gamma) * parent2[i] + gamma * parent1[i]
    return child1, child2

def mutate(individual, mutation_rate=0.1, noise_magnitude=0.03):
    for i in range(len(individual)):
        if np.random.random() < mutation_rate:
            individual[i] += np.random.normal(0, noise_magnitude)
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2.0) / np.sum(h_parent)
        for _ in range(population_size // 2):
            h = h_parent + np.random.normal(0, 0.02, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA parameters
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Main loop
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        new_population = [best_h.copy()]
        fitnesses = [compute_fitness(best_h, dx)]
        
        # Selection, crossover, mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            if np.random.random() < crossover_rate:
                child1, child2 = blend_crossover(parent1, parent2, alpha=0.8)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate, 0.03)
            child2 = mutate(child2, mutation_rate, 0.03)
            
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Periodic local refinement (every 50 generations)
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for idx in top_indices:
                h = population[idx].copy()
                # Simple gradient descent
                for _ in range(50):
                    grad = np.zeros_like(h)
                    eps = 1e-4
                    for i in range(n_points):
                        h_plus = h.copy()
                        h_plus[i] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        c5_plus = compute_fitness(h_plus, dx)
                        grad[i] = (c5_plus - fitnesses[idx]) / eps
                    h = h - 0.001 * grad
                    h = project_to_feasible(h, n_points)
                    new_c5 = compute_fitness(h, dx)
                    if new_c5 < fitnesses[idx]:
                        fitnesses[idx] = new_c5
                        population[idx] = h.copy()
                        if new_c5 < best_c5:
                            best_c5 = new_c5
                            best_h = h.copy()
                            
    # Final refinement passes
    for _ in range(5):
        h_ref = best_h.copy()
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = compute_fitness(h_plus, dx)
                grad[i] = (c5_plus - best_c5) / eps
            h_ref = h_ref - 0.0005 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            c5 = compute_fitness(h_ref, dx)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
