import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Add random particles
    for _ in range(99):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Personal bests and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * 100
    
    # Compute initial fitness for all particles
    for i in range(100):
        correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(correlation) * dx
        pbest_fitness[i] = fitness
        if i == 0 or fitness < pbest_fitness[0]:
            gbest_idx = i
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    max_iter = 500
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(100)]
    
    current_iter = 0
    stagnation_count = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        for i in range(100):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest_particles[i] - particles[i]) +
                            c2 * r2 * (gbest_particle - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = repair(particles[i])
            
            # Fitness computation
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(correlation) * dx
            
            # Update pbest
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_particles[i] = particles[i].copy()
                
                # Update gbest
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[i].copy()
                    stagnation_count = 0
        
        # Stagnation detection and rejuvenation
        stagnation_count += 1
        if stagnation_count > 50:
            for i in range(10):
                idx = np.random.randint(0, 100)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
            
        current_iter += 1
    
    # Local refinement using L-BFGS-B
    def obj_func(h):
        h = h.reshape(-1)
        h = repair(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def grad_func(h):
        h = h.reshape(-1)
        h = repair(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        
        # Gradient approximation
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - np.max(corr) * dx) / eps
            
        return -grad.flatten()
    
    # Refine the best solution
    refined_h = gbest_particle.copy()
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        result = minimize(obj_func, refined_h, method='L-BFGS-B', jac=grad_func, bounds=bounds, options={'maxiter': 100, 'gtol': 1e-6})
        refined_h = result.x.reshape(-1)
        refined_h = repair(refined_h)
    except:
        pass
    
    # Final computation
    final_corr = np.correlate(refined_h, 1 - refined_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (refined_h.tolist(), final_c5, n_points)
