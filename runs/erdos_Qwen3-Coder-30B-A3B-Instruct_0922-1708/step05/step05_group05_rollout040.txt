import numpy as np
from scipy.stats import halton
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population using Halton sequence
    def initialize_population():
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population = [h_init.copy()]
        else:
            population = []
        
        if len(population) < pop_size:
            # Fill with Halton-generated individuals
            seq = halton.Halton(1)
            for i in range(pop_size - len(population)):
                h = np.zeros(n_points)
                for j in range(n_points):
                    h[j] = seq.random(1)[0]
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                if h_sum > 0:
                    h = h * (n_points / 2) / h_sum
                population.append(h)
        
        return population
    
    # Repair individual to meet constraints
    def repair(individual):
        individual = np.clip(individual, 0, 1)
        current_sum = np.sum(individual)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-6:
            diff = current_sum - target_sum
            
            if diff > 0:
                # Reduce values that are too high
                sorted_indices = np.argsort(individual)[::-1]
                for idx in sorted_indices:
                    if diff <= 0:
                        break
                    available = individual[idx]
                    reduce = min(diff, available)
                    individual[idx] -= reduce
                    diff -= reduce
            else:
                # Increase values that are too low
                sorted_indices = np.argsort(individual)
                for idx in sorted_indices:
                    if diff >= 0:
                        break
                    available = 1.0 - individual[idx]
                    increase = min(-diff, available)
                    individual[idx] += increase
                    diff += increase
            
            # Ensure normalization to exact sum
            individual = individual * target_sum / np.sum(individual)
        
        return individual
    
    # Evaluate fitness of an individual
    def evaluate_fitness(individual):
        corr = np.correlate(individual, 1 - individual, mode="full")
        return np.max(corr) * dx
    
    # Tournament selection
    def tournament_select(population, fitnesses):
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx])
        return selected
    
    # Single point crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    # Gaussian mutation
    def mutate(individual, rate):
        if np.random.rand() > rate:
            return individual
        
        mutant = individual.copy()
        num_mutations = np.random.poisson(1)
        for _ in range(num_mutations):
            idx = np.random.randint(n_points)
            delta = np.random.normal(0, 0.02)
            mutant[idx] += delta
        return repair(mutant)
    
    # Initialize
    population = initialize_population()
    fitnesses = [evaluate_fitness(ind) for ind in population]
    best_idx = np.argmin(fitnesses)
    best_individual = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Generation loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Selection
        selected = tournament_select(population, fitnesses)
        
        # Create offspring
        offspring = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[min(i+1, pop_size-1)]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            offspring.extend([child1, child2])
        
        # Replace worst with offspring and repair
        offsprings = offspring[:pop_size]
        for i, child in enumerate(offsprings):
            offsprings[i] = repair(child)
        
        # Combine with elite
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elite = [population[i] for i in elite_indices]
        combined = elite + offsprings[elite_size:]
        
        # Evaluate new population
        fitnesses = [evaluate_fitness(ind) for ind in combined]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = combined[min_fitness_idx].copy()
            
        population = combined
        
        # Adaptive mutation rate (decay)
        if gen % 10 == 0:
            mutation_rate *= 0.99
            mutation_rate = max(0.01, mutation_rate)
    
    # Final validation
    best_individual = repair(best_individual)
    final_fitness = evaluate_fitness(best_individual)
    
    return (best_individual.tolist(), final_fitness, n_points)
