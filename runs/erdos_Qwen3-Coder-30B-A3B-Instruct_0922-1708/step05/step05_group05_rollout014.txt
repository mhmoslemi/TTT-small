import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
   learning_rate = 0.001
    perturb_freq = 50
    perturb_mag = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = 0.5
    h = h * (0.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        min_idx = np.argmin(corr)
        min_val = corr[min_idx]
        
        # Compute gradient (approximate)
        grad = np.zeros(n_points)
        # For the minimal correlation index, we estimate gradient
        # This is simplified: we just compute finite differences
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (0.5 / np.sum(h_plus))  # Maintain constraint
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (corr_plus[min_idx] - min_val) / eps
        
        # Update h with gradient ascent (we want to maximize min correlation)
        h = h + learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.sum(h))  # Maintain sum constraint
        
        # Check if this is an improvement
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Apply random perturbation periodically
        if iteration % perturb_freq == 0:
            noise = np.random.normal(0, perturb_mag * (1.0 / (1.0 + iteration/100)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.sum(h))
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
