import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    lr = initial_learning_rate
    patience_counter = 0
    prev_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Plateau detection and learning rate adjustment
        if patience_counter > plateau_patience:
            lr *= learning_rate_decay
            lr = max(lr, min_learning_rate)
            patience_counter = 0
            
            # Perturb h slightly if plateaued
            noise = np.random.randn(n_points) * perturb_fraction
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Gradient computation for max lag
        max_lag_idx = np.argmax(corr_full)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradients efficiently
        # For correlation at lag k: (h ⋆ (1-h))[k] = sum_{i} h[i] * (1-h[i+k])
        # Gradient w.r.t. h[j]: (1-h[j+k]) if k >= 0, and -h[j-k] if k < 0
        
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # Forward case: h[j] contributes to (1-h[j+lag]) term
            for j in range(n_points - max_lag):
                grad[j] += 1 - h[j + max_lag]
            # Add contribution from h[j+lag] term
            for j in range(max_lag, n_points):
                grad[j] -= h[j - max_lag]
        else:
            # Backward case, but I check indices carefully
            lag = -max_lag  # lag now positive
            for j in range(lag, n_points):
                grad[j] += 1 - h[j - lag]
            for j in range(lag):
                grad[j] -= h[j + lag]
        
        # Update step with learning rate
        h = h - lr * grad
        
        # Repair: bound and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        if c5 < prev_c5:
            prev_c5 = c5
        else:
            # Slight irregularity in improvement, could also do early break
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
