import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Fourier basis size
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Fourier coefficients initialization (real and imaginary parts)
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Get Fourier coefficients
    def get_fourier_coeffs(h):
        # We'll use a real Fourier basis for simplicity
        # h = sum_{k=0}^{n_fourier-1} (a_k * cos(k*x) + b_k * sin(k*x))
        # But let's do rectangular approximation into fourier series
        
        # Use DFT on the sampled data to get initial coeffs
        # But we'll work with real coeffs directly
        fft_coeffs = np.fft.rfft(h)
        n_coeffs = len(fft_coeffs)
        real_coeffs = np.real(fft_coeffs[:n_coeffs//2+1])
        imag_coeffs = np.imag(fft_coeffs[:n_coeffs//2+1])
        return np.concatenate([real_coeffs, imag_coeffs])
    
    def from_fourier_coeffs(coeffs):
        # reconstruct h from fourier coeffs
        real_coeffs = coeffs[:n_fourier+1] 
        imag_coeffs = coeffs[n_fourier+1:]
        
        # Construct fft frequency components
        fft_coeffs = np.zeros(n_points, dtype=complex)
        fft_coeffs[0] = real_coeffs[0]
        if len(real_coeffs) > 1:
            fft_coeffs[1:n_fourier+1] = real_coeffs[1:] + 1j * imag_coeffs[:n_fourier]
            # conjugate for real-valued output
            fft_coeffs[-n_fourier+1:] = real_coeffs[1:][::-1] - 1j * imag_coeffs[:n_fourier][::-1]
        
        # inverse FFT
        h = np.fft.irfft(fft_coeffs, n_points)
        return np.maximum(0, np.minimum(1, h))
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Get initial point
    initial_coeffs = get_fourier_coeffs(h_init)
    
    # Optimization
    best_coeffs = initial_coeffs.copy()
    best_c5 = compute_c5(from_fourier_coeffs(initial_coeffs))
    
    # Stopping criteria
    stagnation_count = 0
    max_stagnation = 100
    
    # Gradient descent
    def opt_step(coeffs):
        h = from_fourier_coeffs(coeffs)
        c5 = compute_c5(h)
        
        # Compute gradient using finite differences
        grad = np.zeros_like(coeffs)
        eps = 1e-4
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            h_plus = from_fourier_coeffs(coeffs_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
        
        # Update with learning rate
        new_coeffs = coeffs - 0.01 * grad
        return new_coeffs, c5
    
    # Optimization loop
    current_iter = 0
    while current_iter < 1000 and time.time() - start_time < budget_s - 10:
        new_coeffs, old_c5 = opt_step(best_coeffs)
        new_h = from_fourier_coeffs(new_coeffs)
        new_c5 = compute_c5(new_h) 
        
        if new_c5 < best_c5:
            best_coeffs = new_coeffs.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        if stagnation_count > max_stagnation:
            break
            
        current_iter += 1
    
    final_h = from_fourier_coeffs(best_coeffs)
    
    # Final validity check and adjustment
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
