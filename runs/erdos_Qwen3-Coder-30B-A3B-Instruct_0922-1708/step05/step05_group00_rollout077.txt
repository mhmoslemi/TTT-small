import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_fine = 200
    dx_coarse = 2.0 / n_start
    dx_fine = 2.0 / n_fine
    max_iter_coarse = 2000
    max_iter_fine = 2000
    patience_limit = 200
    
    # Initialize h with parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to sum = n_start / 2
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    last_c5 = float('inf')
    patience = 0
    
    # Coarse grid optimization
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx_coarse
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-4
        
        for i in range(n_start):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps_fd
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h = h_new
        
    # Transfer to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_start), best_h)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization
    best_h = h_fine.copy()
    best_c5 = float('inf')
    last_c5 = float('inf')
    patience = 0
    lr = 0.01
    
    # Reinitialize Adam moments for fine grid
    m = np.zeros_like(h_fine)
    v = np.zeros_like(h_fine)
    
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_fine
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h_fine)
        eps_fd = 1e-4
        
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps_fd
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h_fine - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_fine / 2) / np.sum(h_new)
        
        h_fine = h_new
        
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx_fine
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_fine)
