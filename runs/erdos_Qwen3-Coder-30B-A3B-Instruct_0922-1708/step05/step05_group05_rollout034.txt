import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Simple block construction as fallback
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to target sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # GLM-based approach with adaptive learning rate
    max_iter = 1000
    learning_rate = 0.01
    decay_factor = 0.995
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current correlation and max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        # Find which index contributes most to max
        max_idx = np.argmax(base_corr)
        max_corr = base_corr[max_idx]
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic random perturbations to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            perturb_magnitude *= 0.9
            for i in range(n_points):
                if random.random() < 0.2:  # 20% chance of perturbation
                    h[i] += random.uniform(-perturb_magnitude, perturb_magnitude)
            
            # Re-project
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Adaptive learning rate
        learning_rate *= decay_factor
    
    # Final validation and recomputation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
