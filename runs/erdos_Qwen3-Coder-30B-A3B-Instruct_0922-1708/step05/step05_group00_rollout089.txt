import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-start approach
    starts = []
    starts.append(h.copy())
    
    # Add 5 random initializations
    for _ in range(5):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        starts.append(h_rand.copy())
    
    # Refinement with L-BFGS-B
    max_refine_iter = 2000
    patience_limit = 100
    
    for i, h_start in enumerate(starts):
        h_current = h_start.copy()
        
        def objective(h_vec):
            if time.time() - start_time > budget_s - 1:
                return best_c5
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(correlation)
            
        def constraint_func(h_vec):
            return np.sum(h_vec) - n_points / 2
            
        cons = {'type': 'eq', 'fun': constraint_func}
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Initial evaluation
        current_c5 = objective(h_current)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
            
        # Optimization with bounds
        try:
            res = minimize(
                objective,
                h_current,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=cons,
                options={'maxiter': max_refine_iter},
                callback=lambda x: None
            )
            
            if res.success:
                final_h = np.clip(res.x, 0, 1)
                final_h = final_h * (n_points / 2) / np.sum(final_h)
                correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
                final_c5 = np.max(correlation)
                
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = final_h.copy()
                    
        except Exception:
            pass
            
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Verify sum
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
