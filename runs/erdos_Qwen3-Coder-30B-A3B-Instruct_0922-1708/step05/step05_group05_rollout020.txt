import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_region_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    refinement_iterations = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Block construction as default
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find dominant lag
        max_k = np.argmax(corr)
        
        # Compute gradient
        # Gradient with respect to h_i is the sum of 1-h_{i-k} and 1-h_{i+k} terms
        # Actually, for overlap_k = sum h_i * (1-h_{i+k}) 
        # d/dh_i overlap_k = 1 - h_{i+k} when i corresponds to the contribution
        # Reconstructing the gradient by analytical derivation:
        # For each position i and lag k, we look at the term h_i * (1 - h_{i+k})
        # Grad_i exists only if terms exist in the convolution, we compute it for dominant k
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Add contribution from h_i * (1 - h_{i+k}) term  
            j = (i + max_k) % n_points
            grad[i] += (1 - h[j]) * dx  # Forward contribution
            
            # Add contribution from h_{i-k} * (1 - h_i) term  
            j = (i - max_k) % n_points
            grad[i] += (1 - h[j]) * dx  # Backward contribution
            
        # Propose step, safer than a full gradient step
        proposed_h = h - trust_region_radius * grad
        
        # Enforce constraints via feasible projection
        proposed_h = np.clip(proposed_h, 0, 1)
        sum_diff = np.sum(proposed_h) - (n_points / 2)
        if abs(sum_diff) > 1e-6:
            # Adjust proportionally
            scale_factor = (n_points / 2) / np.sum(proposed_h)
            proposed_h = proposed_h * scale_factor
        
        # Check feasibility
        new_corr = np.correlate(proposed_h, 1 - proposed_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        if new_c5 < current_c5:
            h = proposed_h.copy()
            current_c5 = new_c5
            trust_region_radius = min(1.0, trust_region_radius * 1.2)
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            trust_region_radius = max(min_trust_radius, trust_region_radius * 0.8)
        
        # Apply perturbation with simulated annealing
        if iteration % perturbation_freq == 0 and iteration > 0:
            perturbation_magnitude = temperature * (1 - h) * h
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% chance of perturbation
                    delta = np.random.normal(0, perturbation_magnitude[i])
                    h[i] += delta
                    h[i] = np.clip(h[i], 0, 1)
            
            # Preserve sum
            h = h * (n_points / 2) / np.sum(h)
            
            # Re-evaluate
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5

        temperature *= cooling_rate
        if temperature < 1e-6:
            temperature = 1e-6
    
    # Intensive local refinement of best solution
    h = best_h.copy()
    
    for _ in range(refinement_iterations):
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        improved = False
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - base_fitness) / eps
            
            if fitness_plus < base_fitness:
                improved = True
                
        if not improved:
            break
            
        # Update 
        h -= 0.001 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        corr = np.correlate(h, 1 - h, mode='full')
        new_fitness = np.max(corr) * dx
        
        if new_fitness < base_fitness:
            base_fitness = new_fitness
        else:
            break
    
    # Final check
    corr_final = np.correlate(h, 1 - h, mode='full')
    final_fitness = np.max(corr_final) * dx
    
    return (h.tolist(), final_fitness, n_points)
