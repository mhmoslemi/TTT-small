import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize multiple starting points
    initial_solutions = []
    
    # Use the parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        initial_solutions.append(h_parent.copy())
    else:
        initial_solutions.append(np.ones(n_points) * 0.5)
    
    # Add random initializations
    for _ in range(4):
        h_rand = np.random.rand(n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_solutions.append(h_rand)
    
    best_h = None
    best_c5 = float('inf')
    
    # For each initialization, do convex relaxation + refinement
    for h_start in initial_solutions:
        h = h_start.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Convex relaxation step using a quadratic approximation
        # For simplicity, we'll do a simple feasibility + gradient method
        
        # Keep track of the best solution found in this run
        current_best_h = h.copy()
        current_best_c5 = float('inf')
        
        # Perform limited refinement with L-BFGS
        max_iter = 1500
        patience = 0
        last_c5 = float('inf')
        lr = 0.005
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current overlap
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            current_c5 = np.max(correlation)
            
            # Update best
            if current_c5 < current_best_c5:
                current_best_c5 = current_c5
                current_best_h = h.copy()
                
            # Check for convergence
            if abs(current_c5 - last_c5) < 1e-6:
                patience += 1
                if patience > 100:
                    break
            else:
                patience = 0
                
            last_c5 = current_c5
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                c5_plus = np.max(cor_plus)
                
                grad[i] = (c5_plus - current_c5) / eps
                
            # Update step with learning rate decrease
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            lr = max(1e-6, lr * 0.999)
            
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
    
    # Final refinement of best solution
    if best_h is not None:
        h = best_h.copy()
        
        # Do one final numerical optimization using scipy
        def c5_obj(h):
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(correlation)
        
        # Use SLSQP for constrained optimization
        bounds = [(0, 1) for _ in range(n_points)]
        # Constraint for sum
        def constraint_sum(h):
            return np.sum(h) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            res = minimize(c5_obj, h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 500, 'ftol': 1e-9})
            if res.success:
                h = res.x
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                corr = np.correlate(h, 1 - h, mode='full') * dx
                best_c5 = np.max(corr)
        except:
            # If optimization failed, stick with current best
            pass
    
    # Final verification
    if best_h is not None:
        final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = np.max(final_corr)
        best_h = best_h * (n_points / 2) / np.sum(best_h)  # Ensure proper sum
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Return a fallback if nothing found
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        corr_fallback = np.correlate(h_fallback, 1 - h_fallback, mode='full') * dx
        c5_fallback = np.max(corr_fallback)
        return (h_fallback.tolist(), c5_fallback, n_points)
