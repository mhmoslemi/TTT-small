import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
    else:
        n_points = 200
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5
    
    # Normalize initial solution to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse-to-fine optimization
    n_coarse = min(50, n_points // 2)
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse optimization
    if n_coarse > 10:
        try:
            # Quadratic Programming approach on coarse grid
            h_coarse_var = cp.Variable(n_coarse)
            C5 = cp.Variable()
            
            # Constraints  
            constraints = [
                h_coarse_var >= 0,
                h_coarse_var <= 1,
                cp.sum(h_coarse_var) == n_coarse / 2,
                C5 >= 0
            ]
            
            # Add quadratic regularization and convex relaxation
            # Simplified but robust approach
            prob = cp.Problem(cp.Minimize(C5), constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_coarse_solution = h_coarse_var.value
                if h_coarse_solution is not None:
                    h_coarse = np.array(h_coarse_solution)
                    h_coarse = np.clip(h_coarse, 0, 1)
                    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
                    
                    # Upsample to fine grid
                    h_upsampled = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
                    h = np.clip(h_upsampled, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass
    
    # Fine optimization with local search
    h_refined = h.copy()
    
    # Calculate overlap function efficiently
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(correlation)
    
    # Gradient-driven optimization with momentum
    lr = 0.005
    momentum = 0.9
    velocity = np.zeros_like(h_refined)
    last_c5 = float('inf')
    patience = 0
    max_patience = 300
    
    # Early stopping in case of no improvement
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h_refined)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_refined.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > max_patience:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with momentum
        velocity = momentum * velocity - lr * grad
        h_new = h_refined + velocity
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_refined = h_new
        
        # Anneal learning rate
        lr = max(1e-6, lr * 0.9995)
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
