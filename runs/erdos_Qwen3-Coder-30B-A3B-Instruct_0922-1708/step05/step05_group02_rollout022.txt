import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    start_time = time.time()

        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a random feasible solution
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Hyperparameters
    learning_rate = 0.001
    perturb_rate = 0.1
    max_iter = 1000
    patience = 100
    
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update h
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Evaluate new solution
        c5 = evaluate_h(h, dx)
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Perturb a fraction of elements
        if np.random.rand() < perturb_rate or stagnation_count > 20:
            # Perturb 10% of the elements
            indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            for i in indices:
                h[i] += np.random.normal(0, 0.01)
                h[i] = np.clip(h[i], 0, 1)
            
            # Repair and normalize
            h = repair_h(h, n_points)
            
        # Sometimes, restart from best solution with slight perturbation
        if np.random.rand() < 0.05:
            h = best_h.copy() + np.random.normal(0, 0.001, n_points)
            h = repair_h(h, n_points)

    for _ in range(100):
