import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    adaptive_lr_decay = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute overlap for all lags
        corr = np.correlate(h, 1-h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Compute gradient w.r.t maximum correlation
        max_idx = np.argmax(corr)
        # Approximate gradient by finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - c5) / eps
        
        # Update using gradient descent
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbations
        if iteration % perturb_freq == 0:
            perturb_magnitude *= 0.95  # Decay perturbation
            # Apply small random perturbations
            noise = np.random.normal(0, perturb_magnitude, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        learning_rate *= adaptive_lr_decay
    
    # Final validation
    final_corr = np.correlate(best_h, 1-best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
