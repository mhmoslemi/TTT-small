import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Scale pyramid
    scales = [10, 20, 40, 80, 100]
    
    for i, n in enumerate(scales):
        if time.time() - start_time > budget_s - 50:
            break
            
        # Create new h for this scale
        if i == 0:
            h_new = np.ones(n) * 0.5
        else:
            # Interpolate from previous scale
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n)
            h_new = np.interp(x_new, x_old, h)
            
        # Relaxation at this scale
        h_var = Variable(n)
        # Soft constraint penalty
        lambda_pen = 0.1
        
        # Objective: average overlap over central lags
        # Compute lags for central region
        lag_range = min(n // 4, n - 1)
        lags = list(range(1, lag_range + 1))
        
        # Construct CVX problem
        objective = Minimize(0)  # placeholder to replace
        constraints = [h_var >= 0, h_var <= 1]
        
        # Approximate C5 using average of limited lags
        avg_overlaps = []
        for k in lags:
            corr_k = 0
            for j in range(n - k):
                corr_k += h_var[j] * (1 - h_var[j + k])
            avg_overlaps.append(corr_k * dx / (n - k))
        
        # Minimize average overlap
        objective = Minimize(sum(avg_overlaps))
        # Add soft constraints
        obj = objective + lambda_pen * sum(h_var) - (n / 2) * lambda_pen
        constraints += [sum(h_var) == n / 2]
        
        # Solve with ECOS
        prob = Problem(obj, constraints)
        try:
            prob.solve(solver=ECOS, verbose=False)
            if prob.status == 'optimal':
                h_sol = np.array(h_var.value).flatten()
                # Fix constraints
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n / 2) / np.sum(h_sol)
                
                # Evaluate actual C5; test if better
                corr = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5_eval = np.max(corr) * dx
                
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    if n == n_points:
                        best_h = h_sol.copy()
                    
                    # Refine with local gradient descent
                    if n == n_points:
                        h_ref = h_sol.copy()
                        for _ in range(50):
                            if time.time() - start_time > budget_s - 10:
                                break
                            grad = np.zeros_like(h_ref)
                            eps = 1e-4
                            for j in range(n):
                                h_plus = h_ref.copy()
                                h_plus[j] += eps
                                h_plus = np.clip(h_plus, 0, 1)
                                h_plus = h_plus * (n / 2) / np.sum(h_plus)
                                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                                c5_plus = np.max(corr_plus) * dx
                                grad[j] = (c5_plus - c5_eval) / eps
                            h_ref = h_ref - 0.001 * grad
                            h_ref = np.clip(h_ref, 0, 1)
                            h_ref = h_ref * (n / 2) / np.sum(h_ref)
                            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                            c5_new = np.max(corr_new) * dx
                            if c5_new < best_c5:
                                best_c5 = c5_new
                                best_h = h_ref.copy()
        except:
            pass
        
        # Update h for next scale
        h = h_new
    
    # Final evaluation
    if 'best_h' not in locals():
        h_final = best_h
    else:
        h_final = best_h.copy()
    
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    # Final check
    if final_c5 > best_c5:
        corr_final = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
