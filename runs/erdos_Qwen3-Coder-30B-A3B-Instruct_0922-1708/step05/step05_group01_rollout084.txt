import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or create random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Scale to ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iter = 5000
    lr = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    perturb_fraction = 0.01
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find the index of the maximum correlation
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        if max_lag > 0:
            # Forward lag: h[i] * (1 - h[i + max_lag])
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
            for i in range(max_lag, n_points):
                grad[i] -= h[i - max_lag]
        elif max_lag < 0:
            # Backward lag: h[i] * (1 - h[i - max_lag])
            for i in range(-max_lag, n_points):
                grad[i] += 1 - h[i + max_lag]
            for i in range(0, n_points + max_lag):
                grad[i] -= h[i - max_lag]
        else: 
            # Zero lag: h[i] * (1 - h[i]) - we handle this as special case
            grad[:] = 1 - 2 * h
            
        # Step size line search
        alpha = lr
        prev_c5 = c5
        while True:
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new_sum = np.sum(h_new)
            h_new = h_new * (n_points / 2) / h_new_sum
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < prev_c5:
                h = h_new
                c5 = c5_new
                break
            else:
                alpha *= line_search_factor
                
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Random perturbation to escape local minimum
        if iteration % 100 == 0:
            perturb = np.random.rand(n_points) * perturb_fraction * 2 - perturb_fraction
            h = h + perturb
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
