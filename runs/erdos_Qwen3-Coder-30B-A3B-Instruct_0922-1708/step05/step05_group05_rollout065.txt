import numpy as np
from scipy.stats import halton
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population using Halton sequences or initial construction
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Use Halton-based initialization
        seq = halton.Halton(1)
        samples = seq.random(population_size - 1)
        for i in range(population_size - 1):
            h = np.zeros(n_points)
            indices = np.floor(samples[i] * n_points).astype(int)
            indices = np.clip(indices, 0, n_points-1)
            for idx in indices:
                h[idx] = 1.0
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    # Add one random solution to start with
    h_random = np.random.rand(n_points)
    h_random = h_random * (n_points / 2) / np.sum(h_random)
    population.append(h_random)
    
    best_individual = None
    best_fitness = float('inf')
    
    # Genetic algorithm main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
            
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_select(population, fitnesses, tournament_size)
            parent2 = tournament_select(population, fitnesses, tournament_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate, n_points, dx)
            child2 = mutate(child2, mutation_rate, n_points, dx)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Adapt mutation rate
        if gen % 10 == 0:
            mutation_rate *= 0.99
    
    # Final refinement using local search
    refined_individual = best_individual.copy()
    best_fitness = evaluate_h(refined_individual, n_points, dx)
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient-based refinement
        grad = estimate_gradient(refined_individual, n_points, dx)
        step_size = 0.001
        refined_individual = refined_individual - step_size * grad
        refined_individual = np.clip(refined_individual, 0, 1)
        
        # Repair sum constraint
        current_sum = np.sum(refined_individual)
        target_sum = n_points / 2
        refined_individual = refined_individual * target_sum / current_sum
        
        new_fitness = evaluate_h(refined_individual, n_points, dx)
        if new_fitness < best_fitness:
            best_fitness = new_fitness
        else:
            break
    
    return (refined_individual.tolist(), best_fitness, n_points)

def tournament_select(population, fitnesses, tournament_size):
    """Select an individual using tournament selection"""
    k = min(tournament_size, len(population))
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2):
    """Single-point crossover"""
    crossover_point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(individual, mutation_rate, n_points, dx):
    """Gaussian mutation with constraint repair"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    
    # Repair bounds
    mutated = np.clip(mutated, 0, 1)
    
    # Repair sum constraint
    current_sum = np.sum(mutated)
    target_sum = n_points / 2
    mutated = mutated * target_sum / current_sum
    
    return mutated

def evaluate_h(h, n_points, dx):
    """Evaluate the fitness of a solution"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def estimate_gradient(h, n_points, dx):
    """Estimate gradient using finite differences"""
    grad = np.zeros_like(h)
    eps = 1e-5
    base_fitness = evaluate_h(h, n_points, dx)
    
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = np.clip(h_plus, 0, 1)
        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
        
        fitness_plus = evaluate_h(h_plus, n_points, dx)
        
        grad[i] = (fitness_plus - base_fitness) / eps
    
    return grad
