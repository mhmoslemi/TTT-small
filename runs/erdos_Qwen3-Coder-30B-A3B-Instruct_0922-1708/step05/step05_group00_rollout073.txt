import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        # Initialize with a structured approach: step function
        h = np.ones(n_points) * 0.5
        # Perturb a bit to avoid symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        
    dx = 2.0 / n_points
    
    # Normalize to sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Smooth maximum parameter
    beta = 10.0
    eps = 1e-5
    
    # Adaptor for learning rate
    def cosine_anneal(step, total_steps, min_lr=1e-5, max_lr=0.01):
        return min_lr + (max_lr - min_lr) * (1 + math.cos(math.pi * step / total_steps)) / 2
        
    # Warm-up optimization with coarse grid
    coarse_n = 50
    coarse_dx = 2.0 / coarse_n
    coarse_h = np.interp(np.linspace(0, 2, coarse_n), np.linspace(0, 2, n_points), h)
    coarse_h = np.clip(coarse_h, 0, 1)
    coarse_h = coarse_h * (coarse_n / 2) / np.sum(coarse_h)
    tmp_h = coarse_h.copy()
    
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute smooth max correlation
        corr = correlate(tmp_h, 1 - tmp_h, mode='full')
        smooth_max = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))
        current_c5 = smooth_max * coarse_dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(tmp_h)
        for i in range(len(tmp_h)):
            h_plus = tmp_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (coarse_n / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            smooth_max_plus = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr_plus)))
            c5_plus = smooth_max_plus * coarse_dx
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with adaptive learning rate
        lr = cosine_anneal(step, 2000)
        tmp_h = tmp_h - lr * grad
        
        # Project to feasible region
        tmp_h = np.clip(tmp_h, 0, 1)
        tmp_h = tmp_h * (coarse_n / 2) / np.sum(tmp_h)
        
        # Check if better solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            # Interpolate back to fine grid
            coarse_to_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_n), tmp_h)
            coarse_to_fine = np.clip(coarse_to_fine, 0, 1)
            best_h = coarse_to_fine * (n_points / 2) / np.sum(coarse_to_fine)
    
    # Transfer coarse solution to fine grid
    fine_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_n), tmp_h)
    fine_h = np.clip(fine_h, 0, 1)
    fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
    h = fine_h.copy()
    
    # Fine grid optimization
    patience = 0
    last_c5 = float('inf')
    
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute smooth max correlation
        corr = correlate(h, 1 - h, mode='full')
        smooth_max = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr)))
        current_c5 = smooth_max * dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            smooth_max_plus = (1.0 / beta) * np.log(np.sum(np.exp(beta * corr_plus)))
            c5_plus = smooth_max_plus * dx
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with adaptive learning rate
        lr = cosine_anneal(step, 2000)
        h = h - lr * grad

        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping check
        if abs(current_c5 - last_c5) < 1e-8:
            patience += 1
            if patience > 1000:
                break
        else:
            patience = 0
        last_c5 = current_c5
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final verification
    final_corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    
    # Ensure sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
