import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)

            # Downsample if too large
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 100)
            h = np.interp(x_new, x_orig, h)
            n_points = 100
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint sum(h)*dx = 1 => sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    def smooth_max(x, beta=10.0):
        """Smooth approximation of max using log-sum-exp"""
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    def compute_overlap(h_vec):
        """Compute all overlaps between h and (1-h)"""
        corr = correlate(h_vec, 1 - h_vec, mode='full') * dx
        return corr
    
    def compute_c5(h_vec):
        """Compute C5 as max of all overlaps"""
        overlap = compute_overlap(h_vec)
        return np.max(overlap)
    
    # Cache evaluation
    last_c5 = float('inf')
    best_h = h.copy()
    best_c5 = compute_c5(h)
    patience = 0
    
    # Track time
    elapsed = time.time() - start_time
    if elapsed > budget_s - 1:
        return (best_h.tolist(), best_c5, n_points)
    
    # Try convex relaxation approach: optimize max directly
    # Alternative heuristic optimization with fine-grained gradient descent
    max_iter = 100000
    lr = 0.01
    
    # Use gradient descent with projected updates
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        # Current value
        current_c5 = compute_c5(h)
        
        # Approximate gradient via finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update via gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update
        h = h_new
        
        # Evaluate new solution
        new_c5 = compute_c5(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 1000:
                break
                
        # Adapt learning rate
        if abs(new_c5 - current_c5) < 1e-8:
            lr *= 0.98
            
    # Final validation
    final_c5 = compute_c5(best_h)
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
