import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_value = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    alpha = 0.25  # Discovery rate
    epsilon = 1e-5  # Perturbation step
    max_iter = 1000
    min_improvement = 1e-8
    stagnation_limit = 200
    iteration = 0
    stagnation_count = 0
    current_best_value = best_value
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        # Generate new solution via random walk
        new_h = best_h.copy()
        # Randomly choose a subset of indices to perturb
        num_perturb = max(1, int(0.15 * n_points))  # Perturb 15% of points
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for i in indices:
            # Add Gaussian noise
            new_h[i] += np.random.normal(0, epsilon)
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_value = evaluate_h(new_h, dx)
        if new_value < best_value:
            best_h = new_h.copy()
            best_value = new_value
            stagnation_count = 0
            current_best_value = best_value
        else:
            # Probabilistic acceptance (cuckoo search variant)
            if np.random.rand() < alpha:
                best_h = new_h.copy()
                best_value = new_value
                stagnation_count = 0
                current_best_value = best_value
            else:
                stagnation_count += 1
                
        if stagnation_count >= stagnation_limit:
            # Restart with new solution
            h_temp = np.random.rand(n_points)
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            temp_value = evaluate_h(h_temp, dx)
            if temp_value < best_value:
                best_h = h_temp.copy()
                best_value = temp_value
                current_best_value = best_value
            stagnation_count = 0

    # Final gradient descent on best solution
    for _ in range(1000):
