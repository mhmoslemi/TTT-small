import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 6000
    lr = 0.01
    stagnation_limit = 200
    improvement_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute gradients via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            if h_plus[i] < 0:
                h_plus[i] = 0
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update step
        h_new = h - lr * grad
        
        # Repair
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            # Scale to correct sum
            h_new = h_new * target_sum / h_sum
        
        # Evaluate new solution
        h = h_new
        corr = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            improvement_count = 0
            last_improvement = iteration
        else:
            improvement_count += 1
            
        # Adaptive learning rate
        if iteration % 200 == 0 and iteration > 0:
            lr *= 0.5
            
        # Perturb to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            perturb_mask = np.random.rand(n_points) < 0.01
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Check for stagnation
        if improvement_count >= stagnation_limit:
            # Reinitialize with a small random perturbation
            h = best_h.copy()
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            improvement_count = 0
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
