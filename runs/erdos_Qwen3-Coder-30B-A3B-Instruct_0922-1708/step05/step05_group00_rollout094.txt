        base_max = np.max(base_correlation)

            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)

            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - base_max) / eps
            
        return grad
    
    # Optimization loop with adaptive learning rate
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-stage optimization
    # Stage 1: Low-res coarse optimization first
    n_coarse = max(32, n_points // 2)

    # Ensure sum constraint

    return (best_h.tolist(), final_c5, n_points)
