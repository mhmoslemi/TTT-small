import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure sum constraint is met
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    # Gradient-based optimization with perturbations
    max_iter = 1000
    lr = 0.001
    perturb_freq = 50
    perturb_mag = 0.1
    patience = 0
    patience_limit = 20
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Correlation computation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        min_k_idx = np.argmax(corr)
        
        # Compute gradient for the minimal correlation component
        grad = np.zeros_like(h)
        k_offset = min_k_idx - (len(h) - 1)  # compute true offset
        
        # The derivative of correlation at k=wrt h is h[j-k] + h[j+k]
        # But we are computing d/dh_j of corr[k] = sum_i h[i] * h[i+k]
        # which is h[j-k] + h[j+k] for k = j - i where we sum over i
        
        # Solve for minimal correlation idx so that we update h accordingly
        # corr[k] = sum_{i=0}^{n_points-1} h[i] * (1 - h[i+k])
        # For fixed k, we want to compute gradient wrt h[j] 
        if k_offset >= 0:
            for j in range(len(h)):
                if j >= k_offset:
                    grad[j] += (1 - h[j - k_offset])
                if j + k_offset < len(h):
                    grad[j] += (1 - h[j + k_offset])
        else:
            k_offset = -k_offset
            for j in range(len(h)):
                if j >= k_offset:
                    grad[j] += (1 - h[j - k_offset])
                if j + k_offset < len(h):
                    grad[j] += (1 - h[j + k_offset])
        
        # Update h using gradient ascent (since we're minimizing)
        h = h + lr * grad
        
        # Project h into bounds [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new C5
        corr_new = correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Reduce learning rate on stagnation
        if patience > patience_limit:
            lr *= 0.5
            patience = 0
            # Add small random perturbation
            if i % perturb_freq == 0:
                noise = np.random.normal(0, perturb_mag, len(h)) 
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        # Occasionally jitter h to escape local minima
        if i % perturb_freq == 0 and i > 0:
            noise = np.random.normal(0, perturb_mag * (1.0 / (1.0 + i / 100)), len(h))
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final computation of C5 for best solution
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
