import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    stagnation_patience = 100
    min_improvement = 1e-8
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Start with a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to meet sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Find the minimizing k (index of minimum correlation)
        # The full correlation has 2*n_points - 1 elements
        # The indices corresponding to k in [0, 2] are around n_points - 1
        # We care about k in [0, 2], which corresponds to indices [n_points-1, n_points+1]
        # Let's more carefully extract the correlation for valid k values
        k_vals = np.arange(0, 2, dx)
        corr_vals = corr[n_points-1:n_points+1]  # we're restricted to k in [0, 2], so we only consider relevant positions
        # Actually, we should go through the full range and take the minimum
        # For now, let's refine and search more rigorously in a shorter range
        min_idx = np.argmin(corr)
        k_min = min_idx - (n_points - 1)
        # k_min can be -n_points+1 to +n_points-1, but we care about postive k where k in [0,2]
        # Actually, for simplicity, let's directly compute the actual values as if we do the full  
        # evaluation
        # For k=0 it's largest size
        # Let us instead compute around the value closest to k_min where k_min = (idx - (n_points-1))
        if min_idx != n_points - 1:
            # We compute around the k where this min occurs
            k_idx = min_idx
        else:
            k_idx = min_idx
        
        # To update, we compute an approximate gradient:
        # For each h_i, the contribution to correlation at position k_idx depends on:
        # the positions of elements. Specifically for the overlap integral,
        # the derivative of the correlation with respect to h[i] is the sum of
        # derivatives over all j in [0, n_points) such that j + k = i, but when 
        # we compute correlation via FFT, the complex nature of the indexing is tricky
        # Instead, let's compute via finite differences gradients correctly
        
        # Compute the gradient vector (approximate) via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Use gradient ascent to move in positive direction  
        h += lr * grad

        # Project back to feasible (clamp to [0,1] then renormalize)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

        # Periodic perturbations
        if iteration % perturbation_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration / 100.0)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)

        iteration += 1
        
        # Early stopping if no significant improvements
        if iteration - last_improvement > stagnation_patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
