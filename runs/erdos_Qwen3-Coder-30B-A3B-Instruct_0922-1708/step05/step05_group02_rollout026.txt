import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Simulated Annealing Parameters
    T0 = 0.1 * evaluate_h(h_init, dx)
    alpha = 0.99
    max_iter = 10000
    perturb_rate = 0.1
    min_temp = 1e-8
    progress_count = 0
    max_progress = 100

    # Initialize
    h_current = h_init.copy()
    C5_current = evaluate_h(h_current, dx)
    h_best = h_current.copy()
    C5_best = C5_current
    
    T = T0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb
        h_new = h_current.copy()
        n_perturb = max(1, int(perturb_rate * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for idx in indices:
            h_new[idx] += np.random.normal(0, 0.01)
        
        # Repair
        h_new = repair_h(h_new, n_points)
        C5_new = evaluate_h(h_new, dx)
        
        # Metropolis criterion
        delta = C5_new - C5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            C5_current = C5_new
        
        # Update best
        if C5_current < C5_best:
            h_best = h_current.copy()
            C5_best = C5_current
            progress_count = 0
        else:
            progress_count += 1
            
        # Cool
        if T > min_temp:
            T *= alpha
            
        # Adaptive cooling if no progress
        if progress_count >= max_progress:
            T = max(min_temp, T * 0.9)
            progress_count = 0
    
    # Final gradient descent refinement
    best_h = h_best.copy()
    best_C5 = C5_best
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)

            grad[j] = (C5_plus - best_C5) / eps
            
        best_h = best_h - 0.0005 * grad
