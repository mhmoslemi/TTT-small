import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize several starting points
    initial_points = []
    
    # Use provided parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_points.append(h_parent.copy())
    else:
        # Create a simple initial guess
        h_base = np.ones(n_points) * 0.5
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        initial_points.append(h_base.copy())
    
    # Add random initializations
    for _ in range(4):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        initial_points.append(h_rand.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # For each initialization point
    for i, h0 in enumerate(initial_points):
        h = h0.copy()
        
        # Convex relaxation to get a starting point
        try:
            h_var = cp.Variable(n_points)
            # Try to solve a relaxation by minimizing sum of overlaps
            # This does not directly solve our problem, but finds a feasible point
            # For now we just project to satisfy constraints and refine below
            
            # Refine using L-BFGS-B with bounds
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                return np.max(corr)
            
            def gradient(h_vec):
                # Compute gradient using finite differences
                eps = 1e-5
                grad = np.zeros_like(h_vec)
                for j in range(n_points):
                    h_plus = h_vec.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    c_plus = objective(h_plus)
                    
                    h_minus = h_vec.copy()
                    h_minus[j] -= eps
                    h_minus = np.clip(h_minus, 0, 1)
                    h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
                    c_minus = objective(h_minus)
                    
                    grad[j] = (c_plus - c_minus) / (2 * eps)
                return grad
                
            # Use a few steps of a local optimizer like L-BFGS to improve
            bounds = [(0, 1) for _ in range(n_points)]
            # Make into a continuous-to-discrete iterator 
            # But because our values are converging more stably, let us optimize
            # One iteration should be enough to improve from the relaxed convex solution
            res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, jac=gradient, 
                           options={'maxiter': 2000, 'gtol': 1e-6, 'ftol': 1e-6})
            
            if res.success:
                h_opt = res.x
            else:
                h_opt = h.copy()
            
            # Ensure it's feasible 
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            c5 = objective(h_opt)
            
            if c5 < best_c5:
                best_h = h_opt.copy()
                best_c5 = c5
                
        except Exception:
            # If optimization fails, just use the starting point
            h_proj = np.clip(h, 0, 1)
            h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
            corr = np.correlate(h_proj, 1 - h_proj, mode='full') * dx
            c5 = np.max(corr)
            if c5 < best_c5:
                best_h = h_proj.copy()
                best_c5 = c5
                
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final check to make sure best_h is valid
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Sanitize
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), final_c5, n_points)
