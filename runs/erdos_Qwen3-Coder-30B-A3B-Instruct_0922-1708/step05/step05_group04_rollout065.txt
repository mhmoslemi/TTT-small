import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem constants
    n_points = 100
    dx = 2.0 / n_points
    k_grid = np.linspace(-2, 2, 50)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Function to compute C5
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Function to repair h
    def repair_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_vals = h_vals * (n_points / 2) / h_sum
        return h_vals
    
    # Binary search on the bound
    low = 0.0
    high = 0.5
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    while high - low > 1e-5 and time.time() - start_time < budget_s - 10:
        mu = (low + high) / 2.0
        
        # Try to construct h that satisfies overlap constraints
        h_attempt = h.copy()
        improved = True
        iteration = 0
        
        while improved and iteration < 50 and time.time() - start_time < budget_s - 10:
            improved = False
            # For each k in the grid, check overlap
            violations = []
            for k in k_grid:
                # Compute overlap constraint
                if k >= 0:
                    h_shifted = np.concatenate([np.zeros(int(k/dx)), h_attempt[:-int(k/dx)]])
                else:
                    h_shifted = np.concatenate([h_attempt[-int(-k/dx):], np.zeros(int(-k/dx))])
                
                overlap = np.sum(h_attempt * h_shifted) * dx
                if overlap < mu:
                    violations.append((k, overlap, mu - overlap))
            
            if len(violations) > 0:
                improved = True
                # Boost the underperforming regions
                for k, overlap, diff in violations:
                    indices = np.arange(len(h_attempt))
                    if k >= 0:
                        shifted_indices = np.arange(int(k/dx), len(h_attempt))
                        to_boost = shifted_indices 
                    else:
                        shifted_indices = np.arange(0, len(h_attempt) - int(-k/dx))
                        to_boost = shifted_indices
                    
                    for i in to_boost:
                        h_attempt[i] = min(1.0, h_attempt[i] + 0.01 * diff / len(to_boost))
                
                h_attempt = repair_h(h_attempt)
                iteration += 1
            else:
                break
                
        # Re-evaluate
        c5_current = compute_c5(h_attempt)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_attempt.copy()
        
        if c5_current < mu:
            high = mu
        else:
            low = mu
    
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    learning_rate = 0.001
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_refined -= learning_rate * grad
        h_refined = repair_h(h_refined)
        
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    final_c5 = compute_c5(h_refined)
    return (h_refined.tolist(), final_c5, n_points)
