from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mutation_strength = 0.5
    initial_sigma = 0.1
    sigma = initial_sigma
    elite_ratio = 0.5
    stagnation_limit = 200
    restart_interval = 100
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            indices = np.argsort(h)
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Fitness calculation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main ES loop
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update the best solution
        if sorted_fitnesses[0] < best_fitness:
            best_fitness = sorted_fitnesses[0]
            best_individual = sorted_population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Select elite
        elite_size = int(elite_ratio * population_size)
        elite = sorted_population[:elite_size]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            # Select parent from elite
            parent = elite[np.random.randint(0, elite_size)]
            
            # Gaussian mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mutation_strength * noise
            
            # Repair
            child = repair(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[-offspring_size:]
        for i, off in enumerate(offspring):
            population[worst_indices[i]] = off

                    population[i] = h
            stagnation_count = 0
        
        # Reset mutation strength periodically
        if current_iter % restart_interval == 0:
            mutation_strength *= 0.9
    
    # Final refinement using gradient descent
    if best_individual is not None:
        h_ref = best_individual.copy()
        for _ in range(100):
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad
            h_ref = repair(h_ref)
            
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        
        final_corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        final_fitness = np.max(final_corr) * dx
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_individual = h_ref.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
