import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the C5 bound for a given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    return h

def initialize_population(n_points, n_pop, initial_h):
    """Create initial population with mix of parent and random solutions"""
    pop = []
    if initial_h is not None and len(initial_h) == n_points:
        # Use parent solution as base
        for _ in range(n_pop // 2):
            perturbed = initial_h + np.random.normal(0, 0.05, n_points)
            pop.append(project_to_feasible(perturbed, n_points))
        # Add random solutions
        for _ in range(n_pop // 2):
            rand_h = np.random.rand(n_points)
            pop.append(project_to_feasible(rand_h, n_points))
    else:
        # All random
        for _ in range(n_pop):
            rand_h = np.random.rand(n_points)
            pop.append(project_to_feasible(rand_h, n_points))
    return pop

def tournament_selection(pop, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(pop), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return pop[best_idx]

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.03):
    """Mutate individual"""
    mask = np.random.rand(len(individual)) < mutation_rate
    individual = individual.copy()
    individual[mask] += np.random.normal(0, sigma, np.sum(mask))
    return individual

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine h using gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        # Evaluate current fitness and estimate gradient
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        
        # Update learning rate based on progress
        lr *= 0.995
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Get initial solution
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            # Interpolate
            x_orig = np.linspace(0, 2, len(initial_h))
            x_target = np.linspace(0, 2, n_points)
            initial_h = np.interp(x_target, x_orig, initial_h)
    
    # Initialize
    n_pop = 100
    population = initialize_population(n_points, n_pop, initial_h)
    
    start_time = time.time()
    best_c5 = float('inf')
    best_h = None
    
    # GA parameters
    tournament_size = 5
    crossover_rate = 0.8
    mutation_rate = 0.1
    
    # Main loop
    gen = 0
    while time.time() - start_time < budget_s - 10:
        gen += 1
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Selection, crossover, mutation
        new_pop = []
        
        # Elitism: keep best
        elite_idx = np.argsort(fitnesses)[:n_pop//10]
        for idx in elite_idx:
            new_pop.append(population[idx].copy())
        
        # Generate rest of population
        while len(new_pop) < n_pop:
            # Selection
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            # Crossover
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Projection
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_pop.extend([child1, child2])
            
        population = new_pop[:n_pop]
        
        # Periodic gradient refinement on top individuals
        if gen % 50 == 0:
            # Sort by fitness
            sorted_indices = np.argsort([compute_fitness(h, dx) for h in population])
            top_indices = sorted_indices[:n_pop//10]
            for idx in top_indices:
                refined = gradient_refinement(population[idx], dx)
                fitness = compute_fitness(refined, dx)
                old_fitness = compute_fitness(population[idx], dx)
                if fitness < old_fitness:
                    population[idx] = refined
                    if fitness < best_c5:
                        best_c5 = fitness
                        best_h = refined.copy()
    
    # Final refinement
    if best_h is not None:
        refined = gradient_refinement(best_h, dx)
        final_fitness = compute_fitness(refined, dx)
        if final_fitness < best_c5:
            best_c5 = final_fitness
            best_h = refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
