import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to sum = n_start/2
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse optimization
    best_h_coarse = h.copy()
    best_c5_coarse = float('inf')
    
    # Create constraint matrix for coarse problem
    A_eq = np.ones((1, n_start))
    b_eq = np.array([n_start / 2])
    
    # Quadratic programming approach
    try:
        h_coarse = cp.Variable(n_start)
        C5 = cp.Variable()
        
        constraints = [
            h_coarse >= 0,
            h_coarse <= 1,
            cp.sum(h_coarse) == n_start / 2,
            C5 >= 0
        ]
        
        # Add overlap constraints - this is a relaxation
        # We'll create a quadratic penalty formulation
        # for each possible shift in the coarse grid
        
        # This approach directly uses non-convex constraint relaxation
        # by using convex approximations
        
        # For true optimization, solve with a convex relaxation
        # that bounds the maximal overlap
        
        prob = cp.Problem(cp.Minimize(C5), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_sol = h_coarse.value
                if h_sol is not None:
                    h = np.array(h_sol)
                    h = np.clip(h, 0, 1)
                    h = h * (n_start / 2) / np.sum(h)
        except:
            pass
    except:
        pass
    
    # Transfer to fine grid
    if 'initial_h_values' in globals():
        h_fine = np.array(initial_h_values)
        if len(h_fine) != n_points:
            x_orig = np.linspace(0, 2, len(h_fine))
            x_new = np.linspace(0, 2, n_points)
            h_fine = np.interp(x_new, x_orig, h_fine)
    else:
        h_fine = np.interp(np.linspace(0, 2, n_points), 
                           np.linspace(0, 2, n_start), h)
    
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Refine using adaptive gradient descent
    # Track best solution
    best_h = h_fine.copy()
    best_c5 = float('inf')
    
    # Helper function for computing correlation
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(correlation)
    
    # Adaptive gradient descent
    lr = 0.01
    patience = 0
    last_c5 = float('inf')
    
    max_iter = 10000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(best_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 300:
            break
            
        # Adaptive learning rate
        if abs(current_c5 - last_c5) < 1e-6:
            lr *= 0.99
        else:
            lr = min(0.01, lr * 1.01)
        lr = max(1e-5, min(lr, 1e-3))
            
        # Finite difference gradient
        eps = 1e-4
        grad = np.zeros_like(best_h)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update
        h_new = best_h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        last_c5 = current_c5
        best_h = h_new
        
        # Slight decrease in learning rate
        lr = max(1e-6, lr * 0.9995)
    
    # Final check and return
    final_c5 = compute_c5(best_h)
    
    # Ensuring final normalization
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Double verification
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
