import numpy as np
import time
from scipy.signal import correlate

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def mutate_h(h, n_points, mutation_rate=0.15):
    """Generate a mutated version of h."""
    h_new = h.copy()
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_new[i] = 1 - h_new[i]  # flip
    return repair_h(h_new, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_best = h_init.copy()
    else:
        current_best = np.random.rand(n_points)
        current_best = current_best * (n_points / 2) / np.sum(current_best)
    
    current_c5 = evaluate_h(current_best, dx)
    
    best_h = current_best.copy()
    best_c5 = current_c5
    
    # Main optimization loop
    no_improvement_count = 0
    max_no_improvement = 100
    n_mutations = 100
    
    while time.time() - start_time < budget_s - 10 and no_improvement_count < max_no_improvement:
        # Generate mutations
        for _ in range(n_mutations):
            if time.time() - start_time > budget_s - 10:
                break
                
            h_mut = mutate_h(current_best, n_points)
            c5_mut = evaluate_h(h_mut, dx)
            
            if c5_mut < current_c5:
                current_best = h_mut.copy()
                current_c5 = c5_mut
                no_improvement_count = 0
                if c5_mut < best_c5:
                    best_c5 = c5_mut
                    best_h = h_mut.copy()
            else:
                no_improvement_count += 1
                
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final gradient-based refinement
    max_refinements = 500
    for _ in range(max_refinements):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
