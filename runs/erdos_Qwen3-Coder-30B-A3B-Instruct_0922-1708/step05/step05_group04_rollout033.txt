from scipy.optimize import minimize
import time

    stagnation_limit = 200
    restart_interval = 100

        correlation = np.correlate(h_init, 1 - h_init, mode='full')

            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize with random solutions
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        scaling = (n_points / 2) / h_sum
        h = h * scaling
        # Ensure sum is exactly correct
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Evaluate fitness
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    iteration = 0
    stagnation_count = 0
    current_mu = mutation_strength
    current_sigma = mutation_variance
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate population
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update best solution
        if sorted_fitnesses[0] < best_c5:

            best_h = sorted_population[0].copy()
            stagnation_count = 0
        else:

            child = repair(child)
            offspring.append(child)

            grad = np.zeros(n_points)

                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            # Update
            h_ref -= 0.001 * grad
            h_ref = repair(h_ref)
            
            # Check improvement
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        
        # Final verification
        corr_final = np.correlate(h_ref, 1 - h_ref, mode='full')
        final_c5 = np.max(corr_final) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
            best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
