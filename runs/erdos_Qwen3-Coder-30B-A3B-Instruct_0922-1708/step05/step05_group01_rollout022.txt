    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize 
    h = np.ones(n_points) * 0.5
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h = h0 * (n_points / 2) / np.sum(h0)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Gradient descent parameters
    lr = 0.001
    penalty_weight = 0.1
    penalty_factor = 1.2
    min_penalty = 1e-4
    penalty = penalty_weight
    max_iter = 5000
    plateau_patience = 200
    perturb_factor = 0.01
    decay_factor = 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    prev_h = h.copy()
    prev_c5 = float('inf')
    patience = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations 
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Compute gradient using central differences
        grad = np.zeros_like(h)
        eps = 1e-6
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            if h_plus[j] > 1: h_plus[j] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - c5) / eps

        # Apply penalty gradient
        penalty_grad = np.zeros_like(h)
        constraint_violation = np.sum(h) - n_points / 2
        if abs(constraint_violation) > 1e-8:
            penalty_grad += penalty * 2 * (np.sum(h) - n_points / 2) 
        penalty_grad = np.clip(penalty_grad, -1e6, 1e6)  # prevent blowup
        
        # Composite gradient
        grad += penalty_grad
        
        # Line search step - simple backtracking linesearch
        step_size = lr
        h_new = h - step_size * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale exactly
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new cost
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Update penalty dynamically
        constraint_violation = abs(np.sum(h_new) - n_points/2)
        if constraint_violation > 1e-3: 
            penalty = penalty * penalty_factor
        else:
            penalty = max(penalty / penalty_factor, min_penalty)

        # Greedy descent update
        if c5_new < c5:
            h = h_new.copy()
            prev_c5 = c5_new
        else:
            # Try making it conditionally better
            h = prev_h.copy()
            patience += 1
            if patience > plateau_patience:
                # Reduce learning rate
                lr *= decay_factor
                patience = 0
                # Add small perturbation
                noise = np.random.normal(0, perturb_factor, n_points)
                h = h + noise
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        prev_h = h.copy()
                    
    return (best_h.tolist(), best_c5, n_points)
