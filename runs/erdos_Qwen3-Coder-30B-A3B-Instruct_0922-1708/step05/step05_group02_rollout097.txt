import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h = h_init.copy()
    else:
        # Initialize randomly with constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 10:

        # Compute gradient using finite differences

        grad = np.zeros(n_points)
        h_base = h.copy()
        c5_base = evaluate_h(h_base, dx)
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - c5_base) / eps
        
        # Update h using gradient descent
        h = h - 0.001 * grad
        h = repair_h(h, n_points)
        
        # Occasionally apply random perturbations
        if iteration % 20 == 0:
            perturb_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)

        c5 = evaluate_h(h, dx)
        
        # Update best solution if improved
        if c5 < best_c5:

            last_improvement = iteration
        
        # Early stopping if no improvement for many iterations
        if iteration - last_improvement > 100:

    for _ in range(500):
