import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    max_iter = 5000
    mu = 0.1
    lr = 0.001
    perturbation_fraction = 0.01
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute C5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximizing lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (2 * n_points - 1)
        
        # Compute gradient for current lag
        grad = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag] - h[i]
        else:
            for i in range(-max_lag, n_points):
                grad[i] += 1 - h[i + max_lag] - h[i]
        
        # Update h using gradient descent
        h = h - lr * grad
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to exact sum
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Occasionally add some randomness
        if iteration % 100 == 0:
            noise = np.random.normal(0, perturbation_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
