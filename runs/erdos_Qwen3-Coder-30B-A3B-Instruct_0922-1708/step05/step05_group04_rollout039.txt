import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, n_points, target_sum):
    """Repair h to ensure sum = target_sum and bounds [0,1]"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    
    diff = current_sum - target_sum
    if diff > 0:
        # Remove excess from largest elements
        indices = np.argsort(h)[::-1]
        for idx in indices:
            if diff <= 0:
                break
            dec = min(h[idx], diff)
            h[idx] -= dec
            diff -= dec
    else:
        # Add deficit to smallest elements
        indices = np.argsort(h)
        for idx in indices:
            if diff >= 0:
                break
            inc = min(1 - h[idx], -diff)
            h[idx] += inc
            diff += inc
    
    # Scale to exact target sum
    if np.sum(h) > 0:
        h = h * target_sum / np.sum(h)
    else:
        h = np.full_like(h, target_sum / n_points)
        
    return h

def de_search(seed, budget_s, n_points=100):
    """Differential Evolution search for the Erdos problem"""
    np.random.seed(seed)
    start_time = time.time()
    
    dx = 2.0 / n_points
    target_sum = n_points / 2
    population_size = 100
    max_generations = 1000
    F = 0.8  # Mutation factor
    CR = 0.9  # Crossover probability
    mutation_decay = 0.99
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        
        h_init = repair_h(h_init * (n_points / 2) / np.sum(h_init), n_points, target_sum)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = repair_h(h * (n_points / 2) / np.sum(h), n_points, target_sum)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = repair_h(h * (n_points / 2) / np.sum(h), n_points, target_sum)
            population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_c5(p, dx) for p in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    generation = 0
    while generation < max_generations and time.time() - start_time < budget_s - 10:
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three different individuals
            indices = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair mutant
            mutant = repair_h(mutant, n_points, target_sum)
            
            # Crossover
            trial = np.zeros(n_points)
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < CR or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Evaluate trial
            trial_fitness = evaluate_c5(trial, dx)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
                
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Update best solution
        new_best_idx = np.argmin(new_fitness)
        if new_fitness[new_best_idx] < best_c5:
            best_c5 = new_fitness[new_best_idx]
            best_h = population[new_best_idx].copy()
        
        # Adjust mutation factor for exploitation
        if generation % 50 == 0:
            F *= mutation_decay
            
        generation += 1
    
    # Final gradient refinement
    best_h_refined = best_h.copy()
    for _ in range(100):
        corr = np.correlate(best_h_refined, 1 - best_h_refined, mode='full')
        base_fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points, target_sum)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[j] = (fitness_plus - base_fitness) / eps
            
        best_h_refined -= 0.001 * grad
        best_h_refined = repair_h(best_h_refined, n_points, target_sum)
        
        # Check for improvement
        corr_new = np.correlate(best_h_refined, 1 - best_h_refined, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= base_fitness:
            break
        else:
            base_fitness = fitness_new
    
    # Final evaluation
    final_corr = np.correlate(best_h_refined, 1 - best_h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return best_h_refined.tolist(), final_c5, n_points

def run(seed=42, budget_s=600, **kwargs):
    return de_search(seed, budget_s)
