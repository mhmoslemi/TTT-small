import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        fit = np.max(corr) * dx
        fitness.append(fit)
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Main DE loop
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        new_fitness = []
        
        # Update mutation factor
        current_mutation = mutation_factor * (0.99 ** (current_iter // 50))
        
        for i in range(population_size):
            # Select three different individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            idx1, idx2, idx3 = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[idx1] + current_mutation * (population[idx2] - population[idx3])
            
            # Repair
            mutant = repair(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_prob or j == j_rand:
                    trial[j] = mutant[j]
            
            # Repair trial
            trial = repair(trial)
            
            # Evaluate
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(corr) * dx
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
            
            # Update global best
            if trial_fitness < best_fitness:
                best_fitness = trial_fitness
                best_h = trial.copy()
        
        population = new_population
        fitness = new_fitness
        current_iter += 1
    
    # Final gradient-based refinement
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    def gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        grad = np.zeros_like(h)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
        return grad
    
    h_ref = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        # Gradient descent step
        grad = gradient(h_ref)
        h_ref -= 0.001 * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
    
    # Final evaluation
    corr_final = np.correlate(h_ref, 1 - h_ref, mode='full')
    c5_final = np.max(corr_final) * dx
    
    return (h_ref.tolist(), c5_final, n_points)
