import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Simulated Annealing parameters
    T = best_c5 * 0.1  # Initial temperature
    alpha = 0.99       # Cooling rate
    max_iter = 20000
    perturb_rate = 0.1 # Fraction of points to perturb
    
    # Track progress
    last_improvement = 0
    stagnation_count = 0
    max_stagnation = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Determine number of indices to perturb
        n_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        # Create candidate solution
        h_new = best_h.copy()
        for i in indices:
            # Small random perturbation
            delta = np.random.normal(0, 0.05)
            h_new[i] += delta
        h_new = repair_h(h_new, n_points)
        
        # Evaluate candidate
        new_c5 = evaluate_h(h_new, dx)
        
        # Acceptance criterion
        delta = new_c5 - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            best_h = h_new.copy()
            best_c5 = new_c5
            
            # Reset stagnation counter on improvement
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive cooling
        if stagnation_count > 10:
            T *= alpha
        
        # Early termination if no improvement in a while
        if stagnation_count >= max_stagnation:
            break
    
    # Fine-tune with gradient descent
    h_final = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        best_c5 = np.max(corr) * dx
        
    return (h_final.tolist(), best_c5, n_points)
