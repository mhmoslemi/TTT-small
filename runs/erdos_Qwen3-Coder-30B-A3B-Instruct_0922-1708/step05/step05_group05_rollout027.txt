import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block + ramps
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize sum to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    # Gradient-based optimization with trust region
    step_size = 0.01
    last_improvement = 0
    perturb_period = 50
    perturb_decay = 0.95
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current correlation and gradient estimate
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (np.max(corr_plus) * dx - current_c5) / eps
        
        # Move in gradient direction
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Correct sum constraint using scaling
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept if improvement or accept with probability
        if new_c5 < current_c5:
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            step_size *= 1.2  # Increase step if successful
            last_improvement = i
        else:
            step_size *= 0.5  # Reduce step if no improvement
            if np.random.random() < 0.01:  # Small probability of accepting worse
                h = h_new
                
        # Perturb periodically to avoid local minima
        if i % perturb_period == 0 and i > last_improvement + 100:
            noise = np.random.normal(0, step_size * 0.1, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Perturb step size a bit
            step_size *= perturb_decay
            
    return (best_h.tolist(), best_c5, n_points)
