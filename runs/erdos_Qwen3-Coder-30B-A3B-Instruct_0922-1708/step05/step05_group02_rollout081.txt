import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Adjust smallest elements to correct the sum
        if current_sum < target_sum:
            # Need to increase sum
            indices = np.argsort(h)
            diff = target_sum - current_sum
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            diff = current_sum - target_sum
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] - diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    alpha = 0.25  # Discovery rate
    epsilon = 1e-5
    max_iter = 1000
    stagnation_limit = 200
    stagnation_count = 0
    best_c5_history = []
    
    # Cuckoo search main loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new solution via random walk
        new_h = best_h.copy()
        
        # Randomly perturb a subset of elements
        mask = np.random.random(n_points) < 0.3  # Perturb 30% of elements
        if np.any(mask):
            for i in range(n_points):
                if mask[i]:
                    # Add Gaussian noise
                    new_h[i] += np.random.normal(0, 0.05)
        
        # Repair solution
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the new solution with some probability
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            if np.random.random() < alpha:
                best_h = new_h.copy()
                best_c5 = new_c5
                stagnation_count = 0
                
        stagnation_count += 1
        if stagnation_count >= stagnation_limit:
            # Perturb best solution more aggressively if stuck
            for i in range(n_points):
                if np.random.random() < 0.1:
                    best_h[i] += np.random.normal(0, 0.01)
            best_h = repair_h(best_h, n_points)
            best_c5 = evaluate_h(best_h, dx)
            stagnation_count = 0
            
    # Final refinement using gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:

        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
