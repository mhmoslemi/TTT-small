import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from provided construction or use block construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to 0.5
    h = h * (0.5 / np.sum(h))
    
    # Compute initial correlation and C5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Gradient-based optimization parameters
    learning_rate = 0.001
    max_iterations = 10000
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Keep track of recent improvements for adaptive learning rate
    recent_improvements = []
    improvement_window = 10
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the correlation to find the minimum
        corr = correlate(h, 1 - h, mode='full')
        min_idx = np.argmin(corr)
        min_val = corr[min_idx]
        min_k = min_idx - (len(corr) - 1) // 2
        
        # Compute gradient: d/dh_j of min_k term
        gradient = np.zeros(n_points)
        
        # For the minimum k, compute gradient for relevant positions
        # The derivative of h[i] * (1 - h[i+k]) w.r.t. h[i] is (1 - h[i+k])
        # But we only care about min_k, which is a convolution
        
        # Simple approximation: finite difference gradient
        # This is high variance, but effective for this non-convex problem
        eps = 1e-5
        base_c5 = min_val * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to [0,1] and renormalize
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (0.5 / np.sum(h_plus))
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            min_val_plus = corr_plus[min_idx]
            c5_plus = min_val_plus * dx
            
            gradient[i] = (c5_plus - base_c5) / eps
            
        # Update step with projection
        h_new = h - learning_rate * gradient
        h_new = np.clip(h_new, 0, 1)
        # Renormalize
        h_new = h_new * (0.5 / np.sum(h_new))
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Only accept improvements or with some probability for exploration
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            recent_improvements.append(1)
        else:
            recent_improvements.append(0)
            
        # Keep best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Adaptive learning rate
        if len(recent_improvements) > improvement_window:
            recent_improvements.pop(0)
            if sum(recent_improvements) / len(recent_improvements) > 0.8:
                learning_rate *= 1.1
            elif sum(recent_improvements) / len(recent_improvements) < 0.3:
                learning_rate *= 0.9
                
        # Random perturbations to escape local minima
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration / 100)), n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.sum(h))
    
    # Final validation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
