import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Population-based approach with genetic operators
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    gradient_steps = 30
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = []
        # Start from initial solution with some perturbation
        population.append(h.copy())
        for _ in range(population_size - 1):
            # Perturb solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Initialize with random feasible solutions
        population = []
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    # Main evolutionary loop
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Evaluate fitness
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Keep track of best solution so far
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Select parents using tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create next generation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness)[:population_size//10]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring via crossover and mutation
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate and len(new_population) + 1 < population_size:
                # Crossover
                parent1 = selected[np.random.randint(0, len(selected))]
                parent2 = selected[np.random.randint(0, len(selected))]
                
                # Blend crossover
                alpha = 0.7 + np.random.rand() * 0.25  # 0.7 to 0.95
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
                
                # Mutation
                for idx in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[idx] += np.random.normal(0, 0.02)
                    if np.random.rand() < mutation_rate:
                        child2[idx] += np.random.normal(0, 0.02)
                
                # Repair
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Mutation only
                parent = selected[np.random.randint(0, len(selected))]
                child = parent.copy()
                for idx in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[idx] += np.random.normal(0, 0.02)
                # Repair
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population[:population_size]
        
        # Local refinement
        if iteration % refinement_interval == 0 and iteration > 0:
            # Select top individuals for refinement
            top_indices = np.argsort(fitness)[:population_size//5]
            for idx in top_indices:
                # Gradient descent refinement
                h_ref = population[idx].copy()
                for _ in range(gradient_steps):
                    # Estimate gradient using finite differences
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    # Update with gradient
                    h_ref = h_ref - 0.0005 * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                
                # Update population if better
                corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                new_c5 = np.max(corr_new) * dx
                if new_c5 < fitness[idx]:
                    population[idx] = h_ref.copy()
                    fitness[idx] = new_c5
        
        # Periodic keep best
        if iteration % 100 == 0:
            min_fitness_idx = np.argmin(fitness)
            if fitness[min_fitness_idx] < best_c5:
                best_c5 = fitness[min_fitness_idx]
                best_h = population[min_fitness_idx].copy()
    
    # Final refinement of top individuals
    top_indices = np.argsort(fitness)[:min(5, len(fitness))]
    for idx in top_indices:
        h_ref = population[idx].copy()
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - fitness[idx]) / eps
            
            # Update
            h_ref = h_ref - 0.0002 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        # Final check
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
