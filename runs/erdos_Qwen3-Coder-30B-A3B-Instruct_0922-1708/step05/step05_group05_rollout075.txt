import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    initial_alpha = 0.01
    max_perturb = 0.05
    perturb_decay = 0.95
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Piecewise constant construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        
    # Normalize to sum = n_points/2
    h = h_init * (n_points / 2) / np.sum(h_init)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    alpha = initial_alpha
    perturb_prob = 0.1
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Project gradient to feasible direction
        h_candidate = h - alpha * grad
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Correct sum
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        
        # Accept or reject
        if c5_candidate < current_c5:
            h = h_candidate.copy()
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
            alpha = min(alpha * 1.2, 0.1)
        else:
            alpha = max(alpha * 0.5, 1e-6)
        
        # Perturb occasionally
        if random.random() < perturb_prob:
            noise = np.random.normal(0, max_perturb * alpha, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            corr_perturbed = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_perturbed = np.max(corr_perturbed) * dx
            
            if c5_perturbed < current_c5:
                h = h_perturbed.copy()
                current_c5 = c5_perturbed
                if c5_perturbed < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_perturbed
        
        # Decay perturbation
        perturb_prob *= perturb_decay
    
    return (best_h.tolist(), best_c5, n_points)
