import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing
    T0 = 10.0
    iteration = 0
    T = T0
    
    # Main SA loop
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        # Update temperature
        T = T0 / np.log(1 + iteration)
        
        # Perturb h
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        # Evaluate current
        correlation_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation_current) * dx
        
        # Accept or reject
        if c5_candidate < c5_current or np.random.rand() < np.exp(-(c5_candidate - c5_current) / T):
            h = h_candidate.copy()
            c5_current = c5_candidate
            
        # Update best
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    
    # Gradient-based refinement on best solution
    h_refine = best_h.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 15:
            break
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        h_refine = h_refine - 0.001 * grad
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        # Recompute c5
        corr_new = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refine.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
