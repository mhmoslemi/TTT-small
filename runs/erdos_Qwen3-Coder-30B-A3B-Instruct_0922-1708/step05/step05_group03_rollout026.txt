import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    max_refinement_steps = 50
    learning_rate = 0.001
    
    # Generate initial population
    population = []
    
    if 'initial_h_values' in globals():
        # Start with parent solution
        parent = np.array(initial_h_values)
        parent = parent * (n_points / 2.0) / np.sum(parent)
        
        # Add variations with noise
        for _ in range(population_size // 2):
            perturbed = parent + np.random.normal(0, 0.05, n_points)
            population.append(project_to_feasible(perturbed, n_points))
        
        # Add random solutions
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            population.append(h)
    else:
        # All random solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            population.append(h)
    
    # Main GA loop
    start_time = time.time()
    best_fitness = float('inf')
    best_individual = None
    
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitnesses = [compute_fitness(ind, dx) for ind in population]
        
        # Update best individual
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Selection via tournament
        def tournament_selection():
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            return population[winner_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_individual.copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection and crossover
            parent1 = tournament_selection()
            parent2 = tournament_selection()
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.uniform(0.7, 0.95)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_magnitude = np.random.uniform(0.01, 0.05)
                child1 += np.random.normal(0, mutation_magnitude, n_points)
            if np.random.rand() < mutation_rate:
                mutation_magnitude = np.random.uniform(0.01, 0.05)
                child2 += np.random.normal(0, mutation_magnitude, n_points)
            
            # Project to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = new_population[:population_size]
        
        # Periodic refinement with gradient descent
        if generation % refinement_interval == 0 and generation > 0:
            # Select top individuals
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                
                for _ in range(max_refinement_steps):
                    # Numerical gradient
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    base_fitness = compute_fitness(h_ref, dx)
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - base_fitness) / eps
                    
                    # Update with gradient descent
                    h_ref -= learning_rate * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    # Evaluate
                    new_fitness = compute_fitness(h_ref, dx)
                    if new_fitness < base_fitness:
                        base_fitness = new_fitness
                        population[idx] = h_ref.copy()
                        if new_fitness < best_fitness:
                            best_fitness = new_fitness
                            best_individual = h_ref.copy()
                    else:
                        break  # No improvement, stop this refinement
    
    # Final refinement on best individual
    h_final = best_individual.copy()
    for _ in range(100):
        grad = np.zeros_like(h_final)
        eps = 1e-4
        base_fitness = compute_fitness(h_final, dx)
        
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, n_points)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - base_fitness) / eps
        
        h_final -= 0.0005 * grad
        h_final = project_to_feasible(h_final, n_points)
        
        new_fitness = compute_fitness(h_final, dx)
        if new_fitness < best_fitness:
            best_fitness = new_fitness
            best_individual = h_final.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
