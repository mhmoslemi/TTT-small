import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Ensure sum constraint and clip
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse-to-fine approach
    n_coarse = min(50, n_points)
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), 
                         np.linspace(0, 2, n_points), h)
    
    # Optimize on coarse grid
    try:
        h_coarse_var = cp.Variable(n_coarse)
        C5_var = cp.Variable()
        
        constraints = [
            h_coarse_var >= 0,
            h_coarse_var <= 1,
            cp.sum(h_coarse_var) == n_coarse / 2,
            C5_var >= 0
        ]
        
        # Simplifying approach: substitute coarse solution into full problem  
        # and optimize directly using gradient descent on that
        h_coarse_opt = h_coarse.copy()
        lr = 0.01
        iters = 0
        patience = 0
        
        while iters < 2000 and time.time() - start_time < budget_s - 5:
            # Evaluate current coarse solution
            corr = np.correlate(h_coarse_opt, 1 - h_coarse_opt, mode='full') * dx_coarse
            curr_c5 = np.max(corr)
            
            if curr_c5 < best_c5:
                best_c5 = curr_c5
                best_h = np.interp(np.linspace(0, 2, n_points), 
                                   np.linspace(0, 2, n_coarse), h_coarse_opt)
            
            old_c5 = curr_c5
            
            # Gradient step
            grad = np.zeros_like(h_coarse_opt)
            eps = 1e-4
            
            for i in range(n_coarse):
                h_plus = h_coarse_opt.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_coarse / 2) / np.sum(h_plus)
                
                cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
                c5_plus = np.max(cor_plus)
                
                grad[i] = (c5_plus - curr_c5) / eps
                
            h_coarse_opt = h_coarse_opt - lr * grad
            h_coarse_opt = np.clip(h_coarse_opt, 0, 1)
            h_coarse_opt = h_coarse_opt * (n_coarse / 2) / np.sum(h_coarse_opt)
            
            # Adaptive learning rate
            if abs(curr_c5 - old_c5) < 1e-6:
                patience += 1
                if patience > 50:
                    break
            else:
                patience = 0
            
            lr = max(1e-6, lr * 0.999)
            iters += 1
            
    except Exception as e:
        pass
    
    # Refinement on fine grid with same setup
    if time.time() - start_time > budget_s - 10:
        # Return best found if insufficient time
        correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = np.max(correlation)
        return (best_h.tolist(), final_c5, n_points)
    
    # Final refinement on fine grid
    h_fine = best_h.copy()
    lr_fine = 0.005
    patience = 0
    
    max_iter = 5000
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        curr_c5 = np.max(corr)
        
        if curr_c5 < best_c5:
            best_c5 = curr_c5
            best_h = h_fine.copy()
        
        if abs(curr_c5 - best_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
            
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - curr_c5) / eps
            
        h_fine = h_fine - lr_fine * grad
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        lr_fine = max(1e-6, lr_fine * 0.9995)
    
    final_c5 = best_c5
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
