import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    initial_alpha = 0.01
    max_perturb_iter = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    # Gradient descent
    alpha = initial_alpha
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        # Compute base correlation
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        # Estimate gradient
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - base_c5) / eps
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new correlation
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept if improvement
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            last_improvement = i
            
            # Adjust alpha
            if i > 0 and i - last_improvement < 20:
                alpha = min(alpha * 1.1, 0.1)
        else:
            # Reduce alpha
            alpha *= 0.5
            
            # If stuck, do perturbation
            if i - last_improvement > max_perturb_iter:
                # Add small random perturbation
                perturbation = np.random.normal(0, 0.01, n_points)
                h_pert = h + perturbation
                h_pert = np.clip(h_pert, 0, 1)
                h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
                
                corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
                c5_pert = np.max(corr_pert) * dx
                
                if c5_pert < best_c5:
                    h = h_pert.copy()
                    best_c5 = c5_pert
                    last_improvement = i
        
        # Periodic validation
        if i % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
    
    # Final verification
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
