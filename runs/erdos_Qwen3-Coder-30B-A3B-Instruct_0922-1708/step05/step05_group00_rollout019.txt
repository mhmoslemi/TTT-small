from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth approximation to max for gradient computation
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    def compute_overlap(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(corr)
    
    def compute_smooth_overlap(h_vec, beta=10.0):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return smooth_max(corr, beta)
    
    def compute_gradient(h_vec, beta=10.0):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        # Find the index of maximum
        max_idx = np.argmax(corr)
        # Compute finite difference gradient
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
        return grad
    
    # Adam optimizer
    def adam_update(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        return h_new, m, v
    
    # Optimizer
    max_iter = 5000
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    patience = 0
    last_c5 = float('inf')
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        t += 1
        
        # Compute current overlap
        current_c5 = compute_overlap(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                break
                
        # Compute gradient using finite differences
        grad = compute_gradient(h)
        
        # Update using Adam
        h, m, v = adam_update(h, grad, m, v, t, lr, beta1, beta2, eps)
        
        # Enforce bounds and normalization
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Update learning rate
        lr = max(1e-6, lr * 0.9995)
        
    # Final evaluation
    if best_h is None:
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        final_c5 = np.max(correlation)
        return (h.tolist(), final_c5, n_points)
    else:
        correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = np.max(correlation)
        return (best_h.tolist(), final_c5, n_points)
