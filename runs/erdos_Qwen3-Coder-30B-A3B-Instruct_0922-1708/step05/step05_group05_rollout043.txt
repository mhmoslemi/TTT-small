import numpy as np
from scipy.stats import halton
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    generations = 600
    mutation_rate = 0.1
    crossover_rate = 0.8
    elite_size = 5
    
    # Generate Halton sequence for initialization
    def generate_halton_points(n, dim):
        seq = halton.Halton(dim)
        return seq.random(n)
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
    else:
        # Use Halton sequences for initialization
        halton_points = generate_halton_points(pop_size - 1, 1)
        for i in range(pop_size - 1):
            h = np.clip(halton_points[i, 0] * 2, 0, 1)
            h = np.full(n_points, h)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Add one good initial approach
    h_block = np.zeros(n_points)
    h_block[:n_points//2] = 2.0
    h_block = h_block * (n_points / 2) / np.sum(h_block)
    population.append(h_block)
    
    # Evaluate initial population
    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    fitnesses = [evaluate_fitness(h) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # GA main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Update global best
        if sorted_fitnesses[0] < best_fitness:
            best_fitness = sorted_fitnesses[0]
            best_h = sorted_population[0].copy()
        
        # Create new population
        new_population = []
        
        # Elitism
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Tournament selection and crossover
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 5
            selected_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitnesses[i] for i in selected_indices]
            winner_index = selected_indices[np.argmin(tournament_fitness)]
            
            parent1 = population[winner_index]
            
            # Another parent
            selected_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [fitnesses[i] for i in selected_indices]
            winner_index = selected_indices[np.argmin(tournament_fitness)]
            
            parent2 = population[winner_index]
            
            # Crossover
            if np.random.random() < crossover_rate:
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for child in [child1, child2]:
                if np.random.random() < mutation_rate:
                    idx = np.random.randint(0, n_points)
                    child[idx] += np.random.normal(0, 0.05)
                    
                # Repair constraints
                child = np.clip(child, 0, 1)
                child_sum = np.sum(child)
                if abs(child_sum - n_points/2) > 1e-6:
                    child = child * (n_points/2) / child_sum
                
                new_population.append(child)
                
                if len(new_population) >= pop_size:
                    break
        
        # Update population
        population = new_population[:pop_size]
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Adapt mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate *= 0.99
    
    # Final refinement with local search
    best_h_refined = best_h.copy()
    best_fitness_refined = best_fitness
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Gradient-like refinement by perturbing indices
        h_test = best_h_refined.copy()
        idx = np.random.randint(0, n_points)
        step = np.random.normal(0, 0.01)
        
        h_test[idx] += step
        h_test = np.clip(h_test, 0, 1)
        h_test_sum = np.sum(h_test)
        h_test = h_test * (n_points/2) / h_test_sum
        
        new_fitness = evaluate_fitness(h_test)
        if new_fitness < best_fitness_refined:
            best_fitness_refined = new_fitness
            best_h_refined = h_test.copy()
    
    # Final validation
    final_corr = np.correlate(best_h_refined, 1 - best_h_refined, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    return (best_h_refined.tolist(), final_fitness, n_points)
