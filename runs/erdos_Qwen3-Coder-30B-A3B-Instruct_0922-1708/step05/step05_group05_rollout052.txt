import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    alpha = 0.01
    epsilon = 1e-8
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation and c5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project h_plus to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - current_c5) / eps
        
        # Compute candidate step
        h_candidate = h - alpha * grad
        
        # Project candidate to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate candidate
        corr_candidate = correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr_candidate) * dx
        
        # Accept or reject step
        if candidate_c5 < current_c5:
            h = h_candidate.copy()
            current_c5 = candidate_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
            alpha = min(alpha * 1.1, 0.1)  # Increase learning rate
        else:
            alpha = max(alpha * 0.5, 1e-6)  # Decrease learning rate
            
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0 and iteration > 0:
            perturbation_magnitude = max(0.001, 0.01 * (1 - (time.time() - start_time) / budget_s))
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            
        # Occasionally increase alpha to explore
        if iteration % 200 == 0 and alpha < 0.05:
            alpha *= 1.5
    
    # Final validation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
