import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.9):
    offspring = parent1.copy()
    for i in range(len(parent1)):
        if np.random.rand() < 0.5:
            lb = min(parent1[i], parent2[i])
            ub = max(parent1[i], parent2[i])
            range_diff = ub - lb
            if range_diff > 0:
                offspring[i] = lb + alpha * range_diff * np.random.rand()
    return offspring

def mutate(individual, mutation_rate=0.1, mutation_strength=0.03):
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, mutation_strength)
    return mutated

def gradient_refinement(h, dx, steps=50, lr=0.001):
    h = h.copy()
    for _ in range(steps):
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = compute_fitness(h_plus, dx)
            grad[j] = (c5_plus - compute_fitness(h, dx)) / eps
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
    return h
