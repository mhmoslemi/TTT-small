def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    max_iter = 200

    if 'initial_h_values' in globals():

            x_orig = np.linspace(0, 2, len(h))

    else:
        h = np.random.uniform(0, 0.5, n_points)

    h = h * (n_points / 2) / np.sum(h)

    low = 0.0
    high = 0.5
    best_c5 = 1.0
    best_h = h.copy()
    
    # Sample shifts
    shifts = np.linspace(-2, 2, 50)
    
    def compute_overlap(h_vector):
        """Compute max overlap with current h"""
        overlaps = []
        for shift in shifts:
            if shift >= 0:
                # Convolution with shifted version
                h_shift = np.concatenate([np.zeros(int(shift/dx)), h_vector[:-int(shift/dx)]])
            else:
                h_shift = np.concatenate([h_vector[-int(-shift/dx):], np.zeros(int(-shift/dx))])
            overlap = np.sum(h_vector * h_shift) * dx
            overlaps.append(overlap)
        return np.max(overlaps)
    
    def repair_h(h_vector):
        """Repair h to satisfy constraints"""
        h_vector = np.clip(h_vector, 0, 1)
        h_sum = np.sum(h_vector)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_vector = h_vector * (n_points / 2) / h_sum
        return h_vector
    
    # Gradient descent for refinement
    def refine_with_gradients(h_init, max_iter=50, lr=0.001):
        h = h_init.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            current_c5 = compute_overlap(h)
            
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus)
                c5_plus = compute_overlap(h_plus)
                grad[i] = (c5_plus - current_c5) / eps
            
            h -= lr * grad
            h = repair_h(h)
            
            c5_new = compute_overlap(h)
            if c5_new >= current_c5:
                break
                
        return h
    
    # Try different initialization strategies
    
    # Strategy 1: Binary jump pattern
    h1 = np.zeros(n_points)

    h1[n_points//4:3*n_points//4] = 0
    h1[3*n_points//4:] = 1
    h1 = h1 * (n_points / 2) / np.sum(h1)
    h1 = refine_with_gradients(h1)
    c5_1 = compute_overlap(h1)
    
    if c5_1 < best_c5:

        best_h = h1.copy()
    
    # Strategy 2: Spike pattern
    h2 = np.zeros(n_points)
    h2[n_points//3] = 1
    h2[2*n_points//3] = 1

    h2 = h2 * (n_points / 2) / np.sum(h2)
    h2 = refine_with_gradients(h2)
    c5_2 = compute_overlap(h2)
    
    if c5_2 < best_c5:

        best_h = h2.copy()
    
    # Strategy 3: Random with repair
    h3 = np.random.uniform(0, 1, n_points)

    h3 = refine_with_gradients(h3)
    c5_3 = compute_overlap(h3)
    
    if c5_3 < best_c5:
        best_c5 = c5_3
        best_h = h3.copy()
    
    # Use best of initial strategies
    h = best_h.copy()
    best_c5 = compute_overlap(h)
    
    # Continue with gradient refinement
    h_refined = refine_with_gradients(h, max_iter=100)
    refined_c5 = compute_overlap(h_refined)
    
    if refined_c5 < best_c5:
        best_c5 = refined_c5
        best_h = h_refined.copy()
    
    # Refine further using simple line search
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        # Do a small step in gradient direction
        grad = np.zeros_like(best_h)
        eps = 1e-5
        base_c5 = compute_overlap(best_h)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            c5_plus = compute_overlap(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        best_h -= 0.001 * grad
        best_h = repair_h(best_h)
        
        new_c5 = compute_overlap(best_h)
        if new_c5 >= base_c5:
            break
        else:
            base_c5 = new_c5
    
    return (best_h.tolist(), compute_overlap(best_h), n_points)
