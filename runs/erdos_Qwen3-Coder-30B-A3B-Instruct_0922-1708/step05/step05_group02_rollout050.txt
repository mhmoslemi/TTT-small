import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Random initialization
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 0.1 * current_c5
    T = T0
    alpha = 0.99
    max_iter = 10000
    perturbation_rate = 0.1
    stagnation_limit = 100
    stagnation_count = 0
    eps = 1e-8
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the solution
        h_new = h.copy()
        num_perturb = max(1, int(perturbation_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for i in indices:
            # Add small random perturbation
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] = h_new[i] + delta
            # Ensure feasibility
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Rescale to maintain integral constraint
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject the new solution
        delta_c5 = new_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / (T + eps)):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        T = T * alpha
        if stagnation_count > stagnation_limit:
            T = max(T * 0.95, 1e-10)
            stagnation_count = 0
            
        # Adjust perturbation rate
        perturbation_rate = max(0.01, perturbation_rate * 0.999)
    
    # Final refinement with gradient descent
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps_step = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps_step
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps_step
            
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
