import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-scale approach
    scales = [10, 20, 40, 80, 100]
    start_time = time.time()
    
    # Convex relaxation helper
    def solve_convex_relaxation(h_start, n):
        # Set up convex optimization
        h = cp.Variable(n)
        # Objective: minimize the average overlap over central lags
        # We'll approximate over lags ±1 to ±n//4
        avg_overlap = 0.0
        count = 0
        
        # Use a subset of lags to make it tractable
        max_lag = n // 4
        if max_lag < 1:
            max_lag = 1
        
        for k in range(1, max_lag + 1):
            # Use convolution of h and (1-h) efficiently
            lagged_h = cp.hstack([cp.Variable(k), h])
            lagged_one_minus_h = cp.hstack([cp.Variable(k), 1-h])
            # Approximate average overlap over lag k
            overlap_term = cp.sum(cp.multiply(lagged_h[:-k], lagged_one_minus_h[k:]))
            avg_overlap += overlap_term
            count += 1
        
        if count > 0:
            avg_overlap /= count
        
        # Soft constraints
        lambda_reg = 0.1
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n / 2
        ]
        
        # Add soft penalties for boundary violations
        penalty = lambda_reg * (cp.sum(cp.pos(-h)) + cp.sum(cp.pos(h - 1)))
        objective = cp.Minimize(avg_overlap + penalty)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=1000)
            if prob.status == cp.OPTIMAL:
                return h.value
        except:
            pass
        return h_start
    
    # Multi-scale optimization
    prev_h = h.copy()
    for scale in scales:
        if time.time() - start_time > budget_s - 5:
            break
        curr_n = scale
        dx = 2.0 / curr_n
        
        if curr_n == 100:
            h = prev_h.copy()
        else:
            # Upsample the previous solution
            x_prev = np.linspace(0, 2, len(prev_h))
            x_curr = np.linspace(0, 2, curr_n)
            h = np.interp(x_curr, x_prev, prev_h)
            
        # Apply convex relaxation if not sparse
        if curr_n <= 100:  # Relaxation only up to 100 points
            h_relaxed = solve_convex_relaxation(h, curr_n)
            if h_relaxed is not None:
                h = h_relaxed
        
        # Project to feasibility
        h = np.clip(h, 0, 1)
        h = h * (curr_n / 2) / np.sum(h)
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        prev_h = h.copy()
    
    # Local refinement with gradient descent using finite differences
    if time.time() - start_time < budget_s - 10:
        h_refined = best_h.copy()
        lr = 0.001
        eps = 1e-4
        
        for _ in range(50):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = np.zeros_like(h_refined)
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
            
            h_refined = h_refined - lr * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_refined.copy()
    
    # Final validation
    h_final = best_h.copy()
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_final.tolist(), final_c5, n_points)
