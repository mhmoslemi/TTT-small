import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

    dx = 2.0 / n_points
    start_time = time.time()

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_best = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_best = np.random.rand(n_points)
        h_best = h_best * (n_points / 2) / np.sum(h_best)
    
    c5_best = evaluate_h(h_best, dx)
    
    # Gradient descent with periodic perturbations
    max_iter = 10000
    learning_rate = 0.001
    perturb_rate = 0.1
    perturb_freq = 100
    stagnation_threshold = 100
    
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        c5_current = evaluate_h(h_best, dx)
        
        for j in range(n_points):
            h_plus = h_best.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[j] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h_best - learning_rate * grad
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement or perturb
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            last_improvement = i
        else:
            # Occasionally perturb to escape local minima
            if i % perturb_freq == 0:
                perturb_count = int(perturb_rate * n_points)
                indices = np.random.choice(n_points, perturb_count, replace=False)
                for idx in indices:
                    h_new[idx] = np.random.rand()
                h_new = repair_h(h_new, n_points)
                c5_new = evaluate_h(h_new, dx)
                if c5_new < c5_best:
                    h_best = h_new.copy()
                    c5_best = c5_new
                    last_improvement = i
                
        # Early stopping if no improvement in a while
        if i - last_improvement > stagnation_threshold:
            break
    
    # Final local search
    for _ in range(500):

        grad = np.zeros(n_points)
