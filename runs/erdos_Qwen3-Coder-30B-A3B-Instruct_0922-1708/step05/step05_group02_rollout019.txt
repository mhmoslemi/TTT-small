import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate Lévy distributed random variables
    sigma1 = np.power(scale, 1/alpha)
    sigma2 = np.power(scale, 1/alpha)
    u = np.random.normal(0, sigma1, size)
    v = np.random.exponential(1, size)
    return u / np.power(np.abs(v), 1/alpha)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    num_candidates = 50
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add more candidates
    for _ in range(num_candidates - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Evaluate initial candidates
    fitness_values = [evaluate_h(candidates[i], dx) for i in range(len(candidates))]
    
    # Find best initial solution
    best_idx = np.argmin(fitness_values)
    best_h = candidates[best_idx].copy()
    best_value = fitness_values[best_idx]
    
    # Parameters for Lévy flight
    alpha = 1.5
    scale = 0.5
    perturbation_rate = 0.1
    top_candidates = 10
    max_iter = 1000
    
    # Main optimization loop
    no_improvement_count = 0
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new candidates using Lévy flight
        new_candidates = []
        for i in range(len(candidates)):
            # Generate Lévy perturbation
            perturbation = levy_flight(n_points, alpha, scale)
            new_candidate = candidates[i] + perturbation_rate * perturbation
            
            # Repair and normalize
            new_candidate = repair_h(new_candidate, n_points)
            new_candidates.append(new_candidate)
        
        # Evaluate new candidates
        new_fitness = [evaluate_h(c, dx) for c in new_candidates]
        
        # Update with better candidates
        for i in range(len(new_candidates)):
            if new_fitness[i] < fitness_values[i]:
                candidates[i] = new_candidates[i].copy()
                fitness_values[i] = new_fitness[i]
                if new_fitness[i] < best_value:
                    best_h = new_candidates[i].copy()
                    best_value = new_fitness[i]
                    no_improvement_count = 0
        
        # Gradient descent on top candidates
        if iter_count % 5 == 0:
            top_indices = np.argsort(fitness_values)[:top_candidates]
            for idx in top_indices:
                h = candidates[idx].copy()
                current_c5 = fitness_values[idx]
                
                # Compute finite differences gradient
                grad = np.zeros(n_points)
                eps = 1e-6
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Update with gradient descent
                h = h - 0.001 * grad
                h = repair_h(h, n_points)
                
                # Evaluate and update if better
                corr = correlate(h, 1 - h, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < fitness_values[idx]:
                    candidates[idx] = h.copy()
                    fitness_values[idx] = new_c5
                    if new_c5 < best_value:
                        best_h = h.copy()
                        best_value = new_c5
                        no_improvement_count = 0
        
        no_improvement_count += 1
        if no_improvement_count > 100:
            break
    
    # Final refinement with intensive gradient descent on best solution
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_value = np.max(corr) * dx
    
    return (best_h.tolist(), best_value, n_points)
