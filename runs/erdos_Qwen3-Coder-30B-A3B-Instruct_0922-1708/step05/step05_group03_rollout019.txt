import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Precompute correlation indices
    max_lag = n_points - 1
    
    # Simulated Annealing parameters
    T0 = 10.0
    cooling_rate = 0.9999
    min_T = 1e-8
    
    # Gradient descent parameters
    lr = 0.001
    max_grad_iter = 50
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def project_to_feasible(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def evaluate(h_vals):
        return compute_c5(h_vals)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    
    # Simulated Annealing
    T = T0
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        if T < min_T:
            break
            
        # Perturb solution
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        h_candidate = project_to_feasible(h_candidate)
        
        # Evaluate candidate
        c5_candidate = evaluate(h_candidate)
        
        # Accept or reject
        delta = c5_candidate - evaluate(h)
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
        
        # Cool down
        T *= cooling_rate
    
    # Gradient descent refinement on best solution
    h_ref = best_h.copy()
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h_ref)
        
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = evaluate(h_plus)
            
            grad[i] = (c5_plus - best_c5) / eps
            
        # Update
        h_ref = h_ref - lr * grad
        h_ref = project_to_feasible(h_ref)
        
        # Recompute
        new_c5 = evaluate(h_ref)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_ref.copy()
        else:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
