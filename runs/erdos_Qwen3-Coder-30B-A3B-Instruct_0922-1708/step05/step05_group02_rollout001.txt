import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = repair_h(h, n_points)
    else:
        h = np.random.rand(n_points)
        h = repair_h(h, n_points)
    
    # Initial evaluation
    current_c5 = evaluate_h(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T0 = 0.1 * current_c5
    T = T0
    alpha = 0.99
    max_iter = 1000
    min_T = 1e-8
    no_improve_count = 0
    max_no_improve = 100
    
    # Perturbation strategy
    perturb_rate = 0.1
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new neighbor
        h_new = h.copy()
        n_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for idx in indices:
            # Perturb by small amount
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
            
        # Repair and rescale
        h_new = repair_h(h_new, n_points)
        
        # Evaluate neighbor
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
            
        # Cooling
        T = max(min_T, T * alpha)
        
        # Adaptive cooling: if no improvement for a while, cool faster
        if no_improve_count > 50:
            T = max(min_T, T * (alpha * 1.1))
            
        # Occasionally reset cooling
        if iter_count % 100 == 0 and T > min_T:
            T = min(T0, T * 1.1)
    
    # Final gradient descent refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
