import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to 1/2
    h = h * (0.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation and find minimum
        corr = signal.correlate(h, 1-h, mode='full')
        min_k = np.argmin(corr)
        min_val = corr[min_k]
        
        # Gradient ascent: for min_k, compute gradient of min_val
        # The gradient w.r.t. h[j] when min_val = sum_k h[k]*h[k+j] or similar
        # The derivative of h[j] in the term h[j]*h[j+k] is h[j+k]
        # But since we want to minimize the overlap, we multiply by -1
        # We can approximate gradient by finite differences or analytical
        
        # Computing gradient analytically based on corr computation:
        # Let c[k] = sum_{i} h[i] * (1 - h[i+k])
        # So c'[h[j]] = -1 * (h[j+k] + h[j-k] if both exist else 0)  
        # The gradient direction should be opposite to minimize c[k]
        # But since there may be other terms affecting the minimum, we need to 
        # be more careful. We'll focus on finite differences.
        
        # More accurate gradient estimate for the most sensitive part
        # Loosen than gradient, but directly work on objectives
        # Compute approximate gradient by finite difference for current angle
        
        # The full correlation is: c[k] = \sum_i h[i] * (1 - h[i+k])
        # Used to compute minimum
        # So when we increase h[j] by eps, the effect on c[k] mainly comes from
        # - the term h[j] * (1 - h[j+k]) for k such that j+k in valid range
        # - the term h[j+k] * (1 - h[j+k+k]) for k such that j+k+k in valid range  
        # etc.
        # But this is a bit messy to get right
        # Let's use a more direct approach:
        # We can also treat the problem of minimizing max_k c[k]
        # So we are essentially looking at that particular minimum
        # Gradients would be:
        # d/dh[j] c[k] = 1 if j == k => 0
        # d/dh[j] c[k] = - (1 - h[j+k]) if j+k is valid
        # d/dh[j] c[k] = - h[j-k] if j-k is valid
        # This is approximate how it modifies the k-th term of correlation...
        # Hmm, actually the correlation matrix has indices from reversed h
        # Let's simplify and try gradient descent as a local search in optimization space
        
        # Instead compute a proper gradient
        # For minimizing max_k c[k], there's a dominating term where we 
        # want to move h such that the min value is pushed up, which means
        # modifying h in directions that most affect that min k component.
        # Let's use projected gradient descent: 
        # Compute finite differences for gradient approximation
        h_grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute a derivative-like vector for min_k term
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (0.5 / np.sum(h_plus))
            
            corr_plus = signal.correlate(h_plus, 1 - h_plus, mode='full')
            corr_base = signal.correlate(h, 1 - h, mode='full')
            
            der = (corr_plus[min_k] - corr_base[min_k]) / eps
            h_grad[j] = -der  # Minimize => use negative gradient
            
        # Update h along gradient direction
        h_new = h + learning_rate * h_grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (0.5 / np.sum(h_new))
        
        # Update
        h = h_new
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Periodic perturbations to escape local minima
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h_perturbed = h + noise
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (0.5 / np.sum(h_perturbed))
            
            c5_perturbed = compute_c5(h_perturbed, dx)
            if c5_perturbed < best_c5:
                best_c5 = c5_perturbed
                best_h = h_perturbed.copy()
            h = h_perturbed
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = signal.correlate(h, 1-h, mode='full')
    return np.max(corr) * dx
