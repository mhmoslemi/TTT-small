import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Multi-start approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Number of restarts
    n_starts = 5
    
    # Generate different initializations
    initializations = [h.copy()]
    
    # Add random initializations
    for _ in range(n_starts - 1):
        h_rand = np.random.uniform(0, 1, n_points)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        h_rand = np.clip(h_rand, 0, 1)
        initializations.append(h_rand)
    
    # For each initialization, try convex relaxation
    for h_start in initializations:
        h = h_start.copy()
        
        # Convex relaxation approach: Formulate QCQP
        # but actually we'll use a different approach - do not solve QP
        # instead, apply a more practical refinement step
        
        # Refinement with L-BFGS
        def obj_func(h_vec):
            # This should be the C5 value
            h_local = h_vec.copy()
            h_local = np.clip(h_local, 0, 1)
            h_local = h_local * (n_points / 2) / np.sum(h_local)
            correlation = np.correlate(h_local, 1 - h_local, mode='full') * dx
            return np.max(correlation)
        
        def callback_func(h_vec):
            pass  # No action needed
            
        # Optimize using L-BFGS-B with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        optimal_result = minimize(
            obj_func,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 2000, 'ftol': 1e-9, 'gtol': 1e-9},
            callback=callback_func
        )
        
        # Recover optimized solution and evaluate
        if optimal_result.success or optimal_result.fun < best_c5:
            h_opt = optimal_result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full') * dx
            current_c5 = np.max(correlation)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_opt.copy()
        
        # Check time
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final refinement if needed
    h = best_h.copy()
    obj_func = lambda h_vec: np.max(np.correlate(h_vec, 1 - h_vec, mode='full') * dx)
    
    bounds = [(0, 1) for _ in range(n_points)]
    optimal_result = minimize(
        obj_func,
        h,
        method='L-BFGS-B',
        bounds=bounds,
        options={'maxiter': 1000, 'ftol': 1e-10, 'gtol': 1e-10}
    )
    
    # Final validation and return
    h_final = optimal_result.x
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    correlation = np.correlate(h_final, 1 - h_final, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (h_final.tolist(), final_c5, n_points)
