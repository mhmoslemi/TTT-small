import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    tolerance = 1e-6
    perturb_freq = 50
    perturb_magnitude = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: piecewise constant
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def project(h):
        # Project to feasible region
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if abs(sum_h - n_points / 2) > 1e-6:
            # Adjust to meet sum constraint
            factor = (n_points / 2) / sum_h
            h = h * factor
        return h
    
    best_h = h.copy()
    best_c5 = evaluate(h)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h)
        
        # Estimate gradient using finite differences
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - base_c5) / eps
        
        # Update step
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = project(h_new)
        
        # Evaluate new solution
        c5_new = evaluate(h_new)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        else:
            # Reduce step size
            alpha *= 0.5
            if alpha < 1e-8:
                alpha = 1e-6
                
        # Apply perturbation to escape local minima
        if i % perturb_freq == 0 and i > 0:
            perturb = np.random.normal(0, perturb_magnitude, n_points)
            h = h + perturb
            h = project(h)
            
            # Recompute the best solution so far
            new_c5 = evaluate(h)
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
