import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    lr = 0.001
    mu = 0.1
    mu_update = 1.2
    perturb_fraction = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize z (binary approximation)
    z = (h > 0.5).astype(float)
    
    # Current best
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Find best lag
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        # For lag k, derivative is (1 - h[i] - h[i+k])
        grad = np.zeros_like(h)
        if max_lag >= 0:
            # h[i] contributes to (1 - h[i] - h[i+k])
            grad[:-max_lag] -= 1
            grad[max_lag:] -= 1
        else:
            # h[i] contributes to (1 - h[i] - h[i+k]) where k < 0
            grad[-max_lag:] -= 1
            grad[:max_lag] -= 1
            
        # Add penalty for deviation from binary
        penalty_grad = mu * (h - z)
        grad += penalty_grad
        
        # Gradient descent step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to required sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update binary approximation
        z_new = (h_new > 0.5).astype(float)
        
        # Repair step
        h = h_new.copy()
        z = z_new.copy()
        
        # Test for improvement
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update penalty parameter
        mu *= mu_update
        
        # Add some random noise occasionally
        if iteration % 100 == 0:
            noise = np.random.randn(n_points) * perturb_fraction
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
