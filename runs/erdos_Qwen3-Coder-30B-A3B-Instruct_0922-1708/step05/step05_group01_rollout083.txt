import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to exact sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    learning_rate = 1e-3
    line_search_factor = 0.5
    max_iterations = 5000
    convergence_threshold = 1e-6
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Gradient-based optimization
    for iteration in range(max_iterations):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_c5 = corr[max_lag_idx] * dx
        
        # Identify the lag that achieves max
        # index 0 corresponds to lag=-(n_points-1), ..., n_points-1 corresponds to lag=0, ..., 2*n_points-2 corresponds to lag=n_points-1
        lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for the max lag
        grad = np.zeros(n_points)
        if lag >= 0:
            # h[i] * (1 - h[i + lag]) contributes to overlap
            for i in range(n_points - lag):
                # gradient of h[i] w.r.t. h[i] is 1 - h[i + lag] 
                grad[i] += (1 - h[i + lag])
                # gradient of h[i + lag] w.r.t. h[i] is -h[i]
                if i + lag < n_points:
                    grad[i + lag] -= h[i]
        else:
            # lag < 0
            for i in range(n_points + lag):
                # gradient of h[i] w.r.t. h[i] is 1 - h[i - lag] 
                grad[i] += (1 - h[i - lag])
                # gradient of h[i - lag] w.r.t. h[i] is -h[i]
                if i - lag < n_points:
                    grad[i - lag] -= h[i]
        
        # Scale by 2/n
        grad *= 2 * dx / n_points
        
        # Line search
        alpha = learning_rate
        improved = False
        for _ in range(10):  # Limit line search steps
            h_new = h - alpha * grad
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to sum(h)=n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Check new C5
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < max_c5:
                h = h_new
                max_c5 = new_c5
                improved = True
                break
            else:
                alpha *= line_search_factor
                
        # Update best solution
        if max_c5 < best_c5:
            best_c5 = max_c5
            best_h = h.copy()
            
        # Convergence check
        if not improved and alpha < 1e-10:
            break
            
        # Adaptive learning rate
        if improved:
            learning_rate *= 1.1
        else:
            learning_rate *= 0.9
            
        # Early stopping based on convergence
        if iteration > 100 and abs(max_c5 - current_c5) < convergence_threshold:
            break
            
        current_c5 = max_c5
    
    # Final evaluation
    try:
        corr_final = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr_final) * dx
    except:
        final_c5 = best_c5
    
    # Ensure valid bounds
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
