import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_factor = 1.2
    penalty_min = 1e-4
    penalty_init = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize penalty
    penalty_weight = penalty_init
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent optimization
    plateau_count = 0
    current_lr = learning_rate
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        c5 = np.max(corr) * dx
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Adaptive learning rate
        if plateau_count > plateau_patience:
            current_lr *= learning_rate_decay
            plateau_count = 0
        
        # Compute gradient for max correlation lag
        max_lag = max_corr_idx - (len(h) - 1)
        
        # Construct gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute overlap contribution for this point
            # Distinguish cases: positive (when h[i] > 0) and negative (when h[i] < 1)
            # For the negative lag case where k is negative (k < 0):
            # We are looking at h[i] * (1 - h[i + |k| - 1]) 
            # where k = -|k| and |k| = 1 - (i - (i + |k| - 1)) = 1 + i - (i + |k| - 1) = 2 - |k|
            # Actually, k = max_lag = max_corr_idx - (n_points - 1)
            # Let `k` be such that `h[i] * (1 - h[i+k])` contributes to the overlap
            if max_lag > 0:
                # i + k is valid
                if i + max_lag < n_points:
                    grad[i] += 1 - h[i + max_lag]
                # handle backward references
                if i - max_lag >= 0:
                    grad[i] += 1 - h[i - max_lag]
            elif max_lag < 0:
                # i + k is valid with modulo
                if i - abs(max_lag) < n_points:
                    grad[i] += 1 - h[i - abs(max_lag)]
                if i + abs(max_lag) >= 0:
                    grad[i] += 1 - h[i + abs(max_lag)]
            else:
                # k == 0 case
                grad[i] += 1 - h[i]
        
        # Compute penalty gradient components
        penalty_grad = np.zeros(n_points)
        # If any h[i] < 0, penalty_grad[i] = -2 * mu * h[i] 
        # If any h[i] > 1, penalty_grad[i] = 2 * mu * (h[i] - 1)
        # If sum(h) != n/2, penalty_grad[i] = -mu * (sum(h) - n/2) * (1/n)
        sum_h = np.sum(h)
        if sum_h < n_points/2 - 1e-8:
            # penalty on sum being too small
            penalty_grad += -penalty_weight * (sum_h - n_points/2) / n_points
        elif sum_h > n_points/2 + 1e-8:
            # penalty on sum being too large
            penalty_grad += penalty_weight * (sum_h - n_points/2) / n_points
            
        # Clip bounds
        penalty_grad[h < 0] += -2 * penalty_weight * h[h < 0]
        penalty_grad[h > 1] += 2 * penalty_weight * (h[h > 1] - 1)
        
        # Combine and apply
        total_grad = grad + penalty_grad
        
        # Update step with current learning rate
        h_new = h - current_lr * total_grad
        
        # Repair feasibility
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Update h
        h = h_new
        
        # Adjust penalty based on violations
        violation = 0
        if h_sum < n_points/2 - 1e-8:
            violation += (n_points/2 - h_sum) ** 2
        elif h_sum > n_points/2 + 1e-8:
            violation += (h_sum - n_points/2) ** 2
            
        for i in range(n_points):
            if h[i] < 0:
                violation += h[i] ** 2
            elif h[i] > 1:
                violation += (h[i] - 1) ** 2
                
        if violation > 1e-6:
            penalty_weight = min(penalty_weight * penalty_factor, 1000)
        else:
            penalty_weight = max(penalty_weight / penalty_factor, penalty_min)
            
        # Stochastic perturbation if needed
        if plateau_count > 100 and np.random.rand() < 0.1:
            perturb = np.random.randn(n_points) * perturbation_fraction
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
