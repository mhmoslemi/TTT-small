import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define fine grid size
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize from parent or construct a good starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create a better initial guess using a structured approach
        h = np.ones(n_points) * 0.5
        # Apply some smoothing and structure to the initial guess
        t = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * t)  # Sine wave modulation
        h = np.clip(h, 0, 1)

    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Define smooth maximum approximation for gradient descent
    def smooth_max(arr, beta=10.0):
        """Approximate maximum using soft maximum"""
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def compute_c5(h):
        """Compute C5 using correlation"""
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    def compute_gradient(h, eps=1e-5):
        """Compute gradient via finite difference"""
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Optimizer parameters
    max_iter = 5000
    lr = 0.01
    patience = 0
    patience_limit = 1000
    
    # Optimizer loop
    for iteration in range(max_iter):

        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient
        grad = compute_gradient(best_h)
        
        # Update with gradient descent
        new_h = best_h - lr * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution  
        try:
            c5 = compute_c5(new_h)
        except:
            c5 = float('inf')
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = new_h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping with patience
        if patience > patience_limit:
            break
            
        # Update learning rate with cosine annealing
        lr = 0.01 * (0.5 * (1 + math.cos(math.pi * iteration / max_iter)))
    
    # Final validation
    try:
        final_c5 = compute_c5(best_h)
    except:
        final_c5 = float('inf') 
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
