import numpy as np
from scipy.stats import halton
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population using Halton sequence
    def init_population():
        pop = []
        # Generate halton points
        seq = halton.Halton(1)
        halton_points = seq.random(population_size * n_points).reshape(population_size, n_points)
        
        for i in range(population_size):
            # Scale and clip halton points to [0, 1]
            h = np.clip(halton_points[i], 0, 1)
            # Adjust to meet sum constraint
            h = h * (n_points / 2) / np.sum(h)
            pop.append(h)
        return pop
    
    # If initial_h_values available, use it as first individual
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population = [h_init]
        population.extend(init_population()[1:])
    else:
        population = init_population()
    
    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Tournament selection
    def tournament_selection(pop, fitnesses):
        selected_idx = np.random.choice(len(pop), tournament_size)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return pop[best_idx]
    
    # Crossover
    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    # Mutation
    def mutate(h, rate):
        h = h.copy()
        for i in range(n_points):
            if np.random.random() < rate:
                delta = np.random.normal(0, 0.05)
                h[i] += delta
        # Repair constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Main GA loop
    best_h = None
    best_fitness = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            fitness = evaluate_fitness(h)
            fitnesses.append(fitness)
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = h.copy()
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        # Adaptive mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate *= 0.99
    
    # Final refinement with local search
    def local_optimize(h, max_iter=100):
        h = h.copy()
        for _ in range(max_iter):
            # Finite difference gradient estimation
            grad = np.zeros_like(h)
            base_fitness = evaluate_fitness(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                fitness_plus = evaluate_fitness(h_plus)
                grad[i] = (fitness_plus - base_fitness) / eps
            
            # Update with learning rate
            h -= 0.001 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            if evaluate_fitness(h) >= base_fitness:
                break
                
        return h
    
    # Refine best solution
    if best_h is not None:
        refined_h = local_optimize(best_h)
        refined_fitness = evaluate_fitness(refined_h)
        if refined_fitness < best_fitness:
            best_h = refined_h
            best_fitness = refined_fitness
    
    return (best_h.tolist(), best_fitness, n_points)
