import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Parameters
    n_mutations = 100
    patience = 100
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 10:
        # Generate mutations
        new_solutions = []
        for _ in range(n_mutations):
            h_candidate = best_h.copy()
            # Flip 10% of elements randomly
            flip_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            h_candidate[flip_indices] = 1 - h_candidate[flip_indices]
            h_candidate = repair_h(h_candidate, n_points)
            new_solutions.append(h_candidate)
        
        # Evaluate all new solutions
        for h in new_solutions:
            c5 = evaluate_h(h, dx)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        if no_improvement_count >= patience:
            # Restart with random solution
            h_init = np.random.rand(n_points)
            h_init = repair_h(h_init, n_points)
            best_h = h_init.copy()
            best_c5 = evaluate_h(best_h, dx)
            no_improvement_count = 0
    
    # Final gradient descent refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
