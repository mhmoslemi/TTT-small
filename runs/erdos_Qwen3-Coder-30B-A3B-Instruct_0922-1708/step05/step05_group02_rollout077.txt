import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize
    current_h = h_init.copy()
    current_h = repair_h(current_h, n_points)
    current_c5 = evaluate_h(current_h, dx)
    
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Simulated Annealing parameters
    T = 0.1 * current_c5 if current_c5 > 0 else 0.1
    alpha = 0.99
    max_iter = 10000
    iterations_without_improvement = 0
    max_no_improvement = 100
    
    # Neighbors to perturb at each step
    perturb_size = max(1, n_points // 10)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create a neighbor solution
        new_h = current_h.copy()
        
        # Randomly choose indices to perturb
        indices = np.random.choice(n_points, size=perturb_size, replace=False)
        for i in indices:
            # Add small random change
            delta = np.random.uniform(-0.1, 0.1)
            new_h[i] += delta
            
        # Repair neighbor
        new_h = repair_h(new_h, n_points)
        
        # Evaluate neighbor
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject the neighbor
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = new_h
            current_c5 = new_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 1
        else:
            iterations_without_improvement += 1
            
        # Cooling schedule
        T *= alpha
        
        # Adaptive cooling: increase cooling if no improvement for a while
        if iterations_without_improvement > 50:
            T *= 0.995
            iterations_without_improvement = 0
            
        # Early termination
        if iterations_without_improvement > max_no_improvement:
            break
    
    # Final polish with gradient descent
    h_polished = best_h.copy()
    current_c5_polished = best_c5
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)

            h_plus = h_polished.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5_polished) / eps
            
        h_polished = h_polished - 0.001 * grad
        h_polished = repair_h(h_polished, n_points)
        corr = correlate(h_polished, 1 - h_polished, mode='full')
        current_c5_polished = np.max(corr) * dx
        
        if current_c5_polished < best_c5:
            best_h = h_polished.copy()
            best_c5 = current_c5_polished
    
    return (best_h.tolist(), best_c5, n_points)
