    else:
        h_init = np.random.uniform(0.1, 0.9, n_points)

    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_best = h_init.copy()
    c5_best = float('inf')
    
    # Binary search for the optimal C5
    low, high = 0.0, 0.5
    best_mu = low
    
    # Grid of shifts k to test
    k_grid = np.linspace(-2, 2, 50)
    
    for _ in range(max_search_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        mu = (low + high) / 2
        feasible = True
        
        # Try to solve feasibility problem with this mu
        # Use a gradient-based approach to try to find a feasible solution
        h_trial = h_init.copy()
        for _ in range(max_grad_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute correlation
            corr = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(corr) * dx
            
            # Check constraints: we want all correlations to be >= mu
            # We only check a representative sample of k values
            valid = True
            for k_idx in range(0, len(k_grid), max(1, len(k_grid)//10)):
                k = k_grid[k_idx]
                # This is approximate due to discretization
                shifts = np.arange(n_points, dtype=int)
                if k >= 0:
                    # For positive k, h(x) -> h(x+k) means x -> x+k

                    start1 = 0
                    end1 = int(n_points - k/dx)
                    start2 = int(k/dx)
                    end2 = n_points
                else:
                    # For negative k, h(x) -> h(x+k) means x -> x+k
                    # Start from -k, go to len(h) 
                    start1 = int(-k/dx)
                    end1 = n_points
                    start2 = 0
                    end2 = int(n_points + k/dx)

                if end1 > start1 and end2 > start2:
                    s = 0

                    # by convolving with a shifted version
                    # Note this is an approximate evaluation of the integral
                    try:
                        s = np.sum(h_trial[start1:end1] * (1 - h_trial[start2:end2]))
                    except Exception:
                        s = 0
                    if s * dx < mu:
                        valid = False
                        break
            
            if not valid:
                # Apply gradient descent
                grad = np.zeros(n_points)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_trial.copy()
                    h_plus[i] += eps

            else:

                break
        
        if valid:
            high = mu

        else:
            low = mu
            
    # With the estimated optimal mu, do final optimization

    # Try a local search refinement approach
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current C5
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)

            grad[i] = (fitness_plus - current_c5) / eps

        # Update h_refined with gradient descent
        h_refined -= learning_rate * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined_sum = np.sum(h_refined)
        h_refined = h_refined * (n_points / 2) / h_refined_sum
    
    # Final evaluation
    final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
