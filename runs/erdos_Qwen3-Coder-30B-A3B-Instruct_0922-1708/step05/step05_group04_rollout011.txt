import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 1000
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Helper function to compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Main DE loop
    current_iter = 0
    best_fitness = float('inf')
    best_h = None
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        
        for i in range(population_size):
            # Select three distinct random individuals
            idxs = list(range(population_size))
            idxs.remove(i)
            a, b, c = np.random.choice(idxs, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = repair(mutant)
            
            # Crossover
            trial = np.zeros(n_points)
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Repair after crossover
            trial = repair(trial)
            
            # Evaluate fitness
            fit_trial = compute_c5(trial)
            fit_current = compute_c5(population[i])
            
            # Selection
            if fit_trial < fit_current:
                new_population.append(trial)
            else:
                new_population.append(population[i])
            
            # Update best
            if fit_trial < best_fitness:
                best_fitness = fit_trial
                best_h = trial.copy()
        
        population = new_population
        
        # Adjust mutation factor
        if current_iter % 50 == 0 and current_iter > 0:
            F = max(0.1, F * 0.99)
        
        current_iter += 1
    
    # Final refinement with gradient descent
    if best_h is not None:
        h_opt = best_h.copy()
        for _ in range(100):
            fitness = compute_c5(h_opt)
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                fitness_plus = compute_c5(h_plus)
                grad[i] = (fitness_plus - fitness) / eps
            
            h_opt -= 0.001 * grad
            h_opt = repair(h_opt)
            
            new_fitness = compute_c5(h_opt)
            if new_fitness >= fitness:
                break
            else:
                fitness = new_fitness
        best_h = h_opt
        best_fitness = compute_c5(best_h)
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
