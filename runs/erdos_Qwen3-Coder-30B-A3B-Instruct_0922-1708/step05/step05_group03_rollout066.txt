import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Start with uniform
        
    # Project to feasible region
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 10.0
    cooling_rate = 0.9995
    min_T = 1e-6
    
    # Gradient descent parameters
    lr = 0.001
    max_grad_iter = 50
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 10:
        # Temperature-based perturbation
        T = max(T0 / (1 + iteration), min_T)
        
        # Generate candidate by perturbing current solution
        perturbation = np.random.normal(0, 0.02, n_points)
        h_candidate = h + perturbation
        
        # Ensure feasibility
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate fitness (C5)
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation) * dx
        current_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Accept or reject
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
            current_c5 = candidate_c5
            
        # Update best solution found so far
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        iteration += 1
        
    # Local gradient descent refinement
    h_refined = best_h.copy()
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 15:
            break
            
        # Centered finite difference to estimate gradient
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Evaluate
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refined.copy()
    
    # Final check and return
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
