import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    total_lags = 2 * n_points - 1
    
    # Set up genetic algorithm parameters
    population_size = 100
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    
    # Initialize population
    def create_individual():
        individual = np.random.rand(n_points)
        individual = individual * (n_points / 2) / np.sum(individual)
        return np.clip(individual, 0, 1)
    
    def project_to_feasible(x):
        x = np.clip(x, 0, 1)
        sum_x = np.sum(x)
        if abs(sum_x - n_points / 2) > 1e-8:
            x = x * (n_points / 2) / sum_x
        return np.clip(x, 0, 1)
    
    if 'initial_h_values' in globals():
        population = [h.copy()]
        for _ in range(population_size - 1):
            # Add perturbed version
            perturbed = h.copy() + np.random.normal(0, 0.03, n_points)
            perturbed = project_to_feasible(perturbed)
            population.append(perturbed)
    else:
        population = [create_individual() for _ in range(population_size)]
    
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Evaluate population
        fitness = [compute_c5(individual) for individual in population]
        
        # Find best so far
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_c5:
            best_c5 = fitness[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Selection
        def tournament_select():
            indices = np.random.choice(population_size, tournament_size, replace=False)
            best_idx = indices[np.argmin([fitness[i] for i in indices])]
            return population[best_idx].copy()
            
        # Crossover, Mutation
        new_population = []
        
        # Keep elites
        elite_indices = np.argsort(fitness)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
            
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.02)
                        
                child = project_to_feasible(child)
                new_population.append(child)
            else:
                # Just mutate a single parent
                parent = tournament_select()
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.02)
                child = project_to_feasible(child)
                new_population.append(child)
                
        population = new_population
        
        # Local refinement periodically
        if iteration % refinement_interval == 0:
            # Select top individuals for gradient refinement
            top_indices = np.argsort(fitness)[:elite_size]
            for idx in top_indices:
                h_ref = population[idx].copy()
                lr = 0.001
                
                # Simple gradient descent-like refinement with finite differences
                for _ in range(20):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    h_ref = h_ref - lr * grad
                    h_ref = project_to_feasible(h_ref)
                    
                    corr = np.correlate(h_ref, 1 - h_ref, mode='full')
                    new_c5 = np.max(corr) * dx
                    
                    if new_c5 < fitness[idx]:
                        fitness[idx] = new_c5
                        population[idx] = h_ref.copy()
        
    # Final refinement on best individuals
    rank_indices = np.argsort(fitness)
    for i in range(min(5, len(rank_indices))):
        idx = rank_indices[i]
        h_ref = population[idx].copy()
        lr = 0.0005
        
        for _ in range(30):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - fitness[idx]) / eps
            
            h_ref = h_ref - lr * grad
            h_ref = project_to_feasible(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            new_c5 = np.max(corr) * dx
            
            if new_c5 < fitness[idx]:
                fitness[idx] = new_c5
                population[idx] = h_ref.copy()
    
    # Select final best
    final_rank = np.argsort(fitness)
    final_best_idx = final_rank[0]
    final_c5 = fitness[final_best_idx]
    
    return (population[final_best_idx].tolist(), final_c5, n_points)
