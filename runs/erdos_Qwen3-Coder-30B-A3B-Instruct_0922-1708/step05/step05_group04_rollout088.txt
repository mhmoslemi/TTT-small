import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    max_iter = 1000
    F = 0.8
    CR = 0.9
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(pop_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Fitness evaluation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    best_h = None
    best_fitness = float('inf')
    
    # Evolutionary loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        new_population = []
        for i in range(pop_size):
            # DE mutation
            idxs = np.random.choice(pop_size, 3, replace=False)
            a, b, c = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            mutant = a + F * (b - c)
            mutant = np.clip(mutant, 0, 1)
            mutant = repair(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            trial = repair(trial)
            
            # Selection
            current_fitness = evaluate_fitness(population[i])
            trial_fitness = evaluate_fitness(trial)
            
            if trial_fitness < current_fitness:
                new_population.append(trial)
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                
        population = new_population
        
        # Adaptive mutation
        if iteration % 50 == 0 and iteration > 0:
            F = max(0.1, F * 0.99)
    
    # Final gradient descent refinement
    if best_h is not None:
        h_ref = best_h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 5:
                break
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad
            h_ref = repair(h_ref)
            
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        
        corr_final = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5_final = np.max(corr_final) * dx
        if c5_final < best_fitness:
            best_fitness = c5_final
            best_h = h_ref.copy()
    
    return (best_h.tolist(), best_fitness, n_points)
