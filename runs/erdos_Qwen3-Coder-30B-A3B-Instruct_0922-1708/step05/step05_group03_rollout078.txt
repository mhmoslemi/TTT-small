import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the C5 value for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    else:
        h = np.ones(n_points) * (n_points / 2.0) / n_points
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual via tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover between two parents"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.03):
    """Apply Gaussian mutation with given rate and std dev"""
    mask = np.random.rand(len(individual)) < mutation_rate
    individual[mask] += np.random.normal(0, sigma, np.sum(mask))
    return individual

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine solution with gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = compute_fitness(h_plus, dx)
            grad[i] = (c5_plus - compute_fitness(h, dx)) / eps
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        # Use initial solution, perturb it
        h_base = np.array(initial_h_values)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
        for i in range(population_size // 2):
            h = h_base + np.random.normal(0, 0.03, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        # Add random solutions
        for i in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # All random
        for i in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Selection, crossover, mutation
        new_population = []
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < 0.8:  # Crossover probability
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                child1 = project_to_feasible(child1, n_points)
                child2 = project_to_feasible(child2, n_points)
                new_population.extend([child1, child2])
            else:
                # Just mutate and keep parents with some chance
                parent1 = mutate(parent1)
                parent1 = project_to_feasible(parent1, n_points)
                parent2 = mutate(parent2)
                parent2 = project_to_feasible(parent2, n_points)
                new_population.extend([parent1, parent2])
        
        # Trim to population_size
        new_population = new_population[:population_size]
        
        # Update population and fitnesses
        population = new_population
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best solution found
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Periodic refinement of top individuals
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for idx in top_indices:
                refined = gradient_refinement(population[idx], dx)
                refined_fitness = compute_fitness(refined, dx)
                if refined_fitness < fitnesses[idx]:
                    population[idx] = refined
                    fitnesses[idx] = refined_fitness
                    if refined_fitness < best_c5:
                        best_c5 = refined_fitness
                        best_h = refined.copy()
    
    # Final refinement pass
    for i in range(3):
        h_ref = best_h + np.random.normal(0, 0.01, n_points)
        h_ref = project_to_feasible(h_ref, n_points)
        h_ref = gradient_refinement(h_ref, dx)
        c5 = compute_fitness(h_ref, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
