import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def evaluate_c5(h, n_points):
    """Evaluate the C5 bound for given h"""
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(correlation)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start strategy with various initializations
    initial_solutions = []
    
    # 1. Use provided parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_solutions.append(h_parent.copy())
    
    # 2. Generate random initializations
    for _ in range(5):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Convex relaxation and refinement for each initialization
    for i, h_start in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Start with solution
        h = h_start.copy()
        
        # Convex relaxation step - simpler approach without quadratic program
        # Just use a warm-start from a convex relaxation to model the overlap
        # Relax to bound the product terms: h[i] * (1 - h[i+k]) <= (h[i] + (1-h[i+k])) / 2  
        # But that's still not directly convex, so let's just do direct minimization 
        # Throwing in gradient-based approach with safe bounds
        
        # Refinement with L-BFGS-B
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return evaluate_c5(h_vec, n_points)
            
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
    
        # Use scipy.optimize with bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Bound the search space to improve convergence
        try:
            result = minimize(
                objective, 
                h, 
                method='L-BFGS-B',
                bounds=bounds, 
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-6, 'gtol': 1e-6},
                callback=lambda x: None
            )
            
            # Check if it improved
            if result.success:
                h_refined = result.x
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                c5_refined = evaluate_c5(h_refined, n_points)
                
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h_refined.copy()
                    # Early stopping if improvement is not significant
                    if best_c5 < 0.38085:
                        break
                        
        except Exception:
            continue
            
        # If no refinement, check initial solution for improvement
        c5_start = evaluate_c5(h_start, n_points)
        if c5_start < best_c5:
            best_c5 = c5_start
            best_h = h_start.copy()
    
    # Final verification that we have a valid and feasible solution
    if best_h is None:
        # Fallback to uniform solution if nothing was found
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        best_c5 = evaluate_c5(best_h, n_points)
    
    # Make sure solution is normalized
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
