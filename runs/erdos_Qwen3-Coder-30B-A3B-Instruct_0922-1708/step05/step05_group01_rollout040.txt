import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize z (binary approximation)
    z = (h > 0.5).astype(float)
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Augmented Lagrangian parameters
    mu = 0.1
    max_iter = 5000
    lr = 0.001
    perturb_frac = 0.01
    
    # Main optimization loop
    for i in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlation for current h
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        
        # Determine the lag that gives maximum overlap
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) // 2)
        
        # Compute gradient with respect to h for the maximal lag
        grad = np.zeros(n_points)
        # For maximized lag k*, the contribution to overlap integral is:
        # integral h(x)(1 - h(x+k*)) dx
        # Grad = (1 - h(x+k*)) - h(x)
        
        # Construct shifted h vector
        h_shifted = np.roll(h, max_lag)
        if max_lag > 0:
            h_shifted[:max_lag] = 0
        elif max_lag < 0:
            h_shifted[max_lag:] = 0
            
        # Compute finite difference gradient for current setup
        # We use the fact that d/dh_i [h_i (1 - h_{i+k})]  = (1 - h_{i+k}) - h_i 
        # And that we want to maximize the integral, so we minimize the negative
        grad = (1 - h_shifted) - h
        grad = grad * dx  # Scale by dx
        
        # Apply penalty term gradient
        penalty_grad = mu * (h - z)
        grad += penalty_grad
        
        # Gradient descent step
        h_new = h - lr * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z (binary approximation)
        z_new = (h_new > 0.5).astype(float)
        
        # Update h and z
        h = h_new
        z = z_new
        
        # Update penalty parameter
        mu = min(mu * 1.2, 10)
        
        # Occasionally introduce small random perturbations to escape local minima
        if i % 100 == 0 and np.random.random() < 0.1:
            noise = np.random.normal(0, 0.001, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final verification of the result
    corr = correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr) * dx
    
    return (best_h.tolist(), c5_final, n_points)
