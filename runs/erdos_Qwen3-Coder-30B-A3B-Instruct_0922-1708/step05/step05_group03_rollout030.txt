import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Memetic algorithm parameters
    population_size = 100
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    refinement_interval = 50
    max_generations = 10000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as basis, add some variation
        population.append(h.copy())
        for _ in range(population_size - 1):
            # Add variation around initial solution
            perturbed = h.copy() + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        # Initialize population with random feasible solutions
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    def evaluate_fitness(h_vector):
        correlation = np.correlate(h_vector, 1 - h_vector, mode='full')
        return np.max(correlation) * dx
    
    # Evaluate initial population
    fitnesses = [evaluate_fitness(ind) for ind in population]
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_c5 = fitnesses[best_idx]
        best_h = population[best_idx].copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1
        
        # Selection (tournament)
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation to create offspring
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover (arithmetic)
                alpha = np.random.uniform(0.7, 0.95)
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.03)
            
            # Repair solutions to maintain constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only valid population size
        population = new_population[:population_size]
        
        # Evaluate fitness of new population
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Update best solution
        best_idx = np.argmin(fitnesses)
        if fitnesses[best_idx] < best_c5:
            best_c5 = fitnesses[best_idx]
            best_h = population[best_idx].copy()
        
        # Local refinement occasionally
        if generation % refinement_interval == 0:
            # Perform gradient-based refinement on top individuals
            top_indices = np.argsort(fitnesses)[:population_size//10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                step_size = 0.001
                
                for _ in range(20):
                    # Estimate gradient via finite differences
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitnesses[idx]) / eps
                    
                    # Update solution
                    h_ref = h_ref - step_size * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                    
                    # Re-evaluate
                    corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                    new_fitness = np.max(corr_new) * dx
                    
                    if new_fitness < fitnesses[idx]:
                        fitnesses[idx] = new_fitness
                        population[idx] = h_ref.copy()
                        
                        if new_fitness < best_c5:
                            best_c5 = new_fitness
                            best_h = h_ref.copy()
    
    # Final gradient refinement of best solutions
    candidates = [best_h.copy()]
    for _ in range(2):
        # Add some random perturbations
        h_candidate = best_h.copy() + np.random.normal(0, 0.02, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        candidates.append(h_candidate)
    
    # Add top of current population
    top_indices = np.argsort(fitnesses)[:min(2, len(fitnesses))]
    for i in top_indices:
        candidates.append(population[i].copy())
    
    # Gradient refinement
    for i, h_ref in enumerate(candidates):
        for _ in range(30):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
