import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    start_time = time.time()
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        best_h = h_parent.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    else:
        best_h = np.ones(n_points) * 0.5
        best_c5 = float('inf')
    
    # Scale pyramid [10, 20, 40, 80, 100]
    scales = [10, 20, 40, 80, 100]
    
    h_current = None
    for s in scales:
        if time.time() - start_time > budget_s - 50:
            break
            
        # Create discretization and step size
        n = s
        dx_s = 2.0 / n
        
        # Initialize
        if h_current is None:
            h_current = np.ones(n) * 0.5
        else:
            # Linear interpolation from previous scale
            x_prev = np.linspace(0, 2, len(h_current))
            x_new = np.linspace(0, 2, n)
            h_current = np.interp(x_new, x_prev, h_current)
        
        # Relaxation parameters
        lambda_reg = 0.1
        max_lags = n // 4  # Use half the range for average

        # Create variables
        h = Variable(n)
        
        # Objective: approximate max overlap with average over 4 interior lags
        obj = 0
        for k in range(1, max_lags+1):
            # Approximate using selected lags
            obj += sum(h[i] * (1 - h[i+k]) for i in range(n-k))
        obj = obj / (max_lags * (n - max_lags))  # normalize
        
        # Constraints
        constraints = [
            h >= 0,
            h <= 1,
            sum(h) == n / 2
        ]
        
        # Soft constraints for robustness
        for i in range(n):
            constraints.append(h[i] >= 0 - 1e-6)
            constraints.append(h[i] <= 1 + 1e-6)
        
        # Create problem with soft regularization
        # Minimize a relaxed objective with penalty constraints
        problem = Problem(Minimize(obj), constraints)
        try:
            problem.solve(solver=ECOS, verbose=False, max_iters=1000)
            if problem.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_val = np.array([v.value for v in h])
                h_current = h_val.copy()
            else:
                # As fallback, try coordinate descent or random restart
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n / 2) / np.sum(h_current)
        except:
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n / 2) / np.sum(h_current)
        
        # Ensure feasibility
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n / 2) / np.sum(h_current)
        
        if n == 100:
            # Final refinement with gradient descent
            if time.time() - start_time > budget_s - 30:
                break
            local_h = h_current.copy()
            eta = 0.001
            eps = 1e-4
            for _ in range(50):
                if time.time() - start_time > budget_s - 30:
                    break
                grad = np.zeros_like(local_h)
                for i in range(n):
                    h_plus = local_h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - np.max(np.correlate(local_h, 1 - local_h, mode='full')) * dx) / eps
                local_h = local_h - eta * grad
                local_h = np.clip(local_h, 0, 1)
                local_h = local_h * (n / 2) / np.sum(local_h)
                corr = np.correlate(local_h, 1 - local_h, mode='full')
                c5 = np.max(corr) * dx
                if c5 < best_c5:
                    best_h = local_h.copy()
                    best_c5 = c5
    
    if best_h is None:
        best_h = np.ones(n_points) * 0.5
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    # Final validation and return
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    # Ensure it meets all requirements
    assert 0 <= np.min(best_h) and np.max(best_h) <= 1
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-6
    
    return (best_h.tolist(), final_c5, n_points)
