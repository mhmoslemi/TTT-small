import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Random initialization
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Simulated Annealing parameters
    T0 = 0.1 * evaluate_h(h_init, dx)
    T = T0
    alpha = 0.99
    max_iter = 1000
    perturbation_rate = 0.1
    
    # Best solution tracking
    current_h = h_init.copy()
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Early stopping
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main SA loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate neighbor: perturb a random subset of elements
        num_perturb = max(1, int(perturbation_rate * n_points))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Create candidate solution
        candidate_h = current_h.copy()
        for i in indices:
            # Add small random perturbation
            candidate_h[i] += np.random.normal(0, 0.01)
        
        # Repair and normalize
        candidate_h = repair_h(candidate_h, n_points)
        candidate_c5 = evaluate_h(candidate_h, dx)
        
        # Acceptance criterion
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            # Update best solution
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cool down
        T *= alpha
        
        # Adaptive cooling
        if no_improvement_count > 50:
            T *= 0.95
            no_improvement_count = 0
            
        # Early stop if no improvement for too long
        if no_improvement_count > max_no_improvement:
            break
    
    # Final gradient refinement on best solution
    best_h = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
