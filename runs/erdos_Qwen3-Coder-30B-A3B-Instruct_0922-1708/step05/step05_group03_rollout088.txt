import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points_list = [10, 20, 40, 80, 100]
    start_time = time.time()
    
    # Parent fallback
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (100 / 2) / np.sum(h_parent)
        best_h = h_parent.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / 100)
    
    # Multi-scale convex optimization
    h_prev = None
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 50:
            break
            
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5  # Initial guess at scale
        
        if h_prev is not None:
            # Upsample from previous scale
            h_prev_up = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, len(h_prev)), 
                                 h_prev)
            h = h_prev_up.copy()
        
        # Convex relaxation
        h_cvx = cp.Variable(n_points)
        # Soft constraints
        objective = cp.Minimize(
            cp.sum(
                cp.vstack([
                    cp.multiply(h_cvx[i], 1 - h_cvx[i + k]) 
                    for i in range(n_points - 1) 
                    for k in range(1, min(n_points // 4, n_points - i))
                ])
            ) * dx / (n_points * (n_points // 4))
        )
        
        # Constraint: sum = 1
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        
        # Solve convex relaxed problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            h_solution = h_cvx.value
        except:
            h_solution = h.copy()
        
        h_solution = np.clip(h_solution, 0, 1)
        h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
        h_prev = h_solution.copy()
        
        # Evaluate
        corr = np.correlate(h_solution, 1 - h_solution, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_solution.copy()
    
    # Local gradient refinement
    if time.time() - start_time > budget_s - 10:
        pass
    else:
        h_ref = best_h.copy()
        lr = 0.001
        steps = 50
        eps = 1e-4
        
        for _ in range(steps):
            if time.time() - start_time > budget_s - 5:
                break
                
            grad = np.zeros_like(h_ref)
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
                
            h_ref = h_ref - lr * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    if best_h is None:
        # Fallback if nothing found
        h_fallback = np.ones(100) * 0.5
        h_fallback = h_fallback * (100 / 2) / np.sum(h_fallback)
        best_h = h_fallback
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / 100)
    
    return (best_h.tolist(), best_c5, 100)
