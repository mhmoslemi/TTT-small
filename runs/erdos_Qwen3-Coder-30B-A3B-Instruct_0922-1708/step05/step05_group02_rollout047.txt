import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Initialize with a random feasible solution
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Gradient descent with random perturbations
    max_iter = 5000
    lr = 0.001
    perturb_rate = 0.1
    stagnation_threshold = 100
    
    last_improvement = 0
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        h_base = h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update using gradient
        h = h - lr * grad
        h = repair_h(h, n_points)
        
        # Check if we should increase learning rate or perturb
        c5_new = evaluate_h(h, dx)
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
            last_improvement = iter_count
        else:
            # Apply random perturbation to escape local minima
            num_perturb = int(perturb_rate * n_points)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                h[i] += np.random.normal(0, 0.01)
            
            h = repair_h(h, n_points)
            c5_new = evaluate_h(h, dx)
            if c5_new < best_c5:
                best_h = h.copy()
                best_c5 = c5_new
                last_improvement = iter_count
                
        # Early stopping if no improvement in a while
        if iter_count - last_improvement > stagnation_threshold:
            break
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        h_base = best_h.copy()
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
