import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h, dx):
    """Evaluate C5 for given h values"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_constraints(h, target_sum):
    """Project h to satisfy sum constraint"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    
    # Scale to adjust sum
    h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def compute_gradient(h, k_best):
    """Compute gradient with respect to the worst lag"""
    n = len(h)
    grad = np.zeros_like(h)
    
    # Gradient computation based on dominant lag k_best
    for i in range(n):
        # Gradient contribution from overlap at k_best
        j1 = (i - k_best) % n
        j2 = (i + k_best) % n
        grad[i] = (1 - h[j1]) + (1 - h[j2])
    
    return grad

def trust_region_step(h, grad, alpha, target_sum):
    """Compute trust region step and project to feasible set"""
    # Compute proposed step
    delta_h = -alpha * grad
    
    # Apply projection
    h_new = h + delta_h
    h_new = np.clip(h_new, 0, 1)
    h_new = project_to_constraints(h_new, target_sum)
    
    return h_new

    # Problem setup
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

            h = np.interp(x_target, x_orig, h)
    else:
        # Simple initial construction
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet sum constraint
    h = project_to_constraints(h, target_sum)
    
    # Initial evaluation
    c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    best_c5 = c5
    
    # Trust region parameters
    alpha = 0.01
    min_alpha = 1e-5
    
    # Simulated Annealing params
    temp = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    
    iteration = 0
    success_count = 0
    max_success = 5
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute correlations to determine worst lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        c5_current = np.max(corr) * dx
        k_best = max_k - (n_points - 1)  # Convert to actual lag
        
        # Compute gradient
        grad = compute_gradient(h, k_best)
        
        # Try trust region step
        h_trial = trust_region_step(h, grad, alpha, target_sum)
        c5_trial = evaluate_c5(h_trial, dx)
        
        # Accept or reject step
        if c5_trial < c5:
            h = h_trial
            c5 = c5_trial
            success_count += 1
            if alpha > min_alpha:
                alpha *= 1.2  # Expand trust region
        else:
            success_count = 0
            alpha *= 0.8  # Shrink trust region
            
        # Update best solution
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Simulated Annealing perturbations
        if iteration % perturbation_freq == 0 and iteration > 0:
            for i in range(n_points):

                if np.random.random() < temp:
                    # Delta based on temperature and value
                    delta = temp * (1 - h[i]) * h[i] * (np.random.random() - 0.5)
                    h[i] += delta
                    h[i] = np.clip(h[i], 0, 1)
            h = project_to_constraints(h, target_sum)
            
        # Adjust temperature
        temp *= cooling_rate
        
        # Early termination on success
        if success_count >= max_success:

    # Final intensive refinement using scipy
    def objective(h_vec):
        # Reshape to 1D
        h = h_vec.reshape(-1)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

        return np.sum(h_vec) - target_sum

    h_guess = best_h.copy()
    bounds = [(0, 1) for _ in range(n_points)]

        c5_final = evaluate_c5(h_final, dx)
        
        # Update if better
        if c5_final < best_c5:
            best_h = h_final
            best_c5 = c5_final
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)
