import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Feasibility projection function
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        return h_vec * (n_points / 2) / np.sum(h_vec)
    
    # Compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Initial temperature
    T0 = 10.0
    iteration = 0
    max_iterations = int(400 / 0.001)  # About 400 seconds at 1ms per iter
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Perturb solution
        epsilon = 0.02
        candidate_h = current_h + np.random.normal(0, epsilon, n_points)
        candidate_h = project_h(candidate_h)
        
        # Compute candidate C5
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        T = T0 / np.log(1 + iteration + 1)
        
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        
        iteration += 1
    
    # Local gradient descent refinement
    h_refine = best_h.copy()
    lr = 0.001
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_refine = h_refine - lr * grad
        h_refine = project_h(h_refine)
        
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refine.copy()
    
    # Final validation
    final_h = project_h(best_h)
    final_corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
