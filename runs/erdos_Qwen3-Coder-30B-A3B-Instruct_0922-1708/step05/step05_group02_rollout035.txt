import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        best_h = h_init.copy()
    else:
        # Initialize a feasible random solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Mutation parameters
    max_iter = 5000
    mutation_rate = 0.15
    num_mutations = 100
    patience = 100
    
    # Track patience
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate mutations
        new_candidates = []
        for _ in range(num_mutations):
            # Create a mutated version of best_h
            candidate = best_h.copy()
            
            # Randomly flip some elements
            num_flip = int(mutation_rate * n_points)
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            for idx in flip_indices:
                candidate[idx] = 1 - candidate[idx]
                
            # Repair and evaluate
            candidate = repair_h(candidate, n_points)
            c5 = evaluate_h(candidate, dx)
            new_candidates.append((candidate, c5))
            
        # Find the best among new candidates
        candidates_vals = [c[1] for c in new_candidates]
        best_new_idx = np.argmin(candidates_vals)
        if candidates_vals[best_new_idx] < best_c5:
            best_h = new_candidates[best_new_idx][0]
            best_c5 = candidates_vals[best_new_idx]
            last_improvement = iteration
        elif iteration - last_improvement > patience:
            # No improvement in a while, try a larger mutation
            mutation_rate = min(0.3, mutation_rate * 1.2)
            
    # Final gradient-based refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h.tolist(), best_c5, n_points)
