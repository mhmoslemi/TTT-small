import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_iter = 1000
    F = 0.8
    CR = 0.9
    mutation_decay = 0.99
    local_search_freq = 50
    
    # Initialize population
    population = []
    best_h = None
    best_c5 = float('inf')
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
        best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
        population.append(h_init.copy())
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        best_h = population[0].copy()
        best_c5 = np.max(np.correlate(best_h, 1 - best_h, mode='full')) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        new_population = []
        
        for i in range(population_size):
            # Selection of three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = repair(mutant)
            
            # Crossover
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Evaluate trial
            trial = repair(trial)
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_c5 = np.max(correlation) * dx
            
            # Selection
            population_c5 = np.max(np.correlate(population[i], 1 - population[i], mode='full')) * dx
            if trial_c5 < population_c5:
                new_population.append(trial)
                if trial_c5 < best_c5:
                    best_c5 = trial_c5
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                
        population = new_population
        F *= mutation_decay
        
        # Local search on best solution
        if iteration % local_search_freq == 0 and iteration > 0:
            h_local = best_h.copy()
            for _ in range(50):
                corr = np.correlate(h_local, 1 - h_local, mode='full')
                fitness = np.max(corr) * dx
                
                grad = np.zeros(n_points)
                eps = 1e-4
                for j in range(n_points):
                    h_plus = h_local.copy()
                    h_plus[j] += eps
                    h_plus = repair(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    fitness_plus = np.max(corr_plus) * dx
                    grad[j] = (fitness_plus - fitness) / eps
                
                h_local -= 0.001 * grad
                h_local = repair(h_local)
                
                corr_new = np.correlate(h_local, 1 - h_local, mode='full')
                fitness_new = np.max(corr_new) * dx
                if fitness_new >= fitness:
                    break
                else:
                    fitness = fitness_new
                    
            corr_final = np.correlate(h_local, 1 - h_local, mode='full')
            c5_final = np.max(corr_final) * dx
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_local.copy()
    
    # Final refinement
    h_ref = best_h.copy()
    for _ in range(100):
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_ref -= 0.001 * grad
        h_ref = repair(h_ref)
        
        corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    correlation = np.correlate(h_ref, 1 - h_ref, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (h_ref.tolist(), final_c5, n_points)
