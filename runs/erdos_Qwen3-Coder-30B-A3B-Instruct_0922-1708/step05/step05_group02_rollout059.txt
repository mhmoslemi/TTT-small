import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    n_nests = 20
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 2000
    stagnation_threshold = 200
    no_improvement_count = 0
    
    # Initialize population
    nests = [best_h.copy()]
    fitness = [best_c5]
    
    # Fill with random solutions
    for _ in range(n_nests - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        nests.append(h)
        fitness.append(evaluate_h(h, dx))
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find the best nest
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_h = nests[best_idx].copy()
            best_c5 = fitness[best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > stagnation_threshold:
            # Restart with new random solutions
            for i in range(n_nests):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                nests[i] = h
                fitness[i] = evaluate_h(h, dx)
            no_improvement_count = 0
            
        # Generate new solutions via Levy flight
        for i in range(n_nests):
            if np.random.rand() > alpha:
                continue  # Skip this nest
                
            # Generate new solution by perturbing current nest
            new_h = nests[i].copy()
            # Select random subset for perturbation
            num_perturb = max(1, int(n_points * 0.1))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for idx in indices:
                # Apply small random perturbation
                delta = np.random.normal(0, 0.01)
                new_h[idx] += delta
                
            # Repair and normalize
            new_h = repair_h(new_h, n_points)
            new_c5 = evaluate_h(new_h, dx)
            
            # Accept or reject based on fitness
            if new_c5 < fitness[i]:
                nests[i] = new_h
                fitness[i] = new_c5
    
    # Final local search
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
        
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
        
    return (best_h.tolist(), best_c5, n_points)
