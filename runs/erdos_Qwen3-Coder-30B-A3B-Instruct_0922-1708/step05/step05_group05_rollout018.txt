import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_count = 5
    
    # Initialize population using Halton sequence
    sampler = qmc.Halton(d=n_points, seed=seed)
    points = sampler.random(n=population_size)
    population = np.zeros((population_size, n_points))
    
    for i in range(population_size):
        # Map to [0,1] and normalize sum
        h = points[i] * 2  # scale to [0,2]
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population[i] = h
    
    # If initial construction available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population[0] = h_init
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
        
        fitnesses = np.array(fitnesses)
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Tournament selection
        selected_indices = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected_indices.append(winner)
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Selection
            idx1, idx2 = np.random.choice(selected_indices, 2, replace=False)
            h1, h2 = population[idx1], population[idx2]
            
            if np.random.random() < crossover_rate:
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
                child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
            else:
                child1, child2 = h1.copy(), h2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.random() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.random() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Repair constraints
            child1 = np.clip(child1, 0, 1)
            child2 = np.clip(child2, 0, 1)
            
            # Normalize sum
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = np.array(new_population[:population_size])
        
        # Adapt mutation rate
        if gen % 10 == 0:
            mutation_rate *= 0.99
    
    # Final refinement of best solution
    if best_h is not None:
        h_refined = best_h.copy()
        for _ in range(1000):
            # Estimate gradient via finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-6
            base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            base_fitness = np.max(base_corr) * dx
            
            improved = False
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                
                grad[j] = (fitness_plus - base_fitness) / eps
                
                if fitness_plus < base_fitness:
                    improved = True
            
            if not improved:
                break
            
            # Update with small step
            step_size = 0.001
            h_refined -= step_size * grad
            h_refined = np.clip(h_refined, 0, 1)
            h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
            
            # Check if this gave improvement
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            new_fitness = np.max(corr) * dx
            
            if new_fitness >= base_fitness:
                break
        
        # Final evaluation
        corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(corr_final) * dx
        
        best_c5 = final_fitness
        best_h = h_refined
    
    return (best_h.tolist(), best_c5, n_points)
