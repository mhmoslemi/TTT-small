import numpy as np
from scipy.stats import halton
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_count = 5
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init.copy())
        else:
            # Create Halton sequence-based initial population
            seq = halton.Halton(1)
            for _ in range(10):
                h = np.zeros(n_points)
                for i in range(n_points):
                    h[i] = seq.random(1)[0]
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
        
        # Fill the rest with random individuals
        for _ in range(population_size - len(population)):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
        return population
    
    def evaluate_individual(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair_individual(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            diff = current_sum - target_sum
            if diff > 0:
                sorted_indices = np.argsort(h)[::-1]
                for idx in sorted_indices:
                    if diff <= 0:
                        break
                    available = h[idx]
                    reduce = min(diff, available)
                    h[idx] -= reduce
                    diff -= reduce
            else:
                sorted_indices = np.argsort(h)
                for idx in sorted_indices:
                    if diff >= 0:
                        break
                    available = 1.0 - h[idx]
                    increase = min(-diff, available)
                    h[idx] += increase
                    diff += increase
            h = h * target_sum / np.sum(h)
        return h
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament = random.sample(list(zip(population, fitnesses)), tournament_size)
            winner = min(tournament, key=lambda x: x[1])
            selected.append(winner[0].copy())
        return selected
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, rate):
        h = h.copy()
        for i in range(n_points):
            if random.random() < rate:
                h[i] += np.random.normal(0, 0.05)
        return h
    
    # Start timing
    start_time = time.time()
    population = initialize_population()
    
    # Evaluate initial population
    fitnesses = [evaluate_individual(h) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Main evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create new population
        new_population = []
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = repair_individual(mutate(child1, mutation_rate))
            child2 = repair_individual(mutate(child2, mutation_rate))
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Evaluate new population
        fitnesses = [evaluate_individual(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_h = population[current_best_idx].copy()
            best_fitness = fitnesses[current_best_idx]
        
        # Adapt mutation rate
        if generation % 10 == 0:
            mutation_rate *= 0.99
    
    # Final validation and return
    best_h = repair_individual(best_h)
    final_fitness = evaluate_individual(best_h)
    
    return (best_h.tolist(), final_fitness, n_points)
