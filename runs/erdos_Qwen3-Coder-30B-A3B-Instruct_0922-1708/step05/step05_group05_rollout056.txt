import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Initial construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current correlation and maximum overlap
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Clamp to [0,1] and re-normalize
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) * dx - current_c5) / eps
            
        # Update h using gradient descent
        h = h - learning_rate * grad
        
        # Project onto feasible region: [0,1] and sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbations
        if iteration % perturb_freq == 0:
            perturb_magnitude_t = perturb_magnitude / (1.0 + iteration / 100.0)
            noise = np.random.normal(0, perturb_magnitude_t, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Update best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
