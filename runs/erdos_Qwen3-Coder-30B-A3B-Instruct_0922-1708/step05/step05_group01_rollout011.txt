import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Optimization loop
    max_iter = 6000
    alpha = 0.01
    stagnation_count = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute current overlaps
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        total_overlap = np.sum(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Check for stagnation
        if stagnation_count > 200:
            # Random perturbation
            perturbation = np.random.normal(0, 0.001, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
            continue
        
        # Compute gradient approximations
        grad_c5 = np.zeros(n_points)
        grad_total = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            total_plus = np.sum(corr_plus) * dx
            
            grad_c5[i] = (c5_plus - c5) / eps
            grad_total[i] = (total_plus - total_overlap) / eps
        
        # Update h using gradient descent
        update = alpha * grad_c5
        h_new = h - update
        
        # Repair feasibility
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-8:
            if h_sum > target_sum:
                # Reduce largest elements
                sorted_indices = np.argsort(h_new)[::-1]
                excess = h_sum - target_sum
                for idx in sorted_indices:
                    if excess <= 0:
                        break
                    reduce_amount = min(excess, h_new[idx])
                    h_new[idx] -= reduce_amount
                    excess -= reduce_amount
            else:
                # Increase smallest elements
                sorted_indices = np.argsort(h_new)
                deficit = target_sum - h_sum
                for idx in sorted_indices:
                    if deficit <= 0:
                        break
                    increase_amount = min(deficit, 1 - h_new[idx])
                    h_new[idx] += increase_amount
                    deficit -= increase_amount
            
            # Rescale to exact sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 200 == 0 and iteration > 0:
            alpha *= 0.5
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
