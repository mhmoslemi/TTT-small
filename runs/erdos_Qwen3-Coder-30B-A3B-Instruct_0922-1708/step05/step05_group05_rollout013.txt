import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    iteration = 0
    
    # Gradient ascent parameters
    learning_rate = 0.001
    decay_factor = 0.5
    patience = 10
    patience_counter = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find index of minimum overlap
        min_idx = np.argmin(corr)
        
        # Compute gradient estimate at that index
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Estimate gradient for correlation at min_idx
            # This approximation only works for a short range - focus on optimizing
            # the key region around min_idx
            grad[i] = 0
            # Add simple derivative estimation
            # For correlation h * (1-h) at offset k, we have:
            # d/dh[i] (h[i] * (1 - h[i+k])) 
            # = 1 - 2*h[i] if i+k is in range and i is valid
            k = min_idx - n_points + 1  # Convert correlation index to offset
            if 0 <= i+k < n_points:
                grad[i] = 1 - 2 * h[i]
        
        # Update
        h = h + learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Project back to sum == n_points/2
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Compute new value
        corr_new = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Check for improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
            
        # Adjust learning rate based on performance
        if patience_counter > patience:
            learning_rate *= decay_factor
            patience_counter = 0
            
        # Periodic stochastic perturbation to escape local minima
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1) 
            h = h * (n_points / 2) / np.sum(h)
            
        # Early exit if no progress for too long
        if iteration - last_improvement > 500:
            break

    return (best_h.tolist(), best_c5, n_points)
