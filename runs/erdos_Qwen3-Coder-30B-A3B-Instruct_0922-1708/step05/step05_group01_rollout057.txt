import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to required sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper for correlation computation
    def compute_c5(h_vec):
        # Compute correlation for all lags
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(corr) * dx
        return c5, corr
    
    # Gradient computation for C5 with respect to h
    def compute_gradient(h_vec, max_lag_idx, corr):
        n = len(h_vec)
        grad = np.zeros_like(h_vec)
        
        # The maximum lag corresponds to the max correlation
        # We're interested in the gradient of the maximum element of correlation
        # For overlap at lag k, the gradient contribution is 
        # d/dh_i [ h[i] * (1 - h[i + k]) ] = 1 - h[i + k] if i + k < n
        # and -h[i] if i - k >= 0
        # But we must be more careful: compute all possible overlap pairs
        
        # The derivative of C5 is approximately: 
        # ∂C5/∂h[i] = 2*dx * (sum_{k} ∂/∂h[i] corr[i,k])
        # where corr[i,k] = h[i] * (1 - h[i+k]) if both exist, 0 otherwise
        
        # More directly, the gradient at position i for the max overlap
        # is influenced by the regions where gradient is non-zero
        
        # For now, we will compute the gradient for the corresponding max lag
        # The effect is more complex because the max is a non-smooth function
        # So we'll approximate it by assuming it occurs at a particular lag
        max_lag = max_lag_idx - (n - 1)
        if max_lag == 0:
            # max overlap at lag 0, that's correlation of h with 1-h, but it's self-overlap
            # Actually no, we're looking at lag where h[i]* (1 - h[i+k]) contributes
            # Correlation of h with 1-h at lag -k is h[i] * (1 - h[i+k]) for i+k < n 
            # lag k = 0: h[i]*(1-h[i]) is self-correlation in forward products  
            # But we want convolution with flip: h[i] * (1 - h[i+k]) = h[i] * (1 - h[i+k])
            # So maximum lag is the one contributing to overlap:
            # For each i, affecting the term h[i] * (1 - h[i+k]) where k is the max lag 
            # Let's recompute carefully. 
            # Correlation lags from -n+1 to n-1.
            # The vector is of size 2n-1, correlation at lag k = i  pair of  h[i] * (1-h[i+k])
            # where i< n-k and i+k < n  so i < n-k and i < n-k -> i < n-k
            #     i+k >0  -> i>-k  
            #     i+k<n -> i<n-k
            #     i≥0 -> i≥0
            # So for lag k (correlation k = k + (n-1) in array index)
            # For all valid i such that i ≥ 0, i+k ≥ 0, i < n, i+k < n, i+k = i+k,
            # So we want:
            # i ∈ [max(0, -k), min(n-1, n-1-k)]
            # i max to influence the cor at k (position in correlation array)
            # if 0 ≤ i < n and 0 ≤ i+k < n, then both terms exist  
            # Let i+k = i+k
            # correlation at k: element at position k + (n-1) = k + n -1
            # So for i ∈ [0,max(0, -k), min(n-1, n-1-k)], which has overlap with h[i+|k|]  
            # if k > 0, then i goes from 0 to n-k-1,  each contributes h[i] * (1-h[i+k])
            # if k < 0, then i goes from -k to n-1,  each contributes h[i] * (1-h[i+k])
            # if k = 0, then n terms i from 0 to n-1 each contributes h[i] * (1-h[i]) 
            # We should compute the gradient anti-symmetrically bi-directionally:
            # At any point, the overlap values add to contrib to C5.
            # There exists a definite k which maximizes the positive sum.
            # For i = 0 to n-1, only if both are within domain:
            # in the corr, if i+k causes entry [i, i+k] to exist.
            # so we can compute exactly  which term in correlation is maximum:
            # It can be thought of as the products of h[i]* (1 - h[i+k]) over k 
            # we'll compute from the oth index of correlation, NOT parse it to see which is max
            # Because max is not smooth, we use chain rule estimate. 
            # But it's better to compute element-wise anti-derivative of overlap contributions
            
            # Let's start with a very simple estimate.
            # The contribution to the max (say peak prev lag inferred) is just 1 - h[i] for lag 0
            # This is an approximation
        else:
            # lag= -k or +k case
            # correlation index is max_lag + (n-1)
            abs_lag = abs(max_lag)
            # the elements such that h[i]*(1-h[i±abs_lag]) are contributing  
            # When we're at max lag = abs_lag, 
            # we're effectively summing h[i]* (1-h[i+abs_lag]) for i = 0 to n-1-abs_lag or  i = 0 to n-1-abs_lag
            # so for i in [0, n-1-abs_lag], h[i] * (1 - h[i+abs_lag]) contributes  
            # Each h[i] contributes grad[i] = -1 * (1 - h[i+abs_lag])
            # Each h[i+abs_lag] contributes grad[i+abs_lag] = h[i]
            # For backward lag: i in [abs_lag, n-1], h[i] * (1 - h[i-abs_lag]) 
            # contribution to i: - (1 - h[i-abs_lag]) 
            # contribution to i-abs_lag: h[i] 
            # Since i+k is not necessarily within [0,n-1], we need to 
            # also ignore i such that i+k >= n, i < 0.
            # So the overlap h[i] * (1-h[i+k])  
            # For positive lag k:  
            # i in [0, n-1-k]
            # h[i] contributes according to derivative of h[i] * (1-h[i+k])
            # = (1-h[i+k]) for i=0..n-1-k and  
            # = -h[i] for i+k = 0..n-1.  
            # >= 0, <= n-1   
            # For the info of which position contributes in what way:
            # start_index = 0 if k>=0; start_index = k if k<0 
            # For positive lag k:
            # i ranges from 0 to n-1-k,  
            # For negative lag k: 
            # i ranges from -k to n-1.
            # Only h[i] where values of both exist in the corr process. 
            
            # we identify lag and just add appropriate gradient
                
            if max_lag > 0:  # Lag k > 0, so i ranges from 0 to n-k-1
                for i in range(n - abs_lag):
                    grad[i] += -(1 - h[i + abs_lag]) 
                    grad[i + abs_lag] += h[i] 
            else:  # Lag k < 0, so i ranges from -k to n-1  
                # Equivalent to i+k >=0, i <= n-1-k  
                # i >= -k and i <= n-1
                for i in range(-max_lag, n):  
                    grad[i] += -(1 - h[i + abs_lag]) 
                    grad[i + abs_lag] += h[i]
                    
        return grad
    
    # Function to project h back to feasible set
    def project_h(h_in):
        h_proj = np.clip(h_in, 0, 1)
        h_sum = np.sum(h_proj)
        if abs(h_sum - n_points/2) > 1e-8:
            # Rescale to target sum
            factor = (n_points/2) / h_sum
            h_proj = h_proj * factor
        return h_proj
    
    # Optimization loop
    c5, corr = compute_c5(h)
    best_c5 = c5 
    best_h = h.copy()
    max_lag = np.argmax(corr) - (n_points - 1)
    
    # Line search + gradient step loop
    learning_rate = 1e-3
    maxiter = 5000
    it = 0
    convergence_threshold = 1e-8
    
    while it < maxiter and (time.time() - start_time) < budget_s - 5:
        it += 1
        
        c5, corr = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        max_lag = np.argmax(corr) - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, np.argmax(corr), corr) 
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        h_new = project_h(h_new)
        
        # Update
        h = h_new
        
        # Adaptive learning rate
        if it > 500 and it % 200 == 0:
            learning_rate *= 0.8
    
    # Final check
    final_c5, _ = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
