import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Constraint handling penalty
    penalty_weight = 0.1
    penalty_weight_min = 1e-4
    penalty_update_factor = 1.2
    
    # Optimizer parameters
    learning_rate = 1e-3
    learning_rate_decay = 0.95
    plateau_patience = 200
    perturbation_fraction = 0.01
    max_iterations = 5000
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def compute_penalty(h):
        penalty = 0
        # Bound constraints
        penalty += np.sum(np.maximum(0, -h) ** 2)
        penalty += np.maximum(0, np.sum(h) - n_points / 2) ** 2
        return penalty_weight * penalty
    
    def gradient_h(h, k_star):
        # This is a simplified gradient approach
        # Equivalent to analytical gradient of C5 for one k
        grad = np.zeros_like(h)
        corr = correlate(h, 1 - h, mode='full')
        # Find maximum lag contribution
        # Just approximate one direction
        # This is an approximate gradient from numerics
        eps = 1e-6
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - compute_c5(h)) / eps
        return grad

    best_h = h.copy()
    best_c5 = compute_c5(h)
    best_penalty = compute_penalty(h)
    
    current_h = h.copy()
    last_improvement = 0
    plateau_count = 0
    old_loss = float('inf')
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute the main objective and its gradient
        corr = correlate(current_h, 1 - current_h, mode='full')
        c5 = np.max(corr) * dx
        k_star = np.argmax(corr)
        
        # Compute penalty gradient
        penalty_grad = np.zeros_like(current_h)
        for i in range(len(current_h)):
            eps = 1e-6
            h_plus = current_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            penalty_plus = compute_penalty(h_plus)
            penalty_grad[i] = (penalty_plus - compute_penalty(current_h)) / eps
        
        # Compute objective gradient for current k*
        # Approximate gradient using finite-differences
        grad_c5 = gradient_h(current_h, k_star)
        
        # Combined gradient
        total_grad = grad_c5 + penalty_grad
        
        # Line search
        alpha = learning_rate
        new_h = current_h - alpha * total_grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        new_c5 = compute_c5(new_h)
        
        # Update penalty weight adaptively
        constraint_violation = max(0, np.sum(current_h) - n_points / 2)
        if constraint_violation > 1e-4:
            penalty_weight = penalty_weight * penalty_update_factor
        else:
            penalty_weight = max(penalty_weight / penalty_update_factor, penalty_weight_min)
        
        # Accept or reject step
        if new_c5 < c5 or (np.sum((new_h - current_h) ** 2) < 1e-10 and iteration > 100):
            current_h = new_h
            c5 = new_c5
        else:
            # If no improvement, try smaller steps
            for _ in range(3):
                alpha *= 0.5
                new_h = current_h - alpha * total_grad
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = compute_c5(new_h)
                if new_c5 < c5:
                    current_h = new_h
                    c5 = new_c5
                    break
        
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
            last_improvement = iteration
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Early termination if no improvement
        if plateau_count > plateau_patience:
            # Try random perturbations
            if np.random.rand() < 0.5:
                noise = np.random.normal(0, 0.01, len(current_h))
                current_h = current_h + noise
                current_h = np.clip(current_h, 0, 1)
                current_h = current_h * (n_points / 2) / np.sum(current_h)
                plateau_count = 0
                learning_rate *= learning_rate_decay  # Reduce step size
            
    # Final check
    actual_c5 = compute_c5(best_h)
    return (best_h.tolist(), actual_c5, n_points)
