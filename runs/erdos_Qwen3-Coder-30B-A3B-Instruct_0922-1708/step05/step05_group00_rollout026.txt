import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with coarse resolution
    n_start = 50
    n_points = 200  # Final resolution
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to sum = n_start / 2
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Momentum and velocity for Adam
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Coarse optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Coarse run
    max_iter_coarse = 2000
    patience = 0
    last_c5 = float('inf')
    
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * (2.0 / n_start)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 200:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_start)
            c5_plus = np.max(cor_plus)
            grad[i] = (c5_plus - current_c5) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h = h_new
        
    # Transfer to fine resolution
    if len(best_h) < n_points:
        x_coarse = np.linspace(0, 2, len(best_h))
        x_fine = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_fine, x_coarse, best_h)
    else:
        h_fine = best_h.copy()
    
    h = h_fine
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Fine optimization
    best_c5 = float('inf')
    best_h = h.copy()
    patience = 0
    last_c5 = float('inf')
    
    for iteration in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 200:
            break
            
        # Compute gradient
        grad = np.zeros_like(h)
        eps_fd = 1e-4
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            grad[i] = (c5_plus - current_c5) / eps_fd
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
