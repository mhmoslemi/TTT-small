import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        # Use structured initialization for better spread
        h = np.ones(n_points) * 0.5
        # Perturb slightly to break symmetry
        h += np.sin(np.linspace(0, 4*np.pi, n_points)) * 0.05
        h = np.clip(h, 0, 1)
        
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use a smooth approximation to the max for better gradients
    # We'll approximate max with log-sum-exp for smooth optimization
    def smooth_max(arr, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return np.max(correlation)
    
    def compute_smooth_c5(h_vals, beta=10.0):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full') * dx
        return smooth_max(correlation, beta)
    
    # Gradient computation using finite differences
    def gradient_step(h_vals, eps=1e-5, beta=10.0):
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_smooth_c5(h_plus, beta)
            grad[i] = (c5_plus - compute_smooth_c5(h_vals, beta)) / eps
        return grad
    
    # Optimizer: Adam with adaptive learning rate
    def adam_optimize(h_vals, steps=2000, lr=0.01, beta1=0.9, beta2=0.999, epsilon=1e-8):
        m = np.zeros_like(h_vals)
        v = np.zeros_like(h_vals)
        best_c5 = float('inf')
        best_h = h_vals.copy()
        
        for t in range(1, steps + 1):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient
            g = gradient_step(h_vals)
            
            # Adam updates
            m = beta1 * m + (1 - beta1) * g
            v = beta2 * v + (1 - beta2) * (g ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            
            # Update parameters
            h_vals = h_vals - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Project to feasible set
            h_vals = np.clip(h_vals, 0, 1)
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            
            # Track best
            current_c5 = compute_c5(h_vals)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h_vals.copy()
                
        return best_h, best_c5
    
    # Coarse optimization
    n_coarse = min(50, n_points)
    if n_coarse < n_points:
        coarse_h = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    else:
        coarse_h = h.copy()
    
    coarse_h = coarse_h * (n_coarse / 2) / np.sum(coarse_h)
    coarse_h = np.clip(coarse_h, 0, 1)
    
    best_coarse_h, best_c5_coarse = adam_optimize(coarse_h, steps=2000, lr=0.01)
    
    # Transfer to fine resolution
    if n_points > n_coarse:
        fine_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), best_coarse_h)
    else:
        fine_h = best_coarse_h.copy()
        
    fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
    fine_h = np.clip(fine_h, 0, 1)
    
    # Fine optimization
    final_h, final_c5 = adam_optimize(fine_h, steps=2000, lr=0.01)
    
    # Final validation
    correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
    actual_c5 = np.max(correlation)
    
    # Ensure sum constraint and bounds
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), actual_c5, n_points)
