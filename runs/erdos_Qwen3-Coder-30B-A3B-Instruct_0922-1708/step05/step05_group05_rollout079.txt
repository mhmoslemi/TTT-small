import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with provided construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: piecewise constant
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy the constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial value
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    start_time = time.time()
    iteration = 0
    alpha = 0.01
    max_iter = 10000
    
    # Gradient estimation using finite differences
    def estimate_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_value = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to feasible set
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            value_plus = np.max(corr_plus) * dx
            
            grad[i] = (value_plus - base_value) / eps
            
        return grad
    
    # Project onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        return h * (n_points / 2) / np.sum(h)
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 10 and iteration < max_iter:
        # Estimate gradient
        grad = estimate_gradient(best_h)
        
        # Compute candidate step
        candidate_h = best_h - alpha * grad
        candidate_h = project(candidate_h)
        
        # Evaluate candidate
        corr = np.correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(corr) * dx
        
        # Accept or reject
        if candidate_c5 < best_c5:
            best_h = candidate_h
            best_c5 = candidate_c5
            alpha *= 1.2  # Increase step size
        else:
            alpha *= 0.5  # Decrease step size
            
        # Occasionally perturb to escape local minima
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            perturbed_h = best_h + perturbation
            perturbed_h = project(perturbed_h)
            
            corr = np.correlate(perturbed_h, 1 - perturbed_h, mode='full')
            perturbed_c5 = np.max(corr) * dx
            
            if perturbed_c5 < best_c5:
                best_h = perturbed_h
                best_c5 = perturbed_c5
                
        iteration += 1
    
    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
