import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mu_init = 0.5
    sigma_init = 0.1
    mutation_decay = 0.999
    restart_threshold = 200
    local_search_freq = 50
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        # Create diverse initial population around initial solution
        for _ in range(population_size):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Best solution tracking
    best_fitness = float('inf')
    best_individual = None
    stagnation_counter = 0
    mu = mu_init
    sigma = sigma_init
    
    # Repair function for constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) < 1e-10:
            return h
        # Scale to meet constraint
        scaled = h * (n_points / 2) / h_sum
        return np.clip(scaled, 0, 1)
    
    # Fitness evaluation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        return fitness
    
    # Main evolution loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Selection: keep top 50%
        selected_indices = np.argsort(fitnesses)[:population_size//2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent = selected_population[np.random.randint(0, len(selected_population))]
            # Gaussian mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mu * noise
            child = repair(child)
            offspring.append(child)
            
        # Replace worst with best offspring
        worst_indices = np.argsort(fitnesses)[::-1][:offspring_size]
        for i, worst_idx in enumerate(worst_indices):
            if i < len(offspring):
                population[worst_idx] = offspring[i]
        
        # Adapt mutation parameters
        mu = max(0.01, mu * mutation_decay)
        sigma = mu / 5.0
        
        # Restart mechanism
        if stagnation_counter > restart_threshold:
            # Reset mutation and add some diversity
            mu = mu_init * 0.9
            for i in range(population_size // 5):
                idx = np.random.randint(0, population_size)
                population[idx] = np.random.uniform(0, 1, n_points)
                population[idx] = population[idx] * (n_points / 2) / np.sum(population[idx])
            stagnation_counter = 0
            
        # Local search every few iterations
        if iteration % local_search_freq == 0 and iteration > 0:
            if best_individual is not None:
                # Simple local search with small perturbations
                h_current = best_individual.copy()
                current_fitness = best_fitness
                for _ in range(100):
                    # Gradient estimation
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    base_corr = np.correlate(h_current, 1 - h_current, mode='full')
                    base_fitness = np.max(base_corr) * dx
                    for i in range(n_points):
                        h_plus = h_current.copy()
                        h_plus[i] += eps
                        h_plus = repair(h_plus)
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        fitness_plus = np.max(corr_plus) * dx
                        grad[i] = (fitness_plus - base_fitness) / eps
                    
                    # Update
                    h_new = h_current - 0.001 * grad
                    h_new = repair(h_new)
                    
                    corr_new = np.correlate(h_new, 1 - h_new, mode='full')
                    fitness_new = np.max(corr_new) * dx
                    if fitness_new < current_fitness:
                        h_current = h_new
                        current_fitness = fitness_new
                        if current_fitness < best_fitness:
                            best_fitness = current_fitness
                            best_individual = h_current.copy()
                    else:
                        break
    
    # Final refinement of best solution
    if best_individual is not None:
        h_final = best_individual.copy()
        current_fitness = best_fitness
        
        for _ in range(100):
            grad = np.zeros(n_points)
            eps = 1e-4
            base_corr = np.correlate(h_final, 1 - h_final, mode='full')
            base_fitness = np.max(base_corr) * dx
            for i in range(n_points):
                h_plus = h_final.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - base_fitness) / eps
            
            h_new = h_final - 0.001 * grad
            h_new = repair(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new < current_fitness:
                h_final = h_new
                current_fitness = fitness_new
            else:
                break
                
        best_fitness = current_fitness
        best_individual = h_final
    
    return (best_individual.tolist(), best_fitness, n_points)
