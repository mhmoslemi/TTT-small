import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    swarm_size = 50
    inertia_weight = 0.7
    phi_p = 1.5
    phi_g = 1.5
    local_search_freq = 50
    rejuvenation_rate = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
        for _ in range(swarm_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    else:
        for _ in range(swarm_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            particles.append(h)
            velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal bests and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * swarm_size
    
    # Compute initial fitness for all particles
    for i in range(swarm_size):
        correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(correlation) * dx
        pbest_fitness[i] = fitness
        if i == 0 or fitness < pbest_fitness[0]:
            gbest_idx = i
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Optimization variables
    current_iter = 0
    stagnation_count = 0
    
    # Main loop
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Update particles
        for p_idx in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[p_idx] = (inertia_weight * velocities[p_idx] + 
                                phi_p * r1 * (pbest_particles[p_idx] - particles[p_idx]) +
                                phi_g * r2 * (gbest_particle - particles[p_idx]))
            
            # Position update
            particles[p_idx] += velocities[p_idx]
            
            # Repair
            particles[p_idx] = repair(particles[p_idx])
            
            # Fitness computation
            correlation = np.correlate(particles[p_idx], 1 - particles[p_idx], mode='full')
            fitness = np.max(correlation) * dx
            
            # Update pbest
            if fitness < pbest_fitness[p_idx]:
                pbest_fitness[p_idx] = fitness
                pbest_particles[p_idx] = particles[p_idx].copy()
                
                # Update gbest
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[p_idx].copy()
                    stagnation_count = 0
        
        # Stagnation check
        stagnation_count += 1
        if stagnation_count > 100:
            # Inject random particles
            for _ in range(int(swarm_size * rejuvenation_rate)):
                idx = np.random.randint(0, swarm_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.05, 0.05, n_points)
            stagnation_count = 0
        
        # Local search
        if current_iter % local_search_freq == 0 and current_iter > 0:
            # Local search on top 10% of particles
            sorted_indices = np.argsort([pbest_fitness[i] for i in range(swarm_size)])[:swarm_size//10]
            for idx in sorted_indices:
                h_local = pbest_particles[idx].copy()
                for _ in range(20):
                    # Gradient estimation
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    base_corr = np.correlate(h_local, 1 - h_local, mode='full')
                    base_fitness = np.max(base_corr) * dx
                    for i in range(n_points):
                        h_plus = h_local.copy()
                        h_plus[i] += eps
                        h_plus = repair(h_plus)
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        fitness_plus = np.max(corr_plus) * dx
                        grad[i] = (fitness_plus - base_fitness) / eps
                    
                    # Update
                    h_local -= 0.001 * grad
                    h_local = repair(h_local)
                
                # Update pbest if better
                correlation = np.correlate(h_local, 1 - h_local, mode='full')
                fitness = np.max(correlation) * dx
                if fitness < pbest_fitness[idx]:
                    pbest_fitness[idx] = fitness
                    pbest_particles[idx] = h_local.copy()
                    
                    if fitness < gbest_fitness:
                        gbest_fitness = fitness
                        gbest_particle = h_local.copy()
                        stagnation_count = 0
        
        current_iter += 1
    
    # Final refinement with local optimization
    best_h = gbest_particle.copy()
    best_c5 = gbest_fitness
    
    # Final optimization using scipy minimize
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a subset of the best particles for L-BFGS refinement
    candidates = [pbest_particles[i] for i in range(swarm_size) if pbest_fitness[i] < gbest_fitness * 1.2]
    candidates = candidates[:5] if candidates else [gbest_particle]
    
    for h_ref in candidates:
        try:
            result = minimize(objective, h_ref, method='L-BFGS-B', bounds=bounds,
                            constraints={'type': 'eq', 'fun': constraint}, 
                            options={'maxiter': 100, 'ftol': 1e-8})
            if result.success:
                h_opt = result.x.copy()
                h_opt = repair(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = -np.max(corr) * dx
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            continue
    
    return (best_h.tolist(), best_c5, n_points)
