import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    initial_lr = 0.01
    perturb_freq = 50
    perturb_mag_init = 0.1
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_corr = corr[max_idx]
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_idx] - max_corr) / eps
            
        # Gradient descent step
        lr = initial_lr / (1.0 + i / 100.0)
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodic random perturbations
        if i % perturb_freq == 0:
            perturb_mag = perturb_mag_init / (1.0 + i / 100.0)
            perturb = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation and computation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
