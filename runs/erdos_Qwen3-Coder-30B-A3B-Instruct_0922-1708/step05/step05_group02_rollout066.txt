import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, beta=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate Lévy distribution samples
    sigma = (np.math.gamma(1 + beta) * np.sin(np.pi * beta / 2) / 
             (np.math.gamma((1 + beta) / 2) * beta * 2**((beta - 1) / 2)))**(1 / beta)
    u = np.random.normal(0, 1, size)
    v = np.random.normal(0, 1, size)
    step = scale * u / (np.abs(v)**(1 / beta)) * sigma
    return step

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Set up population
    population_size = 50
    population = [h_init.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(h, dx) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Hyperparameters
    max_iter = 1000
    levy_scale = 0.5
    levy_beta = 1.5
    learning_rate = 0.001
    perturb_rate = 0.1
    gradient_descent_steps = 100
    patience = 100
    
    no_improvement_count = 0
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Generate new population via mutation
        new_population = [best_h.copy()]
        for h in population:
            # Lévy flight mutation
            mutation = levy_flight(n_points, levy_beta, levy_scale)
            mutated = h + mutation
            
            # Perturb a fraction of components
            mask = np.random.rand(n_points) < perturb_rate
            mutated[mask] = h[mask] + np.random.normal(0, 0.01, np.sum(mask))
            
            # Repair the mutated solution
            mutated = repair_h(mutated, n_points)
            new_population.append(mutated)
        
        # Evaluate new population
        new_fitness = [evaluate_h(h, dx) for h in new_population]
        
        # Local refinement for top solutions
        top_indices = np.argsort(new_fitness)[:10]
        for idx in top_indices:
            h_refined = new_population[idx].copy()
            current_c5 = new_fitness[idx]
            
            # Gradient descent refinement
            for _ in range(gradient_descent_steps):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Update using gradient
                h_refined = h_refined - learning_rate * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < new_fitness[idx]:
                    new_fitness[idx] = current_c5
                    new_population[idx] = h_refined.copy()
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:
            best_h = population[current_best_idx].copy()
            best_value = fitness[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            # Reinitialize part of population
            for i in range(population_size // 2):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population[i] = h
                fitness[i] = evaluate_h(h, dx)
            no_improvement_count = 0

    # Final gradient refinement of best solution
    h_final = best_h.copy()
    current_c5 = best_value
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        h_final = h_final - 0.0005 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        current_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), current_c5, n_points)
