import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mutation_strength = 0.5
    mutation_variance = 0.1
    elite_ratio = 0.5
    stagnation_threshold = 200
    restart_threshold = 100
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Initialize with small random perturbations around initial
        for _ in range(population_size):
            h = h_init + np.random.normal(0, 0.02, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Initialize completely random population
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        h = h * (n_points / 2) / h_sum
        return h
    
    # Fitness evaluation
    def evaluate_fitness(h):
        h = repair(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Find initial best
    fitnesses = [evaluate_fitness(p) for p in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolutionary Strategy
    generation = 0
    stagnation_count = 0
    current_mutation_strength = mutation_strength
    
    while generation < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate all individuals
        fitnesses = [evaluate_fitness(p) for p in population]
        
        # Update best
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = int(elite_ratio * population_size)
        elite_indices = sorted_indices[:elite_size]
        elite_population = [population[i] for i in elite_indices]
        
        # Generate offspring with Gaussian mutation
        offspring = []
        for _ in range(offspring_size):
            parent = elite_population[np.random.randint(0, elite_size)]
            # Sample mutation
            mutation = np.random.normal(0, current_mutation_strength * mutation_variance, n_points)
            child = parent + mutation
            # Apply repair
            child = repair(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[elite_size:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Adaptive mutation strength
        current_mutation_strength *= 0.999
        mutation_variance = current_mutation_strength / 5.0
        
        # Stagnation-based restart
        if stagnation_count > restart_threshold:
            current_mutation_strength = mutation_strength * 0.9
            for i in range(population_size):
                if i not in elite_indices:
                    h = np.random.uniform(0, 1, n_points)
                    h_sum = np.sum(h)
                    h = h * (n_points / 2) / h_sum
                    population[i] = h
        
        generation += 1
    
    # Final gradient-based refinement on best solution
    h_opt = best_h.copy()
    
    for _ in range(100):
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        fitness = np.max(corr) * dx
        
        # Compute gradient numerically
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_opt.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        # Step
        h_opt -= 0.001 * grad
        h_opt = repair(h_opt)
        
        # Check for improvement
        corr_new = np.correlate(h_opt, 1 - h_opt, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    final_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    if final_fitness < best_fitness:
        best_h = h_opt
        best_fitness = final_fitness
    
    return (best_h.tolist(), best_fitness, n_points)
