import numpy as np
from scipy.stats import halton
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Generate Halton-based initialization
        seq = halton.Halton(1)
        samples = seq.random(pop_size-1)[:, 0]
        for i in range(pop_size-1):
            h = np.interp(np.linspace(0, 1, n_points), samples[i] + np.arange(n_points)/(n_points-1) - 0.5, np.clip(np.sin(np.linspace(0, 2*np.pi, n_points)), 0, 1))
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    # Add first individual
    h_first = np.zeros(n_points)
    h_first[:n_points//2] = 2.0
    h_first = h_first * (n_points / 2) / np.sum(h_first)
    population.append(h_first.copy())
        
    # Fitness evaluation
    def evaluate(h):
        if time.time() - start_time > budget_s - 10:
            return float('inf')
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Genetic operators
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, rate):
        h_mut = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                delta = np.random.normal(0, 0.1)
                h_mut[i] += delta
        # Repair constraint
        h_mut = np.clip(h_mut, 0, 1)
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    # Main GA loop
    best_fitness = float('inf')
    best_individual = None
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [evaluate(ind) for ind in population]
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_fitness:
            best_fitness = fitnesses[min_idx]
            best_individual = population[min_idx].copy()
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Create new population
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Adjust mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate *= 0.99
    
    # Final refinement
    if best_individual is not None:
        # Gradient based refinement
        best_h = best_individual.copy()
        for _ in range(200):
            if time.time() - start_time > budget_s - 10:
                break
            grad = np.zeros(n_points)
            eps = 1e-5
            base_fitness = evaluate(best_h)
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                fitness_plus = evaluate(h_plus)
                grad[i] = (fitness_plus - base_fitness) / eps
            # Update
            best_h = best_h - 0.005 * grad
            best_h = np.clip(best_h, 0, 1)
            best_h = best_h * (n_points / 2) / np.sum(best_h)
            # Reevaluate
            base_fitness = evaluate(best_h)
        final_fitness = evaluate(best_h)
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_individual = best_h.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
