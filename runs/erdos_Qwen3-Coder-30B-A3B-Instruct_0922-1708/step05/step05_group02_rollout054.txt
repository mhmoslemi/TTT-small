import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h_current = h_init.copy()
    current_c5 = evaluate_h(h_current, dx)
    
    # SA parameters
    T0 = 0.1 * current_c5
    T = T0
    alpha = 0.99
    max_iter = 10000
    min_T = 1e-8
    no_improve_count = 0
    max_no_improve = 1000
    
    best_h = h_current.copy()
    best_c5 = current_c5
    
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        iter_count += 1
        
        # Generate neighbor by perturbing a random subset of elements
        # Perturb about 10% of the elements
        num_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        h_new = h_current.copy()
        
        # Perturb values
        for idx in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[idx] += delta
        
        # Repair and rescale
        h_new = repair_h(h_new, n_points)
        
        # Compute new C5
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject the new solution
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h_current = h_new
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = h_current.copy()
                best_c5 = current_c5
                no_improve_count = 0
            else:
                no_improve_count += 1
        else:
            no_improve_count += 1
        
        # Adaptive cooling
        if T > min_T:
            T *= alpha
            
        # Reset cooling if no improvement for a while
        if no_improve_count > 200:
            T = max(min_T, T * 0.9)
            no_improve_count = 0
    
    # Final gradient-based refinement
    h_final = best_h.copy()
    final_c5 = best_c5
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - final_c5) / eps
        
        h_final = h_final - 0.001 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        final_c5 = np.max(corr) * dx
    
    return (h_final.tolist(), final_c5, n_points)
