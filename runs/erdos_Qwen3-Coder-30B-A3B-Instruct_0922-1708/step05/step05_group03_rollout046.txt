import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Simple local optimizer for refinement
    def local_refine(h_init):
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return -np.max(corr) * dx  # negative for minimization
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        cons = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(objective, h_init, method='SLSQP', bounds=bounds, 
                             constraints=cons, options={'maxiter': 50})
            if result.success:
                h_final = np.clip(result.x, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                corr = np.correlate(h_final, 1 - h_final, mode='full')
                return h_final, -result.fun
        except:
            pass
        return h_init, float('inf')
    
    # Gradient-based local search
    def gradient_refine(h_vec, max_iter=30):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        
        for _ in range(max_iter):
            # Compute gradient using finite differences
            grad = np.zeros_like(h_vec)
            eps = 1e-4
            base_c5 = float('inf')
            
            # Evaluate base
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            base_c5 = np.max(corr) * dx
            
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - base_c5) / eps
                
            # Gradient step
            lr = 0.005
            h_vec = h_vec - lr * grad
            
            # Project to feasible region
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            
            # Evaluate
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            new_c5 = np.max(corr) * dx
            
            if new_c5 >= base_c5:
                break  # No improvement
                
        return h_vec
    
    # Main search loop
    iterations = 0
    while time.time() - start_time < budget_s - 10:
        iterations += 1
        
        # Random perturbations
        candidates = []
        
        # Start with current best
        candidates.append(best_h.copy())
        
        # Add random perturbations
        for _ in range(10):
            h_pert = best_h.copy()
            noise = np.random.normal(0, 0.05, n_points)
            h_pert = h_pert + noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            candidates.append(h_pert)
            
        # Add random feasible solutions
        for _ in range(5):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            candidates.append(h_rand)
            
        # Local refinement of candidates
        refined = []
        for h_cand in candidates:
            h_refined = gradient_refine(h_cand, max_iter=20)
            refined.append(h_refined)
            
        # Evaluate all
        for h_test in refined:
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_test.copy()
                
        # Occasionally do hill climbing
        if iterations % 10 == 0:
            # Hill climb from current best
            h_current = best_h.copy()
            h_next = h_current.copy()
            
            # Random small step
            step_size = 0.02
            for i in range(n_points):
                if np.random.random() < 0.1:
                    h_next[i] += np.random.normal(0, step_size)
                    
            h_next = np.clip(h_next, 0, 1)
            h_next = h_next * (n_points / 2) / np.sum(h_next)
            
            corr = np.correlate(h_next, 1 - h_next, mode='full')
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_next.copy()

    # Final validation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
