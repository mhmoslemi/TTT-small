import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight step"""
    # Use stable distribution sampler (simplified version)
    sigma = scale
    u = np.random.normal(0, 1, size)
    v = np.random.exponential(1, size)
    step = sigma * u / (np.abs(v) ** (1/alpha))
    return step

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 50
    population = []
    population_values = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
    
    # Add more random initial candidates
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        
    # Evaluate initial population
    for h in population:
        value = evaluate_h(h, dx)
        population_values.append(value)
    
    # Track best solution
    best_idx = np.argmin(population_values)
    best_h = population[best_idx].copy()
    best_value = population_values[best_idx]
    
    # Parameters
    max_iter = 1000
    levy_alpha = 1.5
    levy_scale = 0.5
    learning_rate = 0.001
    perturbation_rate = 0.1
    top_candidates = 10
    patience = 100
    
    # Main loop
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new candidates via Lévy flights
        new_candidates = []
        new_values = []
        
        for h in population:
            # Lévy flight perturbation
            step = levy_flight(n_points, levy_alpha, levy_scale)
            perturbed = h + step * perturbation_rate
            perturbed = repair_h(perturbed, n_points)
            new_candidates.append(perturbed)
            
        # Evaluate new candidates
        for h in new_candidates:
            value = evaluate_h(h, dx)
            new_values.append(value)
            
        # Combine with current population
        all_candidates = population + new_candidates
        all_values = population_values + new_values
        
        # Keep top candidates
        top_indices = np.argsort(all_values)[:population_size]
        population = [all_candidates[i] for i in top_indices]
        population_values = [all_values[i] for i in top_indices]
        
        # Gradient descent on top candidates
        top_indices = np.argsort(population_values)[:top_candidates]
        for idx in top_indices:
            h = population[idx].copy()
            current_value = population_values[idx]
            
            for _ in range(50):  # Limited gradient steps per candidate
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                
                h = h - learning_rate * grad
                h = repair_h(h, n_points)
                corr = correlate(h, 1 - h, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < population_values[idx]:
                    population[idx] = h.copy()
                    population_values[idx] = current_value
                    
                    if current_value < best_value:
                        best_h = h.copy()
                        best_value = current_value
                        last_improvement = iteration
        
        # Early stopping if no improvement
        if iteration - last_improvement > patience:
            break
            
    # Final refinement with gradient descent
    h_final = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_final.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        h_final = h_final - 0.0005 * grad
        h_final = repair_h(h_final, n_points)
        corr = correlate(h_final, 1 - h_final, mode='full')
        best_value = np.max(corr) * dx
    
    return (h_final.tolist(), best_value, n_points)
