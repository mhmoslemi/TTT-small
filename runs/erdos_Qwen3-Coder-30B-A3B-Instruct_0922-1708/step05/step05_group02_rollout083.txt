import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_h = h_init.copy()
    else:
        # Initialize with random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_h = h_init.copy()
    
    # Optimizer parameters
    alpha = 0.25               # Discovery rate
    epsilon = 1e-5             # Perturbation step
    max_iter = 1000
    patience = 200             # For early stopping
    best_h = current_h.copy()
    best_c5 = evaluate_h(current_h, dx)
    no_improvement_count = 0
    local_search_freq = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Cuckoo search: generate new solution by perturbing current
        new_h = current_h.copy()
        # Randomly select a subset to perturb
        num_perturb = max(1, int(n_points * 0.1))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            new_val = current_h[i] + np.random.normal(0, epsilon)
            new_h[i] = new_val
            
        # Repair and rescale
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject based on probability
        if new_c5 < best_c5:
            # Accept immediately if better
            current_h = new_h.copy()
            if new_c5 < best_c5:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
        else:
            # Accept with probability based on discovery rate
            if np.random.rand() < alpha:
                current_h = new_h.copy()
        
        # Local search occasionally
        if iteration % local_search_freq == 0 and iteration > 0:
            # Gradient descent refinement for best solution
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = current_h.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / epsilon
                
                current_h = current_h - 0.0005 * grad
                current_h = repair_h(current_h, n_points)
                corr = correlate(current_h, 1 - current_h, mode='full')
                best_c5 = np.max(corr) * dx
                
                if time.time() - start_time > budget_s - 10:
                    break
        
        no_improvement_count += 1
        if no_improvement_count > patience:
            # Early stopping
            break
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / epsilon
        
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
