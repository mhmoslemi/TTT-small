import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    local_search_freq = 50
    rejuvenation_rate = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        best_h = np.random.uniform(0, 1, n_points)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    best_c5 = compute_c5(best_h)
    
    # Simple hill climbing with local search
    current_h = best_h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Local search step
        if iteration % local_search_freq == 0:
            # Gradient-based local search
            h_candidate = current_h.copy()
            for _ in range(20):
                corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                fitness = np.max(corr) * dx
                
                grad = np.zeros(n_points)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h_candidate.copy()
                    h_plus[i] += eps
                    h_plus = repair(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    fitness_plus = np.max(corr_plus) * dx
                    grad[i] = (fitness_plus - fitness) / eps
                
                h_candidate -= 0.001 * grad
                h_candidate = repair(h_candidate)
                
                new_corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
                new_fitness = np.max(new_corr) * dx
                if new_fitness >= fitness:
                    break
                else:
                    fitness = new_fitness
            
            current_h = h_candidate.copy()
            c5_candidate = compute_c5(current_h)
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = current_h.copy()
        
        # Random perturbation to escape local optima
        if np.random.random() < rejuvenation_rate:
            current_h = np.random.uniform(0, 1, n_points)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Basic random walk with some exploration
        if np.random.random() < 0.5:
            idx = np.random.randint(0, n_points)
            current_h[idx] = np.random.uniform(0, 1)
            current_h = repair(current_h)
        
        # Compute new C5 value
        new_c5 = compute_c5(current_h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
