import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, scale=0.5, alpha=1.5):
    """Generate Lévy flight perturbations."""
    return np.random.laplace(0, scale, size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    swarm_size = 50
    particles = []
    best_h = None
    best_c5 = float('inf')
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Add random initial particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Main loop
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 10 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Evaluate all particles
        particle_values = [evaluate_h(p, dx) for p in particles]
        
        # Update best solution
        min_idx = np.argmin(particle_values)
        if particle_values[min_idx] < best_c5:
            best_c5 = particle_values[min_idx]
            best_h = particles[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Generate new particles with Lévy flights
        new_particles = []
        for p in particles:
            # Generate Lévy flight perturbation
            perturbation = levy_flight(n_points, scale=0.1, alpha=1.5)
            new_p = p + perturbation
            
            # Apply repair
            new_p = repair_h(new_p, n_points)
            new_particles.append(new_p)
            
            # Gradient descent on best particles
            if iteration % 10 == 0 and len(particle_values) > 0:
                top_k = min(10, len(particle_values) // 5)
                indices = np.argsort(particle_values)[:top_k]
                for idx in indices:
                    h = particles[idx].copy()
                    current_c5 = particle_values[idx]
                    
                    # Simple gradient approximation
                    eps = 1e-5
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        h_plus = h.copy()
                        h_plus[j] += eps
                        h_plus = repair_h(h_plus, n_points)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - current_c5) / eps
                    
                    # Update with gradient descent
                    h = h - 0.001 * grad
                    h = repair_h(h, n_points)
                    corr = correlate(h, 1 - h, mode='full')
                    current_c5 = np.max(corr) * dx
                    
                    if current_c5 < particle_values[idx]:
                        new_particles[idx] = h.copy()
                        particle_values[idx] = current_c5
        
        particles = new_particles.copy()
        
    # Final refinement
    if best_h is not None:
        h_refined = best_h.copy()
        prev_best = best_c5
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - prev_best) / eps
            
            h_refined = h_refined - 0.0001 * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_c5 = np.max(corr) * dx
            
            if current_c5 < prev_best:
                prev_best = current_c5
            else:
                break
        best_h = h_refined
        best_c5 = prev_best
    
    return (best_h.tolist(), best_c5, n_points)
