import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent or fallback
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Scale pyramid: 10, 20, 40, 80, 100 points
    scales = [10, 20, 40, 80, 100]
    start_time = time.time()
    
    for scale in scales:
        if time.time() - start_time > budget_s - 50:
            break
            
        # Upsample if needed
        if scale > len(h):
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, scale)
            h_interp = np.interp(x_target, x_orig, h)
            h = h_interp
        else:
            h = h[:scale]
            
        # Convex relaxation approach at each scale
        n = len(h)
        h_var = Variable(n)
        penalty_weight = 0.1
        
        # Soft constraints with penalties
        constraints = [h_var >= 0, h_var <= 1]
        
        # Approximate objective with truncated average over a few lags
        lags_to_consider = min(n // 4, 10)
        lags = range(1, lags_to_consider + 1)
        avg_overlap = 0
        total_lags = 0
        
        for k in lags:
            # Compute correlation for a single lag
            corr_k = 0
            for i in range(n-k):
                corr_k += h_var[i] * (1 - h_var[i+k])
            avg_overlap += corr_k
            total_lags += 1
            
        if total_lags > 0:
            avg_overlap /= total_lags
        
        # Add regularization terms to penalize constraint violations
        penalty_h_min = penalty_weight * sum(pos(-h_var))
        penalty_h_max = penalty_weight * sum(pos(h_var - 1))
        total_objective = avg_overlap + penalty_h_min + penalty_h_max
        
        # Add sum constraint
        constraints.append(sum(h_var) == n/2)
        
        # Solve optimization problem
        prob = Problem(Minimize(total_objective), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status == 'optimal':
                h_new = np.array(h_var.value).flatten()
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n / 2) / np.sum(h_new)
                h = h_new
        except:
            pass
            
        # Local gradient-based refinement
        max_refinement_steps = 50
        lr = 0.001
        
        for step in range(max_refinement_steps):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute gradient via finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
                
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (n / 2) / np.sum(h)
            
            # Evaluate and track best
            corr = np.correlate(h, 1 - h, mode="full")
            c5_new = np.max(corr) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    # Final check
    corr_final = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(corr_final) * dx
    
    # Fallback to current best
    if final_c5 >= best_c5:
        final_c5 = best_c5
        h_final = best_h
    else:
        h_final = best_h
        
    # Final validation
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
