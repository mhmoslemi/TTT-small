import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_region_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    gradient_backtrack = 0.5
    max_time = time.time() + budget_s - 10
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Get initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    # Main loop
    for iteration in range(max_iterations):
        if time.time() > max_time:
            break
            
        # Compute gradient
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        dominant_k = np.argmax(corr)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient: derivative of overlap with respect to h[i]
            # ∂/∂h_i [ h[i] (1 - h[i+k]) + h[i-1] (1 - h[i+k-1]) + ...]
            # For dominant lag k, this is simpler
            k = dominant_k - n_points + 1  # convert full correlation index
            if k >= 0:
                grad[i] = (1 - h[(i - k) % n_points]) + (1 - h[(i + k) % n_points])
            else:
                grad[i] = (1 - h[(i - k) % n_points]) + (1 - h[(i + k) % n_points])
        
        # Normalize gradient
        grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else grad
        grad = grad * dx
        
        # Trust region step
        step = -trust_region_radius * grad
        h_new = h + step
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject step
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            trust_region_radius = min(trust_region_radius * 1.2, 1.0)
        else:
            trust_region_radius = max(trust_region_radius * 0.8, min_trust_radius)
        
        # Perturb using Simulated Annealing
        if iteration % perturbation_freq == 0 and iteration > 0:
            # SA perturbation proportional to temperature
            perturbation = np.random.normal(0, temperature, n_points)
            h_perturbed = h + perturbation
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            # Accept with probability based on temperature and energy difference
            if c5_pert < best_c5 or np.random.rand() < np.exp(-(c5_pert - best_c5) / (temperature + 1e-10)):
                h = h_perturbed.copy()
                if c5_pert < best_c5:
                    best_c5 = c5_pert
        
        # Update temperature
        temperature *= cooling_rate
    
    # Final refinement using gradient descent
    for _ in range(500):
        corr = np.correlate(h, 1 - h, mode='full')
        dominant_k = np.argmax(corr)
        k = dominant_k - n_points + 1
        
        # Gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k >= 0:
                grad[i] = (1 - h[(i - k) % n_points]) + (1 - h[(i + k) % n_points])
            else:
                grad[i] = (1 - h[(i - k) % n_points]) + (1 - h[(i + k) % n_points])
        
        # Normalize and step
        grad = grad / np.max(np.abs(grad)) if np.max(np.abs(grad)) > 0 else grad
        grad = grad * dx
        
        h_new = h - 0.001 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            h = h_new.copy()
        else:
            break
    
    # Final check
    if best_c5 < 0.3807:
        # Bug fix: correct implementation
        h = best_h.copy()
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (h.tolist(), float(best_c5), n_points)
