import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    initial_alpha = 0.01
    min_alpha = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    alpha = initial_alpha
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find current dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        current_max_k = max_k
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Gradient for h[i] with respect to overlap at k
            above_idx = (i - current_max_k) % n_points
            below_idx = (i + current_max_k) % n_points
            gradient[i] = (1 - h[above_idx]) + (1 - h[below_idx])
        
        # Normalize gradient
        gradient = gradient / n_points
        
        # Take trust region step
        step = -alpha * gradient
        
        # Project to feasible region
        h_candidate = h + step
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr_candidate) * dx
        
        if candidate_c5 < current_c5:
            h = h_candidate
            current_c5 = candidate_c5
            # Increase trust region
            alpha = min(alpha * 1.2, 1.0)
            if candidate_c5 < best_c5:
                best_h = h.copy()
                best_c5 = candidate_c5
        else:
            # Decrease trust region
            alpha = max(alpha * 0.8, min_alpha)
            
        # Apply simulated annealing perturbation
        if iteration % perturb_freq == 0:
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    perturb_mag = temperature * h[i] * (1 - h[i])
                    delta = np.random.uniform(-perturb_mag, perturb_mag)
                    h[i] += delta
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Update temperature
        temperature *= cooling_rate
        
    # Final refinement with gradient descent
    for _ in range(500):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            above_idx = (i - max_k) % n_points
            below_idx = (i + max_k) % n_points
            gradient[i] = (1 - best_h[above_idx]) + (1 - best_h[below_idx])
        gradient = gradient / n_points
        step = -0.001 * gradient
        h_new = best_h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
        else:
            break
            
    return (best_h.tolist(), best_c5, n_points)
