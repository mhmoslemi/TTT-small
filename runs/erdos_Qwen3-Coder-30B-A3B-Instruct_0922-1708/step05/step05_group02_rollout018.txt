import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight steps."""
    # Generate random steps from Lévy distribution
    sigma = scale * (np.random.gamma(1 + alpha, 1) / (np.random.gamma(1, 1) * np.cos(np.pi * alpha / 2)))**(1/alpha)
    return np.random.standard_normal(size) * sigma

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize candidates
    num_candidates = 50
    candidates = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
    
    # Add more random initial candidates
    for _ in range(num_candidates - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Evaluate initial candidates
    values = [evaluate_h(c, dx) for c in candidates]
    
    # Find best initial solution
    best_idx = np.argmin(values)
    best_h = candidates[best_idx].copy()
    best_value = values[best_idx]
    
    max_iter = 1000
    alpha = 1.5  # Lévy flight exponent
    scale = 0.5  # Lévy flight scale
    learning_rate = 0.001
    perturb_rate = 0.1
    patience = 100
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create new candidates using Lévy flight
        new_candidates = []
        for idx, h in enumerate(candidates):
            # Apply Lévy flight perturbation
            perturbation = levy_flight(n_points, alpha, scale)
            new_h = h + perturbation * perturb_rate
            new_h = repair_h(new_h, n_points)
            new_candidates.append(new_h)
        
        # Evaluate new candidates
        new_values = [evaluate_h(c, dx) for c in new_candidates]
        
        # Update top candidates with gradient descent
        top_k = min(10, len(candidates) // 5)
        indices = np.argsort(new_values)[:top_k]
        
        for idx in indices:
            h = new_candidates[idx].copy()
            current_c5 = new_values[idx]
            
            # Gradient descent to refine
            for _ in range(20):  # Number of gradient steps
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Update with gradient
                h = h - learning_rate * grad
                h = repair_h(h, n_points)
                corr = correlate(h, 1 - h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_value:
                    best_h = h.copy()
                    best_value = current_c5
                    last_improvement = iteration
                
            new_candidates[idx] = h.copy()
            new_values[idx] = current_c5
        
        # Update candidates with new bests
        candidates = new_candidates
        values = new_values
        
        # Update best
        current_best_idx = np.argmin(values)
        if values[current_best_idx] < best_value:
            best_h = candidates[current_best_idx].copy()
            best_value = values[current_best_idx]
            last_improvement = iteration
        
        # Early stopping
        if iteration - last_improvement > patience:
            break
            
    # Final intensive refinement on best solution
    best_h_refined = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h_refined = best_h_refined - 0.0005 * grad
        best_h_refined = repair_h(best_h_refined, n_points)
        corr = correlate(best_h_refined, 1 - best_h_refined, mode='full')
        best_value = np.max(corr) * dx
    
    return (best_h_refined.tolist(), best_value, n_points)
