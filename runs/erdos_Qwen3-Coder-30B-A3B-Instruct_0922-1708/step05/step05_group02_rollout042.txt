import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(n_points, scale=0.5, alpha=1.5):
    """Generate a Lévy flight perturbation."""
    # Generate Lévy distributed random numbers
    # Using the stable distribution approach
    sigma = scale
    # Simple approximation of Lévy distribution
    # This is a basic implementation, more sophisticated versions exist
    u = np.random.normal(0, 1, n_points)
    v = np.random.normal(0, 1, n_points)
    # Basic stable distribution for alpha = 1.5
    step = sigma * (u / np.abs(v)**(1/alpha))
    return step

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Generate initial candidates
    candidates = []
    num_initial = 50
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        candidates.append(h_init)
        
    # Add random initial candidates
    for _ in range(num_initial - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        candidates.append(h)
    
    # Store objective values
    values = [evaluate_h(c, dx) for c in candidates]
    
    # Best solution tracking
    best_idx = np.argmin(values)
    best_h = candidates[best_idx].copy()
    best_value = values[best_idx]
    
    # Parameters
    max_iter = 1000
    top_candidates = 10
    levy_scale = 0.5
    levy_alpha = 1.5
    learning_rate = 0.001
    max_no_improve = 100
    
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create new candidates via Lévy flights
        new_candidates = []
        
        for i in range(len(candidates)):
            # Create perturbation
            perturb = levy_flight(n_points, levy_scale, levy_alpha)
            
            # Apply perturbation
            new_h = candidates[i] + perturb
            
            # Repair
            new_h = repair_h(new_h, n_points)
            
            new_candidates.append(new_h)
        
        # Evaluate new candidates
        new_values = [evaluate_h(c, dx) for c in new_candidates]
        
        # Update candidates with better ones
        for i in range(len(candidates)):
            if new_values[i] < values[i]:
                candidates[i] = new_candidates[i]
                values[i] = new_values[i]
                
                # Update best if necessary
                if values[i] < best_value:
                    best_value = values[i]
                    best_h = candidates[i].copy()
                    no_improve_count = 0
                else:
                    no_improve_count += 1
    
        # Perform gradient descent on top candidates
        top_indices = np.argsort(values)[:top_candidates]
        
        for idx in top_indices:
            h = candidates[idx].copy()
            current_value = values[idx]
            
            # Gradient descent iterations
            for _ in range(20):  # Fewer iterations for speed

                h = h - learning_rate * grad

                corr = correlate(h, 1 - h, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < values[idx]:
                    candidates[idx] = h.copy()
                    values[idx] = current_value
                    
                    if current_value < best_value:
                        best_value = current_value
                        best_h = h.copy()
                        no_improve_count = 0
                    else:
                        no_improve_count += 1
                
                if time.time() - start_time > budget_s - 10:
                    break
                
        # Early stopping if no improvement for many iterations
        if no_improve_count >= max_no_improve:
            break
    
    # Final refinement with gradient descent on best solution
    best_h_refined = best_h.copy()

        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        best_h_refined = best_h_refined - 0.0005 * grad
        best_h_refined = repair_h(best_h_refined, n_points)
        corr = correlate(best_h_refined, 1 - best_h_refined, mode='full')
        best_value = np.max(corr) * dx
        
    return (best_h_refined.tolist(), best_value, n_points)
