import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Population-based genetic algorithm with local refinement
    population_size = 100
    generations = 500
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Create initial population
    population = []
    for i in range(population_size):
        if 'initial_h_values' in globals() and i < population_size // 2:
            # Start with perturbed version of initial solution
            individual = h + np.random.normal(0, 0.05, n_points)
        else:
            individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Main loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            corr = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Track best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Selection (tournament)
        selected = []
        tournament_size = 5
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner].copy())
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1, parent2 = selected[i], selected[i+1]
            
            if np.random.rand() < crossover_rate:
                # Blend crossover with alpha in [0.7, 0.95]
                alpha = 0.7 + 0.25 * np.random.rand()
                child1 = alpha * parent1 + (1 - alpha) * parent2
                child2 = (1 - alpha) * parent1 + alpha * parent2
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            for j in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[j] += np.random.normal(0, 0.03)
                if np.random.rand() < mutation_rate:
                    child2[j] += np.random.normal(0, 0.03)
            
            # Feasibility projection
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.extend([child1, child2])
        
        # Keep only the required population size
        population = new_population[:population_size]
        
        # Local refinement every 50 generations
        if gen % 50 == 0 and gen > 0:
            # Get top 10% of the population
            top_indices = np.argsort(fitness)[:population_size//10]
            for idx in top_indices:
                individual = population[idx].copy()
                # Gradient ascent to improve local solution
                for _ in range(30):
                    grad = np.zeros_like(individual)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = individual.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    individual -= 0.001 * grad
                    individual = np.clip(individual, 0, 1)
                    individual = individual * (n_points / 2) / np.sum(individual)
                    corr = np.correlate(individual, 1 - individual, mode='full')
                    new_fitness = np.max(corr) * dx
                    if new_fitness < fitness[idx]:
                        fitness[idx] = new_fitness
                        population[idx] = individual.copy()
    
    # Final refinement
    for _ in range(30):
        temp_fitness = []
        for individual in population:
            corr = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(corr) * dx
            temp_fitness.append(c5)
        
        best_idx = np.argmin(temp_fitness)
        if temp_fitness[best_idx] < best_c5:
            best_c5 = temp_fitness[best_idx]
            best_h = population[best_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
