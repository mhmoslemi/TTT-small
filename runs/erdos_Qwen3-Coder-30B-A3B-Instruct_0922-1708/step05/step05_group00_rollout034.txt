import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Resolution parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy sum constraint and clip
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse grid optimization - take a low-res solution to escape local minima
    dx_coarse = 2.0 / n_start
    h_coarse = np.array([np.mean(h[i:i+2]) for i in range(0, n_points, 2)])
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Use CVXPY to optimize over coarse grid
    try:
        h_coarse_var = cp.Variable(n_start)
        C5_var = cp.Variable()
        
        constraints = [
            h_coarse_var >= 0,
            h_coarse_var <= 1,
            cp.sum(h_coarse_var) == n_start / 2,
            C5_var >= 0
        ]
        
        # Use coarse model to estimate bounds
        h_coarse_cvx = h_coarse_var.value
        if h_coarse_cvx is not None:
            h_coarse = np.array(h_coarse_cvx)
            h_coarse = np.clip(h_coarse, 0, 1)
            
    except Exception:
        pass
    
    # Up-sample to fine grid and refine
    h = np.array([h_coarse[i//2] for i in range(n_points)])
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Gradient-based refinement of the finer solution
    lr = 0.01
    max_iter = 3000
    patience = 0
    last_c5 = float('inf')
    
    for iterations in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with adaptive learning rate
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        lr = max(1e-5, lr * 0.9995)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
