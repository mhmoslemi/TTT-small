import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    
    return h

def mutate_solution(h, n_points, mutation_rate=0.15):
    h_new = h.copy()
    num_mutations = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, num_mutations, replace=False)
    
    for idx in indices:
        h_new[idx] = 1 - h_new[idx]
    
    return h_new

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_best = h_init.copy()
    else:
        current_best = np.random.rand(n_points)
        current_best = current_best * (n_points / 2) / np.sum(current_best)
        
    current_c5 = evaluate_h(current_best, dx)
    best_h = current_best.copy()
    best_c5 = current_c5
    
    max_no_improvement = 100
    no_improvement_count = 0
    
    mutation_rate = 0.15
    num_mutations = 100
    
    while time.time() - start_time < budget_s - 10:
        improved = False
        
        for _ in range(num_mutations):
            if time.time() - start_time > budget_s - 10:
                break
            
            mutated = mutate_solution(current_best, n_points, mutation_rate)
            mutated = repair_h(mutated, n_points)
            mutated_c5 = evaluate_h(mutated, dx)
            
            if mutated_c5 < current_c5:
                current_best = mutated
                current_c5 = mutated_c5
                improved = True
                
                if mutated_c5 < best_c5:
                    best_h = mutated.copy()
                    best_c5 = mutated_c5
                    no_improvement_count = 0
                break
        
        if not improved:
            no_improvement_count += 1
            if no_improvement_count > max_no_improvement:
                break
                
            # Adjust mutation rate
            mutation_rate = min(0.5, mutation_rate * 1.1)
            
    # Final gradient descent refinement
    def gradient_step(h, dx, learning_rate=0.001):
        eps = 1e-6
        grad = np.zeros_like(h)
        orig_value = evaluate_h(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            plus_value = evaluate_h(h_plus, dx)
            grad[i] = (plus_value - orig_value) / eps
            
        h_new = h - learning_rate * grad
        return repair_h(h_new, n_points)
    
    ref_h = best_h.copy()
    ref_c5 = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        ref_h = gradient_step(ref_h, dx)
        new_c5 = evaluate_h(ref_h, dx)
        if new_c5 < ref_c5:
            ref_c5 = new_c5
            best_h = ref_h.copy()
            best_c5 = ref_c5
    
    return (best_h.tolist(), best_c5, n_points)
