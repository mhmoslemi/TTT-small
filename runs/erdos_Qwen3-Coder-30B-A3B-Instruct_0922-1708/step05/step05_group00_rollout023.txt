        n_points = len(h)

        dx = 2.0 / n_points
        # Use a structured initialization to explore better regions
        h = np.zeros(n_points)
        for i in range(n_points):
            x = i * dx
            h[i] = 0.5 + 0.3 * math.sin(2 * math.pi * x)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def Correlation(h):
        # np.correlate with mode='full' gives us overlap for all shifts
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    def smooth_max(x, beta=10):
        # Smooth approximation to maximum
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Optimizer state
    best_h = h.copy()
    best_c5 = Correlation(h)
    
    def gradient_step(h, lr=0.01, beta=10):
        # Compute gradiens using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        current_val = Correlation(h)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            val_plus = Correlation(h_plus)
            grad[i] = (val_plus - current_val) / eps
            
        # Update with gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    # Main optimization loop
    lr = 0.01
    patience = 0
    patience_limit = 1000
    last_c5 = best_c5
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Gradient descent step
        h = gradient_step(h, lr)
        c5 = Correlation(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= patience_limit:
                break
                
        if abs(c5 - last_c5) < 1e-6:
            patience += 1
            if patience >= patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = c5
        lr = max(1e-6, lr * 0.9995)
        iteration += 1
    
    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = Correlation(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
