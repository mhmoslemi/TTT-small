import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 6000
    dx = 2.0 / n_points
    α = 0.01
    decay_freq = 200
    eval_freq = 50
    stagnation_limit = 200
    perturb_freq = 100
    perturb_fraction = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_s = np.sum(corr) * dx
    
    # Optimization loop
    last_improvement = 0
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Decay learning rate
        if iteration % decay_freq == 0 and iteration > 0:
            α /= 2
            
        # Evaluate current
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        s = np.sum(corr) * dx
        
        # Update best
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            best_s = s
            last_improvement = iteration
            
        # Check for stagnation
        if iteration - last_improvement > stagnation_limit:
            break
            
        # Sample every eval_freq iterations
        if iteration % eval_freq == 0:
            # Compute gradient-like adjustments
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                elif h_plus[i] < 0:
                    h_plus[i] = 0
                # Rescale
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Adjust h
            h = h - α * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally perturb
        if iteration % perturb_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturb_fraction, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
