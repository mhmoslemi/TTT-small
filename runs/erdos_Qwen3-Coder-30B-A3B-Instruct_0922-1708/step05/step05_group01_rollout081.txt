import time

        h = np.ones(n_points) * 0.5
    
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    def compute_c5(h):
        # Correlate h with (1-h) for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        # Maximize over all lags
        c5 = np.max(corr) * dx
        return c5
    
    c5 = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5
    
    # Gradient-based optimization with adaptive step size
    max_iter = 5000
    alpha = 1e-3
    decay_factor = 0.5
    grad_threshold = 1e-6
    line_search_factor = 0.5
    random_perturbation = 0.01
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute all correlations and find max lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Compute gradients analytically for that specific lag
        # Get the lag value from index
        lag = max_lag_idx - (n_points - 1)
        # Special case: when lag > 0, we have h[i] * (1 - h[i+lag])
        # When lag < 0, we have h[i+|lag|] * (1 - h[i])
        gradient = np.zeros_like(h)
        
        # For the provided lag
        if lag >= 0:
            # h[i] * (1 - h[i+lag]) 
            for i in range(n_points - lag):
                gradient[i] += 1 - h[i + lag]
                if i < n_points - lag:  # Swapped for correct indexing
                    gradient[i + lag] -= h[i]
        else:
            # h[i] * (1 - h[i+lag]) where lag is negative, so h[i+|lag|] term is involved
            for i in range(-lag, n_points):
                gradient[i] += 1 - h[i + lag]
                if i + lag < n_points:
                    gradient[i + lag] -= h[i]
        
        # Normalize gradient for correct scaling factor
        gradient = dx * gradient
        
        # Line search
        step_size = alpha
        trial_h = h - step_size * gradient
        # Project to feasible region
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
        trial_c5 = compute_c5(trial_h)
        
        # Backtracking line search
        max_attempts = 10
        attempts = 0
        while trial_c5 >= c5 and attempts < max_attempts:
            step_size *= line_search_factor
            trial_h = h - step_size * gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
            trial_c5 = compute_c5(trial_h)
            attempts += 1

        if trial_c5 < c5:
            h = trial_h
            c5 = trial_c5
            if c5 < best_c5:

                best_h = h.copy()
        else:
            # Reduce step size if no progress
            alpha *= 0.9
        
        # Early stopping if improvement is small
        if abs(c5 - best_c5) < 1e-8:
            break
            
        # Occasionally perturb for escape from local minima
        if iteration % 100 == 0:
            perturbation = np.random.normal(0, random_perturbation, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
