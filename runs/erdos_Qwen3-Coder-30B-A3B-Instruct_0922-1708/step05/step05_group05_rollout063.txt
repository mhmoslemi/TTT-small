import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.001
    perturbation_frequency = 50
    perturbation_magnitude = 0.1
    decay_factor = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy constraint sum(h) * dx = 0.5
    h = h * (0.5 / np.sum(h))
    
    # Precompute indices for correlation computation
    full_corr_length = 2 * n_points - 1
    zero_idx = n_points - 1
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute correlation 
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find the minimum correlation value
        min_corr_idx = np.argmin(corr)
        min_corr_val = corr[min_corr_idx]
        
        # Compute gradient for the minimum correlation
        gradient = np.zeros(n_points)
        
        # Compute gradient at each point based on the correlation positions
        for i in range(n_points):
            # The correlation at position min_corr_idx involves h[i] with 1-h[i+shift]
            # Compute which position in the correlation corresponds to this interaction
            # This is a simplified approximation using direct finite differences
            shift = min_corr_idx - zero_idx
            if i + shift >= 0 and i + shift < n_points:
                # Gradient contribution from the interaction h[i] and h[i+shift]
                gradient[i] += corr[min_corr_idx] * dx
                
        # Update h with gradient ascent (we want to maximize the minimum correlation)
        h = h + learning_rate * gradient
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.sum(h))
        
        # Periodic perturbations to escape local optima
        if iteration % perturbation_frequency == 0:
            noise = np.random.normal(0, perturbation_magnitude / (1.0 + iteration / decay_factor), n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.sum(h))
            
        # Evaluate current solution
        c5_test = compute_c5(h, dx)
        
        if c5_test < best_c5:
            best_c5 = c5_test
            best_h = h.copy()
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def compute_c5(h, dx):
    """Compute C5 from h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
