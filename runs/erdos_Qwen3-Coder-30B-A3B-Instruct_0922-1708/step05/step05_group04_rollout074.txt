import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    max_iter = 500
    learning_rate = 0.01
    eps = 1e-4
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients
    h_fft = fft(h_init)
    coeffs = np.concatenate([h_fft.real[:n_fourier], h_fft.imag[:n_fourier]])
    
    def h_from_coeffs(c):
        # Reconstruct h from Fourier coefficients
        real_coeffs = c[:n_fourier]
        imag_coeffs = c[n_fourier:]
        h_fft = np.zeros(n_points, dtype=complex)
        h_fft[0] = real_coeffs[0]
        h_fft[1:n_fourier] = real_coeffs[1:] + 1j * imag_coeffs[1:]
        h_fft[-1:-n_fourier:-1] = real_coeffs[1:][::-1] - 1j * imag_coeffs[1:][::-1]
        h = np.real(ifft(h_fft))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient_step(c):
        h = h_from_coeffs(c)
        base_c5 = compute_c5(h)
        grad = np.zeros_like(c)
        
        for i in range(len(c)):
            c_plus = c.copy()
            c_plus[i] += eps
            h_plus = h_from_coeffs(c_plus)
            plus_c5 = compute_c5(h_plus)
            grad[i] = (plus_c5 - base_c5) / eps
        
        return grad
    
    # Optimization loop
    current_coeffs = coeffs.copy()
    best_c5 = float('inf')
    best_h = h_init.copy()
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 10:
            break
            
        grad = gradient_step(current_coeffs)
        current_coeffs -= learning_rate * grad
        
        # Clip coefficients
        current_coeffs = np.clip(current_coeffs, -10, 10)
        
        h = h_from_coeffs(current_coeffs)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
    
    return (best_h.tolist(), best_c5, n_points)
