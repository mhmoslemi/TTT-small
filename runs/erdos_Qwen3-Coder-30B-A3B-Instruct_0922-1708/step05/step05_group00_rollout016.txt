        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))

        # Compute overlap with 1-h for all shifts
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return correlation * (2.0 / n_points)
    
    def objective(h_vals):
        overlap = compute_overlap(h_vals)
        return smooth_max(overlap, beta)
    
    def gradient(h_vals):
        overlap = compute_overlap(h_vals)
        exp_beta = np.exp(beta * overlap)
        sum_exp_beta = np.sum(exp_beta)
        # Gradients for each shift
        grad = np.zeros_like(h_vals)
        # Manually compute gradient for overlap term
        for i in range(len(h_vals)):
            # Forward difference approximation
            eps = 1e-5
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            overlap_plus = compute_overlap(h_plus)

    beta1 = 0.9
    beta2 = 0.999

        overlap = compute_overlap(h)
        c5_current = np.max(overlap)

            patience = 0
        else:
            patience += 1
