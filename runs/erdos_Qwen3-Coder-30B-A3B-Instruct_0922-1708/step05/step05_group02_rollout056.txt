import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_value = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters

    current_h = best_h.copy()
    current_value = best_value
    
    no_improve_count = 0
    iteration = 0
    
    # Global optimization loop
    while iteration < max_iter and time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Cuckoo search: Generate new solution via Lévy flight
        # Modified random walk with step size drawn from heavy-tailed distribution
        step_size = np.random.standard_cauchy(n_points) * 0.1
        new_h = current_h + step_size
        
        # Repair new solution to maintain constraints
        new_h = repair_h(new_h, n_points)
        new_value = evaluate_h(new_h, dx)
        
        # Accept new solution with probability
        if new_value < current_value or np.random.rand() < alpha:

            current_value = new_value
        
        # Update global best if better
        if current_value < best_value:
            best_h = current_h.copy()

            no_improve_count = 0
