import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, target_sum, dx):
    """Repair h to ensure it satisfies constraints"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    diff = current_sum - target_sum
    if diff > 0:
        # Remove excess
        indices = np.argsort(h)[::-1]
        for idx in indices:
            if diff <= 0:
                break
            dec = min(h[idx], diff)
            h[idx] -= dec
            diff -= dec
    else:
        # Add deficit
        indices = np.argsort(h)
        for idx in indices:
            if diff >= 0:
                break
            inc = min(1 - h[idx], -diff)
            h[idx] += inc
            diff += inc
    return h * (target_sum / np.sum(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (target_sum / np.sum(h_init))
        h = h_init.copy()
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (target_sum / np.sum(h))
    
    # PSO parameters
    swarm_size = 50
    max_iter = 1000
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    local_search_freq = 50
    perturb_rate = 0.2
    max_local_iter = 50
    
    # PSO initialization
    particles = [h.copy() for _ in range(swarm_size)]
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(swarm_size)]
    
    pbest_positions = [p.copy() for p in particles]
    pbest_values = [evaluate_h(p, dx) for p in particles]
    
    gbest_idx = np.argmin(pbest_values)
    gbest_position = particles[gbest_idx].copy()
    gbest_value = pbest_values[gbest_idx]
    
    # PSO main loop
    for iteration in range(max_iter):
        if time.time() - start_time >= budget_s - 10:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                           phi_p * r1 * (pbest_positions[i] - particles[i]) +
                           phi_g * r2 * (gbest_position - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = repair_h(particles[i], target_sum, dx)
            
            # Evaluate fitness
            fitness = evaluate_h(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_values[i]:
                pbest_values[i] = fitness
                pbest_positions[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_value:
                    gbest_value = fitness
                    gbest_position = particles[i].copy()
        
        # Local search
        if iteration % local_search_freq == 0 and iteration > 0:
            # Optimize best particles
            sorted_indices = np.argsort(pbest_values)[:swarm_size//2]
            for idx in sorted_indices:
                h_current = pbest_positions[idx].copy()
                
                # L-BFGS-B optimization
                bounds = Bounds([0.0]*n_points, [1.0]*n_points)
                def objective(x):
                    return evaluate_h(x, dx)
                
                def gradient(x):
                    eps = 1e-6
                    grad = np.zeros(n_points)
                    for j in range(n_points):
                        x_plus = x.copy()
                        x_plus[j] += eps
                        x_plus = repair_h(x_plus, target_sum, dx)
                        grad[j] = (objective(x_plus) - objective(x)) / eps
                    return grad
                
                try:
                    res = minimize(objective, h_current, method='L-BFGS-B', 
                                 bounds=bounds, jac=gradient, 
                                 options={'maxiter': max_local_iter, 'ftol': 1e-8})
                    if res.success:
                        h_opt = repair_h(res.x, target_sum, dx)
                        new_value = evaluate_h(h_opt, dx)
                        if new_value < pbest_values[idx]:
                            pbest_values[idx] = new_value
                            pbest_positions[idx] = h_opt.copy()
                            if new_value < gbest_value:
                                gbest_value = new_value
                                gbest_position = h_opt.copy()
                except:
                    pass
        
        # Perturbation
        if iteration % 100 == 0 and iteration > 0:
            for _ in range(int(swarm_size * perturb_rate)):
                idx = np.random.randint(0, swarm_size)
                h_new = np.random.uniform(0, 1, n_points)
                h_new = h_new * (target_sum / np.sum(h_new))
                particles[idx] = h_new
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final check with direct optimization
    h_final = gbest_position.copy()
    bounds = Bounds([0.0]*n_points, [1.0]*n_points)
    
    def objective(x):
        return evaluate_h(x, dx)
    
    def gradient(x):
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            x_plus = x.copy()
            x_plus[j] += eps
            x_plus = repair_h(x_plus, target_sum, dx)
            grad[j] = (objective(x_plus) - objective(x)) / eps
        return grad
    
    try:
        res = minimize(objective, h_final, method='L-BFGS-B', 
                      bounds=bounds, jac=gradient, 
                      options={'maxiter': 50, 'ftol': 1e-10})
        if res.success:
            h_final = repair_h(res.x, target_sum, dx)
            gbest_value = evaluate_h(h_final, dx)
    except:
        pass
    
    return (h_final.tolist(), gbest_value, n_points)
