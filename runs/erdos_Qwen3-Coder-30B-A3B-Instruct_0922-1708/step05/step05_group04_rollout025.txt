import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 3000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.random.uniform(0, 1, n_points)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    best_h = h_current.copy()
    best_c5 = compute_c5(h_current, dx)
    
    # PSO parameters
    swarm_size = 50
    inertia_weight = 0.7
    phi_p = 2.0
    phi_g = 2.0
    max_vel = 0.1
    local_search_freq = 50
    
    # Initialize particles
    particles = [h_current.copy()]
    velocities = [np.random.uniform(-0.05, 0.05, n_points)]
    
    for _ in range(swarm_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.05, 0.05, n_points))
    
    # Personal bests
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [compute_c5(p, dx) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Helper function for feasibility
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
                if diff <= 0:
                    break
        else:
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
                if diff >= 0:
                    break
        return h * (n_points / 2) / np.sum(h)
    
    # PSO main loop
    for iter_ in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        for i in range(swarm_size):
            # Velocity update
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia_weight * velocities[i] + 
                            phi_p * r1 * (pbest_particles[i] - particles[i]) +
                            phi_g * r2 * (gbest_particle - particles[i]))
            
            # Clamp velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Position update
            particles[i] += velocities[i]
            
            # Repair
            particles[i] = repair(particles[i])
            
            # Evaluate fitness
            fitness = compute_c5(particles[i], dx)
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest_fitness[i] = fitness
                pbest_particles[i] = particles[i].copy()
                
                # Update global best
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[i].copy()
        
        # Periodic local search
        if iter_ % local_search_freq == 0 and iter_ > 0:
            for i in range(swarm_size):
                # L-BFGS-B local refinement
                try:
                    h_ref = pbest_particles[i].copy()
                    bounds = Bounds([0] * n_points, [1] * n_points)
                    res = minimize(
                        fun=lambda x: compute_c5(x, dx),
                        x0=h_ref,
                        method='L-BFGS-B',
                        bounds=bounds,
                        options={'maxiter': 10, 'ftol': 1e-6, 'gtol': 1e-6}
                    )
                    if res.success:
                        h_opt = res.x.copy()
                        h_opt = repair(h_opt)
                        fitness_new = compute_c5(h_opt, dx)
                        if fitness_new < pbest_fitness[i]:
                            pbest_fitness[i] = fitness_new
                            pbest_particles[i] = h_opt.copy()
                            if fitness_new < gbest_fitness:
                                gbest_fitness = fitness_new
                                gbest_particle = h_opt.copy()
                except:
                    pass
        
        # Update best if necessary
        for i in range(swarm_size):
            if pbest_fitness[i] < best_c5:
                best_c5 = pbest_fitness[i]
                best_h = pbest_particles[i].copy()
    
    # Final refinement with L-BFGS-B
    try:
        bounds = Bounds([0] * n_points, [1] * n_points)
        res = minimize(
            fun=lambda x: compute_c5(x, dx),
            x0=gbest_particle,
            method='L-BFGS-B',
            bounds=bounds,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        if res.success:
            h_final = res.x.copy()
            h_final = repair(h_final)
            c5_final = compute_c5(h_final, dx)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx
