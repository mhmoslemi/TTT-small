import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    initial_mutation_rate = 0.1
    tournament_size = 5
    elite_count = 5
    hall_of_fame_size = 10
    
    # Generate Halton sequence for initial population
    sampler = qmc.Halton(d=n_points, seed=seed)
    samples = sampler.random(n=population_size)
    
    # Convert to step function candidates
    population = []
    for i in range(population_size):
        h = samples[i] * 2  # Scale to [0,2]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population[0] = h_init.copy()

    # Track hall of fame
    hall_of_fame = []
    best_solution = None
    best_fitness = float('inf')

    start_time = time.time()
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_solution = population[min_fitness_idx].copy()
            
        # Add to hall of fame
        sorted_indices = np.argsort(fitnesses)
        for i in range(min(elite_count, len(sorted_indices))):
            h = population[sorted_indices[i]]
            if not any(np.allclose(h, hof_h, atol=1e-6) for hof_h in hall_of_fame):
                hall_of_fame.append(h.copy())
                if len(hall_of_fame) > hall_of_fame_size:
                    hall_of_fame.pop(0)
        
        # Selection via tournament
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Crossover and mutation
        new_population = []
        
        # Elitism
        elite_indices = sorted_indices[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest via crossover and mutation
        while len(new_population) < population_size:
            # Selection
            parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
            parent1, parent2 = selected[parent1_idx], selected[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            mutation_rate = initial_mutation_rate * (0.99 ** (gen // 10))
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                    child2[i] += np.random.normal(0, 0.01)
            
            # Repair
            for child in [child1, child2]:
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child.copy())
                
                if len(new_population) >= population_size:
                    break
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    # Final local refinement of best solution
    if best_solution is not None:
        h_final = best_solution.copy()
        best_fitness_final = best_fitness
        
        # Gradient-based local search (finite differences)
        for _ in range(1000):
            if time.time() - start_time > budget_s - 10:
                break
            
            grad = np.zeros(n_points)
            eps = 1e-5
            base_corr = np.correlate(h_final, 1 - h_final, mode='full')
            fitness_base = np.max(base_corr) * dx
            
            improved = False
            for j in range(n_points):
                h_plus = h_final.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                
                grad[j] = (fitness_plus - fitness_base) / eps
                
                if fitness_plus < fitness_base:
                    improved = True
            
            if not improved:
                break
            
            # Update with step size
            step_size = 0.001
            h_final -= step_size * grad
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            corr = np.correlate(h_final, 1 - h_final, mode='full')
            fitness_new = np.max(corr) * dx
            
            if fitness_new < fitness_base:
                best_fitness_final = fitness_new
            else:
                break
                
        best_fitness = best_fitness_final
    
    # Final validation and reporting
    if best_solution is not None:
        # Check constraints
        assert np.all(best_solution >= 0) and np.all(best_solution <= 1)
        assert abs(np.sum(best_solution) - n_points/2) < 1e-6, "Sum constraint violated"
        
        # Final computation
        corr = np.correlate(best_solution, 1 - best_solution, mode='full')
        final_fitness = np.max(corr) * dx
        return (best_solution.tolist(), final_fitness, n_points)
    
    # Fallback if no solution found
    fallback = np.zeros(n_points)
    fallback[:n_points//2] = 2.0
    fallback = fallback * (n_points / 2) / np.sum(fallback)
    corr = np.correlate(fallback, 1 - fallback, mode='full')
    fallback_fitness = np.max(corr) * dx
    return (fallback.tolist(), fallback_fitness, n_points)
