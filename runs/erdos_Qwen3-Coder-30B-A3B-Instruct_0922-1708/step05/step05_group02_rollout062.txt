import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    no_improvement_count = 0
    max_no_improvement = 100
    mutation_rate = 0.15
    
    # Main search loop
    while time.time() - start_time < budget_s - 10:
        # Generate mutations
        num_mutations = 100
        for _ in range(num_mutations):
            # Create a mutated version
            h_mut = best_h.copy()
            # Flip a random subset of elements
            num_flips = int(mutation_rate * n_points)
            flip_indices = np.random.choice(n_points, num_flips, replace=False)
            for idx in flip_indices:
                h_mut[idx] = 1 - h_mut[idx]
            
            # Repair to maintain constraints
            h_mut = repair_h(h_mut, n_points)
            
            # Evaluate
            c5_mut = evaluate_h(h_mut, dx)
            
            if c5_mut < best_c5:
                best_h = h_mut.copy()
                best_c5 = c5_mut
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        # Check termination condition
        if no_improvement_count >= max_no_improvement:
            break
            
        # Adapt mutation rate
        if no_improvement_count % 20 == 0 and mutation_rate < 0.3:
            mutation_rate *= 1.1
    
    # Final gradient descent refinement
    max_refinements = 1000
    for _ in range(max_refinements):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
