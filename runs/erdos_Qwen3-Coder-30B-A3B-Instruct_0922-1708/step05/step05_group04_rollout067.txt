import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import quad_form, Variable, Minimize, Problem

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    k_points = 50
    k_range = np.linspace(-2, 2, k_points)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0.1, 0.9, n_points)
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_opt = h_init.copy()
    
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    def feasible(h, mu):
        # Check if all constraints ∫h(x)h(x+k)dx ≥ mu are satisfied
        h = repair_h(h)
        violations = []
        for k in k_range:
            shift = int(k / dx)
            if shift >= 0:
                corr = np.sum(h[:-shift] * (1 - h[shift:]))
            else:
                corr = np.sum(h[-shift:] * (1 - h[:shift]))
            violations.append(mu - corr)
        return min(violations) >= 0
    
    # Binary search on C5 bound
    low = 0.0
    high = 0.5
    best_bound = 1.0
    best_h = h_opt.copy()
    
    iterations = 0
    max_iterations = 20
    
    while high - low > 1e-5 and iterations < max_iterations and time.time() - start_time < budget_s - 10:
        mu = (low + high) / 2
        
        # Solving dual problem using CVXPY
        h_var = Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2,
        ]
        
        # Objective function inspired by max overlap constraint
        # This encourages the minimum overlap in each constraint
        objective = Minimize(cp.sum_squares(h_var - 0.5) - mu * cp.sum(h_var))
        
        # Create problem
        problem = Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.SCS, verbose=False)
            h_sol = h_var.value
            
            if h_sol is not None:
                h_sol = repair_h(h_sol)
                if feasible(h_sol, mu):
                    low = mu
                    if mu < best_bound:
                        best_bound = mu
                        best_h = h_sol.copy()
                else:
                    high = mu
            else:
                high = mu
        except:
            high = mu
            
        iterations += 1
    
    # To get the final optimized value, do a refinement with gradient descent
    h_refined = best_h.copy()
    min_c5 = compute_overlap(h_refined)
    
    # Gradient-based refinement
    for iter_ref in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-4
        
        base_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - base_fitness) / eps
        
        h_refined -= 0.001 * grad
        h_refined = repair_h(h_refined)
        
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_fitness = np.max(corr) * dx
        
        if new_fitness >= min_c5:
            break
        min_c5 = new_fitness
    
    final_c5 = compute_overlap(h_refined)
    return (h_refined.tolist(), final_c5, n_points)
