import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Ensure feasibility
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T0 = 10.0
    cooling_rate = 0.9995
    
    # Annealing loop
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        # Temperature
        T = T0 / np.log(1 + iteration)
        
        # Perturb h
        h_trial = h + np.random.normal(0, 0.02, n_points)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5 values
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(corr_trial) * dx
        
        corr_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr_current) * dx
        
        # Accept or reject
        delta = c5_trial - c5_current
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_trial.copy()
            c5 = c5_trial
        else:
            c5 = c5_current
            
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        iteration += 1
        
        # Occasionally refine with gradient descent
        if iteration % 10 == 0 and iteration > 100:
            h_refined = best_h.copy()
            lr = 0.001
            for _ in range(50):
                if time.time() - start_time > budget_s - 15:
                    break
                grad = np.zeros_like(h_refined)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - best_c5) / eps
                
                h_refined = h_refined - lr * grad
                h_refined = np.clip(h_refined, 0, 1)
                h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
                
                corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5_new = np.max(corr_new) * dx
                
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_refined.copy()
                
                # Early stopping if no improvement
                if c5_new > c5_new:
                    break
            
            h = best_h.copy()
    
    # Final refinement using gradient descent on best solution
    h_refined = best_h.copy()
    lr = 0.001
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refined.copy()
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
