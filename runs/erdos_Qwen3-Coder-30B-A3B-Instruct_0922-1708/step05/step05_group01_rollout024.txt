import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Parameters
    learning_rate = 1e-3
    penalty_weight = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    max_iter = 5000
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize penalty
    penalty_mu = penalty_weight
    
    # Optimization loop
    last_improvement = 0
    iteration = 0
    plateau_count = 0
    current_c5 = float('inf')
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        current_c5 = c5
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            plateau_count = 0
        else:
            plateau_count += 1
        
        # Check for stagnation
        if plateau_count >= plateau_patience:
            # Reduce learning rate
            learning_rate *= learning_rate_decay
            # Apply random perturbation
            pert = np.random.normal(0, perturbation_fraction, n_points)
            h += pert
            # Repair
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            plateau_count = 0
            penalty_mu = penalty_weight  # reset penalty
        
        # Compute gradient for objective and penalty
        # Find the max lag k*
        max_idx = np.argmax(corr)
        k_star = max_idx - (n_points - 1)
        
        # Create gradient approximation
        grad = np.zeros(n_points)
        eps = 1e-5
        
        # For each element h[i], compute partial derivative
        for i in range(n_points):
            # Forward difference approximation
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Gradient for penalty: sum constraint
        penalty_grad = np.zeros(n_points)
        sum_violation = np.sum(h) - n_points / 2
        penalty_grad += 2 * penalty_mu * sum_violation
        
        # Add bound constraint penalty gradients
        bound_penalty_grad = np.zeros(n_points)
        bound_penalty_grad += 2 * penalty_mu * np.maximum(0, -h)
        bound_penalty_grad += 2 * penalty_mu * np.maximum(0, h - 1)
        
        total_grad = grad + penalty_grad + bound_penalty_grad
        
        # Update h
        h -= learning_rate * total_grad
        
        # Repair to maintain feasibility
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive penalty
        if abs(sum_violation) > 1e-6 or np.any(h < 0) or np.any(h > 1):
            penalty_mu *= penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_min)
    
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
