import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    initial_alpha = 0.01
    alpha_decay = 0.5
    alpha_growth = 1.2
    perturbation_freq = 50
    max_perturbation = 0.05
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use a simple construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate current solution
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    start_time = time.time()
    
    alpha = initial_alpha
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient approximation via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update h
        h_new = h - alpha * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept step if improvement
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            alpha *= alpha_growth
        else:
            alpha *= alpha_decay
            
        # Apply random perturbation to escape local minima
        if iteration % perturbation_freq == 0:
            perturbation = np.random.uniform(-max_perturbation, max_perturbation, n_points)
            h_pert = h + perturbation
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert) 
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            if c5_pert < best_c5:
                h = h_pert.copy()
                best_c5 = c5_pert
                
    # Final validation
    corr_final = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h.tolist(), final_c5, n_points)
