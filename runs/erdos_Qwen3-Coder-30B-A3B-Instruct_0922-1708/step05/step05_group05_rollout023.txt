import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use a block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)

    # Helper function to compute C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx

    # Gradient estimation via finite differences
    def estimate_gradient(h_vec, eps=1e-5):
        base_c5 = compute_c5(h_vec)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad

    # Trust region optimization with simulated annealing
    max_iter = 1000
    trust_radius = 0.01
    min_trust_radius = 1e-5
    temp = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Estimate gradient
        grad = estimate_gradient(best_h)
        
        # Compute candidate step
        step = -trust_radius * grad
        
        # Project step to feasible region
        new_h = best_h + step
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        new_c5 = compute_c5(new_h)
        
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            trust_radius = min(1.2 * trust_radius, 0.1)
        else:
            trust_radius = max(min_trust_radius, 0.8 * trust_radius)
            
        # Apply simulated annealing perturbation periodically
        if iteration % perturbation_freq == 0:
            perturb_mask = np.random.rand(n_points) < 0.3
            if np.any(perturb_mask):
                # Apply perturbations
                for i in range(n_points):
                    if perturb_mask[i]:
                        delta = temp * (np.random.rand() - 0.5)
                        new_h[i] += delta
                        new_h[i] = np.clip(new_h[i], 0, 1)
                # Normalize after perturbation
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                new_c5 = compute_c5(new_h)
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    
        temp *= cooling_rate
    
    # Final refinement with gradient descent
    for _ in range(500):
        grad = estimate_gradient(best_h)
        step = -0.001 * grad
        new_h = best_h + step
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        new_c5 = compute_c5(new_h)
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
        else:
            break
            
    return (best_h.tolist(), best_c5, n_points)
