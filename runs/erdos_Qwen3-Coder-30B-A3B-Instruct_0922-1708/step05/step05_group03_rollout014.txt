import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points_list = [10, 20, 40, 80, 100]
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Fallback to parent if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (100 / 2) / np.sum(h_parent)
        best_h = h_parent.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / 100)
    
    # Multi-scale optimization
    h_prev = None
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 150:
            break
            
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5  # Default uniform
        
        if h_prev is not None:
            # Interpolate from previous scale
            x_prev = np.linspace(0, 2, len(h_prev))
            x_curr = np.linspace(0, 2, n_points)
            h = np.interp(x_curr, x_prev, h_prev)
        
        # Use convex relaxation for optimization
        try:
            # Formulate problem
            h_var = cp.Variable(n_points)
            # Objective: maximize avg overlap over central lags (to minimize C5)
            # We do this indirectly by minimizing a relaxed sum of overlaps
            # Simplified convex relaxation using average over significant lags
            m = min(n_points // 4, 10)  # reasonable lag window
            lag_indices = list(range(-m, 0)) + list(range(1, m + 1)) 
            overlap_exprs = []
            for k in lag_indices:
                if k < 0:
                    expr = cp.sum(cp.multiply(h_var[:n_points+k], (1 - h_var[-k:])))
                else:
                    expr = cp.sum(cp.multiply(h_var[k:], (1 - h_var[:-k])))
                overlap_exprs.append(expr)
            
            # Sum all overlaps and minimize
            objective = cp.Minimize(cp.sum(overlap_exprs) * dx / len(overlap_exprs))
            
            # Constraints
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # Solve
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h = h_var.value
            else:
                # fallback to previous if solver failed
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception as e:
            # use previous or default if convex solve fails
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Projection to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_prev = h.copy()
        
        # Evaluate
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Local gradient refinement
    if best_h is not None and (time.time() - start_time) < budget_s - 50:
        h = best_h.copy()
        dx = 2.0 / 100
        lr = 0.001
        eps = 1e-4
        steps = 50
        
        for _ in range(steps):
            if time.time() - start_time > budget_s - 30:
                break
            grad = np.zeros(100)
            for i in range(100):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (100 / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5) / eps
            
            h = h - lr * grad
            h = np.clip(h, 0, 1)
            h = h * (100 / 2) / np.sum(h)
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * (2.0 / 100)
    
    return (best_h.tolist(), final_c5, 100)
