import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with the provided initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize randomly
        h_init = np.random.rand(n_points)
    
    h_init = repair_h(h_init, n_points)
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        # Generate a new candidate by applying random mutations
        candidate = best_h.copy()
        # Randomly mutate a percentage of elements
        num_mutations = max(1, int(n_points * 0.15))
        mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
        
        for i in mutation_indices:
            # Flip or slightly modify the value
            if np.random.random() < 0.5:
                # Flip between 0 and 1
                candidate[i] = 1.0 - candidate[i]
            else:
                # Add small random change
                delta = np.random.normal(0, 0.05)
                candidate[i] += delta
        
        # Repair and rescale to maintain constraints
        candidate = repair_h(candidate, n_points)
        candidate_c5 = evaluate_h(candidate, dx)
        
        if candidate_c5 < best_c5:
            best_h = candidate.copy()
            best_c5 = candidate_c5
            last_improvement = iteration
        
        # Terminate if no improvement in last 100 iterations
        if iteration - last_improvement > 100:
            break
    
    # Final refinement using gradient descent
    h_refined = best_h.copy()
    prev_c5 = best_c5
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_c5) / eps
        
        h_refined = h_refined - 0.002 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        prev_c5 = np.max(corr) * dx
        
        if prev_c5 < best_c5:
            best_c5 = prev_c5
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
