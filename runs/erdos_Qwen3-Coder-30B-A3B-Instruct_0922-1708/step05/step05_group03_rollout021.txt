import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute cross-correlation indices
    max_lag = n_points - 1
    
    # Genetic Algorithm parameters
    pop_size = 100
    generations = 50
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = h + np.random.normal(0, 0.02, n_points)
        else:
            individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            corr = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Sort by fitness
        sorted_idx = np.argsort(fitness)
        population = [population[i] for i in sorted_idx]
        fitness = [fitness[i] for i in sorted_idx]
        
        # Create new population
        new_population = population[:elite_size]  # Elitism
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            # Selection (tournament)
            tournament_size = 5
            tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if np.random.rand() < crossover_rate:
                # Pick second parent
                tournament_indices2 = np.random.choice(pop_size, tournament_size, replace=False)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
                
                # Blend crossover
                alpha = 0.7 + 0.25 * np.random.rand()
                child1 = alpha * population[winner_idx] + (1 - alpha) * population[winner_idx2]
                child2 = (1 - alpha) * population[winner_idx] + alpha * population[winner_idx2]
            else:
                child1 = population[winner_idx].copy()
                child2 = population[winner_idx].copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project to feasible region
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            # Add to population
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    # Final local refinement
    def eval_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Refine best solution with gradient descent
    h_ref = best_h.copy()
    lr = 0.001
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros_like(h_ref)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = eval_c5(h_plus)
            grad[j] = (c5_plus - best_c5) / eps
        
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        current_c5 = eval_c5(h_ref)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_ref.copy()
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
