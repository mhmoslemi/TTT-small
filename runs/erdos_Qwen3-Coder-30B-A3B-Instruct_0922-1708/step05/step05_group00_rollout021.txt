    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth approximation of max function
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Gradient computation using finite differences
    def compute_gradient(h, dx, beta=10.0):
        eps = 1e-5
        grad = np.zeros_like(h)
        cor_current = np.correlate(h, 1 - h, mode='full') * dx
        c5_current = smooth_max(cor_current, beta)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = smooth_max(cor_plus, beta)
            grad[i] = (c5_plus - c5_current) / eps
        
        return grad
    
    # Project to feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Optimizer with adaptive learning rate and early stopping
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    max_patience = 500
    
    # Main optimization loop
    lr = 0.01
    beta = 10.0
    
    while time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = compute_gradient(best_h, dx, beta)
        
        # Update with adaptive learning rate
        h_new = best_h - lr * grad
        h_new = project(h_new)
        
        # Evaluate new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full') * dx
        current_c5 = np.max(correlation)
        
        # Check improvement
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping
        if patience > max_patience:
            break
            
        # Update learning rate with cosine annealing
        lr = 0.01 * (0.5 * (1 + np.cos(np.pi * (len(best_h) - patience) / max_patience)))
        lr = max(1e-6, lr)
        
        # Occasionally refresh with better starting points
        if patience % 200 == 0 and patience > 0:
            # Start from scratch or use a better initialization
            h_trial = np.random.rand(n_points) * 0.5 + 0.25  # Spread out

            for _ in range(100):
                grad_trial = compute_gradient(h_trial, dx, beta)
                h_trial = h_trial - 0.005 * grad_trial
                h_trial = project(h_trial)
            
            trial_correlation = np.correlate(h_trial, 1 - h_trial, mode='full') * dx

                best_c5 = trial_c5
                best_h = h_trial.copy()
                patience = 0
    
    # Final verification and cleanup
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_correlation)
    
    # Ensure correct normalization and bounds

        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), float(final_c5), n_points)
