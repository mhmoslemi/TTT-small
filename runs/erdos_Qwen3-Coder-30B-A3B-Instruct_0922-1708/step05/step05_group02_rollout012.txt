import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Mutation-based search
    max_no_improve = 100
    no_improve_count = 0
    mutation_size = int(n_points * 0.15)  # 15% of elements
    
    while time.time() - start_time < budget_s - 10 and no_improve_count < max_no_improve:
        # Generate multiple mutations
        num_mutations = 100
        for _ in range(num_mutations):
            if time.time() - start_time > budget_s - 10:
                break
            # Create a copy of the best solution
            mut_h = best_h.copy()
            
            # Randomly flip or modify elements
            indices = np.random.choice(n_points, mutation_size, replace=False)
            for idx in indices:
                # Flip value with 50% probability or modify slightly
                if np.random.random() < 0.5:
                    mut_h[idx] = 1 - mut_h[idx]
                else:
                    # Add a small perturbation
                    perturbation = np.random.normal(0, 0.02)
                    mut_h[idx] += perturbation
            
            # Repair the mutation
            mut_h = repair_h(mut_h, n_points)
            
            # Evaluate the mutated solution
            mut_c5 = evaluate_h(mut_h, dx)
            
            # If better, update the best
            if mut_c5 < best_c5:
                best_h = mut_h.copy()
                best_c5 = mut_c5
                no_improve_count = 0  # Reset counter
            else:
                no_improve_count += 1
    
    # Final gradient descent refinement on best solution
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
