import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mutation_strength = 0.5
    initial_sigma = 0.1
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function to repair solutions
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        scale = (n_points / 2) / h_sum
        h = h * scale
        return h
    
    # Evaluation function
    def evaluate(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main evolution loop
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    sigma = initial_sigma
    mu = mutation_strength
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness of entire population
        fitnesses = [evaluate(ind) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection: keep top 50% (elitism)
        selected_indices = np.argsort(fitnesses)[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring with Gaussian mutation
        offspring = []
        for _ in range(offspring_size):
            parent = selected_population[np.random.randint(0, len(selected_population))]
            # Mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mu * noise
            # Repair
            child = repair(child)
            offspring.append(child)
            
        # Replace worst individuals
        worst_indices = np.argsort(fitnesses)[population_size // 2:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Adaptive mutation parameters
        mu *= 0.999
        sigma = mu / 5.0
        
        # Reset if stagnation
        if stagnation_count > 200:
            mu = mutation_strength * 0.9
            sigma = mu / 5.0
            for i in range(population_size):
                if i != current_best_idx:
                    population[i] = population[current_best_idx] + np.random.normal(0, 0.05, n_points)
                    population[i] = repair(population[i])
            stagnation_count = 0
    
    # Final refinement with gradient descent
    if best_individual is not None:
        h_ref = best_individual.copy()
        for _ in range(100):
            # Gradient approximation
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_ref.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad  
            h_ref = repair(h_ref)
            
            # Check for convergence
            corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
                
        best_individual = h_ref
        best_fitness = evaluate(best_individual)
    
    return (best_individual.tolist(), best_fitness, n_points)
