import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize z to represent binary decisions
    z = (h > 0.5).astype(float)
    
    # Current best
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Augmented Lagrangian parameters
    mu = 0.1
    max_iter = 5000
    lr = 0.001
    perturb_frac = 0.01
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute the lag that gives max correlation
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        c5 = max_corr * dx
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient for selected lag
        # For this lag we compute the gradient for all points
        lag = max_lag_idx - n_points + 1  # Convert to shift index
        if lag >= 0:
            # y_prev shifted: y[i] where i = lag + j
            grad = np.zeros(n_points)
            for j in range(n_points - lag):
                grad[j] += 1 - h[j + lag]
            for j in range(lag, n_points):
                grad[j] += 1 - h[j - lag]
            grad -= 1
        else:
            # y_next shifted: y[i] where i = -lag + j
            grad = np.zeros(n_points)
            for j in range(-lag, n_points):
                grad[j] += 1 - h[j + lag]
            for j in range(n_points + lag):
                grad[j] += 1 - h[j - lag]
            grad -= 1
        
        # Gradient descent step for h
        h_new = h - lr * grad
        
        # Project h_new to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to enforce constraint sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z based on new h_new
        z = (h_new > 0.5).astype(float)
        
        # Update h
        h = h_new
        
        # Increase penalty parameter
        mu *= 1.05
    
    # Final validation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
