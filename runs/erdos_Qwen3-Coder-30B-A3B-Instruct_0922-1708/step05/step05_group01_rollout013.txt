import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Optimization parameters
    max_iter = 6000
    lr = 0.01
    lr_decay = 0.5
    decay_interval = 200
    stagnation_limit = 200
    perturb_freq = 100
    perturb_fraction = 0.01
    
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Decay learning rate
        if iteration % decay_interval == 0:
            lr *= lr_decay
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find the maximum overlap position(s)
        max_corr = np.max(corr)
        max_positions = np.where(corr == max_corr)[0]
        max_lag = max_positions[len(max_positions)//2] - n_points + 1  # middlemost lag
        
        # Create gradient estimates
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update h
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        h = h_new
        corr = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Perturbation to escape local minima
        if iteration % perturb_freq == 0:
            mask = np.random.rand(n_points) < perturb_fraction
            h[mask] += np.random.normal(0, 0.01, np.sum(mask))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Stagnation check
        if iteration - last_improvement > stagnation_limit:
            # Add stochastic perturbation
            noise = np.random.normal(0, 0.01, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
    
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
