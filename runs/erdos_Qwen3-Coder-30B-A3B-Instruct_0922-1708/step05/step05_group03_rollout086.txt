import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Scale sequence
    scales = [10, 20, 40, 80, 100]
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Start with parent or uniform
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
    else:
        h_parent = np.ones(100) * 0.5

    best_h = h_parent.copy()
    best_c5 = float('inf')
    
    # Multi-scale optimization
    h_current = None
    for scale in scales:
        if time.time() - start_time > budget_s - 100:
            break
            
        n_scale = scale
        dx_scale = 2.0 / n_scale
        # Initialize at current scale
        if h_current is None:
            h = np.ones(n_scale) * 0.5
        else:
            # Upsample from previous scale
            h = np.interp(np.linspace(0, 2, n_scale), np.linspace(0, 2, len(h_current)), h_current)
            # Clip and rescale
            h = np.clip(h, 0, 1)
            h = h * (n_scale / 2) / np.sum(h)
        
        # Convex relaxation
        h_var = cp.Variable(n_scale)
        
        # Objective: approximate max overlap with average over central lags
        # Using lags from 1 to n_scale/4 for smoothing (as approximation)
        max_lag = min(n_scale//4, n_scale-1)
        if max_lag <= 0:
            max_lag = 1
            
        # Build approximation of overlap in bounds
        # Approximate C5 as mean of overlaps with small lags
        # Just use a few lags for reshape
        if max_lag > 0:
            lag_weights = np.ones(max_lag)
            lag_weights = lag_weights / np.sum(lag_weights)
            mean_overlap = 0
            for k in range(1, max_lag+1):
                corr_coeffs = np.concatenate([np.ones(k), np.zeros(n_scale - k)]) 
                corr_coeffs_full = np.concatenate([np.zeros(n_scale), corr_coeffs])
                corr_coeffs_full = corr_coeffs_full[:n_scale]
                mean_overlap += lag_weights[k-1] * cp.sum(cp.multiply(h_var, corr_coeffs_full))
            objective = cp.Minimize(mean_overlap)
        else:
            objective = cp.Minimize(cp.sum(h_var))
            
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_scale / 2
        ]
        
        # Solve with CVXPY
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000)
            if prob.status in ['optimal', 'optimal_inaccurate']:
                h = h_var.value
            else:
                # fallback
                h = h * (n_scale / 2) / np.sum(h)
        except:
            h = h * (n_scale / 2) / np.sum(h)
            
        # Project to valid range
        h = np.clip(h, 0, 1)
        h = h * (n_scale / 2) / np.sum(h)
        
        # Back-projection if necessary
        if scale == 100:
            h_current = h.copy()
        else:
            h_current = h.copy()

    # Local refinement using gradient descent on finite differences
    if time.time() - start_time > budget_s - 100:
        pass
    else:
        h_ref = h_current.copy()
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        lrate = 0.001
        for _ in range(50):
            if time.time() - start_time > budget_s - 50:
                break
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx 
                grad[j] = (c5_plus - best_c5) / eps
            h_ref = h_ref - lrate * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    # Final evaluation and return
    if best_c5 == float('inf'):
        corr = np.correlate(h_current, 1 - h_current, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h_current.copy()
        
    return (best_h.tolist(), best_c5, n_points)
