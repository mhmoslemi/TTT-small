import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a simple block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 10.0
    iteration = 0
    start_time = time.time()
    
    # Annealing loop
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        T = T0 / np.log(1 + iteration)
        
        # Perturb solution
        h_new = h + np.random.normal(0, 0.02, n_points)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject based on SA criteria
        delta_c5 = c5_new - best_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    # Refinement with gradient descent
    h_refine = best_h.copy()
    lr = 0.001
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_refine = h_refine - lr * grad
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_refine.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
