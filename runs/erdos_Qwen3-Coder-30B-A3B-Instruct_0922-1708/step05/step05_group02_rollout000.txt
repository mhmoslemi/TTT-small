import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    h = repair_h(h, n_points)
    current_c5 = evaluate_h(h, dx)
    
    # SA parameters
    T = current_c5 * 0.1
    alpha = 0.99
    max_iter = 10000
    min_T = 1e-8
    stagnation_count = 0
    max_stagnation = 100
    
    best_h = h.copy()
    best_c5 = current_c5
    
    # Adaptive cooling params
    last_improvement = 0
    improvement_window = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the solution
        h_new = h.copy()
        # Perturb about 10% of elements
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        
        h_new = repair_h(h_new, n_points)
        new_c5 = evaluate_h(h_new, dx)
        
        # Accept or reject
        delta = new_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                last_improvement = iteration
                stagnation_count = 0
            else:
                stagnation_count += 1
        else:
            stagnation_count += 1
            
        # Cooling schedule
        if T > min_T:
            T *= alpha
            
        # Adaptive cooling: if no progress, slow down cooling
        if iteration - last_improvement > improvement_window:
            alpha = max(0.99, alpha * 0.999)
            
        # Early exit if stagnation
        if stagnation_count > max_stagnation:
            break
    
    # Final gradient-based refinement
    h_refined = best_h.copy()
    current_c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5_refined) / eps
        
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        current_c5_refined = np.max(corr) * dx
        
        if current_c5_refined < best_c5:
            best_h = h_refined.copy()
            best_c5 = current_c5_refined
    
    return (best_h.tolist(), best_c5, n_points)
