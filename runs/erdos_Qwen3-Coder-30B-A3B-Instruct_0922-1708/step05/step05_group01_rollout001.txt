import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    lr = initial_learning_rate
    no_improvement_count = 0
    
    # Gradient descent optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_val = corr[max_lag]
        c5 = max_val * dx
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for plateau
        if no_improvement_count >= plateau_patience:
            if lr > min_learning_rate:
                lr *= learning_rate_decay
                no_improvement_count = 0
            else:
                # Random perturbation
                perturb_mask = np.random.random(n_points) < perturb_fraction
                h[perturb_mask] = np.random.uniform(0, 1, np.sum(perturb_mask))
                h = h * (n_points / 2) / np.sum(h)
                no_improvement_count = 0
                continue
        
        # Compute gradient for the maximum lag
        # Correlation at k-th lag corresponds to convolution: sum(h[i] * (1-h[i+k]))
        # The gradient is roughly: (1 - h shifted)*I_{k} - h*I_{k}
        # But we simplify to taking derivatives of the max element
        # using contribution of only the maximum lag
        
        # For Lag k, we focus on region of overlap
        # The max lag gives us how the function should change
        # We approximate gradient of max according to which elements contribute
        # Simplified: Approximate gradient contribution as difference of correlation
        # shape of correlation: 2*n_points - 1
        # max_lag from 0 to 2*n_points - 2
        # the actual convolution is from h[-k] to h[n_points - 1]
        # Coefficient k:
        #   max_lag = n_points - 1 + k => k = max_lag - (n_points - 1)
        # We want derivative of convolution at k = max_lag
        # To compute the gradient: use derivative of convolution
        
        # For a brute-force approximation:
        # Since correlation is sum_h[i]*h[i+k], 
        # gradient w.r.t h[i] = sign of the convolution contribution
        # which is 1 - h[i+k] when h[i] is involved in convolution
        # reverse convolution results: Exact gradient seems too complex for dynamic
        # Instead use simple finite difference on the correlation max
        
        # Instead take simpler approach: use alternating block incidence matrix
        # Alternate between:
        # Take elements of h_i such that h[i] is part of max_k correlation  
        # This is more complex to compute;

        # Approximate gradient with simple steepest descent derivative 
        # gradient = correlation derivative wrt h
        # Approximate it directly from the maximum indices
        # For correlation at margin k:
        
        # Simplified gradient:
        # Each h[i] affects an entry in convolution at positions i and i+k (when valid)
        # For actual iterated gradient, approximate: 
        # dir_derivative = corr(i + h[i], 1-h[i+k]) - corr(i , 1-h[i+k])
        # But this contextually is from causal build.
        # gradient computation for max corresponds to:
        # gradient = direct derivative of convolution of h with (1-h) at max lags
        # This is non-trivial analytically  
        # Here we compute finite difference of max hinge w.r.t. change of h
        # Approximate by forming a single gradient at max corr:
        # over the structure where entry contributes to maximum lag correlation
        
        # Use approximate gradient by focusing on max_lag
        # This defines positions where derivative should be built
        gradient = np.zeros_like(h)
        
        # Truncation with proper indices for correlation
        # lag being from -n+1 to n-1
        k = max_lag - (n_points - 1)  # corresponds to actual lag
        
        # For the maximum correlation at position k
        # we have contribution to the overlap integral.
        # The indices of h involved in the overlap sum:
        # f(x) = h[i] * (1 - h[i+k]) 
        # for i in [0, n_points-1]  
        # where i+k >= 0 and i+k < n_points  
        # i+ k in [0, n_points - 1]
        # so i in [max(0, -k), min(n_points-1, n_points-1-k)]  
        k_start = max(0, -k)
        k_end = min(n_points - 1, n_points - 1 - k)
        if k >= 0:
            # k_start = 0, k_end = n_points - 1 - k
            for i in range(k_start, k_end + 1): 
                gradient[i] += 1 - h[i + k]
                gradient[i + k] -= h[i] 
        else:  # k < 0  
            # k_start = -k, k_end = n_points - 1
            for i in range(k_start, k_end + 1):
                gradient[i] -= h[i - k]
                gradient[i - k] += 1 - h[i]
        
        # Gradient descent step
        h -= lr * gradient
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
