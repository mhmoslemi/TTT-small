import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary approximation
    z = (h > 0.5).astype(float)
    
    # Augmented Lagrangian parameters
    mu = 0.1
    learning_rate = 0.001
    max_iter = 5000
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Identify the maximum lag
        max_lag_idx = np.argmax(corr)
        k = max_lag_idx - (len(h) - 1)
        
        # Construct gradient for the identified maximizer
        # We optimize y directly using gradient information
        grad = np.zeros_like(h)
        # Approximate gradient for overlap at lag k
        # For each i, compute how much h[i] affects the overlap at k
        # This is a simplified approximation for efficiency
        
        # We'll consider the direct correlation component
        # Simple overall gradient via finite differences for noted lag
        for i in range(len(h)):
            # Increment h[i] slightly to estimate influence
            h_plus = h.copy()
            h_plus[i] += 1e-6
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / 1e-6
        
        # Update h by gradient descent
        h = h - learning_rate * grad
        
        # Project h to [0,1] and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(h_sum - target_sum) > 1e-8:
            h = h * target_sum / h_sum
        
        # Update binary approximation
        z = (h > 0.5).astype(float)
        
        # Update penalty
        mu *= 1.01
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
