import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Simple block construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = np.max(np.correlate(h_init, 1 - h_init, mode='full')) * dx
    
    start_time = time.time()
    
    # Gradient descent with projected steps
    h = h_init.copy()
    alpha = 0.01
    iteration = 0
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 5:
        # Compute current correlation
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-8:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - current_c5) / eps
        
        # Compute next step
        h_new = h - alpha * grad
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new_sum = np.sum(h_new)
        if abs(h_new_sum - n_points / 2) > 1e-8:
            h_new = h_new * (n_points / 2) / h_new_sum
        
        # Evaluate new point
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject step
        if new_c5 < current_c5:
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            alpha *= 1.1  # Increase step size
        else:
            alpha *= 0.5  # Decrease step size
            
        # Occasionally perturb to escape optima
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.01, n_points)
            h_pert = h + noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert_sum = np.sum(h_pert)
            if abs(h_pert_sum - n_points / 2) > 1e-8:
                h_pert = h_pert * (n_points / 2) / h_pert_sum
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            pert_c5 = np.max(corr_pert) * dx
            
            if pert_c5 < best_c5:
                best_h = h_pert.copy()
                best_c5 = pert_c5
                h = h_pert.copy()
        
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
