import numpy as np
import time
from scipy.signal import correlate

def compute_gradient(h, k_star, n_points, dx):
    """Compute gradient for the maximum correlation lag k_star"""
    grad = np.zeros(n_points)
    
    # Compute the correlation for the specific lag
    # For lag k_star, we compare h[i] with h[i + k_star] 
    # but we care about overlaps where both are defined
    
    # For positive lag k_star, h[i] is compared with h[i + k_star]
    # For negative lag, h[i - k_star] is compared with h[i]
    # For k_star=0, h[i] is compared with h[i]
    
    if k_star >= 0:
        # Compare h[i] with h[i + k_star]
        for i in range(n_points - k_star):
            grad[i] += 1 - h[i + k_star]
            grad[i + k_star] += -h[i]
    else:
        # Compare h[i - k_star] with h[i]
        k_abs = -k_star
        for i in range(k_abs, n_points):
            grad[i - k_abs] += 1 - h[i]
            grad[i] += -h[i - k_abs] 
    
    return grad * (2 / n_points)

def line_search(h, grad, dx, step_size=1e-3, max_iter=10):
    """Perform line search to find optimal step size"""
    best_h = h.copy()
    best_c5 = float('inf')
    
    for _ in range(max_iter):
        h_new = h - step_size * grad
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to sum = n_points / 2
        h_new = h_new * (len(h_new) / 2) / np.sum(h_new)
        
        # Evaluate
        corr = correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
        else:
            break
            
        step_size *= 0.5  # Reduce step size
        
    return best_h, best_c5

def project_to_feasible(h, n_points):
    """Project h to feasible region [0,1] with sum = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-8:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    step_size = 1e-3
    stagnation_counter = 0
    stagnation_threshold = 20
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        k_star = np.argmax(corr) - n_points + 1  # Convert full to lag index
        
        # Compute gradient
        grad = compute_gradient(h, k_star, n_points, dx)
        
        # Line search for step size
        h_new, c5_new = line_search(h, grad, dx, step_size)
        
        # Update h
        h = h_new
        c5 = c5_new
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Adapt step size
        if stagnation_counter > 5:
            step_size *= 0.5
            stagnation_counter = 0
            
        # Occasionally perturb
        if iteration % 100 == 0:
            perturb = np.random.normal(0, 1e-4, n_points)
            h = h + perturb
            h = project_to_feasible(h, n_points)
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
