import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_count = 5
    
    # Initialize Halton quasirandom sequence
    sampler = qmc.Halton(d=n_points, seed=seed)
    
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init.copy())
        else:
            # Use block construction as default
            h_init = np.zeros(n_points)
            h_init[:n_points//2] = 2.0
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init.copy())
        
        # Fill rest with Halton-based samples
        samples = sampler.random(population_size - 1)
        for i in range(population_size - 1):
            h = samples[i] * 2  # Scale to [0,2]
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
        
        return population
    
    def evaluate_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            scale = target_sum / current_sum
            h = h * scale
        return h
    
    def tournament_selection(population, fitnesses):
        selected_idx = 0
        best_fitness = float('inf')
        for _ in range(tournament_size):
            idx = np.random.randint(0, len(population))
            if fitnesses[idx] < best_fitness:
                best_fitness = fitnesses[idx]
                selected_idx = idx
        return population[selected_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, rate):
        new_h = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                delta = np.random.normal(0, 0.05)
                new_h[i] += delta
        return repair(new_h)
    
    # Initialize population
    population = initialize_population()
    fitnesses = [evaluate_fitness(h) for h in population]
    
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    start_time = time.time()
    
    # GA iteration
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Rank population
        ranked_indices = np.argsort(fitnesses)
        new_population = [population[i].copy() for i in ranked_indices[:elite_count]]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            new_population.extend([child1, child2])
        
        new_population = new_population[:population_size]
        population = new_population
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Adaptive mutation rate
        if gen % 10 == 0:
            mutation_rate *= 0.99
    
    # Local refinement
    h_refined = best_h.copy()
    for _ in range(1000):
        base_fitness = evaluate_fitness(h_refined)
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            fitness_plus = evaluate_fitness(h_plus)
            grad[i] = (fitness_plus - base_fitness) / eps
        
        h_refined -= 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        new_fitness = evaluate_fitness(h_refined)
        if new_fitness >= base_fitness:
            break
    
    final_fitness = evaluate_fitness(h_refined)
    if final_fitness < best_fitness:
        best_fitness = final_fitness
        best_h = h_refined.copy()
        
    return (best_h.tolist(), best_fitness, n_points)
