import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Initialize population
    population_size = 100
    population = []
    for _ in range(population_size):
        if 'initial_h_values' in globals():
            noise = np.random.normal(0, 0.05, n_points)
            new_h = h + noise
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        else:
            new_h = np.random.rand(n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    # Adaptive search parameters
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    max_generations = 1000
    generations = 0
    
    # Main evolutionary loop
    while time.time() - start_time < budget_s - 10 and generations < max_generations:
        generations += 1
        
        # Evaluate fitness of each individual
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection and breeding
        new_population = []
        fitness = np.array(fitness)
        
        # Elitism: keep best 10%
        elite_indices = np.argsort(fitness)[:population_size//10]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Generate offspring
        for _ in range(population_size - len(elite_indices)):
            # Tournament selection
            selected = []
            for _ in range(2):
                tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
                tournament_fitness = [fitness[i] for i in tournament_indices]
                winner_idx = tournament_indices[np.argmin(tournament_fitness)]
                selected.append(population[winner_idx])
            
            # Crossover
            if np.random.rand() < crossover_rate:
                alpha = np.random.uniform(0.7, 0.95)
                child = alpha * selected[0] + (1 - alpha) * selected[1]
            else:
                child = selected[0].copy()
            
            # Mutation
            for i in range(len(child)):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.03)
            
            # Repair constraints
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Local refinement every 50 generations
        if generations % 50 == 0:
            # Select top individuals
            top_indices = np.argsort(fitness)[:population_size//10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                lr = 0.001
                
                for _ in range(20):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    h_ref = h_ref - lr * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                    
                    corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                    new_c5 = np.max(corr_new) * dx
                    
                    if new_c5 < fitness[idx]:
                        fitness[idx] = new_c5
                        population[idx] = h_ref.copy()
                    
                    if new_c5 < best_c5:
                        best_c5 = new_c5
                        best_h = h_ref.copy()

    # Final gradient refinement
    final_candidates = []
    for _ in range(3):
        if 'initial_h_values' in globals():
            h_candidate = best_h.copy() + np.random.normal(0, 0.02, n_points)
        else:
            h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        final_candidates.append(h_candidate)
    
    # Add top from population
    top_indices = np.argsort(fitness)[:min(2, len(fitness))]
    for i in top_indices:
        final_candidates.append(population[i].copy())
        
    # Gradient refinement
    for h_ref in final_candidates:
        for _ in range(30):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
