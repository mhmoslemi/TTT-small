import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    best_h = h.copy()
    
    # Gradient descent parameters
    learning_rate = 1e-3
    penalty_weight = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    max_iterations = 5000
    
    # Penalty parameter
    mu = penalty_weight
    
    # Optimization loop
    iteration = 0
    patience = 0
    last_c5 = c5
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 5:
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient w.r.t. h for the maximum lag
        max_lag = max_lag_idx - (n_points - 1)
        grad = np.zeros(n_points)
        
        # Compute contribution from the maximum overlap term
        if max_lag > 0:
            # h[i] * (1 - h[i + max_lag]) for i from 0 to n_points - max_lag - 1
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
                grad[i + max_lag] -= h[i]
        elif max_lag < 0:
            # h[i + |max_lag]| * (1 - h[i]) for i from 0 to n_points + max_lag - 1
            for i in range(n_points + max_lag):
                grad[i] -= h[i - max_lag]
                grad[i - max_lag] += 1 - h[i]
        else:
            # h[i] * (1 - h[i]) for all i
            for i in range(n_points):
                grad[i] = 1 - 2 * h[i]
        
        # Penalty gradient
        penalty_grad = np.zeros(n_points)
        for i in range(n_points):
            penalty_grad[i] += 2 * max(0, -h[i])
        penalty_grad += 2 * (np.sum(h) - n_points / 2) * np.ones(n_points)
        
        # Total gradient
        total_grad = grad + mu * penalty_grad
        
        # Line search or direct update
        h_new = h - learning_rate * total_grad
        
        # Feasibility repair
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Re-evaluate
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            last_c5 = c5_new
            patience = 0
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # Check if we should reduce learning rate or perturb
            patience += 1
            if patience >= plateau_patience:
                learning_rate *= 0.5
                # Perturb
                perturb = np.random.normal(0, perturbation_fraction, n_points)
                h = h + perturb
                h = np.clip(h, 0, 1)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                patience = 0
        
        # Update penalty based on constraint violations
        violation_sum = abs(np.sum(h) - n_points / 2)
        if violation_sum > 1e-6:
            mu = mu * penalty_update_factor
        else:
            mu = max(mu / penalty_update_factor, penalty_min)
        
        iteration += 1
    
    # Final evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
