import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize multiple starting points
    initial_solutions = []
    
    # Use parent solution if available
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        initial_solutions.append(h_parent.copy())
    else:
        # Create a baseline
        h_base = np.ones(n_points) * 0.5
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        initial_solutions.append(h_base.copy())
    
    # Add 4 random initializations
    for _ in range(4):
        h_random = np.random.rand(n_points)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        initial_solutions.append(h_random.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Process each initialization
    for i, h_start in enumerate(initial_solutions):
        h = h_start.copy()
        
        # Convex relaxation to improve initial solution
        try:
            # Solution from convex relaxation
            h_relaxed = cp.Variable(n_points)
            constraints = [
                h_relaxed >= 0,
                h_relaxed <= 1,
                cp.sum(h_relaxed) == n_points / 2
            ]
            
            # Objective: minimize sum(h_i * h_{i+k}) over all shifts k
            # For now, simpler: minimize sum(h_i * (1 - h_i)) which approximates our target
            # Better: we want to minimize max of overlap across shifts
            
            # Use SCIP or other solver to get an approximation
            # Tie it to a convex upper bound, here we'll 
            # work directly in the refinement phase
            
        except Exception:
            pass  # Just continue to gradient descent
        
        # Refinement using L-BFGS
        max_iter = 2000
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            def objective(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                return -np.max(correlation)  # Negative because we minimize
            
            def gradient(h_vec):
                h_vec = np.clip(h_vec, 0, 1)
                h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                max_idx = np.argmax(correlation)
                # Approximate gradient using finite differences
                grad = np.zeros_like(h_vec)
                eps = 1e-4
                base_val = objective(h_vec)
                for i in range(len(h_vec)):
                    h_plus = h_vec.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                    c5_plus = -np.max(corr_plus)
                    grad[i] = (c5_plus - base_val) / eps
                return grad
            
            # Minimization with bound constraints
            bounds = [(0, 1) for _ in range(n_points)]
            try:
                res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                               jac=gradient, options={'maxiter': 1})
                if res.success:
                    h = res.x.copy()
                else:
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except Exception:
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            
            # Final evaluation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            current_c5 = np.max(correlation)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
    
    # Ensure we have a valid solution
    if best_h is None:
        # Revert to parent solution or base if nothing was found
        if 'initial_h_values' in globals():
            best_h = np.array(initial_h_values)
            if len(best_h) != n_points:
                x_orig = np.linspace(0, 2, len(best_h))
                x_new = np.linspace(0, 2, n_points)
                best_h = np.interp(x_new, x_orig, best_h)
        else:
            best_h = np.ones(n_points) * 0.5
            best_h = best_h * (n_points / 2) / np.sum(best_h)
        correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
        best_c5 = np.max(correlation)
        
    # Final normalization and check
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
