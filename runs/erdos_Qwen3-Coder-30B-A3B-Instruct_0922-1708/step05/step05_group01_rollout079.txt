import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturbation_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    lr = initial_learning_rate
    patience_counter = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        max_value = corr[max_lag]
        
        # Compute gradient for the maximum lag
        # The gradient is computed for the convolution at that lag
        # For convolution (h * (1-h))[k] = sum h[i] * (1-h[i+k]) for valid i
        # Gradient w.r.t. h[j] is -1 if j = i and j+k is valid, and 1 if j+k = i and i is valid
        # The contribution to correlation max_lag is from:
        # h[i] * (1 - h[i + max_lag - n_points + 1]) for i = 0, ..., n_points-1
        # but only when both indices are valid
        
        gradient = np.zeros(n_points)
        # This is the term proportional to d(h_i) * (1 - h_{i + k})
        # Its gradient wrt h_i = - (1 - h[i + k]) = h[i + k] - 1
        # and wrt h_{i + k} = - h[i]
        
        # Identify nonzero contributions in the overlap
        n_corr = 2 * n_points - 1
        lag_offset = max_lag - (n_points - 1)
        
        for i in range(n_points):
            # Check if we use positive and negative term in the overlap
            j = i + lag_offset
            if 0 <= j < n_points:
                # This i-th element contributes to the max lag correlation
                # its contribution to the correlation is by definition h[i]* (1 - h[j]) 
                # gradient wrt h[i] = (1 - h[j])
                # gradient wrt h[j] = -h[i]
                gradient[i] += 1 - h[j]
                gradient[j] -= h[i]
        
        # Gradient needs scaling for dx
        gradient *= 2 * dx / n_corr
        
        # Gradient descent step
        h_new = h - lr * gradient
        
        # Repair phase
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept if improvement
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            patience_counter = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            patience_counter += 1
            if patience_counter >= plateau_patience:
                lr *= learning_rate_decay
                patience_counter = 0
                if lr < min_learning_rate:
                    # Perturb if we are stuck
                    h += np.random.normal(0, perturbation_fraction, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        
        # Allow learning rate to decay
        if iteration % 50 == 0:
            if lr > min_learning_rate:
                lr *= learning_rate_decay
        
    return (best_h.tolist(), best_c5, n_points)
