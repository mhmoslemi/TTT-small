import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
    else:
        n_points = 100
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5
    
    # Ensure the sum constraint (exact for initial case)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-resolution optimization
    coarse_n = 50
    dx_coarse = 2.0 / coarse_n
    h_coarse = np.interp(np.linspace(0, 2, coarse_n), np.linspace(0, 2, n_points), h)
    h_coarse = h_coarse * (coarse_n / 2) / np.sum(h_coarse)
    
    # Coarse grid optimization
    # Use Adam-like optimization with adaptive learning
    h_c = h_coarse.copy()
    m = np.zeros_like(h_c)
    v = np.zeros_like(h_c)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute gradient using finite differences
        grad = np.zeros_like(h_c)
        eps_f = 1e-4
        for j in range(len(h_c)):
            h_plus = h_c.copy()
            h_plus[j] += eps_f
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (coarse_n / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_coarse
            c5_plus = np.max(corr_plus)
            
            h_minus = h_c.copy()
            h_minus[j] -= eps_f
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (coarse_n / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx_coarse
            c5_minus = np.max(corr_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps_f)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h_c = h_c - lr * m_hat / (np.sqrt(v_hat) + eps)
        h_c = np.clip(h_c, 0, 1)
        h_c = h_c * (coarse_n / 2) / np.sum(h_c)
        
        # Evaluate solution
        corr = np.correlate(h_c, 1 - h_c, mode='full') * dx_coarse
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h_coarse = h_c.copy()
            
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, coarse_n), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization
    h = h_fine.copy()
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
        # Compute gradient
        grad = np.zeros_like(h)
        eps_f = 1e-4
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps_f
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(corr_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps_f
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx
            c5_minus = np.max(corr_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps_f)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate solution
        corr = np.correlate(h, 1 - h, mode='full') * dx
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final check
    final_corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    
    # Adjust to exact sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
