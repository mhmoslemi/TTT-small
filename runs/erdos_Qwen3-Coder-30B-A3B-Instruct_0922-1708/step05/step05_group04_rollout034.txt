from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    mu_initial = 0.5
    sigma_initial = 0.1

    population = []
    
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Adjust to meet constraint
        factor = (n_points / 2) / h_sum
        h = h * factor
        return h
    
    # Fitness evaluation
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(correlation) * dx
        return fitness
    
    # Main Evolution Strategy loop
    current_mu = mu_initial
    current_sigma = sigma_initial
    best_fitness = float('inf')
    best_individual = None
    iter_without_improvement = 0
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Evaluate all individuals
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Find best individual so far
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
            iter_without_improvement = 0
        else:
            iter_without_improvement += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_indices = sorted_indices[:population_size // 2]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx]
            
            # Gaussian mutation
            noise = np.random.normal(0, current_sigma, n_points)
            child = parent + current_mu * noise
            child = repair(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[population_size // 2:]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Update strategy parameters
        current_mu *= 0.999
        current_sigma = current_mu / 5.0
        
        # Decay mutation strength periodically
        if current_iter % 100 == 0:
            current_mu = mu_initial * 0.9
        
        # Restart if no improvement in 200 iterations
        if iter_without_improvement > 200:
            current_mu = mu_initial
            for i in range(len(population)):
                population[i] = repair(population[i] + np.random.normal(0, 0.1, n_points))
            iter_without_improvement = 0
        
        current_iter += 1
    
    # Final refinement with gradient-based method
    if best_individual is not None:
        h_opt = best_individual.copy()
        for _ in range(100):
            # Compute gradient via finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            base_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            base_fitness = np.max(base_corr) * dx
            improved = False
            for i in range(n_points):
                h_plus = h_opt.copy()

                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - base_fitness) / eps
            
            # Update and repair
            h_opt -= 0.001 * grad
            h_opt = repair(h_opt)
            
            corr_new = np.correlate(h_opt, 1 - h_opt, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= base_fitness:
                break
            else:
                base_fitness = fitness_new
                improved = True
        
        # Final verification
        final_corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        final_c5 = np.max(final_corr) * dx
        
        if final_c5 < best_fitness:
            best_fitness = final_c5
            best_individual = h_opt.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
