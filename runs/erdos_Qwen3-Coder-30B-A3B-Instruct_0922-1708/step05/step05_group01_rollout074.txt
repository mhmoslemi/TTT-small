import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_factor = 1.2
    penalty_min = 1e-4
    penalty_initial = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize penalty
    penalty_weight = penalty_initial
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Track plateau
    last_improvement = 0
    current_lr = learning_rate
    last_c5 = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute constraint violations
        violation_sum = np.sum(h) - n_points / 2
        violation_bounds = np.sum(np.maximum(0, -h)) + np.sum(np.maximum(0, h - 1))
        
        # Compute penalty
        penalty = penalty_weight * (violation_sum**2 + violation_bounds)
        
        # Augmented objective
        augmented_c5 = c5 + penalty
        
        # Compute gradient
        max_idx = np.argmax(corr)
        lag = max_idx - (n_points - 1)
        gradient = np.zeros(n_points)
        
        # Compute gradient for the identified maximum lag
        if lag >= 0:
            for i in range(n_points - lag):
                gradient[i] += (1 - h[i + lag])
        else:
            for i in range(-lag, n_points):
                gradient[i] += (1 - h[i + lag])
        
        # Add penalty gradient
        penalty_grad = np.zeros(n_points)
        penalty_grad += 2 * penalty_weight * h  # grad penalty w.r.t. h
        penalty_grad -= 2 * penalty_weight * violation_sum  # grad penalty w.r.t. sum
        
        # Clip penalties at bounds
        penalty_grad += 2 * penalty_weight * np.maximum(0, -h)  # penalty for lower bound
        penalty_grad -= 2 * penalty_weight * np.maximum(0, h - 1)  # penalty for upper bound
        
        # Final gradient and update
        grad = gradient * dx + penalty_grad
        
        # Line search step size
        step_size = current_lr
        h_new = h - step_size * grad
        
        # Feasibility repair
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        
        if abs(h_sum - target_sum) > 1e-10:
            scale = target_sum / h_sum
            h_new = h_new * scale
        
        # Update h
        h = h_new
        
        # Update penalty weight based on constraints
        if violation_sum != 0 or violation_bounds != 0:
            penalty_weight = penalty_weight * penalty_factor
        else:
            penalty_weight = max(penalty_weight / penalty_factor, penalty_min)
        
        # Adaptive learning rate adjustment and perturbation
        if iteration - last_improvement > plateau_patience:
            current_lr *= learning_rate_decay
            # Random perturbation
            perturbation = np.random.uniform(-perturbation_fraction, perturbation_fraction, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            
        # Force re-evaluation for next iteration
        last_c5 = c5
    
    # Final evaluation of best solution
    best_corr = correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(best_corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
