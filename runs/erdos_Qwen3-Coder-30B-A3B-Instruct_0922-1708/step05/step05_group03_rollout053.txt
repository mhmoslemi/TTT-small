import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    selected = []
    for _ in range(len(population)):
        indices = np.random.choice(len(population), tournament_size)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, alpha=0.8):
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(individual, mutation_rate=0.1, std=0.03):
    mask = np.random.rand(len(individual)) < mutation_rate
    if np.any(mask):
        noise = np.random.normal(0, std, len(individual))
        individual += noise * mask
    return individual

def gradient_refinement(h, dx, steps=50, lr=0.001):
    h = h.copy()
    for _ in range(steps):
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - compute_fitness(h, dx)) / eps
        
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2.0) / np.sum(h_parent)
        # Create variations around parent
        for _ in range(population_size // 2):
            h_var = h_parent + np.random.normal(0, 0.05, n_points)
            h_var = project_to_feasible(h_var, n_points)
            population.append(h_var)
        # Add some random valid individuals
        for _ in range(population_size // 2):
            h_rand = np.random.rand(n_points)
            h_rand = project_to_feasible(h_rand, n_points)
            population.append(h_rand)
    else:
        # All random
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for generation in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = [compute_fitness(ind, dx) for ind in population]
        
        # Find best in current population
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and Mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            if np.random.rand() < 0.8:  # crossover rate
                child1, child2 = crossover(parent1, parent2, alpha=0.8)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate=0.1, std=0.03)
            child2 = mutate(child2, mutation_rate=0.1, std=0.03)
            
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Periodic gradient refinement
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for i in top_indices:
                refined = gradient_refinement(population[i], dx, steps=50, lr=0.001)
                refined_fitness = compute_fitness(refined, dx)
                if refined_fitness < fitnesses[i]:
                    population[i] = refined
                    fitnesses[i] = refined_fitness
                    if refined_fitness < best_c5:
                        best_c5 = refined_fitness
                        best_h = refined.copy()
    
    # Final local refinement of the best solution
    if best_h is not None:
        best_h = gradient_refinement(best_h, dx, steps=100, lr=0.0005)
        best_c5 = compute_fitness(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
