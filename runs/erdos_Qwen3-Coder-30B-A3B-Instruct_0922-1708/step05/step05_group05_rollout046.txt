import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    alpha = 0.01
    min_alpha = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    best_h = None
    best_c5 = float('inf')
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    h = h * (n_points / 2) / np.sum(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlations and find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        c5 = max_corr * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient
        # For each h_i, compute gradient contribution from dominant k
        gradient = np.zeros(n_points)
        # Since max_corr is over all lags k, we use k = max_k to approximate
        # In case of full correlation, we have:
        # d C5/d h_i = (1 - h_{i-k}) + (1 - h_{i+k}) for the dominant correlation
        
        # For each position i, compute contribution from the dominant lag
        for i in range(n_points):
            # Look at the two terms in the correlation
            # h_i * (1 - h_{i+k}) and h_{i-k} * (1 - h_i)
            # Since h_{i-k} appears in the summation for correlation at k,
            # and the overall expression we're differentiating on uses 
            # correlation = sum(h[j] * (1-h[j+k])), index calculation
            # should consider i and i+k in circular fashion
            
            # Let's take derivative w.r.t. h_i
            # In the simple model of one particular k, it's
            # d/dh_i [ sum_j h_j * (1-h_{j+k}) ] = (1 - h_{i+k})
            
            # But we are optimizing the max over all k
            # The closest approximation of gradient:
            # the gradient that maximizes objective is approximate by
            # derivative of sum of h_i (1-h_{i+k})
            
            # We take the gradient of the dominant term
            # d/dh_i Sum_j h_j * (1 - h_{j+k}) 
            # When summing over j, let j = i-k => h_{i-k} * (1-h_i)  
            # When summing over j, let j = i => h_i * (1-h_{i+k})
            # So gradient w.r.t. h_i = (1-h_{i+k}) + (1-h_{i-k})
            # modified to respect circular indexing
            
            j_plus_k = (i + max_k - (n_points - 1)) % n_points
            j_minus_k = (i - max_k) % n_points
            gradient[i] = (1 - h[j_plus_k]) + (1 - h[j_minus_k])
            
        # Step update
        step = -alpha * gradient
        new_h = h + step
        
        # Project to [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Ensure sum constraint ∫ h(x) dx = 1 (so sum h = n/2)
        current_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / current_sum
        
        # Update h
        h = new_h.copy()
        
        # Trust region adjustment
        if c5 < best_c5:
            alpha = min(alpha * 1.2, 1.0)
        else:
            alpha = max(alpha * 0.8, min_alpha)
            
        # Simulated Annealing perturbation
        if iteration % perturb_freq == 0:
            # Apply random perturbation with temperature
            for i in range(n_points):
                if np.random.rand() < 0.1:  # Small chance to update
                    perturb = temperature * (np.random.rand() - 0.5) * (1 - h[i]) * h[i]
                    h[i] += perturb
                    h[i] = np.clip(h[i], 0, 1)
                    
            # Renormalize
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
            # Ensure energy is high enough for perturbations
            temperature *= cooling_rate
    
    # Final optimization with local search
    if best_h is None:
        best_h = h.copy()
    
    # Perform a more refined local search
    for _ in range(500):
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        c5 = max_corr * dx
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j_plus_k = (i + max_k - (n_points - 1)) % n_points
            j_minus_k = (i - max_k) % n_points
            gradient[i] = (1 - best_h[j_plus_k]) + (1 - best_h[j_minus_k])
            
        step = -0.001 * gradient
        new_h = best_h + step
        new_h = np.clip(new_h, 0, 1)
        current_sum = np.sum(new_h)
        new_h = new_h * (n_points / 2) / current_sum
        
        # Evaluate new solution
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_max_corr = new_corr[np.argmax(new_corr)]
        new_c5 = new_max_corr * dx
        
        if new_c5 < c5:
            best_h = new_h.copy()
        else:
            break
            
        # Small additional allowance for re-optimization
        if np.random.rand() < 0.01:
            break    
    
    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
