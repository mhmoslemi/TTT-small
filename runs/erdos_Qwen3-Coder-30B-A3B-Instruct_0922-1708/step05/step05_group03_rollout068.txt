import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        
    # Ensure initial h is feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 10.0
    max_iter = 100000
    start_time = time.time()
    
    # SA loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Temperature schedule
        T = T0 / np.log(1 + iteration + 1)
        
        # Create candidate by perturbing h
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C5
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(corr_candidate) * dx
        
        # Accept or reject candidate
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
    
    # Gradient descent refinement of the best solution
    h_refined = best_h.copy()
    lr = 0.001
    eps = 1e-4
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros_like(h_refined)
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        h_refined = h_refined - lr * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
        refined_c5 = np.max(corr_refined) * dx
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_refined.copy()
    
    # Final evaluation 
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
