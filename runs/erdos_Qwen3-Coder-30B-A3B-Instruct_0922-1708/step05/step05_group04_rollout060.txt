from scipy.optimize import minimize
import time

    population_size = 100
    max_iter = 1000
    F = 0.8
    CR = 0.9
    mutation_decay = 0.99

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    fitness = []
    for h in population:
        corr = np.correlate(h, 1 - h, mode='full')
        fit = np.max(corr) * dx
        fitness.append(fit)
    
    # Find best individual
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1]
            for idx in indices:
                if diff <= 0:
                    break
                dec = min(h[idx], diff)
                h[idx] -= dec
                diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)
            for idx in indices:
                if diff >= 0:
                    break
                inc = min(1 - h[idx], -diff)
                h[idx] += inc
                diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Main DE loop
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair mutant
            mutant = repair(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            trial = repair(trial)
            
            # Evaluate trial
            corr = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(corr) * dx
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                new_fitness.append(trial_fitness)
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_h = trial.copy()
            else:
                new_population.append(population[i])
                new_fitness.append(fitness[i])
        
        # Update population
        population = new_population
        fitness = new_fitness
        
        # Adapt mutation factor
        F *= mutation_decay
        
        iteration += 1
    
    # Final gradient-based refinement
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        base_fitness = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            fitness_plus = objective(h_plus)
            grad[i] = (fitness_plus - base_fitness) / eps
        return grad
    
    # Gradient descent refinement
    h_ref = best_h.copy()
    for _ in range(100):
        if time.time() - start_time >= budget_s - 10:
            break
        grad = gradient(h_ref)
        h_ref -= 0.001 * grad
        h_ref = repair(h_ref)
        
        corr_ref = np.correlate(h_ref, 1 - h_ref, mode='full')
        new_fitness = np.max(corr_ref) * dx
        if new_fitness >= objective(best_h):
            break
        else:
            best_h = h_ref.copy()
            best_fitness = new_fitness
    
    return (best_h.tolist(), best_fitness, n_points)
