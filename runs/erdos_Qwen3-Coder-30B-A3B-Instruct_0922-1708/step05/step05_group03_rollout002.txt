import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute C5 value for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover with parameter alpha"""
    return alpha * parent1 + (1 - alpha) * parent2

def mutate(h, mutation_rate=0.1, sigma=0.03):
    """Apply Gaussian mutation"""
    mask = np.random.random(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + mask * noise
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 1000
    crossover_rate = 0.8
    mutation_rate = 0.1
    refinement_interval = 50
    
    start_time = time.time()
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as basis
        parent = np.array(initial_h_values)
        parent = parent * (n_points / 2.0) / np.sum(parent)
        
        # Create variations around parent
        for _ in range(population_size // 2):
            h = parent + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
            
        # Add some random individuals
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Initialize entirely with random individuals
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Selection, crossover, mutation
        new_population = []
        
        for _ in range(population_size):
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_rate:
                child = crossover(parent1, parent2)
            else:
                child = parent1.copy()
                
            # Mutation
            child = mutate(child, mutation_rate, 0.03)
            
            # Project to feasible region
            child = project_to_feasible(child, n_points)
            
            new_population.append(child)
        
        # Evaluate fitness of new population
        population = new_population
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Periodic refinement
        if gen % refinement_interval == 0 and gen > 0:
            # Gradient descent on top individuals
            top_indices = np.argsort(fitnesses)[:population_size//10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                
                # Simple gradient descent step
                for _ in range(20):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        
                        c5_plus = compute_fitness(h_plus, dx)
                        grad[j] = (c5_plus - fitnesses[idx]) / eps
                    
                    h_ref = h_ref - 0.001 * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    c5_new = compute_fitness(h_ref, dx)
                    if c5_new < fitnesses[idx]:
                        fitnesses[idx] = c5_new
                        population[idx] = h_ref.copy()
                        
                # Update best if improved
                if fitnesses[idx] < best_c5:
                    best_c5 = fitnesses[idx]
                    best_h = population[idx].copy()
    
    # Final refinement pass
    refined_candidates = []
    
    # Refine best individual from population
    top_indices = np.argsort(fitnesses)[:5]
    for idx in top_indices:
        h_ref = population[idx].copy()
        
        for _ in range(50):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                
                c5_plus = compute_fitness(h_plus, dx)
                grad[j] = (c5_plus - fitnesses[idx]) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            c5_new = compute_fitness(h_ref, dx)
            if c5_new < fitnesses[idx]:
                fitnesses[idx] = c5_new
                population[idx] = h_ref.copy()
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_ref.copy()
    
    # Final validation
    final_c5 = compute_fitness(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
