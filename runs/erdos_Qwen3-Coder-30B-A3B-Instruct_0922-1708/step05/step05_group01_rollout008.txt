import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    perturb_fraction = 0.01
    
    # Initialize h from initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Ensure exact sum
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h):
        # Compute all correlations for overlap integral
        conv = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(conv)
        return max_corr * dx
    
    def compute_gradient(h, k_star):
        # Compute gradient for the maximum correlation lag k_star
        # This is a simplified gradient approximated using finite differences
        n = len(h)
        grad = np.zeros(n)
        
        # Compute the difference in correlation when 1 unit is added to h[i]
        eps = 1e-5
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr = np.correlate(h, 1 - h, mode='full')
            
            grad[i] = (np.max(corr_plus) - np.max(corr)) / eps
        
        return grad
    
    def project_h(h):
        # Clip to [0,1] and rescale to sum = n/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        return h
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute all correlations and find the maximum
        conv = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(conv)
        k_star = np.argmax(conv) - (n_points - 1)
        
        # Compute gradient
        grad = compute_gradient(h, k_star)
        
        # Adaptive line search
        current_c5 = compute_c5(h)
        step_size = alpha
        improved = False
        
        while step_size > 1e-10:
            h_new = h - step_size * grad
            h_new = project_h(h_new)
            new_c5 = compute_c5(h_new)
            if new_c5 < current_c5:
                h = h_new
                current_c5 = new_c5
                improved = True
                break
            step_size *= line_search_factor
        
        # If no improvement, reduce learning rate
        if not improved:
            alpha *= 0.9
        
        # Check for improvement
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Small random perturbation for diversification
        if iteration % 100 == 0:
            perturb_indices = np.random.choice(n_points, size=int(perturb_fraction * n_points), replace=False)
            for idx in perturb_indices:
                h[idx] += np.random.normal(0, 0.001)
            h = project_h(h)
    
    return (best_h.tolist(), best_c5, n_points)
