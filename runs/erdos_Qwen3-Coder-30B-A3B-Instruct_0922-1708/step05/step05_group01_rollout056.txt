import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization
    alpha = 1e-3
    max_iter = 5000
    grad_thresh = 1e-6
    line_search_factor = 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute correlations for all lags
        correlations = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlations)
        c5 = correlations[k_max] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient for the maximizing lag
        # k_max corresponds to lag k = k_max - (n_points - 1)
        k = k_max - (n_points - 1)
        
        # Create shifted version of h
        if k >= 0:
            h_shifted = h[k:]
            h_orig = h[:len(h_shifted)]
        else:
            h_shifted = h[:k]
            h_orig = h[-k:]
        
        # Compute gradient components
        grad = np.zeros(n_points)
        if k >= 0:
            grad[k:k+len(h_shifted)] = (1 - h_shifted) - h_orig
        else:
            grad[:-k] = (1 - h_shifted) - h_orig
        
        # Propagate gradient back to full h
        if k >= 0:
            # Shifted part contributes to indices k:k+len(h_shifted)
            grad_full = np.zeros(n_points)
            grad_full[k:k+len(h_shifted)] = (1 - h_shifted) - h_orig
        else:
            # Shifted part contributes to indices -k:n_points
            grad_full = np.zeros(n_points)
            grad_full[:-k] = (1 - h_shifted) - h_orig
        
        # Evaluate gradient norm
        grad_norm = np.linalg.norm(grad_full)
        if grad_norm < grad_thresh:
            break
        
        # Line search
        while True:
            h_new = h - alpha * grad_full
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = corr_new[np.argmax(corr_new)] * dx
            
            if c5_new <= c5:
                h = h_new.copy()
                break
            else:
                alpha *= line_search_factor
                if alpha < 1e-10:
                    break
        
        # Adaptive step size
        if c5_new <= c5:
            alpha *= 1.2
        else:
            alpha *= 0.5
    
    # Final check
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = corr_final[np.argmax(corr_final)] * dx
    
    return (best_h.tolist(), final_c5, n_points)
