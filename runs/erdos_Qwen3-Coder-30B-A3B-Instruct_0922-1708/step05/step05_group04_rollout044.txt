import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Correlation computation
    def compute_correlation(h_vals):
        h_vals = np.array(h_vals)
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    ### Binary search on the bound
    def feasible(h_vals, mu):
        h_vals = np.array(h_vals)
        # Sample points for the constraint
        k_samples = np.linspace(-2, 2, 50)
        # For each k, compute the convolution implicitly using overlaps
        # Here we compute ∫h(x)(1-h(x+k))dx for various k using discrete approximation
        n = len(h_vals)
        total = 0
        count = 0
        for k in k_samples:
            k_idx = int((k + 2) / (4.0 / (n-1)))  # Convert continuous k to discrete index
            k_idx = max(0, min(k_idx, n-1))
            
            # Compute convolution (dot product with shifted sequence)
            if k_idx < n:
                # h(x)(1-h(x+k)) = h(x) - h(x)*h(x+k)
                # For now, we compute the integral ∫h(x)*(1-h(x+k))dx directly
                t = 0
                for i in range(n):
                    j = i + k_idx
                    if 0 <= j < n:
                        t += h_vals[i] * (1 - h_vals[j])
                total += t * dx
                count += 1
                
        if count > 0:
            avg_overlap = total / count
            return avg_overlap >= mu
        else:
            return True  # Should never happen
    
    # Modified version using cvxpy
    def solve_convex_optimization(h_guess, mu):
        # This is a simplified version with assumed convexity
        h_var = Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            sum(h_var) == n_points / 2
        ]
        
        # Add constraint same as in original way (simplified approach) 
        # k samples
        k_list = np.linspace(-2, 2, 30)
        for k in k_list:
            shifted_h = np.roll(h_guess, int(-k / dx))
            # Need differently supported version of the constraint
            # Just do it by hand:

        # We directly try to solve it:
        obj = Minimize(0)  # Dummy objective - just feasibility check
        prob = Problem(obj, constraints)
        
        try:
            prob.solve(solver=OSQP, verbose=False)
            if prob.status == 'optimal':
                return h_var.value
        except:
            pass
        return h_guess
    
    def refine(h_vals):
        h_vals = np.array(h_vals)
        
        def grad_func(x):
            # Approximate gradient using finite differences
            eps = 1e-4
            grad = np.zeros_like(x)
            base_val = compute_correlation(x)
            for i in range(len(x)):
                x_plus = x.copy()
                x_plus[i] += eps
                x_plus = np.clip(x_plus, 0, 1)
                x_plus = x_plus * (n_points / 2) / np.sum(x_plus)
                grad[i] = (compute_correlation(x_plus) - base_val) / eps
            return grad
        
        # Gradient descent
        h_opt = h_vals.copy()
        lr = 0.001
        for _ in range(200):
            if time.time() - start_time > budget_s - 10:
                break
            grad = grad_func(h_opt)
            h_opt = h_opt - lr * grad
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
        return h_opt
    
    best_h = h.copy()
    best_C5 = compute_correlation(h)
    
    # Start of local search 
    for iter in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
        
        improved = False
        # Try different search directions:
        # Random neighborhood search
        for _ in range(10):
            h_neigh = h.copy() + np.random.normal(0, 0.01, n_points) 
            h_neigh = np.clip(h_neigh, 0, 1)
            h_neigh = h_neigh * (n_points / 2) / np.sum(h_neigh)
            
            c5 = compute_correlation(h_neigh)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h_neigh.copy()
                improved = True

        # Gradient-based refinement 
        if improved:
            h = refine(best_h)
            c5 = compute_correlation(h)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h.copy()
                
        else:
            # Random restart if no improvement for a while
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            c5 = compute_correlation(h)
            if c5 < best_C5:
                best_C5 = c5
                best_h = h.copy()

    # Final refinement via gradient descent
    best_h = refine(best_h)
    best_C5 = compute_correlation(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
