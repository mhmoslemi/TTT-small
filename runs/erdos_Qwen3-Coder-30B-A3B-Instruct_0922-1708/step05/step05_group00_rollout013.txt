import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize the starting solutions
    initial_solutions = []
    
    # Use provided parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    else:
        # If no parent solution, start with uniform
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    
    # Add several random restarts
    for _ in range(4):
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # For each initial solution, perform refinement
    for i, h in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Convex relaxation approach using quadratic programming
        try:
            h_var = cp.Variable(n_points)
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # We're optimizing a quadratic form, which isn't directly solvable in CVXPY
            # Instead, we'll treat it as a feasibility problem with relaxed constraints
            # and then use local optimization
            prob = cp.Problem(cp.Minimize(0), constraints)
            try:
                prob.solve(solver=ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL]:
                    h_init = h_var.value
                    if h_init is not None:
                        h = np.clip(np.array(h_init), 0, 1)
                        h = h * (n_points / 2) / np.sum(h)
            except:
                pass
        except:
            pass
            
        # Refine using local optimization
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(corr)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Optimize using L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        try:
            result = minimize(
                objective, 
                h, 
                method='L-BFGS-B', 
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 2000, 'ftol': 1e-6},
                callback=lambda x: None,
                tol=1e-6
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_val = objective(h_opt)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_opt.copy()
        except:
            pass
    
    # Final check
    if best_h is None:
        # Fallback to initial best
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            h = np.ones(n_points) * 0.5
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        best_h = h.copy()
        corr = np.correlate(best_h, 1 - best_h, mode='full') * dx
        best_c5 = np.max(corr)
    
    # Ensure sum constraint
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
