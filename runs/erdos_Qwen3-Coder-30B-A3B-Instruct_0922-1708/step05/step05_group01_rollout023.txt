import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    learning_rate = 1e-3
    penalty_param = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Gradient descent loop
    iteration = 0
    no_improvement_count = 0
    last_improvement = 0
    
    while (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Check for stagnation
        if no_improvement_count > plateau_patience:
            # Reduce learning rate
            learning_rate *= learning_rate_decay
            # Apply perturbations
            perturbation_mask = np.random.rand(n_points) < perturbation_fraction
            h[perturbation_mask] += np.random.normal(0, 0.01, np.sum(perturbation_mask))
            # Repair h
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            continue
            
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        k_max = np.argmax(corr)
        corr_max = np.max(corr)
        
        # Reconstruct the actual maximizing correlation index
        # Full correlation corresponds to indices [0, 2*n_points-2]
        # The actual lag k in [-n+1, n-1] is k = i - (n-1) where i is full correlation index
        # For k_max = i, k = i - (n-1)
        actual_k = k_max - (n_points - 1)
        
        # Choosing correct lags for gradient computation
        if actual_k >= 0:
            # lag is positive, h[i] * (1 - h[i + actual_k])
            # But correlation is computed as sum over i for index j = i + actual_k, so the
            # gradient w.r.t. h[i] is computed being interaction with (1 - h[i + actual_k])
            # yet we're interested in h[i] term with the actual_k shift. 
            for i in range(n_points):
                if i + actual_k < n_points:
                    grad[i] += 1 - h[i + actual_k]
                if i - actual_k >= 0:
                    grad[i] += 1 - h[i - actual_k]
        else:
            # lag is negative, we compute sum over j for h[j + (-actual_k)]
            for i in range(n_points):
                if i - actual_k < n_points:
                    grad[i] += 1 - h[i - actual_k]
                if i + actual_k >= 0:
                    grad[i] += 1 - h[i + actual_k]
        grad = 2 * grad * dx / n_points
        
        # Compute penalty gradient
        penalty_grad = np.zeros(n_points)
        constraint_violation_sum = np.sum(h) - n_points / 2
        penalty_grad += penalty_param * 2 * constraint_violation_sum
        
        # Clip gradient
        penalty_grad = np.clip(penalty_grad, -100, 100)
        
        # Apply gradient
        h = h - learning_rate * (grad + penalty_grad)
        
        # Repair constraints
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-8:
            # Adjust values to rescale to exact sum
            diff = n_points / 2 - current_sum
            indices = np.argsort(h)
            # If diff > 0, we increase smaller values
            # If diff < 0, we decrease larger values
            for i in indices:
                if abs(diff) < 1e-10:
                    break
                if diff > 0:
                    increase = min(diff, 1 - h[i])
                    h[i] += increase
                    diff -= increase
                else:
                    decrease = min(-diff, h[i])
                    h[i] -= decrease
                    diff += decrease
        
        # Adaptive penalty update
        sum_violation = np.sum(h) - n_points / 2
        sum_violation = max(0, abs(sum_violation))
        if sum_violation > 1e-5:
            penalty_param *= penalty_update_factor
        else:
            penalty_param = max(penalty_param / penalty_update_factor, penalty_min)
    
    # Final verification and return
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
