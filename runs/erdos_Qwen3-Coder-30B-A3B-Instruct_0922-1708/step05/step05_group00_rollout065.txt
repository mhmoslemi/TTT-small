import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

    dx = 2.0 / n_points
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Define smooth max for gradient computation
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

    # Gradient-based optimization with Adam
    def adam_update(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        return h_new, m, v

    # Optimizer that matches problem structure
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    max_patience = 500
    
    # Strategies:
    # 1. Uniform random + gradient ascent for exploring solutions  
    # 2. Projected gradient descent to handle constraints
    # 3. Use of smooth max for gradient computation
    
    # Initial mock optimization for tuning
    # Use a very light exploration with large steps initially
    
    # Take a better approach - discrete convolution-based gradient descent
    h = best_h.copy()
    t = 1
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # For parameters, we adjust a bit  
    lr = 0.005
    beta = 15.0  # Smoothness of max approximation
    
    # Adam optimization main loop
    while time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        current_smooth_c5 = smooth_max(correlation, beta)

        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            smooth_plus = smooth_max(cor_plus, beta)
            grad[i] = (smooth_plus - current_smooth_c5) / eps

        # Update with Adam
        h_new, m, v = adam_update(h, grad, m, v, t, lr)
        t += 1
        
        # Project to feasible region (constraint: bounds + sum)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate
        lr = max(1e-6, lr * 0.9995)

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Ensure constraint satisfaction
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
