import numpy as np
from scipy.optimize import minimize
import time

import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent or use structured initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        if n_points < 100:
            # Upsample to required size
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 100)
            h = np.interp(x_new, x_orig, h)
            n_points = 100
    else:
        # Create structured low-discrepancy initialization
        n_points = 100
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(3 * np.pi * x))  # Smooth variation
        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Smooth max approximation parameter
    beta = 10.0
    
    def smooth_max(arr, beta):
        """Numerically stable smooth approximation to max"""
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def evaluate_c5(h_vec):
        """Evaluate current bound"""
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
        return np.max(correlation)
    
    def evaluate_c5_smooth(h_vec, beta):
        """Evaluate smooth approximation"""
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx  
        return smooth_max(correlation, beta)
    
    def project_to_feasible(h_vec):
        """Project onto feasible region"""
        h_proj = np.clip(h_vec, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    def compute_gradient(h_vec, eps=1e-5):
        """Finite difference gradient approximation"""
        grad = np.zeros_like(h_vec)
        curr_val = evaluate_c5_smooth(h_vec, beta)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            grad[i] = (evaluate_c5_smooth(h_plus, beta) - curr_val) / eps
        return grad
    
    # Main optimization loop
    max_iter = 5000
    lr = 0.01
    patience = 0
    max_patience = 1000
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_c5 = evaluate_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Compute gradient using finite differences
        grad = compute_gradient(best_h)
        
        # Update via projected gradient descent
        h_new = best_h - lr * grad
        h_new = project_to_feasible(h_new)
        
        # Adaptive learning rate
        lr = max(1e-6, lr * 0.9995)
        
        # Check convergence via patience
        if abs(current_c5 - best_c5) < 1e-7:
            patience += 1
            if patience > max_patience:
                break
        else:
            patience = 0
            
        best_h = h_new
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = best_h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
