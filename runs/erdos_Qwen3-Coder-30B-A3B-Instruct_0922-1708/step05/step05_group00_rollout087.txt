import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start approach with convex relaxation
    best_h = None
    best_c5 = float('inf')
    
    # Generate initial solutions
    initial_solutions = []
    
    # 1. Use parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    # 2. Random starting points
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    # Process each initialization
    for idx, h in enumerate(initial_solutions):
        # Convex relaxation - approximate by treating h[i]*(1-h[i+k]) 
        # as bounded by convex combinations
        
        # The fundamental insight: we want to minimize
        # max_k integral h(x)*(1-h(x+k)) dx
        # This is difficult due to the non-linearity and the max over k
        
        # Try first optimizing via a linear relaxation
        # i.e. convex programming on (h * (1-h_blocked)) 
        # is not straightforward, but we attempt a stayed  approach:
        
        # Use L-BFGS-B to refine the candidate
        def objective(h_vec):
            h = h_vec.copy()
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            corr = correlate(h, 1 - h, mode='full') * dx
            return np.max(corr)
        
        def gradient(h_vec):
            # Approximate using finite differences
            eps = 1e-6
            grad = np.zeros_like(h_vec)
            base_c5 = objective(h_vec)
            for i in range(len(h_vec)):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                grad[i] = (objective(h_plus) - base_c5) / eps
            return grad
        
        # Optimization with bounds
        bounds = [(0, 1) for _ in range(n_points)]
        
        # L-BFGS-B is best for bounds and smooth objectives
        try:
            res = minimize(
                objective, 
                h, 
                method='L-BFGS-B', 
                bounds=bounds,
                jac=gradient,
                options={'maxiter': 2000, 'ftol': 1e-6, 'gtol': 1e-6}
            )
            
            if res.success:
                h_opt = res.x.copy()
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5 = objective(h_opt)
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_opt.copy()
                    
        except Exception:
            continue
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final validation - ensure it's good
    if best_h is None:
        # Fallback to uniform distribution
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        h_fallback = np.clip(h_fallback, 0, 1)
        corr = correlate(h_fallback, 1 - h_fallback, mode='full') * dx
        best_c5 = np.max(corr)
        best_h = h_fallback
    
    # Ensure that sum constraint is met
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Final overlap check
    corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), final_c5, n_points)
