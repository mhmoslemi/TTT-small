import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use scipy.optimize.minimize with method='L-BFGS-B' for better performance
    n_start = 50
    n_points = 200  # Higher resolution for final refinement
    dx = 2.0 / n_points
    
    # Initialize based on provided parent or uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        # Downsample to coarse grid for initial optimization
        if len(h_init) > n_start:
            x_orig = np.linspace(0, 2, len(h_init))
            x_coarse = np.linspace(0, 2, n_start)
            f = interp1d(x_orig, h_init, kind='linear', fill_value='extrapolate')
            h = f(x_coarse)
        else:
            h = h_init
    else:
        h = np.ones(n_start) * 0.5
        
    # Ensure valid initial values
    h = np.clip(h, 0, 1)
    h = h * (n_start / 2) / np.sum(h)
    
    def objective(h_vec):
        # Reshape h_vec
        h = h_vec.copy()
        h = h * (n_points / 2) / np.sum(h)  # This aligns with full resolution later
        h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h)), h)
        # Compute correlation of h with 1-h
        correlation = np.correlate(h_full, 1 - h_full, mode='full') * dx
        max_c5 = np.max(correlation)
        return max_c5
    
    def constraint_sum(h_vec):
        # Normalize to satisfy sum constraint
        h_normalized = h_vec * (n_points / 2) / np.sum(h_vec)
        return np.sum(h_normalized) - n_points / 2
    
    def project_h(h_vec):
        # Clip h to [0,1] and ensure sum constraint
        h_proj = np.clip(h_vec, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    # First, coarse optimization
    best_c5 = float('inf')
    best_h = None
    patience = 0
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        h_base = h.copy()
        
        for j in range(len(h)):
            h_plus = h_base.copy()
            h_plus[j] += eps
            c5_plus = objective(h_plus)
            grad[j] = (c5_plus - objective(h_base)) / eps
        
        # Adam optimization step
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        h = h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project back to feasible set
        h = project_h(h)
        
        # Evaluate solution
        c5 = objective(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 200:
            break
    
    # Second phase: refine the solution with higher resolution
    if best_h is not None:
        fine_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
        fine_h = np.clip(fine_h, 0, 1)
    else:
        fine_h = np.ones(n_points) * 0.5
        fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
        fine_h = np.clip(fine_h, 0, 1)
        
    # Refine the fine grid with Adam
    best_fine_h = fine_h.copy()
    best_fine_c5 = float('inf')
    patience = 0
    
    m = np.zeros_like(fine_h)
    v = np.zeros_like(fine_h)
    lr = 0.01
    
    for i in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        eps = 1e-4
        grad = np.zeros_like(fine_h)
        
        for j in range(len(fine_h)):
            h_plus = fine_h.copy()
            h_plus[j] += eps
            c5_plus = objective(h_plus)
            grad[j] = (c5_plus - objective(fine_h)) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        fine_h = fine_h - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        
        # Project
        fine_h = project_h(fine_h)
        
        # Evaluate
        c5 = objective(fine_h)
        if c5 < best_fine_c5:
            best_fine_c5 = c5
            best_fine_h = fine_h.copy()
            patience = 0
        else:
            patience += 1
        if patience > 200:
            break
    
    # Final check
    final_h = best_fine_h.copy()
    final_c5 = best_fine_c5
    
    # Ensure sum is correct
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
