import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Optimization parameters
    max_iter = 10000
    learning_rate = 0.001
    perturb_rate = 0.1
    stagnation_threshold = 100
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update h using gradient
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Periodic perturbation to escape local minima
        if iteration % 50 == 0:
            num_perturb = max(1, int(n_points * perturb_rate))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in indices:
                h[i] = np.random.rand()
            h = repair_h(h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_h(h, dx)
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if no progress
        if stagnation_count > stagnation_threshold:
            break
    
    # Final refinement
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(best_h, dx)
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        new_c5 = evaluate_h(best_h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
