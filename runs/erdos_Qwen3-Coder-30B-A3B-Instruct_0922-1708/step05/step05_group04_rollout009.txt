import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Add initial solution with some perturbation
        population.append(h_init.copy())
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Initialize with random feasible solutions
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Evaluation function
    def evaluate(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Repair function
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Mutation function
    def mutate(parent, mu, sigma):
        child = parent + mu * np.random.normal(0, sigma, n_points)
        return repair(child)
    
    best_h = None
    best_fitness = float('inf')
    stagnation_count = 0
    mu = 0.5
    sigma = mu / 5.0
    
    # Main ES loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all population members
        fitnesses = [evaluate(ind) for ind in population]
        
        # Update best
        for i, fit in enumerate(fitnesses):
            if fit < best_fitness:
                best_fitness = fit
                best_h = population[i].copy()
                stagnation_count = 0
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        top_indices = sorted_indices[:population_size // 2]
        
        # Create offspring
        offspring = []
        
        # Keep elite
        for idx in top_indices:
            offspring.append(population[idx].copy())
        
        # Generate new offspring
        while len(offspring) < offspring_size:
            parent = population[np.random.choice(top_indices)]
            child = mutate(parent, mu, sigma)
            offspring.append(child)
        
        # Replace worst candidates with best offspring
        # Keep top 50% from original population and top 50% from offspring
        combined = [population[i] for i in sorted_indices[:population_size // 2]] + offspring
        combined_fitness = [evaluate(ind) for ind in combined]
        sorted_combined = np.argsort(combined_fitness)
        
        # Replace population with best half
        new_population = [combined[i] for i in sorted_combined[:population_size]]
        population = new_population
        
        # Update mutation parameters
        mu *= 0.999
        sigma = mu / 5.0
        
        # Occasionally reset mutation strength
        if iteration % 100 == 0:
            mu = 0.5 * 0.9 ** (iteration // 100)
            # Perturb population to avoid local minima
            for i in range(population_size):
                if np.random.rand() < 0.1:
                    population[i] = mutate(population[i], mu * 0.5, sigma * 0.5)
        
        # Stagnation check
        stagnation_count += 1
        if stagnation_count > 200:
            # Restart strategy with higher mutation
            mu = 0.7
            for i in range(population_size):
                population[i] = repair(np.random.uniform(0, 1, n_points))
            stagnation_count = 0
    
    # Final refinement with gradient descent on best solution
    if best_h is not None:
        h_opt = best_h.copy()
        # Simple gradient-based refinement
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            fitness = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_opt -= 0.001 * grad
            h_opt = repair(h_opt)
            
            # Check if improvement was made
            corr_new = np.correlate(h_opt, 1 - h_opt, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        
        # Final evaluation
        corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5_final = np.max(corr_final) * dx
        
        best_h = h_opt.copy()
        best_fitness = c5_final
    
    # Final safety check
    if best_h is None:
        best_h = population[0]
        corr_final = np.correlate(best_h, 1 - best_h, mode='full')
        best_fitness = np.max(corr_final) * dx
    
    return (best_h.tolist(), best_fitness, n_points)
