import numpy as np
import time
from scipy.optimize import minimize
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    beta = 10.0  # Smooth max parameter
    adam_lr = 0.01
    adam_b1 = 0.9
    adam_b2 = 0.999
    epsilon = 1e-8
    max_iter = 5000
    patience = 500
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    t = 0
    
    def smooth_max(arr, beta):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h, beta):
        # Compute the gradient of smooth max overlap  
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx] * dx
        
        # Compute gradient with respect to smooth max
        exp_vals = np.exp(beta * corr * dx)
        sum_exp = np.sum(exp_vals)
        
        # Gradient of smooth max with respect to each element of h
        grad = np.zeros_like(h)
        
        # Overlap computation for all shifts
        # Points where h[i] appears in correlation: for k = 0 to 2n - 2
        # If k is such that h[i] * (1 - h[i+k]) contributes to the max, 
        # then the derivative that matters is for h[i] and (1-h[i+k])
        
        # The complexity of directly computing the gradient makes this infeasible
        # We fall back to finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - compute_c5(h)) / eps
            
        return grad
    
    # Iterations
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
            
        if i - last_improvement > patience:
            break
            
        # Gradient
        grad = compute_gradient(h, beta)
        
        # Adam update
        t += 1
        m = adam_b1 * m + (1 - adam_b1) * grad
        v = adam_b2 * v + (1 - adam_b2) * (grad * grad)
        m_hat = m / (1 - adam_b1**t)
        v_hat = v / (1 - adam_b2**t)
        
        # Update
        h = h - adam_lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final verification
    final_c5 = compute_c5(best_h)
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
