import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 1000
    F = 0.8
    CR = 0.9
    mutation_adjustment_freq = 50
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Compute initial fitness for all population members
    fitnesses = []
    for i in range(population_size):
        correlation = np.correlate(population[i], 1 - population[i], mode='full')
        fitness = np.max(correlation) * dx
        fitnesses.append(fitness)
    
    # Track best solution
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Main DE loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        if generation % mutation_adjustment_freq == 0 and generation > 0:
            F *= 0.99
            
        new_population = []
        new_fitnesses = []
        
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = x1 + F * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to maintain sum constraint
            mutant_sum = np.sum(mutant)
            if abs(mutant_sum - n_points / 2) > 1e-10:
                diff = mutant_sum - n_points / 2
                if diff > 0:
                    # Remove excess
                    indices = np.argsort(mutant)[::-1]
                    for idx in indices:
                        if diff <= 0:
                            break
                        dec = min(mutant[idx], diff)
                        mutant[idx] -= dec
                        diff -= dec
                else:
                    # Add deficit
                    indices = np.argsort(mutant)
                    for idx in indices:
                        if diff >= 0:
                            break
                        inc = min(1 - mutant[idx], -diff)
                        mutant[idx] += inc
                        diff += inc
                # Scale to sum to n_points/2
                mutant = mutant * (n_points / 2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial[j] = mutant[j]
            
            # Evaluate trial
            correlation = np.correlate(trial, 1 - trial, mode='full')
            trial_fitness = np.max(correlation) * dx
            
            # Selection
            if trial_fitness < fitnesses[i]:
                new_population.append(trial)
                new_fitnesses.append(trial_fitness)
            else:
                new_population.append(population[i])
                new_fitnesses.append(fitnesses[i])
            
            # Update global best
            if trial_fitness < best_fitness:
                best_fitness = trial_fitness
                best_h = trial.copy()
        
        population = new_population
        fitnesses = new_fitnesses
        
        if generation % 50 == 0:
            # Progress check
            if time.time() - start_time > budget_s - 50:
                break
    
    # Final refinement with gradient descent
    h_opt = best_h.copy()
    
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def gradient(h):
        eps = 1e-4
        grad = np.zeros_like(h)
        base_fitness = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            grad[i] = (objective(h_plus) - base_fitness) / eps
        return grad
    
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        grad = gradient(h_opt)
        h_opt -= 0.001 * grad
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = np.sum(h_opt)
        if abs(h_opt_sum - n_points / 2) > 1e-10:
            h_opt = h_opt * (n_points / 2) / h_opt_sum
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        new_fitness = np.max(corr) * dx
        if new_fitness >= objective(h_opt):
            break
    
    # Final evaluation
    correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
    final_fitness = np.max(correlation) * dx
    
    return (h_opt.tolist(), final_fitness, n_points)
