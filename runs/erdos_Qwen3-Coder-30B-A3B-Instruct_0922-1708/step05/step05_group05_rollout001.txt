import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Default construction: step function
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        
    # Normalize to sum = n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Helper function to compute correlation
    def compute_overlap(h):
        corr = np.correlate(h, 1 - h, mode="full")
        return np.max(corr) * dx
    
    # Helper function to project h onto feasible region
    def project_h(h):
        # Project onto [0,1]^n
        h = np.clip(h, 0, 1)
        # Project onto sum = n_points/2
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            scale_factor = (n_points / 2) / current_sum
            h = h * scale_factor
        return h
    
    # Gradient approximation via finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode="full")
        base_fitness = np.max(base_corr) * dx
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - base_fitness) / eps
            
        return grad
    
    # Projected gradient descent with adaptive step size
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = compute_overlap(h)
    
    start_time = time.time()
    
    alpha = 0.01
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Try step
        h_candidate = h - alpha * grad
        h_candidate = project_h(h_candidate)
        
        # Evaluate
        candidate_c5 = compute_overlap(h_candidate)
        
        if candidate_c5 < best_c5:
            h = h_candidate.copy()
            best_c5 = candidate_c5
            best_h = h.copy()
        else:
            # Reduce step size
            alpha *= 0.5
            
        # Occasionally apply random perturbation to escape local minima
        if iteration % 50 == 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h_perturbed = h + perturbation
            h_perturbed = project_h(h_perturbed)
            perturbed_c5 = compute_overlap(h_perturbed)
            if perturbed_c5 < best_c5:
                h = h_perturbed.copy()
                best_c5 = perturbed_c5
                best_h = h.copy()
            # Reset alpha for fresh search
            alpha = 0.01
    
    # Final check
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
