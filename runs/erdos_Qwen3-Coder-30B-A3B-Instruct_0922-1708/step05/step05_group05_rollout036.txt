import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to 1/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        min_k = np.argmin(corr)
        min_corr = corr[min_k]
        
        # Compute gradient w.r.t. h for the minimal correlation
        grad = np.zeros_like(h)
        # The minimal correlation between h and (1-h) is at index min_k
        # This is the result of: h[i] * (1 - h[i+min_k]) 
        # For that specific k, we compute gradient
        # h is a step function, so we need to consider k shifts
        for i in range(n_points):
            k = min_k - n_points + 1  # Corresponds to offset in correlation
            if 0 <= i - k < n_points:
                grad[i] += 1.0
            if 0 <= i + k < n_points:
                grad[i] += 1.0
        
        # Apply gradient step (we want to minimize C5, so we maximize the minimal correlation)
        # But C5 is proportional to the max of correlation with 1-h
        # So we actually want to maximize that. Since the max is over k, we consider the minimal correlation
        # to be the most critical
        h_new = h + learning_rate * grad
        
        # Project back to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to keep sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            
        # Replace with new h
        h = h_new.copy()
        
        # Occasionally apply a random perturbation to escape local minima
        if iteration % perturbation_freq == 0:
            # Add controlled random noise
            noise_magnitude = perturbation_magnitude / (1.0 + iteration / 100.0)
            noise = np.random.normal(0, noise_magnitude, n_points)
            h = h + noise
            
            # Project again
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.99
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    """Computes the C5 value given h."""
    corr = correlate(h, 1 - h, mode='full') * dx
    return np.max(corr)
