import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Adjust smallest elements to correct the sum
        if current_sum < target_sum:
            indices = np.argsort(h)
            diff = target_sum - current_sum
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            diff = current_sum - target_sum
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] - diff / (len(indices) - i), 0.0)
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    # Initial evaluation
    best_c5 = evaluate_h(best_h, dx)
    best_h = repair_h(best_h, n_points)
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    discovery_rate = 0.25  # α
    max_generations = 1000
    local_search_freq = 20
    
    # Main search loop
    no_improvement_count = 0
    max_no_improvement = 200  # Stop after no improvement for 200 steps
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new solution (random walk)
        new_h = best_h.copy()
        
        # Perturb a random subset of elements
        n_perturb = max(1, int(n_points * 0.1))  # 10% of elements
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Add a small random perturbation
            perturbation = np.random.normal(0, 0.01)
            new_h[i] += perturbation
        
        # Repair the solution
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject the new solution
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            # Accept worse solutions with probability
            prob = np.exp(-10 * (new_c5 - best_c5))
            if np.random.rand() < prob:
                best_h = new_h.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Local refinement every local_search_freq iterations
        if generation % local_search_freq == 0 and generation > 0:
            # Gradient-based refinement
            for _ in range(20):
                if time.time() - start_time > budget_s - 10:
                    break
                
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = best_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / eps
                
                # Update using gradient
                best_h = best_h - 0.001 * grad
                best_h = repair_h(best_h, n_points)
                corr = correlate(best_h, 1 - best_h, mode='full')
                best_c5 = np.max(corr) * dx

        # Early stopping if no improvement
        if no_improvement_count > max_no_improvement:
            break

    # Final intensive gradient descent on the best solution
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0002 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h.tolist(), best_c5, n_points)
