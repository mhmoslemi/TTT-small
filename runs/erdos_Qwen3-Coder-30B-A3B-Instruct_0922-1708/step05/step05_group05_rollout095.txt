import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    trust_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find dominating lag
        lag_idx = np.argmax(corr)
        max_lag = lag_idx - (len(corr) - 1) // 2
        
        # Compute gradient
        gradient = np.zeros(n_points)
        for j in range(n_points):
            # Compute gradient contribution from max_lag
            k = max_lag
            if k >= 0:
                gradient[j] += (1 - h[(j - k) % n_points])
                gradient[j] += (1 - h[(j + k) % n_points])
            else:
                k = -k
                gradient[j] += (1 - h[(j + k) % n_points])
                gradient[j] += (1 - h[(j - k) % n_points])
        
        # Normalize gradient
        gradient = gradient / n_points
        
        # Compute proposed step
        proposed_h = h - trust_radius * gradient
        
        # Project to feasible region
        proposed_h = np.clip(proposed_h, 0, 1)
        current_sum = np.sum(proposed_h)
        if abs(current_sum - n_points/2) > 1e-6:
            proposed_h = proposed_h * (n_points / 2) / current_sum
        
        # Evaluate proposal
        proposed_corr = np.correlate(proposed_h, 1 - proposed_h, mode='full')
        proposed_c5 = np.max(proposed_corr) * dx
        
        # Accept or reject
        if proposed_c5 < current_c5:
            h = proposed_h
            current_c5 = proposed_c5
            trust_radius = min(1.2 * trust_radius, 1.0)
        else:
            trust_radius = max(0.8 * trust_radius, min_trust_radius)
        current_c5 = proposed_c5
        
        # Simulated Annealing perturbation
        if i % perturb_freq == 0:
            rand_mask = np.random.random(n_points) < 0.1
            perturbation = (np.random.random(n_points) - 0.5) * temperature
            h += perturbation * rand_mask * h * (1 - h)
            
            # Project to feasible region
            h = np.clip(h, 0, 1)
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
            # Compute updated C5
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Cooling
        temperature *= cooling_rate
    
    # Intensive local refinement
    h_refined = best_h.copy()
    
    # Simple gradient descent refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        max_lag = np.argmax(corr) - (len(corr) - 1) // 2
        gradient = np.zeros(n_points)
        for j in range(n_points):
            k = max_lag
            if k >= 0:
                gradient[j] += (1 - h_refined[(j - k) % n_points])
                gradient[j] += (1 - h_refined[(j + k) % n_points])
            else:
                k = -k
                gradient[j] += (1 - h_refined[(j + k) % n_points])
                gradient[j] += (1 - h_refined[(j - k) % n_points])
        
        gradient = gradient / n_points
        h_new = h_refined - 0.001 * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < current_c5:
            h_refined = h_new
            current_c5 = c5_new
        else:
            break
    
    corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h_refined.tolist(), final_c5, n_points)
