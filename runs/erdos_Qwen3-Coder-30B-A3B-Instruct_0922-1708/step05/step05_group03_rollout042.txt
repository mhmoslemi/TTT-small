import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_time = budget_s - 10  # Leave 10 seconds for cleanup
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Evaluate initial solution
    correlation = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(correlation) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    start_time = time.time()
    
    # Simulated Annealing
    temperature = 10.0
    cooling_rate = 0.9995
    perturbation_std = 0.02
    
    iteration = 0
    while time.time() - start_time < max_time:
        iteration += 1
        
        # Generate candidate by perturbing current solution
        candidate_h = h + np.random.normal(0, perturbation_std, n_points)
        candidate_h = np.clip(candidate_h, 0, 1)
        # Renormalize to sum constraint
        candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
        
        # Compute candidate's C5
        correlation = np.correlate(candidate_h, 1 - candidate_h, mode='full')
        candidate_c5 = np.max(correlation) * dx
        
        # Accept or reject the candidate
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / temperature):
            h = candidate_h.copy()
            current_c5 = candidate_c5
        
        # Update best solution if necessary
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Update temperature
        temperature *= cooling_rate
        
        # Occasionally do gradient descent refinement
        if iteration % 10 == 0:
            h_ref = best_h.copy()
            lr = 0.001
            for _ in range(50):
                if time.time() - start_time > max_time:
                    break
                grad = np.zeros_like(h_ref)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = h_ref.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - best_c5) / eps
                
                h_ref = h_ref - lr * grad
                h_ref = np.clip(h_ref, 0, 1)
                h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                
                corr = np.correlate(h_ref, 1 - h_ref, mode='full')
                new_c5 = np.max(corr) * dx
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
