import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    initial_alpha = 0.01
    adaptive_step = True
    perturb_freq = 50
    perturb_strength = 0.05
    perturb_decay = 0.95
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:

            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Default construction
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
    
    # Normalize to correct sum
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h = h_init.copy()
    
    # Track best solution
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    
    start_time = time.time()
    
    # Gradient-based optimization
    alpha = initial_alpha
    iteration = 0
    
    while iteration < max_iterations and (time.time() - start_time) < budget_s - 10:
        iteration += 1
        
        # Compute current value and gradient via finite differences
        corr = np.correlate(h, 1-h, mode="full")
        current_c5 = np.max(corr) * dx
        
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1-h_plus, mode="full")
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - current_c5) / eps
        
        # Update step
        h_new = h - alpha * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new fitness
        corr_new = np.correlate(h_new, 1-h_new, mode="full")
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new.copy()
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
            if adaptive_step:
                alpha *= 1.1
        else:
            if adaptive_step:
                alpha *= 0.5
        
        # Periodic perturbation to escape local minimum
        if iteration % perturb_freq == 0:
            perturb = np.random.normal(0, perturb_strength, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Decay perturbation strength
            perturb_strength *= perturb_decay
    
    return (best_h.tolist(), best_c5, n_points)
