import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Multi-resolution approach
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to sum = n_start / 2
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse optimization
    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    for iteration in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * (2.0 / n_start)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                break
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_start)
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust learning rate
        if iteration % 100 == 0:
            lr = max(1e-6, lr * 0.95)
    
    # Transfer to fine grid
    if 'initial_h_values' in globals():
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_new, x_orig, best_h)
    else:
        # If no initial, just interpolate
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_fine = np.interp(x_new, x_orig, best_h)
    
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine optimization using simple gradient descent
    patience = 0
    lr = 0.005
    
    for iteration in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            patience = 0
        else:
            patience += 1
            if patience > 200:
                break
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(len(h_fine)):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update
        h_new = h_fine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_fine = h_new
        
        # Decrease learning rate
        lr = max(1e-6, lr * 0.9995)
    
    # Final evaluation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_correlation)
    
    return (best_h.tolist(), final_c5, n_points)
