import numpy as np
import time
from scipy.optimize import minimize
from cvxpy import GLPK

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    beta = 10.0  # smoothing parameter for max approximation
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to compute smooth max
    def smooth_max(x, beta):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Helper function to compute gradient
    def compute_gradient(h, beta, dx):
        n = len(h)
        grad = np.zeros(n)
        eps = 1e-5
        
        # Compute base correlation
        corr_base = np.correlate(h, 1 - h, mode='full') * dx
        
        for i in range(n):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            grad[i] = (smooth_max(corr_plus, beta) - smooth_max(corr_base, beta)) / eps
            
        return grad
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse optimization
    n_coarse = 50
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    # Optimize coarse resolution
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h_coarse, beta, dx)
        
        # Update with gradient descent
        lr = 0.01
        h_coarse_new = h_coarse - lr * grad
        
        # Project onto feasible region
        h_coarse_new = np.clip(h_coarse_new, 0, 1)
        h_coarse_new = h_coarse_new * (n_coarse / 2) / np.sum(h_coarse_new)
        
        h_coarse = h_coarse_new
        
        # Evaluate
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h_coarse.copy()
    
    # Transfer to fine grid
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), best_h_coarse)
    
    # Optimize fine resolution
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = compute_gradient(h, beta, dx)
        
        # Update with gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Project onto feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate
        corr = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Ensure final constraint satisfaction
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
