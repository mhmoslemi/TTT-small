import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Optimization loop
    max_iter = 5000
    mu = 0.1
    lr = 0.001
    perturb_frac = 0.01
    ftol = 1e-6
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5 and corresponding lag
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        # Track best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient for the max lag
        # Correlation is symmetric, so focus on the main diagonal
        # We'll target the overlap term corresponding to the max_lag
        lag = max_lag_idx - (n_points - 1)
        # Compute the gradient w.r.t. h at each point when lag is fixed
        grad = np.zeros(n_points)
        
        # Use finite differences for gradient computation or analytical method
        # Analytical: gradient w.r.t. h[i] is (sum of 1-h[j] at positions such that j-i = lag) - sum of h[j] ...
        # This is complex so use finite difference:
        eps = 1e-5
        grad = np.zeros(n_points)
        h_plus = h.copy()
        h_minus = h.copy()
        for i in range(n_points):
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            h_minus[i] -= eps
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            grad[i] = (np.max(corr_plus) - np.max(corr_minus)) * dx / (2 * eps)
            
            h_plus[i] -= eps
            h_minus[i] += eps
        
        # Perform update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Apply perturbation occasionally
        if iteration % 100 == 0:
            perturb_mask = np.random.rand(n_points) < perturb_frac
            h_new[perturb_mask] = np.random.rand(np.sum(perturb_mask)) * 0.5 + 0.25
        
        h = h_new
        
        # Update mu
        mu *= 1.01
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
