import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse optimization
    best_c5 = float('inf')
    best_h_coarse = h.copy()
    
    # Try multiple restarts
    for _ in range(5):
        if time.time() - start_time > budget_s - 10:
            break
        h_restart = h.copy()
        np.random.shuffle(h_restart)
        h_restart = h_restart * (n_start / 2) / np.sum(h_restart)
        h_restart = np.clip(h_restart, 0, 1)
        
        # Quadratic program approach
        h_var = cp.Variable(n_start)
        C5_var = cp.Variable()
        
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_start / 2,
            C5_var >= 0
        ]
        
        try:
            prob = cp.Problem(cp.Minimize(C5_var), constraints)
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_norm = np.array(h_sol)
                    h_norm = h_norm * (n_start / 2) / np.sum(h_norm)
                    h_norm = np.clip(h_norm, 0, 1)
                    
                    # Evaluate
                    correlation = np.correlate(h_norm, 1 - h_norm, mode='full') * (2.0 / n_start)
                    current_c5 = np.max(correlation)
                    
                    if current_c5 < best_c5:
                        best_c5 = current_c5
                        best_h_coarse = h_norm.copy()
        except:
            pass
    
    # Transfer to fine grid and refine
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization with gradient descent
    max_iter = 5000
    lr = 0.01
    patience = 0
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_fine = h_fine.copy()
            
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient estimate
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_fine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[j] = (c5_plus - current_c5) / eps
            
        # Update
        h_fine = h_fine - lr * grad
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        lr = max(1e-6, lr * 0.999)
    
    # Final check
    if 'best_h_fine' not in locals():
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        final_c5 = np.max(correlation)
        best_h_fine = h_fine
    else:
        final_c5 = best_c5

    # Ensure sum constraint exactly
    best_h_fine = best_h_fine * (n_points / 2) / np.sum(best_h_fine)
    
    return (best_h_fine.tolist(), final_c5, n_points)
