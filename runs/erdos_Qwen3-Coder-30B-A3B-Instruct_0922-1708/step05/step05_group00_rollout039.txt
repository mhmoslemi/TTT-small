import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Collect initial solutions
    initial_solutions = []
    
    # Parent solution (if available)
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        if len(h_parent) != n_points:
            x_orig = np.linspace(0, 2, len(h_parent))
            x_new = np.linspace(0, 2, n_points)
            h_parent = np.interp(x_new, x_orig, h_parent)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        h_parent = np.clip(h_parent, 0, 1)
        initial_solutions.append(h_parent.copy())
    else:
        # Default initialization
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    # Random restarts
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Process each initial solution
    for i, h in enumerate(initial_solutions):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Convex relaxation approach
        try:
            # Create a relaxed version using convex optimization
            h_cvx = cp.Variable(n_points)
            sum_constraint = cp.sum(h_cvx) == n_points / 2
            bounds = [h_cvx >= 0, h_cvx <= 1]
            
            # Use a simple convex relaxation of the correlation term
            # Objective: min of sum of products over all shifts
            # This is over-simplified but provides a useful starting point
            
            prob = cp.Problem(cp.Minimize(cp.sum(h_cvx)), bounds + [sum_constraint])
            
            try:
                prob.solve(solver=ECOS, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                    h = np.array(h_cvx.value)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            except Exception:
                pass
        except Exception:
            pass
            
        # Refinement with L-BFGS-B
        def objective(h_flat):
            h = h_flat.reshape(-1)
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            return np.max(correlation)
            
        def gradient(h_flat):
            h = h_flat.reshape(-1)
            grad = np.zeros_like(h)
            eps = 1e-6
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                c5_plus = np.max(cor_plus)
                
                grad[i] = (c5_plus - objective(h_flat)) / eps
            return grad.flatten()
            
        # Optimize using L-BFGS-B
        h_flat = h.flatten()
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize
        options = {'maxiter': 2000}
        
        # Early stopping
        for _ in range(2000):
            if time.time() - start_time > budget_s - 1:
                break
                
            obj_val = objective(h_flat)
            if obj_val < best_c5:
                best_c5 = obj_val
                best_h = h_flat.copy()
                
            # Clip and normalize
            h = h_flat.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            h_flat = h.flatten()
            
            # Check convergence
            # Simple difference check won't work reliably with L-BFGS
            # Just iterate for time budget
            improvement = False
            for i in range(n_points):
                h_plus = h_flat.copy()
                h_plus[i] += 1e-4
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                if objective(h_plus) < obj_val - 1e-8:
                    improvement = True
                    break
                    
            if not improvement:
                break
                
            # Do a crude step in gradient direction
            try:
                grad = gradient(h_flat)
                h_new = h_flat - 0.01 * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h_flat = h_new
            except Exception:
                break
        
        # Check final value for the best solution so far
        if time.time() - start_time < budget_s - 1:
            final_obj = objective(h_flat)
            if final_obj < best_c5:
                best_c5 = final_obj
                best_h = h_flat.copy()
    
    # Ensure best_h is properly formed
    if best_h is None:
        # Fallback
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        best_c5 = np.max(correlation)
        best_h = h
    
    # Final computation to ensure best value
    final_h = best_h.reshape(-1)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
    final_c5 = np.max(final_correlation)
    
    return (final_h.tolist(), final_c5, n_points)
