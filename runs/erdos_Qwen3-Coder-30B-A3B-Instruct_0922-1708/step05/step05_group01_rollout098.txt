import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_update_factor = 1.2
    penalty_mu_min = 1e-4
    penalty_mu_initial = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize penalty parameter
    penalty_mu = penalty_mu_initial
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        iter_count += 1
        
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
        
        # Check for stagnation and apply learning rate decay or perturbation
        if iter_count - last_improvement > plateau_patience:
            learning_rate *= learning_rate_decay
            # Add small random perturbations
            perturb = np.random.normal(0, perturbation_fraction * learning_rate, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iter_count
        
        # Compute gradient for current c5
        # Find the lag with maximum overlap
        k_max = np.argmax(corr)
        # The overlap correspond to lag k_max in the full correlation array
        # Full correlation length is 2*n_points - 1
        # The actual lag is k_max - (n_points - 1)
        actual_lag = k_max - (n_points - 1)
        
        # Recompute correlation to be exact
        corr_exact = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr_exact)
        actual_lag = max_idx - (n_points - 1) 
        
        # gradient calculation for one lag
        grad = np.zeros_like(h)
        if actual_lag >= 0:  # positive lag
            # h[i] * (1 - h[i + actual_lag])
            # So gradient w.r.t h[i] = 1 - h[i + actual_lag] 
            # and w.r.t h[i + actual_lag] = -h[i]
            for i in range(n_points - actual_lag):
                grad[i] += 1.0 - h[i + actual_lag]
            for i in range(actual_lag, n_points):
                grad[i] -= h[i - actual_lag]
        else:  # negative lag 
            actual_lag = -actual_lag
            for i in range(n_points - actual_lag):
                grad[i] -= h[i + actual_lag]
            for i in range(actual_lag, n_points):
                grad[i] += 1.0 - h[i - actual_lag]
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        # Constraint violation penalties on bounds and sum
        bound_violation = np.sum(np.maximum(0, -h) + np.maximum(0, h - 1))
        sum_violation = np.abs(np.sum(h) - n_points / 2)
        
        # Inject penalty gradient into bound constraint
        penalty_grad = penalty_grad + 2 * penalty_mu * (np.maximum(0, -h) - np.maximum(0, h - 1)) 
        penalty_grad[0] += 2 * penalty_mu * (np.sum(h) - n_points / 2)  # gradient from sum constraint
        
        # Total gradient
        total_grad = grad + penalty_grad
        
        # Clip gradient
        total_grad = np.clip(total_grad, -10, 10)
        
        # Apply gradient
        h = h - learning_rate * total_grad
        
        # Repair feasibility
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to ensure sum(h) = n_points/2
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Update penalty
        if sum_violation > 1e-6 or bound_violation > 1e-6:
            penalty_mu = penalty_mu * penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_mu_min)
    
    # Final evaluation of best solution to ensure accuracy
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
