import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_mag = 0.1
    decay_factor = 1.0 / (1.0 + 0.01 * max_iter)
    
    # Initialize from parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: piecewise constant
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum to 0.5
    h = h * (0.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find index of minimum correlation
        min_idx = np.argmin(corr)
        min_k = min_idx - (len(corr) - 1) // 2
        
        # Compute gradient of correlation w.r.t h
        grad = np.zeros(n_points)
        for i in range(n_points):
            # For k-step correlation, derivatives of h[i] 
            # with respect to terms at indices i-k and i+k
            if i - min_k >= 0 and i - min_k < n_points:
                grad[i] += 1.0
            if i + min_k >= 0 and i + min_k < n_points:
                grad[i] += 1.0
                
        # Apply gradient ascent (maximize the minimum)
        h += learning_rate * grad
        h = np.clip(h, 0, 1)
        # Normalize to maintain sum constraint
        h = h * (0.5 / np.sum(h))
        
        # Check for improvement
        corr_new = correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Perturb periodically
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_mag * np.exp(-decay_factor * iteration), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.sum(h))
    
    # Final check
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
