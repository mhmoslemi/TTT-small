import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    initial_alpha = 0.01
    min_alpha = 1e-5
    temp_start = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction - block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Normalize to exact sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Initial evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    alpha = initial_alpha
    temperature = temp_start
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        dominant_k = np.argmax(corr)
        max_overlap = corr[dominant_k]
        
        # Compute gradient (simplified)
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Gradient w.r.t. h[i]
            # We approximate this by finite differences
            # df/dh_i = (1 - h[(i-k) mod n]) + (1 - h[(i+k) mod n])
            # But for now we just approximate based on the local impact
            
            # Estimate gradient more simply using finite differences
            eps = 1e-6
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            
            gradient[i] = (max_plus - max_overlap) / eps
        
        # Step in gradient direction
        step = -alpha * gradient
        new_h = h + step
        
        # Project new_h back to feasible region  
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Evaluate new solution
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject step
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            alpha = min(alpha * 1.2, 1.0)  # Expand trust region
        else:
            alpha = max(alpha * 0.8, min_alpha)  # Shrink trust region
            
        # Periodic perturbation by Simulated Annealing
        if iteration % perturb_freq == 0:
            # Apply random perturbations
            for i in range(n_points):
                if np.random.rand() < 0.1:  # 10% chance per point
                    perturb_mag = temperature * h[i] * (1 - h[i])
                    delta = np.random.uniform(-perturb_mag, perturb_mag)
                    new_h[i] = h[i] + delta
                    new_h[i] = np.clip(new_h[i], 0, 1)
                    
            # Renormalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate new solution
            corr_new = np.correlate(new_h, 1 - new_h, mode='full')
            new_c5 = np.max(corr_new) * dx
            
            if new_c5 < best_c5:
                h = new_h.copy()
                best_c5 = new_c5
                
        temperature *= cooling_rate
        
    # Final intensive refinement with gradient descent
    for _ in range(500):
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        
        # Recompute gradient with full domain
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Estimate contribution to gradient from h[i] to max_k
            # By finite differences
            eps = 1e-6
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fast_max_plus = np.max(corr_plus)
            grad[i] = (fast_max_plus - np.max(corr)) / eps
            
        # Update
        h = h - 0.001 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        corr = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
