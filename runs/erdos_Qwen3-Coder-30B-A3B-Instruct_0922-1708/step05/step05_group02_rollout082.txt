import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Parameter setup
    alpha = 0.25  # Discovery rate
    epsilon = 1e-5  # Perturbation step
    max_iter = 1000  # Maximum iterations
    min_improvement_iter = 200  # For early stopping
    
    # Initialize solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Track improvement for early stopping
    last_improvement = 0
    last_c5 = best_c5
    
    # Main cuckoo search loop
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution by random walk (cuckoo)
        h_new = h.copy()
        # Perturb 10% of elements
        indices_to_change = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices_to_change:
            h_new[i] += np.random.normal(0, epsilon)
        
        # Repair new solution
        h_new = repair_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        # Accept or reject the new solution
        if c5_new < best_c5:
            h = h_new.copy()
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iter_count
            last_c5 = c5_new
        else:
            # Probabilistic acceptance with discovery rate
            if np.random.rand() < alpha:
                h = h_new.copy()
                c5_new = evaluate_h(h, dx)
                if c5_new < best_c5:
                    best_h = h.copy()
                    best_c5 = c5_new
                    last_improvement = iter_count
                    last_c5 = c5_new
        
        # Local search periodically
        if iter_count % 50 == 0 and iter_count > 0:
            # Gradient-based refinement on the best solution
            h_refined = best_h.copy()
            for _ in range(50):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / eps
                
                h_refined = h_refined - 0.001 * grad
                h_refined = repair_h(h_refined, n_points)
                corr = correlate(h_refined, 1 - h_refined, mode='full')
                new_c5 = np.max(corr) * dx
                
                if new_c5 < best_c5:
                    best_h = h_refined.copy()
                    best_c5 = new_c5
                    
        # Early stopping if no improvement for a while
        if iter_count - last_improvement > min_improvement_iter:
            break
            
    # Final refinement with gradient descent on the best solution
    h_refined = best_h.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        h_refined = h_refined - 0.0005 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5 = np.max(corr) * dx
        
    return (h_refined.tolist(), best_c5, n_points)
