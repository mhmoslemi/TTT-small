import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use block construction as default
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Current best
    best_h = h.copy()
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Trust region parameters
    alpha = 0.01
    min_alpha = 1e-5
    success_count = 0
    max_success = 5
    
    # Simulated Annealing params
    temperature = 0.1
    cooling_rate = 0.99
    perturbation_freq = 50
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlation and find max_k
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        
        # Compute gradient for the max_k lag
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Add contribution from both directions
            idx1 = (i - max_k) % n_points
            idx2 = (i + max_k) % n_points
            gradient[i] = (1 - h[idx1]) + (1 - h[idx2])
        
        # Normalize gradient
        gradient = gradient / n_points
        
        # Take gradient step
        new_h = h - alpha * gradient
        
        # Project to feasible region [0,1]
        new_h = np.clip(new_h, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(new_h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-6:
            new_h = new_h * target_sum / current_sum
        
        # Evaluate new solution
        new_corr = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(new_corr) * dx
        
        # Accept or reject
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            success_count += 1
            alpha = min(0.1, alpha * 1.2)  # Expand trust region
        else:
            alpha = max(min_alpha, alpha * 0.8)  # Shrink trust region
            success_count = 0
        
        # Simulated Annealing perturbation
        if iteration % perturbation_freq == 0 and iteration > 0:
            # Perturb some variables
            perturb_mask = np.random.rand(n_points) < 0.3
            h_perturbed = h.copy()
            for i in range(n_points):
                if perturb_mask[i]:
                    # Apply small random perturbation with temperature-scaled magnitude
                    delta = np.random.normal(0, temperature) * (1 - h[i])
                    h_perturbed[i] = h[i] + delta
            h_perturbed = np.clip(h_perturbed, 0, 1)
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            new_corr = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            new_c5 = np.max(new_corr) * dx
            
            if new_c5 < best_c5:
                h = h_perturbed.copy()
                best_c5 = new_c5
                
        # Decay temperature
        temperature *= cooling_rate
        
        # Early stopping based on convergence
        if success_count >= max_success:
            break
    
    # Final intensive gradient descent
    for _ in range(500):
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        
        gradient = np.zeros(n_points)
        for i in range(n_points):
            idx1 = (i - max_k) % n_points
            idx2 = (i + max_k) % n_points
            gradient[i] = (1 - h[idx1]) + (1 - h[idx2])
        
        gradient = gradient / n_points
        h_new = h - 0.001 * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
        else:
            break
    
    # Final correction to make sure constraints are satisfied
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    return (h.tolist(), best_c5, n_points)
