import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_points = 200  # Final resolution
    dx = 2.0 / n_points
    dx_start = 2.0 / n_start
    max_iter = 3000
    patience_limit = 200
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h_start = np.interp(x_new, x_orig, h)
        else:
            h_start = h.copy()
    else:
        h_start = np.ones(n_start) * 0.5
        
    # Normalize to sum to n_start/2
    h_start = h_start * (n_start / 2) / np.sum(h_start)
    h_start = np.clip(h_start, 0, 1)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros_like(h_start)
    v = np.zeros_like(h_start)
    
    # For coarse grid
    best_h_coarse = h_start.copy()
    best_c5_coarse = float('inf')
    
    # Coarse grid optimization with Adam
    last_c5 = float('inf')
    patience = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_start, 1 - h_start, mode='full') * dx_start
        current_c5 = np.max(correlation)
        
        # Update best solution
        if current_c5 < best_c5_coarse:
            best_c5_coarse = current_c5
            best_h_coarse = h_start.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_start)
        eps = 1e-4
        
        for i in range(n_start):
            h_plus = h_start.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_start
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_count + 1))
        v_hat = v / (1 - beta2**(iter_count + 1))
        
        # Update parameters
        h_start = h_start - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible bounds
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_start / 2) / np.sum(h_start)
        
        iter_count += 1
    
    # Interpolate to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization with Adam
    # Initialize Adam moments
    m_fine = np.zeros_like(h_fine)
    v_fine = np.zeros_like(h_fine)
    
    best_h_fine = h_fine.copy()
    best_c5_fine = float('inf')
    
    last_c5 = float('inf')
    patience = 0
    iter_count = 0
    max_iter_fine = 2500
    
    while iter_count < max_iter_fine and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        # Update best solution
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h_fine.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m_fine = beta1 * m_fine + (1 - beta1) * grad
        v_fine = beta2 * v_fine + (1 - beta2) * grad**2
        m_hat = m_fine / (1 - beta1**(iter_count + 1))
        v_hat = v_fine / (1 - beta2**(iter_count + 1))
        
        # Update parameters
        h_fine = h_fine - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible bounds
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        iter_count += 1
    
    # Final evaluation
    final_h = best_h_fine
    correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (final_h.tolist(), final_c5, n_points)
