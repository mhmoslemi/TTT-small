import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 1e-3
    penalty_update_factor = 1.2
    μ_min = 1e-4
    μ_initial = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Penalty parameter
    μ = μ_initial
    
    # Optimization loop
    iteration = 0
    no_improvement_count = 0
    current_lr = learning_rate
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best if better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Check for stagnation and adjust learning rate
        if no_improvement_count > plateau_patience:
            current_lr *= learning_rate_decay
            no_improvement_count = 0
            
            # Add random perturbation
            perturbation = np.random.normal(0, perturbation_fraction, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Compute gradient for correlation term
        max_k = np.argmax(corr)
        max_lag = max_k - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + abs(max_lag) < n_points:
                if max_lag >= 0:
                    grad[i] = 1 - h[i + max_lag]
                else:
                    grad[i] = 1 - h[i - max_lag]
            else:
                grad[i] = 0
        
        # Adjoint method to compute full gradient
        grad_corr = np.zeros(n_points)
        for k in range(n_points):
            if k + abs(max_lag) < n_points:
                if max_lag >= 0:
                    grad_corr[k] += 1 - h[k + max_lag]
                else:
                    grad_corr[k] += 1 - h[k - max_lag]
        
        # Compute penalty gradient and update
        penalty_grad = np.zeros(n_points)
        violation_sum = np.sum(h) - n_points / 2
        for i in range(n_points):
            penalty_grad[i] = 2 * μ * (h[i] - np.clip(h[i], 0, 1))
        penalty_grad += 2 * μ * violation_sum
        
        # Full gradient
        full_grad = grad_corr + penalty_grad
        
        # Update h
        h = h - current_lr * full_grad
        
        # Repair constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        
        iteration += 1
        # Update penalty
        if abs(h_sum - n_points / 2) > 1e-8:
            μ = μ * penalty_update_factor
        else:
            μ = max(μ / penalty_update_factor, μ_min)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
