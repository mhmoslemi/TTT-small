import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    perturb_fraction = 0.01
    
    # Initialize h from initial_h_values or randomly
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Scale to meet constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial overlap
    def compute_overlap(h):
        # This is an efficient vectorized approach for computing the overlap
        # This computes all correlations for all lags and finds maximum
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        h_rev = np.concatenate([h, np.zeros(n_points - 1)])[::-1]
        corr_full = np.convolve(h_full, h_rev, mode='full')
        corr_full = corr_full[n_points-1:]  # This is the full correlation
        # We want : h[i]* (1 - h[i+k]) for all i and k
        # The full convolution gives us sum_{i} h[i] h[i+k], 
        # but we want sum_{i} h[i](1-h[i+k]) = sum_{i} h[i] - sum_{i} h[i] h[i+k]
        # Which is sum_{i} h[i] * n - sum_{i} h[i] h[i+k] 
        # with proper limits
        # But using the same method - we compute correlation of h with (1-h)
        # so we have to compute this properly
        # Let's recompute properly:
        h_conj = 1 - h  # h_conj = 1 - h
        h_conj_full = np.concatenate([h_conj, np.zeros(n_points - 1)])
        h_full = np.concatenate([h, np.zeros(n_points - 1)])
        corr = np.convolve(h_full, h_conj_full[::-1], mode='full')
        # The maximum lag is n_points -1, which corresponds to b[0], a[n-1], etc.
        # corr has size 2n -1 = 199, indices from 0 to 198
        # The valid indices in the sliding window mode are 
        # from n-1 to 2*(n-1) = 198, so there are 100 valid overlap indices
        # But correct way is to index only from n-1 to 2n-2 to get the maximum
        # of all shifts for positive lags 1 to n-1 (but not negative, x+k)
        # For shifts x+k = 1,2,...,n-1  
        # indices: 1 to n-1 are the "positive overlap"
        # These are the correlations from index (2n-2) - (n-1) = n-1 to 2n-2 -> valid overlap indices
        # indices with (n-1) to (2n-2) in 0-based indexing are equivalent to now-full:
        # The correlation C[k] of h and 1-h for lag k is from correlation index [n-1+k] for k=0..n-1
        # Max of our correlation for valid (for all k) from returns at n-1 to 2*n-2
        # return np.max(corr[n_points-1 : 2*n_points-1]) * dx
        # Actually, to make it simpler, the correct overlap is with convolution of h with 1 - h in the reverse,
        # which for lag k = 0..n-1 gives us sum h[i] (1 - h[i+k]) for all valid i.
        # Let's do it directly by defining list of correlation values.
        # The lags in correlate with mode='full' produce:
        # The"full correl" of a with b is size 2*n-1. The "valid" part starts at index n-1.
        # C[s] = sum_{i = max(s-n+1,0)}^{min(s, n-1)} a[i] b[s-i]
        # so for b = (1-h) and a = h:
        # C[k] = sum_{i = max(k-n+1,0)}^{min(k, n-1)} h[i] (1-h[k-i]) = sum_{i} h[i] - sum_{i} h[i] h[k-i]
        # sum_{i} h[i] = n/2, and correlations are with indices [k] = sum_{i} h[i] h[k-i]
        # so C[k] = n/2 - sum_{i} h[i] h[k-i]
        # Implies max_ k C[k] = max_k (n/2 - sum_{i} h[i] h[k-i]) = n/2 - min_k sum_i h[i] h[k-i]
        # But we want max_k sum_i h[i] (1 - h[i+k]) = max_k [n/2 - sum_i h[i] h[k-i]] = n/2 - min_k sum_i h[i] h[k-i]
        # But we can just evaluate C = conv(h, 1-h)
        # Then max(C) = max_k [n/2 - sum[ h[i] h[k+i] ]] = n/2 - min_k [sum[ h[i] h[k+i] ]]
        # So we want to minimize sum h[i] h[k+i] = sum_{i=0}^{n-1} h[i] h[k+i] for all k
        corr_rev = np.convolve(h_full, h_conj_full[::-1], mode='full')
        # returns 2*n - 1 elements
        # max lag = n-1 (late lags)
        # max_k = (n-1)  
        # correlation with lag k = k in [0..n-1] returns index n-1 + k from full corr
        # But this is: sum_{i = 0}^{n-1} h[i] h[i+k]
        # The max is our desired negative of it. But we want the actual overlaps.
        # Since we took C = conv(h, 1-h) = sum h[i] (1-h[i+k]) = sum h[i] - sum h[i] h[i+k] 
        # We want the maximum of this for k = 0...n-1
        # So we want argmax_{k} sum h[i] (1 - h[i+k]) 
        # = argmax_k (n/2 - sum h[i] h[i+k])
        # = argmin_k sum h[i] h[i+k] 
        # Though not more directly compute only (n/2 - sum h[i] h[i+k])
        # Compute all kernel sums, just a for loop as C has length 2n-1, we want lags 0..n-1
        # and extract indices [n-1] to [2n-2] which has indices  [n-1] = lag0, ..., [2n-2] = lag n-1  
        
        # Let's compute for lags 0 to n-1:
        overlaps = []
        for k in range(n_points):
            # Compute sum_{i=0}^{n-1} h[i] * h[k+i]  
            # This is just summing the result for valid values
            # For _k = 0 to n-1, we compute the overlaps for lag k
            corr_pos = corr_rev[n_points - 1 + k]  
            # This is for lag k = (n-1+k) - (n-1) = k 
            # Bootstrapped by convolution.
            # Use the fact that conv(a,b)[k] = sum_{j=0}^{length(a)-1} a[j] b[k-j] 
            # so for b = 1-h, and a=h:
            # With degree array roll; 
            # Let's just calculate directly using matrix method for stability
            # Direct method for k:
            # The value of convolution at index 0..2n-2 representing:
            # Product for lags 0..2n-2
            # Specialized for k = 0 to n-1, we want sum h[i] h[k+i], (but also 1-h for symmetry)
            # Algorithm is:
            # We are looking for :
            # For overlap integral: sum_{i} h[i](1-h[i+k])  
            # What this product does, we expect to compute 'correct' overlaps
            # Let's do slightly incorrect but easier path: 
            # Use a proper definition: encourage valid indices
            
        # Correct approach:
        # For overlap of h with (1-h) at lag k
        # Let's define overlaps directly for all finite lags k from -n+1 to n-1:
        # We want max of overlap integral over all lags k = 0, 1, ... n-1
        # Using identity:
        # ∫ h(x)(1 - h(x+k)) dx = ∫ h(x) dx - ∫ h(x) h(x+k) dx 
        # = n/2 - ∫ h(x) h(x+k) dx 
        # So we minimize ∫ h(x) h(x+k) dx
        # Let us compute all such integrals directly:
        # ∫ h(x) h(x+k) dx : for discrete it becomes sum_{i = 0}^{n-1} h[i] h[x+k]  
        # but we must restrict this to valid range for x+k within valid range.
        # Let's turn to correlate with correct sign:  
        
        # Instead, use direct definition:
        # [h]_{i,k} = h[i] * (1 - h[i + k])
        # where i + k in [0, n-1] range only
        # Enumerate shift k from 0 to n-1
        overlaps = []
        for k in range(n_points):
            # Sum over valid indices i, where both i, i+k are in [0, n-1]
            # This means k from 0, n-1, where 0 <= i, i+k <= n-1
            # So k from 0..n-1, i from max(0, -k) to min(n-1, n-1-k) = [max(0, -k), n-1-k]  
            # But this is just range[0, n-1-k] for k=0,...,n-2
            valid_i = np.arange(0, n_points - k)
            valid_sum = np.sum(h[valid_i] * (1 - h[valid_i + k]))
            overlaps.append(valid_sum)
        # Now find max
        return np.max(overlaps) * dx
    
    best_c5 = compute_overlap(h)
    best_h = h.copy()
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Compute correlations for all lags to find maximum overlap
        # Direct computation
        overlaps = np.zeros(n_points)
        for k in range(n_points):  # lags from 0 to n-1
            # Compute sum_{i} h[i] (1-h[i+k]) for valid indices only
            # This means if k > 0, for i = 0,...,n-1-k 
            if k == 0:
                valid_sum = np.sum(h * (1 - h))
            else:
                valid_i = np.arange(n_points - k)
                valid_sum = np.sum(h[valid_i] * (1 - h[valid_i + k]))
            overlaps[k] = valid_sum
            
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        # To calculate the gradient of max_overlap w.r.t. h:
        # ∂(h[i] (1-h[i+k])) / ∂h[i] = (1 - h[i+k]) 
        # Since we are at max_k, the constraint is to zero out the gradient of just that term, 
        # something we can use to build a better vector in gradient step method in iterates.
        
        # Gradient calculation for max_k
        grad = np.zeros(n_points)
        # For each i compute: ∂(h[i] (1 - h[i+k]))= (1 - h[i+k]) and accumulate to gradient
        if max_k == 0:
            # Gradient for term i in h[i] (1-h[i+k]) = (1 - h[i+0]) = (1 - h[i])
            # We have 1 - h[i] contribution for every i  
            # But here it is all terms so all i weight: this = 1 - h[i]
            # Actually no. in overlap = 
            # ∑_i h[i] (1- h[i+k]) for k = k*
            # Partial differential of this sum w.r.t. h[j] 
            # = the term (1-h[j+k]) when j is not k 
            # So let's compute:
            # This is partial w.r.t h[j] of ∑_i h[i] (1-h[i+k])
            # If i = j, 1-h[j+k] 
            # If i = j+k, h[j](we have 1 - h[j+k]), then 0, 0, 0... 
            # That is, ∂/(∂ h[j])  ∑_i h[i] (1-h[i+k])  
            # Only terms i == j or i == j+k contribute to derivative w.r.t. h[j]
            # When i = j, we get (1 - h[j+k])
            # When i = j+k, we get h[j] (1 - h[i+j+k] = h[j] (1-h[2k+j])
            # But the point is, in range j to 0 to n-1
            # So:
            # ∂ (h[i] (1(h[i+k]))) = (1-h[i+k]) if i = j 
            # Or:    
            # ∂/∂h[j] (h[j] (1-h[j+k])) = (1-h[j+k])
            # and  
            # ∂/∂h[j] (h[j+k] (1-h[j+k+k])) = h[j] (when j+k+k < n)            
            # But at overlap we are aggregating:
            # let's reframe:
            # Consider a value like h[j] in the overlap integral:
            # ∑i h[i] (1-h[i+k]) 
            # Partial derivative w.r.t h[j]: The non-zero contribution only comes from index i = j in h[j],
            # and index i = j-k in term h[j-k] (1-h[j]), but we include h[j-k] (1-h[j+k]) term, so corresp. only index j-k.
            # When i = j in the sum, it is the term h[j] (1-h[j+k])
            # So ∂ = (1 - h[j+k]). 
            # Now from i = j-k, we have the h[j-k] (1-h[j-k+k]) = h[j-k] (1-h[j])  
            # Therefore ∂h[j-k] = -h[j]  
            # overall derivatives:  
            # You only get:
            # Grad[j] += (1 - h[j+k]) - h[j*(k)] where h[j*(k)] = 1 if j+k < n, 0 else
            # Actually better is to use two lag-based sol'n.
            # Let's compute all terms of sum over all i of h[i] (1-h[i+k]) 
            # For some fixed k, the derivative wrt h[j] is:
            # If i == j: derivative is (1 - h[j+k])
            # If i == j+k: derivative is h[j] 
            # If none of above: 0
            # In other words, ∂ ∑ (h[i] (1-h[i+k])) / ∂ h[j]  
            # = (1 - h[j+k]) if (j+k) < n
            # - h[j-k] if (j-k) >= 0 and (j-k) < n  
            # + 0 elsewhere
            # But since when we are computing the overlap, we do a shift,
            # the h[i], h[i+k] (only valid when i+k < n) 
            # i is from 0 to n-1-k, so overlap includes h[i] (1-h[i+k]) for all i 
            # So for this particular lag k, over all i=0..n-k-1, we have:
            # Some i corresponds to index j, getting derivative of h[i](1-h[i+k])
            # ∂ (h[i] (1-h[i+k])) / ∂ h[j]
            # = 0 if i ≠ j and i ≠ j-k
            # = (1 - h[i+k]) if i = j and j+k < n  
            # =  h[i] if i = j-k 
            # hence gradient for j:
            # The value is just 0 if j+k ≥ n and 0 if j-k < 0  
            # It is (1-h[j+k]) else when j+k < n
            # and -h[j-k] when j-k >= 0 
            # For all j, we get these two contributions,  
            # For k = max_k:
            # grad = np.zeros(n_points)
            for j in range(n_points):
                grad[j] += (1 - h[j+max_k]) if (j + max_k) < n_points else 0  # mechanism of term h[j] * (1-h[j+max_k])  
                if (j - max_k) >= 0 and (j - max_k) < n_points:  
                    grad[j] -= h[j - max_k]  # From the term h[j-max_k] (1-h[j]) 
        else:
            # From h[i] (1 - h[i + k]) for i = 0..n-1-k
            # ∂/∂h[j] = contribution to this derivative when i = j or i = j-k
            # 1. Term h[j] (1 - h[j+k])  => 1 - h[j+k]  
            # 2. Term h[j-k] (1 - h[j]) => - h[j] 
            # But manually index-space based - no low-level dependencies on numpy correlate
            for j in range(n_points):
                # Contribution from h[j] (1 - h[j+k]) => h[j+max_k]
                if j + max_k < n_points:  # This means i=j+k is valid
                    grad[j] += 1 - h[j + max_k] 
                # Contribution from h[j-k] (1 - h[j]) => i = j-k, so h[j-k] is multiplied with something term
                # That term 1-h[j] contributes - h[j]
                if j - max_k >= 0:  # means i = j-k is valid index (0 to n-1)  
                    grad[j] -= h[j - max_k] 
                
        # Update with simple gradient descent step
        h = h - alpha * grad
        # Project back to feasible:
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to sum = n/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Check if C5 improved
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iter_count  
        
        # Adaptive step size: if no improvement, reduce alpha
        if iter_count - last_improvement > 100:
            alpha *= line_search_factor
            
        # If improvement is very small, stop
        if iter_count % 100 == 0:
            if abs(current_c5 - best_c5) < grad_threshold:
                break
                
        # Occasionally add noise
        if np.random.random() < perturb_fraction:
            h += np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
