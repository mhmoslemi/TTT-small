import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    alpha_decay = 0.5
    perturbation_freq = 50
    perturbation_magnitude = 0.05
    trust_radius_init = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    # Normalize sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient approximation via finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure bounds
            h_plus = np.clip(h_plus, 0, 1)
            # Correct sum
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Step direction
        new_h = h - alpha * grad
        
        # Project onto feasible set: bounds and sum constraint
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Compute correlation for new point
        corr_new = np.correlate(new_h, 1 - new_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept if improvement
        if new_c5 < best_c5:
            h = new_h.copy()
            best_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h.copy()
                
        elif new_c5 > base_c5 * 1.01:  # If not improving significantly, reduce step size
            alpha *= alpha_decay
            
        # Periodic perturbations to escape local minima
        if iteration % perturbation_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturbation_magnitude, n_points)
            h_pert = h + noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            if c5_pert < best_c5:
                h = h_pert.copy()
                best_c5 = c5_pert
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
