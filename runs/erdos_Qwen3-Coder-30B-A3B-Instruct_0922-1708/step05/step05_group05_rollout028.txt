import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    # Project to feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Compute C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient-based optimization with projection
    alpha = 0.01
    decay = 0.95
    min_alpha = 1e-6
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        
        # Compute current gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Take step
        h_new = h - alpha * grad
        h_new = project_h(h_new)
        
        # Accept step if improvement
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            alpha *= 1.1  # Increase step size
        else:
            alpha *= decay  # Decrease step size
            if alpha < min_alpha:
                alpha = min_alpha
        
        # Periodic perturbation to escape local minima
        if iter_count % 50 == 0:
            perturb_magnitude = 0.01 * (1.0 - (time.time() - start_time) / budget_s)
            if perturb_magnitude > 1e-8:
                h = project_h(h + np.random.normal(0, perturb_magnitude, n_points))
                new_c5 = compute_c5(h)
                if new_c5 < best_c5:
                    best_c5 = new_c5
        
        # Occasionally reinitialize from best
        if iter_count % 200 == 0 and iter_count > 0:
            h = best_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
