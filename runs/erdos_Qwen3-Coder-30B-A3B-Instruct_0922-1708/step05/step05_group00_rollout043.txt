import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5

    # Normalize to satisfy constraint ∫ h(x) dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Smooth approximation to max: log(sum(exp(β * x))) / β
    def smooth_max(x, beta=100):
        return np.log(np.sum(np.exp(beta * x))) / beta

    best_h = h.copy()
    best_c5 = float('inf')
    patience = 0
    last_c5 = float('inf')
    
    # Parameters
    n_coarse = 50
    n_fine = 200
    beta = 50.0  # For smooth max approximation
    
    # Coarse grid optimization - downsample
    if n_points > n_coarse:
        h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    else:
        h_coarse = h[:n_coarse]
        
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Gradient-based optimization on coarse grid
    lr = 0.01
    for it in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap with smooth max
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_coarse)
        c5_approx = smooth_max(correlation, beta)
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h_coarse)
        eps = 1e-5
        for i in range(len(h_coarse)):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_coarse / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_coarse)
            c5_plus = smooth_max(cor_plus, beta)
            grad[i] = (c5_plus - c5_approx) / eps
            
        # Update
        h_new = h_coarse - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_coarse / 2) / np.sum(h_new)
        h_coarse = h_new
        
        # Check improvement
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_coarse)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h_coarse.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                break
                
        lr = max(1e-5, lr * 0.9995)
    
    # Transfer optimal coarse solution to fine grid
    if n_points > n_fine:
        h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_coarse), best_h_coarse)
    else:
        h_fine = best_h_coarse[:n_fine]
        
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine-grid optimization
    lr = 0.01
    patience = 0
    for it in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap with smooth max
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * (2.0 / n_fine)
        c5_approx = smooth_max(correlation, beta)
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h_fine)
        eps = 1e-5
        for i in range(len(h_fine)):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_fine)
            c5_plus = smooth_max(cor_plus, beta)
            grad[i] = (c5_plus - c5_approx) / eps
            
        # Update
        h_new = h_fine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_fine / 2) / np.sum(h_new)
        h_fine = h_new
        
        # Check improvement
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * (2.0 / n_fine)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
            patience = 0
        else:
            patience += 1
            if patience > 500:
                break
                
        lr = max(1e-5, lr * 0.9995)
    
    # Final evaluation at appropriate resolution
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full') * (2.0 / len(best_h))
    final_c5 = np.max(final_correlation)
    
    # Return result
    return (best_h.tolist(), final_c5, len(best_h))
