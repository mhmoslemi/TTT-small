import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK

def smooth_max(x, beta=10.0):
    """Smooth approximation to maximum using log-sum-exp"""
    return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Use structured initialization for better spread
        h = np.ones(n_points) * 0.5
    
    # Normalize to meet sum constraint exactly  
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    best_h = h.copy()
    
    # Coarse grid optimization (n=50) to escape local minima
    n_coarse = 50
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Gradient descent on smooth approximation of C5
    max_iter = 2000
    lr = 0.01
    beta = 10.0
    patience = 0
    last_loss = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap using correlation
        corr = correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_coarse)
        # Smooth approximation to the maximum overlap
        smooth_overlaps = smooth_max(corr, beta)
        
        # Compute gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h_coarse)
        for j in range(len(h_coarse)):
            h_eps = h_coarse.copy()
            h_eps[j] += eps
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = h_eps * (n_coarse / 2) / np.sum(h_eps)
            corr_eps = correlate(h_eps, 1 - h_eps, mode='full') * (2.0 / n_coarse)
            smooth_eps = smooth_max(corr_eps, beta)
            grad[j] = (smooth_eps - smooth_overlaps) / eps
        
        # Update step
        h_coarse -= lr * grad
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
        
        # Check for improvement
        if abs(smooth_overlaps - last_loss) < 1e-8:
            patience += 1
        else:
            patience = 0
        
        if smooth_overlaps < best_c5:
            best_c5 = smooth_overlaps
            # Interpolate back to full resolution
            h_full = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
            h_full = h_full * (n_points / 2) / np.sum(h_full)
            h_full = np.clip(h_full, 0, 1)
            best_h = h_full.copy()
            
        last_loss = smooth_overlaps
        
        # Early stopping
        if patience > 1000:
            break
            
        # Adjust learning rate
        lr = max(1e-6, lr * 0.9996)
    
    # Transfer to fine grid and optimize
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    max_iter = 2000
    lr = 0.005
    beta = 10.0
    patience = 0
    last_loss = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        corr = correlate(h_fine, 1 - h_fine, mode='full') * dx
        smooth_overlaps = smooth_max(corr, beta)
        
        eps = 1e-5
        grad = np.zeros_like(h_fine)
        for j in range(len(h_fine)):
            h_eps = h_fine.copy()
            h_eps[j] += eps
            h_eps = np.clip(h_eps, 0, 1)
            h_eps = h_eps * (n_points / 2) / np.sum(h_eps)
            corr_eps = correlate(h_eps, 1 - h_eps, mode='full') * dx
            smooth_eps = smooth_max(corr_eps, beta)
            grad[j] = (smooth_eps - smooth_overlaps) / eps
            
        h_fine -= lr * grad
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        if abs(smooth_overlaps - last_loss) < 1e-8:
            patience += 1
        else:
            patience = 0
            
        if smooth_overlaps < best_c5:
            best_c5 = smooth_overlaps
            best_h = h_fine.copy()
            
        last_loss = smooth_overlaps
        
        if patience > 1000:
            break
            
        lr = max(1e-6, lr * 0.9996)
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(final_corr)
    
    return (best_h.tolist(), final_c5, n_points)
