import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h based on initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Initial uniform guess
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Temperature schedule parameters
    T0 = 10.0
    temp_decay = 1.0
    
    # Simulated Annealing loop
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Perturb the current solution
        perturbation = np.random.normal(0, 0.02, n_points)
        h_new = h + perturbation
        
        # Project to feasible space
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Evaluate current solution
        corr_old = np.correlate(h, 1 - h, mode='full')
        c5_old = np.max(corr_old) * dx
        
        # Compute temperature
        T = T0 / (1.0 + temp_decay * iteration)
        
        # Accept or reject the new solution
        delta_c5 = c5_new - c5_old
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_new.copy()
            c5_old = c5_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
    
    # Local gradient descent refinement
    h_ref = best_h.copy()
    lr = 0.001
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros_like(h_ref)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
        
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_ref.copy()
    
    return (best_h.tolist(), best_c5, n_points)
