import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial correlation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Bicriteria optimization with restarts
    max_iter = 6000
    alpha = 0.01
    patience = 200
    stagnation_count = 0
    restart_count = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Adaptive learning rate
        if iteration > 0 and iteration % 200 == 0:
            alpha *= 0.5
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Determine which lags contribute to the maximum overlap
        max_val = np.max(corr)
        max_indices = np.where(np.abs(corr - max_val) < 1e-8)[0]
        
        # Create a "gradients" vector that tries to reduce the maximum overlap
        gradient = np.zeros(n_points)
        for i in range(n_points):
            # Simple finite difference approximation of gradient
            h_plus = h.copy()
            h_plus[i] = min(1.0, h_plus[i] + 1e-5)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            gradient[i] = (c5_plus - c5) / 1e-5
        
        # Simple gradient step, make sure it's a descent direction
        h_new = h - alpha * gradient
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Projection to sum = n/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept the new solution if it improves
        if c5_new < c5:
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Per iteration for ensuring diversity
        if iteration % 100 == 0 and stagnation_count > 0:
            # Random perturbation
            h = h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Check for early termination
        if stagnation_count > patience:
            if restart_count < 3:
                # Restart with a small perturbation
                h = np.random.rand(n_points) * 0.1 + 0.45
                h = h * (n_points / 2) / np.sum(h)
                stagnation_count = 0
                restart_count += 1
            else:
                break
    
    # Final evaluation of the best result
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    c5_final = np.max(corr_final) * dx
    
    # Ensure final version respects both constraints
    assert np.all(best_h >= 0) and np.all(best_h <= 1), "Constraints violated"
    
    return (best_h.tolist(), c5_final, n_points)
