import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient-based optimization
    iter_count = 0
    plateau_counter = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 5:
        iter_count += 1
        
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best if needed
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_count
            plateau_counter = 0
        else:
            plateau_counter += 1
        
        # Check for plateau
        if plateau_counter >= plateau_patience:
            # Perturb randomly
            perturb_mask = np.random.rand(n_points) < perturb_fraction
            h[perturb_mask] += np.random.normal(0, 0.01, np.sum(perturb_mask))
            # Repair
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            plateau_counter = 0
            continue
            
        # Find the maximizing lag
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient for this lag
        # The gradient is based on the derivative of the convolution
        # For maximum lag k*, we have:
        # ∂(h ⋆ (1-h))[k*]/∂h[i] = (1 - h shifted by k*)[i] - h[i] 
        # But we want to account for the indicator function over the overlap interval
        # Since it's a full correlation, we compute contributions
    
        grad = np.zeros(n_points)
        k_star = max_lag_idx - (n_points - 1)  # Translate to actual lag
        
        # Create shifted versions for counting
        h_shifted = np.roll(h, -k_star) if k_star < 0 else np.roll(h, k_star)
        one_minus_h_shifted = 1 - h_shifted if k_star < 0 else 1 - np.roll(h, -k_star)

        # Gradient calculation based on involved indices
        # For h[i], the contribution comes from how the lagged 
        # structure changes with small variation of h[i]
        for i in range(n_points):
            # When you shift h by k_star with i, 
            # the overlap region includes specific terms
            # We directly reason about the impacted indices.
            # This kind of stencil-based gradient approximation is more reliable for discrete
            # case than analytical derivative.
            pass
            
        # A more reliable way: directional derivative estimation
        temp_h = h.copy()
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            temp_h[i] += eps
            if temp_h[i] > 1:
                temp_h[i] = 1
            elif temp_h[i] < 0:
                temp_h[i] = 0
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
            
            temp_corr = correlate(temp_h, 1 - temp_h, mode='full')
            temp_c5 = np.max(temp_corr) * dx
            
            grad[i] = (temp_c5 - c5) / eps
            temp_h[i] = h[i]  # restore
            
        # Apply learning rate and clip
        h -= learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Decay learning rate
        if learning_rate > min_learning_rate:
            learning_rate *= learning_rate_decay
            
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
