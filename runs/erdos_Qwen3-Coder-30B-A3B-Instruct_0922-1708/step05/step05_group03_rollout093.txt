import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()

    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # uniform distribution

    # Project to feasible region
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec

    # Compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx

    # Simulated Annealing
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    best_h = current_h.copy()
    best_c5 = current_c5

    # Parameters
    T0 = 10.0
    cooling_rate = 0.9999
    noise_std = 0.02
    max_iter = int(5e5)

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break

        # Annealing temperature
        T = T0 / np.log(1 + iteration + 1)

        # Perturb
        candidate_h = current_h + np.random.normal(0, noise_std, n_points)
        candidate_h = project(candidate_h)

        # Evaluate
        candidate_c5 = compute_c5(candidate_h)

        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate_h
            current_c5 = candidate_c5

        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5

    # Local refinement with gradient descent
    h_refine = best_h.copy()
    lr = 0.001
    eps = 1e-4
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros_like(h_refine)
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = project(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
            
        h_refine = h_refine - lr * grad
        h_refine = project(h_refine)
        
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_refine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
