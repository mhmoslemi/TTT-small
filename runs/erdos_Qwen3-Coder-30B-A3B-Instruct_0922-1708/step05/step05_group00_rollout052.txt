import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h based on provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Ensure n_points is appropriate
    if n_points < 50:
        n_points = 50
    
    dx = 2.0 / n_points
    
    # Normalize the initial h
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Multi-resolution optimization
    # Start with a coarser grid and gradually go finer
    coarse_n = min(50, n_points)
    fine_n = max(200, n_points)
    
    # First, optimize on coarse grid
    coarse_h = np.interp(np.linspace(0, 2, coarse_n), np.linspace(0, 2, n_points), h)
    coarse_h = np.clip(coarse_h, 0, 1)
    
    # Normalize is here too
    coarse_h = coarse_h * (coarse_n / 2) / np.sum(coarse_h)
    coarse_dx = 2.0 / coarse_n
    
    # Adam optimizer for coarse grid with subgradient approximation
    def compute_correlation(h, dx):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    # Helper to project h to feasible region
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (coarse_n / 2) / np.sum(h)
        return h
        
    # Adam optimizer setup  
    m = np.zeros_like(coarse_h)
    v = np.zeros_like(coarse_h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    t = 0
    
    last_c5 = float('inf')
    patience = 0
    best_c5 = float('inf')
    best_coarse_h = coarse_h.copy()
    
    max_iter = 10000
    for iter_i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = compute_correlation(coarse_h, coarse_dx)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_coarse_h = coarse_h.copy()
        
        if abs(c5_current - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
        last_c5 = c5_current
        
        # Compute gradient using finite differences
        grad = np.zeros_like(coarse_h)
        eps_fd = 1e-4
        for i in range(coarse_n):
            h_plus = coarse_h.copy()
            h_plus[i] += eps_fd
            h_plus = project_h(h_plus)
            c5_plus = compute_correlation(h_plus, coarse_dx) 

            h_minus = coarse_h.copy()
            h_minus[i] -= eps_fd
            h_minus = project_h(h_minus)
            c5_minus = compute_correlation(h_minus, coarse_dx)

            grad[i] = (c5_plus - c5_minus) / (2 * eps_fd)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        coarse_h = coarse_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project back to feasible region
        coarse_h = project_h(coarse_h) 
        
        # Decrease learning rate
        lr = max(1e-6, lr * 0.9995)
        
    coarse_h = best_coarse_h.copy()
    
    # Transfer to fine grid and optimize further
    if fine_n > coarse_n:
        # Interpolate from coarse to fine grid
        fine_h = np.interp(np.linspace(0, 2, fine_n), np.linspace(0, 2, coarse_n), coarse_h)
        fine_h = np.clip(fine_h, 0, 1)
        fine_h = fine_h * (fine_n / 2) / np.sum(fine_h)
        fine_dx = 2.0 / fine_n
    else:
        fine_h = coarse_h.copy()
        fine_h = np.clip(fine_h, 0, 1)
        fine_h = fine_h * (n_points / 2) / np.sum(fine_h)
        fine_dx = dx
        
    # Fine optimization
    m = np.zeros_like(fine_h)
    v = np.zeros_like(fine_h)
    lr = 0.005
    
    best_fine_h = fine_h.copy()
    best_c5_fine = compute_correlation(fine_h, fine_dx)
    
    # Optimizing fine grid
    max_iter = 10000
    for iter_i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5_current = compute_correlation(fine_h, fine_dx)
        if c5_current < best_c5_fine:
            best_c5_fine = c5_current
            best_fine_h = fine_h.copy()
            
        # Compute gradient
        grad = np.zeros_like(fine_h)
        eps_fd = 1e-4
        for i in range(fine_n):
            h_plus = fine_h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (fine_n / 2) / np.sum(h_plus)
            c5_plus = compute_correlation(h_plus, fine_dx) 

            h_minus = fine_h.copy()
            h_minus[i] -= eps_fd
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (fine_n / 2) / np.sum(h_minus)
            c5_minus = compute_correlation(h_minus, fine_dx)

            grad[i] = (c5_plus - c5_minus) / (2 * eps_fd)
        
        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        fine_h = fine_h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project back to feasible region
        fine_h = np.clip(fine_h, 0, 1)
        fine_h = fine_h * (fine_n / 2) / np.sum(fine_h)
        
        # Decrease learning rate
        lr = max(1e-6, lr * 0.9995)
    
    final_h = best_fine_h.copy()
    final_c5 = best_c5_fine
    
    # Return the final result
    return (final_h.tolist(), final_c5, fine_n)
