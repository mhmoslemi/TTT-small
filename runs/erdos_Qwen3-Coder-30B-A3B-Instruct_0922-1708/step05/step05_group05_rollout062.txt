import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.001
    perturbation_freq = 50
    perturbation_magnitude = 0.1
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = 1/2
    h = h * (0.5 / np.sum(h))
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation and find minimum
        corr = signal.correlate(h, 1 - h, mode='full')
        min_k = np.argmin(corr)
        min_val = corr[min_k]
        
        # Compute gradient w.r.t. h for the minimum value
        # The gradient at index j for the minimum correlation at offset k
        # is derived from the derivative of the convolution
        grad = np.zeros(n_points)
        
        # For correlation at k, sum over all j where h[j] * (1-h[j+k]) contributes
        # When h is shifted by k, we get h[j] * (1-h[j+k])
        # So gradient of h[j] = (1-h[j+k]) + (1-h[j-k]) 
        # But we want the minimum: we look at which h_j contributes to minimum
        # We compute how the contribution changes when we modify h_j
        k_offset = min_k - (n_points - 1)  # Convert to actual lag
        
        # Compute finite difference gradient
        eps = 1e-5
        base_corr = corr.copy()
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (0.5 / np.sum(h_plus))
            corr_plus = signal.correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[min_k] - base_corr[min_k]) / eps
        
        # Gradient descent step (we are minimizing the max overlap, so we use negative gradient)
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h * (0.5 / np.sum(h))
        
        # Periodic perturbations to avoid local minima
        if iteration % perturbation_freq == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 / (1.0 + iteration / 100.0)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (0.5 / np.sum(h))
        
        # Check if we improved
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = signal.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
