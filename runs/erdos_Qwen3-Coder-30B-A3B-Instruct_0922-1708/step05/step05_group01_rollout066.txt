import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 0.001
    penalty_factor = 1.2
    perturbation_frac = 0.01
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Binary decision variables
    z = (h > 0.5).astype(float)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Penalty parameter
    mu = 0.1
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap integral
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximizing lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        # Gradient w.r.t. h for the maximizing lag
        gradient = np.zeros(n_points)
        if max_lag >= 0:
            for i in range(n_points - max_lag):
                gradient[i] += (1 - h[i + max_lag] - h[i])
        else:
            for i in range(-max_lag, n_points):
                gradient[i] += (1 - h[i + max_lag] - h[i])
        
        # Update h based on gradient
        h_new = h - learning_rate * gradient
        
        # Apply penalty to encourage binary behavior
        penalty_grad = mu * (h_new - z)
        h_new -= learning_rate * penalty_grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z based on new h
        z = (h_new > 0.5).astype(float)
        
        h = h_new
        
        # Update penalty parameter
        mu *= penalty_factor
        
        # Occasionally perturb to escape local minima
        if iteration % 100 == 0:
            perturb = np.random.uniform(-perturbation_frac, perturbation_frac, n_points)
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
