import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate a levy flight step."""
    # Generate levies using stable distribution approximation
    # For simplicity, using normal distribution but with scale adjustment
    return np.random.laplace(0, scale, size)

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Add 49 more random initial particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Keep track of best solutions
    best_h = particles[0].copy()
    best_value = evaluate_h(best_h, dx)
    
    # Main loop
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate all particles
        values = [evaluate_h(p, dx) for p in particles]
        # Find best current
        best_idx_in_particles = np.argmin(values)
        if values[best_idx_in_particles] < best_value:
            best_h = particles[best_idx_in_particles].copy()
            best_value = values[best_idx_in_particles]
        
        # Generate new candidates using Lévy flight
        new_particles = []
        for p in particles:
            # Apply levy flight
            step = levy_flight(n_points, alpha=1.5, scale=0.5)
            new_p = p + step
            
            # Repair and normalize
            new_p = repair_h(new_p, n_points)
            new_particles.append(new_p)
        
        # Add gradient descent refinement on top performers
        top_k = min(10, len(new_particles))
        top_indices = np.argsort(values)[:top_k]
        
        for idx in top_indices:

            # Gradient descent

            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        final_h = final_h - 0.0002 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')

    return (final_h.tolist(), best_value, n_points)
