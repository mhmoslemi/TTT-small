import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial population
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    fitness_scores = [compute_c5(h) for h in population]
    
    # Track best solution
    best_idx = np.argmin(fitness_scores)
    best_h = population[best_idx].copy()
    best_c5 = fitness_scores[best_idx]
    
    # DE main loop
    current_iter = 0
    stagnation_counter = 0
    stagnation_threshold = 50
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Create new population
        new_population = []
        new_fitness = []
        
        for i in range(population_size):
            # Select three distinct individuals
            idxs = np.random.choice(population_size, 3, replace=False)
            x1, x2, x3 = population[idxs[0]], population[idxs[1]], population[idxs[2]]
            
            # Mutation
            mutant = x1 + mutation_factor * (x2 - x3)
            mutant = np.clip(mutant, 0, 1)
            
            # Repair to satisfy constraint sum(h) = n_points/2
            sum_h = np.sum(mutant)
            if abs(sum_h - n_points/2) > 1e-10:
                if sum_h > n_points/2:
                    # Reduce to meet constraint
                    diff = sum_h - n_points/2
                    indices = np.argsort(mutant)[::-1]
                    for idx in indices:
                        if diff <= 0: break
                        dec = min(mutant[idx], diff)
                        mutant[idx] -= dec
                        diff -= dec
                else:
                    # Increase to meet constraint
                    diff = n_points/2 - sum_h
                    indices = np.argsort(mutant)
                    for idx in indices:
                        if diff <= 0: break
                        inc = min(1 - mutant[idx], diff)
                        mutant[idx] += inc
                        diff -= inc
                # Scale to make sum exactly n_points/2
                mutant = mutant * (n_points/2) / np.sum(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Evaluate trial vector
            c5_trial = compute_c5(trial)
            
            # Selection
            if c5_trial < fitness_scores[i]:
                new_population.append(trial)
                new_fitness.append(c5_trial)
            else:
                new_population.append(population[i])
                new_fitness.append(fitness_scores[i])
        
        # Update population
        population = new_population
        fitness_scores = new_fitness
        
        # Update best solution
        current_best_idx = np.argmin(fitness_scores)
        if fitness_scores[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitness_scores[current_best_idx]
            stagnation_counter = 0
        else:
            stagnation_counter += 1
        
        # Decrease mutation factor
        if current_iter % 50 == 0 and current_iter > 0:
            mutation_factor *= 0.99
            
        # Handle stagnation
        if stagnation_counter >= stagnation_threshold:
            # Add some diversity
            for _ in range(population_size // 10):
                idx = np.random.randint(0, population_size)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                population[idx] = h
        
        current_iter += 1
    
    # Final refinement with gradient-based method
    def refine_solution(h_start):
        h_refined = h_start.copy()
        learning_rate = 0.001
        
        for _ in range(100):
            # Compute derivatives
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            base_c5 = np.max(corr) * dx
            
            grad = np.zeros(n_points)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                sum_h = np.sum(h_plus)
                h_plus = h_plus * (n_points/2) / sum_h
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - base_c5) / eps
            
            # Update with gradient
            h_refined -= learning_rate * grad
            
            # Repair
            h_refined = np.clip(h_refined, 0, 1)
            sum_h = np.sum(h_refined)
            if abs(sum_h - n_points/2) > 1e-10:
                h_refined = h_refined * (n_points/2) / sum_h
                
            # Check for improvement
            new_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            new_c5 = np.max(new_corr) * dx
            if new_c5 >= base_c5:
                break
                
        return h_refined
    
    # Refine best solution
    best_h = refine_solution(best_h)
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
