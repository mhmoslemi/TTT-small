import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimizer parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    perturbation_fraction = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlations for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        if max_lag > 0:
            # h[i] * (1 - h[i + max_lag]) overlaps
            for i in range(len(h) - max_lag):
                grad[i] += 1 - h[i + max_lag]
                grad[i + max_lag] -= h[i]
        elif max_lag < 0:
            # h[i + |max_lag|] * (1 - h[i]) overlaps
            for i in range(-max_lag, len(h)):
                grad[i] -= h[i + max_lag]
                grad[i + max_lag] += 1 - h[i]
        else:
            # h[i] * (1 - h[i]) overlaps
            for i in range(len(h)):
                grad[i] += 1 - 2 * h[i]
                
        # Sample gradient for monitoring
        if iteration % 500 == 0:
            grad_norm = np.linalg.norm(grad)
            if grad_norm < grad_threshold:
                # Add random perturbation to escape local minima
                noise = np.random.normal(0, perturbation_fraction, n_points)
                h = np.clip(h + noise, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                alpha = 1e-3  # Reset learning rate
                continue
        
        # Line search
        step_size = alpha
        while True:
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < c5:
                h = h_new
                alpha = min(alpha * 1.2, 1e-1)  # Increase learning rate
                break
            else:
                step_size *= line_search_factor
                if step_size < 1e-10:
                    break
        
        # Occasionally reset learning rate
        if iteration % 1000 == 0 and iteration > 0:
            alpha = 1e-3
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
