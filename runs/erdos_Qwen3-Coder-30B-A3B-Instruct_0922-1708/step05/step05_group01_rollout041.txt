import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 5000
    learning_rate = 0.001
    penalty_factor = 1.2
    repair_fraction = 0.01
    μ = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    z = (h > 0.5).astype(int)
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Find the maximum lag
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient
        # We want to compute gradient of overlap with respect to h
        # The overlap at lag k is sum h[i] * (1 - h[i+k])
        # So gradient w.r.t. h[i] is (1 - h[i+k] - h[i]) if k = max_lag
        # or, more precisely, what the derivative of max over all lags is
        
        # Get window for maximum lag
        h_shifted = np.roll(h, max_lag)
        gradient = (1 - h_shifted - h)
        
        # Gradient descent step on h
        h_new = h - learning_rate * gradient
        
        # Repair step
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum - n_points/2) > 1e-8:
            # Rescale to correct sum
            h_new = h_new * (n_points / 2) / h_sum
            
        h = h_new
        z = (h > 0.5).astype(int)
        
        # Update penalty parameter
        μ *= penalty_factor
        
        # Check if improved
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Random perturbation occasionally
        if iteration % 100 == 0:
            perturb_indices = np.random.choice(n_points, size=int(n_points * repair_fraction), replace=False)
            h[perturb_indices] = np.random.rand(len(perturb_indices))
            h = h * (n_points / 2) / np.sum(h)
            z = (h > 0.5).astype(int)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
