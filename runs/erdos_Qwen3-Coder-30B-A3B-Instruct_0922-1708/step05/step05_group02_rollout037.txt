import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, n_points, mutation_rate=0.15):
    """Generate a mutated version of h."""
    h_new = h.copy()
    n_mutate = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_new[i] = 1 - h_new[i]
    return repair_h(h_new, n_points)

    else:
        # Initialize with random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    patience = 0
    max_patience = 100
    
    # Mutation-based search
    n_mutations = 100
    mutation_rate = 0.15

        # Generate mutations
        candidates = []

            h_mut = mutate_h(best_h, n_points, mutation_rate)
            c5 = evaluate_h(h_mut, dx)
            candidates.append((h_mut, c5))

        candidates.sort(key=lambda x: x[1])
        best_candidate = candidates[0]

            # Reduce mutation rate if stuck
            mutation_rate = min(0.5, mutation_rate * 1.1)
            patience = 0
            # Random restart with new random solution
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_c5 = evaluate_h(h_new, dx)
            if new_c5 < best_c5:

    # Final gradient descent refinement
    h_refined = best_h.copy()
    prev_c5 = best_c5

        if time.time() - start_time > budget_s - 5:

        grad = np.zeros(n_points)

            h_plus[j] += eps
