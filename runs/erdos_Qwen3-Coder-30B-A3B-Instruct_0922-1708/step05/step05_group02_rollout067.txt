import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    return np.random.standard_cauchy(size) * scale

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles
    initial_candidates = 50
    top_candidates = 10
    max_iter = 1000
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles = [h_init.copy()]
    else:
        particles = []
    
    # Fill remaining particles with random feasible solutions
    while len(particles) < initial_candidates:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        
    # Initialize best solution
    best_h = None
    best_value = float('inf')
    
    for i, p in enumerate(particles):
        value = evaluate_h(p, dx)
        if value < best_value:
            best_value = value
            best_h = p.copy()
    
    # Perturbation parameters
    levy_alpha = 1.5
    levy_scale = 0.5
    perturbation_rate = 0.1
    learning_rate = 0.001
    
    # Main loop
    iter_count = 0
    no_improvement_count = 0
    last_best = best_value
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        new_particles = []
        
        # Generate new candidates using Lévy flights
        for p in particles:
            # Generate perturbation
            pert = levy_flight(n_points, levy_alpha, levy_scale)
            perturbed = p + pert * perturbation_rate
            perturbed = repair_h(perturbed, n_points)
            new_particles.append(perturbed)
            
        # Evaluate new candidates
        particle_values = []
        for p in new_particles:
            value = evaluate_h(p, dx)
            particle_values.append(value)
            if value < best_value:
                best_value = value
                best_h = p.copy()
                
        # Update particles with top candidates + gradient descent
        indices = np.argsort(particle_values)[:top_candidates]
        for idx in indices:
            p = new_particles[idx].copy()
            current_value = particle_values[idx]
            
            # Simple gradient descent for local refinement
            for _ in range(10):
                if time.time() - start_time > budget_s - 10:
                    break
                eps = 1e-6
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = p.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_value) / eps
                    
                p = p - learning_rate * grad
                p = repair_h(p, n_points)
                corr = correlate(p, 1 - p, mode='full')
                current_value = np.max(corr) * dx
                
                if current_value < best_value:
                    best_value = current_value
                    best_h = p.copy()
                    
        # Update particles list
        particles = new_particles
        
        # Check for stagnation
        if abs(best_value - last_best) < 1e-8:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        last_best = best_value
        
        if no_improvement_count > 100:
            break
            
    # Final intensive gradient descent on best solution
    if best_h is not None:
        h_refined = best_h.copy()
        current_value = best_value
        
        for _ in range(1000):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h_refined.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            h_refined = h_refined - 0.0001 * grad
            h_refined = repair_h(h_refined, n_points)
            corr = correlate(h_refined, 1 - h_refined, mode='full')
            current_value = np.max(corr) * dx
            
            if current_value < best_value:
                best_value = current_value
                best_h = h_refined.copy()
    
    return (best_h.tolist(), best_value, n_points)
