import numpy as np
from scipy.optimize import minimize
import time

import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Initialize with a structured sequence - low discrepancy
        h = np.full(n_points, 0.5)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Use adaptive gradient descent with projecting
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Surrogate function for smooth max approximation (avoid non-differentiable max)
    def smooth_max(values, beta=10):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * values)))
        
    # Compute overlap with smooth approximation, adaptively
    def compute_overlap(h_func):
        correlation = np.correlate(h_func, 1 - h_func, mode='full') * dx
        return correlation
    
    def compute_c5(h_func):
        correlation = compute_overlap(h_func)
        return np.max(correlation)
        
    def compute_smooth_c5(h_func):
        correlation = compute_overlap(h_func)
        return smooth_max(correlation, beta=10)
        
    # Project back to feasible region: [0,1] and normalize sum
    def project_h(h_in):
        h_proj = np.clip(h_in, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj

    # Parameters
    max_iter = 10000
    patience = 0
    patience_limit = 1000
    lr = 0.01
    
    # Adaptive optimizer: Adam-like with learning rate schedule
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping condition
        if i > 100 and abs(current_c5 - best_c5) < 1e-6:
            patience += 1
            if patience > patience_limit:
                break
        else:
            patience = 0
            
        # Compute gradient using finite differences
        eps_val = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps_val
            h_plus = project_h(h_plus)
            
            c5_plus = compute_smooth_c5(h_plus)
            grad[j] = (c5_plus - compute_smooth_c5(h)) / eps_val

        # Adam-like update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        
        m_hat = m / (1 - beta1**(i+1))
        v_hat = v / (1 - beta2**(i+1))
        
        lr_t = lr / (np.sqrt(v_hat) + eps)
        
        h_new = h - lr_t * m_hat
        h_new = project_h(h_new)
        
        # Cosine decay of learning rate (first 1000 steps)
        if i < 1000:
            lr = 0.01 * 0.5 * (1 + np.cos(np.pi * i / 1000))
        
        h = h_new
        
    # Final check and return
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
