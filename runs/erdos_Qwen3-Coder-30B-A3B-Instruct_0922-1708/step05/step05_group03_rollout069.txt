import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Ensure the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing
    T0 = 10.0
    cooling_rate = 1.0
    max_iterations = int(400000 / n_points)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Temperature schedule
        T = T0 / np.log(1 + iteration)
        
        # Perturb
        h_candidate = h + np.random.normal(0, 0.02, n_points)
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Evaluate
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        
        # Accept or reject
        delta = c5_candidate - best_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_candidate.copy()
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h.copy()
    
    # Gradient descent refinement
    h_ref = best_h.copy()
    lr = 0.001
    eps = 1e-4
    max_grad_iter = 50
    
    for _ in range(max_grad_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
            
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
