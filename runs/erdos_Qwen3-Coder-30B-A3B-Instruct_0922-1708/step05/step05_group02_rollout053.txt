import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_h = h_init.copy()
    else:
        # Initialize randomly
        current_h = np.random.rand(n_points)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
    
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # Parameters for simulated annealing
    initial_temp = 0.1 * current_c5 if current_c5 > 0 else 0.1
    temperature = initial_temp
    cooling_rate = 0.995
    max_iter = 10000
    perturb_fraction = 0.1
    
    # Early stopping
    stagnation_count = 0
    stagnation_limit = 100
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the solution
        old_h = current_h.copy()
        num_perturb = max(1, int(n_points * perturb_fraction))
        perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Apply perturbations
        for i in perturb_indices:
            current_h[i] += np.random.uniform(-0.1, 0.1)
        
        # Repair to maintain constraints
        current_h = repair_h(current_h, n_points)
        
        # Evaluate new value
        new_c5 = evaluate_h(current_h, dx)
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5:
            current_c5 = new_c5
        else:
            # Accept with probability based on temperature
            delta = new_c5 - current_c5
            if np.random.rand() < np.exp(-delta / temperature):
                current_c5 = new_c5
            else:
                current_h = old_h.copy()
        
        # Update best solution
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Cooling and adaptive changes
        temperature *= cooling_rate
        if stagnation_count > stagnation_limit:
            temperature *= 0.9
            stagnation_count = 0
            
        # Progress tracking
        if iteration % 50 == 0 and iteration > 0:
            if stagnation_count > stagnation_limit:
                break
    
    # Final gradient descent refinement
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), best_c5, n_points)
