import numpy as np
from scipy.optimize import minimize
from scipy.optimize import Bounds
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    local_search_freq = 50
    rejuvenation_rate = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    correlation = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Gradient computation
    def compute_gradient(h):
        corr = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        max_val = corr[max_idx]
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[i] = (np.max(corr_plus) - max_val) / eps
        return grad
    
    # Optimization loop
    current_iter = 0
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Local descent with gradient
        h_prev = h.copy()
        for _ in range(10):
            if time.time() - start_time > budget_s - 10:
                break
            grad = compute_gradient(h)
            h -= 0.001 * grad
            h = repair(h)
            
            # Check if we made improvement
            corr = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr) * dx
            if current_c5 >= best_c5:
                h = h_prev.copy()
                break
            else:
                best_c5 = current_c5
                best_h = h.copy()
        
        # Rejuvenation
        if current_iter % 100 == 0 and current_iter > 0:
            if np.random.random() < rejuvenation_rate:
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
        
        current_iter += 1
    
    # Final local optimization
    def objective(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint}
    bounds = Bounds(np.zeros(n_points), np.ones(n_points))
    
    h_opt = h.copy()
    if time.time() - start_time < budget_s - 10:
        try:
            result = minimize(objective, h_opt, method='SLSQP', bounds=bounds, constraints=cons, 
                            options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
            if result.success:
                h_opt = result.x
                h_opt = repair(h_opt)
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                final_c5 = -objective(h_opt)
                if final_c5 < best_c5:
                    best_c5 = final_c5
                    best_h = h_opt
        except:
            pass
    
    # Final check
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
