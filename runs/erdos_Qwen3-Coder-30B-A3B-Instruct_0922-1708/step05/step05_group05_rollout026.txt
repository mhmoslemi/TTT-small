import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    start_time = time.time()
    
    # Gradient-based optimization
    max_iter = 5000
    alpha = 0.01
    decay_rate = 0.95
    perturb_freq = 50
    perturb_magnitude = 0.05
    min_alpha = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current correlation and gradient estimate
        corr = np.correlate(h, 1 - h, mode="full")
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # Estimating gradient via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            fitness_plus = np.max(corr_plus) * dx
            
            grad[i] = (fitness_plus - current_c5) / eps
        
        # Update with gradient descent
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < current_c5:
            h = h_new.copy()
            # Increase learning rate slightly if successful
            alpha = min(alpha * 1.05, 0.1)
        else:
            # Decrease learning rate if failed
            alpha *= decay_rate
            if alpha < min_alpha:
                alpha = min_alpha
        
        # Occasionally apply perturbations to escape local minima
        if iteration % perturb_freq == 0 and iteration > 0:
            perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode="full")
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
