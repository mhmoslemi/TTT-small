import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Track best solution
    best_h = h.copy()
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    # Gradient descent parameters
    learning_rate = 0.001
    max_iter = 5000
    μ = 0.1
    μ_factor = 1.2
    
    # Optimization loop
    for i in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Evaluate current best
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the maximum lag
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) // 2)
        
        # Compute gradient
        grad = np.zeros(n_points)
        if max_lag >= 0:
            # Shift h by max_lag
            h_shifted = np.concatenate([h[max_lag:], np.zeros(max_lag)])
        else:
            # Negative lag
            h_shifted = np.concatenate([np.zeros(-max_lag), h[:n_points + max_lag]])
        
        # Actual gradient computation
        grad = (1 - h_shifted - h) * 2 * dx
        
        # Update h
        h_new = h - learning_rate * grad
        
        # Repair to [0,1] range and ensure sum = n_points/2
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        target_sum = n_points / 2
        h_new = h_new * target_sum / h_sum
        
        h = h_new
        
        # Adjust penalty term
        μ *= μ_factor
    
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
