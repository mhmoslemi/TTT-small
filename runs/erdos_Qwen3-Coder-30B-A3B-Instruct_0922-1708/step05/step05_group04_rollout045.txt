import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    max_iter = 500
    lr = 0.01  # Learning rate
    eps = 1e-4  # Gradient estimation step
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        h_current = h_init.copy()
    else:
        h_current = np.random.uniform(0, 1, n_points)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
    
    # Compute initial C5
    corr = np.correlate(h_current, 1 - h_current, mode='full')
    c5_current = np.max(corr) * dx
    best_c5 = c5_current
    best_h = h_current.copy()
    
    # Fourier representation
    def h_from_fourier(coeffs, n_points):
        x = np.linspace(0, 2, n_points)
        h = coeffs[0] / 2.0 * np.ones_like(x)
        for k in range(1, min(len(coeffs), n_fourier)):
            h += coeffs[k] * np.cos(np.pi * k * x)  # Cosine terms
            if k < len(coeffs) - 1:
                h += coeffs[k + 1] * np.sin(np.pi * k * x)  # Sine terms
        return np.clip(h, 0, 1)
    
    def project_h_to_constraint(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Gradient computation function
    def compute_gradient(coeffs, h_original):
        current_h = h_from_fourier(coeffs, n_points)
        current_h = project_h_to_constraint(current_h)
        corr = np.correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        grad = np.zeros_like(coeffs)
        for i in range(len(coeffs)):
            h_plus = coeffs.copy()
            h_plus[i] += eps
            h_plus_h = h_from_fourier(h_plus, n_points)
            h_plus_h = project_h_to_constraint(h_plus_h)
            corr_plus = np.correlate(h_plus_h, 1 - h_plus_h, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - current_c5) / eps
        return grad
    
    # Optimization
    current_coeffs = np.random.randn(n_fourier)
    stagnation_count = 0
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Compute gradient
        grad = compute_gradient(current_coeffs, h_current)
        
        # Update coefficients
        current_coeffs -= lr * grad
        
        # Reconstruct h
        h_reconstructed = h_from_fourier(current_coeffs, n_points)
        h_reconstructed = project_h_to_constraint(h_reconstructed)
        
        # Compute C5
        corr = np.correlate(h_reconstructed, 1 - h_reconstructed, mode='full')
        c5_reconstructed = np.max(corr) * dx
        
        # Update best solution
        if c5_reconstructed < best_c5:
            best_c5 = c5_reconstructed
            best_h = h_reconstructed.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Early stopping
        if stagnation_count > 100:
            break
            
        current_iter += 1
    
    return (best_h.tolist(), best_c5, n_points)
