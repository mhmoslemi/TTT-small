import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    learning_rate = 0.01
    decay_factor = 0.995
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Check if this is the best solution so far
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        # Estimate gradient with finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            # Ensure sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
        
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Apply random perturbation periodically
        if iteration % perturb_freq == 0:
            perturb_magnitude_t = perturb_magnitude / (1.0 + iteration / 100.0)
            perturb = np.random.normal(0, perturb_magnitude_t, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adapt learning rate
        learning_rate *= decay_factor
    
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
