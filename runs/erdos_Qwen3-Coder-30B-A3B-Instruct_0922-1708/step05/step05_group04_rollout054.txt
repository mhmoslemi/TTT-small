import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    local_search_freq = 50
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for _ in range(99):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal bests and global best
    pbest_particles = [p.copy() for p in particles]
    pbest_fitness = [float('inf')] * 100
    
    # Compute initial fitness for all particles
    for i in range(100):
        corr = correlate(particles[i], 1 - particles[i], mode='full')
        fitness = np.max(corr) * dx
        pbest_fitness[i] = fitness
        if i == 0 or fitness < pbest_fitness[0]:
            gbest_idx = i
    gbest_particle = particles[gbest_idx].copy()
    gbest_fitness = pbest_fitness[gbest_idx]
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Optimization variables
    current_iter = 0
    stagnation_count = 0
    
    # Main loop
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # PSO update
        for p_idx in range(100):
            r1, r2 = np.random.rand(2)
            # PSO velocity update
            inertia_weight = 0.7
            phi_p = 2.0
            phi_g = 2.0
            velocities[p_idx] = (inertia_weight * velocities[p_idx] + 
                                phi_p * r1 * (pbest_particles[p_idx] - particles[p_idx]) +
                                phi_g * r2 * (gbest_particle - particles[p_idx]))
            
            # Position update
            particles[p_idx] += velocities[p_idx]
            
            # Repair
            particles[p_idx] = repair(particles[p_idx])
            
            # Fitness computation
            corr = correlate(particles[p_idx], 1 - particles[p_idx], mode='full')
            fitness = np.max(corr) * dx
            
            # Update pbest
            if fitness < pbest_fitness[p_idx]:
                pbest_fitness[p_idx] = fitness
                pbest_particles[p_idx] = particles[p_idx].copy()
                
                # Update gbest
                if fitness < gbest_fitness:
                    gbest_fitness = fitness
                    gbest_particle = particles[p_idx].copy()
                    stagnation_count = 0
        
        # Stagnation check and perturbation
        stagnation_count += 1
        if stagnation_count > 50:
            # Inject random particles (10%)
            for _ in range(10):
                idx = np.random.randint(0, 100)
                h = np.random.uniform(0, 1, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
        
        # Periodic local search (L-BFGS-B)
        if current_iter % local_search_freq == 0 and current_iter > 0:
            # Local search on best solutions
            sorted_indices = np.argsort([pbest_fitness[i] for i in range(100)])[:10]
            for idx in sorted_indices:
                h_local = pbest_particles[idx].copy()
                # Gradient descent refinement
                for _ in range(50):
                    corr = correlate(h_local, 1 - h_local, mode='full')
                    fitness = np.max(corr) * dx
                    
                    # Numerical gradient
                    grad = np.zeros(n_points)
                    eps = 1e-5
                    for i in range(n_points):
                        h_plus = h_local.copy()
                        h_plus[i] += eps
                        h_plus = repair(h_plus)
                        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                        fitness_plus = np.max(corr_plus) * dx
                        grad[i] = (fitness_plus - fitness) / eps
                    
                    # Update with small step size
                    step_size = 0.0001
                    h_local -= step_size * grad
                    h_local = repair(h_local)
                    
                    corr_new = correlate(h_local, 1 - h_local, mode='full')
                    fitness_new = np.max(corr_new) * dx
                    if fitness_new >= fitness:
                        break
                    else:
                        fitness = fitness_new
                
                # Update pbest if better
                corr_final = correlate(h_local, 1 - h_local, mode='full')
                fitness_final = np.max(corr_final) * dx
                if fitness_final < pbest_fitness[idx]:
                    pbest_fitness[idx] = fitness_final
                    pbest_particles[idx] = h_local.copy()
                    
                    if fitness_final < gbest_fitness:
                        gbest_fitness = fitness_final
                        gbest_particle = h_local.copy()
                        stagnation_count = 0
        
        current_iter += 1
    
    # Final refinement with L-BFGS
    best_h = gbest_particle.copy()
    best_c5 = gbest_fitness
    
    # L-BFGS refinement
    def objective(h):
        corr = correlate(h, 1 - h, mode='full')
        return -np.max(corr) * dx
    
    def constraint(h):
        return np.sum(h) - n_points / 2
    
    # Use best solution as initial guess
    result = minimize(objective, best_h, method='L-BFGS-B', 
                     bounds=[(0, 1) for _ in range(n_points)],
                     constraints={'type': 'eq', 'fun': constraint},
                     options={'maxiter': 100, 'ftol': 1e-6, 'gtol': 1e-6})
    
    if result.success:
        h_refined = result.x.copy()
        h_refined = repair(h_refined)
        corr_refined = correlate(h_refined, 1 - h_refined, mode='full')
        c5_refined = np.max(corr_refined) * dx
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = h_refined
    
    return (best_h.tolist(), best_c5, n_points)
