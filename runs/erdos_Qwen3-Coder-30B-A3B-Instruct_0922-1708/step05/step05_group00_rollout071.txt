import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 200  # Higher resolution for better precision
    dx = 2.0 / n_points
    n_start = 50  # Coarse grid size
    
    # Initialize using provided parent or structured approach
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        # Create a structured initialization - low discrepancy sequence
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points) + 0.2 * np.cos(4 * np.pi * i / n_points)
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Smooth maximum parameter for surrogate
    beta = 10.0
    
    def smooth_max(arr, beta):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))
    
    def compute_overlap(h, dx):
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    def compute_surrogate(h, dx):
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return smooth_max(correlation, beta)
    
    def project(h, n_points):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Coarse grid optimization
    coarse_h = h[::(n_points // n_start)]
    coarse_dx = 2.0 / n_start
    
    # Reduce to coarse resolution
    coarse_h = project(coarse_h, n_start)
    coarse_best_h = coarse_h.copy()
    coarse_best_c5 = float('inf')
    
    # Use Adam optimizer with cosine decay for coarse grid
    lr = 0.01
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    m_t, v_t = 0, 0
    
    coarse_iterations = 2000
    for t in range(coarse_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute surrogate and gradient with finite differences
        current_value = compute_surrogate(coarse_h, coarse_dx)
        
        # Gradient via finite differences
        grad = np.zeros_like(coarse_h)
        eps = 1e-5
        
        for i in range(len(coarse_h)):
            h_plus = coarse_h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus, n_start)
            value_plus = compute_surrogate(h_plus, coarse_dx)
            grad[i] = (value_plus - current_value) / eps
        
        # Adam step
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad ** 2
        m_hat = m_t / (1 - beta1 ** (t + 1))
        v_hat = v_t / (1 - beta2 ** (t + 1))
        
        # Update
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        coarse_h -= update
        coarse_h = project(coarse_h, n_start)
        
        # Check for better solution
        current_c5 = compute_overlap(coarse_h, coarse_dx)
        if current_c5 < coarse_best_c5:
            coarse_best_c5 = current_c5
            coarse_best_h = coarse_h.copy()
    
    # Transfer coarse solution to fine grid
    fine_h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), coarse_best_h)
    fine_h = project(fine_h, n_points)
    
    # Fine grid optimization with Adam
    fine_best_h = fine_h.copy()
    fine_best_c5 = float('inf')
    
    # Reset Adam moments
    m_t, v_t = 0, 0
    lr = 0.01
    
    fine_iterations = 2000
    for t in range(fine_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_value = compute_surrogate(fine_h, dx)
        
        # Gradient via finite differences
        grad = np.zeros_like(fine_h)
        eps = 1e-5
        
        for i in range(len(fine_h)):
            h_plus = fine_h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus, n_points)
            value_plus = compute_surrogate(h_plus, dx)
            grad[i] = (value_plus - current_value) / eps
        
        # Adam step
        m_t = beta1 * m_t + (1 - beta1) * grad
        v_t = beta2 * v_t + (1 - beta2) * grad ** 2
        m_hat = m_t / (1 - beta1 ** (t + 1))
        v_hat = v_t / (1 - beta2 ** (t + 1))
        
        # Update with cosine annealing
        lr = 0.01 * (0.5 * (1 + np.cos(np.pi * t / fine_iterations)))
        update = lr * m_hat / (np.sqrt(v_hat) + epsilon)
        fine_h -= update
        fine_h = project(fine_h, n_points)
        
        # Check for better solution
        current_c5 = compute_overlap(fine_h, dx)
        if current_c5 < fine_best_c5:
            fine_best_c5 = current_c5
            fine_best_h = fine_h.copy()
    
    # Final best solution
    final_h = fine_best_h.copy()
    final_c5 = compute_overlap(final_h, dx)
    
    # Ensure the final solution meets constraint
    final_h = project(final_h, n_points)
    final_c5 = compute_overlap(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
