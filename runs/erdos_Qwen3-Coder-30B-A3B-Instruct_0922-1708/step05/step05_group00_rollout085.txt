import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start strategy with diverse initializations
    initializations = []
    
    # 1. Use parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initializations.append(h.copy())
    
    # 2. Add several random restarts
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initializations.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # For each initialization, try convex relaxation + refinement
    for h in initializations:
        # Convex relaxation approach
        try:
            # Create a problem that approximates some relaxations
            h_var = cp.Variable(n_points)
            
            # Using relaxed sum constraint
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # Try solving via relaxation to get a good starting point
            prob = cp.Problem(cp.Minimize(cp.sum(h_var)), constraints)
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_relaxed = np.array(h_var.value)
                h = np.clip(h_relaxed, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
            
        # Refine with local optimization
        max_iter = 2000
        patience = 0
        last_c5 = float('inf')
        
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
                
            # Compute current C5
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            current_c5 = np.max(correlation)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            # Check convergence
            if abs(current_c5 - last_c5) < 1e-6:
                patience += 1
                if patience > 50:
                    break
            else:
                patience = 0
                
            last_c5 = current_c5
            
            # Compute gradient using finite differences
            grad = np.zeros_like(h)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                c5_plus = np.max(cor_plus)
                
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update h
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Decrease learning rate slowly
            lr = max(1e-6, lr * 0.9995)
    
    # Final validation
    if best_h is None:
        # Fallback to uniform if nothing was found
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
