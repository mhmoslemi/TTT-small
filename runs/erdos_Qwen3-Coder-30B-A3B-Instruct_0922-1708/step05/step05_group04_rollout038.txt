import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    F = 0.8
    CR = 0.9
    max_iter = 1000
    mutation_decay = 0.99
    local_search_freq = 50
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function to compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluate current population
    fitness = [compute_c5(p) for p in population]
    best_idx = np.argmin(fitness)
    gbest = population[best_idx].copy()
    gbest_fitness = fitness[best_idx]
    
    iteration = 0
    
    # DE main loop
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        
        for i in range(population_size):
            # Selection of three different individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Mutation
            mutant = population[a] + F * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Repair
            mutant = repair(mutant)
            
            # Crossover
            trial = []
            for j in range(n_points):
                if np.random.rand() < CR:
                    trial.append(mutant[j])
                else:
                    trial.append(population[i][j])
            trial = np.array(trial)
            
            # Repair trial vector
            trial = repair(trial)
            
            # Evaluation
            trial_fitness = compute_c5(trial)
            
            # Selection
            if trial_fitness < fitness[i]:
                new_population.append(trial)
                fitness[i] = trial_fitness
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update global best
        current_fitness = [compute_c5(p) for p in population]
        best_idx = np.argmin(current_fitness)
        if current_fitness[best_idx] < gbest_fitness:
            gbest = population[best_idx].copy()
            gbest_fitness = current_fitness[best_idx]
        
        # Adjust mutation factor
        if iteration % 50 == 0:
            F *= mutation_decay
            
        iteration += 1
    
    # Final refinement with gradient descent
    best_h = gbest.copy()
    best_c5 = gbest_fitness
    
    for _ in range(30):
        # Simple finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = compute_c5(best_h)
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[j] = (c5_plus - base_c5) / eps
        
        # Update with gradient descent
        best_h -= 0.001 * grad
        best_h = repair(best_h)
        
        new_c5 = compute_c5(best_h)
        if new_c5 >= best_c5:
            break
        best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
