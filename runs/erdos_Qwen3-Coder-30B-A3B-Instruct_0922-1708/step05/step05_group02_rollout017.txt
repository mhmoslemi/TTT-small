import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    # Generate Lévy flight steps
    sigma = scale / (alpha * (alpha - 1)) ** (1 / alpha)
    return np.random.standard_cauchy(size) * sigma

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initial solution
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Initialize population
    population_size = 50
    population = [h_init.copy()]
    fitness = [evaluate_h(h_init, dx)]
    
    # Add random solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
        fitness.append(evaluate_h(h, dx))
    
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    max_iter = 1000
    levy_alpha = 1.5
    levy_scale = 0.5
    learning_rate = 0.001
    perturb_rate = 0.1
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        if time.time() - start_time > budget_s - 10:
            break
            
        # Generate new candidates via Lévy flights
        new_population = []
        new_fitness = []
        
        for i in range(len(population)):
            h = population[i].copy()
            
            # Apply Lévy flight
            perturb = levy_flight(n_points, levy_alpha, levy_scale)
            h_new = h + perturb_rate * perturb
            
            # Repair
            h_new = repair_h(h_new, n_points)
            
            new_population.append(h_new)
            new_fitness.append(evaluate_h(h_new, dx))
        
        # Combine with existing population
        combined_population = population + new_population
        combined_fitness = fitness + new_fitness
        
        # Keep best half
        indices = np.argsort(combined_fitness)[:population_size]
        population = [combined_population[i] for i in indices]
        fitness = [combined_fitness[i] for i in indices]
        
        # Select top candidates for gradient updates
        top_k = min(10, len(population) // 2)
        if top_k > 0:
            indices = np.argsort(fitness)[:top_k]
            
            for idx in indices:
                h = population[idx].copy()
                current_c5 = fitness[idx]
                
                # Compute gradient
                grad = np.zeros(n_points)
                eps = 1e-6
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Update
                h = h - learning_rate * grad
                h = repair_h(h, n_points)
                
                new_c5 = evaluate_h(h, dx)
                if new_c5 < fitness[idx]:
                    population[idx] = h
                    fitness[idx] = new_c5
                    
                    if new_c5 < best_value:
                        best_h = h.copy()
                        best_value = new_c5
                        last_improvement = iter_count
        
        # Check for stagnation
        if iter_count - last_improvement > 100:
            # Inject diversity
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                population.append(h)
                fitness.append(evaluate_h(h, dx))
            
            # Keep best population_size
            indices = np.argsort(fitness)[:population_size]
            population = [population[i] for i in indices]
            fitness = [fitness[i] for i in indices]
            last_improvement = iter_count
            
        iter_count += 1
    
    # Final gradient refinement of best solution
    h_refined = best_h.copy()
    prev_value = best_value
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - prev_value) / eps
        
        h_refined = h_refined - 0.0001 * grad
        h_refined = repair_h(h_refined, n_points)
        prev_value = evaluate_h(h_refined, dx)
        
        if prev_value < best_value:
            best_h = h_refined.copy()
            best_value = prev_value
    
    return (best_h.tolist(), best_value, n_points)
