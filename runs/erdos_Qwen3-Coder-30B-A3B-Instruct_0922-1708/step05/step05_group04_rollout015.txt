import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from math import floor

def evaluate_c5(h, dx):
    """Evaluate C5 bound from h values"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def repair_h(h, n_points, dx):
    """Repair h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-10:
        # Adjust to meet integral constraint
        factor = (n_points / 2) / h_sum
        h = h * factor
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 200
    k_samples = 50
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Binary search for optimal C5
    low, high = 0.0, 0.5
    tol = 1e-5
    
    # Generate shift points
    k_grid = np.linspace(-2, 2, k_samples)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Try mid point
        mid = (low + high) / 2
        c5_target = mid
        
        # Simplified greedy approach with gradient estimation
        current_h = h.copy()
        for _ in range(50):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Estimate gradient
            grad = np.zeros(n_points)
            eps = 1e-4
            
            base_c5 = evaluate_c5(current_h, dx)
            
            for i in range(n_points):
                h_plus = current_h.copy()
                h_plus[i] += eps
                h_plus = repair_h(h_plus, n_points, dx)
                c5_plus = evaluate_c5(h_plus, dx)
                grad[i] = (c5_plus - base_c5) / eps
            
            # Gradient descent step
            current_h -= 0.005 * grad
            current_h = repair_h(current_h, n_points, dx)
            
            if evaluate_c5(current_h, dx) >= base_c5:
                break
                
        current_c5_bound = evaluate_c5(current_h, dx)
        
        if current_c5_bound < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5_bound
            
        if current_c5_bound < c5_target:
            high = c5_target
        else:
            low = c5_target
            
    # Final refinement
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = evaluate_c5(h_final, dx)
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points, dx)
            c5_plus = evaluate_c5(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        h_final -= 0.001 * grad
        h_final = repair_h(h_final, n_points, dx)
        if evaluate_c5(h_final, dx) >= base_c5:
            break
    
    final_c5 = evaluate_c5(h_final, dx)
    
    return (h_final.tolist(), final_c5, n_points)
