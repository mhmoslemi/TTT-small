import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Simulated Annealing parameters
    T = 10.0
    T_min = 1e-4
    alpha = 0.95
    max_iterations = 1000
    
    # Initialize best solution
    current_h = h.copy()
    current_corr = np.correlate(current_h, 1 - current_h, mode='full')
    current_c5 = np.max(current_corr) * dx
    best_c5 = current_c5
    best_h = current_h.copy()
    
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Annealing temperature
        if T > T_min:
            T *= alpha
            
        # Generate neighbor solution
        neighbor_h = current_h.copy()
        # Randomly pick a point to perturb
        idx = np.random.randint(0, n_points)
        # Small perturbation
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Ensure feasibility
        neighbor_h = np.clip(neighbor_h, 0, 1)
        sum_h = np.sum(neighbor_h)
        if abs(sum_h - n_points/2) > 1e-8:
            # Rescale to meet sum constraint
            neighbor_h = neighbor_h * (n_points / 2) / sum_h
            
        # Compute correlation
        neighbor_corr = np.correlate(neighbor_h, 1 - neighbor_h, mode='full')
        neighbor_c5 = np.max(neighbor_corr) * dx
        
        # Accept or reject
        if neighbor_c5 < current_c5:
            current_h = neighbor_h.copy()
            current_c5 = neighbor_c5
        else:
            # Accept with probability based on simulated annealing
            delta_e = neighbor_c5 - current_c5
            if np.random.rand() < np.exp(-delta_e / T):
                current_h = neighbor_h.copy()
                current_c5 = neighbor_c5
                
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()

        if iteration % 50 == 0 and iteration > 0:
            # Local refinement around current best
            h_ref = best_h.copy()
            lr = 0.001

                grad = np.zeros_like(h_ref)
                eps = 1e-4
                
                for i in range(n_points):
                    h_plus = h_ref.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - best_c5) / eps
                
                h_ref = h_ref - lr * grad
                h_ref = np.clip(h_ref, 0, 1)
                h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                
                corr = np.correlate(h_ref, 1 - h_ref, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_ref.copy()
    
    # Final local optimization
    # Local refinement of best solution
    h_ref = best_h.copy()
    lr = 0.0005
    for _ in range(100):
        grad = np.zeros_like(h_ref)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_ref.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
        
        h_ref = h_ref - lr * grad
        h_ref = np.clip(h_ref, 0, 1)
        h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
        
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_ref.copy()
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
