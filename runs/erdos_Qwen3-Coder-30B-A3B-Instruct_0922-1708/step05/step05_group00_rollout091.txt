import numpy as np
from scipy.signal import correlate
import time
import cvxpy as cp
from cvxpy import GLPK

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        if n_points < 100:
            # Resample to 100 points
            x_orig = np.linspace(0, 2, n_points)
            x_new = np.linspace(0, 2, 100)
            h = np.interp(x_new, x_orig, h)
            n_points = 100
    else:
        h = np.ones(100) * 0.5
        n_points = 100
    
    # Normalize to satisfy constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth maximum approximation parameter
    beta = 10.0
    dx = 2.0 / n_points
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Time management
    max_iter = 5000
    patience = 0
    max_patience = 500
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap correlation
        correlation = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        # Update best if improved
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        
        # Compute gradient with finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(cor_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam-like update with adaptive learning rate
        lr = 0.005
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Reduce learning rate
        lr = max(1e-6, lr * 0.9995)
    
    # Final verification
    final_c5 = np.max(correlate(best_h, 1 - best_h, mode='full')) * dx
    
    return (best_h.tolist(), final_c5, n_points)
