import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Evaluate initial solution
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    
    # Gradient descent optimization
    lr = initial_learning_rate
    last_improvement = 0
    current_patience = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient for the maximizing lag
        grad = np.zeros(n_points)
        lag = max_lag_idx - (n_points - 1)
        
        # The gradient for correlation at lag k is:
        # d/dh (h * (1-h))[k] = (1-h) shift(-k) - h shift(-k) 
        # which equals (1-h) * I_{k} - h * I_{k} where I_k indicates overlap
        # But we compute an approximate version:
        # Gradient approximated by finite differences over convolution
        # For deriving exact form we treat convolution as linear:
        # grad = 2*(1 - 2*h) * delta_k + (1 - h) * delta_k - h * delta_k
        # actually, the direct computation using convolution properties.
        
        # Here we compute numerical gradient for the maximum lag:
        # Finding the actual gradient contribution for the corresponding lag:
        # For lag k in full convolution, we compute gradient at k:
        # if k is lag where max is, then current corr[k] = sum h[i] * (1-h[i+k])
        # d/dh_i [h[i] * (1-h[i+k])] = 1 - h[i+k] - h[i]
        # Take this as gradient approximation for max lag = k
        
        # approx direct gradient:
        # d/dh_i [sum h[i] * (1-h[i+k])] = [1 - h[i+k] - h[i]] for all i
        # Let's compute a full gradient
        # Use indicator vectors for clarity:
        # For lag k: corresponds to candidate of h[i] ...
        # (i,i+k) elements of correlation
        # Gradient d/dh_i (h[i] * (1-h[i+k])) = 1 - h[i+k] - h[i]
        
        # Naive approach:
        # Calculate contribution of all terms to gradient for this lag
        # For landk k:
        #  d/dh_i [h[i] * (1 - h[i+k])] = 1 - h[i+k] - h[i] (approximately)
        
        # Most accurate gradient via correlate trick:
        # For 1-h, shift by -k; for h shift by -k
        # get masks and sum over correct window
        # Manually construct gradient for max_lag
        h_shifted = np.zeros_like(h)
        if lag < 0:
            h_shifted[:lag] = h[-lag:]
        elif lag > 0:
            h_shifted[lag:] = h[:-lag]
        else:
            h_shifted = h.copy()
        ones_minus_h_shifted = 1 - h_shifted
        grad = (ones_minus_h_shifted - h)
        
        # Take gradient step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to enforce constraint sum(h) = n_points/2
        h_new_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_new_sum
        
        h = h_new
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            current_patience = 0
        else:
            current_patience += 1
            if current_patience > plateau_patience:
                lr *= learning_rate_decay
                lr = max(lr, min_learning_rate)
                if lr <= min_learning_rate:
                    # Perturb slightly to escape local min or plateau
                    perturb = np.random.uniform(-perturb_fraction, perturb_fraction, n_points)
                    h = np.clip(h + perturb, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    current_patience = 0
                
    return (best_h.tolist(), best_c5, n_points)
