from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):

    # Start with a coarse grid to avoid local minima
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, 100), h)
    
    # Optimization using CVXPY quadratic programming
    # Ballpark optimization for better solution
    
    # Create a reasonable initial step function
    h_coarse = np.array([0.0, 0.0, 0.0, 0.5, 0.5, 0.5, 0.5, 0.0, 0.0, 0.0] * 5)
    h_coarse = np.clip(h_coarse, 0, 1)
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    
    # Refine using gradient descent
    max_iter = 20000
    lr_schedule = np.concatenate([np.linspace(0.05, 0.0005, 10000),

    patience = 0
    best_c5 = float('inf')
    best_h = h_coarse.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Current correlation
        corr = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_coarse
        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_coarse.copy()
            
        # Early stopping
        if i > 100 and abs(corr[-1] - corr[0]) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0

        grad = np.zeros_like(h_coarse)
        eps = 1e-5

        lr = lr_schedule[i] if i < len(lr_schedule) else lr_schedule[-1]
        h_new = h_coarse - lr * grad
        h_new = np.clip(h_new, 0, 1)

    max_iter = 20000
    lr_schedule = np.concatenate([np.linspace(0.05, 0.0005, 10000),

    best_c5 = float('inf')

    for i in range(max_iter):

        current_c5 = np.max(corr)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()

        if i > 100 and np.abs(np.max(corr) - np.max(np.correlate(best_h, 1 - best_h, mode='full') * dx)) < 1e-8:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        # Gradient approximation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for j in range(100):
            h_plus = h.copy()

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (100 / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            grad[j] = (np.max(corr_plus) - current_c5) / eps
            
        # Update
        lr = lr_schedule[i] if i < len(lr_schedule) else lr_schedule[-1]
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (100 / 2) / np.sum(h_new)
        h = h_new
        
    # Final validation
    corr_final = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(corr_final)
    
    # Ensure sum constraint
    sum_constraint = np.sum(best_h) * dx
    if abs(sum_constraint - 1.0) > 1e-6:
        best_h = best_h * (100 / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, 100)
