import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 1000
    mutation_factor = 0.8
    crossover_prob = 0.9
    mutation_decay = 0.99
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Function to repair solutions
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluate fitness
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main DE loop
    generation = 0
    best_fitness = float('inf')
    best_individual = None
    
    while generation < max_generations and time.time() - start_time < budget_s - 5:
        # Evaluate all individuals
        fitnesses = [evaluate_fitness(p) for p in population]
        
        # Track best
        for i, fitness in enumerate(fitnesses):
            if fitness < best_fitness:
                best_fitness = fitness
                best_individual = population[i].copy()
        
        # Create new population
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            indices = list(range(population_size))
            indices.remove(i)
            a, b, c = np.random.choice(indices, 3, replace=False)
            
            # Mutation
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            mutant = repair(mutant)
            
            # Crossover
            trial = population[i].copy()
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Repair trial if needed
            trial = repair(trial)
            
            # Selection
            if evaluate_fitness(trial) < fitnesses[i]:
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Adjust mutation factor
        if generation % 50 == 0:
            mutation_factor *= mutation_decay
            
        generation += 1
    
    # Final refinement with gradient descent
    if best_individual is not None:
        h_ref = best_individual.copy()
        for _ in range(100):
            fitness = evaluate_fitness(h_ref)
            grad = np.zeros(n_points)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = repair(h_plus)
                fitness_plus = evaluate_fitness(h_plus)
                grad[j] = (fitness_plus - fitness) / eps
            
            h_ref -= 0.001 * grad
            h_ref = repair(h_ref)
            
            # Check for convergence
            corr_current = np.correlate(h_ref, 1 - h_ref, mode='full')
            current_fitness = np.max(corr_current) * dx
            if current_fitness >= fitness:
                break
            fitness = current_fitness
        
        # Final evaluation
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        final_c5 = np.max(corr) * dx
        
        return (h_ref.tolist(), final_c5, n_points)
    
    return (population[0].tolist(), evaluate_fitness(population[0]), n_points)
