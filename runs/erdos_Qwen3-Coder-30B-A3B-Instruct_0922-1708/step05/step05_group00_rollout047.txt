            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points)

    h = np.clip(h, 0, 1)

    best_h = h.copy()
    best_c5 = float('inf')
    
    # First attempt with coarse grid
    n_coarse = 50

    h_coarse = np.interp(np.linspace(0, 2, n_coarse),

    # Optimize coarse grid
    def smooth_max(x, beta=10):
        return np.log(np.sum(np.exp(beta * x))) / beta
    
    opt_h = h_coarse.copy()
    lr = 0.01
    patience = 0
    last_c5 = float('inf')
    
    for step in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlap
        overlap = np.correlate(opt_h, 1 - opt_h, mode='full') * dx_coarse

            best_c5 = c5
            best_h = opt_h.copy()

            patience += 1
            if patience > 200:
                break

            patience = 0

    best_h = best_h * (n_points / 2) / np.sum(best_h)
