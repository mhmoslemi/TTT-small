import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx].copy()

def crossover(parent1, parent2, alpha=0.8):
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.1, mutation_strength=0.03):
    mask = np.random.random(len(h)) < mutation_rate
    h += np.random.normal(0, mutation_strength, len(h)) * mask
    return h

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    h = h.copy()
    for _ in range(max_iter):
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            fitness_plus = compute_fitness(h_plus, dx)
            grad[i] = (fitness_plus - compute_fitness(h, dx)) / eps
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        if np.allclose(grad, 0, atol=1e-6):
            break
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2.0) / np.sum(h_parent)
        # Create 50% variations around parent + 50% random
        for _ in range(population_size // 2):
            h = h_parent + np.random.normal(0, 0.03, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Initialize with random valid solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)

    # Evaluate initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]

    # Main GA loop
    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < 1000:
        generation += 1
        
        # Create new population
        new_population = []
        
        # Elitism: keep best solutions
        elite_idx = np.argsort(fitnesses)[:population_size // 10]
        for i in elite_idx:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                child1, child2 = crossover(parent1, parent2, alpha=0.85)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1, mutation_rate=0.1, mutation_strength=0.03)
            child2 = mutate(child2, mutation_rate=0.1, mutation_strength=0.03)
            
            # Projection to feasible region
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Periodic gradient refinement on top individuals
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for i in top_indices:
                refined = gradient_refinement(population[i], dx, max_iter=30)
                refined_fitness = compute_fitness(refined, dx)
                if refined_fitness < fitnesses[i]:
                    population[i] = refined
                    fitnesses[i] = refined_fitness
                    if refined_fitness < best_c5:
                        best_c5 = refined_fitness
                        best_h = refined.copy()

    # Final local refinement on best solution
    final_h = gradient_refinement(best_h, dx, max_iter=100)
    final_c5 = compute_fitness(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
