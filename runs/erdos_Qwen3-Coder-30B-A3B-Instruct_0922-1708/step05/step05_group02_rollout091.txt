import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate random numbers from Lévy distribution
    sigma = scale * (np.random.gamma(1 + alpha, 1) / (np.random.gamma(1, 1) * np.random.beta(1, alpha)))**(1/alpha)
    return np.random.normal(0, sigma, size)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize individuals
    population_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
    
    # Add random initial individuals
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(ind, dx) for ind in population]
    
    # Track best solution
    best_idx = np.argmin(fitness)
    best_individual = population[best_idx].copy()
    best_fitness = fitness[best_idx]
    
    # Parameters
    max_iter = 1000
    levy_alpha = 1.5
    levy_scale = 0.5
    lr = 0.001
    elite_size = 10
    perturb_rate = 0.1
    
    # Evolution loop
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 5:
        iter_count += 1
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite_indices = sorted_indices[:elite_size]
        
        # Generate new population
        new_population = []
        new_population.extend([population[i].copy() for i in elite_indices])
        
        # Lévy flight and gradient-based exploration
        for _ in range(population_size - elite_size):
            # Select parent from elite
            parent_idx = np.random.choice(elite_indices)
            parent = population[parent_idx].copy()
            
            # Apply Lévy flight perturbation
            perturbation = levy_flight(n_points, levy_alpha, levy_scale)
            new_individual = parent + perturbation
            
            # Apply perturbation rate to maintain diversity
            mask = np.random.rand(n_points) < perturb_rate
            if np.any(mask):
                new_individual[mask] += np.random.normal(0, 0.1, np.sum(mask))
            
            # Repair to feasibility
            new_individual = repair_h(new_individual, n_points)
            new_population.append(new_individual)
        
        # Gradient descent on elite candidates
        for idx in elite_indices:
            ind = population[idx].copy()
            current_fitness = fitness[idx]
            
            # Compute gradient
            eps = 1e-6
            gradient = np.zeros(n_points)
            for j in range(n_points):
                ind_plus = ind.copy()
                ind_plus[j] += eps
                ind_plus = repair_h(ind_plus, n_points)
                corr_plus = correlate(ind_plus, 1 - ind_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                gradient[j] = (c5_plus - current_fitness) / eps
            
            # Update using gradient
            ind = ind - lr * gradient
            ind = repair_h(ind, n_points)
            
            # Update if better
            corr = correlate(ind, 1 - ind, mode='full')
            new_fitness = np.max(corr) * dx
            
            if new_fitness < current_fitness:
                population[idx] = ind
                fitness[idx] = new_fitness
                if new_fitness < best_fitness:
                    best_fitness = new_fitness
                    best_individual = ind.copy()
                    no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Update population with new individuals
        population = new_population
        fitness = [evaluate_h(ind, dx) for ind in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_fitness:
            best_fitness = fitness[current_best_idx]
            best_individual = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Early termination
        if no_improvement_count > max_no_improvement:
            break
    
    # Final gradient descent on best solution
    best_h = best_individual.copy()
    for _ in range(1000):
        if time.time() - start_time > budget_s - 10:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_fitness) / eps
        
        # Update and repair
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_fitness = np.max(corr) * dx
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_fitness = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_fitness, n_points)
