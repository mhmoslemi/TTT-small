import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 100
    epsilon = 1e-5
    
    # Try to use initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    # Normalize to sum to n_points / 2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    # Binary search for C5
    low, high = 0.0, 0.5
    while high - low > epsilon and time.time() - start_time < budget_s - 10:
        mid = (low + high) / 2
        h_candidate = optimize_h_for_c5(h_init, mid, n_points, dx, start_time, budget_s)
        
        if h_candidate is not None:
            # Compute actual C5
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            actual_c5 = np.max(corr) * dx
            if actual_c5 < best_c5:
                best_c5 = actual_c5
                best_h = h_candidate.copy()
                
            # Update search bounds
            high = mid
        else:
            low = mid
    
    # Final local refinement
    h_refined = best_h.copy()
    
    # Simple gradient descent refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
            
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        fitness = np.max(corr) * dx
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_refined.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_refined = h_refined - 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_sum = np.sum(h_refined)
        if abs(h_sum - n_points / 2) > 1e-10:
            h_refined = h_refined * (n_points / 2) / h_sum
        corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    # Final computation
    corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
    c5_final = np.max(corr_final) * dx
    
    if c5_final < best_c5:
        best_c5 = c5_final
        best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def optimize_h_for_c5(h_start, target_c5, n_points, dx, start_time, budget_s):
    """Try to find a feasible h for the given target C5"""
    # Simple algorithm: gradient-based adjustment
    h = h_start.copy()
    
    for iteration in range(500):
        if time.time() - start_time > budget_s - 5:
            return None
            
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # If we're close to target, try refining
        if abs(current_c5 - target_c5) < 1e-3:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_sum = np.sum(h_plus)
            if abs(h_sum - n_points / 2) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - current_c5) / eps
            
        # Adjust h
        h = h - 0.0005 * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum

    # Final check
    corr = np.correlate(h, 1 - h, mode='full')
    actual_c5 = np.max(corr) * dx
    
    # Allow small tolerance
    if actual_c5 <= target_c5 * 1.1:
        return h
    else:
        return None
