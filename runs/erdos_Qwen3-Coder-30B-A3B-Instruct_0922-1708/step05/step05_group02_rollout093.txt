import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        if diff > 0:
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(size, alpha=1.5, scale=0.5):
    """Generate Lévy flight perturbations."""
    # Generate Lévy distributed random variables
    # We simulate this using the characteristic function approach
    # For simplicity, we use a normal approximation for small steps
    # and heavy-tailed distribution for larger random jumps
    u = np.random.normal(0, 1, size)
    v = np.random.exponential(1, size)
    return scale * (u / (v ** (1/alpha)))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize particles with better starting solutions
    swarm_size = 50
    particles = []
    best_h = None
    best_c5 = float('inf')
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Add more diverse initial particles
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Keep track of best particle in each generation
    generation_best = []
    stagnation_counter = 0
    max_stagnation = 100
    
    # Main evolutionary loop
    for iter_count in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate all particles
        particle_values = []
        for particle in particles:
            c5 = evaluate_h(particle, dx)
            particle_values.append(c5)
        
        # Update global best
        current_best_idx = np.argmin(particle_values)
        current_best_value = particle_values[current_best_idx]
        if current_best_value < best_c5:
            best_c5 = current_best_value
            best_h = particles[current_best_idx].copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter >= max_stagnation:
            # Restart with new diverse particles
            particles = []
            for _ in range(swarm_size):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles.append(h)
            stagnation_counter = 0
            generation_best = []
            continue
            
        # Create a new generation
        new_particles = []
        
        # Keep top performing particles
        top_indices = np.argsort(particle_values)[:3]
        for idx in top_indices:
            new_particles.append(particles[idx].copy())
            
        # Generate new particles using Levy flights
        for _ in range(swarm_size - len(top_indices)):
            # Select a parent
            parent_idx = np.random.randint(len(particles))
            parent = particles[parent_idx].copy()
            
            # Apply Levy flight perturbation
            perturbation = levy_flight(n_points, alpha=1.5, scale=0.1)
            new_h = parent + perturbation
            
            # Repair
            new_h = repair_h(new_h, n_points)
            new_particles.append(new_h)
            
        # Gradient descent on top performers
        if iter_count % 10 == 0 and len(particle_values) > 0:
            top_indices = np.argsort(particle_values)[:10]
            for idx in top_indices:
                h = particles[idx].copy()
                current_c5 = particle_values[idx]
                
                # Compute gradient
                eps = 1e-5
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                # Update with gradient descent
                h = h - 0.001 * grad
                h = repair_h(h, n_points)
                corr = correlate(h, 1 - h, mode='full')
                new_c5 = np.max(corr) * dx
                
                # Keep improvement
                if new_c5 < current_c5:
                    new_particles[idx] = h.copy()
                    
        particles = new_particles
        
    # Final refinement with better gradient descent
    if best_h is not None:
        for _ in range(500):
            if time.time() - start_time > budget_s - 5:
                break
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = best_h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - best_c5) / eps
            
            best_h = best_h - 0.0002 * grad
            best_h = repair_h(best_h, n_points)
            corr = correlate(best_h, 1 - best_h, mode='full')
            best_c5 = np.max(corr) * dx
            
        # Ensure final values are exactly validated
        corr = correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * dx

    return (best_h.tolist(), best_c5, n_points)
