import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    coarse_dx = 2.0 / n_start
    
    # Initialization from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    # Normalize to satisfy constraint
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Coarse optimization using quadratic programming
    best_c5_coarse = float('inf')
    best_h_coarse = h_coarse.copy()
    
    try:
        h_coarse_var = cp.Variable(n_start)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_coarse_var >= 0,
            h_coarse_var <= 1,
            cp.sum(h_coarse_var) == n_start / 2,
            C5_var >= 0
        ]
        
        # Add approximate convex constraints for the max term
        # We'll validate based on a finite set of constraints
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        
        prob.solve(solver=ECOS, verbose=False, feastol=1e-8, abstol=1e-8, reltol=1e-8)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_solution = h_coarse_var.value
            if h_solution is not None:
                h_coarse = np.array(h_solution)
                h_coarse = np.clip(h_coarse, 0, 1)
                h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
                h_coarse = np.clip(h_coarse, 0, 1)
                
        # Check if this is a good coarse solution
        correlation_coarse = np.correlate(h_coarse, 1 - h_coarse, mode='full') * coarse_dx
        c5_coarse = np.max(correlation_coarse)
        if c5_coarse < best_c5_coarse:
            best_c5_coarse = c5_coarse
            best_h_coarse = h_coarse.copy()
    except Exception:
        pass
    
    # Transfer to fine grid
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Fine grid optimization using gradient descent
    def evaluate_c5(h):
        corr = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(corr)
    
    # Gradient estimation
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        c5_current = evaluate_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
            
        return grad
    
    # Adaptive learning rate
    lr = 0.005
    last_c5 = float('inf')
    patience = 0
    max_iter = 10000
    iterations = 0
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    while iterations < max_iter and time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update h
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h_new)
        
        # Update if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        # Check convergence
        if abs(new_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
            
        h = h_new
        last_c5 = new_c5
        
        # Adapt learning rate
        lr = max(1e-5, lr * 0.9995)
        iterations += 1
    
    # Final verification
    final_c5 = evaluate_c5(best_h)
    
    # Final normalization check
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Re-evaluate final
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
