import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import Minimize, Problem

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    k_grid = np.linspace(-2, 2, 50)
    
    # Initialize h from initial_h_values or random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_current = h_init.copy()
    
    # Binary search on C5
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    best_c5 = float('inf')
    best_h = h_current.copy()
    
    def compute_overlap(h, k):
        # Simulate convolution via direct computation for shift k
        if k >= 0:
            # h(x) * (1 - h(x+k))
            h_left = h[:-int(k/dx)]
            h_right = 1 - h[int(k/dx):]
            if len(h_left) > len(h_right):
                h_left = h_left[:len(h_right)]
            elif len(h_right) > len(h_left):
                h_right = h_right[:len(h_left)]
            return np.sum(h_left * h_right) * dx
        else:
            # h(x+k) * (1 - h(x))
            h_left = h[int(-k/dx):]
            h_right = 1 - h[:-int(-k/dx)]
            if len(h_left) > len(h_right):
                h_left = h_left[:len(h_right)]
            elif len(h_right) > len(h_left):
                h_right = h_right[:len(h_left)]
            return np.sum(h_left * h_right) * dx
    
    def compute_max_overlap(h):
        max_val = 0
        for k in k_grid:
            val = compute_overlap(h, k)
            if val > max_val:
                max_val = val
        return max_val
    
    def solve_cvxopt(h_start, mu):
        h = cp.Variable(n_points)
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Add overlapping constraints
        for idx, k in enumerate(k_grid):
            if k >= 0:
                part1 = h[:-int(k/dx)]
                part2 = 1 - h[int(k/dx):]
                if len(part1) > len(part2):
                    part1 = part1[:len(part2)]
                elif len(part2) > len(part1):
                    part2 = part2[:len(part1)]
                constraints.append(cp.sum(part1 * part2) >= mu * dx * len(part1))
            else:
                part1 = h[int(-k/dx):]
                part2 = 1 - h[:-int(-k/dx)]
                if len(part1) > len(part2):
                    part1 = part1[:len(part2)]
                elif len(part2) > len(part1):
                    part2 = part2[:len(part1)]
                constraints.append(cp.sum(part1 * part2) >= mu * dx * len(part1))
                
        objective = Minimize(cp.sum(h))  # Dummy objective
        prob = Problem(objective, constraints)
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == 'optimal':
                return h.value
        except:
            pass
        return None
    
    # Binary search
    while high - low > tolerance and time.time() - start_time < budget_s - 10:
        mu = (low + high) / 2
        h_val = solve_cvxopt(h_current, mu)
        if h_val is not None:
            low = mu
            h_current = h_val
        else:
            high = mu
            
    # Final refinement using local optimization
    h_opt = h_current.copy()
    max_iter = 200
    learning_rate = 0.001
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Estimate gradient via finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        base_c5 = compute_max_overlap(h_opt)
        
        for i in range(n_points):
            h_plus = h_opt.copy()
            h_plus[i] += eps
            # Repair
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points / 2) > 1e-10:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            c5_plus = compute_max_overlap(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        
        h_opt -= learning_rate * grad
        # Repair
        h_opt = np.clip(h_opt, 0, 1)
        h_opt_sum = np.sum(h_opt)
        if abs(h_opt_sum - n_points / 2) > 1e-10:
            h_opt = h_opt * (n_points / 2) / h_opt_sum
            
        c5_current = compute_max_overlap(h_opt)
        if c5_current < best_c5:
            best_h = h_opt.copy()
            best_c5 = c5_current
    
    return (best_h.tolist(), best_c5, n_points)
