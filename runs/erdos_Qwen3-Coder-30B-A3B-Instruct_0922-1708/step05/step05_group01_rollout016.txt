import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters  
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize binary indicator z
    z = (h > 0.5).astype(float)
    
    # Parameters for optimization
    mu = 0.1
    lr = 0.001
    max_iter = 5000
    perturb_fraction = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute overlap
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the lag that maximizes overlap
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        # Compute gradient w.r.t. h for this lag
        # The gradient of (h * (1 - h_shifted))[k] w.r.t. h[i] is:
        # -1 if i corresponds to a position that contributes to the overlap at k+shift
        # +1 if i corresponds to a position that contributes when shifted by k+shift
        # (let's compute the derivative more carefully)
        grad = np.zeros(n_points)
        
        # Compute overlap integrand manually for the max lag
        for i in range(n_points):
            # For a fixed i, the contribution is h[i] * (1 - h[i + max_lag])
            # unless the shift goes out of bounds
            shifted_i = i + max_lag
            if 0 <= shifted_i < n_points:
                grad[i] -= (1 - h[shifted_i])  # Gradient from h[i] * (1 - h[shifted_i])
                grad[shifted_i] -= h[i]        # Gradient from h[i] * (1 - h[shifted_i])
        
        grad = grad * dx
        
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update z with rounding
        z_new = (h_new > 0.5).astype(float)
        
        # Solve augmented Lagrangian subproblem
        # minimize C5 + mu * sum |h - z|
        # with constraints sum(h) = n/2, 0 <= h <= 1
        # This is a modification of a quadratic penalty approach.
        # Let's do fixed-point-like iterations to optimize:
        for _ in range(5):  # Inner iterations for better convergence
            h_temp = h_new.copy()
            for i in range(n_points):
                # Apply penalty update: -mu * (h[i] - z_new[i])
                penalty_grad = -mu * (h_temp[i] - z_new[i])
                h_temp[i] = h_temp[i] - lr * penalty_grad
                h_temp[i] = np.clip(h_temp[i], 0, 1)
                
            h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
            h_new = h_temp.copy()
        
        h = h_new
        z = z_new
        
        # Occasionally perturb
        if iteration % 100 == 0:
            perturb = np.random.rand(n_points) * perturb_fraction
            h += perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Increase penalty parameter with time
        if iteration % 100 == 0:
            mu *= 1.05
    
    # Final computed value
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
