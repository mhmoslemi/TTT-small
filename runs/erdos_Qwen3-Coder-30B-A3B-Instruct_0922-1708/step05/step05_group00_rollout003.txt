import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a coarse grid to efficient exploration
    n_start = 50
    n_points = 200  # Final resolution
    dx = 2.0 / n_points
    
    # Initialize using parent construction if available
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    # Normalize coarse grid to sum to n_start / 2
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Adam parameters for first-order moments
    m = np.zeros(n_start)
    v = np.zeros(n_start)
    
    last_c5 = float('inf')
    patience = 0
    best_h_coarse = h_coarse.copy()
    best_c5 = float('inf')
    
    # Coarse grid optimization
    max_iter_coarse = 2000
    for iteration in range(max_iter_coarse):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * (2.0 / n_start)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h_coarse.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros(n_start)
        eps = 1e-4
        for i in range(n_start):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * (2.0 / n_start)
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h_coarse -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_coarse = np.clip(h_coarse, 0, 1)
        h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    
    # Transfer to finer grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h_coarse)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine grid optimization
    # Refine with Adam on fine grid
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    last_c5 = float('inf')
    patience = 0
    best_h_fine = h_fine.copy()
    best_c5_fine = float('inf')
    
    max_iter_fine = 2000
    for iteration in range(max_iter_fine):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h_fine.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iteration + 1))
        v_hat = v / (1 - beta2**(iteration + 1))
        h_fine -= lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Final evaluation of best solution
    correlation = np.correlate(best_h_fine, 1 - best_h_fine, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Make sure sum is exactly correct
    final_h = best_h_fine
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
