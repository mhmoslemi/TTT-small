import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start approach
    initial_solutions = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    else:
        # Fallback to uniform initialization
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    
    # Generate random initializations
    for _ in range(4):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    # Process each initialization
    for h_init in initial_solutions:
        if time.time() - start_time > budget_s - 5:
            break
            
        h = h_init.copy()
        
        # Convex relaxation to get a good starting point
        try:
            h_var = cp.Variable(n_points)
            # Objective: minimize sum of products (upper bound on overlap)
            # But this needs more thought for the actual overlap function
            # Just set up the problem in case we can get something useful
            
            constraints = [
                h_var >= 0,
                h_var <= 1,
                cp.sum(h_var) == n_points / 2
            ]
            
            # Try to solve a tractable LP relaxation 
            # We consider:
            # Minimize max_k sum_i h[i] * (1 - h[(i+k)%n]) 
            # This is not convex but can be convexified by using convex ensembles
            # Instead simplify by just solving a minimal reconstruction
            
            # Use simple LP relaxation for minimum correlation:
            # Though the true problem has product terms which are difficult
            # We might run a projected gradient descent with lower bounds directly
            
            # More practical: just start with h and refine
            pass # skip this part for now as we're solving directly
                
        except Exception:
            pass  # Continue with initialization
        
        # Refinement using L-BFGS-B
        def objective(h_flat):
            # Reshape back to vector
            h = h_flat.reshape(n_points)
            # Clip and normalize
            h = np.clip(h, 0, 1)
            # Ensure sum constraint
            h = h * (n_points / 2) / np.sum(h)
            # Compute overlap integral
            corr = np.correlate(h, 1-h, mode='full')
            return np.max(corr) * dx
        
        def grad(h_flat):
            h = h_flat.reshape(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Derivative using finite differences
            eps = 1e-4
            grad_vec = np.zeros_like(h_flat)
            
            for i in range(n_points):
                h_plus = h_flat.copy()
                h_plus[i] += eps
                # Project and compute gradient
                h_plus = h_plus.reshape(n_points)
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus) 
                c5_plus = objective(h_plus.reshape(-1))
                
                grad_vec[i] = (c5_plus - objective(h_flat)) / eps
                
            return grad_vec
            
        # Optimization with bounds and constraints
        try:
            # Use scipy minimize with L-BFGS-B
            bounds = [(0, 1) for _ in range(n_points)]
            opt_result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                options={'maxiter': 2000, 'gtol': 1e-6},
                callback=None
            )
            
            h_final = opt_result.x.reshape(n_points)
            h_final = np.clip(h_final, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            
            # Compute final C5
            corr_final = np.correlate(h_final, 1-h_final, mode='full')
            c5_final = np.max(corr_final) * dx
            
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = h_final.copy()
                
        except Exception as e:
            continue  # Try next initialization
            
    # Ensure final result meets constraint
    if best_h is not None:
        # Normalize again for safety
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        # Compute final overlap
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        
        # Final check
        assert abs(np.sum(best_h) * dx - 1.0) < 1e-6, "Sum constraint violated"
        assert all(0 <= x <= 1 for x in best_h), "Bounds violated"
        
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to the uniform distribution if nothing better found
        h_fallback = np.ones(n_points) * 0.5
        h_fallback = h_fallback * (n_points / 2) / np.sum(h_fallback)
        
        corr = np.correlate(h_fallback, 1 - h_fallback, mode='full')
        fallback_c5 = np.max(corr) * dx
        
        return (h_fallback.tolist(), fallback_c5, n_points)
