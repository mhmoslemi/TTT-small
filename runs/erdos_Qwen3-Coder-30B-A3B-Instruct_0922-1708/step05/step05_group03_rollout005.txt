import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Start with a uniform distribution

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Set up the optimization parameters
    pop_size = 100
    generations = 500
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            # Start with a copy of the initial solution and add some noise
            individual = h.copy() + np.random.normal(0, 0.05, n_points)
        else:
            # Start with random valid solutions
            individual = np.random.rand(n_points)
        individual = np.clip(individual, 0, 1)
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all individuals
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Find best solution in current generation
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection (tournament selection)
        def tournament_select():
            k = 3
            indices = np.random.choice(pop_size, k)
            winner_idx = indices[np.argmin([fitness[i] for i in indices])]
            return population[winner_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep the best individual
        new_population.append(population[min_fitness_idx].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_select()
            parent2 = tournament_select()
            
            if np.random.rand() < crossover_rate:
                # Blend crossover
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
            else:
                # Copy parent
                child = parent1.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child[i] += np.random.normal(0, 0.02)
            
            # Repair solution to maintain constraints
            child = np.clip(child, 0, 1)
            sum_h = np.sum(child)
            if abs(sum_h - n_points/2) > 1e-8:
                child = child * (n_points / 2) / sum_h
            
            new_population.append(child)
        
        # Update population
        population = new_population
        
        # Occasionally perform local search on best individuals
        if gen % 50 == 0:
            # Refine top 10%
            top_indices = np.argsort(fitness)[:pop_size//10]
            for idx in top_indices:
                h_ref = population[idx].copy()
                lr = 0.001
                
                for _ in range(20):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    h_ref = h_ref - lr * grad
                    h_ref = np.clip(h_ref, 0, 1)
                    h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
                    
                    corr_new = np.correlate(h_ref, 1 - h_ref, mode='full')
                    new_c5 = np.max(corr_new) * dx
                    
                    if new_c5 < fitness[idx]:
                        fitness[idx] = new_c5
                        population[idx] = h_ref.copy()
    
    # Final refinement on the best few individuals
    final_candidates = []
    for _ in range(5):
        if 'initial_h_values' in globals():
            h_candidate = best_h.copy() + np.random.normal(0, 0.01, n_points)
        else:
            h_candidate = np.random.rand(n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        final_candidates.append(h_candidate)
    
    # Add best solution from all initial population
    final_candidates.append(population[np.argmin(fitness)].copy())
    
    # Gradient refinement
    for h_ref in final_candidates:
        for _ in range(30):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[j] = (c5_plus - best_c5) / eps
            
            h_ref = h_ref - 0.0005 * grad
            h_ref = np.clip(h_ref, 0, 1)
            h_ref = h_ref * (n_points / 2) / np.sum(h_ref)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_ref.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
