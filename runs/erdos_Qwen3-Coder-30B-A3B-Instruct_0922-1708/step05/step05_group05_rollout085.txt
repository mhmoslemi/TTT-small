import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    learning_rate = 0.001
    perturbation_period = 50
    perturbation_magnitude = 0.1
    decay_factor = 1.0 / (1.0 + 0.01 * max_iterations)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Initial construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Project to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial overlap
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Get the minimum correlation index
        corr = correlate(h, 1 - h, mode='full')
        min_k_idx = np.argmin(corr)
        min_corr_val = corr[min_k_idx]
        
        # Compute gradient for the minimum correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Forward and backward differences
            j_forward = (i + min_k_idx - (n_points - 1)) % n_points
            j_backward = (i - min_k_idx + (n_points - 1)) % n_points
            grad[i] = (h[j_forward] + h[j_backward]) * dx
            
        # Update step
        h_new = h + learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept update if improvement
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            
            if new_c5 < best_c5:
                best_h = h.copy()
                best_c5 = new_c5
        else:
            # Reduce learning rate if no improvement
            learning_rate *= 0.999
            
        # Perturb occasionally to escape local minima
        if iteration % perturbation_period == 0:
            noise = np.random.normal(0, perturbation_magnitude * (1.0 - iteration * decay_factor), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final validation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
