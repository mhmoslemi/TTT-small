import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute C5 value for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2, alpha=0.8):
    """Blend crossover"""
    child1 = alpha * parent1 + (1 - alpha) * parent2
    child2 = (1 - alpha) * parent1 + alpha * parent2
    return child1, child2

def mutate(h, mutation_rate=0.1, sigma=0.03):
    """Gaussian mutation"""
    mask = np.random.rand(len(h)) < mutation_rate
    noise = np.random.normal(0, sigma, len(h))
    h = h + mask * noise
    return h

def gradient_descent_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine h using gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        # Approximate gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_fitness = compute_fitness(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            plus_fitness = compute_fitness(h_plus, dx)
            grad[i] = (plus_fitness - base_fitness) / eps
            
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        
        # Early stopping if improvement is small
        if abs(compute_fitness(h, dx) - base_fitness) < 1e-8:
            break
            
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    if 'initial_h_values' in globals():
        # Start with the provided initial solution
        base_h = np.array(initial_h_values)
        base_h = base_h / np.sum(base_h) * (n_points / 2)
        
        # Add some variation around the initial solution
        for _ in range(population_size // 2):
            noise = np.random.normal(0, 0.05, n_points)
            h = base_h + noise
            h = project_to_feasible(h, n_points)
            population.append(h)
        
        # Fill with random solutions
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        # Initialize entirely with random valid solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Evaluate initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    generation = 0
    while time.time() - start_time < budget_s - 10:
        generation += 1
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best two
        sorted_indices = np.argsort(fitnesses)
        new_population.extend([population[i] for i in sorted_indices[:2]])
        
        # Generate rest
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:  # crossover rate
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Feasibility projection
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best
        curr_best_idx = np.argmin(fitnesses)
        if fitnesses[curr_best_idx] < best_fitness:
            best_fitness = fitnesses[curr_best_idx]
            best_h = population[curr_best_idx].copy()
            
        # Periodic gradient descent refinement on top individuals
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:population_size//10]
            for i in top_indices:
                refined = gradient_descent_refinement(population[i], dx)
                refined_fitness = compute_fitness(refined, dx)
                if refined_fitness < fitnesses[i]:
                    population[i] = refined
                    fitnesses[i] = refined_fitness
                    if refined_fitness < best_fitness:
                        best_fitness = refined_fitness
                        best_h = refined.copy()
    
    # Final refinement pass
    for _ in range(3):
        h_ref = best_h.copy() + np.random.normal(0, 0.02, n_points)
        h_ref = project_to_feasible(h_ref, n_points)
        h_ref = gradient_descent_refinement(h_ref, dx)
        c5 = compute_fitness(h_ref, dx)
        if c5 < best_fitness:
            best_fitness = c5
            best_h = h_ref.copy()
            
    # Return final result
    return (best_h.tolist(), best_fitness, n_points)
