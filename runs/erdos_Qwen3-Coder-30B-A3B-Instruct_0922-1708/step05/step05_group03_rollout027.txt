import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the C5 bound for given h values."""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region with bounds [0,1] and sum = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) > 1e-10:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select individual using tournament selection."""
    idx = np.random.choice(len(population), tournament_size)
    best_idx = idx[np.argmin([fitnesses[i] for i in idx])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.9):
    """Blend crossover operation."""
    offspring = alpha * parent1 + (1 - alpha) * parent2
    return offspring

def mutate(individual, mutation_rate=0.1, sigma=0.03):
    """Apply Gaussian mutation to individual."""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual += noise
    return individual

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        # Use parent solution
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2) / np.sum(h_parent)
        # Create variations with gaussian noise
        for _ in range(population_size // 2):
            h = h_parent + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        # Add random feasible solutions
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Random initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial fitness
    fitnesses = [compute_fitness(h, dx) for h in population]
    
    # Main GA loop
    start_time = time.time()
    best_fitness = min(fitnesses)
    best_individual = population[np.argmin(fitnesses)].copy()
    
    generation = 0
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1
        
        # Selection and reproduction
        new_population = []
        
        # Elitism: keep best 2
        sorted_indices = np.argsort(fitnesses)
        new_population.extend([population[i].copy() for i in sorted_indices[:2]])
        
        # Generate rest through crossover and mutation
        while len(new_population) < population_size:
            # Tournament selection
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                offspring = blend_crossover(p1, p2, alpha=0.9)
            else:
                offspring = p1.copy()
            
            # Mutation
            offspring = mutate(offspring, mutation_rate=0.1, sigma=0.03)
            
            # Maintain feasibility
            offspring = project_to_feasible(offspring, n_points)
            
            new_population.append(offspring)
        
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best
        current_best_fitness = min(fitnesses)
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[np.argmin(fitnesses)].copy()
        
        # Periodic gradient descent on top individuals
        if generation % 50 == 0:
            top_indices = np.argsort(fitnesses)[:10]
            for i in top_indices:
                h_ref = population[i].copy()
                # Simple gradient descent
                for _ in range(50):
                    # Estimate gradient
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        c5_plus = compute_fitness(h_plus, dx)
                        grad[j] = (c5_plus - fitnesses[i]) / eps
                    
                    # Update
                    h_ref = h_ref - 0.0005 * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    # Update fitness if better
                    new_fitness = compute_fitness(h_ref, dx)
                    if new_fitness < fitnesses[i]:
                        fitnesses[i] = new_fitness
                        population[i] = h_ref.copy()
            
            # Re-evaluate top individual
            new_fitness = min(fitnesses)
            if new_fitness < best_fitness:
                best_fitness = new_fitness
                best_individual = population[np.argmin(fitnesses)].copy()
    
    # Final refinement pass
    top_indices = np.argsort(fitnesses)[:5]
    for i in top_indices:
        h_ref = population[i].copy()
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                c5_plus = compute_fitness(h_plus, dx)
                grad[j] = (c5_plus - fitnesses[i]) / eps
            
            h_ref = h_ref - 0.0002 * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            new_fitness = compute_fitness(h_ref, dx)
            if new_fitness < fitnesses[i]:
                fitnesses[i] = new_fitness
                population[i] = h_ref.copy()
    
    # Final evaluation of best individual
    final_fitness = min(fitnesses)
    final_individual = population[np.argmin(fitnesses)].copy()
    
    return (final_individual.tolist(), final_fitness, n_points)
