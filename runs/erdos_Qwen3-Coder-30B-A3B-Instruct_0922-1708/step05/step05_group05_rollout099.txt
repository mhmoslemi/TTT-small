import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    trust_radius = 0.01
    min_trust_radius = 1e-5
    T = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_c5 = current_c5
    best_h = h.copy()
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find the maximum overlap index
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        # The maximum overlap corresponds to the maximum value in correlation array
        current_c5 = np.max(corr) * dx
        
        # Compute gradient using finite differences
        eps = 1e-6
        gradient = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_val = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = np.max(corr_plus) * dx
            
            gradient[i] = (val_plus - base_val) / eps
            
        # Compute new trial step
        trial_h = h - trust_radius * gradient
        # Ensure feasibility
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
        
        # Evaluate trial
        corr_trial = np.correlate(trial_h, 1 - trial_h, mode='full')
        trial_c5 = np.max(corr_trial) * dx
        
        # Accept or reject step
        if trial_c5 < current_c5:
            h = trial_h.copy()
            current_c5 = trial_c5
            # Increase trust region
            trust_radius = min(trust_radius * 1.2, 0.1)
        else:
            # Decrease trust region
            trust_radius = max(trust_radius * 0.8, min_trust_radius)
            
        # Apply Simulated Annealing perturbation
        if iteration % perturb_freq == 0:
            for i in range(n_points):
                if np.random.rand() < 0.05:  # 5% chance to perturb
                    if h[i] < 0.5:
                        perturb = T * (1 - h[i]) * h[i]
                    else:
                        perturb = T * (1 - h[i]) * h[i]
                    h[i] += perturb
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Temperature decay
        T *= cooling_rate
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        gradient = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_val = np.max(base_corr) * dx
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            val_plus = np.max(corr_plus) * dx
            
            gradient[i] = (val_plus - base_val) / eps
            
        h_new = h - 0.001 * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_val = np.max(corr_new) * dx
        
        if new_val < base_val:
            h = h_new
        else:
            break
    
    # Final evaluation
    corr_final = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h.tolist(), final_c5, n_points)
