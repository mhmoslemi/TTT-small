from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Create a reasonable starting point
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//4] = 1.0
        h_init[n_points//4:3*n_points//4] = 0.0
        h_init[3*n_points//4:] = 1.0
    
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert initial h to Fourier coefficients
    h_fft = fft(h_init)
    c_initial = np.concatenate([h_fft.real[:n_fourier], h_fft.imag[:n_fourier]])
    
    # Objective function
    def compute_c5(h):
        # Compute correlation sum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient estimation using finite differences
    def compute_gradient(c, eps=1e-5):
        g = np.zeros_like(c)
        for i in range(len(c)):
            c_plus = c.copy()
            c_plus[i] += eps
            h_plus = ifft(np.concatenate([c_plus[:n_fourier], 1j * c_plus[n_fourier:]]))
            h_plus = np.real(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            
            c_minus = c.copy()
            c_minus[i] -= eps
            h_minus = ifft(np.concatenate([c_minus[:n_fourier], 1j * c_minus[n_fourier:]]))
            h_minus = np.real(h_minus)
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            c5_minus = compute_c5(h_minus)
            
            g[i] = (c5_plus - c5_minus) / (2 * eps)
        return g
    
    # Range for Fourier coefficients
    bounds = []
    for _ in range(n_fourier):
        bounds.append((-1.0, 1.0))
    for _ in range(n_fourier):
        bounds.append((-1.0, 1.0))
    
    # Optimization loop with gradient descent
    c_current = c_initial.copy()
    best_c = c_current.copy()
    best_c5 = compute_c5(h_init)
    iteration = 0
    stagnation = 0
    
    while time.time() - start_time < budget_s - 10 and stagnation < 100:
        h_current = ifft(np.concatenate([c_current[:n_fourier], 1j * c_current[n_fourier:]]))
        h_current = np.real(h_current)
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        c5_current = compute_c5(h_current)
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_c = c_current.copy()
            stagnation = 0
        else:
            stagnation += 1
        
        # Gradient descent step
        grad = compute_gradient(c_current)
        c_current -= 0.01 * grad
        
        # Maintain constraints
        c_current = np.clip(c_current, -1.0, 1.0)
        
        iteration += 1
    
    # Convert final best coefficients to h
    h_final = ifft(np.concatenate([best_c[:n_fourier], 1j * best_c[n_fourier:]]))
    h_final = np.real(h_final)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), best_c5, n_points)
