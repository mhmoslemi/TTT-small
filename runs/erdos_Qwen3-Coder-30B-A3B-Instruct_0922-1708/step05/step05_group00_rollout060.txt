import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import Problem, Minimize, Variable, sum as cp_sum, Maximize, Constraint
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize multiple starting points
    initial_points = []
    
    # Use provided parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_points.append(h.copy())
    else:
        # Create base distribution for fallback
        h_base = np.ones(n_points) * 0.5
        h_base = h_base * (n_points / 2) / np.sum(h_base)
        initial_points.append(h_base.copy())
    
    # Add 4 random starting points
    for i in range(4):
        h_random = np.random.rand(n_points)
        h_random = np.clip(h_random, 0, 1)
        h_random = h_random * (n_points / 2) / np.sum(h_random)
        initial_points.append(h_random.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    for i, h_start in enumerate(initial_points):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Solve convex relaxation for this initialization
        # We will use a regularization approach to handle C5
        h = h_start.copy()
        
        # Refinement using scipy.optimize
        def objective(h_vec):
            # Compute the overlap integral
            correlation = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
            return np.max(correlation)
        
        def constraint_sum(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Use trust-constr for better handling of constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = {'type': 'eq', 'fun': constraint_sum}
        
        # Limit iterations to improve convergence
        options = {'maxiter': 1000, 'disp': False}
        
        try:
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options=options,
                tol=1e-6
            )
            
            if result.success:
                h_final = result.x.copy()
                h_final = np.clip(h_final, 0, 1)
                # Re-normalize to sum(h) = n_points / 2
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                c5_val = objective(h_final)
                
                if c5_val < best_c5:
                    best_c5 = c5_val
                    best_h = h_final.copy()
                    
        except Exception:
            continue
    
    # Final validation check on the result
    if best_h is not None:
        correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
        final_c5 = np.max(correlation)
        # Ensure it's normalized
        final_h = best_h * (n_points / 2) / np.sum(best_h)
    else:
        # Fallback case
        final_h = np.ones(n_points) * 0.5
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        correlation = np.correlate(final_h, 1 - final_h, mode='full') * dx
        final_c5 = np.max(correlation)
    
    return (final_h.tolist(), final_c5, n_points)
