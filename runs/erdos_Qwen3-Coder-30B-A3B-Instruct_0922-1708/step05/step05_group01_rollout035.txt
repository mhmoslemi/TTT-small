import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Initial evaluation
    corr = correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Optimization parameters
    max_iter = 6000
    alpha = 0.01
    decay_rate = 0.5
    decay_every = 200
    eval_freq = 50
    stagnation_limit = 200
    perturb_freq = 100
    perturb_amount = 0.01
    
    iteration = 0
    last_improvement = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Adapt learning rate
        if iteration % decay_every == 0 and iteration > 0:
            alpha *= decay_rate
            
        # Evaluate current solution
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Periodic evaluation
        if iteration % eval_freq == 0:
            curr_corr = correlate(h, 1 - h, mode='full')
            max_val = np.max(curr_corr)
            if max_val * dx < best_c5:
                best_c5 = max_val * dx
                best_h = h.copy()
                last_improvement = iteration
                
        # Stagnation check
        if iteration - last_improvement > stagnation_limit:
            # Reset with better randomness
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            alpha = 0.01
            continue
            
        # Random perturbation
        if iteration % perturb_freq == 0:
            noise = np.random.randn(n_points) * perturb_amount
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            continue
            
        # Gradient-based update
        # Estimate gradient
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update using gradient
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
    # Final refinement with local search
    for _ in range(100):
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            if h_plus[i] < 0:
                h_plus[i] = 0
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update step
        h = h - 0.001 * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Early stopping if no improvement
        corr = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 > current_c5:
            break
            
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
    # final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
