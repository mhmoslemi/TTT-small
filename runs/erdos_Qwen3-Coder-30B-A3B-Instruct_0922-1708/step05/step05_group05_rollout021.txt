import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Core parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    trust_region_radius = 0.01
    min_trust_radius = 1e-5
    sa_temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Evaluate initial
    corr = np.correlate(h, 1 - h, mode='full')
    c5_bound = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Trust-region optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Find dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute gradient for the dominant lag
        grad = np.zeros(n_points)
        # For overlap_k = sum h[i] * (1 - h[i-k]) over all i
        # d(overlap_k)/dh[i] = (1 - h[i-k]) when considering k
        # and we use local max_k for gradient computation
        
        # A more accurate gradient:
        # ∂C₅ / ∂h[i] ≈ (1 - h[i - k]) + (1 - h[i + k])
        # But we update the gradient at each step for current max_k
        # Let's compute gradients properly using finite differences:
        
        # Estimate gradient via finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_fitness = np.max(base_corr) * dx

        for idx in range(n_points):
            h_plus = h.copy()
            h_plus[idx] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[idx] = (fitness_plus - base_fitness) / eps

        # Update with trust region
        # Move in direction of neg gradient with trust region
        step = -trust_region_radius * grad
        step = np.clip(step, -h, 1-h)  # Clip step to [0,1] bound
        
        # Apply step
        h_new = h + step
        
        # Ensure sum constraint is maintained
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject step
        if c5_new < c5_bound:
            h = h_new.copy()
            c5_bound = c5_new
            trust_region_radius = min(1.2 * trust_region_radius, 1.0)
        else:
            trust_region_radius = max(trust_region_radius * 0.8, min_trust_radius)
        
        # Update best
        if c5_bound < best_c5:
            best_h = h.copy()
            best_c5 = c5_bound
        
        # Simulated Annealing Perturbation
        if iteration % perturb_freq == 0 and iteration > 0:
            perturbation = np.random.randn(n_points) * sa_temperature
            h_perturbed = h + perturbation
            
            # Clip to valid range
            h_perturbed = np.clip(h_perturbed, 0, 1)
            
            # Normalize sum
            h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
            
            # Evaluate
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            # Accept with probability based on SA
            if c5_pert < c5_bound:
                h = h_perturbed.copy()
                c5_bound = c5_pert
            else:
                # Possibly accept based on SA probability (only varying temperature)
                delta = c5_pert - c5_bound
                if delta > 0 and np.random.rand() < np.exp(-delta / sa_temperature):
                    h = h_perturbed.copy()
                    c5_bound = c5_pert
                    
            sa_temperature *= cooling_rate

    # Final refinement with gradient descent
    h_fine = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
        
        eps = 1e-5
        grad = np.zeros_like(h_fine)
        base_corr = np.correlate(h_fine, 1 - h_fine, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        for j in range(n_points):
            h_plus = h_fine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[j] = (fitness_plus - base_fitness) / eps
        
        # Update
        h_fine -= 0.001 * grad
        h_fine = np.clip(h_fine, 0, 1)
        h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
        
        # Evaluate
        corr_fine = np.correlate(h_fine, 1 - h_fine, mode='full')
        new_fitness = np.max(corr_fine) * dx
        if new_fitness < base_fitness:
            base_fitness = new_fitness
        else:
            break
            
    # Final evaluation
    final_corr = np.correlate(h_fine, 1 - h_fine, mode='full')
    final_c5 = np.max(final_corr) * dx
    final_h = h_fine
    
    return (final_h.tolist(), final_c5, n_points)
