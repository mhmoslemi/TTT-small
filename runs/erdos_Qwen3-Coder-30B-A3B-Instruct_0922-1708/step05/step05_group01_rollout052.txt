import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    random_perturbation_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    lr = initial_learning_rate
    last_improvement = 0
    current_c5 = c5
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag_idx = np.argmax(corr)
        
        # Compute gradient for the max lag
        # For lag k, the gradient w.r.t. h[i] is:
        #   - (1 - h[i+k]) if k >= 0 and i+k < n (from h[i] * (1-h[i+k]))
        #   + h[i-k] if k <= 0 and i-k >= 0 (from h[i] * h[i-k])
        # But actually, for correlation h ⋆ (1-h), the gradient at index i 
        # for the maximum value is much subtler.
        # We simplify by using finite difference approximation of gradient.
        
        # To make it faster, we take a simple approach:
        # Compute directional derivatives for max contributing lag simply:
        # Gradient w.r.t. h[i] of (h ⋆ (1-h))[k*] is approximately:
        # - (1 - h[i+k*] ) if k* >= 0, i+k* < n
        # + (h[i-k*] ) if k* <= 0, i-k* >= 0 
        # But it's safer to compute the full correlation with shifted versions.
        
        # Simpler approximation: finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            elif h_plus[i] < 0:
                h_plus[i] = 0
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = np.max(correlate(h_plus, 1 - h_plus, mode='full')) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Gradient descent step
        h -= lr * grad
        h = np.clip(h, 0, 1)
        
        # Rescale to ensure sum = n_points / 2
        sum_h = np.sum(h)
        target_sum = n_points / 2
        if abs(sum_h - target_sum) > 1e-8:
            h = h * target_sum / sum_h
        
        # Compute new C5
        corr = correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < current_c5:
            current_c5 = new_c5
            last_improvement = iteration
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        else:
            # Reduce learning rate if no improvement
            if iteration - last_improvement > plateau_patience:
                if lr > min_learning_rate:
                    lr *= learning_rate_decay
                    # Perturb h slightly to escape plateaus
                    perturb = np.random.normal(0, random_perturbation_fraction, n_points)
                    h += perturb
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    last_improvement = iteration
    
    return (best_h.tolist(), best_c5, n_points)
