import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def initial_h(n_points):
    """Create an initial feasible h."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    return repair_h(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize
    h = initial_h(n_points)
    current_c5 = evaluate_h(h, dx)
    
    # Simulated Annealing parameters
    T = 0.1 * current_c5 if current_c5 > 0 else 0.1
    alpha = 0.99
    max_iter = 10000
    min_improvement = 1e-6
    patience = 100
    no_improvement_count = 0
    
    best_h = h.copy()
    best_c5 = current_c5
    current_h = h.copy()
    
    # Main loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturbation: select a random subset of indices to change
        n_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        new_h = current_h.copy()
        
        # Modify the selected indices
        for j in indices:
            delta = np.random.uniform(-0.1, 0.1)
            new_h[j] += delta
            
        # Repair and rescale
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Metropolis acceptance criterion
        delta_e = new_c5 - current_c5
        if delta_e < 0 or np.random.rand() < np.exp(-delta_e / T):
            current_h = new_h.copy()
            current_c5 = new_c5
            
            if current_c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = current_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        # Cooling schedule with adaptive behavior
        if i % 50 == 0 and i > 0:
            T = max(1e-10, T * alpha)
            
        # Early stopping if no improvement
        if no_improvement_count > patience:
            break
    
    # Final refinement with gradient descent
    final_h = best_h.copy()
    final_c5 = best_c5
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - final_c5) / eps
            
        # Update
        final_h = final_h - 0.001 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(corr) * dx
        
    return (final_h.tolist(), final_c5, n_points)
