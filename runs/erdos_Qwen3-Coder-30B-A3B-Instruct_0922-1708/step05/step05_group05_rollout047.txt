import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Load initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Default construction: piecewise constant
        h_init = np.zeros(n_points)
        h_init[:n_points//2] = 2.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h_best = h_init.copy()
    c5_best = np.max(np.correlate(h_best, 1 - h_best, mode="full")) * dx
    
    start_time = time.time()
    max_iter = 1000
    trust_radius = 0.01
    temp = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    min_trust_radius = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient approximation
        corr = np.correlate(h_best, 1 - h_best, mode="full")
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute gradient using finite differences
        gradient = np.zeros(n_points)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h_best.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            gradient[i] = (corr_plus[max_k] - max_corr) / eps
        
        # Compute trust region step
        step = -trust_radius * gradient
        h_new = h_best + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new point
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject step
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            trust_radius = min(1.2 * trust_radius, 0.1)
        else:
            trust_radius = max(min_trust_radius, 0.8 * trust_radius)
        
        # Apply simulated annealing perturbation
        if iteration % perturb_freq == 0 and temp > 1e-6:
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    perturbation = temp * (1 - h_best[i]) * h_best[i]
                    h_best[i] += np.random.normal(0, perturbation)
            
            # Project back to feasible region
            h_best = np.clip(h_best, 0, 1)
            h_best = h_best * (n_points / 2) / np.sum(h_best)
            
            # Update best if improved
            corr_current = np.correlate(h_best, 1 - h_best, mode="full")
            c5_current = np.max(corr_current) * dx
            if c5_current < c5_best:
                c5_best = c5_current
        
        temp *= cooling_rate
    
    # Final gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        corr = np.correlate(h_best, 1 - h_best, mode="full")
        max_k = np.argmax(corr)
        
        gradient = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h_best.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            gradient[i] = (corr_plus[max_k] - corr[max_k]) / eps
        
        h_new = h_best - 0.001 * gradient
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        corr_new = np.correlate(h_new, 1 - h_new, mode="full")
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
        else:
            break
    
    # Final check
    corr_final = np.correlate(h_best, 1 - h_best, mode="full")
    c5_final = np.max(corr_final) * dx
    
    return (h_best.tolist(), c5_final, n_points)
