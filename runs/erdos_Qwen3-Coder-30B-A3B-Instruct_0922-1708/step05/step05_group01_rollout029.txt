import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    max_iter = 5000
    initial_learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    best_c5 = c5
    
    learning_rate = initial_learning_rate
    last_improvement = 0
    patience = 0
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(corr)
        c5 = corr[max_idx] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            patience = 0
        else:
            patience += 1
            
        # Check for plateau and reduce learning rate if needed
        if patience > plateau_patience:
            learning_rate *= learning_rate_decay
            if learning_rate < min_learning_rate:
                learning_rate = min_learning_rate
            patience = 0
            
        # Gradient calculation for the maximum correlation lag
        # For the convolution h * (1-h) at lag max_idx, the contribution
        # to gradient depends on how h is shifted.
        
        # Create indices for the convolution overlap
        k = max_idx - (n_points - 1)  # lag
        # k is the shift amount
        
        grad = np.zeros(n_points)
        
        # Compute gradient based on the positions that contribute to max overlap
        # For correlation at lag k, we consider terms:
        # (1-h)[i] * h[i+k] or h[i] * (1-h)[i+k] for i in valid range
        # since h[i] and (1-h)[i] are both related to the same positions
        
        # Backward pass approach: for a fixed lag k, the gradient will be a function
        # of positions where both h and (1-h) overlap
        
        if k >= 0:
            # h[i] * (1-h)[i+k] term contributes to correlation
            # for i = 0 to n_points - k - 1
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k])
            for i in range(n_points - k):
                grad[i + k] -= h[i]
        else:
            # h[i+k] * (1-h)[i] term
            k = -k
            for i in range(n_points - k):
                grad[i + k] += (1 - h[i])
            for i in range(n_points - k):
                grad[i] -= h[i + k]
        
        grad = grad * (2 / n_points)
        
        # Gradient descent step
        h = h - learning_rate * grad
        
        # Repair: clip and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Add small perturbation occasionally to escape plateaus
        if patience > 100 and np.random.rand() < 0.01:
            noise = np.random.normal(0, perturb_fraction * np.std(h), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
