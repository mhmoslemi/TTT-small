import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a variation of simulated annealing for the optimization
    temp = 10.0
    cooling_rate = 0.995
    iteration = 0
    
    def evaluate(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    def project(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust to maintain sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Simulated Annealing approach with local search
    current_h = h.copy()
    current_c5 = evaluate(current_h)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Generate neighbor
        neighbor_h = current_h.copy()
        # Randomly modify one element
        idx = np.random.randint(0, n_points)
        delta = np.random.normal(0, 0.05)
        neighbor_h[idx] += delta
        
        # Project to feasible region
        neighbor_h = project(neighbor_h)
        
        neighbor_c5 = evaluate(neighbor_h)
        
        # Accept or reject the change
        if neighbor_c5 < current_c5:
            current_h = neighbor_h
            current_c5 = neighbor_c5
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
        else:
            # Accept worse solution with some probability
            if np.random.rand() < np.exp(-(neighbor_c5 - current_c5) / temp):
                current_h = neighbor_h
                current_c5 = neighbor_c5
                
        # Cool down temperature
        temp *= cooling_rate
        
        # Occasionally perform local gradient-based refinement
        if iteration % 20 == 0:
            # Local search around current best
            for _ in range(5):
                # Create small perturbations
                h_candidate = current_h.copy()
                perturb = np.random.normal(0, 0.02, n_points)
                h_candidate += perturb
                h_candidate = project(h_candidate)
                
                c5_candidate = evaluate(h_candidate)
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h_candidate.copy()

    # Final gradient-based refinement
    for _ in range(30):
        # Gradient estimation and update
        h_ref = best_h.copy()
        lr = 0.001
        eps = 1e-5
        
        grad = np.zeros_like(h_ref)
        
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = project(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps

        h_ref = h_ref - lr * grad
        h_ref = project(h_ref)
        
        corr = np.correlate(h_ref, 1 - h_ref, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_ref.copy()

    # Final verification
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    # Ensure all constraints are met
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - (n_points / 2)) < 1e-8
    
    return (best_h.tolist(), final_c5, n_points)
