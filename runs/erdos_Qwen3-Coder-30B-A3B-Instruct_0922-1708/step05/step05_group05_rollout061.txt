import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    lr = 0.001
    perturb_freq = 50
    perturb_mag = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Simple block construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = np.inf
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find which k gives the minimum correlation
        min_k_idx = np.argmax(corr)
        min_k = min_k_idx - (len(corr) // 2)
        
        # Compute gradient (simplified version using finite difference approximation)
        # We want to maximize the minimal value of correlation
        grad = np.zeros(n_points)
        eps = 1e-6
        
        # Approximate gradient for each point
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update
        h = h + lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation to escape local optima
        if i % perturb_freq == 0:
            noise = np.random.normal(0, perturb_mag * (1.0 / (1.0 + i / 100.0)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
