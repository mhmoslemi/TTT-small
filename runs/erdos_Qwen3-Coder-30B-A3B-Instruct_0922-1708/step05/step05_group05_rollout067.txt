import numpy as np
from scipy.stats import halton
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    def evaluate(h):
        """Evaluate the overlap integral for given step function h."""
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        """Repair h to satisfy constraints."""
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to maintain sum = n_points / 2
        current_sum = np.sum(h)
        target_sum = n_points / 2
        
        if abs(current_sum - target_sum) > 1e-6:
            diff = current_sum - target_sum
            
            if diff > 0:
                # Reduce values from highest to lowest
                sorted_indices = np.argsort(h)[::-1]
                for idx in sorted_indices:
                    if diff <= 0:
                        break
                    available = h[idx]
                    reduce = min(diff, available)
                    h[idx] -= reduce
                    diff -= reduce
            else:
                # Increase values from lowest to highest
                sorted_indices = np.argsort(h)
                for idx in sorted_indices:
                    if diff >= 0:
                        break
                    available = 1.0 - h[idx]
                    increase = min(-diff, available)
                    h[idx] += increase
                    diff += increase
            # Normalize to ensure exact sum
            h = h * target_sum / np.sum(h)
        return h
    
    def crossover(parent1, parent2):
        """Single-point crossover."""
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
            
        crossover_point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h, rate):
        """Gaussian mutation."""
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05)
                h[i] += noise
        h = repair(h)
        return h
    
    def tournament_select(population, fitnesses):
        """Tournament selection."""
        selected_idx = np.random.randint(0, len(population))
        for _ in range(tournament_size - 1):
            idx = np.random.randint(0, len(population))
            if fitnesses[idx] < fitnesses[selected_idx]:
                selected_idx = idx
        return population[selected_idx]
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        # Interpolate initial construction to current n_points
        orig_len = len(initial_h_values)
        if orig_len == n_points:
            h0 = np.array(initial_h_values)
        else:
            x_orig = np.linspace(0, 2, orig_len)
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, initial_h_values)
        h0 = h0 * (n_points / 2) / np.sum(h0)
    else:
        # Use Halton sequence for initialization
        # Generate Halton points in [0, 1]^n
        seq = halton.Halton(n_points)
        points = seq.random(n_points)
        
        # Simple threshold-based construction from Halton points
        h0 = np.zeros(n_points)
        for i in range(n_points):
            h0[i] = 1 if points[i][0] < 0.5 else 0
        # Reduce to sum = n_points / 2
        current_sum = np.sum(h0)
        if current_sum < n_points / 2:
            # Increase some values
            indices = np.argsort(h0)
            needed = n_points / 2 - current_sum
            for i in indices:
                if needed <= 0:
                    break
                increase = min(1 - h0[i], needed)
                h0[i] += increase
                needed -= increase
        elif current_sum > n_points / 2:
            # Decrease some values
            indices = np.argsort(h0)[::-1]
            excess = current_sum - n_points / 2
            for i in indices:
                if excess <= 0:
                    break
                reduce = min(h0[i], excess)
                h0[i] -= reduce
                excess -= reduce
        h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Ensure h0 is valid
    h0 = repair(h0)
    population.append(h0)
    
    # Fill the rest of the population with mutated versions
    for _ in range(population_size - 1):
        h = mutate(h0, mutation_rate)
        population.append(h)
    
    # Main evolution loop
    best_h = None
    best_fitness = float('inf')
    
    for g in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate(h) for h in population]
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_fitness:
            best_fitness = fitnesses[min_idx]
            best_h = population[min_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elite selection
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Tournament selection
            parent1 = tournament_select(population, fitnesses)
            parent2 = tournament_select(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            if len(new_population) < population_size:
                child1 = mutate(child1, mutation_rate)
                new_population.append(child1)
            if len(new_population) < population_size:
                child2 = mutate(child2, mutation_rate)
                new_population.append(child2)
        
        population = new_population
        
        # Adapt mutation rate
        if g % 10 == 0:
            mutation_rate *= 0.99
        
        # Debug output every 100 gens
        if g % 100 == 0:
            pass
    
    if best_h is None:
        # Fallback to best in final population
        fitnesses = [evaluate(h) for h in population]
        min_idx = np.argmin(fitnesses)
        best_h = population[min_idx].copy()
        best_fitness = fitnesses[min_idx]
    
    # Final refinement with gradient descent
    h_refined = best_h.copy()
    best_c5 = best_fitness
    
    for _ in range(100):  # Limit gradient steps to 100
        if time.time() - start_time > budget_s - 5:
            break
            
        # Estimate gradient
        grad = np.zeros(n_points)
        eps = 1e-6
        base_fitness = evaluate(h_refined)
        
        improved = False
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair(h_plus)
            
            fitness_plus = evaluate(h_plus)
            grad[j] = (fitness_plus - base_fitness) / eps
            
            if fitness_plus < base_fitness:
                improved = True
        
        if not improved:
            break
            
        # Update
        h_refined -= 0.01 * grad
        h_refined = repair(h_refined)
        
        new_fitness = evaluate(h_refined)
        if new_fitness < base_fitness:
            base_fitness = new_fitness
        else:
            break
    
    final_fitness = evaluate(h_refined)
    
    return (h_refined.tolist(), final_fitness, n_points)
