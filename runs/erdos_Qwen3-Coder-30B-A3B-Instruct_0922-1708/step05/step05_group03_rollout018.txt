import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or generate a starting solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Simulated Annealing parameters
    T0 = 10.0
    T = T0
    cooling_rate = 0.999
    perturbation_std = 0.02
    
    # Core SA loop
    while time.time() - start_time < budget_s - 10:
        iteration += 1
        
        # Compute current C₅
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate candidate by perturbing h
        h_candidate = h + np.random.normal(0, perturbation_std, n_points)
        
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute candidate C₅
        correlation_cand = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        candidate_c5 = np.max(correlation_cand) * dx
        
        # Accept or reject based on SA criteria
        delta_c5 = candidate_c5 - current_c5
        if delta_c5 < 0 or np.random.rand() < np.exp(-delta_c5 / T):
            h = h_candidate.copy()
        
        # Update temperature
        T = T0 / np.log(1 + iteration)
    
    # Gradient descent refinement
    h_refined = best_h.copy()
    for _ in range(50):
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        # Update solution
        h_refined = h_refined - 0.001 * grad
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Recompute C₅ and update if better
        corr_refined = np.correlate(h_refined, 1 - h_refined, mode='full')
        refined_c5 = np.max(corr_refined) * dx
        
        if refined_c5 < best_c5:
            best_c5 = refined_c5
            best_h = h_refined.copy()
    
    # Final verification
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
