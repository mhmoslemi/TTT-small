import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Main optimization loop
    max_iter = 1000
    perturbation_rate = 0.1
    learning_rate = 0.001
    stagnation_limit = 100
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient with finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        h_base = best_h.copy()
        
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update with gradient
        new_h = best_h - learning_rate * grad
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept the improvement
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Occasionally perturb solution
        if np.random.rand() < perturbation_rate or stagnation_count > stagnation_limit:
            perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
            for idx in perturb_indices:
                # Small random change
                delta = np.random.normal(0, 0.05)
                best_h[idx] = np.clip(best_h[idx] + delta, 0, 1)
            best_h = repair_h(best_h, n_points)
            best_c5 = evaluate_h(best_h, dx)
            stagnation_count = 0
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        h_base = best_h.copy()
        for i in range(n_points):
            h_plus = h_base.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - best_c5) / eps
        best_h = best_h - 0.0002 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
        
    return (best_h.tolist(), best_c5, n_points)
