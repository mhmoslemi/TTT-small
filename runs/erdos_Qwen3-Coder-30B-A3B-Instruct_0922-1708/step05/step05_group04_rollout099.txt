import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import rfft, irfft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem setup
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize with a smooth step function
        h_init = np.zeros(n_points)
        mid = n_points // 2
        h_init[:mid] = 1.0
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Convert to Fourier coefficients
    h_fourier = rfft(h_init)
    # Take only the first n_fourier coefficients
    h_fourier = h_fourier[:n_fourier]
    
    # Objective function for optimization
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    def objective(c_fourier):
        # Convert Fourier coefficients to h values
        h_vals = irfft(np.concatenate([c_fourier, np.zeros(n_points - len(c_fourier))]), n_points)
        # Ensure h_values are within [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to sum to n_points/2
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return compute_c5(h_vals)
    
    def gradient(c_fourier):
        eps = 1e-6
        grad = np.zeros_like(c_fourier)
        base_val = objective(c_fourier)
        for i in range(len(c_fourier)):
            c_plus = c_fourier.copy()
            c_plus[i] += eps
            grad[i] = (objective(c_plus) - base_val) / eps
        return grad
    
    # Optimization loop
    best_c5 = float('inf')
    best_c_fourier = h_fourier.copy()
    best_h = h_init.copy()
    
    iterations = 0
    max_iter = 500
    learning_rate = 0.01
    
    while iterations < max_iter and time.time() - start_time < budget_s - 10:
        # Compute gradient
        if iterations == 0:
            # First iteration, compute directly without gradient
            current_h = irfft(np.concatenate([best_c_fourier, np.zeros(n_points - len(best_c_fourier))]), n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            current_c5 = compute_c5(current_h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
                best_c_fourier = best_c_fourier.copy()
        else:
            grad = gradient(best_c_fourier)
            # Update using gradient descent
            new_c_fourier = best_c_fourier - learning_rate * grad
            # Clip to prevent divergence
            new_c_fourier = np.clip(new_c_fourier, -10, 10)
            
            # Convert to h and check
            new_h = irfft(np.concatenate([new_c_fourier, np.zeros(n_points - len(new_c_fourier))]), n_points)
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = compute_c5(new_h)
            
            # Accept the update if better
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_c_fourier = new_c_fourier.copy()
                best_h = new_h.copy()
        
        iterations += 1
        
        # Ensure we have a valid solution within time budget
        if time.time() - start_time > budget_s - 5:
            break
    
    # Final refinement
    final_h = best_h.copy()
    for _ in range(50):
        # Gradient step size
        grad_step = 0.001
        corr = np.correlate(final_h, 1 - final_h, mode='full')
        base_c5 = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = final_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - base_c5) / eps
        
        final_h -= grad_step * grad
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        
        corr_new = np.correlate(final_h, 1 - final_h, mode='full')
        c5_new = np.max(corr_new) * dx
        if c5_new >= base_c5:
            break
        else:
            base_c5 = c5_new
    
    # Compute final C5
    final_corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
