import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Relaxation variables
    y = h_init.copy()
    z = (y > 0.5).astype(float)
    
    # Optimization parameters
    mu = 0.1
    lr = 0.001
    max_iter = 5000
    perturbation_frac = 0.01
    
    best_y = y.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute current C5
        corr = correlate(y, 1 - y, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_y = y.copy()
            best_c5 = c5
        
        # Find the lag that maximizes overlap
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (len(corr) - 1) // 2
        
        # Compute gradient
        grad = np.zeros_like(y)
        # Gradient approximation for the max lag
        if max_lag >= 0:
            # y[i] affects (1 - y[i + max_lag]) 
            for i in range(len(y)):
                if i + max_lag < len(y):
                    grad[i] += -1.0  # Almost direct
        else:
            # y[i] affects (1 - y[i + max_lag]) where max_lag is negative
            for i in range(len(y)):
                if i + max_lag >= 0:
                    grad[i] += -1.0  # Almost direct

        # Add penalty term gradient
        penalty_grad = mu * (y - z)
        grad += penalty_grad
        
        # Update y
        y = y - lr * grad
        
        # Re-project y into [0,1]
        y = np.clip(y, 0, 1)
        
        # Repair to ensure sum = n_points / 2
        y_sum = np.sum(y)
        if abs(y_sum - n_points / 2) > 1e-8:
            # Rescale
            y = y * (n_points / 2) / y_sum
            
        # Update z
        z = (y > 0.5).astype(float)
        
        # Perturb occasionally
        if iteration % 100 == 0:
            perturb_idx = np.random.choice(n_points, int(perturbation_frac * n_points))
            y[perturb_idx] += np.random.normal(0, 0.01, len(perturb_idx))
            y = np.clip(y, 0, 1)
            y_sum = np.sum(y)
            y = y * (n_points / 2) / y_sum
        
        # Update penalty
        mu *= 1.01
    
    # Final evaluation
    corr = correlate(best_y, 1 - best_y, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_y.tolist(), final_c5, n_points)
