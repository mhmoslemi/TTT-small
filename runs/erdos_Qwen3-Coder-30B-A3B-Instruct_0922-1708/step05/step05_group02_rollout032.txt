import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        current_h = h_init.copy()
    else:
        # Random initialization
        current_h = np.random.rand(n_points)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
    
    current_c5 = evaluate_h(current_h, dx)
    best_h = current_h.copy()
    best_c5 = current_c5
    
    # CS parameters
    alpha = 0.25
    epsilon = 1e-5
    max_iter = 1000
    stagnation_count = 0
    max_stagnation = 200
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution by perturbing some elements
        new_h = current_h.copy()
        num_perturb = max(1, int(n_points * 0.1))  # Perturb 10% of elements
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        for idx in indices:
            # Perturb slightly
            new_h[idx] += np.random.normal(0, 0.05)
            
        new_h = repair_h(new_h, n_points)
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject based on CS criteria
        if new_c5 < current_c5 or np.random.rand() < alpha:
            current_h = new_h.copy()
            current_c5 = new_c5
            
        # Update global best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Adaptive alpha
        if stagnation_count > 50:
            alpha = min(0.5, alpha * 1.1)
            
        # Local search on best solution periodically
        if iteration % 50 == 0 and iteration > 0:
            # Refine best solution
            refined_h = best_h.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 10:
                    break
                grad = np.zeros(n_points)
                for j in range(n_points):
                    h_plus = refined_h.copy()
                    h_plus[j] += epsilon
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - best_c5) / epsilon
                    
                refined_h = refined_h - 0.001 * grad
                refined_h = repair_h(refined_h, n_points)
                corr = correlate(refined_h, 1 - refined_h, mode='full')
                new_best_c5 = np.max(corr) * dx
                
                if new_best_c5 < best_c5:
                    best_h = refined_h.copy()
                    best_c5 = new_best_c5
                else:
                    break
        
        # Early stopping if no progress
        if stagnation_count > max_stagnation:
            break
    
    # Final refinement of best solution
    final_h = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = final_h.copy()
            h_plus[j] += epsilon
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5) / epsilon
            
        final_h = final_h - 0.0005 * grad
        final_h = repair_h(final_h, n_points)
        corr = correlate(final_h, 1 - final_h, mode='full')
        new_c5 = np.max(corr) * dx
        
        if new_c5 < best_c5:
            best_h = final_h.copy()
            best_c5 = new_c5
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
