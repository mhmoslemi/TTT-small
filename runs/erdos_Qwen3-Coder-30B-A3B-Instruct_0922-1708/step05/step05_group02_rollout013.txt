import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to hit target sum exactly
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
        current_best = h_init.copy()
    else:
        # Random initial solution
        current_best = np.random.rand(n_points)
        current_best = repair_h(current_best, n_points)
        
    best_value = evaluate_h(current_best, dx)
    
    # Parameters
    num_mutations = 100
    patience = 100
    patience_count = 0
    
    while time.time() - start_time < budget_s - 10:
        # Generate mutations
        mutations = []
        for _ in range(num_mutations):
            # Clone the current best
            h_mut = current_best.copy()
            
            # Apply random perturbations (randomly flip 5-20% of elements)
            num_flip = np.random.randint(5, 21) * n_points // 100
            flip_indices = np.random.choice(n_points, num_flip, replace=False)
            for i in flip_indices:
                h_mut[i] = 1 - h_mut[i]  # Flip between 0 and 1
            
            # Repair and re-scale
            h_mut = repair_h(h_mut, n_points)
            mutations.append(h_mut)
            
        # Evaluate mutations
        best_new = current_best
        best_new_value = best_value
        
        for h_mut in mutations:
            val = evaluate_h(h_mut, dx)
            if val < best_new_value:
                best_new = h_mut
                best_new_value = val
                
        # Update if we found an improvement
        if best_new_value < best_value:
            current_best = best_new
            best_value = best_new_value
            patience_count = 0
        else:
            patience_count += 1
            if patience_count >= patience:
                break
    
    # Final gradient descent refinement
    h_ref = current_best.copy()
    old_c5 = best_value
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_ref.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode="full")
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - old_c5) / eps
            
        h_ref = h_ref - 0.0005 * grad
        h_ref = repair_h(h_ref, n_points)
        corr = correlate(h_ref, 1 - h_ref, mode="full")
        old_c5 = np.max(corr) * dx
        
        if old_c5 < best_value:
            best_value = old_c5
            current_best = h_ref.copy()

    return (current_best.tolist(), best_value, n_points)
