import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the provided parent construction if available
    if 'initial_h_values' in globals():

        n_points = len(h)

        # Initialize with a structured function for better exploration
        n_points = 100
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        # Use a piecewise linear function that respects bounds
        h = np.clip(1.5 * np.sin(np.pi * x / 2) + 0.5, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    def evaluate_c5(h):
        # compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        return np.max(correlation)
    
    # Gradient estimation with finite differences
    def compute_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        current_c5 = evaluate_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        return grad

    # Project to feasible region: clip and normalize
    def project(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

    # Adaptive gradient descent
    i = 0
    patience = 0
    last_c5 = float('inf')
    lr = 0.05
    
    while time.time() - start_time < budget_s - 1:
        # Update learning rate
        lr = max(1e-6, lr * 0.99999)
        
        # Compute gradient
        grad = compute_gradient(best_h)
        
        # Step
        h_new = best_h - lr * grad
        
        # Project to feasible region
        h_new = project(h_new)
        
        # Evaluate new solution
        c5 = evaluate_c5(h_new)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            if patience > 1000:
                break
                
        if abs(c5 - last_c5) < 1e-8:
            patience += 1
            if patience > 500:
                break
        else:
            patience = 0
            
        last_c5 = c5
        
        i += 1
        
        # Occasionally check budget
        if i % 1000 == 0 and time.time() - start_time > budget_s - 5:
            break

    # Final validation 
    final_c5 = evaluate_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
