import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = 1/2
    h = h * (0.5 / np.sum(h))
    
    # Compute initial overlap
    corr = correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    best_h = h.copy()
    
    # Gradient ascent parameters
    learning_rate = 0.001
    decay_rate = 0.5
    patience = 50
    patience_counter = 0
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations and find minimum
        corr = correlate(h, 1 - h, mode='full')
        min_idx = np.argmin(corr)
        min_val = corr[min_idx]
        k_shift = min_idx - (len(corr) // 2)
        
        # Compute gradient for min correlation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # i-th element of gradient corresponds to d/dh[i] of min correlation
            # which is the sum of contributions from h[i+k_shift] and h[i-k_shift]
            if 0 <= i + k_shift < n_points:
                grad[i] += 1.0
            if 0 <= i - k_shift < n_points:
                grad[i] += 1.0
        
        # Update h with gradient ascent step
        h_new = h + learning_rate * grad
        
        # Project back to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (0.5 / np.sum(h_new))
        
        # Evaluate new solution
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject step
        if new_c5 <= best_c5:
            h = h_new.copy()
            best_c5 = new_c5
            patience_counter = 0
            last_improvement = iteration
        else:
            patience_counter += 1
            if patience_counter > patience:
                learning_rate *= decay_rate
                patience_counter = 0
        
        # Random perturbation every 50 iterations to escape local optima
        if iteration % 50 == 0 and iteration > 0:
            pert_mag = 0.1 * (1.0 / (1.0 + iteration / 100.0))
            noise = np.random.normal(0, pert_mag, n_points)
            h_pert = h + noise
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (0.5 / np.sum(h_pert))
            
            corr_pert = correlate(h_pert, 1 - h_pert, mode='full')
            pert_c5 = np.max(corr_pert) * dx
            
            if pert_c5 < best_c5:
                h = h_pert.copy()
                best_c5 = pert_c5
    
    # Final verification
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
