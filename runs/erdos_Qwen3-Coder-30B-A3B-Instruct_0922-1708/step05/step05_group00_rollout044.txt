import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import convolve

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    
    # Normalize to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Smooth approximation of max for differentiable optimization
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Gradient of smooth max approximation
    def smooth_max_grad(x, beta=10.0):
        exp_x = np.exp(beta * x)
        return exp_x / np.sum(exp_x)
    
    # Compute correlation using FFT for faster convolution
    def compute_overlap(h):
        # Using FFT for convolution
        h1 = np.concatenate([h, np.zeros_like(h)])
        h2 = np.concatenate([1 - h, np.zeros_like(h)])
        corr = convolve(h1, h2, mode='valid')[:len(h)]
        return corr * dx
    
    # Adam optimizer helper
    def adam_update(h, grad, m, v, t, lr=0.01, beta1=0.9, beta2=0.999, eps=1e-8):
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        return h_new, m, v
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Optimization loop
    max_iter = 5000
    lr = 0.01
    beta = 10.0
    t = 1
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    patience = 0
    last_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap values
        overlap = compute_overlap(h)
        
        # Compute smooth maximum
        c5_smooth = smooth_max(overlap, beta)
        
        # Compute gradient w.r.t. h
        # This is an approximation since we don't have closed form gradient
        # Use finite difference for gradient computation
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = compute_overlap(h_plus)
            c5_plus = smooth_max(overlap_plus, beta)
            
            grad[j] = (c5_plus - c5_smooth) / eps
            
        # Update with Adam
        h_new, m, v = adam_update(h, grad, m, v, t, lr)
        t += 1
        
        # Projection onto feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluate current solution
        overlap = compute_overlap(h)
        c5 = np.max(overlap)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early stopping
        if patience > 200 or abs(c5 - last_c5) < 1e-6:
            break
            
        last_c5 = c5
        
        # Adjust learning rate
        if i % 100 == 0:
            lr = max(1e-6, lr * 0.95)
    
    return (best_h.tolist(), best_c5, n_points)
