import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 1e-3
    line_search_factor = 0.5
    stagnation_threshold = 1e-6
    max_iter = 5000
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute all correlations
        corr_full = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find which lag gives max correlation
        max_lag_idx = np.argmax(corr_full)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for current lag
        grad = np.zeros(n_points)
        if max_lag > 0:
            # h[i] * (1 - h[i + max_lag]) overlap
            for i in range(n_points - max_lag):
                grad[i] += 1 - h[i + max_lag]
                grad[i + max_lag] -= h[i]
        elif max_lag < 0:
            # h[i - max_lag] * (1 - h[i]) overlap
            for i in range(-max_lag, n_points):
                grad[i] += 1 - h[i + max_lag]
                grad[i + max_lag] -= h[i]
        else:
            # h[i] * (1 - h[i]) overlap
            grad = 1 - 2 * h
        
        # Normalize gradient
        grad = grad * (2 / n_points)
        
        # Line search
        current_c5 = c5
        step_size = alpha
        while step_size > 1e-10:
            # Try step
            h_new = h - step_size * grad
            
            # Project onto feasible region [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to sum to n/2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new C5
            corr_new = correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < current_c5:
                h = h_new
                current_c5 = c5_new
                break
            else:
                step_size *= line_search_factor
        
        # Random perturbation to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            perturbation = np.random.normal(0, 0.01, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation of best solution
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
