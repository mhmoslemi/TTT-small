import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluate fitness
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main DE loop
    best_fitness = float('inf')
    best_h = None
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 10:
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            indices = np.random.choice(population_size, 3, replace=False)
            h1, h2, h3 = population[indices[0]], population[indices[1]], population[indices[2]]
            
            # Mutation
            mutant = h1 + mutation_factor * (h2 - h3)
            mutant = np.clip(mutant, 0, 1)
            mutant = repair(mutant)
            
            # Crossover
            trial = np.copy(population[i])
            for j in range(n_points):
                if np.random.rand() < crossover_prob:
                    trial[j] = mutant[j]
            
            # Repair trial
            trial = repair(trial)
            
            # Evaluate
            trial_fitness = evaluate_fitness(trial)
            
            # Selection
            if trial_fitness < evaluate_fitness(population[i]):
                new_population.append(trial)
            else:
                new_population.append(population[i])
        
        population = new_population
        
        # Update best solution
        for h in population:
            fitness = evaluate_fitness(h)
            if fitness < best_fitness:
                best_fitness = fitness
                best_h = h.copy()
        
        # Decrease mutation factor
        mutation_factor = max(0.1, mutation_factor * 0.99)
        
        iteration += 1
    
    # Final refinement
    if best_h is not None:
        h_opt = best_h.copy()
        for _ in range(100):
            fitness = evaluate_fitness(h_opt)
            
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_opt.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                fitness_plus = evaluate_fitness(h_plus)
                grad[i] = (fitness_plus - fitness) / eps
            
            h_opt -= 0.001 * grad
            h_opt = repair(h_opt)
            
            # Early stopping if no improvement
            new_fitness = evaluate_fitness(h_opt)
            if new_fitness >= fitness:
                break
            else:
                fitness = new_fitness
        
        best_fitness = evaluate_fitness(h_opt)
        best_h = h_opt.copy()
    
    return (best_h.tolist(), best_fitness, n_points)
