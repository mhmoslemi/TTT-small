import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize h - use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    # Learning rate and other parameters
    lr = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute current overlap
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Find which index gives max correlation
        max_idx = np.argmax(corr)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Maintain sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update step
        h = h - lr * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbation to escape local optima
        if iteration % perturb_freq == 0 and iteration > 0:
            noise = np.random.normal(0, perturb_magnitude * (1.0 / (1.0 + iteration/100)), n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate
        corr_new = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < c5:
            lr *= 1.1  # Increase learning rate if we improved
        else:
            lr *= 0.9  # Decrease if no improvement
            
        lr = max(1e-6, min(lr, 0.1))  # Bound learning rate
        
        # Update best solution
        if c5_new < best_c5:
            best_h = h.copy()
            best_c5 = c5_new
    
    return (best_h.tolist(), best_c5, n_points)
