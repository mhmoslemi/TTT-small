import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    learning_rate = 1e-3
    penalty_mu = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    max_iterations = 5000
    learning_rate_decay = 0.5
    
    # Track optimization state
    last_improvement = 0
    plateau_count = 0
    
    # Initialize penalty factor
    penalty_mu = 0.1
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlations
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Check for stagnation
        if plateau_count >= plateau_patience:
            learning_rate *= learning_rate_decay
            # Add random perturbation
            perturbation = np.random.normal(0, perturbation_fraction * learning_rate, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            plateau_count = 0
            continue
            
        # Compute gradient:
        # For max correlation, identify which lag is the maximum
        max_idx = np.argmax(corr)
        k = max_idx - (n_points - 1)  # Lag in range [-n+1, n-1]
        
        # Compute gradient for this correlation
        gradient = np.zeros_like(h)
        
        if k >= 0:
            # h[i] * (1 - h[i + k]) for i = 0 to n-k-1
            for i in range(n_points - k):
                gradient[i] += (1 - h[i + k])
        else:
            # h[i + |k|] * (1 - h[i]) for i = 0 to n+|k|-1  
            k_abs = -k
            for i in range(n_points - k_abs):
                gradient[i + k_abs] += (1 - h[i])
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        # Penalty for bounds
        penalty_grad += 2 * penalty_mu * np.clip(h, 0, 1) * (h < 0) - 2 * penalty_mu * np.clip(h, 0, 1) * (h > 1)
        # Penalty for sum
        sum_diff = np.sum(h) - n_points / 2
        penalty_grad += 2 * penalty_mu * sum_diff
        
        # Combine gradients
        total_grad = gradient + penalty_grad
        
        # Update h
        h = h - learning_rate * total_grad
        
        # Repair constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-8:
            h = h * (n_points / 2) / h_sum
            
        # Update penalty parameter
        violation = 0
        violation += np.sum(np.maximum(0, -h)**2) + np.maximum(0, h_sum - n_points / 2)**2
        if violation > 1e-6:
            penalty_mu *= penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_min)
    
    # Final recomputation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
