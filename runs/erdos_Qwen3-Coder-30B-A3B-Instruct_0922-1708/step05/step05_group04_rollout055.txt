import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 2000
    mutation_strength = 0.5
    mutation_variance = 0.1
    elite_ratio = 0.5
    stagnation_limit = 200
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Add some noise to create diverse initial population
        for _ in range(population_size):
            noise = np.random.normal(0, 0.05, n_points)
            h = np.clip(h_init + noise, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Create initial random population
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper functions
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Scale to target sum
        h = h * (n_points / 2) / h_sum
        return h
    
    def evaluate_fitness(h):
        # Compute correlation for all lags
        correlation = np.correlate(h, 1 - h, mode='full')
        # Get maximum overlap
        max_overlap = np.max(correlation)
        return max_overlap * dx
    
    # Main loop
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    current_mutation_strength = mutation_strength
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness of all individuals
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_individual = population[fitnesses.index(min_fitness)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = int(population_size * elite_ratio)
        
        # Selection: keep top elite_size individuals
        elite_individuals = [population[i] for i in sorted_indices[:elite_size]]
        
        # Generate offspring through mutation
        offspring = []
        for _ in range(offspring_size):
            # Select random parent
            parent = elite_individuals[np.random.randint(0, len(elite_individuals))]
            
            # Gaussian mutation
            noise = np.random.normal(0, current_mutation_strength * mutation_variance, n_points)
            child = parent + noise
            child = repair(child)
            offspring.append(child)
            
        # Replace worst individuals with offspring
        worst_indices = sorted_indices[elite_size:]
        for i, index in enumerate(worst_indices):
            if i < len(offspring):
                population[index] = offspring[i]
                
        # Adaptive mutation strength decay
        current_mutation_strength *= 0.999
        if current_mutation_strength < 0.01:
            current_mutation_strength = 0.01
            
        # Dynamic mutation variance
        mutation_variance = current_mutation_strength / 5.0
        
        # Restart mechanism for stagnation
        if stagnation_count > 100:
            current_mutation_strength *= 0.9
            if current_mutation_strength < 0.01:
                current_mutation_strength = 0.01
            # Perturb some elites
            for i in range(0, len(elite_individuals), 5):
                if i < len(elite_individuals):
                    noise = np.random.normal(0, 0.1, n_points)
                    elite_individuals[i] = np.clip(elite_individuals[i] + noise, 0, 1)
                    elite_individuals[i] = repair(elite_individuals[i])
            stagnation_count = 0
            
        # Occasionally reintroduce random individuals
        if iteration % 50 == 0:
            for i in range(0, population_size, 10):
                h = np.random.uniform(0, 1, n_points)
                h_sum = np.sum(h)
                h = h * (n_points / 2) / h_sum
                population[i] = h

    # Final gradient descent refinement
    if best_individual is not None:
        h_refined = best_individual.copy()
        for _ in range(100):
            # Compute correlation
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            fitness = np.max(corr) * dx
            
            # Compute gradient using finite differences
            grad = np.zeros(n_points)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
                
            # Gradient step
            h_refined -= 0.001 * grad
            h_refined = repair(h_refined)
            
            # Stop if no improvement
            corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            fitness = fitness_new
            
        # Final evaluation
        final_corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(final_corr) * dx
        
        return (h_refined.tolist(), final_fitness, n_points)
    
    return (best_individual.tolist(), best_fitness, n_points) if best_individual is not None else ([0.5] * n_points, 0.5, n_points)
