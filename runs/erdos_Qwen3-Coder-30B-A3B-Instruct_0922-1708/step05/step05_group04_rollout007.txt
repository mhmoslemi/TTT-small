import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 5000
    mu_init = 0.5
    sigma_init = 0.1

    population = []
    
    if 'initial_h_values' in globals():

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        # Create population with small random perturbations around initial
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        # Initialize with random feasible solutions
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        scaling = (n_points / 2) / h_sum
        h = h * scaling
        return h
    
    # Fitness evaluation function
    def evaluate_fitness(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Main ES loop
    mu = mu_init
    sigma = sigma_init
    best_fitness = float('inf')
    best_individual = None
    stagnation_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate all individuals
        fitnesses = [evaluate_fitness(ind) for ind in population]
        
        # Track best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Selection: keep top 50%
        selected_indices = np.argsort(fitnesses)[:population_size // 2]
        selected_population = [population[i] for i in selected_indices]
        
        # Generate offspring via Gaussian mutation
        offspring = []
        for _ in range(offspring_size):
            parent = selected_population[np.random.randint(0, len(selected_population))]
            # Mutation
            noise = np.random.normal(0, sigma, n_points)
            child = parent + mu * noise
            # Repair
            child = repair(child)
            offspring.append(child)
        
        # Replace worst individuals with offspring
        # Combine population and offspring
        combined = population + offspring
        combined_fitness = fitnesses + [evaluate_fitness(ind) for ind in offspring]
        
        # Select best individuals for the new population
        best_indices = np.argsort(combined_fitness)[:population_size]
        population = [combined[i] for i in best_indices]
        
        # Adaptation 
        mu *= 0.999
        if mu < 1e-6:
            mu = 1e-6
        sigma = mu / 5
        
        # Restart if stagnation
        if stagnation_count > 200:
            mu = mu_init * 0.9
            # Add some random perturbations
            for i in range(population_size):
                if np.random.random() < 0.1:
                    population[i] = np.random.uniform(0, 1, n_points)
                    population[i] = population[i] * (n_points / 2) / np.sum(population[i])
            stagnation_count = 0

    if best_individual is not None:

            corr = np.correlate(h, 1 - h, mode='full')
            return np.max(corr) * dx

            eps = 1e-6
            grad = np.zeros_like(h)
            base_corr = np.correlate(h, 1 - h, mode='full')
            base_fitness = np.max(base_corr) * dx
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - base_fitness) / eps
            return grad
        
        # Gradient descent steps
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            grad = gradient(h_refined)
            h_refined -= 0.001 * grad
            h_refined = repair(h_refined)
                    
        # Final computation
        best_h = h_refined
        corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
        final_fitness = np.max(corr_final) * dx
        
        if final_fitness < best_fitness:
            best_fitness = final_fitness
            best_h = h_refined
    
    # Final check for final solution
    if best_individual is None:
        # Re-evaluate final population
        final_fitnesses = [evaluate_fitness(ind) for ind in population]
        best_idx = np.argmin(final_fitnesses)
        best_h = population[best_idx]
        best_fitness = final_fitnesses[best_idx]
    
    return (best_h.tolist(), best_fitness, n_points)
