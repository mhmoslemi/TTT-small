import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    trust_radius_init = 0.01
    trust_radius_min = 1e-5
    temp_init = 0.1
    cooling_rate = 0.99
    pert_freq = 50
    max_corr_evals = 100
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Use a simple construction
        h = np.ones(n_points) * 0.5
    
    # Normalize sum to n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Compute initial C5
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Adaptive trust region parameters
    trust_radius = trust_radius_init
    temp = temp_init
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute correlation and dominant lag
        corr = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient calculation for the dominant case
            # d/dh_i of max_k [sum_j h_j * (1 - h_{j+k})] 
            # approximated by gradient of dominant k
            j1 = (i - k_max) % n_points
            j2 = (i + k_max) % n_points
            grad[i] = (1 - h[j1]) + (1 - h[j2])
        
        # Normalize gradient
        grad = grad / n_points
        
        # Trust region step proposal
        delta_h = -trust_radius * grad
        
        # Project to feasible region and preserve sum
        h_new = h + delta_h
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to preserve sum constraint
        current_sum = np.sum(h_new)
        if abs(current_sum - n_points/2) > 1e-6:
            scale = (n_points / 2) / current_sum
            h_new = h_new * scale
            
        # Evaluate new solution
        c5_new = compute_c5(h_new)
        
        # Accept or reject step
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            trust_radius = min(trust_radius * 1.2, 1.0)  # Expand trust region
        else:
            trust_radius = max(trust_radius * 0.8, trust_radius_min)  # Shrink trust region
            
        # Simulated Annealing perturbations
        if iteration % pert_freq == 0 and iteration > 0:
            for _ in range(10):
                i = np.random.randint(0, n_points)
                # Perturb one element
                pert = temp * np.random.normal(0, 1)
                h[i] = h[i] + pert
                h[i] = max(0, min(1, h[i]))
                
            # Normalize sum again
            current_sum = np.sum(h)
            h = h * (n_points / 2) / current_sum
            
            c5_new = compute_c5(h)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
            elif np.random.random() < np.exp((best_c5 - c5_new) / temp):
                # Accept worse solution with probability
                pass  # Keep current h
                
        # Decay temperature
        temp *= cooling_rate
        
        # Occasionally refine the best solution
        if iteration % 500 == 0:
            # Intensive local search on best_h
            x = best_h.copy()
            for _ in range(100):
                corr = np.correlate(x, 1 - x, mode='full')
                k_max = np.argmax(corr)
                grad = np.zeros(n_points)
                for i in range(n_points):
                    j1 = (i - k_max) % n_points
                    j2 = (i + k_max) % n_points
                    grad[i] = (1 - x[j1]) + (1 - x[j2])
                grad = grad / n_points
                
                # Take step
                x_new = x - 0.001 * grad
                x_new = np.clip(x_new, 0, 1)
                x_new = x_new * (n_points / 2) / np.sum(x_new)
                
                c5_new = compute_c5(x_new)
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = x_new.copy()
                x = x_new.copy()
    
    # Final refinement
    x = best_h.copy()
    for _ in range(500):
        corr = np.correlate(x, 1 - x, mode='full')
        k_max = np.argmax(corr)
        grad = np.zeros(n_points)
        for i in range(n_points):
            j1 = (i - k_max) % n_points
            j2 = (i + k_max) % n_points
            grad[i] = (1 - x[j1]) + (1 - x[j2])
        grad = grad / n_points
        
        x_new = x - 0.001 * grad
        x_new = np.clip(x_new, 0, 1)
        x_new = x_new * (n_points / 2) / np.sum(x_new)
        
        c5_new = compute_c5(x_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = x_new.copy()
        x = x_new.copy()
    
    return (best_h.tolist(), best_c5, n_points)
