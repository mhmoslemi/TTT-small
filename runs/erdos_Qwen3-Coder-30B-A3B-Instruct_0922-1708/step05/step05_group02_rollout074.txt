import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize with initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Main optimization loop
    max_iter = 10000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Gradient descent step
        eps = 1e-5
        grad = np.zeros(n_points)
        
        # Compute gradient using finite differences
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - best_c5) / eps
        
        # Update with gradient
        best_h = best_h - 0.001 * grad
        best_h = repair_h(best_h, n_points)
        
        # Evaluate new value
        new_c5 = evaluate_h(best_h, dx)
        
        # Accept if better
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            # Sometimes accept worse to escape local minima
            # Small probability of accepting worse solution
            if np.random.rand() < 0.01:
                best_h = best_h.copy()
                best_c5 = new_c5
                
        # Occasionally perturb a few points to explore
        if iter_count % 10 == 0:
            num_perturb = max(1, n_points // 10)
            perturb_indices = np.random.choice(n_points, num_perturb, replace=False)
            for i in perturb_indices:
                perturb_mag = np.random.normal(0, 0.05)
                best_h[i] += perturb_mag
            best_h = repair_h(best_h, n_points)
            best_c5 = evaluate_h(best_h, dx)
            
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - best_c5) / eps
        
        best_h = best_h - 0.0001 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
