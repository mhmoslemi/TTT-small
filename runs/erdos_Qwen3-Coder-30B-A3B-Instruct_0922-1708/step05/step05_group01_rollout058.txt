import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Scale to sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 5000
    learning_rate = 1e-3
    line_search_factor = 0.5
    min_grad_norm = 1e-6
    perturb_fraction = 0.01
    
    # Current best
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    for iteration in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        # Compute all correlations
        correlations = []
        for k in range(1, 2 * n_points - 1):
            # For correlation lag k (0-based index from -n+1 to n-1)
            # Need to compute sum over i: h[i] * (1 - h[i + k - (n-1)])
            # Which is equivalent to sum_{0}^{n-1-k} h[i] * (1 - h[i+k])
            if k <= n_points - 1:
                corr = np.sum(h[:n_points-k] * (1 - h[k:]))
            else:
                corr = np.sum(h[k-n_points+1:] * (1 - h[:n_points-(k-n_points+1)]))
            correlations.append(corr)
        
        # Find maximum correlation
        max_corr_idx = np.argmax(correlations)
        max_corr = correlations[max_corr_idx]
        
        # Compute gradient for maximum correlation lag
        k = max_corr_idx + 1  # convert to 1-based lag
        gradient = np.zeros(n_points)
        
        # Gradient computation based on the specific lag k
        if k <= n_points - 1:
            # Normal case: h[i] * (1 - h[i + k]) contributes to corr
            # Gradient with respect to h[i] is: (1 - h[i+k]) 
            # Gradient with respect to h[i+k] is: -h[i]
            for i in range(n_points - k):
                gradient[i] += (1 - h[i+k])
                if i+k < n_points:
                    gradient[i+k] -= h[i]
        else:
            # Reverse case: lag is negative or exceeds valid range
            # Modify indices accordingly
            actual_k = k - (n_points - 1)  # now from -n+1 to -1
            # Corresponds to h[i+actual_k] * (1 - h[i]) 
            for i in range(n_points + actual_k):
                gradient[i] -= h[i - actual_k] if i - actual_k >= 0 else 0
                if i + actual_k < n_points:
                    gradient[i + actual_k] += (1 - h[i])
        
        # Scale gradient by dx * 2 / n_points
        gradient = gradient * 2 * dx / n_points
        
        # Line search
        alpha = learning_rate
        old_c5 = best_c5
        trial_h = h - alpha * gradient
        trial_h = np.clip(trial_h, 0, 1)
        trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
        trial_c5 = evaluate_c5(trial_h, dx)
        
        # Reduce step size if necessary
        while trial_c5 >= old_c5 and alpha > 1e-10:
            alpha *= line_search_factor
            trial_h = h - alpha * gradient
            trial_h = np.clip(trial_h, 0, 1)
            trial_h = trial_h * (n_points / 2) / np.sum(trial_h)
            trial_c5 = evaluate_c5(trial_h, dx)
        
        # Update h based on line search result
        if alpha > 1e-10:
            h = trial_h.copy()
            c5 = trial_c5
            # Update best
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
            # Adaptive learning rate
            if abs(old_c5 - c5) < 1e-8:
                learning_rate *= 0.9
            else:
                learning_rate = min(1e-1, learning_rate * 1.1)
        else:
            # Perturb slightly to escape stagnation
            perturbation = np.random.normal(0, perturb_fraction, n_points) * (1 - 2 * (np.random.rand(n_points) < 0.5))
            h += perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            # Small gradient update
            temp_h = h.copy()
            temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
            temp_c5 = evaluate_c5(temp_h, dx)
            if temp_c5 < best_c5:
                best_h = temp_h.copy()
                best_c5 = temp_c5
    
    # Final evaluation
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)

def evaluate_c5(h_values, dx):
    """Evaluate the maximum correlation C5 using numpy correlation"""
    h = np.array(h_values)
    # Using the scipy.correlate in full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Return the maximum value scaled by dx
    return np.max(corr) * dx
