    # Normalize to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # For coarse-to-fine approach:
    # 1. Try coarse optimization first
    n_coarse = min(50, n_points)
    if n_coarse < n_points:
        # Interpolate coarse grid
        coarse_x = np.linspace(0, 2, n_coarse)
        fine_x = np.linspace(0, 2, n_points)
        h_coarse = np.interp(fine_x, coarse_x, h[:n_coarse])
    else:
        h_coarse = h.copy()
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Gradient descent with projected updates
    def project(h_vec):
        h_proj = np.clip(h_vec, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
        
    # Smooth approximation to max
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Optimizer: Adam-style with adaptive learning rate
    def optimize_with_adam(h_init, max_iter, lr_start=0.02):
        h = h_init.copy()
        m = np.zeros_like(h)
        v = np.zeros_like(h)
        beta1, beta2 = 0.9, 0.999
        epsilon = 1e-8
        
        for t in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute gradient using finite differences
            eps = 1e-5
            grad = np.zeros_like(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = project(h_plus)
                
                # Compute overlap with 1-h for all shifts
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                c5_plus = np.max(corr_plus)
                
                grad[i] = (c5_plus - np.max(np.correlate(h, 1 - h, mode='full') * dx)) / eps
            
            # Adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * grad**2
            m_hat = m / (1 - beta1**(t + 1))
            v_hat = v / (1 - beta2**(t + 1))
            
            # Adaptive lr
            lr = lr_start / (1 + 0.0001 * t)  # decay
            h_new = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
            
            # Project
            h = project(h_new)
            
            # Evaluate
            corr = np.correlate(h, 1 - h, mode='full') * dx
            current_c5 = np.max(corr)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
        
        return best_h, best_c5
        
    # First, run coarse optimization if needed
    if n_coarse < n_points:
        h_coarse, coarse_c5 = optimize_with_adam(h_coarse, max_iter=2000, lr_start=0.02)
        # Transfer to fine grid
        fine_x = np.linspace(0, 2, n_points)
        coarse_x = np.linspace(0, 2, n_coarse)
        h_fine = np.interp(fine_x, coarse_x, h_coarse[:n_coarse])
        h_fine = project(h_fine)
    else:
        h_fine = h.copy()
    
    # Then refine fine grid
    h_final, final_c5 = optimize_with_adam(h_fine, max_iter=2000, lr_start=0.02)
    
    # Final evaluation with exact computation
    final_corr = np.correlate(h_final, 1 - h_final, mode='full') * dx
    actual_c5 = np.max(final_corr)
    
    # Ensure sum constraint
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    return (h_final.tolist(), actual_c5, n_points)
