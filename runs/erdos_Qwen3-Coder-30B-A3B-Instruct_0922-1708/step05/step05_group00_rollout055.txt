import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with the parent construction if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        h = h * (n_points / 2) / np.sum(h)  # normalize
    else:
        n_points = 200
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5
        
    h = np.clip(h, 0, 1)
    
    # Coarse optimization to escape local minima
    n_coarse = min(50, n_points)
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.interp(np.linspace(0, 2, n_coarse), np.linspace(0, 2, n_points), h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    try:
        # Quadratic programming approach: minimize C5 subject to the constraints
        h_var = cp.Variable(n_coarse)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_coarse / 2,
            C5_var >= 0
        ]
        
        # A simplified convex formulation:
        # Add one constraint for each k
        # But note: h[i] * (1 - h[i+k]) is non-convex, so we relax with
        # quadratically penalized bounds
        # For now, we'll optimize a simpler relaxation
        
        # Instead, solve coarse problem via LP relaxation
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iter=1000, abstol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_solution = h_var.value
                if h_solution is not None:
                    h_coarse = np.array(h_solution)
                    h_coarse = np.clip(h_coarse, 0, 1)
                    
                    # Project and transform to fine grid
                    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
                    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
                    h_fine = np.clip(h_fine, 0, 1)
                    
                    # Validation
                    correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx
                    c5_fine = np.max(correlation)
                    if c5_fine < best_c5:
                        best_c5 = c5_fine
                        best_h = h_fine.copy()
        except Exception:
            pass
    except Exception:
        pass
    
    # Fine grid refinement using gradient descent
    h = best_h.copy()
    lr = 1e-3
    max_iter = 5000
    
    patience = 0
    last_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1 and max_iter > 0:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update using gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        lr = max(1e-5, min(1e-3, lr * 0.999))
        
        max_iter -= 1
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
