import numpy as np
from scipy.stats import halton
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    def initialize_population():
        population = []
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init.copy())
        else:
            # Use Halton sequences for diversity
            seq = halton.Halton(1)
            for _ in range(population_size):
                h = np.zeros(n_points)
                for i in range(n_points):
                    h[i] = seq.random(1)[0] if i < n_points else 0
                h = h * (n_points / 2) / np.sum(h)
                population.append(h.copy())
        
        # Fill with random individuals
        for i in range(len(population), population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
            
        return population
    
    def evaluate_fitness(h):
        # Compute correlation and find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair(h):
        # Ensure values are in [0, 1]
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Single-point crossover
        crossover_point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h, mutation_rate_val):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate_val:
                # Gaussian perturbation
                h[i] += np.random.normal(0, 0.05)
        return repair(h)
    
    # Begin optimization
    population = initialize_population()
    
    # Evaluate initial population
    fitnesses = [evaluate_fitness(h) for h in population]
    
    # Track best
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Evolution loop
    for generation in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create offspring
        offspring = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < population_size else selected[0]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            offspring.extend([child1, child2])
        
        # Keep elite
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in range(elite_size):
            offspring[i] = population[elite_indices[i]].copy()
        
        # Replace population
        population = offspring[:population_size]
        
        # Evaluate new population
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_h = population[current_best_idx].copy()
            best_fitness = fitnesses[current_best_idx]
        
        # Adapt mutation rate
        if generation % 10 == 0:
            mutation_rate *= 0.99
    
    # Final refinement with local search around best
    refined_h = best_h.copy()
    best_c5 = best_fitness
    
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        # Local gradient-like search using finite differences
        grad = np.zeros(n_points)
        eps = 1e-5
        base_corr = np.correlate(refined_h, 1 - refined_h, mode='full')
        base_fitness = np.max(base_corr) * dx
        
        improved = False
        for j in range(n_points):
            h_plus = refined_h.copy()
            h_plus[j] += eps
            h_plus = repair(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            
            grad[j] = (fitness_plus - base_fitness) / eps
            
            if fitness_plus < base_fitness:
                improved = True
                
        if not improved:
            break
        
        # Update with step size
        refined_h -= 0.01 * grad
        refined_h = repair(refined_h)
        
        # Check for improvement
        corr = np.correlate(refined_h, 1 - refined_h, mode='full')
        new_fitness = np.max(corr) * dx
        
        if new_fitness < best_c5:
            best_c5 = new_fitness
            best_h = refined_h.copy()
        else:
            break
    
    # Final validation and return
    corr_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_fitness = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_fitness, n_points)
