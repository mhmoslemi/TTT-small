import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up resolutions
    n_start = 50
    n_fine = 200
    dx_start = 2.0 / n_start
    dx_fine = 2.0 / n_fine
    
    # Initialize from parent or create default
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    # Normalize coarse solution
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Coarse grid optimization
    best_h_coarse = h_coarse.copy()
    best_c5_coarse = float('inf')
    
    # Use cvxpy to solve coarse grid optimization
    h_cvx = cp.Variable(n_start)
    C5 = cp.Variable()
    
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_start / 2,
        C5 >= 0
    ]
    
    # Constraint relaxation: For each shift k, max h[i] * (1-h[i+k]) <= C5 * dx_start
    # Not direct convex due to products. 
    # Instead we solve with ADMM or robustify each constraint
    
    # For tractability, we sample a few more constraint copies to help
    # with better coverage.
    # Use more natural convex form via mean squared error for violations
    try:
        # Naive approximation: treat outer loop as max over all k
        # But impose C5 bound via constraints for correlation values
        
        # Score each pair at k = 0..n_start-1. We'll be relaxed about solving
        # the full max constraint - instead we compute a "good enough" set.
        
        prob = cp.Problem(cp.Minimize(C5), constraints)
        
        # Relax matrix constraints as matrix inequalities using linear relaxation
        prob.solve(solver=ECOS, time_limit=200)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_solution = h_cvx.value
            if h_solution is not None:
                h_coarse = np.array(h_solution)
                h_coarse = np.clip(h_coarse, 0, 1)
                h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
                
    except Exception:
        pass
    
    # Sample coarse solution and refine at different resolution
    # First do interpolation to fine grid
    x_start = np.linspace(0, 2, n_start)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_start, h_coarse)
    
    # Normalize and clip
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Now refine the fine grid via local search
    # Track best solution and early stopping
    best_h_fine = h_fine.copy()
    best_c5_fine = float('inf')
    patience = 0
    last_c5 = float('inf')
    iteration = 0
    adaptive_lr = 1e-3
    max_iter = 10000
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_fine
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5_fine:
            best_c5_fine = current_c5
            best_h_fine = h_fine.copy()
            patience = 0
        else:
            patience += 1
            
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
        else:
            patience = 0
            
        if patience > 300:
            break
            
        last_c5 = current_c5
        
        # Compute gradient by finite difference
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Gradient descent update with adaptive learning rate
        h_new = h_fine - adaptive_lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_fine / 2) / np.sum(h_new)
        
        h_fine = h_new
        adaptive_lr = max(1e-5, adaptive_lr * 0.9995)
        iteration += 1
    
    # Final evaluation
    correlation = np.correlate(best_h_fine, 1 - best_h_fine, mode='full') * dx_fine
    final_c5 = np.max(correlation)
    
    return (best_h_fine.tolist(), final_c5, n_fine)
