import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.rand(n_points)
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    # Optimization parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    min_alpha = 1e-8
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Compute all correlations and find maximum
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        max_val = corr[k_max]
        
        # Compute gradient
        # The gradient w.r.t. h for the maximum correlation is the difference
        # in shifted versions, which we approximate directly from the correlation
        # Using finite difference for clarity and correctness
        grad = np.zeros(n_points)
        
        # Compute correlation gradient using finite difference approach
        # For correlation at lag k_max, this is -> h[i] * (1 - h[i+k]) for i = 0...n-k-1
        # We compute the gradient w.r.t. each h[i] considering how it contributes
        # to the max correlation
        
        # Find indices contributing to max correlation
        n_corr = len(corr)
        if k_max < n_points:
            # Positive lag case
            k = k_max
            for i in range(n_points - k):
                # Contribution to correlation at delay k
                grad[i] += (1 - h[i + k])
                grad[i + k] -= h[i]
        else:
            # Negative lag case (negative lag is technically k_max - n_points)
            k = k_max - n_points
            for i in range(n_points + k):
                grad[i] += (1 - h[i - k])
                grad[i - k] -= h[i]
        
        # Normalize gradient by 2(n_points) and by dx
        grad = grad * dx / (2.0 * n_points)
        
        # Line search for step size
        initial_c5 = c5
        step_h = h - alpha * grad
        step_h = np.clip(step_h, 0, 1)
        step_h = step_h * (n_points / 2) / np.sum(step_h)
        
        # Compute new correlation
        step_corr = correlate(step_h, 1 - step_h, mode='full')
        step_c5 = np.max(step_corr) * dx
        
        # Backtracking line search
        while step_c5 >= initial_c5 and alpha > min_alpha:
            alpha *= line_search_factor
            step_h = h - alpha * grad
            step_h = np.clip(step_h, 0, 1)
            step_h = step_h * (n_points / 2) / np.sum(step_h)
            step_corr = correlate(step_h, 1 - step_h, mode='full')
            step_c5 = np.max(step_corr) * dx
        
        # Update h
        h = step_h.copy()
        c5 = step_c5
        
        # Check for convergence
        if np.linalg.norm(grad) < grad_threshold:
            break
    
    # Return final solution
    return (h.tolist(), c5, n_points)
