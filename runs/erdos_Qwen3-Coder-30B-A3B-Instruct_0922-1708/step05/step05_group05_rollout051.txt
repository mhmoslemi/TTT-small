from scipy.optimize import minimize
import time

    dx = 2.0 / n_points

    initial_alpha = 0.01
    alpha_decay = 0.5
    alpha_grow = 1.2
    perturb_freq = 50
    perturb_magnitude = 0.05
    perturb_decay = 0.95

    if 'initial_h_values' in globals():

        if len(h) != n_points:

            h = np.interp(x_target, x_orig, h)
    else:
        # Default initialization
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial correlation
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    start_time = time.time()
    
    # Projected gradient descent with adaptive step size
    alpha = initial_alpha
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            # Project to feasible region
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            if abs(h_plus_sum - n_points/2) > 1e-8:
                h_plus = h_plus * (n_points/2) / h_plus_sum
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - base_c5) / eps
        
        # Compute candidate step
        h_candidate = h - alpha * grad
        # Project to feasible region
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate_sum = np.sum(h_candidate)
        if abs(h_candidate_sum - n_points/2) > 1e-8:
            h_candidate = h_candidate * (n_points/2) / h_candidate_sum
            
        # Evaluate candidate
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        
        # Accept step if improvement
        if c5_candidate < current_c5:
            h = h_candidate.copy()
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = h.copy()
                best_c5 = c5_candidate
            alpha *= alpha_grow  # Increase step size
        else:
            alpha *= alpha_decay  # Decrease step size
            
        # Apply random perturbations periodically
        if i % perturb_freq == 0 and i > 0:
            perturb_magnitude *= perturb_decay
            noise = np.random.uniform(-perturb_magnitude, perturb_magnitude, n_points)

            # Update current value
            corr_new = np.correlate(h, 1 - h, mode='full')
            current_c5 = np.max(corr_new) * dx
            
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
