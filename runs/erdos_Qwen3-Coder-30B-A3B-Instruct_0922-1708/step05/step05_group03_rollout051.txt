import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute C5 value for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum=h_sum"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) < 1e-10:
        h = np.ones_like(h) * (n_points / 2.0) / n_points
    else:
        h = h * (n_points / 2.0) / h_sum
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def blend_crossover(parent1, parent2, alpha=0.8):
    """Blend crossover with parameter alpha"""
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        gamma = (1 + 2 * alpha) * np.random.random() - alpha
        child1[i] = (1 - gamma) * parent1[i] + gamma * parent2[i]
        child2[i] = gamma * parent1[i] + (1 - gamma) * parent2[i]
    return child1, child2

def mutate(h, mutation_rate=0.1, noise_level=0.03):
    """Mutate h with Gaussian noise"""
    if np.random.random() < mutation_rate:
        noise = np.random.normal(0, noise_level, len(h))
        h = h + noise
    return h

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    """Refine h using gradient descent"""
    h = h.copy()
    for _ in range(max_iter):
        # Compute gradient
        eps = 1e-5
        grad = np.zeros_like(h)
        base_fitness = compute_fitness(h, dx)
        
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            fitness_plus = compute_fitness(h_plus, dx)
            grad[i] = (fitness_plus - base_fitness) / eps
            
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        
        # Early stopping if improvement is minimal
        if abs(compute_fitness(h, dx) - base_fitness) < 1e-8:
            break
            
        # Ensure bounds
        h = np.clip(h, 0, 1)
        h = h * (len(h) / 2.0) / np.sum(h)
        
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        parent = np.array(initial_h_values)
        parent = parent * (n_points / 2.0) / np.sum(parent)
        # Create variations around parent
        for _ in range(population_size // 2):
            h = parent + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        # Add random individuals
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    else:
        # Initialize with random feasible solutions
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Main evolution loop
    gen = 0
    while time.time() - start_time < budget_s - 15:
        gen += 1
        new_population = []
        
        # Elitism: keep best individual
        new_population.append(best_h.copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < 0.8:
                child1, child2 = blend_crossover(parent1, parent2, alpha=0.8)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate(child1, noise_level=0.03)
            child2 = mutate(child2, noise_level=0.03)
            
            # Feasibility projection
            child1 = project_to_feasible(child1, n_points)
            child2 = project_to_feasible(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        # Trim population
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            
        # Periodic gradient refinement every 50 generations
        if gen % 50 == 0 and gen > 0:
            top_k = min(10, len(population) // 5)
            top_indices = np.argsort(fitnesses)[:top_k]
            for i in top_indices:
                refined = gradient_refinement(population[i], dx)
                refined_fitness = compute_fitness(refined, dx)
                if refined_fitness < fitnesses[i]:
                    population[i] = refined
                    fitnesses[i] = refined_fitness
                    if refined_fitness < best_c5:
                        best_c5 = refined_fitness
                        best_h = refined.copy()
    
    # Final refinement of the best solution
    best_h = gradient_refinement(best_h, dx, max_iter=100)
    final_c5 = compute_fitness(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
