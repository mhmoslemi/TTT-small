import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    max_generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Use Halton sequence for initialization
        sampler = qmc.Halton(d=n_points, seed=seed)
        samples = sampler.random(n=population_size - 1)
        for i in range(population_size - 1):
            h = samples[i] * 2  # Scale to [0, 2]
            h = np.clip(h, 0, 1)
            # Normalize to sum = n_points/2
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    # Add the first individual to population if not already added
    if len(population) == 0:
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Ensure we have the right number of individuals
    while len(population) < population_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    def evaluate_fitness(h):
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def repair_h(h):
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-6:
            # Apply proportional adjustment
            scale_factor = (n_points / 2) / current_sum
            h = h * scale_factor
        return h
    
    def tournament_selection(pop, fitnesses, k):
        selected_idx = np.random.choice(len(pop), size=k, replace=False)
        best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
        return pop[best_idx].copy()
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Single-point crossover
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2
    
    def mutate(h, mutation_rate):
        h = h.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Add Gaussian noise
                noise = np.random.normal(0, 0.05)
                h[i] += noise
        return repair_h(h)
    
    # Evaluate initial population
    fitnesses = []
    for i, h in enumerate(population):
        fitnesses.append(evaluate_fitness(h))
    
    start_time = time.time()
    
    # Main GA loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Order population by fitness (lower is better)
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep best individuals
        new_population = sorted_pop[:elite_size]
        new_fitnesses = sorted_fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(sorted_pop, sorted_fitnesses, tournament_size)
            parent2 = tournament_selection(sorted_pop, sorted_fitnesses, tournament_size)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Add to new population
            new_population.append(child1)
            new_fitnesses.append(evaluate_fitness(child1))
            
            if len(new_population) < population_size:
                new_population.append(child2)
                new_fitnesses.append(evaluate_fitness(child2))
        
        # Replace population
        population = new_population[:population_size]
        fitnesses = new_fitnesses[:population_size]
        
        # Adaptive mutation rate
        if generation % 10 == 0:
            mutation_rate *= 0.99
            
        # Verbose output for debugging
        # if generation % 50 == 0:
        #     print(f"Gen {generation}, Best: {min(fitnesses):.6f}")
    
    # Final evaluation
    final_fitnesses = []
    for h in population:
        final_fitnesses.append(evaluate_fitness(h))
    
    best_idx = np.argmin(final_fitnesses)
    best_h = population[best_idx]
    best_fitness = final_fitnesses[best_idx]
    
    return (best_h.tolist(), best_fitness, n_points)
