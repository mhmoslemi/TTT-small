import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 5
    elite_size = 2
    
    # Initialize population
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Use Halton sequence for initialization
        sampler = qmc.Halton(d=n_points, scramble=True, seed=seed)
        samples = sampler.random(n=population_size-1)
        for i in range(population_size - 1):
            h = np.clip(samples[i], 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    # Add default construction
    default_h = np.zeros(n_points)
    default_h[:n_points//2] = 2.0
    default_h = default_h * (n_points / 2) / np.sum(default_h)
    population.append(default_h.copy())
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Evaluate initial population
    fitnesses = []
    for i, h in enumerate(population):
        corr = np.correlate(h, 1 - h, mode='full')
        fitness = np.max(corr) * dx
        fitnesses.append(fitness)
        if fitness < best_c5:
            best_c5 = fitness
            best_h = h.copy()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_idx].copy())
        
        # Create new population
        new_population = []
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for idx in elite_indices:
            new_population.append(population[idx].copy())
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.random() < crossover_rate:
                parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
                parent1, parent2 = selected[parent1_idx], selected[parent2_idx]
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                children = [child1, child2]
            else:
                # Clone one parent
                parent = selected[np.random.randint(population_size)]
                children = [parent.copy()]
            
            # Mutation
            for child in children:
                if np.random.random() < mutation_rate:
                    # Gaussian mutation
                    for i in range(n_points):
                        if np.random.random() < 0.1:  # 10% chance to mutate each gene
                            noise = np.random.normal(0, 0.05)
                            child[i] += noise
                    # Repair constraints
                    child = np.clip(child, 0, 1)
                    child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:population_size]
        
        # Evaluate fitness
        fitnesses = []
        for i, h in enumerate(population):
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitnesses.append(fitness)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = h.copy()
        
        # Adjust mutation rate
        if gen % 10 == 0:
            mutation_rate *= 0.99
    
    # Final refinement
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        # Try random perturbations to improve the best solution
        h_refined = best_h.copy()
        for i in range(n_points):
            h_refined[i] += np.random.normal(0, 0.01)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_fitness = np.max(corr) * dx
        if new_fitness < best_c5:
            best_c5 = new_fitness
            best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
