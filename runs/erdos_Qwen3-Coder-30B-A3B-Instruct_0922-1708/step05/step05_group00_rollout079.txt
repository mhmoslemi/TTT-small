import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with a coarse grid
    n_start = 50
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to satisfy constraint
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Coarse grid optimization with Adam
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    learning_rate = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    last_c5 = float('inf')
    eval_count = 0
    patience = 0
    max_patience = 200
    
    # Coarse grid optimization loop
    for iter_num in range(2000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * (2.0 / n_start)
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-8:
            patience += 1
            if patience > max_patience:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient using centered differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_start):
            h_plus = h.copy()
            h_plus[i] += eps

            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_num + 1))
        v_hat = v / (1 - beta2**(iter_num + 1))
        
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)

    max_patience = 200

    for iter_num in range(2000):

        correlation = np.correlate(h, 1 - h, mode='full') * dx

            patience += 1
            if patience > max_patience:
                break

        last_c5 = current_c5
        
        # Compute finite difference gradient using centered differences

        eps = 1e-4

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(iter_num + 1))
        v_hat = v / (1 - beta2**(iter_num + 1))
        
        h_new = h - learning_rate * m_hat / (np.sqrt(v_hat) + epsilon)

        h_new = np.clip(h_new, 0, 1)

        h = h_new
        
    # Final check with exact evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx

    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:

    final_c5 = np.max(final_correlation)
    
    return (best_h.tolist(), final_c5, n_points)
