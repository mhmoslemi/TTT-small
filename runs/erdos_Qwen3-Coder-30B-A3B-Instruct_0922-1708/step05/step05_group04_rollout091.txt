import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper function to clip values and maintain sum constraint
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if not np.isclose(h_sum, n_points / 2):
            h = h * (n_points / 2) / h_sum
        return h
    
    # Helper function to compute C5
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Refinement using gradient descent
    def refine_h(h, max_iter=100):
        h_refined = h.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            corr = np.correlate(h_refined, 1 - h_refined, mode='full')
            fitness = np.max(corr) * dx
            grad = np.zeros(n_points)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                fitness_plus = np.max(corr_plus) * dx
                grad[i] = (fitness_plus - fitness) / eps
            
            h_refined -= 0.001 * grad
            h_refined = repair(h_refined)

            corr_new = np.correlate(h_refined, 1 - h_refined, mode='full')
            fitness_new = np.max(corr_new) * dx
            if fitness_new >= fitness:
                break
            else:
                fitness = fitness_new
        return h_refined
    
    # Initial refinement
    best_h = refine_h(h)
    best_c5 = compute_c5(best_h)
    
    # Binary search over possible values of C5
    low, high = 0.0, 0.5
    tolerance = 1e-5
    max_iterations = 20
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 10:
            break
        mu = (low + high) / 2.0
        
        # Create variables for optimization
        h_vars = Variable(n_points)
        
        # Define constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            sum(h_vars) == n_points / 2
        ]
        
        # Sample shifts for constraints (more dense sampling near zero)
        shifts = np.linspace(-2, 2, 50)
        
        # Add overlap constraints
        for k in shifts:
            k_idx = int((k + 2) / (4.0 / (n_points - 1)))
            k_idx = max(0, min(n_points - 1, k_idx))
            # For k < 0:
            if k < 0:
                # h[i] * (1 - h[i+k]) for valid i ranges
                overlap = 0
                for i in range(n_points):
                    j = i + k_idx
                    if 0 <= j < n_points:
                        overlap += h_vars[i] * (1 - h_vars[j])
            else:
                # k >= 0
                overlap = 0
                for i in range(n_points):
                    j = i + k_idx
                    if 0 <= j < n_points:
                        overlap += h_vars[i] * (1 - h_vars[j])
            # This is just a placeholder - in practice, a more accurate constraint is needed
            # The constraints we are adding are just the redundancy to speed up the process
            # Preferred to use a cleaner method which is not easy to visualize without better access to Convex Optimization Solver
            
        # For now, we only do fine tuning refinement
        # The current approach is to use the refinement with gradient descent and renewal cycle
        
        # Try improvement using local search
        for _ in range(5):
            candidate_h = refine_h(best_h)
            candidate_c5 = compute_c5(candidate_h)
            if candidate_c5 < best_c5:
                best_h = candidate_h
                best_c5 = candidate_c5
        
        # Update binary search bounds
        if best_c5 <= mu:
            low = mu
        else:
            high = mu

    # Final refinement
    final_h = refine_h(best_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
