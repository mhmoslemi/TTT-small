import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Tracking number of iterations without improvement
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main heuristic search loop
    while time.time() - start_time < budget_s - 10:
        # Generate new candidates
        candidates = []
        num_candidates = 100
        
        for _ in range(num_candidates):
            # Random mutation: flip some elements
            h_mut = best_h.copy()
            num_flips = np.random.randint(1, n_points // 10)
            flip_indices = np.random.choice(n_points, num_flips, replace=False)
            
            # Flip values (0 -> 1, 1 -> 0) with probability 0.5
            for idx in flip_indices:
                if np.random.rand() < 0.5:
                    h_mut[idx] = 1.0 - h_mut[idx]
            
            # Repair to maintain constraints
            h_mut = repair_h(h_mut, n_points)
            candidates.append(h_mut)
        
        # Evaluate candidates
        for h_candidate in candidates:
            c5_candidate = evaluate_h(h_candidate, dx)
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        
        # Early termination if no improvement
        if no_improvement_count >= max_no_improvement:
            break
    
    # Final refinement with gradient descent on best solution
    h_refined = best_h.copy()
    best_c5_refined = best_c5
    
    for _ in range(1000):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_refined) / eps
        
        h_refined = h_refined - 0.0005 * grad
        h_refined = repair_h(h_refined, n_points)
        corr = correlate(h_refined, 1 - h_refined, mode='full')
        best_c5_refined = np.max(corr) * dx
    
    return (h_refined.tolist(), best_c5_refined, n_points)
