import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    trust_region_radius = 0.01
    min_trust_radius = 1e-5
    temperature = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    max_time = time.time() + budget_s - 10
    
    # Initialize from precomputed solution or use a simple construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Simple step function approach: first half = 2, second half = 0
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    h = h * (n_points / 2) / np.sum(h)
    best_h = h.copy()
    
    # Compute initial value of C5
    corr = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(corr) * dx
    
    for iteration in range(max_iterations):
        if time.time() > max_time:
            break
            
        # Compute gradient
        # We only approximate gradient: compute correlation of h with (1-h)
        # The steepest direction is along the sum of lagged correlations
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # At each point, we look at how changing h[i] affects the maximum correlation
            grad_i = 0
            for k in range(n_points):
                # k = 0 to n-1, but report as shifted
                lag_idx1 = (i - k) % n_points
                lag_idx2 = (i + k) % n_points
                grad_i += (1 - h[lag_idx1]) + (1 - h[lag_idx2])
            grad[i] = grad_i
        
        # Normalize gradient
        grad = grad / np.linalg.norm(grad) * trust_region_radius
        
        # Compute proposed new h
        h_new = h - grad
        
        # Project into feasible region
        h_new = np.clip(h_new, 0, 1)
        current_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / current_sum
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept if better or with probability
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            trust_region_radius *= 1.2
        else:
            trust_region_radius *= 0.9
            
        trust_region_radius = max(trust_region_radius, min_trust_radius)
        
        # Apply perturbation from Simulated Annealing
        if iteration % perturb_freq == 0 and temperature > 1e-6:
            # Add Gaussian noise scaled by temperature
            noise = np.random.normal(0, temperature, n_points)
            h_perturbed = h + noise
            
            # Project back into feasible region
            h_perturbed = np.clip(h_perturbed, 0, 1)
            current_sum = np.sum(h_perturbed)
            h_perturbed = h_perturbed * (n_points / 2) / current_sum
            
            corr_pert = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
            c5_pert = np.max(corr_pert) * dx
            
            # Accept with Metropolis probability
            if c5_pert < best_c5 or np.random.rand() < np.exp(-(c5_pert - best_c5) / temperature):
                h = h_perturbed.copy()
                if c5_pert < best_c5:
                    best_c5 = c5_pert
                    
        # Cool down temperature
        temperature *= cooling_rate
        
        # Occasionally refine with heavy gradient descent
        if iteration % 20 == 0:
            # Use limited iterations of gradient descent with smaller step size
            for _ in range(50):
                corr = np.correlate(h, 1 - h, mode='full')
                max_lag = np.argmax(corr)
                
                # Simple gradient calculation just for stabilization
                grad = np.zeros(n_points)
                for i in range(n_points):
                    abs_lag_idx = (i + max_lag) % n_points
                    grad[i] = 1 - h[abs_lag_idx]
                
                h = h - 0.001 * grad
                h = np.clip(h, 0, 1)
                current_sum = np.sum(h)
                h = h * (n_points / 2) / current_sum
                
                # Recompute C5
                corr_new = np.correlate(h, 1 - h, mode='full')
                new_c5 = np.max(corr_new) * dx
                
                if new_c5 < best_c5:
                    best_c5 = new_c5
                else:
                    break
    
    # Final intensive gradient descent
    for _ in range(500):
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(corr)
        
        grad = np.zeros(n_points)
        for i in range(n_points):
            abs_lag_idx = (i + max_lag) % n_points
            grad[i] = 1 - h[abs_lag_idx]
            
        h = h - 0.001 * grad
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
        
        corr_new = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
        else:
            break
            
    return (h.tolist(), best_c5, n_points)
