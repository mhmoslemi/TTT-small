import numpy as np
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_start = 50
    n_fine = 200
    dx_start = 2.0 / n_start
    dx_fine = 2.0 / n_fine
    
    # Initialize
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    # Normalize coarse h to meet constraint ∫ h(x) dx = 1
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Store best solution
    best_h_coarse = h_coarse.copy()
    best_c5 = float('inf')
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    lr = 0.01
    
    # Initialize Adam moments
    m = np.zeros_like(h_coarse)
    v = np.zeros_like(h_coarse)
    
    # Coarse grid optimization
    iterations = 0
    patience = 0
    last_c5 = float('inf')
    
    while iterations < 2000 and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_start
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_coarse = h_coarse.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Estimate gradient using centered finite differences
        grad = np.zeros_like(h_coarse)
        eps_fd = 1e-4
        
        for i in range(n_start):
            h_plus = h_coarse.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_start
            c5_plus = np.max(cor_plus)
            
            h_minus = h_coarse.copy()
            h_minus[i] -= eps_fd
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_start / 2) / np.sum(h_minus)
            
            cor_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx_start
            c5_minus = np.max(cor_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps_fd)
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(iterations+1))
        v_hat = v / (1 - beta2**(iterations+1))
        
        h_new = h_coarse - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h_coarse = h_new
        iterations += 1
    
    # Transfer to fine grid
    x_coarse = np.linspace(0, 2, n_start)
    x_fine = np.linspace(0, 2, n_fine)
    h_fine = np.interp(x_fine, x_coarse, best_h_coarse)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Store best solution
    best_h_fine = h_fine.copy()
    best_c5 = float('inf')
    
    # Initialize Adam moments for fine grid
    m = np.zeros_like(h_fine)
    v = np.zeros_like(h_fine)
    
    # Fine grid optimization
    iterations = 0
    patience = 0
    last_c5 = float('inf')
    
    while iterations < 2000 and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_fine
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h_fine = h_fine.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Estimate gradient using centered finite differences
        grad = np.zeros_like(h_fine)
        eps_fd = 1e-4
        
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            h_minus = h_fine.copy()
            h_minus[i] -= eps_fd
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_fine / 2) / np.sum(h_minus)
            
            cor_minus = np.correlate(h_minus, 1 - h_minus, mode='full') * dx_fine
            c5_minus = np.max(cor_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps_fd)
        
        # Adam update with lower learning rate for fine grid
        lr_fine = 0.005
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        
        m_hat = m / (1 - beta1**(iterations+1))
        v_hat = v / (1 - beta2**(iterations+1))
        
        h_new = h_fine - lr_fine * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_fine / 2) / np.sum(h_new)
        
        h_fine = h_new
        iterations += 1
    
    # Final validation and return
    correlation = np.correlate(best_h_fine, 1 - best_h_fine, mode='full') * dx_fine
    final_c5 = np.max(correlation)
    
    return (best_h_fine.tolist(), final_c5, n_fine)
