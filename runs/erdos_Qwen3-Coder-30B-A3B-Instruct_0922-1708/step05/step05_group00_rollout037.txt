import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Multi-start with both initial_h_values and random restarts
    initial_solutions = []
    
    # First, try using the provided parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    else:
        # Standard uniform initialization
        h = np.ones(n_points) * 0.5
        initial_solutions.append(h.copy())
    
    # Generate 4 random restarts
    for _ in range(4):
        h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        initial_solutions.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    for h_init in initial_solutions:
        h = h_init.copy()
        
        # Convex relaxation step to get a better starting point
        try:
            h_relax = cp.Variable(n_points)
            # Relax the product term using epigraph formulation
            # This is a heuristic approach for the non-convex problem
            
            # Constrain sum to be exactly n_points/2
            constraints = [
                h_relax >= 0,
                h_relax <= 1,
                cp.sum(h_relax) == n_points / 2
            ]
            
            # Use a simple convex relaxation - just minimize sum of values with some penalty
            # We'll minimize sum(h) subject to normalization, which gives a quick good bound
            prob = cp.Problem(cp.Minimize(cp.sum(h_relax)), constraints)
            
            prob.solve(solver=ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h = h_relax.value
                if h is not None:
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
        except Exception:
            pass  # Keep the original h_init
            
        # Refinement with L-BFGS-B
        def objective(h_vec):
            # Compute correlation using np.correlate
            correlation = np.correlate(h_vec, 1-h_vec, mode='full') * dx
            return np.max(correlation)
            
        def gradient(h_vec):
            # Approximate gradient via finite differences for numerical stability
            eps = 1e-5
            grad = np.zeros_like(h_vec)
            f0 = objective(h_vec)
            for i in range(n_points):
                h_plus = h_vec.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                f_plus = objective(h_plus)
                grad[i] = (f_plus - f0) / eps
            return grad
            
        # Project back onto feasible set 
        def project_h(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            return h_vec
        
        # Minimize using L-BFGS-B with bounds [0,1]
        h_opt = h.copy()
        max_iter = 2000
        patience = 0
        last_c5 = float('inf')
        
        # Discrete optimization for binary or modular problems
        for iteration in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            c5 = objective(h_opt)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
            if abs(c5 - last_c5) < 1e-6:
                patience += 1
                if patience > 100:
                    break
            else:
                patience = 0
            last_c5 = c5
            # Compute gradient and perform update
            grad = gradient(h_opt)
            # Manual gradient update, project after each step  
            lr = 0.001
            h_opt = h_opt - lr * grad
            h_opt = project_h(h_opt)
            
        # Return the best solution found
        if best_h is not None:
            correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
            final_c5 = np.max(correlation)
            if final_c5 < best_c5:
                best_c5 = final_c5
        
        # Maximum iterations based on available budget
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final refinement if we found a solution
    if best_h is None:
        # Fall back to uniform initialization if nothing was found 
        best_h = np.ones(n_points) * 0.5
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
    return (best_h.tolist(), final_c5, n_points)
