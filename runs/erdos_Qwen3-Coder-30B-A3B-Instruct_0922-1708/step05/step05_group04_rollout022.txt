import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        # Initialize with a simple construction
        h_init = np.ones(n_points) * 0.5
    
    # Convert to Fourier coefficients
    def h_to_fourier(h):
        # Pad to n_points for FFT
        h_padded = np.pad(h, (0, n_points), mode='constant')
        fft_coeffs = fft(h_padded)
        # Return real and imaginary parts of first n_fourier coefficients
        return np.concatenate([fft_coeffs[:n_fourier].real, fft_coeffs[:n_fourier].imag])
    
    def fourier_to_h(coeffs):
        # Reconstruct from fourier coeffs
        real_coeffs = coeffs[:n_fourier]
        imag_coeffs = coeffs[n_fourier:]
        # Ensure we have the right number of coefficients for real signal
        fft_coeffs = real_coeffs + 1j * imag_coeffs
        # Since we want a real signal, use symmetry
        # Original FFT for n_points: 0, 1, ..., n_points-1
        # For real signal, FFT(k) = conj(FFT(n_points-k))
        # We simulate this by making sure half is mirror
        full_coeffs = np.zeros(n_points * 2, dtype=complex)
        full_coeffs[:n_fourier] = fft_coeffs
        full_coeffs[n_points:n_points+n_fourier] = fft_coeffs[:0:-1].conj()
        h_rec = ifft(full_coeffs)
        h_ret = h_rec[:n_points].real
        # Clip and rescale
        h_ret = np.clip(h_ret, 0, 1)
        h_ret = h_ret * (n_points / 2) / np.sum(h_ret)
        return h_ret
    
    # Objective function: compute C5 from Fourier coefficients
    def objective(coeffs):
        h = fourier_to_h(coeffs)
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Gradient approximation with finite differences
    def gradient_approx(coeffs, eps=1e-4):
        grad = np.zeros_like(coeffs)
        base_c5 = objective(coeffs)
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            c5_plus = objective(coeffs_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Initial Fourier coefficients
    init_coeffs = h_to_fourier(h_init)
    
    # Optimization loop
    best_c5 = float('inf')
    best_coeffs = init_coeffs.copy()
    stagnation_count = 0
    max_stagnation = 100
    
    current_coeffs = init_coeffs.copy()
    last_c5 = objective(current_coeffs)
    
    # Step size
    lr = 0.01
    
    # Main optimization loop
    iters = 0
    while time.time() - start_time < budget_s - 10 and stagnation_count < max_stagnation:
        # Compute gradient
        grad = gradient_approx(current_coeffs)
        
        # Update coefficients
        current_coeffs -= lr * grad
        
        # Constraint maintaining
        current_coeffs = np.clip(current_coeffs, -10, 10)
        
        # Evaluate objective
        new_c5 = objective(current_coeffs)
        iters += 1
        
        # Check improvement
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_coeffs = current_coeffs.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

    # Final refinement with local optimization
    h_final = fourier_to_h(best_coeffs)
    h_best = h_final.copy()
    c5_best = best_c5
    
    # Local refinement with gradient descent
    for _ in range(50):
        # Compute correlation
        correlation = np.correlate(h_best, 1 - h_best, mode='full')
        base_c5 = np.max(correlation) * dx
        
        # Estimate gradient with finite differences
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_best.copy()
            h_plus[i] += eps
            # Correct the scaling to maintain sum constraint
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - base_c5) / eps
        
        # Update
        h_best -= 0.001 * grad
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best * (n_points / 2) / np.sum(h_best)
        
        # Check if it improved
        corr = np.correlate(h_best, 1 - h_best, mode='full')
        new_c5 = np.max(corr) * dx
        if new_c5 < c5_best:
            c5_best = new_c5
        else:
            break

    return (h_best.tolist(), c5_best, n_points)
