import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    k_grid = np.linspace(-2, 2, 50)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.random.uniform(0, 1, n_points)
    
    # Scale to meet constraint sum(h) = n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_best = h_init.copy()
    c5_best = float('inf')
    
    def compute_c5(h):
        # Compute the overlap integral maximum
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Repair function to maintain constraints
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) > 1e-10:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Binary search on target overlap value
    low = 0.0
    high = 0.5
    tolerance = 1e-5
    
    # Comparison function for C5 value
    def compare_c5(h, target):
        # Check if all k shifts satisfy constraint
        for k in k_grid:
            idx = int((k + 2) / (4.0 / n_points))
            if idx < 0 or idx >= n_points:
                continue
            # Represents the number of overlapping units, simplified
            # For now, just compute one direct value
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx <= target
    
    # Gradient descent refinement
    def refine_solution(h_start, max_iter=100):
        h = h_start.copy()
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
            # Compute gradient numerically
            grad = np.zeros(n_points)
            eps = 1e-4
            base_corr = np.correlate(h, 1 - h, mode='full')
            base_c5 = np.max(base_corr) * dx
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = repair(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - base_c5) / eps
            
            # Update step
            h -= 0.001 * grad
            h = repair(h)
            
            # Check convergence
            new_corr = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(new_corr) * dx
            if abs(new_c5 - base_c5) < 1e-8:
                break
        return h
    
    # Main optimization loop
    iterations = 0
    while time.time() - start_time < budget_s - 10 and iterations < 20:
        # Binary search step
        target = (low + high) / 2.0
        
        # Try to improve h for this target using gradient descent
        # Initialize with current best solution
        h_current = h_best.copy()
        
        # First try with tracked best, then random starting points
        h_ref = h_current.copy()
        h_ref = refine_solution(h_ref, 100)
        
        test_c5 = compute_c5(h_ref)
        if test_c5 <= target and test_c5 < c5_best:
            h_best = h_ref.copy()
            c5_best = test_c5
        
        # Adjust binary search
        if test_c5 > target:
            low = target
        else:
            high = target
            
        iterations += 1
        
        # Occasionally take a random restart
        if iterations % 5 == 0:
            h_random = np.random.uniform(0, 1, n_points)
            h_random = h_random * (n_points / 2) / np.sum(h_random)
            h_random = refine_solution(h_random, 50)
            test_c5_random = compute_c5(h_random)
            if test_c5_random < c5_best:
                c5_best = test_c5_random
                h_best = h_random.copy()
    
    # Final refinement
    h_final = refine_solution(h_best, 100)
    final_c5 = compute_c5(h_final)
    
    # Ensure final solution meets constraints
    h_final = repair(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
