import numpy as np
from scipy.stats import qmc
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate_init = 0.1
    tournament_size = 5
    elite_size = 5
    
    # Initialize population with Halton sequences or initial construction
    sampler = qmc.Halton(d=n_points, seed=seed)
    points = sampler.random(n=population_size)
    
    population = np.zeros((population_size, n_points))
    for i in range(population_size):
        # Map Halton points to [0, 1] and scale to satisfy sum constraint
        h = points[i]
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population[i] = h
    
    # If initial construction exists, use it as first individual
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population[0] = h_init
    
    best_individual = None
    best_fitness = float('inf')
    
    # Main GA loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness for all individuals
        fitness_scores = []
        for h in population:
            corr = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(corr) * dx
            fitness_scores.append(fitness)
        
        fitness_scores = np.array(fitness_scores)
        
        # Track best individual
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_fitness:
            best_fitness = fitness_scores[min_idx]
            best_individual = population[min_idx].copy()
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fitness = [fitness_scores[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create next generation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitness_scores)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            # Select parents
            parent1 = selected[np.random.randint(0, population_size)]
            parent2 = selected[np.random.randint(0, population_size)]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            mutation_rate = mutation_rate_init * (0.99 ** (gen // 10))
            
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.01)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.01)
            
            # Repair constraints
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        population = np.array(new_population[:population_size])
    
    # Final evaluation of best individual
    if best_individual is not None:
        corr = np.correlate(best_individual, 1 - best_individual, mode='full')
        final_fitness = np.max(corr) * dx
        return (best_individual.tolist(), final_fitness, n_points)
    else:
        # Fallback to last evaluated population
        corr = np.correlate(population[0], 1 - population[0], mode='full')
        fitness = np.max(corr) * dx
        return (population[0].tolist(), fitness, n_points)
