import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    max_iter = 5000
    learning_rate = 1e-3
    learning_rate_decay = 0.999
    min_learning_rate = 1e-6
    plateau_patience = 200
    perturb_fraction = 0.01
    
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization loop
    last_improvement = 0
    current_lr = learning_rate
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Compute overlaps for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        max_lag = np.argmax(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute gradient for the maximum lag
        grad = np.zeros(n_points)
        # The lag k corresponds to indices k to k + n_points - 1 in the full correlation
        # We want to identify the positions in the overlap and compute the gradient
        # For lag k in full correlation, k = 0 to 2*n_points - 2
        # The actual lag is k - (n_points - 1), which runs from -(n_points-1) to (n_points-1)
        actual_lag = max_lag - (n_points - 1)
        
        # For the correlation point max_lag, we have:
        # correlation[k] = sum_{i=0}^{n_points-1} h[i] * (1 - h[i + k - (n_points-1)])
        # We need gradient terms for h[i] which affect this sum.
        # Gradient w.r.t. h[i] = (1 - h[i + k - (n_points-1)]) - h[i + k - (n_points-1)]
        
        # The derivation is more intricate, let's compute finite differences gradient
        h_plus = h.copy()
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus[i] = h[i] + eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus_rescaled = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus_rescaled, 1 - h_plus_rescaled, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
            h_plus[i] = h[i]  # reset
        
        # Gradient descent step
        h = h - current_lr * grad
        h = np.clip(h, 0, 1)
        
        # Rescale to sum = n_points/2
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Adapt learning rate
        if iteration - last_improvement > plateau_patience:
            current_lr *= learning_rate_decay
            if current_lr < min_learning_rate:
                current_lr = min_learning_rate
            # Add perturbation
            noise = np.random.normal(0, perturb_fraction * current_lr, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
