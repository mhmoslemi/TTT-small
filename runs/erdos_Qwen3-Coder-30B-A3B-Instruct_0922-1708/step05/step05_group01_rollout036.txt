import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Compute initial correlation
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Gradient-based optimization
    max_iter = 6000
    learning_rate = 0.01
    stagnation_count = 0
    max_stagnation = 200
    adjustment_freq = 50
    perturbation_freq = 100
    perturbation_magnitude = 0.01
    
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        # Adaptive learning rate
        if iteration > 0 and iteration % 200 == 0:
            learning_rate *= 0.5
            
        # Compute current c5 and correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # If improved, update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if no improvement
        if stagnation_count > max_stagnation:
            break
            
        # Every adjustment_freq iterations, do gradient update
        if iteration % adjustment_freq == 0:
            # Compute approximate gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                elif h_plus[i] < 0:
                    h_plus[i] = 0
                
                # Rescale to target sum
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
                
            # Update with gradient
            h = h - learning_rate * grad
            h = np.clip(h, 0, 1)
            
            # Rescale to target sum
            h = h * (n_points / 2) / np.sum(h)
            
        # Apply random perturbation occasionally
        if iteration % perturbation_freq == 0:
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            
            # Rescale to target sum
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
