import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Helper to compute C5
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    # Initial evaluation
    c5 = compute_c5(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Optimization parameters
    max_iter = 6000
    lr = 0.01
    stagnation_threshold = 200
    perturb_freq = 100
    eval_freq = 50
    
    last_improvement = 0
    for iteration in range(max_iter):
        if (time.time() - start_time) >= budget_s - 5:
            break
            
        if iteration % eval_freq == 0:
            # Compute all correlations
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement = iteration
        
        # Adaptive learning rate
        if iteration > 0 and iteration % 200 == 0:
            lr *= 0.5
        
        # Identify critical positions
        max_lag = np.argmax(corr)
        max_corr_val = corr[max_lag]
        
        # Gradient approximation - we examine how the correlation changes when we change each h[i]
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] = min(1.0, h[i] + eps)
            # Rescale
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        # Rescale to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally add random perturbation to escape local minima
        if iteration % perturb_freq == 0:
            perturb = np.random.normal(0, 0.005, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
