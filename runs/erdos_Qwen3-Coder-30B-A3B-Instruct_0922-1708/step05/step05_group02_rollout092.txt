import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def levy_flight(scale=0.5, alpha=1.5):
    """Generate a Lévy flight step."""
    # Using stable distribution properties for α = 1.5
    # For computational simplicity, using a truncated Cauchy distribution
    return np.random.standard_cauchy() * scale

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population with diverse strategies
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
    
    # Add more diverse initial particles
    for _ in range(49):
        # Mix of random and structured patterns
        if np.random.random() < 0.5:
            # Random
            h = np.random.rand(n_points)
        else:
            # Step pattern
            h = np.zeros(n_points)
            pos = np.random.randint(0, n_points)
            h[pos] = 1.0
            # Add some variance
            for i in range(max(0, pos-2), min(n_points, pos+3)):
                if np.random.random() < 0.3:
                    h[i] = np.random.rand() * 0.5 + 0.25
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Evaluate initial population
    fitness = [evaluate_h(p, dx) for p in population]
    
    # Track best
    best_idx = np.argmin(fitness)
    best_h = population[best_idx].copy()
    best_value = fitness[best_idx]
    
    # Design parameters
    max_iter = 1000
    top_k = 10
    perturb_rate = 0.1
    levy_scale = 0.5
    lr = 0.001
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        iter_count += 1
        
        # Generate new population via Lévy flights
        new_population = []
        
        # Keep best solutions
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:top_k]]
        new_population.extend(elite)
        
        # Generate rest via Levy flights
        for i in range(50):  # Generate 50 new candidates
            # Select parent from top performers
            parent = population[np.random.choice(sorted_indices[:20])]
            
            # Create perturbed version
            new_h = parent.copy()
            
            # Apply Lévy flight perturbations
            for j in range(n_points):
                if np.random.random() < perturb_rate:
                    step = levy_flight(levy_scale)
                    new_h[j] += step
                    
            # Repair to constraints
            new_h = repair_h(new_h, n_points)
            new_population.append(new_h)
        
        # Add gradient based improvements for top performers
        sorted_indices = np.argsort([evaluate_h(p, dx) for p in new_population])
        for i in range(top_k):
            idx = sorted_indices[i]
            h = new_population[idx].copy()
            current_value = evaluate_h(h, dx)
            
            # Simple gradient estimation via finite differences
            eps = 1e-6
            grad = np.zeros(n_points)
            for j in range(n_points):
                h_plus = h.copy()
                h_plus[j] += eps
                h_plus = repair_h(h_plus, n_points)
                corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - current_value) / eps
                
            # Update with gradient
            h = h - lr * grad
            h = repair_h(h, n_points)
            
            # Accept if better
            new_value = evaluate_h(h, dx)
            if new_value < current_value:
                new_population[idx] = h
            
        # Evaluate new population
        fitness = [evaluate_h(p, dx) for p in new_population]
        
        # Update best
        current_best_idx = np.argmin(fitness)
        if fitness[current_best_idx] < best_value:
            best_h = new_population[current_best_idx].copy()
            best_value = fitness[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Replace worst with new random
        if no_improvement_count > 50:
            worst_idx = np.argmax(fitness)
            h_new = np.random.rand(n_points)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            new_population[worst_idx] = h_new
            fitness[worst_idx] = evaluate_h(h_new, dx)
        
        if no_improvement_count > max_no_improvement:
            break
            
        population = new_population
    
    # Final gradient refinement on best solution
    h_refined = best_h.copy()
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_value) / eps
            
        h_refined = h_refined - 0.001 * grad
        h_refined = repair_h(h_refined, n_points)
        best_value = evaluate_h(h_refined, dx)
    
    return (h_refined.tolist(), best_value, n_points)
