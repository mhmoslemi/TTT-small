import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    initial_lr = 0.01
    perturb_freq = 50
    perturb_decay = 100.0
    perturb_magnitude = 0.1
    tolerance = 1e-8
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based approach with perturbations
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate current C5
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        max_idx = np.argmax(base_corr)
        max_val = base_corr[max_idx]
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Apply gradient descent
        lr = initial_lr / (1.0 + iteration / 100.0)
        h = h - lr * grad
        
        # Project back to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > tolerance:
            h = h * (n_points / 2) / h_sum
            
        # Apply random perturbations periodically
        if iteration % perturb_freq == 0:
            perturb_mag = perturb_magnitude / (1.0 + iteration / perturb_decay)
            noise = np.random.uniform(-perturb_mag, perturb_mag, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final eval of best solution
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
