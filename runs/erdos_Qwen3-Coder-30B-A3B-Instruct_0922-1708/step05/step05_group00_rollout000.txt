import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with coarse resolution to explore efficiently
    n_start = 50
    n_points = 200  # Finer grid for better precision
    dx = 2.0 / n_points
    dx_start = 2.0 / n_start
    
    # Initialize h - use provided construction or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_start:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_start)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_start) * 0.5
    
    # Normalize to satisfy sum constraint
    h = h * (n_start / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Adaptive Adam optimizer parameters
    lr = 0.01
    beta1 = 0.9
    beta2 = 0.999
    epsilon = 1e-8
    
    # Initial Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Coarse optimization phase
    coarse_iter = 0
    max_coarse_iter = 2000
    patience = 0
    last_c5 = float('inf')
    
    while coarse_iter < max_coarse_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        # Compute correlations using full convolution
        h_extended = np.concatenate([h, h])
        h1_minus = 1 - h
        h1_minus_extended = np.concatenate([h1_minus, h1_minus])
        
        # Compute correlation for all shifts
        correlation = np.correlate(h_extended, h1_minus_extended, mode='valid')
        correlation = correlation[:len(h)] * dx_start
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_start):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            # Recompute correlation
            h_plus_extended = np.concatenate([h_plus, h_plus])
            h1_plus = 1 - h_plus
            h1_plus_extended = np.concatenate([h1_plus, h1_plus])
            
            cor_plus = np.correlate(h_plus_extended, h1_plus_extended, mode='valid')
            cor_plus = cor_plus[:len(h)] * dx_start
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(coarse_iter + 1))
        v_hat = v / (1 - beta2**(coarse_iter + 1))
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_start / 2) / np.sum(h)
        
        coarse_iter += 1
        
        # Debug information
        if coarse_iter % 500 == 0:
            print(f"Coarse iter {coarse_iter}, C5 = {current_c5:.6f}")
    
    # Transfer to fine grid
    # Interpolate best coarse solution to fine grid
    h_fine = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_start), best_h)
    h_fine = np.clip(h_fine, 0, 1)
    h_fine = h_fine * (n_points / 2) / np.sum(h_fine)
    
    # Fine grid optimization
    h = h_fine.copy()
    best_h = h.copy()
    best_c5 = float('inf')
    fine_iter = 0
    max_fine_iter = 2000
    patience = 0
    last_c5 = float('inf')
    
    # Reset Adam moments for fine grid
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    while fine_iter < max_fine_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current solution with full correlation
        h_extended = np.concatenate([h, h])
        h1_minus = 1 - h
        h1_minus_extended = np.concatenate([h1_minus, h1_minus])
        
        correlation = np.correlate(h_extended, h1_minus_extended, mode='valid')
        correlation = correlation[:len(h)] * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite difference gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            # Recompute correlation
            h_plus_extended = np.concatenate([h_plus, h_plus])
            h1_plus = 1 - h_plus
            h1_plus_extended = np.concatenate([h1_plus, h1_plus])
            
            cor_plus = np.correlate(h_plus_extended, h1_plus_extended, mode='valid')
            cor_plus = cor_plus[:len(h)] * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**(fine_iter + 1))
        v_hat = v / (1 - beta2**(fine_iter + 1))
        
        h = h - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection to feasible region
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        fine_iter += 1
        
        # Debug info
        if fine_iter % 500 == 0:
            print(f"Fine iter {fine_iter}, C5 = {current_c5:.6f}")
    
    # Final update of best
    h_extended = np.concatenate([best_h, best_h])
    h1_minus = 1 - best_h
    h1_minus_extended = np.concatenate([h1_minus, h1_minus])
    
    correlation = np.correlate(h_extended, h1_minus_extended, mode='valid')
    correlation = correlation[:len(best_h)] * dx
    final_c5 = np.max(correlation)
    
    # Ensure sum constraint is satisfied
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    # Clean up h for final return
    h = best_h.tolist()
    return (h, final_c5, n_points)
