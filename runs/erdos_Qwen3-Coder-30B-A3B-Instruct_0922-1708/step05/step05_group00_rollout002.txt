import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import GLPK, ECOS, SCS
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Configuration
    n_start = 50  # Initial coarse grid
    n_fine = 200  # Final fine grid
    dx_start = 2.0 / n_start
    dx_fine = 2.0 / n_fine
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_coarse = np.array(initial_h_values)
        if len(h_coarse) != n_start:
            x_orig = np.linspace(0, 2, len(h_coarse))
            x_new = np.linspace(0, 2, n_start)
            h_coarse = np.interp(x_new, x_orig, h_coarse)
    else:
        h_coarse = np.ones(n_start) * 0.5
    
    h_coarse = h_coarse * (n_start / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    epsilon = 1e-8
    lr = 0.01
    
    # Adam moment estimates
    m = np.zeros(n_start)
    v = np.zeros(n_start)
    
    best_h = h_coarse.copy()
    best_c5 = float('inf')
    
    # Coarse optimization
    iter_coarse = 0
    max_iter_coarse = 2000
    patience_coarse = 0
    last_c5_coarse = float('inf')
    lr_decay = 0.9999
    
    while iter_coarse < max_iter_coarse and time.time() - start_time < budget_s - 1:
        correlation = np.correlate(h_coarse, 1 - h_coarse, mode='full') * dx_start
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_coarse.copy()
        
        if abs(current_c5 - last_c5_coarse) < 1e-6:
            patience_coarse += 1
            if patience_coarse > 200:
                break
        else:
            patience_coarse = 0
            
        last_c5_coarse = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_coarse)
        eps = 1e-4
        for i in range(n_start):
            h_plus = h_coarse.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_start / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_start
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        
        m_hat = m / (1 - beta1**(iter_coarse + 1))
        v_hat = v / (1 - beta2**(iter_coarse + 1))
        
        h_new = h_coarse - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_start / 2) / np.sum(h_new)
        
        h_coarse = h_new
        lr = max(1e-6, lr * lr_decay)
        iter_coarse += 1
    
    # Interpolate to finer grid
    h_fine = np.interp(np.linspace(0, 2, n_fine), np.linspace(0, 2, n_start), best_h)
    h_fine = h_fine * (n_fine / 2) / np.sum(h_fine)
    h_fine = np.clip(h_fine, 0, 1)
    
    # Fine optimization
    m = np.zeros(n_fine)
    v = np.zeros(n_fine)
    lr = 0.01
    
    best_h = h_fine.copy()
    iter_fine = 0
    max_iter_fine = 2000
    patience_fine = 0
    last_c5_fine = float('inf')
    
    while iter_fine < max_iter_fine and time.time() - start_time < budget_s - 1:
        correlation = np.correlate(h_fine, 1 - h_fine, mode='full') * dx_fine
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_fine.copy()
        
        if abs(current_c5 - last_c5_fine) < 1e-6:
            patience_fine += 1
            if patience_fine > 200:
                break
        else:
            patience_fine = 0
            
        last_c5_fine = current_c5
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_fine)
        eps = 1e-4
        for i in range(n_fine):
            h_plus = h_fine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad * grad
        
        m_hat = m / (1 - beta1**(iter_fine + 1))
        v_hat = v / (1 - beta2**(iter_fine + 1))
        
        h_new = h_fine - lr * m_hat / (np.sqrt(v_hat) + epsilon)
        
        # Projection step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_fine / 2) / np.sum(h_new)
        
        h_fine = h_new
        lr = max(1e-6, lr * lr_decay)
        iter_fine += 1
    
    # Final validation
    final_c5 = best_c5
    best_h = best_h * (n_fine / 2) / np.sum(best_h)  # Ensure constraint
    if abs(np.sum(best_h) * dx_fine - 1.0) > 1e-6:
        best_h = best_h * (n_fine / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_fine)
