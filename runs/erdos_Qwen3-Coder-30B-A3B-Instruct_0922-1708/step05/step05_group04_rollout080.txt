import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 200
    offspring_size = 200
    max_iter = 1000
    initial_mu = 0.5
    mutation_decay = 0.999
    restart_interval = 100
    stagnation_limit = 200
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        # Create diverse initial population
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h_sum = np.sum(h)
            h = h * (n_points / 2) / h_sum
            population.append(h)
    
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        # Adjust based on deficit/excess
        diff = h_sum - n_points / 2
        if diff > 0:
            # Scale down to meet constraint
            h = h * (n_points / 2) / h_sum
        else:
            # Scale up to meet constraint
            h = h * (n_points / 2) / h_sum
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Initial evaluation
    fitnesses = [compute_c5(p) for p in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    mu = initial_mu
    sigma = mu / 5
    
    iter_count = 0
    stagnation_count = 0
    best_fitness_history = [best_fitness]
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 10:
        # Selection: keep top 50%
        selected_indices = np.argsort(fitnesses)[:population_size // 2]
        selected = [population[i] for i in selected_indices]
        selected_fitness = [fitnesses[i] for i in selected_indices]
        
        # Generate offspring
        offspring = []
        for _ in range(offspring_size):
            parent = np.random.choice(selected)
            # Gaussian mutation with adaptive sigma
            delta = np.random.normal(0, sigma, n_points)
            child = parent + mu * delta
            child = repair(child)
            offspring.append(child)
        
        # Evaluate offspring
        offspring_fitness = [compute_c5(o) for o in offspring]
        
        # Combine population and offspring
        all_candidates = population + offspring
        all_fitness = fitnesses + offspring_fitness
        
        # Select best individuals to form new population
        top_indices = np.argsort(all_fitness)[:population_size]
        population = [all_candidates[i] for i in top_indices]
        fitnesses = [all_fitness[i] for i in top_indices]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = population[current_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Store history and decay parameters
        best_fitness_history.append(best_fitness)
        
        # Adjust mutation parameters
        mu *= mutation_decay
        sigma = mu / 5
        
        # Restart mechanism
        if iter_count % restart_interval == 0 and iter_count > 0:
            mu *= 0.9
            
        if stagnation_count > stagnation_limit:
            # Restart population with adapted versions of current best
            mu = initial_mu
            sigma = mu / 5
            stagnation_count = 0
            # Perturb current best and refill population
            best_perturbed = best_h + np.random.normal(0, 0.05, n_points)
            best_perturbed = repair(best_perturbed)
            population = [best_perturbed.copy()]
            population.extend([best_h + np.random.normal(0, 0.05, n_points) for _ in range(population_size - 1)])
            population = [repair(p) for p in population]
            # Re-evaluate population
            fitnesses = [compute_c5(p) for p in population]
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_fitness = fitnesses[best_idx]
        
        iter_count += 1
    
    # Final gradient-based refinement of best candidate
    h_ref = best_h.copy()
    
    def objective(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient(h):
        eps = 1e-5
        grad = np.zeros_like(h)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_fitness = np.max(base_corr) * dx
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - base_fitness) / eps
        return grad
    
    # Gradient descent
    for _ in range(100):
        grad = gradient(h_ref)
        h_ref -= 0.001 * grad
        h_ref = repair(h_ref)
        
        if _ % 20 == 0:
            current_fitness = objective(h_ref)
            if current_fitness >= best_fitness * 1.001:  # Early stopping if no improvement
                break
    
    # Final check
    final_corr = np.correlate(h_ref, 1 - h_ref, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (h_ref.tolist(), final_c5, n_points)
