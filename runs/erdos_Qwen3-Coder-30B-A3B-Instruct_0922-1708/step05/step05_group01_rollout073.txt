import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Compute initial C5
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    
    # Track best solution
    best_h = h.copy()
    best_c5 = c5
    
    # Gradient descent parameters
    lr = 1e-3
    penalty_weight = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.5
    
    # Penalty parameter
    penalty_mu = penalty_weight
    
    # Track convergence
    last_improvement = 0
    plateau_count = 0
    
    iteration = 0
    while (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlation for all lags
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        # Find the lag that maximizes the overlap
        max_idx = np.argmax(corr)
        max_lag = max_idx - (n_points - 1)
        
        # Compute gradient for current lag only
        # This is a simplified approach: only influence the terms 
        # that contribute to the maximum correlation
        grad_h = np.zeros_like(h)
        
        # Handle the case where max_lag is positive (h[i] and h[i-max_lag] are correlated)
        if max_lag > 0:
            for i in range(max_lag, n_points):
                grad_h[i] += (1 - h[i - max_lag])
                grad_h[i - max_lag] += (1 - h[i])
        else:
            # max_lag < 0
            abs_lag = -max_lag
            for i in range(abs_lag, n_points):
                grad_h[i] += (1 - h[i - abs_lag])
                grad_h[i - abs_lag] += (1 - h[i])
                
        # Gradient of penalty term
        penalty_grad = np.zeros_like(h)
        # Grad for bounds
        for i in range(n_points):
            if h[i] < 0:
                penalty_grad[i] += -2 * penalty_mu * h[i]
            elif h[i] > 1:
                penalty_grad[i] += -2 * penalty_mu * (h[i] - 1)
        
        # Grad for sum constraint
        sum_violation = np.sum(h) - n_points / 2
        for i in range(n_points):
            penalty_grad[i] += 2 * penalty_mu * sum_violation
        
        # Combine gradients
        total_grad = grad_h + penalty_grad
        
        # Apply learning rate
        h_new = h - lr * total_grad
        
        # Repair feasibility
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        if abs(h_sum - n_points/2) > 1e-6:
            h_new = h_new * (n_points / 2) / h_sum
            
        # Check for constraint violations and update penalty
        sum_violation_new = np.sum(h_new) - n_points / 2
        bounds_violation_new = np.sum(np.maximum(0, -h_new)) + np.sum(np.maximum(0, h_new - 1))
        
        if bounds_violation_new > 0 or abs(sum_violation_new) > 1e-6:
            penalty_mu = penalty_mu * penalty_update_factor
        else:
            penalty_mu = max(penalty_mu / penalty_update_factor, penalty_min)
        
        # Replace h with h_new
        h = h_new
        
        # Update best solution
        corr_new = correlate(h, 1 - h, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iteration
            plateau_count = 0
        else:
            plateau_count += 1
            
        # Check for stagnation and apply fallback
        if plateau_count > plateau_patience:
            if lr > 1e-5:
                lr *= learning_rate_decay
            if iteration - last_improvement > 300:
                # Reinitialize with a random perturbation
                h = np.random.rand(n_points) * 0.5
                h = h * (n_points / 2) / np.sum(h) 
                plateau_count = 0
                lr = 1e-3
        
    return (best_h.tolist(), best_c5, n_points)
