import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Initialize coarse grid if needed
    n_start = 50
    if n_points > n_start:
        x_orig = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_start)
        h = np.interp(x_new, x_orig, h)
        n_points = n_start

    dx = 2.0 / n_points
    
    # Normalize to constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # LP-based initialization using CVXPY with a simpler approach 
    try:
        h_cvx = cp.Variable(n_points)
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        # Simple pooling objective -- try to minimize average overlap
        # This won't solve exact problem but sets up at least feasible direction:
        prob = cp.Problem(cp.Minimize(cp.sum(h_cvx)), constraints)
        prob.solve(solver=ECOS, verbose=False)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h = np.array(h_cvx.value)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass

    # Coarse optimization with quadratic programming approach
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimize coarse grid
    max_iter = 2000
    lr = 0.01
    patience = 0
    last_c5 = float('inf')
    
    # Multi-scale optimization: start with coarse grid
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check stopping condition
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 200:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # FD gradient estimate
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with clipping and normalization
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        lr = max(1e-6, lr * 0.999)

    # Refine on fine grid if needed
    if len(best_h) < 100:
        # Interpolate to fine grid
        x_fine = np.linspace(0, 2, 100)
        x_coarse = np.linspace(0, 2, len(best_h))
        h_coarse = best_h
        h_fine = np.interp(x_fine, x_coarse, h_coarse)
        n_points = 100
        dx = 2.0 / n_points
        h = h_fine
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        best_h = h.copy()
        best_c5 = float('inf')
        
        # Fine grid optimization
        max_iter = 3000
        lr = 0.005
        last_c5 = float('inf')
        patience = 0
        
        for iter_count in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            correlation = np.correlate(h, 1 - h, mode='full') * dx
            current_c5 = np.max(correlation)
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                
            if abs(current_c5 - last_c5) < 1e-6:
                patience += 1
                if patience > 200:
                    break
            else:
                patience = 0
                
            last_c5 = current_c5
            
            # Finite difference gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
                c5_plus = np.max(cor_plus)
                
                grad[i] = (c5_plus - current_c5) / eps
                
            # Update
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            lr = max(1e-6, lr * 0.9995)

    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Exact constraint enforcement
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
