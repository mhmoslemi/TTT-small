import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points_list = [10, 20, 40, 80, 100]
    best_h = None
    best_c5 = float('inf')
    
    # Parent fallback
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = np.clip(h_parent, 0, 1)
        h_parent = h_parent * (100 / 2) / np.sum(h_parent)
        # Compute C5 for parent
        corr = np.correlate(h_parent, 1 - h_parent, mode='full')
        c5_parent = np.max(corr) * (2.0 / 100)
        if c5_parent < best_c5:
            best_c5 = c5_parent
            best_h = h_parent.copy()
    
    start_time = time.time()
    
    # Scale pyramid optimization
    h_prev = None
    for n_points in n_points_list:
        if time.time() - start_time > budget_s - 100:
            break
            
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5  # Initial guess
        if h_prev is not None:
            # Upsample from previous scale
            h_upsample = np.interp(np.linspace(0, 2, n_points), 
                                 np.linspace(0, 2, len(h_prev)), h_prev)
            h = h_upsample.copy()
        
        # Relaxation and optimization
        h_var = Variable(n_points)
        # Set constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)
        
        # Approximate the max constraint with average over middle lags
        # Compute lags from -n/4 to n/4
        half_n = n_points // 4
        lags = list(range(-half_n, 0)) + list(range(1, half_n + 1))
        overlaps = []
        for k in lags:
            if k > 0:
                left = h_var[:-k]
                right = 1 - h_var[k:]
            else:
                left = h_var[-k:]
                right = 1 - h_var[:k]
            overlaps.append(sum(left * right) * dx)
        avg_overlap = sum(overlaps) / len(overlaps)
        
        # Convex optimization
        prob = Problem(Minimize(avg_overlap), constraints)
        try:
            prob.solve(solver=ECOS, verbose=False, max_iters=1000)
        except:
            pass
            
        if prob.status == 'optimal':
            h_opt = np.array(h_var.value).flatten()
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            corr = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_opt = np.max(corr) * dx
            if c5_opt < best_c5:
                best_c5 = c5_opt
                best_h = h_opt.copy()
            h_prev = h_opt
        else:
            h_prev = h
            
    # Local gradient refinement on best solution
    if best_h is not None:
        h_refine = best_h.copy()
        lr = 0.001
        for _ in range(50):
            if time.time() - start_time > budget_s - 10:
                break
            grad = np.zeros_like(h_refine)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_refine.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - best_c5) / eps
            h_refine = h_refine - lr * grad
            h_refine = np.clip(h_refine, 0, 1)
            h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
            corr = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_refine = np.max(corr) * dx
            if c5_refine < best_c5:
                best_c5 = c5_refine
                best_h = h_refine.copy()
    
    # Final check
    if best_h is None:
        # Fallback to uniform
        best_h = np.ones(100) * 0.5  
        best_h = best_h * (100 / 2) / np.sum(best_h)
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr) * (2.0 / 100)
        
    return (best_h.tolist(), best_c5, 100)
