import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent construction or create a simple start
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple piecewise constant function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0

    # Ensure constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)

    # Objective function: return C5 value
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5

    # Project to feasible region
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec

    # Simulated Annealing with adaptive cooling
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    start_time = time.time()
    current_h = h.copy()
    current_c5 = compute_c5(current_h)
    
    # Parameters
    T0 = 10.0
    iterations = 0
    max_iterations = int(1e6)  # Prevent infinite loop
    
    # Main SA loop
    while time.time() - start_time < budget_s - 10 and iterations < max_iterations:
        iterations += 1
        
        # Temperature schedule
        T = T0 / np.log(1 + iterations)
        
        # Mutation: perturb with Gaussian noise
        candidate_h = current_h + np.random.normal(0, 0.02, n_points)
        candidate_h = project(candidate_h)
        candidate_c5 = compute_c5(candidate_h)
        
        # Accept or reject
        delta = candidate_c5 - current_c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            current_h = candidate_h
            current_c5 = candidate_c5
            
        # Update best
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    # Gradient descent local refinement
    h_refine = best_h.copy()
    lr = 0.001
    eps = 1e-4
    
    for _ in range(50):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros_like(h_refine)
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - best_c5) / eps
        
        h_refine = h_refine - lr * grad
        h_refine = project(h_refine)
        new_c5 = compute_c5(h_refine)
        if new_c5 < best_c5:
            best_h = h_refine.copy()
            best_c5 = new_c5

    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
