import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    corr = np.correlate(h, 1 - h, mode='full')
    current_c5 = np.max(corr) * dx
    best_h = h.copy()
    best_c5 = current_c5
    
    # Trust region parameters
    trust_radius = 0.01
    min_trust_radius = 1e-5
    trust_shrink_factor = 0.8
    trust_expand_factor = 1.2
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.99
    perturb_freq = 50
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient estimate by finite differences
        eps = 1e-5
        gradient = np.zeros(n_points)
        base_corr = np.correlate(h, 1 - h, mode='full')
        base_max_k = np.argmax(base_corr)
        base_value = base_corr[base_max_k]
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_k_plus = np.argmax(corr_plus)
            value_plus = corr_plus[max_k_plus]
            
            gradient[i] = (value_plus - base_value) / eps
                
        # Compute candidate step within trust region
        step = -trust_radius * gradient
        
        # Project step to satisfy constraints
        h_new = h + step
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        if new_c5 < current_c5:
            h = h_new.copy()
            current_c5 = new_c5
            trust_radius = min(trust_radius * trust_expand_factor, 0.1)
            if current_c5 < best_c5:
                best_h = h.copy()
                best_c5 = current_c5
        else:
            trust_radius = max(trust_radius * trust_shrink_factor, min_trust_radius)
            
        # Apply Simulated Annealing perturbation every few iterations
        if iteration % perturb_freq == 0:
            perturbation = np.random.normal(0, temp, n_points)
            h_pert = h + perturbation
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            corr_pert = np.correlate(h_pert, 1 - h_pert, mode='full')
            pert_c5 = np.max(corr_pert) * dx
            
            if pert_c5 < current_c5:
                h = h_pert.copy()
                current_c5 = pert_c5
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                
        # Update temperature
        temp *= cooling_rate
    
    # Final refinement with gradient descent
    for _ in range(500):
        if time.time() - start_time > budget_s - 10:
            break
            
        eps = 1e-6
        gradient = np.zeros(n_points)
        base_corr = np.correlate(best_h, 1 - best_h, mode='full')
        base_max_k = np.argmax(base_corr)
        base_value = base_corr[base_max_k]
        
        for i in range(n_points):
            h_plus = best_h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_k_plus = np.argmax(corr_plus)
            value_plus = corr_plus[max_k_plus]
            
            gradient[i] = (value_plus - base_value) / eps
            
        step = -0.001 * gradient
        best_h = best_h + step
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        corr_new = np.correlate(best_h, 1 - best_h, mode='full')
        new_c5 = np.max(corr_new) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            
    # Final validation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
