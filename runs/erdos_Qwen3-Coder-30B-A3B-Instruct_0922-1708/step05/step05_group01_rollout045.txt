import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Constraints: 0 <= h[i] <= 1, sum(h) = n_points/2
    def repair_h(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points/2) > 1e-8:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Compute overlap integral
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Gradient computation using finite differences
    def compute_gradient(h, k_max):
        # Approximate gradient using finite differences
        eps = 1e-5
        g = np.zeros_like(h)
        c5_current = compute_c5(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            if h_plus[i] > 1:
                h_plus[i] = 1
            h_plus = repair_h(h_plus)
            c5_plus = compute_c5(h_plus)
            g[i] = (c5_plus - c5_current) / eps
            
        return g
    
    # Adaptive penalized gradient descent
    learning_rate = 1e-3
    penalty_param = 0.1
    penalty_update_factor = 1.2
    penalty_min = 1e-4
    plateau_patience = 200
    perturbation_fraction = 0.01
    max_iter = 5000
    best_h = h.copy()
    best_c5 = compute_c5(h)
    patience = 0
    last_c5 = best_c5
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Find maximum overlap lag
        corr = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Update with gradient descent
        h = h - learning_rate * grad
        
        # Repair feasibility
        h = repair_h(h)
        
        # Compute new C5
        new_c5 = compute_c5(h)
        
        # Update penalty parameter
        violation = max(0, np.sum(h) - n_points/2)
        if violation > 1e-6:
            penalty_param *= penalty_update_factor
        else:
            penalty_param = max(penalty_param / penalty_update_factor, penalty_min)
        
        # Update best solution
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Early termination or perturbation
        if patience > plateau_patience:
            patience = 0
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                break
            # Perturb solution
            perturb = np.random.normal(0, perturbation_fraction * 0.1, n_points)
            h = h + perturb
            h = repair_h(h)
            
    return (best_h.tolist(), best_c5, n_points)
