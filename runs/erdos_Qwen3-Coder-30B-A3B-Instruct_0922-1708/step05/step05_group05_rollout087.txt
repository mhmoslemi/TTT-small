import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from global if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to sum = 1/2
    h = h * (0.5 / np.sum(h))
    
    # Track best solution
    best_h = h.copy()
    best_c5 = np.max(correlate(h, 1 - h, mode='full')) * dx
    
    start_time = time.time()
    
    # Gradient ascent parameters
    learning_rate = 0.001
    patience = 50
    patience_counter = 0
    last_improvement = 0
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        # Compute current correlation
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        k_min = np.argmax(corr)
        
        # Compute gradient for the minimum correlation value
        # The gradient for entry j of h is the sum of contributions from all shifts k
        # where the correlation is minimized
        grad = np.zeros_like(h)
        
        # For all shifts k in valid range
        for k in range(len(corr)):
            if k >= len(h):
                break
            # Contribution to h[j] is from 1-h[j-k] if k > 0, or h[j+k] if k < 0
            # More correctly, the derivative of correlation is related to:
            # if h*(1-h) correlation at k, then d/dh_i[correlation at k] = 
            # terms where the kernel and signal transition wrt h_i
            # The gradient computation is careful:
            # Let R(k) = sum_j h[j] * (1 - h[j+k])
            # dR/dh_i = (1 - h[i+k]) if k >= 0, 1 - h[i-k] if k < 0
            # So we only add the contributed weights
            
            # We need the partial derivative of correlation R_k = sum_j h[j] * (1 - h[j+k])
            # This means the derivative wrt h[i] is:
            # 1 - h[i+k] if i+k < len(h) (and i+k >= 0), else 0
            # but this is when we have the offset k. For fixed k, we look at:
            # dR_k/dh[i] = (1 - h[i+k]) if k >= 0, else hmm
            
            # Let shift_k = k - (len(h) - 1), then shift_k ranges from -len(h)+1 to len(h)-1
            # So return value of correlate is like R[k] = sum_{j} h[j] * (1 - h[j + shift_k])
            # for shift_k = 0, 1, ..., 2*len(h) - 2
            # When shift_k = k - (len(h) - 1), that's k from the perspective of the shift
            
            shift_k = k - (len(h) - 1)
            
            if shift_k >= 0:
                # for h[j], h[j+shift_k] contributes to R_k => dR_k/dh[j] = 1-h[j+shift_k]
                # but we're adding them up as derivatives of the maximal. Gradient points to
                # direction of *larger* max correlation (not minimal), so we're doing gradient descent
                # on C5, which is max(R) * dx
                # Thus, the direction to move h to decrease this correlation is to 
                # move towards lower 1 - h[j + shift_k]. So that's to move toward higher h[j+shift_k]
                # So if we are at position j, and correlation max is at offset k, we care about what we can
                # scale h[j+shift_k]. 
                # Actually, detailed analysis shows that dC5/dh[j] is:
                # d(C5)/dh[j] = derived from only few terms.
                
                # The gradient direction to reduce the maximum correlation value:
                # We are looking at specific correlation value R_k where k is such that max_k R_k = C5
                # If h[i] contributes to R_k through term h[i] * (1 - h[i + shift_k]) 
                # Then dR_k/dh[i] = 1 - h[i + shift_k] 
                # and for our max, dC5/dh[i] = sign(R_k - C5) * (1 - h[i + shift_k])
                # We'll need to estimate it consistently.
                
                # So loop over all entries, track gradient components
                pass  # Simplified approach - we're treating this as a gradient ascent on C5
                
            else:
                # negative shift k < 0 (shift_k + len(h) - 1 < len(h) - 1 + shift_k)
                pass  # Same logic - not using this direction in this simplified approach
        
        # Simplified approach:
        # Take a less refined gradient attempt - the analytical derivative often leads to a 
        # very involved truncation; we'll go with a numerical estimate by finite diffs,
        # but on just one of the critical k values
        grad = np.zeros_like(h)
        steps = 1e-4
        base_corr = correlate(h, 1 - h, mode='full')
        base_c5 = np.max(base_corr) * dx
        
        # Pick the match as the one that gives maximum correlation (likely tricky to compute via grad directly)
        # So we'll assume latest argmax can be treated as local neighborhood and do backward optimization.
        
        # Direct approach:
        for i in range(len(h)):
            # Compute slight positive perturbation
            h_p = h.copy()
            h_p[i] += steps
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (0.5 / np.sum(h_p))
            corr_p = correlate(h_p, 1 - h_p, mode='full')
            c5_p = np.max(corr_p) * dx
            
            # Gradient estimate from finite differences
            grad[i] = (c5_p - base_c5) / steps
    
        # Function to adjust h to sum constraint
        def project_sum(h):
            current_sum = np.sum(h)
            target_sum = 0.5
            if abs(current_sum - target_sum) < 1e-10:
                return h
            # Rescale
            h = h * target_sum / current_sum
            return h
        
        # Update with gradient step
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = project_sum(h_new)
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Adaptive step size
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            learning_rate = min(learning_rate * 1.1, 0.01)
            patience_counter = 0
            last_improvement = iteration
        else:
            learning_rate = max(learning_rate * 0.5, 1e-6)
            patience_counter += 1
            if patience_counter > patience:
                # Add small random perturbation to escape local minimum
                perturb_mag = 0.01 * np.exp(-iteration / 100)
                h += np.random.normal(0, perturb_mag, len(h))
                h = np.clip(h, 0, 1)
                h = project_sum(h)
                patience_counter = 0
                last_improvement = iteration

        # Normalize to sum constraint
        h = project_sum(h)

        iteration += 1
        
        # Early termination by budget
        if iteration % 100 == 0:
            if time.time() - start_time > budget_s - 5:
                break
    
    # Final validation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
