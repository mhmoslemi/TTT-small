import numpy as np
from scipy.optimize import minimize
from scipy.fftpack import dct, idct
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    n_fourier = 20  # Number of Fourier coefficients to optimize
    
    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init[20:80] = 1.0
    
    # Fourier representation: use DCT-II for real signals
    # Convert initial h to Fourier domain
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    fourier_coeffs = dct(h_init, type=2, norm='ortho')[:n_fourier]
    
    # Optimization variables (Fourier coefficients)
    coeffs = fourier_coeffs.copy()
    
    # Previous best
    best_c5 = float('inf')
    best_coeffs = coeffs.copy()
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 1000
    patience = 100
    stagnation_count = 0
    eps = 1e-4
    
    def compute_c5_from_coeffs(coeffs):
        # Convert back to time domain
        h = idct(coeffs, type=2, norm='ortho')
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        return c5, h
    
    def compute_gradient(coeffs, eps=1e-4):
        grad = np.zeros_like(coeffs)
        base_c5, _ = compute_c5_from_coeffs(coeffs)
        
        for i in range(len(coeffs)):
            coeffs_plus = coeffs.copy()
            coeffs_plus[i] += eps
            c5_plus, _ = compute_c5_from_coeffs(coeffs_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        return grad
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient
        grad = compute_gradient(coeffs, eps)
        
        # Update coefficients
        coeffs -= learning_rate * grad
        
        # Project back to feasible region for working coeffs
        coeffs = np.clip(coeffs, -1, 1)
        
        # Compute current c5
        c5, h = compute_c5_from_coeffs(coeffs)
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_coeffs = coeffs.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Early termination if no improvement
        if stagnation_count > patience:
            break
            
    # Final refinement with gradient descent on best
    coeffs = best_coeffs.copy()
    for _ in range(50):
        if time.time() - start_time > budget_s - 10:
            break
        grad = compute_gradient(coeffs, eps)
        coeffs -= 0.005 * grad
        coeffs = np.clip(coeffs, -1, 1)
        
        c5, h = compute_c5_from_coeffs(coeffs)
        if c5 < best_c5:
            best_c5 = c5
            best_coeffs = coeffs.copy()
    
    # Final conversion to h_values and return
    final_h, _ = compute_c5_from_coeffs(best_coeffs)
    
    return (final_h.tolist(), best_c5, n_points)
