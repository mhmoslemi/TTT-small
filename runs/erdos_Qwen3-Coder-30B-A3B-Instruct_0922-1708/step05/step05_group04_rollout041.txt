import numpy as np
from scipy.optimize import minimize
import time
import math
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    tolerance = 1e-5
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.random.uniform(0, 1, n_points)
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Binary search on the maximum overlap
    low, high = 0.0, 0.5
    best_c5 = float('inf')
    best_h = h.copy()
    
    while high - low > tolerance and time.time() - start_time < budget_s - 10:
        mid = (low + high) / 2.0
        
        # Optimization problem: find h such that constraints are satisfied
        h_var = Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, sum(h_var) == n_points / 2]
        
        # Generate shift samples
        shift_samples = np.linspace(-2, 2, 50)
        
        # Add overlap constraints for all shift samples
        for k in shift_samples:
            # This is a tough constraint, but we approximate it.
            # For now, we'll compute the worst-case overlap in the discrete setting.
            k_idx = int((k + 2) / (4.0 / n_points))
            if k_idx >= 0 and k_idx < n_points:
                # Approximate the correlation constraint
                # Correlation as a sum, not a vector operation due to discrete nature
                corr = 0.0
                for i in range(n_points):
                    j = i + k_idx
                    if 0 <= j < n_points:
                        corr += h_var[i] * (1 - h_var[j])
                constraints.append(corr >= mid)
        
        # Note: Direct optimization with these constraints is complex due to non-convexity
        # Instead, we do a more targeted optimization approach
        try:
            # We'll iteratively improve h using a local search approach with CVX problem
            # For now, just do gradient descent and clipping for feasibility
            
            # Simple local refinement
            h_refined = h.copy()
            for _ in range(100):
                if time.time() - start_time > budget_s - 10:
                    break
                # Compute current C5
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                current_c5 = np.max(corr) * dx
                
                # Gradient estimation
                grad = np.zeros(n_points)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_refined.copy()
                    h_plus[i] += eps
                    # Clip and normalize
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus_sum = np.sum(h_plus)
                    h_plus = h_plus * (n_points / 2) / h_plus_sum
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    fitness_plus = np.max(corr_plus) * dx
                    grad[i] = (fitness_plus - current_c5) / eps
                
                # Update h
                h_refined = h_refined - 0.001 * grad
                # Repair
                h_refined = np.clip(h_refined, 0, 1)
                h_refined_sum = np.sum(h_refined)
                h_refined = h_refined * (n_points / 2) / h_refined_sum
            
            # Evaluate final solution
            corr_final = np.correlate(h_refined, 1 - h_refined, mode='full')
            final_c5 = np.max(corr_final) * dx
            
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = h_refined.copy()
                
            # Update binary search bounds
            if current_c5 >= mid:
                low = mid
            else:
                high = mid
                
        except Exception:
            # If CVX fails, fall back to another heuristic
            high = mid  # Assume not feasible
            
        h = best_h.copy()
    
    # Final refinement pass
    h_final = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 10:
            break
        corr = np.correlate(h_final, 1 - h_final, mode='full')
        current_c5 = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_final.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus)
            h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - current_c5) / eps
        
        h_final = h_final - 0.001 * grad
        h_final = np.clip(h_final, 0, 1)
        h_final_sum = np.sum(h_final)
        h_final = h_final * (n_points / 2) / h_final_sum

    # Final evaluation
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (h_final.tolist(), final_c5, n_points)
