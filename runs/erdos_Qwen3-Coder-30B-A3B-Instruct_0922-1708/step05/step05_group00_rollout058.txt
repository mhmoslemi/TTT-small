import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points_fine = 200
    n_points_coarse = 50
    dx_fine = 2.0 / n_points_fine
    dx_coarse = 2.0 / n_points_coarse
    
    # Initialize with provided parent or uniform
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points_fine:
            x_orig = np.linspace(0, 2, len(h_init))
            x_new = np.linspace(0, 2, n_points_fine)
            h_init = np.interp(x_new, x_orig, h_init)
        h_coarse = np.interp(np.linspace(0, 2, n_points_coarse), x_new, h_init)
    else:
        h_init = np.ones(n_points_fine) * 0.5
        h_coarse = np.ones(n_points_coarse) * 0.5

    h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    best_h = h_coarse.copy()
    best_c5 = float('inf')
    
    # Coarse grid optimization
    h_var = cp.Variable(n_points_coarse)
    C5 = cp.Variable()

    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) == n_points_coarse / 2
    ]

    # LP relaxation with bounds on max correlation
    try:
        prob = cp.Problem(cp.Minimize(C5), constraints)
        prob.solve(solver=ECOS, max_iters=1000, feastol=1e-6)
        if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
            h_solution = h_var.value
            if h_solution is not None:
                h_coarse = np.array(h_solution)
                h_coarse = np.clip(h_coarse, 0, 1)
                h_coarse = h_coarse * (n_points_coarse / 2) / np.sum(h_coarse)
                # Transfer to fine grid
                x_coarse = np.linspace(0, 2, n_points_coarse)
                x_fine = np.linspace(0, 2, n_points_fine)
                h_tmp = np.interp(x_fine, x_coarse, h_coarse)
                h_fine = h_tmp * (n_points_fine / 2) / np.sum(h_tmp)
                h_fine = np.clip(h_fine, 0, 1)
    except Exception:
        h_fine = h_init.copy()
    
    # Refine with gradient descent
    h = h_fine.copy()
    lr_base = 0.01
    lr = lr_base
    patience = 0
    last_c5 = float('inf')
    max_iter = 5000
    total_iter = 0
    
    while total_iter < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate
        correlation = np.correlate(h, 1 - h, mode='full') * dx_fine
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        if abs(current_c5 - last_c5) < 1e-6:
            patience += 1
            if patience > 300:
                break
        else:
            patience = 0
            
        last_c5 = current_c5
        
        # Compute finite diff gradient
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points_fine):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points_fine / 2) / np.sum(h_plus)
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx_fine
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps
            
        # Update with adaptive learning rate
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points_fine / 2) / np.sum(h_new)
        h = h_new
        
        lr = max(1e-5, lr * 0.999)
        total_iter += 1
    
    return (best_h.tolist(), best_c5, n_points_fine)
