import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 5000
    alpha = 1e-3
    line_search_factor = 0.5
    grad_threshold = 1e-6
    random_perturbation = 0.01
    
    def compute_c5(h):
        # Compute correlation in full mode
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def compute_gradient(h, k_star):
        # For lag k_star (0-based index), get the overlap indices
        # k_star goes from 0 to 2*n-2 where 0 corresponds to k=0
        # we convert to actual lag k = k_star - (n-1)
        k_actual = k_star - (n_points - 1)
        
        # Indices for h and 1-h
        if k_actual >= 0:
            # h_i and (1-h_{i+k_actual})
            start1 = 0
            stop1 = n_points - k_actual
            start2 = k_actual
            stop2 = n_points
        else:
            # h_{i-k_actual} and (1-h_i)
            k_actual = -k_actual
            start1 = k_actual
            stop1 = n_points
            start2 = 0
            stop2 = n_points - k_actual

        # gradient contribution
        grad = np.zeros_like(h)
        if k_actual == 0:
            # This is when h[i] * (1-h[i]) is considered
            grad = -2 * h + 1
        else:
            # gradient computation based on correlation at this specific lag
            for i in range(start1, stop1):
                j = i + k_actual
                grad[i] += (1 - h[j])
                grad[j] -= h[i]

        return grad
    
    # Main optimization loop
    c5_val = compute_c5(h)
    best_h = h.copy()
    best_c5 = c5_val
    iteration = 0
    last_improvement = 0
    stagnation_count = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 5:
        iteration += 1
        
        # Compute correlations for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        k_star = np.argmax(corr)
        max_corr = corr[k_star]
        
        # Compute gradient for this k*
        grad = compute_gradient(h, k_star)
        
        # Line search
        current_alpha = alpha
        improved = False
        new_h = h.copy()
        
        while True:
            new_h = h - current_alpha * grad
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            new_c5 = compute_c5(new_h)
            
            if new_c5 < c5_val:
                h = new_h
                c5_val = new_c5
                improved = True
                break
            else:
                current_alpha *= line_search_factor
                if current_alpha < 1e-10:
                    break
        
        # Adaptive alpha update
        if improved:
            stagnation_count = 0
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                last_improvement = iteration
        else:
            stagnation_count += 1
            if stagnation_count > 20:
                # Perturb slightly if stuck
                perturbation = np.random.normal(0, random_perturbation, n_points)
                h = h + perturbation
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                c5_val = compute_c5(h)
                stagnation_count = 0

        # Early termination if no improvement in a while
        if iteration - last_improvement > 500 and iteration > 1000:
            break
    
    # Final check
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
