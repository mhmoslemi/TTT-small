import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 1000
    learning_rate = 0.01
    perturb_freq = 50
    perturb_magnitude = 0.1
    
    # Initialize h from initial_h_values or default construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        # Default construction: step function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to target sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5 value and gradient
        corr = correlate(h, 1 - h, mode='full')
        max_corr_idx = np.argmax(corr)
        max_corr_value = corr[max_corr_idx]
        c5_value = max_corr_value * dx
        
        # Compute gradient with respect to the maximum correlation
        # We compute finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            # Ensure h_plus remains feasible
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            max_corr_plus = np.max(corr_plus)
            grad[i] = (max_corr_plus - max_corr_value) / eps
        
        # Apply gradient descent update
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Periodic perturbations to escape local minima
        if iteration % perturb_freq == 0:
            perturb_magnitude_t = perturb_magnitude / (1.0 + iteration / 100)
            noise = np.random.normal(0, perturb_magnitude_t, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = compute_c5(h, dx)
        
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            
        # Adapt learning rate based on recent improvement
        if iteration > 10:
            if abs(new_c5 - best_c5) < 1e-8:
                learning_rate *= 0.9
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
