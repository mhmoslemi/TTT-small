import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    mutation_factor = 0.8
    crossover_prob = 0.9
    max_iter = 1000
    adjustment_freq = 50
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init)
        for _ in range(population_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if abs(h_sum - n_points / 2) > 1e-10:
                h = h * (n_points / 2) / h_sum
            population.append(h)
    else:
        for _ in range(population_size):
            h = np.random.uniform(0, 1, n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h)
    
    # Helper function for repair
    def repair(h):
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2) < 1e-10:
            return h
        diff = h_sum - n_points / 2
        if diff > 0:
            # Remove excess from largest elements
            indices = np.argsort(h)[::-1][:int(diff/dx)]
            for idx in indices:
                if h[idx] > 0 and diff > 0:
                    dec = min(h[idx], diff)
                    h[idx] -= dec
                    diff -= dec
        else:
            # Add deficit to smallest elements
            indices = np.argsort(h)[:int(-diff/dx)]
            for idx in indices:
                if h[idx] < 1 and diff < 0:
                    inc = min(1 - h[idx], -diff)
                    h[idx] += inc
                    diff += inc
        return h * (n_points / 2) / np.sum(h)
    
    # Evaluate fitness function
    def evaluate_fitness(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Main DE loop
    best_individual = None
    best_fitness = float('inf')
    current_iter = 0
    
    while current_iter < max_iter and time.time() - start_time < budget_s - 10:
        # Update mutation factor
        if current_iter % adjustment_freq == 0 and current_iter > 0:
            mutation_factor *= 0.99
        
        new_population = []
        
        for i in range(population_size):
            # Select three distinct individuals
            candidates = list(range(population_size))
            candidates.remove(i)
            a, b, c = np.random.choice(candidates, 3, replace=False)
            
            # Create mutant vector
            mutant = population[a] + mutation_factor * (population[b] - population[c])
            mutant = np.clip(mutant, 0, 1)
            
            # Repair mutant
            mutant = repair(mutant)
            
            # Binomial crossover
            trial = np.zeros(n_points)
            j_rand = np.random.randint(0, n_points)
            for j in range(n_points):
                if np.random.rand() < crossover_prob or j == j_rand:
                    trial[j] = mutant[j]
                else:
                    trial[j] = population[i][j]
            
            # Repair trial
            trial = repair(trial)
            
            # Evaluate fitness and evolve if better
            current_fitness = evaluate_fitness(population[i])
            trial_fitness = evaluate_fitness(trial)
            
            if trial_fitness < current_fitness:
                new_population.append(trial)
                if trial_fitness < best_fitness:
                    best_fitness = trial_fitness
                    best_individual = trial.copy()
            else:
                new_population.append(population[i])
        
        population = new_population
        current_iter += 1
    
    # Final gradient descent refinement
    h_opt = best_individual.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 5:
            break
        corr = np.correlate(h_opt, 1 - h_opt, mode='full')
        fitness = np.max(corr) * dx
        
        grad = np.zeros(n_points)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h_opt.copy()
            h_plus[i] += eps
            h_plus = repair(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            fitness_plus = np.max(corr_plus) * dx
            grad[i] = (fitness_plus - fitness) / eps
        
        h_opt -= 0.001 * grad
        h_opt = repair(h_opt)
        
        corr_new = np.correlate(h_opt, 1 - h_opt, mode='full')
        fitness_new = np.max(corr_new) * dx
        if fitness_new >= fitness:
            break
        else:
            fitness = fitness_new
    
    # Final check
    corr_final = np.correlate(h_opt, 1 - h_opt, mode='full')
    c5_final = np.max(corr_final) * dx
    
    # Ensure final result meets constraints
    h_opt = np.clip(h_opt, 0, 1)
    sum_h = np.sum(h_opt)
    h_opt = h_opt * (n_points / 2) / sum_h
    
    return (h_opt.tolist(), c5_final, n_points)
