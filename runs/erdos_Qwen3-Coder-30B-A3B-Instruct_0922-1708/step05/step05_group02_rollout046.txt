import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    diff = target_sum - current_sum
    
    if abs(diff) > 1e-8:
        # Adjust smallest elements to correct the sum
        if diff > 0:
            # Need to increase sum
            indices = np.argsort(h)
            for i in range(int(np.ceil(diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = min(h[idx] + diff / (len(indices) - i), 1.0)
        else:
            # Need to decrease sum
            indices = np.argsort(h)[::-1]
            for i in range(int(np.ceil(-diff))):
                if i < len(indices):
                    idx = indices[i]
                    h[idx] = max(h[idx] + diff / (len(indices) - i), 0.0)
    
    # Final correction to ensure exact sum
    h = h * target_sum / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize solution
    if 'initial_h_values' in globals():

        if len(h_init) != n_points:

            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Parameters
    learning_rate = 0.001
    perturb_rate = 0.1
    perturb_magnitude = 0.05
    max_iter = 1000
    stagnation_threshold = 100
    
    last_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 10:
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(h, dx)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = repair_h(h_plus, n_points)
            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update using gradient
        h = h - learning_rate * grad
        h = repair_h(h, n_points)
        
        # Occasionally perturb solution to escape local minima
        if iter_count % 20 == 0:
            indices = np.random.choice(n_points, int(n_points * perturb_rate), replace=False)
            for i in indices:
                perturb = np.random.uniform(-perturb_magnitude, perturb_magnitude)
                h[i] = np.clip(h[i] + perturb, 0, 1)
            h = repair_h(h, n_points)
        
        # Evaluate new solution
        current_c5 = evaluate_h(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            last_improvement = iter_count
            
        iter_count += 1
        
        if iter_count - last_improvement > stagnation_threshold:
            # Random restart
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iter_count
    
    # Final gradient descent refinement
    for _ in range(500):
        if (time.time() - start_time) > budget_s - 5:

        eps = 1e-6
        grad = np.zeros(n_points)
        base_c5 = evaluate_h(best_h, dx)

            c5_plus = evaluate_h(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
            
        best_h = best_h - 0.0005 * grad
        best_h = repair_h(best_h, n_points)
        best_c5 = evaluate_h(best_h, dx)
