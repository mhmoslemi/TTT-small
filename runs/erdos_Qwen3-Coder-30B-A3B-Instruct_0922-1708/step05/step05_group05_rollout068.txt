import numpy as np
from scipy.stats import qmc
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    population_size = 100
    generations = 600
    crossover_rate = 0.8
    mutation_rate = 0.1
    elite_size = 5
    tournament_size = 5
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_target = np.linspace(0, 2, n_points)
                h = np.interp(x_target, x_orig, h)
        else:
            # Use Halton sequence for diverse initialization
            sampler = qmc.Halton(d=1, seed=seed)
            samples = sampler.random(n_points)
            h = samples.flatten()
        
        # Scale and normalize
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        return h

    def compute_fitness(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    def repair_individual(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - n_points / 2) > 1e-6:
            # Adjust to meet sum constraint
            scaling = (n_points / 2) / current_sum
            h = h * scaling
        return np.clip(h, 0, 1)

    def crossover(parent1, parent2):
        if np.random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # Single-point crossover
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        return child1, child2

    def mutate(h, rate):
        if np.random.random() > rate:
            return h.copy()
            
        # Gaussian mutation
        noise = np.random.normal(0, 0.05, n_points)
        h_mut = h + noise
        return repair_individual(h_mut)
    
    def tournament_selection(population, fitnesses):
        selected_indices = np.random.choice(len(population), tournament_size)
        best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
        return population[best_idx].copy()

    # Initialize population
    population = [create_individual() for _ in range(population_size)]
    fitnesses = [compute_fitness(h) for h in population]
    
    best_individual = population[np.argmin(fitnesses)].copy()
    best_fitness = min(fitnesses)
    
    start_time = time.time()
    
    # Evolution loop
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Track best
        if fitnesses[0] < best_fitness:
            best_individual = population[0].copy()
            best_fitness = fitnesses[0]
            
        # Keep elite
        new_population = population[:elite_size]
        new_fitnesses = fitnesses[:elite_size]
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            new_fitnesses.extend([compute_fitness(child1), compute_fitness(child2)])
        
        # Trim to exact size
        population = new_population[:population_size]
        fitnesses = new_fitnesses[:population_size]
        
        # Adapt mutation rate
        if gen % 10 == 0 and gen > 0:
            mutation_rate *= 0.99
    
    # Final refinement using gradient-based method
    h_refined = best_individual.copy()
    best_final_fitness = best_fitness
    
    for _ in range(300):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Estimate gradient via finite differences
        grad = np.zeros_like(h_refined)
        eps = 1e-5
        base_fitness = compute_fitness(h_refined)
        
        improved = False
        for j in range(n_points):
            h_plus = h_refined.copy()
            h_plus[j] += eps
            h_plus = repair_individual(h_plus)
            
            fitness_plus = compute_fitness(h_plus)
            grad[j] = (fitness_plus - base_fitness) / eps
            
            if fitness_plus < base_fitness:
                improved = True
                
        if not improved:
            break
            
        # Update
        h_refined -= 0.001 * grad
        h_refined = repair_individual(h_refined)
        
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        new_fitness = np.max(corr) * dx
        
        if new_fitness < base_fitness:
            base_fitness = new_fitness
        else:
            break
            
    # Final validation and recompute fitness
    h_final = repair_individual(h_refined)
    final_fitness = compute_fitness(h_final)
    
    return (h_final.tolist(), final_fitness, n_points)
