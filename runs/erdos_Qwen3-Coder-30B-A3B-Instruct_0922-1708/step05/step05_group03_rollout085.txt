import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # uniform guess
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    
    scale_sequence = [10, 20, 40, 80, 100]
    scales = []
    for s in scale_sequence:
        scales.append(s)
    
    # Measure total time at start
    start_time = time.time()
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    # Multi-scale optimization
    current_h = h.copy()
    for scale in scales:
        resolution = scale
        if resolution > n_points:
            break
            
        # Upsample if needed
        if resolution < len(current_h):
            original_points = len(current_h)
            x_original = np.linspace(0, 2, original_points)
            x_target = np.linspace(0, 2, resolution)
            current_h = np.interp(x_target, x_original, current_h)
        elif resolution > len(current_h):
            # interpolate to higher resolution
            original_points = len(current_h)
            x_original = np.linspace(0, 2, original_points)
            x_target = np.linspace(0, 2, resolution)
            current_h = np.interp(x_target, x_original, current_h)
            
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (resolution / 2) / np.sum(current_h)
        
        # Convex relaxation for improvement
        if resolution <= 40:
            # Use CVXPY only for smaller problems
            try:
                h_var = cp.Variable(resolution)
                # Soft constraints in convex solver
                constraints = [
                    h_var >= 0, 
                    h_var <= 1,
                    cp.sum(h_var) == resolution / 2
                ]
                
                # Simplified objective: average over relevant lags
                objective_terms = []
                step = max(1, resolution // 8)
                max_lag = resolution // 4
                
                for i in range(-max_lag, max_lag + 1):
                    if i == 0:
                        continue
                    if i > 0:
                        l = slice(0, resolution - i)
                        r = slice(i, resolution)
                    else:
                        l = slice(-i, resolution)
                        r = slice(0, resolution + i)
                    term = cp.sum(cp.multiply(h_var[l], 1 - h_var[r]))
                    objective_terms.append(term)
                
                # Average the fewer terms for faster solving
                mean_obj = cp.sum(objective_terms) / len(objective_terms) if objective_terms else 0
                objective = cp.Minimize(mean_obj)
                
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    new_h = h_var.value
                    new_h = np.array(new_h)
                    new_h = project_to_feasible(new_h)
                    current_h = new_h
            except:
                # fall back to project
                pass
        
        # Project and validate
        current_h = project_to_feasible(current_h)
        
        # Evaluate and update
        c5 = compute_c5(current_h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = current_h.copy()
            
        # Early exit if time is nearly up
        if time.time() - start_time > budget_s - 20:
            break
    
    # Refinement with gradient descent
    def finite_diff_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_to_feasible(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        return grad
    
    # Local refinement with gradient
    current_refine = best_h.copy()
    steps = 300
    eta = 0.001
    for i in range(steps):
        grad = finite_diff_gradient(current_refine)
        current_refine -= eta * grad
        current_refine = project_to_feasible(current_refine)
        
        # Recompute c5 for this refinement
        c5_refined = compute_c5(current_refine)
        if c5_refined < best_c5:
            best_c5 = c5_refined
            best_h = current_refine.copy()
            
        if time.time() - start_time > budget_s - 10:
            break
    
    # Final check of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
