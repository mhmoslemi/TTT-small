import numpy as np
from scipy.optimize import minimize
import time

def compute_fitness(h, dx):
    """Compute the maximum overlap integral for given h"""
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0<=h<=1 and sum(h)=n_points/2"""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx].copy()

def blend_crossover(parent1, parent2, alpha=0.8):
    """Blend crossover with alpha parameter"""
    child1 = parent1.copy()
    child2 = parent2.copy()
    for i in range(len(parent1)):
        c1, c2 = parent1[i], parent2[i]
        range_ = alpha * abs(c1 - c2)
        lower = min(c1, c2) - range_
        upper = max(c1, c2) + range_
        child1[i] = np.random.uniform(lower, upper)
        child2[i] = np.random.uniform(lower, upper)
    return child1, child2

def mutate(individual, mutation_rate=0.1, sigma=0.03):
    """Mutate individual with Gaussian noise"""
    if np.random.rand() < mutation_rate:
        noise = np.random.normal(0, sigma, len(individual))
        individual = individual + noise
    return individual

def gradient_refinement(h, dx, max_iter=50, lr=0.001):
    """Local gradient-based refinement"""
    h = h.copy()
    for _ in range(max_iter):
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        base_c5 = compute_fitness(h, dx)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus, len(h))
            c5_plus = compute_fitness(h_plus, dx)
            grad[i] = (c5_plus - base_c5) / eps
        # Update
        h = h - lr * grad
        h = project_to_feasible(h, len(h))
        # Recompute to avoid numerical issues
        base_c5 = compute_fitness(h, dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        # Start with the initial solution
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2.0) / np.sum(h_parent)
        # Add some injected noise to diversify
        for _ in range(population_size // 2):
            h = h_parent + np.random.normal(0, 0.03, n_points)
            h = project_to_feasible(h, n_points)
            population.append(h)
        # Add random individuals
        for _ in range(population_size // 2):
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            population.append(h)
    else:
        # Initialize entirely with random individuals
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            population.append(h)
    
    # Evaluate initial fitnesses
    fitnesses = [compute_fitness(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_c5 = fitnesses[best_idx]
    
    # Main evolution loop
    generations = 0
    max_generations = 10000
    refinement_interval = 50
    
    while time.time() - start_time < budget_s - 10 and generations < max_generations:
        generations += 1
        
        # Selection, crossover, mutation
        new_population = []
        while len(new_population) < population_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < 0.8:
                c1, c2 = blend_crossover(p1, p2)
            else:
                c1, c2 = p1.copy(), p2.copy()
            
            # Mutation
            c1 = mutate(c1)
            c2 = mutate(c2)
            
            # Projection to feasible region
            c1 = project_to_feasible(c1, n_points)
            c2 = project_to_feasible(c2, n_points)
            
            new_population.extend([c1, c2])
        
        # Trim to proper size
        population = new_population[:population_size]
        
        # Evaluate fitnesses
        fitnesses = [compute_fitness(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
        
        # Optional periodic refinement of top individuals
        if generations % refinement_interval == 0:
            # Refine top 10%
            top_indices = np.argsort(fitnesses)[:population_size // 10]
            for i in top_indices:
                population[i] = gradient_refinement(population[i], dx)
            # Re-evaluate after refinement
            fitnesses = [compute_fitness(h, dx) for h in population]
            current_best_idx = np.argmin(fitnesses)
            if fitnesses[current_best_idx] < best_c5:
                best_c5 = fitnesses[current_best_idx]
                best_h = population[current_best_idx].copy()
        
        # Occasionally add new random individuals to maintain diversity
        if generations % 100 == 0:
            for _ in range(5):
                h = np.random.rand(n_points)
                h = h * (n_points / 2.0) / np.sum(h)
                population.append(h)
                fitnesses.append(compute_fitness(h, dx))
    
    # Final refinement pass on best solution
    best_h = gradient_refinement(best_h, dx)
    best_c5 = compute_fitness(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
