import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Genetic Algorithm parameters
    population_size = 100
    generations = 500
    elite_size = 10
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 5
    
    # Initialize population
    if 'initial_h_values' in globals():
        population = [h.copy()]
        for _ in range(population_size - 1):
            # Perturb initial solution
            perturbed = h.copy() + np.random.normal(0, 0.03, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            population.append(perturbed)
    else:
        population = []
        for _ in range(population_size):
            particle = np.random.rand(n_points)
            particle = particle * (n_points / 2) / np.sum(particle)
            population.append(particle)
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitness_scores = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness_scores.append(c5)
        
        # Update best
        min_idx = np.argmin(fitness_scores)
        if fitness_scores[min_idx] < best_c5:
            best_c5 = fitness_scores[min_idx]
            best_h = population[min_idx].copy()
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness_scores)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        
        # Tournament selection and crossover
        while len(new_population) < population_size:
            if np.random.rand() < crossover_rate:
                # Tournament selection
                tourn_indices = np.random.choice(sorted_indices[:population_size//2], 
                                                tournament_size, replace=False)
                parent1_idx = tourn_indices[np.argmin([fitness_scores[i] for i in tourn_indices])]
                
                tourn_indices = np.random.choice(sorted_indices[:population_size//2], 
                                                tournament_size, replace=False)
                parent2_idx = tourn_indices[np.argmin([fitness_scores[i] for i in tourn_indices])]
                
                # Blend crossover
                alpha = np.random.rand()
                child1 = alpha * population[parent1_idx] + (1 - alpha) * population[parent2_idx]
                child2 = (1 - alpha) * population[parent1_idx] + alpha * population[parent2_idx]
                
                # Mutation
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child1[i] += np.random.normal(0, 0.02)
                    if np.random.rand() < mutation_rate:
                        child2[i] += np.random.normal(0, 0.02)
                
                # Repair
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < population_size:
                    new_population.append(child2)
            else:
                # Random individual
                new_population.append(np.random.rand(n_points))
                new_population[-1] = new_population[-1] * (n_points / 2) / np.sum(new_population[-1])
        
        population = new_population
        
        # Periodic local refinement (every 20 generations)
        if gen % 20 == 0:
            # Select top 10% for refinement
            top_indices = sorted_indices[:population_size//10]
            for i in top_indices:
                # Gradient-based refinement
                h_refine = population[i].copy()
                lr = 0.001
                for _ in range(20):
                    grad = np.zeros(n_points)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_refine.copy()
                        h_plus[j] += eps
                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - fitness_scores[i]) / eps
                    
                    h_refine = h_refine - lr * grad
                    h_refine = np.clip(h_refine, 0, 1)
                    h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
                    
                    corr_new = np.correlate(h_refine, 1 - h_refine, mode='full')
                    new_c5 = np.max(corr_new) * dx
                    
                    if new_c5 < fitness_scores[i]:
                        fitness_scores[i] = new_c5
                        population[i] = h_refine.copy()
                    else:
                        break
    
    # Final gradient refinement of best solution
    h_refine = best_h.copy()
    for _ in range(100):
        grad = np.zeros(n_points)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        h_refine = h_refine - 0.0005 * grad
        h_refine = np.clip(h_refine, 0, 1)
        h_refine = h_refine * (n_points / 2) / np.sum(h_refine)
        
        corr = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
