import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper to compute C5
    def compute_c5(h_array):
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        c5 = np.max(corr) * dx
        return c5
    
    # Gradient computation for a specific lag
    def compute_gradient(h_array, lag):
        n = len(h_array)
        grad = np.zeros_like(h_array)
        
        # Compute overlap window for this lag
        # For lag k, we're looking at h[i] * (1 - h[i+k])
        if lag >= 0:
            # Left-to-right correlation
            for i in range(n - lag):
                grad[i] += 1 - h_array[i + lag]
                grad[i + lag] -= h_array[i]
        else:
            # Right-to-left correlation (negative lag)
            lag_abs = -lag
            for i in range(n - lag_abs):
                grad[i] += 1 - h_array[i + lag_abs]
                grad[i + lag_abs] -= h_array[i]
        
        return grad * dx
    
    # Penalty function for constraints
    def compute_penalty(h_array):
        # Violations
        violation_lower = np.sum(np.maximum(0, -h_array)**2)
        violation_upper = np.sum(np.maximum(0, h_array - 1)**2)
        violation_sum = np.maximum(0, np.abs(np.sum(h_array) - n_points / 2))**2
        return 0.1 * (violation_lower + violation_upper + violation_sum)
    
    # Repair h to satisfy constraints
    def repair_h(h_array):
        # Clip to [0, 1]
        h_array = np.clip(h_array, 0, 1)
        # Rescale to exact sum
        h_sum = np.sum(h_array)
        if np.abs(h_sum - n_points / 2) > 1e-8:
            h_array = h_array * (n_points / 2) / h_sum
        return h_array
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Parameters
    learning_rate = 1e-3
    penalty_factor = 0.1
    plateau_patience = 200
    perturbation_fraction = 0.01
    learning_rate_decay = 0.9
    
    last_improvement = 0
    current_plateau = 0
    current_lr = learning_rate
    
    iteration_count = 0
    max_iter = 5000
    
    while (time.time() - start_time) < budget_s - 1 and iteration_count < max_iter:
        # Compute all correlations to find maximum
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_lag = max_lag_idx - (n_points - 1)
        
        # Compute gradient for max lag
        grad = compute_gradient(h, max_lag)
        
        # Add penalty gradient
        penalty_grad = np.zeros_like(h)
        violation_lower = np.maximum(0, -h)
        violation_upper = np.maximum(0, h - 1)
        violation_sum = np.sum(h) - n_points / 2
        if violation_sum > 0:
            penalty_grad += penalty_factor * violation_sum
        else:
            penalty_grad -= penalty_factor * violation_sum
            
        # Apply constraints to penalty
        penalty_grad += 2 * penalty_factor * (violation_lower - violation_upper)
        
        # Combine gradients
        grad_total = grad + penalty_grad
        
        # Line search for α (simple step size)
        alpha = current_lr
        test_h = h - alpha * grad_total
        test_h = repair_h(test_h)
        test_c5 = compute_c5(test_h)
        
        # If not improving, reduce learning rate and try again or perturb
        improved = test_c5 < best_c5
        
        if improved:
            h = test_h
            if test_c5 < best_c5:
                best_h = h.copy()
                best_c5 = test_c5
                last_improvement = iteration_count
        else:
            # Check if plateauing
            current_plateau += 1
            if current_plateau > plateau_patience:
                # Reduce learning rate
                current_lr *= learning_rate_decay
                # Perturb and restart plateau
                h = h + np.random.normal(0, perturbation_fraction, n_points)
                h = repair_h(h)
                current_plateau = 0
                    
        iteration_count += 1
    
    # Final computation of best C5
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
