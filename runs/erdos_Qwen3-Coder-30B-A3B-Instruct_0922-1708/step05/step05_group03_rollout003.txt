import numpy as np
from scipy.optimize import minimize
import time
import random

def compute_fitness(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_to_feasible(h, n_points):
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 1000
    
    # Initialize population
    population_size = 100
    population = []
    
    if 'initial_h_values' in globals():
        h_parent = np.array(initial_h_values)
        h_parent = h_parent * (n_points / 2.0) / np.sum(h_parent)
        # Create variations around parent
        for _ in range(population_size // 2):
            noise = np.random.normal(0, 0.05, n_points)
            h_var = h_parent + noise
            population.append(project_to_feasible(h_var, n_points))
        # Add some random solutions
        for _ in range(population_size - population_size // 2):
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2.0) / np.sum(h_rand)
            population.append(h_rand)
    else:
        # Purely random initialization
        for _ in range(population_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2.0) / np.sum(h)
            population.append(h)
    
    # Track best solution
    best_fitness = float('inf')
    best_individual = None
    
    start_time = time.time()
    
    # GA main loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitness = [compute_fitness(ind, dx) for ind in population]
        
        # Update global best
        min_idx = np.argmin(fitness)
        if fitness[min_idx] < best_fitness:
            best_fitness = fitness[min_idx]
            best_individual = population[min_idx].copy()
        
        # Selection (tournament)
        selected = []
        tournament_size = 5
        for _ in range(population_size):
            tournament = random.sample(list(enumerate(fitness)), tournament_size)
            winner_idx = min(tournament, key=lambda x: x[1])[0]
            selected.append(population[winner_idx])
        
        # Crossover and mutation
        new_population = []
        for i in range(0, population_size, 2):
            parent1 = selected[i]
            parent2 = selected[(i + 1) % population_size]
            
            # Blend crossover
            alpha = random.uniform(0.7, 0.95)
            child1 = alpha * parent1 + (1 - alpha) * parent2
            child2 = (1 - alpha) * parent1 + alpha * parent2
            
            # Mutation
            if random.random() < 0.1:  # 10% mutation rate
                mutation_magnitude = random.uniform(0.01, 0.05)
                noise1 = np.random.normal(0, mutation_magnitude, n_points)
                child1 += noise1
            if random.random() < 0.1:
                mutation_magnitude = random.uniform(0.01, 0.05)
                noise2 = np.random.normal(0, mutation_magnitude, n_points)
                child2 += noise2
            
            new_population.append(project_to_feasible(child1, n_points))
            new_population.append(project_to_feasible(child2, n_points))
        
        population = new_population
        
        # Periodic gradient refinement
        if generation % 50 == 0 and generation > 0:
            # Refine top individuals with gradient descent
            sorted_indices = np.argsort(fitness)
            top_k = min(10, len(sorted_indices))
            for i in range(top_k):
                idx = sorted_indices[i]
                h_ref = population[idx].copy()
                lr = 0.001
                
                for _ in range(50):
                    grad = np.zeros_like(h_ref)
                    eps = 1e-4
                    for j in range(n_points):
                        h_plus = h_ref.copy()
                        h_plus[j] += eps
                        h_plus = project_to_feasible(h_plus, n_points)
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        grad[j] = (c5_plus - fitness[idx]) / eps
                    
                    h_ref = h_ref - lr * grad
                    h_ref = project_to_feasible(h_ref, n_points)
                    
                    corr = np.correlate(h_ref, 1 - h_ref, mode='full')
                    new_fitness = np.max(corr) * dx
                    
                    if new_fitness < fitness[idx]:
                        fitness[idx] = new_fitness
                        population[idx] = h_ref.copy()
                        if new_fitness < best_fitness:
                            best_fitness = new_fitness
                            best_individual = h_ref.copy()
    
    # Final refinement of the best solution found
    if best_individual is not None:
        h_ref = best_individual.copy()
        lr = 0.0005
        for _ in range(100):
            grad = np.zeros_like(h_ref)
            eps = 1e-4
            for j in range(n_points):
                h_plus = h_ref.copy()
                h_plus[j] += eps
                h_plus = project_to_feasible(h_plus, n_points)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[j] = (c5_plus - best_fitness) / eps
            
            h_ref = h_ref - lr * grad
            h_ref = project_to_feasible(h_ref, n_points)
            
            corr = np.correlate(h_ref, 1 - h_ref, mode='full')
            new_fitness = np.max(corr) * dx
            
            if new_fitness < best_fitness:
                best_fitness = new_fitness
                best_individual = h_ref.copy()
    
    return (best_individual.tolist(), best_fitness, n_points)
