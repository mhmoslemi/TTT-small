import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        # Rescale to hit target sum
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    else:
        # Random initialization
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        best_h = h_init.copy()
    
    best_c5 = evaluate_h(best_h, dx)
    
    # Cuckoo Search parameters
    alpha = 0.25  # Discovery rate
    epsilon = 1e-5  # Small perturbation
    
    max_iter = 1000
    stagnation_limit = 200
    stagnation_count = 0
    
    # Main loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate new solution via Lévy flight (simplified)
        # Perturb a random subset of elements
        num_perturb = max(1, int(n_points * 0.1))  # Perturb ~10%
        indices = np.random.choice(n_points, num_perturb, replace=False)
        
        # Create new solution
        new_h = best_h.copy()
        for idx in indices:
            # Add random perturbation
            new_h[idx] += np.random.normal(0, epsilon)
        
        # Repair and normalize
        new_h = repair_h(new_h, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_h(new_h, dx)
        
        # Accept or reject based on discovery rate
        if new_c5 < best_c5:
            best_h = new_h.copy()
            best_c5 = new_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Occasionally apply local search refinement to best solution
        if iteration % 20 == 0:
            # Gradient descent refinement
            current_h = best_h.copy()
            current_c5 = best_c5
            for _ in range(50):  # Limited iterations
                if time.time() - start_time > budget_s - 10:
                    break
                
                grad = np.zeros(n_points)
                eps = 1e-6
                for j in range(n_points):
                    h_plus = current_h.copy()
                    h_plus[j] += eps
                    h_plus = repair_h(h_plus, n_points)
                    corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[j] = (c5_plus - current_c5) / eps
                
                current_h = current_h - 0.001 * grad
                current_h = repair_h(current_h, n_points)
                corr = correlate(current_h, 1 - current_h, mode='full')
                current_c5 = np.max(corr) * dx
                
                if current_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
        # Check for stagnation and adjust discovery rate
        if stagnation_count > stagnation_limit:
            alpha *= 0.95
            stagnation_count = 0
            if alpha < 0.01:
                alpha = 0.01
    
    # Final refinement
    current_h = best_h.copy()
    current_c5 = best_c5
    for _ in range(200):
        if time.time() - start_time > budget_s - 5:
            break
        grad = np.zeros(n_points)
        eps = 1e-6
        for j in range(n_points):
            h_plus = current_h.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - current_c5) / eps
        
        current_h = current_h - 0.0005 * grad
        current_h = repair_h(current_h, n_points)
        corr = correlate(current_h, 1 - current_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = current_c5
    
    return (best_h.tolist(), best_c5, n_points)
