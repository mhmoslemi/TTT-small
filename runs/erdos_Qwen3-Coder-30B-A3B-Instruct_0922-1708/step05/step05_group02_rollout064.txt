import numpy as np
import time
from scipy.signal import correlate

def repair_h(h, n_points):
    """Repair h to satisfy the constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-8:
        h = h * target_sum / current_sum
    return h

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = repair_h(h_init, n_points)
    else:
        # Initialize with a random feasible solution
        h_init = np.random.rand(n_points)
        h_init = repair_h(h_init, n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Track the last improvement
    last_improvement = 0
    consecutive_no_improvement = 0
    max_no_improvement = 200
    
    # Main search loop
    iteration = 0
    while time.time() - start_time < budget_s - 10 and consecutive_no_improvement < max_no_improvement:
        iteration += 1
        # Generate ~100 candidate solutions by perturbing best_h
        candidates = []
        for _ in range(100):
            # Create a mutation by flipping some values
            h_mut = best_h.copy()
            # Flip 5-15% of positions
            num_flip = np.random.randint(int(0.05 * n_points), int(0.15 * n_points) + 1)
            flip_indices = np.random.choice(n_points, size=num_flip, replace=False)
            for i in flip_indices:
                h_mut[i] = 1 - h_mut[i]
            
            # Add some randomness
            noise_scale = 0.05
            noise = np.random.normal(0, noise_scale, n_points)
            h_mut += noise
            
            # Repair to ensure feasibility
            h_mut = repair_h(h_mut, n_points)
            candidates.append(h_mut)
        
        # Evaluate all candidates
        for candidate in candidates:
            c5_candidate = evaluate_h(candidate, dx)
            if c5_candidate < best_c5:
                best_h = candidate.copy()
                best_c5 = c5_candidate
                consecutive_no_improvement = 0
                last_improvement = iteration
            else:
                consecutive_no_improvement += 1
        
        # Occasionally rescale to escape local optima
        if iteration % 50 == 0:
            # Randomly reduce all elements by a small amount
            scale_factor = 0.95 + np.random.rand() * 0.1
            best_h *= scale_factor
            best_h = repair_h(best_h, n_points)
            best_c5 = evaluate_h(best_h, dx)
    
    # Final fine-tuning with gradient descent
    h_fine = best_h.copy()
    best_c5_fine = best_c5
    for _ in range(500):
        if time.time() - start_time > budget_s - 5:
            break
        eps = 1e-6
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h_fine.copy()
            h_plus[j] += eps
            h_plus = repair_h(h_plus, n_points)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[j] = (c5_plus - best_c5_fine) / eps
        
        h_fine = h_fine - 0.0005 * grad
        h_fine = repair_h(h_fine, n_points)
        corr = correlate(h_fine, 1 - h_fine, mode='full')
        best_c5_fine = np.max(corr) * dx
    
    return (h_fine.tolist(), best_c5_fine, n_points)
