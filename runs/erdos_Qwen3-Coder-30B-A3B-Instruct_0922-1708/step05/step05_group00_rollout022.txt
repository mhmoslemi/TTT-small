        x = np.linspace(0, 2, n_points)
        h = 0.5 + 0.3 * np.sin(2 * np.pi * x)  # simple smooth variation
        
    # Normalize to meet constraint sum h * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Define a smooth approximation to the maximum function for optimization
    def smooth_max(x, beta=10.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * x)))
    
    # Gradient-based refinement using a differentiable surrogate objective
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimizer constants
    max_iter = 6000

    patience = 0

        # Compute overlap values for current h
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 1000:
                break
        
        # Compute gradient using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)

            h_plus = h.copy()

        # Update with learning rate
        lr = lr_schedule(t)
        h_new = h - lr * grad

        h_new = h_new * (n_points / 2) / np.sum(h_new)

        t += 1
    
    # Final validation

    # Ensure final constraint satisfaction
