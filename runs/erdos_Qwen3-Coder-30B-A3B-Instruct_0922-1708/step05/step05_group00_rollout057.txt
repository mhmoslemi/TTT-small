import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp
from cvxpy import ECOS, SCS

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy sum constraint exactly
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Multi-scale optimization
    # First, coarse optimization to avoid local minima
    n_coarse = 50
    dx_coarse = 2.0 / n_coarse
    h_coarse = np.array([np.mean(h[i*n_points//n_coarse:(i+1)*n_points//n_coarse]) 
                        for i in range(n_coarse)])
    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
    h_coarse = np.clip(h_coarse, 0, 1)
    
    # Solve coarse problem using quadratic programming relaxation
    try:
        h_var_coarse = cp.Variable(n_coarse)
        C5_var = cp.Variable()
        
        # Constraints
        constraints = [
            h_var_coarse >= 0,
            h_var_coarse <= 1,
            cp.sum(h_var_coarse) == n_coarse / 2,
            C5_var >= 0
        ]
        
        # We'll construct constraint matrix for all shift combinations in coarse grid
        # Compute all overlap terms exactly in coarse domain - aproximate
        # For large n_approximate full correlation computation, this is expensive
        # Approximation: Use first-order approx as a convex relaxation
        
        # No direct convex formulation, so we use a heuristic
        # But let's try a greedy greedy refinement
        prob = cp.Problem(cp.Minimize(C5_var), constraints)
        try:
            prob.solve(solver=ECOS, max_iters=1000, abstol=1e-6, reltol=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMIZER_FOUND]:
                h_sol = h_var_coarse.value
                if h_sol is not None:
                    h_coarse = np.array(h_sol)
                    h_coarse = h_coarse * (n_coarse / 2) / np.sum(h_coarse)
                    h_coarse = np.clip(h_coarse, 0, 1)
        except Exception:
            pass
    except Exception:
        pass
    
    # Transfer coarse solution to fine grid by interpolation
    h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, n_coarse), h_coarse)
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Refine at fine grid using gradient-based approach with adaptive learning rate
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Use an adaptive learning rate that decreases with each iteration
    lr = 1e-3
    max_iter = int(1e6)
    patience = 0
    worst_improvement = 1e-5
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full') * dx
        current_c5 = np.max(correlation)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 300:
                break  # Early stop if no improvement for 300 iterations
            
        if abs(current_c5 - last_improvement) < worst_improvement:
            # If improvement is getting too small, reduce learning rate
            lr *= 0.99
            if lr < 1e-6:
                break  # Stop if learning rate is too small
                
        last_improvement = current_c5

        # Compute finite difference gradient for update
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Maintain constraint
            
            cor_plus = np.correlate(h_plus, 1 - h_plus, mode='full') * dx
            c5_plus = np.max(cor_plus)
            
            grad[i] = (c5_plus - current_c5) / eps

        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Further refine the learning rate if needed
        lr = max(1e-6, lr * 0.9995)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full') * dx
    final_c5 = np.max(correlation)
    
    # Final sum check
    if abs(np.sum(best_h) * dx - 1.0) > 1e-6:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
