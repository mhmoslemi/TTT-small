import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    # Normalize h to have integral 1 over [0,2]
    dx = 2.0 / len(h)
    h = np.array(h)
    h = h * (1.0 / np.sum(h)) * 2.0  # Make integral = 1
    
    # Compute the correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the maximum correlation value
    c5 = np.max(corr) * dx
    
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_particles = 20
    max_iter = int(budget_s / 0.05)  # Estimate iterations
    w = 0.7
    c1 = c2 = 2.0
    v_max = 0.1
    perturb_scale = 0.01
    
    # Initialize particles
    if 'initial_h_values' in globals():

    else:
        # Create a simple initial guess (constant + noise)
        h_best = np.ones(n_points) * 0.5
        h_best = h_best + np.random.normal(0, 0.1, n_points)
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best * (n_points/2.0) / np.sum(h_best)
    
    h_best = np.clip(h_best, 0, 1)
    h_best = h_best * (n_points/2.0) / np.sum(h_best)
    c_best = evaluate_erdos_solution(h_best)
    h_particles = [h_best.copy()]
    v_particles = [np.random.uniform(-0.1, 0.1, n_points)]
    
    # Add more particles
    for _ in range(n_particles - 1):
        h_p = h_best + np.random.normal(0, 0.1, n_points)
        h_p = np.clip(h_p, 0, 1)
        h_p = h_p * (n_points/2.0) / np.sum(h_p)
        h_particles.append(h_p)
        v_particles.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Global best
    h_global = h_best.copy()
    c_global = c_best
    
    # Remove duplicates if any
    h_particles = [h_particles[0]] + [h for h in h_particles[1:] if not np.allclose(h, h_particles[0])]
    
    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    
    for _ in range(max_iter):
        if time.time() - start_time > budget_s:
            break
        
        iter_count += 1
        
        # Update each particle
        for i in range(len(h_particles)):
            h_p = h_particles[i]
            v_p = v_particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_p_new = w * v_p + c1 * r1 * (h_particles[i] - h_p) + c2 * r2 * (h_global - h_p)
            v_p_new = np.clip(v_p_new, -v_max, v_max)
            
            # Update position
            h_p_new = h_p + v_p_new
            h_p_new = np.clip(h_p_new, 0, 1)
            
            # Normalize sum to n_points / 2
            sum_h_new = np.sum(h_p_new)
            if sum_h_new != 0:
                h_p_new = h_p_new * (n_points/2.0) / sum_h_new
            
            # Evaluate
            c_p_new = evaluate_erdos_solution(h_p_new)
            
            # Update particle's personal best
            if c_p_new < c_best:
                c_best = c_p_new
                h_best = h_p_new.copy()
            
            # Update global best
            if c_p_new < c_global:
                c_global = c_p_new
                h_global = h_p_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            h_particles[i] = h_p_new
            v_particles[i] = v_p_new
            
        # Tournament selection: replace bottom half with new random solutions
        if iter_count % 20 == 0:
            idxs = np.argsort([evaluate_erdos_solution(h) for h in h_particles])
            top_half = idxs[:len(idxs)//2]
            bottom_half = idxs[len(idxs)//2:]
            for j in bottom_half:
                h_new = h_global + np.random.normal(0, perturb_scale, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points/2.0) / np.sum(h_new)
                h_particles[j] = h_new
        
        # Introduce diversity if stuck
        if no_improvement_count > 100:
            for i in range(len(h_particles)):
                h_particles[i] = h_global + np.random.normal(0, perturb_scale, n_points)
                h_particles[i] = np.clip(h_particles[i], 0, 1)
                h_particles[i] = h_particles[i] * (n_points/2.0) / np.sum(h_particles[i])
            no_improvement_count = 0

    # Return final best solution
    return (h_best.tolist(), c_global, n_points)
