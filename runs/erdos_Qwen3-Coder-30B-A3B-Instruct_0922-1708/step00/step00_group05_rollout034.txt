import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Create random initial point with sum = n/2
        n_points = 70
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Refine using QCQP approach
    max_iter = 100
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try different random initializations
        for _ in range(5):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Initialize h with small perturbation of best solution or fresh
            h_init = best_h.copy() + np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init / np.sum(h_init) * (n_points / 2) 
            
            # Define variables
            h = cp.Variable(n_points)
            
            # Objective: minimize C
            C = cp.Variable()
            
            # Constraints
            constraints = [h >= 0, h <= 1]
            constraints.append(cp.sum(h) == n_points / 2)
            
            # Overlap constraints for all lags (excluding zero lag, it's 0 anyway)
            for k in range(1, n_points):
                if k <= n_points - 1:
                    # Correlation with shifted version of (1-h)
                    sum_overlap = 0
                    for i in range(n_points):
                        j = i + k
                        if 0 <= j < n_points:
                            sum_overlap += h[i] * (1 - h[j])
                    constraints.append(sum_overlap <= C * dx)
            
            # Objective function: minimize C
            prob = cp.Problem(cp.Minimize(C), constraints)
            
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status in ("optimal", "optimal_inaccurate"):
                    h_val = h.value
                    if h_val is not None:
                        h_val = np.clip(h_val, 0, 1)
                        # Renormalize
                        h_val = h_val / np.sum(h_val) * (n_points / 2)
                        c5_val = evaluate_erdos_solution(h_val)
                        if c5_val < best_c5:
                            best_c5 = c5_val
                            best_h = h_val.copy()
            except:
                pass  # Ignore solver failures and continue

    return (best_h, best_c5, n_points)
