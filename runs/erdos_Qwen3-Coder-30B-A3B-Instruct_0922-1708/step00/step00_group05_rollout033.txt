import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    best_h = h_init.copy()
    best_C = float('inf')
    
    start_time = time.time()
    max_iterations = 50
    iteration_count = 0
    
    def evaluate_C(h):
        # Compute overlaps for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        # Take the maximum value across all lags
        max_overlap = np.max(corr)
        return max_overlap * dx
    
    def solve_with_qp(h_start):
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Add overlap constraints for all lags
        for k in range(-n_points + 1, n_points):
            # Compute indices for correlation
            if k >= 0:
                h1 = h[k:]
                h2 = h[:n_points - k]
            else:
                h1 = h[:n_points + k]
                h2 = h[-k:]
            
            if len(h1) > 0:
                overlap = cp.sum(cp.multiply(h1, 1 - h2))
                constraints.append(overlap <= C * dx)
        
        # Objective
        objective = cp.Minimize(C)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                return h.value, prob.value
            else:
                return None, None
        except:
            return None, None
    
    # Try multiple random initializations
    for trial in range(5):
        if time.time() - start_time > budget_s - 1:
            break
            
        h = h_start = h_init.copy()
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Refine with QP optimization
        for attempt in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            
            h_refined, C_val = solve_with_qp(h)
            if h_refined is not None:
                C = evaluate_C(h_refined)
                if C < best_C:
                    best_C = C
                    best_h = h_refined.copy()
                    if best_C < 0.3807:  # Early termination
                        return (best_h.tolist(), best_C, n_points)
            else:
                # Perturb randomly if QP fails
                pert = np.random.normal(0, 0.005, n_points)
                h = h + pert
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
        
    # Final refinement
    try:
        h_final, C_final = solve_with_qp(best_h)
        if h_final is not None:
            final_C = evaluate_C(h_final)
            if final_C < best_C:
                best_C = final_C
                best_h = h_final.copy()
    except:
        pass
    
    return (best_h.tolist(), best_C, n_points)
