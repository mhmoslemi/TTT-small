def run(seed=42, budget_s=600, **kwargs):

        # Compute the overlap integral
        # We use numpy's correlate to compute all overlaps
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # The full correlation is of length 2*n_points - 1
        # The maximum is at the middle
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_to_feasible(h_vals):
        # Project to [0,1] and ensure integral constraint
        h_proj = np.clip(h_vals, 0, 1)
        # Scale to make integral equal to 1
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj

        # This function optimizes h to minimize overlap for shift k
        # Only optimize the elements that are involved in shift k
        n = len(h_vals)
        # Indices involved in shift k
        if k >= 0:
            indices = np.arange(n - k)
        else:
            indices = np.arange(-k, n)
        
        indices = indices.astype(int)

            # Recompute overlap with updated h_part
            h_new = h_vals.copy()
            h_new[indices] = h_part
            return np.sum(h_new * (1 - np.roll(h_new, k))) * dx

    return (best_h.tolist(), final_c5, n_points)
