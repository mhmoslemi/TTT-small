import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70  # Use the provided initial construction size
    dx = 2.0 / n_points
    
    # Get initial guess
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h_init = np.ones(n_points) * 0.5
    
    # Normalize initial guess to satisfy integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Function to evaluate C5 for a given h
    def evaluate_c5(h):
        h = np.array(h)
        # Compute all overlaps using correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum overlap (excluding the zero lag which is always 0)
        max_overlap = np.max(corr)
        return max_overlap * dx
        
    # Define the optimization problem as a QCQP
    def optimize_h():
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Bound constraints
        constraints = [h >= 0, h <= 1]
        
        # Integral constraint
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Overlap constraints (for all lags)
        for k in range(-n_points + 1, n_points):
            # Compute overlap for lag k
            if k >= 0:
                # h[i] * (1 - h[i + k]) for valid indices
                overlap_expr = cp.sum(cp.multiply(h[k:], 1 - h[:n_points-k]))
            else:
                # k < 0: h[i + k] * (1 - h[i]) for valid indices
                overlap_expr = cp.sum(cp.multiply(h[-k:], 1 - h[:n_points+k]))
            
            # Add constraint for this lag
            constraints.append(overlap_expr <= C * dx)
        
        # Objective function
        objective = cp.Minimize(C)
        
        # Create and solve the problem
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers
        for solver in [cp.GUROBI, cp.CVXOPT, cp.SCS]:
            try:
                prob.solve(solver=solver, verbose=False)
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    return h.value
            except:
                continue
        
        # Fallback to Scipy if CVXPY fails
        def obj(h_vec):
            return evaluate_c5(h_vec)
        
        # Using scipy.optimize to get a feasible solution
        bounds = [(0, 1) for _ in range(n_points)]
        # Constraint on the integral
        def integral_constraint(h_vec):
            return np.sum(h_vec) - n_points / 2
        
        # Optimize using scipy
        result = minimize(obj, h_init, bounds=bounds, method='SLSQP',
                          constraints={'type': 'eq', 'fun': integral_constraint},
                          options={'maxiter': 1000, 'ftol': 1e-6})
        
        if result.success:
            return result.x
        else:
            # Return initial if no optimization succeeded
            return h_init
    
    best_h = None
    best_c5 = float('inf')
    
    # Multiple random restarts
    for i in range(10):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Get better starting point
        h = optimize_h()
        
        # Validate that h is in valid range
        h = np.clip(h, 0, 1)
        
        # Remap so that integral constraint is satisfied
        h = h * (n_points / 2) / np.sum(h)
        
        c5 = evaluate_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final check    
    if best_h is None:
        best_h = h_init
        best_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
