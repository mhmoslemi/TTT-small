import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the maximum overlap integral."""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project(h, n_points):
    """Project h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 100
    dx = 2.0 / n_points
    max_iter = 500
    swarm_size = 50
    w = 0.7
    c1 = c2 = 2.0
    stagnation_threshold = 50
    diversity_threshold = 25
    reset_fraction = 0.2
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project(h0, n_points)
        particles = [h0.copy() for _ in range(swarm_size)]
        for i in range(1, swarm_size):
            noise = np.random.normal(0, 0.05, n_points)
            particles[i] = project(h0 + noise, n_points)
    else:
        particles = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = project(h, n_points)
            particles.append(h)
    
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    personal_best = [particles[i].copy() for i in range(swarm_size)]
    personal_best_values = [evaluate(particles[i], dx) for i in range(swarm_size)]
    global_best = personal_best[np.argmin(personal_best_values)].copy()
    global_best_value = min(personal_best_values)
    
    iter_no_improve = 0
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        for i in range(swarm_size):
            # Evaluate current particle
            current_value = evaluate(particles[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = particles[i].copy()
                global_best_value = current_value
                iter_no_improve = 0
            else:
                iter_no_improve += 1
                
            # Update velocity and position
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (personal_best[i] - particles[i]) + 
                             c2 * r2 * (global_best - particles[i]))
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i], n_points)
            
        # Inject diversity if stagnating
        if iter_no_improve > diversity_threshold:
            num_to_reset = int(swarm_size * reset_fraction)
            indices_to_reset = np.random.choice(swarm_size, num_to_reset, replace=False)
            for idx in indices_to_reset:
                h = np.random.rand(n_points)
                h = project(h, n_points)
                particles[idx] = h
                velocities[idx] = np.zeros(n_points)
                personal_best[idx] = h.copy()
                personal_best_values[idx] = evaluate(h, dx)
            iter_no_improve = 0
    
    return (global_best, global_best_value, n_points)
