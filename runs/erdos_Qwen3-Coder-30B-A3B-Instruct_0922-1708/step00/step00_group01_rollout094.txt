import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for a given step function h."""
    correlation = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # reasonable compromise
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    inertia_w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset_frac = 0.2
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = project_h(h_start, n_points)
    else:
        h_start = np.ones(n_points) * 0.5
    
    particles_h = np.tile(h_start, (swarm_size, 1))
    
    # Add noise to particles
    for i in range(swarm_size):
        noise = np.random.normal(0, 0.1, n_points)
        particles_h[i] += noise
        particles_h[i] = project_h(particles_h[i], n_points)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (swarm_size, n_points))
    
    # Initialize personal best
    personal_best_h = particles_h.copy()
    personal_best_c5 = np.array([evaluate_h(ph, dx) for ph in personal_best_h])
    
    # Initialize global best
    best_idx = np.argmin(personal_best_c5)
    global_best_h = personal_best_h[best_idx].copy()
    global_best_c5 = personal_best_c5[best_idx]
    
    start_time = time.time()
    iter_count = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate current swarm
        current_c5 = np.array([evaluate_h(ph, dx) for ph in particles_h])
        
        # Update personal bests
        improved = current_c5 < personal_best_c5
        personal_best_h[improved] = particles_h[improved]
        personal_best_c5[improved] = current_c5[improved]
        
        # Update global best
        new_best_idx = np.argmin(personal_best_c5)
        if personal_best_c5[new_best_idx] < global_best_c5:
            global_best_c5 = personal_best_c5[new_best_idx]
            global_best_h = personal_best_h[new_best_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Diversify if stagnating
        if stagnation_count >= stagnation_limit:
            num_to_reset = int(swarm_size * diversity_reset_frac)
            reset_indices = np.random.choice(swarm_size, num_to_reset, replace=False)
            for idx in reset_indices:
                particles_h[idx] = project_h(np.random.rand(n_points), n_points)
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
        
        # Update positions and velocities
        r1, r2 = np.random.rand(swarm_size, 1), np.random.rand(swarm_size, 1)
        velocities = (
            inertia_w * velocities +
            c1 * r1 * (personal_best_h - particles_h) +
            c2 * r2 * (global_best_h - particles_h)
        )
        particles_h += velocities
        
        # Project back to feasible region
        for i in range(swarm_size):
            particles_h[i] = project_h(particles_h[i], n_points)
            
    # Final re-evaluation of best solution
    final_c5 = evaluate_h(global_best_h, dx)
    
    return (global_best_h.tolist(), final_c5, n_points)
