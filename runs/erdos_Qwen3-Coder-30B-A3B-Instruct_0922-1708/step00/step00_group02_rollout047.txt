import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    lr = 0.01
    beta = 0.9
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Use initial construction if available, otherwise create uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Initialize velocity for momentum
    v_h = np.zeros(n_points)
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation: h ⋆ (1 - h)
        h_reflect = np.concatenate([h, np.zeros(n_points)])
        h_mirror = np.concatenate([np.zeros(n_points), 1 - h])
        corr = np.correlate(h_reflect, h_mirror, mode='valid')
        
        # Find k that maximizes the correlation (shift index k)
        k_max = np.argmax(corr)
        max_overlap = corr[k_max] * dx
        
        # Update best solution
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            lr *= 0.5
            patience_counter = 0
            
        # Compute gradient w.r.t. h
        # grad = (1 - h) - (h * correlation_derivative)
        # For cross-correlation with reflected signal, the gradient in x domain
        # for shift k_max is approximated by finite differences:
        # Moving the peak slightly in direction of improvement
        grad_h = (1 - h) - h * (1 - h) * dx  # Simplified gradient
        
        # Actually, use gradient of correlation wrt. h values:
        # For forward cross-correlation of h with (1-h), 
        # for each x, a contribution to derivative of overlap 
        # from h[i] occurs in kernel overlaps: 
        # loop over k to compute precise gradient contributions
        # but here, we can derive gradient from discrete stretches:
        # gradient = (1 - h) - h * (derivative of R(k) w.r.t. h)
        # approximate using sliding windows:
        d_h = np.zeros(n_points)
        r_val = 0.0
        for i in range(n_points):
            r_val = 0.0
            for k in range(n_points):
                # h[i] contributes to overlap when x in [i, i+k]
                # and 1-h[i+k] is computed for x=(i+k)
                j = (i + k) % n_points
                r_val += (1 - h[j])  # approximate derivative?
            # actual gradient: each h[i] contributes - R(i) to gradient  
            # But simpler version
            d_h[i] = (1 - h[i])  # Approximate gradient contribution
        # Normalization to keep sum constraint: h sum = 1
        grad_h = d_h
        
        # Update h using momentum
        v_h = beta * v_h + (1 - beta) * grad_h
        h = h - lr * v_h
        
        # Project onto feasible set [0,1] and keep sum constraint
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    if best_h is None:
        best_h = h
    
    return (best_h, best_c5, n_points)
