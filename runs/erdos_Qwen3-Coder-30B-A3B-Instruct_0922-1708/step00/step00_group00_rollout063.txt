import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h_values):
    # Compute overlap integral (C5)
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute cross-correlation between h and (1-h)
    corr = correlate(h_values, 1 - h_values, mode="full")
    # Scale by dx to get true overlap values
    overlaps = corr * dx
    # Take maximum overlap over all shifts
    max_overlap = np.max(overlaps)
    return max_overlap

def initialize_population(n_points, population_size, initial_h=None):
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill remaining with random step functions
    while len(population) < population_size:
        h = np.random.random(n_points)
        # Scale to meet constraint: sum(h) = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def fitness(h):
    return evaluate_h(h)

def tournament_selection(population, fitnesses, tournament_size=5):
    selected_indices = np.random.choice(len(population), size=tournament_size)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    if np.random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Normalize to meet constraint and clip
    child1 = np.clip(child1 * (n_points / 2) / np.sum(child1), 0, 1)
    child2 = np.clip(child2 * (n_points / 2) / np.sum(child2), 0, 1)
    return child1, child2

def mutate(h, mutation_rate=0.05):
    n_points = len(h)
    mask = np.random.random(n_points) < mutation_rate
    if np.any(mask):
        h = h.copy()
        h[mask] += np.random.normal(0, 0.05, np.sum(mask))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use default sampling resolution
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize population
    population_size = 100
    initial_h = getattr(__builtins__, 'initial_h_values', None)
    population = initialize_population(n_points, population_size, initial_h)

    best_h = None
    best_c5 = float('inf')
    
    # Start evolution
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 0.1:
        generation += 1
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
        
        # Create next generation
        new_population = []
        
        # Elitism: Keep best 20%
        elites_idx = np.argsort(fitnesses)[:int(population_size * 0.2)]
        for idx in elites_idx:
            new_population.append(population[idx].copy())
        
        # Generate rest using selection, crossover, mutation
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        population = new_population[:population_size]
    
    assert best_h is not None
    # Final safety check
    assert np.all(best_h >= 0) and np.all(best_h <= 1)
    assert abs(np.sum(best_h) - n_points/2) < 1e-6
    
    return best_h, best_c5, n_points
