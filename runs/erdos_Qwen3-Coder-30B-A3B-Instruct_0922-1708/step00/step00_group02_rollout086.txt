import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm_phase(initial_h, n_points, budget_s, dx):
    """Genetic algorithm to explore the space"""
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.2
    generations = 0
    start_time = time.time()
    
    # Initialize population
    population_h = [initial_h.copy()]
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population_h.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    while time.time() - start_time < 0.7 * budget_s:
        generations += 1
        # Evaluate all individuals
        fitness = [evaluate_h(h, dx) for h in population_h]
        
        # Track best
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_h = population_h[min_fitness_idx].copy()
            best_c5 = fitness[min_fitness_idx]
        
        # Selection - elitism
        elite_count = int(elitism_rate * pop_size)
        sorted_indices = np.argsort(fitness)
        elite = [population_h[i] for i in sorted_indices[:elite_count]]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            # Tournament selection
            a_idx = sorted_indices[np.random.randint(0, pop_size // 2)]
            b_idx = sorted_indices[np.random.randint(0, pop_size // 2)]
            a, b = population_h[a_idx], population_h[b_idx]
            
            if np.random.rand() < crossover_prob:
                # Single point crossover
                cross_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([a[:cross_point], b[cross_point:]])
            else:
                child = a.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mut_idx = np.random.randint(0, n_points)
                child[mut_idx] = np.random.rand()
            
            child = project_h(child, n_points)
            offspring.append(child)
        
        # Create new population
        population_h = elite + offspring
        
    return best_h, best_c5

def gradient_descent_phase(initial_h, n_points, budget_s, dx):
    """Gradient descent on the worst overlap"""
    start_time = time.time()
    h = initial_h.copy()
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    learning_rate = 0.01
    patience = 0
    max_patience = 5000
    
    while time.time() - start_time < 0.3 * budget_s:
        # Compute overlap for all shifts and find worst one
        corr = correlate(h, 1 - h, mode="full")
        max_shift_idx = np.argmax(corr)
        worst_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient w.r.t. h
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c_plus = evaluate_h(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c_minus = evaluate_h(h_minus, dx)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
        
        # Update
        h -= learning_rate * grad
        h = project_h(h, n_points)
        
        # Evaluate
        new_c5 = evaluate_h(h, dx)
        if new_c5 < best_c5:
            best_h = h.copy()
            best_c5 = new_c5
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
        if patience > 0 and patience % 1000 == 0:
            learning_rate *= 0.5
            patience = 0
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function"""
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        # Initialize with sinusoidal
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = project_h(h, n_points)
    
    # Genetic algorithm phase
    best_h, best_c5 = genetic_algorithm_phase(h, n_points, budget_s, dx)
    
    # Gradient descent phase
    final_h, final_c5 = gradient_descent_phase(best_h, n_points, budget_s, dx)
    
    return (final_h, final_c5, n_points)
