import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h"""
    dx = 2.0 / len(h)
    # Compute all overlaps using correlation
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral is the correlation at different lags
    # We want maximum overlap over all lags and the valid range of lags
    # The valid lags are from the minimum overlap to maximum overlap
    # Correct way: For each valid lag k, find the overlap
    c5 = max(corr[len(h)-1:]) * dx
    return c5

def project_to_feasible(h, n_points):
    """Projects h to [0,1] and ensures sum = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / n_points)
    return h

def solve_erdos_frank_wolfe(initial_h, n_points, max_time=600):
    """Solve the Erdős minimum overlap problem using Frank-Wolfe"""
    h = initial_h.copy()
    h = project_to_feasible(h, n_points)
    
    start_time = time.time()
    best_c5 = evaluate_h(h)
    best_h = h.copy()
    
    # Frank-Wolfe parameters
    eta = 0.1
    max_iter = 1000
    tol = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time - 1:
            break
            
        # Compute all overlaps (correlations)
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode='full')
        # Extract valid overlaps
        overlap_values = corr[len(h)-1:]
        k_max = np.argmax(overlap_values)
        max_overlap = overlap_values[k_max] * dx
        
        # Gradient for k_max (simplified)
        # For a fixed k_max, compute gradient w.r.t. h[i]
        # Gradient at i is: 1 - h[i+k_max] if valid else 0
        grad = np.zeros(n_points)
        # Iteration over i: compute gradient for each i
        # An efficient way to compute the gradient vector:
        # we're looking at the convolution h[i] * (1 - h[i+k])
        # So for lag k_max, the gradient of the overlap at k_max w.r.t. h[i]
        # is the coefficient of h[i] in the system, which is 1 - h[i+k_max]
        # But wait: that's the Jacobian. Let's discretely compute the derivative:
        # For a fixed lag k_max, overlap = sum(h[i] * (1 - h[i+k_max]))
        # d/dh[i] overlap = 1 - h[i+k_max] if i + k_max is valid
        # Define gradient at i:
        if k_max > 0:  # h[i] appears in h[i] * (1 - h[i + k_max])
            for i in range(n_points):
                j = (i + k_max) % n_points
                if j >= 0 and j < n_points:
                    grad[i] = -h[j]
            grad = grad + 1.0  # Simplified methodology of gradient for approximate update
        else:
            grad = -h.copy()
            grad = grad + 1.0
            
        # Solve linear subproblem: minimize grad^T h = sum(grad[i] * h[i])
        # subject to 0 <= h[i] <= 1 and sum(h) = n_points / 2  
        # This is a simple projected gradient approach, but to solve exact subproblem efficiently:
        # We want to minimize grad^T h s.t. h in feasible set.
        # Since the cost is linear and constraints are simple, 
        # we find d such that it minimizes d^T grad and is feasible.
        # Find direction that gives most negative dot product: d = argmin d^T grad
        # d should be feasible: 0 <= d[i] <= 1, sum d = n_points/2
        # Let's use the standard Frank-Wolfe approach: 
        # find simplified direction d minimizing d^T grad subject to constraints
        # A simple greedy approach for descent direction in feasible set:
        # set d = 0, then set components with largest grad to 1 until we hit sum constraint
        sign_grad = np.sign(grad)
        sign_grad = np.clip(sign_grad, 0, 1)  # Just ensure non negativity for direction
        # Fletcher's case: letterweight flipped sign for minimization direction:
        # Let's take simpler but similar approach:
        # Let d be projection of -grad to feasible region:
        # It is favorable to make the component with the most negative gradient to be 
        # at its lower bound (0) or upper bound (1) to do the descent
        # To instead simplify: update h towards a point with large negative gradient values:
        d = -grad  # roughly. Larger negative d will decrease obj faster
        d = np.clip(d, 0, 1)
        # Ensure sum constraint:
        current_sum = np.sum(d)
        if current_sum > 0:
            d = d * (n_points / 2) / current_sum
        else:
            # If all values are zero or invalid, use uniform perturbation
            d = np.full(n_points, 1.0 / n_points)
            
        # Perform line search (or simple update in FW)
        # Update: h = (1 - eta) * h + eta * d
        h = (1 - eta) * h + eta * d
        h = project_to_feasible(h, n_points)
        
        # Evaluate
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Maybe add early stopping?
        if abs(best_c5 - c5) < tol:
            break
            
    return best_h, best_c5
