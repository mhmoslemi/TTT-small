import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: initialize with uniform distribution
        n_points = 45
        h = np.ones(n_points) * 0.5

    # Ensure initial sum is correct
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.01
    min_alpha = 1e-6
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Helper to compute C5
    def compute_c5(h):
        # Correlate h with (1 - h) for all shifts
        corr = correlate(h, 1 - h, mode="full")
        # Use only valid correlations (not extended)
        half_len = len(h) - 1
        corr = corr[half_len:half_len + len(h)]
        return np.max(corr) * (2.0 / n_points)

    # Gradient-based update
    def update_h(h, alpha):
        h = h.copy()
        # Compute C5 gradient w.r.t. the worst shift
        corr = correlate(h, 1 - h, mode="full")
        half_len = len(h) - 1
        corr = corr[half_len:half_len + len(h)]
        k_max = np.argmax(corr)
        # Gradient computation (simplified)
        grad = np.zeros_like(h)
        # Refer to shift k_max in correlation
        if k_max < len(h):
            # Gradient for indices affected by shift k_max
            for i in range(len(h)):
                j = i + k_max
                if j < len(h):
                    grad[i] += (1 - h[j])  # contribution of h[j] to h[i]
                if i >= k_max:
                    grad[i] -= h[i - k_max]  # contribution of h[i - k_max] to h[i]
        # Update h
        h = h - alpha * grad
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to sum to n/2 exactly
        h = h * (n_points / 2) / np.sum(h)
        return h

    iter_count = 0
    last_improvement = 0
    current_c5 = compute_c5(h)
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 2:
        iter_count += 1
        
        # Save old value
        old_c5 = current_c5

        # Update h
        h = update_h(h, alpha)
        current_c5 = compute_c5(h)

        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1

        # Adapt learning rate
        if last_improvement > patience:
            alpha /= 2
            last_improvement = 0
            if alpha < min_alpha:
                break

    return (best_h.tolist(), best_c5, n_points)
