import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # compute all correlations for full mode
    corr = correlate(h, 1 - h, mode='full')
    # maximum correlation over all shifts
    max_corr = np.max(corr)
    # multiply by dx
    c5 = max_corr * dx
    return c5

def normalize(h, n_points):
    """Ensure sum(h) = n_points / 2"""
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def initialize_population(n_points, pop_size, initial_h_values=None):
    """Initialize population with diverse candidates"""
    pop = []
    # Add initial solution if available
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = normalize(h, n_points)
        pop.append(h.copy())
    
    # Add random candidates to reach pop_size
    while len(pop) < pop_size:
        h = np.random.rand(n_points)
        h = normalize(h, n_points)
        pop.append(h.copy())
    
    return pop

def crossover(parent1, parent2):
    """Perform crossover by swapping segments"""
    n_points = len(parent1)
    # Random crossover point
    cx_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, noise_scale=0.1):
    """Mutate h with noise, and re-normalize"""
    n_points = len(h)
    if np.random.rand() < mutation_rate:
        # Select random points to mutate
        n_mutate = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, n_mutate, replace=False)
        # Add noise
        for i in indices:
            noise = np.random.uniform(-noise_scale, noise_scale)
            h[i] += noise  # Could be negative, so need to clip and renormalize later
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    return h

def tournament_select(pop, fitnesses, tournament_size=0.2):
    """Tournament selection based on fitness (lower is better)"""
    n_pop = len(pop)
    n_select = max(1, int(n_pop * tournament_size))
    selected_indices = np.random.choice(n_pop, size=n_select, replace=False)
    selected_fitnesses = [fitnesses[i] for i in selected_indices]
    winner_idx = selected_indices[np.argmin(selected_fitnesses)]
    return pop[winner_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    pop_size = 100
    max_generations = 300
    mutation_rate = 0.2
    noise_scale = 0.1
    
    # Initialize tracking
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    last_improvement_gen = 0
    
    # Initialize population
    pop = initialize_population(n_points, pop_size, initial_h_values)
    fitnesses = [evaluate_erdos_solution(h) for h in pop]
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        pop = [pop[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = pop[0].copy()
            last_improvement_gen = gen
            
        # Elitism: keep top 10%
        elite_count = max(1, pop_size // 10)
        next_pop = pop[:elite_count]
        next_fitness = fitnesses[:elite_count]
        
        # Generate children
        while len(next_pop) < pop_size:
            # Selection
            p1 = tournament_select(pop, fitnesses)
            p2 = tournament_select(pop, fitnesses)
            
            # Crossover
            c1, c2 = crossover(p1, p2)
            
            # Mutation
            c1 = mutate(c1, mutation_rate, noise_scale)
            c1 = normalize(c1, n_points)
            c2 = mutate(c2, mutation_rate, noise_scale)
            c2 = normalize(c2, n_points)
            
            next_pop.extend([c1, c2])
            if len(next_pop) > pop_size:
                break
            
        # Update population and fitness
        pop = next_pop[:pop_size]
        fitnesses = [evaluate_erdos_solution(h) for h in pop]
        
        # Early exit
        if gen - last_improvement_gen > 50:
            break
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_erdos_solution(best_h)
        return (list(best_h), final_c5, n_points)
    else:
        # Fallback: return population best
        best_idx = np.argmin(fitnesses)
        return (list(pop[best_idx]), fitnesses[best_idx], n_points)
