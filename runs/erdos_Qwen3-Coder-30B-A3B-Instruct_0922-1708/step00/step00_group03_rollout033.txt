import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use the initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 70
        h0 = np.full(n_points, 1.0 / n_points)
    
    # Normalize initial guess to satisfy constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Track best solution
    best_h = h0.copy()
    best_C5 = evaluate_erdos_solution(h0)[1]
    
    # Use CVXPY to solve the convex relaxation
    # Variables
    x = cvx.Variable(n_points)
    Z = cvx.Variable((n_points, n_points))
    t = cvx.Variable()
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == n_points / 2,
        t >= 0
    ]
    
    # Z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i,j] <= x[i],
                Z[i,j] <= 1 - x[j],
                Z[i,j] >= x[i] + x[j] - 1
            ]
    
    # Correlation constraints
    C_values = []
    for k in range(-n_points + 1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        sum_z = cvx.sum(Z[np.arange(n_points)[valid], idx[valid]])
        C_values.append(sum_z)
    
    # Maximum over all correlations
    for c in C_values:
        constraints.append(c <= t)
        
    # Objective
    objective = cvx.Minimize(t)
    prob = cvx.Problem(objective, constraints)
    
    # Try to solve with different solvers
    solvers = ['MOSEK', 'SCS', 'OSQP', 'ECOS']
    solver_used = "unknown"
    
    for solver in solvers:
        try:
            if solver == "MOSEK":
                prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
            else:
                prob.solve(solver=solver, verbose=False, max_iters=5000)
            solver_used = solver
            break
        except:
            continue
    
    # If no solver succeeded, try the best we've found
    if not prob.is_dcp() or prob.status in ["infeasible", "unbounded"]:
        # Return the best solution found
        h_opt = best_h.copy()
        C5_bound = best_C5
    else:
        # Extract solution
        x_sol = x.value
        if x_sol is None:
            h_opt = best_h.copy()
            C5_bound = best_C5
        else:
            # Clip to [0,1] and rescale
            x_sol = np.clip(x_sol, 0, 1)
            x_sol = x_sol * (n_points / 2) / np.sum(x_sol)
            h_opt = x_sol
            C5_bound = evaluate_erdos_solution(h_opt)[1]
    
    # Ensure constraint is satisfied
    correct_sum = np.sum(h_opt)
    if abs(correct_sum - n_points / 2) > 1e-5:
        h_opt = h_opt * (n_points / 2) / correct_sum
    
    return (h_opt, C5_bound, n_points)
