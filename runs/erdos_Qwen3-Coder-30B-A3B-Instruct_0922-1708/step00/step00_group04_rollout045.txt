import numpy as np
from scipy import signal
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except NameError:
        # Default initial construction
        n_points = 42
        h_init = np.linspace(0, 1, n_points)
        h_init = np.minimum(h_init, 1 - h_init)
        h_init = np.clip(h_init, 0, 1)
        # Make sure it integrates to 1
        h_init = h_init * 2.0 / np.sum(h_init)
    
    dx = 2.0 / n_points
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h, dx)
    start_time = time.time()
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5:  # Keep some time for final eval
        if iter_count % 10 == 0:
            # Re-evaluate best solution to make sure we're not stuck
            current_c5 = evaluate_erdos_solution(best_h, dx)
            if abs(current_c5 - best_c5) < 1e-6:
                last_improvement += 1
            else:
                last_improvement = 0
                best_c5 = current_c5
            if last_improvement > 10:
                break

        # Linearize
        h_current = best_h.copy()
        
        # Evaluate the overlap
        corr = np.correlate(h_current, 1 - h_current, mode="full")
        k_indices = np.arange(len(corr) - n_points + 1, len(corr))
        max_k_idx = k_indices[np.argmax(corr[k_indices])]
        
        # Construct the linearized problem: minimize the largest overlap
        h = cp.Variable(n_points)
        
        # Constraint: 0 <= h <= 1
        constraints = [h >= 0, h <= 1]
        
        # Constraint: sum(h) * dx = 1
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Objective: minimize the linearized overlap at max_k_idx
        # Linearize h(x)(1-h(x+k)) at current h
        # We linearize at h_current, so the linearization is h(x) + (1-h(x+k)) - 1
        # This corresponds to h(x) - h(x+k) = h(x) - (h_current[(x + max_k_idx) % n_points])
        
        # Since the constraint is to maximize and we're minimizing the negative,
        # the linearization is h[x] + h[(x + max_k_idx) % n_points] - 1
        # so we minimize h[x] + h[(x + max_k_idx) % n_points] - 1 for each x
        # However, we actually want to minimize the max of the above over all k
        
        # So let's just linearize the integral at a selected k
        linearized_term = cp.sum(cp.multiply(h, 1 - h_current[max_k_idx:] + 
                                             h_current[:n_points - max_k_idx]))
        # Adjust for the fact that it's the middle-of-overlap, and we sum all over k
        # This approximation is direct for each span
        # Instead, just use the computed maximum to linearize at that k
        
        # To make it exact for a particular k:
        # For maximum k that gives max overlap (say k_opt), 
        # we minimize over all i: h[i] - h[i+k_opt] + 1 * dx = constant for that k
        # Since for the integral we sum over all k and take the maximum, 
        # we use the current maximum k to linearize and thus minimize the overlap term
        
        # Lets compute the actual line of the linearization of a part of integral
        # h(x)(1 - h(x + k)) at h(x) = current_h[x]
        # Linearize to get: h(x) * (1 - current_h[x + k]) + current_h[x] * (1 - h(x + k))
        # Total resulting term is: h(x) + current_h[x] - h(x) * current_h[x + k] - current_h[x] * h(x + k) 
        # We linearize to h(x) + current_h[x] - current_h[x + k] * h(x) 
        # But it's simpler to approximate the objective directly.
        # For overlap at lag k_opt: \sum h[i] * (1 - h[i + k_opt]) dx = \sum h[i] - h[i] * h[i + k_opt] dx
        # Linearized: h[i] - current_h[i + k_opt] * h[i] = h[i] * (1 - current_h[i + k_opt])  
        # Which is same as above: for each term with h_current, we linearize differently
        # But we want to approximate the evaluation of overlap at k_opt, and that's h[i] (1-h[i + k_opt])
        
        # The linearized expression is (at max_k_idx):
        # h[i] (1 - h[i + k_opt]) 
        # Linearized around hat_h = h_current: 
        # hat_h[i] (1 - hat_h[i + k_opt]) + (1 - hat_h[i + k_opt]) (h[i] - hat_h[i])  
        # = hat_h[i] (1 - hat_h[i + k_opt]) +  (h[i] - hat_h[i]) (1 - hat_h[i + k_opt]) 
        # Actually, to make it a linear function:
        #  = hat_h[i] (1 - hat_h[i + k_opt]) + (1 - hat_h[i + k_opt]) * h[i] - (1 - hat_h[i + k_opt]) * hat_h[i]  
        # = (1 - hat_h[i + k_opt]) * h[i]  
        # But that isn't good either. Let's change to simplest convex method...
        
        # Let's take a more direct path: 
        # Look at idx_k = k_opt, get sum(h[i] * (1 - h[i+k_opt])) * dx
        # Linearized around h_current: 
        # terms[i] = h[i] * (1 - h_current[i+k_opt]) + h_current[i] * (1 - h[i+k_opt])
        # We want to minimize the sum of those terms integrated = sum_{i} term[i] 
        # which is \sum h[i] * (1 - h_current[i+k_opt]) + \sum h_current[i] *(1 - h[i+k_opt])
        # We fix h_current and let h vary => minimizing in h:
        # sum_{i} ( h[i] * (1 - h_current[i+k_opt]) ) 
        # corresponding to minimizing: - \sum h[i] * h_current[i+k_opt] 
        # The values h_current[i+k_opt] are fixed.
        
        # That's why the linearization is just the summation of h[i] * (1 - h_current[i+k_opt])
        # Set the objective to minimize that = 
        # == minimize sum h[i] - sum h[i] * h_current[i + max_k_idx] = 
        # sum h[i] - sum h[i] * h_current[i + max_k_idx]
        
        # Objective is \sum h[i] (1 - h[i + k_opt]) |opt = \sum h[i] - sum h[i] * h_current[i + k_opt]
        # This is convex, and we can just solve it directly (i.e., not to approximate the overlaps)
        # Let's just solve minimize \sum h[i] * (1 - h_current[i + k_opt])
        # Be careful with circular indexing
        h_shift = np.roll(h_current, -max_k_idx)
        objective = cp.sum(cp.multiply(h, 1 - h_shift))
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h.value
            else:
                h_new = h_current
        except:
            h_new = h_current

        # Apply constraints (clip and rescale)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)  # Rescale to sum to n_points/2

        # Evaluate the function
        c5 = evaluate_erdos_solution(h_new, dx)
        
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5
            last_improvement = 0 
            if c5 < 0.3807:
                break
        
        iter_count += 1
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h_values, dx):
    # Compute correlation
    corr = signal.correlate(h_values, 1 - h_values, mode='full')
    # Take valid lags from full correlation
    valid_corr = corr[len(h_values)-1:]
    # Return maximum scaled by dx
    return np.max(valid_corr) * dx
