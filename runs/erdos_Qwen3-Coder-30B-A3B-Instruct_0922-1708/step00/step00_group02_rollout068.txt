import numpy as np
from scipy.special import expit
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 100  # smaller for faster optimization
    dx = 2.0 / n_points
    max_iter = 500
    
    def sigmoid(x):
        return expit(x)
    
    def basis_functions(x, centers, scales):
        return np.array([sigmoid((x - c) / s) for c, s in zip(centers, scales)])
    
    def eval_h(a, x, centers, scales):
        return a[0] + np.dot(a[1:], basis_functions(x, centers, scales))
    
    def compute_overlap(a, centers, scales, x_vals):
        h_vals = eval_h(a, x_vals, centers, scales)
        h_vals = np.clip(h_vals, 0, 1)
        overlap_vals = []
        for k in range(len(x_vals)):
            # correlation for lag k
            if k == 0:
                corr_val = np.sum(h_vals * (1 - h_vals)) * dx
            else:
                corr_val = np.sum(h_vals[:-k] * (1 - h_vals[k:])) * dx
            overlap_vals.append(corr_val)
        return np.max(overlap_vals)
    
    def compute_gradient(a, centers, scales, x_vals):
        h_vals = eval_h(a, x_vals, centers, scales)
        h_vals = np.clip(h_vals, 1e-6, 1 - 1e-6)
        
        # Gradient w.r.t. a for max_k : integral h(x)(1 - h(x+k)) dx
        # Approximate using average gradient over lags
        grad = np.zeros_like(a)
        for k in range(1, len(x_vals) // 2):
            if k >= len(x_vals): continue
            h_slice = h_vals[:-k]
            h_k_slice = h_vals[k:]
            # derivative of integral h(x)(1 - h(x+k))
            # w.r.t a: h(x)(-h'(x+k)) + (1-h(x+k))h'(x)
            h_prime = np.gradient(h_vals, dx)
            grad += -h_slice * h_prime[k:] - (1 - h_k_slice) * h_prime[:-k]
        return grad * dx
    
    # Setup basis functions
    m = int(np.sqrt(n_points) * 2)  # number of basis functions
    centers = np.linspace(0, 2, m)
    scales = np.full(m, 2.0/m)
    x_vals = np.linspace(0, 2, n_points)
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = initial_h_values
        # Fit a coefficients
        A = np.array([np.concatenate(([1], basis_functions(x, centers, scales))) for x in x_vals])
        # Solve least squares
        a = np.linalg.lstsq(A, h_init, rcond=None)[0]
        # So h approximates initial_h_values
        a = np.clip(a, -1.0, 1.0)
    else:
        a = np.random.uniform(-0.1, 0.1, m+1)
    
    # Ensure integral constraint: sum(h)*dx = 1 => sum(h) = n_points/2
    h_init = eval_h(a, x_vals, centers, scales)
    h_init = np.clip(h_init, 0, 1)
    norm_factor = np.sum(h_init) / (n_points / 2)
    a[0] -= np.log(norm_factor)  # approximate effect on constant term
    
    # Optimizer parameters
    lr = 0.01
    gamma = 0.8  # Elastic net
    best_c5 = float('inf')
    best_a = a.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break
            
        # Compute current overlap
        c5 = compute_overlap(a, centers, scales, x_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a.copy()
        
        # Compute gradient
        grad = compute_gradient(a, centers, scales, x_vals)
        
        # Update a
        a -= lr * grad
        # Elastic net: L1 + L2
        a[1:] -= lr * (gamma * np.sign(a[1:]) + (1 - gamma) * a[1:])
        
        # Project to [0,1]
        h_vals = eval_h(a, x_vals, centers, scales)
        h_vals = np.clip(h_vals, 0, 1)
        
        # Rescale to meet constraint
        h_sum = np.sum(h_vals) * dx
        if abs(h_sum - 1.0) > 1e-6:
            # Linear scaling
            scale_factor = 1.0 / h_sum * (n_points/2) / n_points
            a[0] += np.log(scale_factor)
    
    # Final discretization
    h_vals = eval_h(best_a, x_vals, centers, scales)
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals / np.sum(h_vals) * (n_points / 2)
    
    # Evaluate final C5
    overlap_full = np.correlate(h_vals, 1 - h_vals, mode='full')
    c5_final = np.max(overlap_full) * dx
    
    # Ensure result is within constraints
    assert np.all(h_vals >= 0) and np.all(h_vals <= 1)
    assert abs(np.sum(h_vals) - n_points / 2) < 1e-6
    
    return h_vals, c5_final, n_points
