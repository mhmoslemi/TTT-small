from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        return 1.0 / (1.0 + np.exp(-x))

        return sigmoid((x - mu[j]) / sigma)

        return np.clip(a[0] + np.sum([a[j+1] * basis(x, j) for j in range(m)]), 0, 1)

    # Optimization loop with surrogate model
    
    # We will optimize the coefficient vector a that parameterizes h via basis functions
    def compute_overlap(a):
        # Construct h
        x_vals = np.linspace(0, 2, n_points)
        h_vals = np.array([h_func(a, x) for x in x_vals])
        
        # Compute cross-correlations
        h_rev = np.flip(h_vals)
        corr = np.correlate(h_vals, h_rev, mode="full")
        # adjust for the order of correlations
        correlations = []
        for k in range(len(corr)):
            lag = k - (len(h_vals) - 1)
            if lag >= 0:
                # integral becomes sum * dx over the shifted overlap
                overlap = np.sum(h_vals[:len(h_vals)-lag] * (1 - h_vals[lag:]))
            else:
                overlap = np.sum(h_vals[-lag:] * (1 - h_vals[:len(h_vals)+lag]))
            correlations.append(overlap * dx)
        return max(correlations)
    
    # Elastic net regularized optimization
    def objective(a):
        return compute_overlap(a)
    
    # Initial coefficients
    initial_a = np.random.normal(0, 0.1, m+1)
    initial_a[0] = 0.5  # start with average value
    
    if 'initial_h_values' in globals():

        # A and b from A @ a = b for a linear system representing h in basis
        A = np.zeros((n_points, m+1))
        x_vals = np.linspace(0, 2, n_points)
        A[:,0] = 1
        for j in range(m):
            A[:,j+1] = [basis(x, j) for x in x_vals]

            a_init = np.linalg.lstsq(A, h_init, rcond=None)[0]
            initial_a = a_init

    def gradient(a):
        # approximate gradient using finite differences
        epsilon = 1e-4
        grad = np.zeros_like(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += epsilon
            a_minus = a.copy()
            a_minus[i] -= epsilon
            grad[i] = (compute_overlap(a_plus) - compute_overlap(a_minus)) / (2*epsilon)

            elastic_coeff = 0.1

            l2_penalty = (1 - elastic_coeff) * best_a
            
            d_a = grad - l1_penalty - l2_penalty
            new_a = best_a - step_size * d_a

    h_final = h_final * scale_factor
    c5_final = compute_overlap(best_a)
    
    # Final validation
    c5_final = max(np.correlate(h_final, 1 - h_final, mode="full")) * dx
    
    return (h_final, c5_final, n_points)
