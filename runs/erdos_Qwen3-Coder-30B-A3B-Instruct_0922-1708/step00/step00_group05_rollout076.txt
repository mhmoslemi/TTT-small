import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1-h, mode='full')
    # The maximum value of the correlation corresponds to the maximum overlap
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use the provided initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback: create a basic step function
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        # Normalize to have sum = n_points / 2
        h *= (n_points / 2) / np.sum(h)
    
    # Ensure h is valid
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / np.sum(h)
    
    # Evaluate initial C5
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Parameters
    perturbation_delta = 0.05
    num_trials = 100
    smoothing_window = 3
    max_no_improvement = 100
    no_improvement_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s:
        improved = False
        
        for _ in range(num_trials):
            if time.time() - start_time >= budget_s:
                break
                
            # Create a perturbed version of h
            h_trial = h.copy()
            
            # Randomly select points to perturb
            num_perturb = max(1, int(0.1 * n_points))
            indices = np.random.choice(n_points, num_perturb, replace=False)
            
            for i in indices:
                # Random perturbation
                delta = np.random.uniform(-perturbation_delta, perturbation_delta)
                h_trial[i] += delta
                
            # Ensure values stay in [0,1]
            h_trial = np.clip(h_trial, 0, 1)
            
            # Adjust the sum to be exactly n_points / 2
            current_sum = np.sum(h_trial)
            correction = (n_points / 2) - current_sum
            # Distribute correction uniformly
            h_trial += correction / n_points
            
            # Ensure again after adjustment
            h_trial = np.clip(h_trial, 0, 1)
            h_trial *= (n_points / 2) / np.sum(h_trial)
            
            # Apply smoothing to maintain structure
            if smoothing_window > 0:
                # Simple moving average
                smoothed = np.zeros_like(h_trial)
                for i in range(len(h_trial)):
                    left = max(0, i - smoothing_window//2)
                    right = min(len(h_trial), i + smoothing_window//2 + 1)
                    smoothed[i] = np.mean(h_trial[left:right])
                h_trial = smoothed
                
            # Evaluate new C5
            new_c5 = evaluate_erdos_solution(h_trial)
            
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h_trial.copy()
                improved = True
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
            # Update h for next iteration if improved
            if improved:
                h = h_trial.copy()
                
        if not improved:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            
        if no_improvement_count >= max_no_improvement:
            break
            
    return (best_h.tolist(), best_c5, n_points)
