import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h."""
    # Compute correlation of h with 1-h for all shifts
    correlation = correlate(h, 1 - h, mode='full')
    # Multiply by dx to get actual overlap values
    overlaps = correlation * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with some diversity."""
    population = []
    
    # Add initial solution if available
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill the rest with random individuals
    while len(population) < pop_size:
        # Create a random vector, then normalize to meet constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1] to ensure bounds
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select a parent using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    # Single-point crossover
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Mutate an individual."""
    n = len(individual)
    for i in range(n):
        if np.random.rand() < mutation_rate:
            individual[i] += np.random.normal(0, 0.05)
    
    # Clip values to [0,1]
    individual = np.clip(individual, 0, 1)
    
    # Normalize to meet constraint
    individual = individual * (len(individual) / 2) / np.sum(individual)
    return individual

def ensure_constraints(h, n_points):
    """Ensure h is in [0,1] and sum equals n_points/2."""
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

    max_generations = 120
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Select parents and create offspring
        new_population = []
        
        # Elitism: keep top 20%
        elite_count = pop_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = ensure_constraints(child1, n_points)
            child2 = ensure_constraints(child2, n_points)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
    
    # Final evaluation of best individual
    if best_h is not None:
