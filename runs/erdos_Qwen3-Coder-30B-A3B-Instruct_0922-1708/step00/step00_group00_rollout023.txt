import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Get the maximum value (corresponds to max k)
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy the constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to maintain sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def perturb_h(h, sigma):
    """Add Gaussian noise with standard deviation sigma."""
    noise = np.random.normal(0, sigma, len(h))
    return h + noise

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)  # Initial guess
        # Slight perturbation
        h += np.random.normal(0, 0.01, n_points)
    
    # Ensure sum constraint is satisfied
    h = project_h(h, n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Decreasing variance for noise
    sigma = 0.1
    decay = 0.99
    
    while time.time() - start_time < budget_s:
        iteration += 1
        # Generate perturbed h
        h_new = perturb_h(best_h, sigma)
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update sigma
        sigma *= decay
        
        # Early termination if no improvement after many iterations
        if no_improvement_count > max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
