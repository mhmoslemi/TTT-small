import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial step function
        h_init = np.ones(n_points)
        h_init[n_points//2:] = 0.0
    
    # Fourier representation initialization with even symmetry
    m = 10  # Number of cosine modes (do not exceed 20 for fast convergence)
    
    # Initialize coefficients: a_k for k=0 to m-1
    # Use a0 = 0.5 to enforce integral constraint
    coeffs = np.random.rand(m) * 0.5
    coeffs[0] = 0.5  # integral constraint
    
    # Optimization loop
    start_time = time.time()
    best_c = float('inf')
    best_h = None
    eval_count = 0
    
    def compute_h_from_coeffs(coeffs):
        # Compute h(x) from Fourier coeffs a_k
        # h(x) = sum_{k=0}^{m-1} a_k * cos(pi * k * x / 2)
        x_vals = np.linspace(0, 2, n_points, endpoint=False)
        h_vals = np.zeros(n_points)
        for k in range(m):
            h_vals += coeffs[k] * np.cos(np.pi * k * x_vals / 2)
        return np.clip(h_vals, 0, 1)
    
    def enforce_integral(h_vals):
        # Ensure ∫ h(x) dx = 1
        current_sum = np.sum(h_vals) * dx
        if current_sum > 0:
            h_vals = h_vals * target_sum / current_sum
        return h_vals
    
    def compute_c5(h_vals):
        # Compute the maximum overlap
        h_padded = np.concatenate([h_vals, np.zeros(n_points)])
        h_rev = np.flip(h_vals)
        conv = np.convolve(h_padded, h_rev, mode='full')
        overlap = conv[:2*n_points-1]
        c5_val = np.max(overlap) * dx
        return c5_val
    
    # Gradient computation for update
    def compute_gradient(coeffs, k_idx):
        # Feedforward to get h_vals approx
        h_vals = compute_h_from_coeffs(coeffs)
        h_vals = enforce_integral(h_vals)
        
        # Compute the overlap integral for a stretch of indices around k_idx
        x_vals = np.linspace(0, 2, n_points, endpoint=False)
        # Use finite difference here for gradient
        eps = 1e-5
        grad_coeffs = np.zeros(m)
        base_c = compute_c5(h_vals)
        
        for k in range(m):
            coeffs_pert = coeffs.copy()
            coeffs_pert[k] += eps
            h_pert = compute_h_from_coeffs(coeffs_pert)
            h_pert = enforce_integral(h_pert)
            c_pert = compute_c5(h_pert)
            grad_coeffs[k] = (c_pert - base_c) / eps
            
        return grad_coeffs
    
    # Optimization parameters
    learning_rate = 0.01
    patience = 100
    no_improvement_count = 0
    
    # Main loop
    while time.time() - start_time < budget_s - 1:
        # Compute current h from coeffs
        h = compute_h_from_coeffs(coeffs)
        h = enforce_integral(h)
        c5_current = compute_c5(h)
        eval_count += 1
        
        # Track best
        if c5_current < best_c:
            best_c = c5_current
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > patience:
            break
            
        # Select worst k
        # We'll use a quick method to sample k
        x_vals = np.linspace(0, 2, n_points, endpoint=False)
        idx_to_check = np.arange(0, n_points)  # Sample a few indices
        max_overlap = 0
        worst_idx = 0
        for idx in idx_to_check:
            h_x = np.roll(h, -idx) * (1 - h)
            overlap = np.sum(h_x) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                worst_idx = idx
                
        # Compute gradient for worst case
        grad = compute_gradient(coeffs, worst_idx)
        # Update step
        coeffs -= learning_rate * grad
        coeffs = np.clip(coeffs, 0, 1)
        
        # Periodically rescale to ensure integral
        if eval_count % 10 == 0:
            h = compute_h_from_coeffs(coeffs)
            h = enforce_integral(h)
            c5_check = compute_c5(h)
            if c5_check < best_c:
                best_c = c5_check
                best_h = h.copy()
            
    if best_h is None:
        h = compute_h_from_coeffs(coeffs)
        h = enforce_integral(h)
        best_h = h
        best_c = compute_c5(h)
    
    return (best_h.tolist(), best_c, n_points)
