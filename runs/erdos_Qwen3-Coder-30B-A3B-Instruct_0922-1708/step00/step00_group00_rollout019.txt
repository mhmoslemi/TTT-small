import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    # Compute the correlation of h with (1-h) in full mode
    correlation = correlate(h, 1 - h, mode='full')
    # Scale by dx to get the actual overlap
    overlaps = correlation * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize the population of step functions."""
    population = []
    
    # If initial solution exists, add it with some variation
    if initial_h is not None:
        population.append(initial_h.copy())
        # Add variations
        for _ in range(pop_size // 5):
            h = initial_h.copy()
            # Add small random perturbations
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
            # Clip and renormalize
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h)
            if h_sum > 0:
                h = h * (n_points / 2) / h_sum
            population.append(h.copy())
    
    # Fill the rest with random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual via tournament selection."""
    # Select tournament_size individuals at random
    selected_indices = np.random.choice(len(population), size=tournament_size)
    # Choose the best one from the tournament
    best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return population[best_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    # Random crossover point
    cx_point = np.random.randint(1, n_points)
    
    # Create offspring
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05, clip_range=(0, 1)):
    """Apply mutation to an individual."""
    # Create a copy of the individual
    mutated = h.copy()
    # Determine how many positions to mutate
    n_mutations = int(mutation_rate * len(h))
    mutation_indices = np.random.choice(len(h), size=n_mutations, replace=False)
    
    for idx in mutation_indices:
        # Add small random change
        mutated[idx] += np.random.normal(0, 0.05)
        # Clip and renormalize
        mutated = np.clip(mutated, clip_range[0], clip_range[1])
        # Re-normalize
        h_sum = np.sum(mutated)
        if h_sum > 0:
            mutated = mutated * (len(h) / 2) / h_sum
    
    return mutated

    start_time = time.time()

        if time.time() - start_time > budget_s * 0.95:
            break
        
        h, c5 = genetic_algorithm(n_points=n_points, time_budget=budget_s)
        
        if c5 < best_c5:
            best_h = h
            best_c5 = c5
            best_n_points = n_points
    
    return best_h, best_c5, best_n_points
