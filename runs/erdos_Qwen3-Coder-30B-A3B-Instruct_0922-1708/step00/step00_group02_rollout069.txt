import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time
from math import sqrt

# Precomputed initial_h_values if available
try:
    initial_h_values
except NameError:
    initial_h_values = None

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 200  # A reasonable balance between resolution and optimization time
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Handle initial construction
    if initial_h_values is not None:
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * target_sum
    else:
        # Create a reasonable initial function
        h_start = np.ones(n_points) * 0.5
        # Make it more step-like
        h_start[:n_points//4] = 0.9
        h_start[n_points//4:3*n_points//4] = 0.1
        h_start[3*n_points//4:] = 0.9
        # Normalize
        h_start = h_start / np.sum(h_start) * target_sum

    # Optimization variables
    h_current = h_start.copy()
    best_h = h_current.copy()
    best_c5 = float('inf')
    start_time = time.time()

    # Early termination if time is almost out
    def check_time():
        return time.time() - start_time > 0.95 * budget_s

    # Basis functions definition
    # Use sigmoid-based basis functions for smooth representations
    def build_basis(n_basis, domain=(0, 2)):
        x = np.linspace(domain[0], domain[1], n_points)
        centers = np.linspace(domain[0], domain[1], n_basis)
        scales = 0.1 * (domain[1] - domain[0]) / n_basis
        basis = []
        for i in range(n_basis):
            b = expit((x - centers[i]) / scales)
            basis.append(b)
        return np.array(basis).T  # shape (n_points, n_basis)

    # Build basis with number of basis functions proportional to n_points
    n_basis = max(10, int(2 * sqrt(n_points)))
    basis = build_basis(n_basis, (0, 2))
    
    # Process: Doing iterative descent on coefficients
    # To do so: Map h_current -> coefficients in basis space
    def to_coefficients(h):
        # Use least squares to find coefficients that best approximate h
        # using the basis functions
        # We solve: basis @ coeffs = h
        try:
            coeffs = np.linalg.lstsq(basis, h, rcond=None)[0]
        except np.linalg.LinAlgError:
            coeffs = np.random.rand(n_basis)  # fallback
        return coeffs

    def from_coefficients(coeffs):
        # Reconstruct h from coefficients
        h = basis @ coeffs
        # Clip and rescale to ensure sum and bounds
        h = np.clip(h, 0, 1)
        rescale_factor = target_sum / np.sum(h)
        h = h * rescale_factor
        return h

    # Compute overlap
    def compute_overlap(h):
        # Overlap is the maximum over all lags
        # Use scipy convolve for speed
        conv = np.correlate(h, 1 - h, mode='full')
        # We want overlaps for lag k in [0, n_points]
        # But it's symmetric due to [0 .. 2] padding, so just check full length
        # and extract appropriate range
        k_indices = np.arange(len(conv))
        # Which values correspond to k in [0, 2]?
        # The sizes are such that:
        # conv length is 2*n_points - 1
        # We want overlaps for lag k = 0 to n_points - 1 (corresponding to shift)
        # The convolution results are indexed from -(n_points-1) ... to n_points-1
        # i.e. for indices 0 to n_points-1, odds are negative lags, evens are positive
        # We map it properly to make 
        overlaps = []
        for k in range(n_points):
            idx = len(conv) // 2 - (n_points - 1) + k  # This should be the index for lag=k
            if 0 <= idx < len(conv):
                overlaps.append(conv[idx] * dx)
            else:
                overlaps.append(0)  # bounds error would be rare but just in case
                
        # Or just compute based on where the actual lags are:
        # The convolution for mode='full' spans from -n_points+1 to n_points-1
        # Representing lags from -n_points+1 to n_points-1
        # so for lag k, we use index (n_points-1) + k
        # But it's simpler to just do it correctly:
        # concat h = h + [0*0.1]
        idx_offset = n_points - 1
        # So lag k result is at index idx_offset + k 
        # For k = 0 to n_points-1
        overlaps = []
        for k in range(n_points):
            conv_idx = idx_offset + k
            overlaps.append(conv[conv_idx] * dx)
        return max(overlaps)

    # Optimization loop
    coeffs = to_coefficients(h_current)
    coeffs = np.clip(coeffs, -1, 1)  # allow reasonable bounds

    # Mini-batch annealing for learning rate
    learning_rate = 0.02
    n_iter = 500
    
    # Optimization with time tracking
    for i in range(n_iter):
        if check_time():
            break
            
        h_current = from_coefficients(coeffs)
        
        # Compute current C5 - the maximum overlap
        current_c5 = compute_overlap(h_current)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h_current.copy()
        
        # Compute gradient using perturbation method
        # This will help in direction of decreasing overlaps
        # Stochastic approx of gradient to compute descent
        # Compute gradient approx of overlap wrt h with finite diffs
        # but we don't have it analytically so approximate it.
        
        # Simple gradient estimation with subgradient-like update
        # based on slight changes to coefficients and measure change in overlap
        
        # Helper to compute specific overlap term corresponding to a lag
        # Let's compute specific gradient components 
        # for minimizing overlap wrt to coefficients
        
        # For simplicity, compute direct finite difference for gradient
        epsilon = 1e-5
        grad = np.zeros_like(coeffs)
        for j in range(n_basis):
            # Perturb one coefficient
            coeffs_plus = coeffs.copy()
            coeffs_plus[j] += epsilon
            h_plus = from_coefficients(coeffs_plus) 
            overlap_plus = compute_overlap(h_plus)
            
            coeffs_minus = coeffs.copy()
            coeffs_minus[j] -= epsilon
            h_minus = from_coefficients(coeffs_minus)
            overlap_minus = compute_overlap(h_minus)
            
            grad[j] = (overlap_plus - overlap_minus) / (2*epsilon)
            
        # Apply gradient update with elastic net regularization
        # L1 + L2 penalty
        gamma = 0.7
        elastic_penalty = gamma * np.sign(coeffs) + (1-gamma) * coeffs
        
        # Optimistic idea: try using only the steepest coefficient updates
        # but at least ensure it's descending
        coeffs = coeffs - learning_rate * grad - learning_rate * elastic_penalty
        # Keep coeffs within some safe range in terms of soft upper/lower bounds
        coeffs = np.clip(coeffs, -2.0, 2.0)
        
        # Reduce learning rate if stall
        if i > 0 and i % 100 == 0:
            learning_rate *= 0.8
        
    # Final rescale just in case to fix minor normalization issues
    h_final = from_coefficients(coeffs)
    current_c5 = compute_overlap(h_final)
    
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h_final.copy()

    return (best_h, best_c5, n_points)
