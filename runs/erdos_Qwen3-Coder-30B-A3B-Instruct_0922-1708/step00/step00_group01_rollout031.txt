import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem
    n_points = 47  # Use the provided initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert len(h) == n_points
        # Normalize to match sum constraint
        h = h * target_sum / np.sum(h)
    else:
        # Create a starting configuration
        h = np.ones(n_points) * 0.5
        # Centered plateau with tapering
        half_width = n_points // 3
        center = n_points // 2
        for i in range(n_points):
            distance = min(abs(i - center), abs(i - (center + half_width)), abs(i - (center - half_width)))
            if distance > half_width:
                h[i] = 0.1
            else:
                h[i] = 0.1 + (0.9 * (1.0 - distance / half_width))
        h = h * target_sum / np.sum(h)
        
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * target_sum / np.sum(h)
    
    # Parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iterations = 3000
    convergence_tol = 1e-5
    
    # Track best solution
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    start_time = time.time()
    
    # Gradient descent loop
    for iteration in range(max_iterations):
        # Check time budget
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute overlaps for all lags
        h_flip = 1 - h
        # Cross-correlation with "full" mode for all lags
        overlaps = signal.correlate(h, h_flip, mode='full')
        # Adjust for the indexing shift and scaling
        overlaps = overlaps * dx
        # The offset of the correlation is len(h)-1
        overlap_shift = len(h) - 1
        # Remove offset leads to correct lags
        overlap_values = overlaps[overlap_shift:]
        
        # Smooth approx to max using log-sum-exp
        smooth_max = np.log(np.sum(np.exp(tau * overlap_values))) / tau
        C5_approx = smooth_max
        
        # Compute gradient w.r.t h
        # Gradient of log-sum-exp is weighted mean of gradients
        exp_term = np.exp(tau * overlap_values)
        exp_sum = np.sum(exp_term)
        
        # Gradient of overlap(k) w.r.t h[i]
        # For overlap(k) = sum(h[i] * (1-h[(i+k)%n])) * dx
        # d/dh[i] overlap(k) = (1 - h[(i-k)%n]) * dx
        # We compute this efficiently using cross-correlation of gradient terms
        
        # Precompute terms for gradient calculation
        grad = np.zeros(n_points)
        for k in range(n_points):
            # For the k-th overlap:
            # gradient contribution from h[i] is (1 - h[(i-k)%n])
            # which can be computed as one cross-correlation of (1 - h) with a shifted h
            shifted_h = np.roll(h, k)
            grad_term = (1 - shifted_h)
            # Cross correlation of per-element differences with current h
            # So we parallelize this with correcting indices
            # The derivative with respect to h[i] of overlap(k) is (1 - h[(i-k)%n])
            # This is (1-h) convolved with a k-shifted h (no shift needed in interpret)
            # Cross-corr implementation for this:
            # But let's compute dC5/dh[i] directly as sum over all overlaps:
            # gradient contribution of each overlap component due to h[i] is: 
            # contribution from overlap_k = (1 - h[(i-k)%n]) * dx
            # So for gradient of log-sum-exp with respect to h[i]
            # we sum for all k of (exp(tau*overlap_k)/exp_sum) * -(1-h[(i-k)%n]) * dx
            
            # Compute what I actually need:
            # We want d/dh[i] of sum over k of (exp(tau*overlap_k)/exp_sum) * overlap_k
            # But it's just a weighted sum of per-k components
            
            # For each overlap:
            for j in range(n_points):
                this_offset = (j - k) % n_points
                grad[j] += exp_term[k] / exp_sum * (-1.0) * dx * (1.0 - h[this_offset])
        
        # Final gradient: 
        grad = grad / exp_sum
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Maintain sum constraint
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Check for improvement
        C5_new = evaluate_C5(h, dx)
        if C5_new < best_C5:
            best_C5 = C5_new
            best_h = h.copy()
            
        # Anneal τ and reduce learning rate
        tau *= anneal_factor
        if tau < 1e-6:
            tau = 1e-6
        if time.time() - start_time > budget_s * 0.9:
            learning_rate *= 0.9995
            
    return (best_h.tolist(), float(best_C5), int(n_points))

def evaluate_C5(h, dx):
    # Using signal.correlate for full cross-correlation approach to get maximum overlap
    h_flip = 1 - np.array(h)
    overlaps_full = signal.correlate(h, h_flip, mode='full') * dx
    # Extract the valid middle overlaps (that correspond to lags from 0 to n-1)
    n = len(h)
    overlap_shift = n - 1
    overlaps = overlaps_full[overlap_shift:]
    return float(np.max(overlaps))
