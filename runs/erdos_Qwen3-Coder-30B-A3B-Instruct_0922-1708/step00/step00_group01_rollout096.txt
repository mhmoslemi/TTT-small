import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C5 constant for a given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for all lags
    corr = correlate(h, 1 - h, mode="full")
    # The full correlation has 2*n - 1 elements. We want lags from -n+1 to n-1
    # But we only care about the valid overlaps within the 0,2 interval
    # The relevant part is from index n-1 to 2*n-2, which corresponds to lag 0 to n-1
    # And from index 0 to n-2, which corresponds to lag -n+1 to -1
    # But for max computation across all positive lags, let's check all lags 0 to n-1
    # and via symmetry we can consider negative lags up to -n+1
    # Since we're looking at lags in -n+1 to n-1:
    # For lag k, indices are i and i+k, we want 0 <= i < n, 0 <= i+k < n
    # This gives -k <= i < n-k, so for k>=0: 0 <= i < n-k, for k<0: -k <= i < n
    # But since we want max of abs(lag) up to n-1:
    # The maximum overlap for each k will be at the middle
    # To make it exact: looking at the full convolution and extracting correct parts
    result = np.max(corr)
    return result * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region with sum = n_points/2 and bounds [0,1]"""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def solve_erdos_problem(n_points, initial_h, max_time_s):
    """Frank-Wolfe style optimization for Erdős problem"""
    start_time = time.time()
    h = initial_h.copy()
    h = project_to_feasible(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    # Learning rate
    eta = 0.1
    max_iter = 5000
    tol = 1e-6
    
    for iteration in range(max_iter):
        if time.time() - start_time > max_time_s - 1:
            break
            
        # Compute the current maximum overlap and the corresponding lag
        dx = 2.0 / n_points
        corr = correlate(h, 1 - h, mode="full")
        # Extract all overlaps 
        overlap_values = []
        for k in range(-n_points + 1, n_points):
            # Compute overlap for lag k
            if k >= 0:
                # This is from indices 0 to n-k-1, to indices k to n-1
                if k < n_points:
                    overlap = np.sum(h[:n_points-k] * (1 - h[k:]))
                else:
                    overlap = 0
            else:
                # k < 0, indices -k to n-1, to indices 0 to n+1+k-1
                #y = h[-k:] * (1 - h[:n_points+k])
                if -k < n_points:
                    overlap = np.sum(h[-k:] * (1 - h[:n_points+k]))
                else:
                    overlap = 0
            overlap_values.append(overlap * dx)
            
        k_max = np.argmax(overlap_values)
        lag = k_max - n_points + 1
        
        # Gradient of overlap w.r.t. h
        # For lag k_max, we have to compute one-step derivative
        # This forces 1-h[(i+k_max)%n] for valid indices i
        # So: gradient at i is -1 if i+k_max < n, then 1 if i-k_max >= 0.  
        # This is more complex, better to work based on derivative
        # Gradient of sum_i h_i*(1-h_{i+k}) 
        # d/dh_i = (1 - h_{i+k})  if i+k >= 0, 0 else
        #       If i + k < n, d/dh_i = 1 - h_{i+k} 
        #       If i+k < 0, then h_{i+k} doesn't exist. treat like 0
        #       If i+k >= n, then h_{i+k} doesn't exist. treat like 0
        # But the idea: we compute gradient as the first derivative
            
        grad = np.zeros_like(h)
        # For each position i, the contribution to gradient is the derivative
        # of that term. If we consider h_i at position i:
        # We sum h_i * (1 - h_{i+k}) over all i. So d/dh_i = (1 - h_{i+k}) 
        # But that depends on the value of k_max.
        # The derivative at j of a term h_i (1-h_{i+k}) is: 
        # If j = i+k, then derivative = -h_i
        # If j = i, then derivative = 1-h_{i+k}  
        # We can compute different sign cases compactly:
        for i in range(n_points):
            # d/dh[i] = 1 - h[(i + k_max) % n] if i+k_max is valid, i.k. keep it in correct range
            # For a lag k_max: h' = (1 - h_{i+k_max})
            # But since we are in the derivatives, what we're maximizing by moving h towards a
            # direction is gradient f = - h[i] (1 - h[(i+k)%n]) - etc.  
            # That is, the coefficient of h[i] in the derivative of sum over all valid i,k
            # Actually, the method is the more direct approach:
            # Compute the terms that show up in derivative like -h[i+k] for some k. But
            # We're differentiating sum over i of h[i] * (1 - h[i+k])
            # Gradient of term w.r.t. h_j is:
            #     if j=i, then derivative = (1 - h[i+k]),  i.e. we set to 1 - h[(i+k)%n]
            #     if j=i+k, then derivative = -h[i] 
            #     if j=/=i and j=/=i+k, then 0
            # Let me compute direction directly from derivative of overlap:
            # Just compute derivative at position i of the overlap as derivative of sum_i h[i] * (1 - h[i+k_max])
            # This is: d / h[i] = 1 - h[(i+k_max)%n] when valid. 
            # We compute gradient as: 1 - h[(i+k_max)%n], except when k_max = 0 and i=0 (or more accurately, for lag = 0, all terms  
            if k_max >= 0:
                if i + k_max < n_points:
                    # There is valid h_{i+k_max}
                    grad[i] = 1 - h[i + k_max]
                else:
                    grad[i] = 0  # added for safety
            else:
                # k_max < 0
                if i + k_max >= 0:
                    grad[i] = 1 - h[i + k_max]
                else:
                    grad[i] = 0  # added for safety
        # Solving linear subproblem: minimize <grad, d> subject to sum(d) = n_points/2, 0 <= d <= 1
        # So solve min_i grad[i]*d[i] s.t. \sum d[i] = n_points/2, 0<= d[i] <= 1.  
        # It's known that optimal direction is 0 or 1 because it's a linear cost over simplex
        # The min cost direction is d such that d[i] = 0 when grad[i] > 0, 1 when grad[i] < 0
        # For equal cost we choose one that is more feasible.
        # The sign can be problematic if it's at boundary, but let's be careful:
        d = np.zeros_like(h)
        # Sort indices by |grad| to find ones with most impact, so "reverse.Gradient"
        sorted_indices = np.argsort(-np.abs(grad))  # descending
        # Set d[i] = 1 if grad[i] < 0
        # Set d[i] = 0 if grad[i] > 0  
        # Then spread remaining fraction as required
        d = (grad < 0).astype(float)
        sum_d = np.sum(d)
        if abs(sum_d - n_points / 2) > 1e-8:
            # Need to adjust by setting some to 0 or 1 according to sign of grad
            # We set all where grad is negative to 1, all positive to 0
            # And then we problably will not get exact sum
            # We should adjust by setting the right number of 
            # the smallest (in absolute value) ones to 0 or 1 to be 1/2
            # But here we just choose by sign
            if np.sum(d) > n_points/2:
                # More than required. Set the largest entries (with |grad| small) to 0
                # Let's select first (sum_d - n_points/2) from sorted list
                indices_set_to_zero = sorted_indices[:int(sum_d - n_points/2)]
                d[indices_set_to_zero] = 0
            else:
                # Less than, set ones with |grad| smallest to 1  
                indices_set_to_one = sorted_indices[:int(n_points/2 - sum_d)]
                d[indices_set_to_one] = 1
        # Ensure it's exactly the required sum
        d = project_to_feasible(d, n_points)
        
        # Update solution
        h = (1 - eta) * h + eta * d
        h = project_to_feasible(h, n_points)
        
        c5 = evaluate_h(h)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
        # Check for convergence
        if iteration > 0 and abs(c5 - evaluate_h(h)) < tol:
            break
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    global initial_h_values
    np.random.seed(seed)
    # Use the provided initial construction
    init_h = np.array(initial_h_values)
    n_points = len(init_h)
    h_opt, c5_bound = solve_erdos_problem(n_points, init_h, budget_s)
    return (h_opt.tolist(), float(c5_bound), int(n_points))
