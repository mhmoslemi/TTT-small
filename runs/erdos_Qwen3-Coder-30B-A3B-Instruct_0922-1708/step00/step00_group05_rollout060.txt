import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given step function h."""
    dx = 2.0 / len(h)
    # Compute overlap integral for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_h(h, n_points):
    """Ensure h stays feasible: [0,1] and sum==n/2"""
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Adjust sum to exactly n/2
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2.0) > 1e-10:
        adjustment = (n_points / 2.0 - current_sum) / n_points
        h += adjustment
        # Re-clip to ensure bounds
        h = np.clip(h, 0, 1)
    return h
