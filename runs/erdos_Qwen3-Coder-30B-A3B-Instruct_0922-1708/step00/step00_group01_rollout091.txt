import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    """Ensure h is in [0,1] and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h > 0:
        h = h * (n_points / 2) / sum_h
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_inject = 20
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = project_h(h_start, n_points)
    else:
        h_start = np.random.rand(n_points)
        h_start = project_h(h_start, n_points)
    
    particles = []
    velocities = []
    for _ in range(swarm_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.6:
            h = h_start.copy() + np.random.normal(0, 0.05, n_points)
        else:
            h = np.random.rand(n_points)
        h = project_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
    
    # Personal best particles
    p_best = [p.copy() for p in particles]
    p_best_scores = [evaluate_h(p, dx) for p in particles]
    
    # Global best
    g_best_idx = np.argmin(p_best_scores)
    g_best = p_best[g_best_idx].copy()
    g_best_score = p_best_scores[g_best_idx]
    
    # PSO iteration
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            score = evaluate_h(particles[i], dx)
            
            # Update personal best
            if score < p_best_scores[i]:
                p_best_scores[i] = score
                p_best[i] = particles[i].copy()
                if score < g_best_score:
                    g_best_score = score
                    g_best = particles[i].copy()
                    last_improvement = iteration
        
        # Early stopping if no improvement
        if iteration - last_improvement > stagnation_limit:
            # Inject diversity
            for _ in range(swarm_size * diversity_inject // 100):
                idx = np.random.randint(swarm_size)
                particles[idx] = np.random.rand(n_points)
                particles[idx] = project_h(particles[idx], n_points)
                p_best[idx] = particles[idx].copy()
                p_best_scores[idx] = evaluate_h(particles[idx], dx)
            last_improvement = iteration
        
        iteration += 1
    
    return (g_best, g_best_score, n_points)
