import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum and multiply by dx
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, population_size, initial_h):
    """Initialize a population of step functions."""
    population = []
    # Add the initial solution
    population.append(initial_h.copy())
    # Add random solutions
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        # Normalize sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select parents using tournament selection."""
    selected = []
    indices = list(range(len(population)))
    np.random.shuffle(indices)
    for i in range(0, len(indices), tournament_size):
        group_indices = indices[i:i+tournament_size]
        # Sort group by fitness (ascending)
        group_indices.sort(key=lambda idx: fitnesses[idx])
        # Select top 50%
        selected.extend(group_indices[:tournament_size//2])
    return [population[i] for i in selected]

def crossover(parent1, parent2):
    """Perform crossover of two parents."""
    n_points = len(parent1)
    # Random cut point
    cut = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:cut], parent2[cut:]])
    child2 = np.concatenate([parent2[:cut], parent1[cut:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, delta=0.1):
    """Mutate a step function."""
    if np.random.rand() > mutation_rate:
        return h
    h_copy = h.copy()
    # Choose points to mutate
    n_points = len(h_copy)
    n_mutate = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    for i in indices:
        h_copy[i] += np.random.uniform(-delta, delta)
    # Clip to [0, 1]
    h_copy = np.clip(h_copy, 0, 1)
    # Rescale to satisfy sum constraint
    target_sum = n_points / 2
    current_sum = np.sum(h_copy)
    if abs(current_sum - target_sum) > 1e-6:
        h_copy *= target_sum / current_sum
    return h_copy

def ga_search(n_points, initial_h, budget_s):
    """Perform genetic algorithm search."""
    start_time = time.time()
    population_size = 100
    max_generations = 300
    population = initialize_population(n_points, population_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Check for no improvement
        if abs(best_c5 - last_best_c5) < 1e-6:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
            last_best_c5 = best_c5
            
        if no_improvement_count >= 50:
            break
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover
        new_population = [best_h]  # Keep best
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Keep top population_size
        fitnesses = [evaluate_erdos_solution(h) for h in new_population]
        sorted_indices = np.argsort(fitnesses)
        population = [new_population[i] for i in sorted_indices[:population_size]]
    
    return best_h, best_c5

        initial_h = np.clip(initial_h, 0, 1)
    
    # Run GA
    best_h, best_c5 = ga_search(len(initial_h), initial_h, budget_s)
    
    # Final evaluation to make sure
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
