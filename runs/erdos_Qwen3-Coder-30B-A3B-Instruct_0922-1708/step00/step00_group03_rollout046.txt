import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation for all lags
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value is the C5 bound
    c5 = np.max(corr) * dx
    return c5

def initialize_population(n_points, initial_h_values=None, pop_size=100):
    """Initialize a population of step functions."""
    population = []
    # Add the initial solution if available
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        population.append(h)
    else:
        # Generate a random solution that satisfies the constraint
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill the rest with random solutions
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    return population

def mutation(h, rate=0.2, step_size=0.1):
    """Apply mutation to a solution."""
    n = len(h)
    # Determine which points to mutate
    num_mutations = int(n * rate)
    indices = np.random.choice(n, num_mutations, replace=False)
    
    # Apply mutations
    for idx in indices:
        h[idx] += np.random.uniform(-step_size, step_size) * (2.0 / n)
    
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    
    # Rescale so sum is correct
    h *= (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2, prob_swap=0.5):
    """Perform crossover between two parents."""
    n = len(parent1)
    if np.random.rand() > prob_swap:
        return parent1.copy()
    
    # Create a crossover point
    point = np.random.randint(1, n)
    child = np.concatenate([parent1[:point], parent2[point:]])
    
    # Rescale to satisfy the sum constraint
    child *= (n_points / 2) / np.sum(child)
    return child

def tournament_selection(population, fitnesses, k=20):
    """Select a parent via tournament selection."""
    selected = []
    for _ in range(len(population)):
        idxs = np.random.choice(len(population), k, replace=False)
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[best_idx].copy())
    return selected

def ga_optimize_c5(n_points, initial_h_values=None, pop_size=100, max_generations=300, budget_s=600):
    """Use genetic algorithm to optimize C5 bound."""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, initial_h_values, pop_size)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Evaluate fitness (C5) for all individuals
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Track best solution
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 50:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = [best_h]  # Keep the best
        while len(new_population) < pop_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            child = crossover(parent1, parent2)
            child = mutation(child)
            new_population.append(child)
            
        population = new_population[:pop_size]
        
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Run the optimization."""
    np.random.seed(seed)
    
    # Use a reasonable number of points (less than 1000)
    n_points = 70
    
    # Start with initial_h_values if available
    h, c5 = ga_optimize_c5(n_points, initial_h_values, budget_s=budget_s)
    
    return h, c5, n_points
