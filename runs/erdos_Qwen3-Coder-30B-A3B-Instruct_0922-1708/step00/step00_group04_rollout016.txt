import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(2*x - 1))
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # This mimics the evaluate_erdos_solution behavior for consistency
        h_vec = np.array(h_vec)
        # Extended array for full convolution
        h_extended = np.concatenate([h_vec, np.zeros(n_points)])
        h_neg_extended = np.concatenate([1 - h_vec, np.zeros(n_points)])
        # Compute correlations for all shifts k from -n+1 to n-1
        corr = correlate(h_extended, h_neg_extended, mode='full')
        # The actual overlaps are for k = 0 to 2*n-2, corresponding to shift indices n-1 to 2*n-2
        overlaps = corr[n_points - 1:]
        return np.max(overlaps * dx)
    
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def constraint_bounds(h_vec):
        return np.concatenate([h_vec, 1 - h_vec])  # Ensuring all h[i] in [0,1]
    
    # Set up bounding box constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization settings
    max_iter = 1000
    learning_rate = 0.1
    patience = 100
    best_C5 = compute_C5(h)
    iterations_without_improvement = 0
    last_h = h.copy()
    last_C5 = best_C5
    
    # Add some initial variations for exploration
    h_start = h.copy()
    h_best = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Identify the worst-case shift
        # This is a greedy heuristic: slightly cheaper than full evaluation
        h_current = h.copy()
        h_extended = np.concatenate([h_current, np.zeros(n_points)])
        h_neg_extended = np.concatenate([1 - h_current, np.zeros(n_points)])
        corr = correlate(h_extended, h_neg_extended, mode='full')
        overlaps = corr[n_points - 1:] * dx
        k_star = np.argmax(overlaps)
        max_overlap = overlaps[k_star]
        
        # Generate candidate updates around current h
        h_new = h_current.copy()
        
        # Simple noise-based perturbation
        noise = np.random.normal(0, 0.01, n_points)
        h_new += noise
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if this improved the configuration
        C5_new = compute_C5(h_new)
        if C5_new < best_C5:
            h_best = h_new.copy()
            best_C5 = C5_new
            iterations_without_improvement = 0
        else:
            iterations_without_improvement += 1
            
        h = h_new.copy()
        
        if iterations_without_improvement > patience:
            # Perturb slightly if no improvement
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            iterations_without_improvement = 0
            
        # Early termination if target met
        if best_C5 < 0.3807:
            break
            
    return (list(h_best), round(best_C5, 6), n_points)
