import numpy as np
import time
from scipy.signal import correlate
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, else create a symmetric plateau
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create symmetric step function with plateau in middle
        n_points = 47
        h = np.ones(n_points)
        mid = n_points // 2
        # Ensure integral = 1
        h[mid-5:mid+5] = 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # Hyperparameters
    eta = 0.1
    decay_every = 1000
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations
        corr = correlate(h, 1 - h, mode='full')
        # The full correlation has length 2*n-1, we care about lags from -n+1 to n-1
        # The positive lags are from index n-1 to 2*n-2, negative from 0 to n-2
        # But we only care about positive lags [0, n-1] for the overlap integrals
        # Which correspond to indices n-1 to 2*n-2 in full array.
        # The overlap integral for lag k corresponds to index n-1+k  
        max_idx = np.argmax(corr[n_points-1:])
        lag = max_idx - (n_points - 1)
        max_corr_val = corr[max_idx]
        
        # Compute analytic gradient for the dominant lag
        # Gradient is d/dh_i of integral h(x)(1-h(x+k)) dx = 1-h(x+k) when x=i, 
        # or equivalently, -h(i+k) for the term where h(i) is being differentiated.
        # So the gradient with respect to h[i] is:
        # For i+k in [0,n-1]: grad[i] = -h[i+k]
        # But we are minimizing -max_corr_val so we need to subtract the gradient
        # and then take the negative for promotion.

        # Actually, to make gradient descent minimize this, we need to compute:
        # Loss = - maximum overlap, so dLoss/dh[i] = - d(max_overlap)/dh[i].
        # But d(max_overlap)/dh[i] = -h[i+k] for the term with maximum overlap
        # So the gradient used should be:
        # grad[i] = h[i+k] (so that we're going in the right direction to minimize the term)
        # But we worry about the constraint imposed. We use one-sided gradients based on the fact
        # that corr[k] = sum h[i] (1 - h[i+k]), so d corr[k]/dh[j] = -h[j+k] if j+k=k so j=0,
        # or -h[k] if j=0 and we're looking at k, etc.
        # Actually, just compute it manually:
        # We'll compute gradient at each point to reduce cost:
        grad = np.zeros_like(h)
        # For dominent lag k, we compute d/dh[i] of h[i](1-h[i+k]), which is:
        # 1 - h[i+k] if i+k is within bounds and i=0,...n-1
        # But that's wrong, the actual derivative is:
        # d/dh[i] integrate h(x)(1-h(x+k)) dx = (1-h(x+k)) | x=i = (1-h[i+k]) if i+k < n
        # But when computing gradient for h[i], only depends on the term with h[i]
        # The term h[i](1-h[i+k]) contributes a gradient of (1-h[i+k]) wrt h[i]
        # So we update the gradient correspondingly.
        # More precise:
        # The overlaps for different lags k are:
        # O_k = integral from 0 to 2 of h(x)(1-h(x+k))dx
        # dO_k/dh[i] = (1-h[i+k]) if i+k is in [0,2] (which means i in [0, 2-k] and i+k in [0,2])
        # Actually, focus on (horizontally) coding correct method.
        # Step 1: find the cov for dominant lag
        # Step 2: tangent to h[i] that contributes to loss is (1 - h[i+k]) where x+k=i, so x=i-k
        # So when computing derivative with respect to h[i], we have:
        # the integral is sum over all x of h(x)(1-h(x+k))
        # derivative w.r.t h[i] = (1-h[i+k]) if x+k=i (i.e., x=i-k), which is just (1-h[i+k]), 
        # only if the lag is valid. This only happens when i+k < n (since valid x in range [0,n-1])
        # We only consider the term h[i](1-h[i+k]) in the integral for this gradient computation
        
        # This is tricky but we can treat the Hilbert space one by one:
        # The loss we're talking about is max_k O_k where O_k = integral h(x)(1-h(x+k))
        # Gradient is with respect to h[i]: sum_k dO_k/dh[i] * sign of gradient for that k
        # But since we're trying to minimize the max value, let's consider that the dominant `k` is causing loss
        # Updated formula for the gradient of the overlap (for dominant lag)
        # Gradient is: idx 0, ..., n-1 where h[i] enters as h[i](1-h[i+k]) term
        # Derivative towards h[i] = (1-h[i+k]) 
        # But the true update should be toward reducing dominant overlap term
        # So gradient = -(1-h[i+k]) for each valid i and k
        # So we compute gradient vector for the given lag k
        # But also remember we don't want to touch i where i+k goes out of bounds.
        # Correct: derivative of the integral O_k w.r.t h[i] = (1-h[i+k]) for valid i i.e. i+k < n
        target_lag = lag
        grad = np.zeros_like(h)
        # Only consider valid i where i+k < n (so i < n-k): 
        k = target_lag
        # i can range from max(0, -k) to min(n-1, n-1-k) = n-1-k or n-1 (but the -k shifts)
        # i+k must be in [0,n-1]
        # i in [0,n-1]
        # i+k in [0,n-1]  =>  i in [-k,n-1-k] and [0,n-1]
        # i in max(0,-k) to min(n-1,n-1-k) = n-1 and n-1-k, 
        # so it's from max(0,-k) to min(n-1,n-1-k) = max(0,-k) to n-1-k
        # let's just be explicit:
        for i in range(n_points):
            j = i + k
            if 0 <= j < n_points:
                grad[i] += -(1 - h[j])
        
        # SGD step with leapfrog improvement
        h += eta * grad 
        # Project h into [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to make integral = 1:
        h = h * (n_points/2) / np.sum(h)
        
        # Check if we have improved solution
        c5 = evaluate_h(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Anneal rate
        if i > 0 and i % decay_every == 0:
            eta *= 0.5
            
    # Final validation
    final_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))

def evaluate_h(h, dx):
    # Compute all correlations and return max
    corr = correlate(h, 1-h, mode="full")
    # Max is over lags from j=0 to j=2*n-2 (how numpy's full works)
    # We're interested in the valid positive lags, which in the original define
    # have index n-1 to 2*n-2  
    # The overlap integral for lag k corresponds to index n-1+k where -n+1 <= k <= n-1  
    # Positive lags are: k = 0 to k = n-1 
    # Indices of positive lags in full array: from n-1 to 2*n-2
    return np.max(corr[len(h)-1:])
