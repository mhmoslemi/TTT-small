import numpy as np
import time
from scipy import signal

def evaluate_h(h_values):
    """Computes C5 for given step function h."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute overlap integral using correlation
    conv = signal.correlate(h_values, 1 - h_values, mode='full')
    # Maximize over all possible shifts
    max_overlap = np.max(conv) * dx
    return max_overlap

    """Initialize population of step functions."""
    population = []
    if initial_h is not None:
        # Use initial construction as first individual
        population.append(initial_h.copy())
        pop_size -= 1
    
    # Generate diverse random individuals
    for _ in range(pop_size):
        # Start with uniform distribution around 0.5
        h = np.full(n_points, 0.5)
        # Randomly perturb
        for i in range(n_points):
            if np.random.rand() < 0.5:
                h[i] += (np.random.rand() - 0.5) * 0.2
        # Clip and renormalize
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        scale = target_sum / current_sum
        h = h * scale
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    n = len(population)
    best_idx = np.random.randint(n)
    best_fitness = fitnesses[best_idx]
    for _ in range(k - 1):
        idx = np.random.randint(n)
        if fitnesses[idx] < best_fitness:
            best_idx = idx
            best_fitness = fitnesses[idx]
    return population[best_idx]

def crossover(parent1, parent2):
    """Single-point crossover."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    # Enforce sum constraint
    def renormalize(h):
        current_sum = np.sum(h)
        target_sum = len(h) / 2.0
        scale = target_sum / current_sum
        h = h * scale
        return np.clip(h, 0, 1)
    child1 = renormalize(child1)
    child2 = renormalize(child2)
    return child1, child2

def mutate(h, rate=0.05):
    """Gaussian mutation."""
    n = len(h)
    for i in range(n):
        if np.random.rand() < rate:
            h[i] += np.random.normal(0, 0.05)

    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = len(h) / 2.0
    scale = target_sum / current_sum
    h = h * scale
    return h
