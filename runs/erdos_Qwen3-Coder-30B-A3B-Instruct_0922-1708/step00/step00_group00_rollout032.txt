import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h0)
    except:
        # Default construction with 45 points
        n_points = 45
        h0 = np.full(n_points, 1.0 / n_points * 2.0)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Objective function to minimize
    def obj(h):
        # This is equivalent to max_k int h(x)(1-h(x+k))dx
        # Using full convolution
        h_full = np.concatenate([h, np.zeros(n_points-1)])
        one_minus_h = 1 - h
        one_minus_h_full = np.concatenate([one_minus_h, np.zeros(n_points-1)])
        corr = np.correlate(h_full, one_minus_h_full, mode="full")
        c5 = np.max(corr[:2*n_points-1]) * dx
        return c5
    
    # Constraints
    def constraint_int(h):
        return np.sum(h) * dx - 1.0
    
    def constraint_bounds(h):
        return np.concatenate([h, 1.0 - h])
    
    # Optimization using scipy minimize
    # A hybrid approach with gradient-based optimization
    cons = [
        {'type': 'eq', 'fun': constraint_int},
    ]
    
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    
    best_h = h0.copy()
    best_c5 = obj(best_h)
    
    # Start with initial
    if obj(h0) < best_c5:
        best_h = h0.copy()
        best_c5 = obj(h0)
    
    # Gradient based local optimization with restarts
    options = {'maxiter': 1000, 'disp': False}
    
    # Try different restart strategies
    for i in range(100):  # limited iterations to avoid timeout
        if time.time() - start_time > budget_s - 1:
            break
            
        # Random restart
        h_start = np.random.rand(n_points) * 0.5
        h_start = h_start / np.sum(h_start) * 2.0 * n_points / 2.0  # scale to correct integral
        h_start = np.clip(h_start, 0, 1)
        
        result = minimize(
            obj, 
            h_start, 
            method='L-BFGS-B', 
            bounds=bounds, 
            constraints=cons,
            options=options,
            tol=1e-8
        )
        
        h_opt = result.x
        c5_opt = obj(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            
    # Final validation
    h_validation = best_h.copy()
    
    # Final evaluation
    final_c5 = obj(h_validation)
    
    # Enforce the constraint exactly
    integral = np.sum(h_validation) * dx
    h_validation = h_validation * (1.0 / integral)
    
    return (h_validation, final_c5, n_points)
