        correlations = np.correlate(h, 1 - h, mode="full")
        k_indices = np.arange(-n_points + 1, n_points)
        worst_k_idx = np.argmax(correlations)
        worst_k = k_indices[worst_k_idx]
        
        # Compute relevant indices for optimization
        min_idx = max(0, -worst_k)
        max_idx = min(n_points, n_points - worst_k)
        relevant_indices = list(range(min_idx, max_idx))
        
        # Compute gradient w.r.t. h for worst overlap
        grad = np.zeros_like(h)

        for j in relevant_indices:
            k_idx = j + worst_k
            if 0 <= k_idx < n_points:
                grad[j] += (1 - h[k_idx])
        
        # Update h with gradient descent
        h_new = h - learning_rate * grad

        h_new = np.clip(h_new, 0, 1)
