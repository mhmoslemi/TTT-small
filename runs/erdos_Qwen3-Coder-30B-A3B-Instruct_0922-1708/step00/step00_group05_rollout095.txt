import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initial h values
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a symmetric initial guess
        h_init = np.ones(n_points)
        h_init[:n_points//2] = 0.5
        h_init[n_points//2:] = 0.5
    
    # Ensure integral constraint
    current_sum = np.sum(h_init)
    h_init = h_init * target_sum / current_sum
    
    # Project h_init to [0,1]
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Use Fourier approach for optimization
    # Parameters
    n_modes = 10  # Number of cosine modes to use
    # Fourier coefficients for symmetric function: a_k for k=0,1,...,n_modes-1
    a_init = np.random.rand(n_modes)  # random start in [0,1]
    a_init[0] = 0.5  # integral constraint
    
    # Optimization setup
    best_c5 = float('inf')
    best_h = None
    iteration = 0
    
    learning_rate = 0.01
    patience = 0
    max_patience = 100
    
    def forward_transform(a_coeffs):
        # Construct h from cosine modes
        x = np.linspace(0, 2, n_points)
        h_vals = np.zeros_like(x)
        h_vals += a_coeffs[0]
        for i in range(1, len(a_coeffs)):
            h_vals += a_coeffs[i] * np.cos(np.pi * i * x / 2.0)
        h_vals = np.clip(h_vals, 0, 1)
        # Ensure correct sum
        current_sum = np.sum(h_vals) * dx
        h_vals = h_vals * target_sum / current_sum
        return h_vals
    
    def check_constraints(h_vec):
        integral = np.sum(h_vec) * dx
        return abs(integral - 1.0) < 1e-10
    
    def compute_overlap(h_vec):
        # Use scipy correlate for full mode
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Get the maximum overlap over all shifts
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Gradient descent loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Forward transform for current a_coeffs
        h_current = forward_transform(a_init)
        
        # Compute current C5
        c5_current = compute_overlap(h_current)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h_current.copy()
            patience = 0
        else:
            patience += 1
        
        if patience > max_patience:
            break
            
        # Compute gradients using finite differences
        eps = 1e-5
        grad = np.zeros_like(a_init)
        for i in range(len(a_init)):
            a_plus = a_init.copy()
            a_plus[i] += eps
            h_plus = forward_transform(a_plus)
            c5_plus = compute_overlap(h_plus)
            
            a_minus = a_init.copy()
            a_minus[i] -= eps
            h_minus = forward_transform(a_minus)
            c5_minus = compute_overlap(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update step
        a_init -= learning_rate * grad
        a_init = np.clip(a_init, 0, 1)
        a_init[0] = 0.5  # integral constraint
        
        # Adapt learning rate
        if patience > 10:
            learning_rate *= 0.95
    
    if best_h is None:
        # fallback
        h_final = h_init
        c5_final = compute_overlap(h_final)
    else:
        h_final = best_h
        c5_final = best_c5
    
    return (h_final.tolist(), c5_final, n_points)
