import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, else create uniform
    if 'initial_h_values' in globals():

        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    max_iter = 800
    lambda_reg = 1e-4
    line_search_iter = 10
    time_start = time.time()
    
    def compute_C5(h):
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Take only valid overlaps (excluding padding)
        valid_corr = corr[len(h)-1:]  
        return np.max(valid_corr) * dx
    
    def grad_C5(h):
        # Compute maximum overlap index k
        corr = np.correlate(h, 1 - h, mode='full')
        valid_corr = corr[len(h)-1:]
        k_max = np.argmax(valid_corr)
        
        # Gradient computation
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad[i] = 1 - 2 * h[j]  # Gradient of h[i] * (1 - h[j])
            # Add regularization term
            grad[i] += 2 * lambda_reg * h[i]
        return grad
    
    def line_search(h, grad, alpha_max=1.0):
        best_h = h.copy()
        best_C5 = compute_C5(h)
        for i in range(line_search_iter):
            alpha = alpha_max * (0.5 ** i)
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            C5_new = compute_C5(h_new)
            if C5_new < best_C5:
                best_C5 = C5_new
                best_h = h_new.copy()
        return best_h, best_C5
    
    # Main optimization loop
    for iteration in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
            
        # Compute objective and gradient
        C5 = compute_C5(h)
        grad = grad_C5(h)
        
        # Line search
        h, _ = line_search(h, grad)
    
    # Final refinement with shift search
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Try some shifts to refine
    for shift in range(-10, 11):
        if time.time() - time_start > budget_s - 1:
            break
        h_shifted = np.roll(h, shift)
        h_shifted = h_shifted * (n_points / 2) / np.sum(h_shifted)
        h_shifted = np.clip(h_shifted, 0, 1)
        C5_shifted = compute_C5(h_shifted)
        if C5_shifted < best_C5:
            best_C5 = C5_shifted
            best_h = h_shifted
    
    # Final check and normalization
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
