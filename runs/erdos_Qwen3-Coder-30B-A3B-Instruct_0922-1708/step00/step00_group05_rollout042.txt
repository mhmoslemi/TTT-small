import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given step function h."""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all lags
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def normalize_h(h, n_points):
    """Normalize h to have sum = n_points / 2."""
    h_sum = np.sum(h)
    if h_sum == 0:
        return np.ones_like(h) * 0.5
    target_sum = n_points / 2
    scale = target_sum / h_sum
    h = h * scale
    return np.clip(h, 0, 1)

def project_to_feasible(h, n_points):
    """Project h to be feasible (clipping and normalizing)."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction: simple step function
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    h0 = project_to_feasible(h0, n_points)
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for _ in range(n_particles - 1):
        h_perturb = h0 + np.random.normal(0, 0.05, n_points)
        h_perturb = project_to_feasible(h_perturb, n_points)
        particles.append(h_perturb.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.7
    c1, c2 = 2.0, 2.0
    max_iter = int(budget_s / 0.05)  # Approximate
    
    # Track best solutions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_erdos_solution(p) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    start_time = time.time()
    
    # PSO iterations
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v_new = (w * velocities[i] +
                     c1 * r1 * (personal_best[i] - particles[i]) +
                     c2 * r2 * (global_best - particles[i]))
            
            # Apply velocity limits
            v_new = np.clip(v_new, -0.1, 0.1)
            velocities[i] = v_new
            
            # Update position
            h_new = particles[i] + v_new
            h_new = project_to_feasible(h_new, n_points)
            
            particles[i] = h_new
            
            # Evaluate new function
            new_score = evaluate_erdos_solution(h_new)
            
            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_best[i] = h_new.copy()
                personal_best_scores[i] = new_score
                
                # Update global best
                if new_score < global_best_score:
                    global_best = h_new.copy()
                    global_best_score = new_score
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Tournament selection and diversity maintenance
        if iteration % 20 == 0 and iteration > 0:
            # Sort by fitness
            sorted_indices = np.argsort(personal_best_scores)
            keep_indices = sorted_indices[:n_particles // 2]
            
            # Replace half the worst
            for i in range(n_particles // 2, n_particles):
                # Perturb a good particle
                parent_idx = np.random.choice(keep_indices)
                h_perturb = particles[parent_idx] + np.random.normal(0, 0.02, n_points)
                h_perturb = project_to_feasible(h_perturb, n_points)
                particles[i] = h_perturb
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                personal_best_scores[i] = evaluate_erdos_solution(h_perturb)
                personal_best[i] = h_perturb.copy()
    
    # Final evaluation
    final_score = evaluate_erdos_solution(global_best)
    
    return (global_best.tolist(), final_score, n_points)
