    alpha = 0.01
    beta = 0.9

    patience_counter = 0

        # Compute cross-correlation: h * (1 - h)
        corr = signal.correlate(h, 1 - h, mode='full')

        # The full correlation has length 2*n - 1

        shifts = k_indices * dx / n_points  # Convert to actual shifts in [0,2]

        k_max_idx = np.argmax(valid_corr)

        overlap = valid_corr[k_max_idx] * dx

            patience_counter = 0
