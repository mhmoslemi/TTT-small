    lambda_reg = 1e-4
    step_size = 0.01
    
    def eval_C5(h):
        # Compute all correlations
        h_extended = np.concatenate([h, h])
        overlaps = np.correlate(h_extended, 1 - h, mode='full')
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
        return h
    
    best_h = h.copy()
    best_C5 = eval_C5(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        
        h = project_h(h)
        
        # Estimate gradient using finite differences
        eps = 1e-4
        grad = np.zeros_like(h)
        h_plus = h.copy()
        h_minus = h.copy()
        
        for i in range(n_points):
            h_plus[i] = min(1, h[i] + eps)
            h_minus[i] = max(0, h[i] - eps)
            
            # For this specific case, we'll estimate gradient by sampling overlaps
            C5_plus = eval_C5(h_plus)
            C5_minus = eval_C5(h_minus)
            
            grad[i] = (C5_plus - C5_minus) / (2 * eps)
            
            h_plus[i] = h[i]
            h_minus[i] = h[i]
        
        # Apply gradient step
        h = h - step_size * (grad + lambda_reg * h)
        
        # Regular projection
        h = project_h(h)

        new_C5 = eval_C5(h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
    
    # Local search for shifts
    best_shifted = best_h.copy()
    best_C5_shifted = best_C5
