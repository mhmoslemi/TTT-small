import numpy as np
import time
from scipy import signal

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for a given h."""
    # Compute the correlation of h with (1 - h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # The overlap integral for each shift k is corr[k + len(h) - 1] * dx
    # So we need to take the maximum over all these values
    overlaps = corr * dx
    return np.max(overlaps)

def enforce_constraints(h, n_points):
    """Ensure h is within [0,1] and sum equals n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-6:
        # Normalize so sum is exactly target_sum
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with some diversity."""
    population = []
    # Add initial solution
    if initial_h is not None:
        population.append(enforce_constraints(initial_h.copy(), n_points))
    
    # Add random solutions to fill population
    remaining = pop_size - len(population)
    for _ in range(remaining):
        # Random start: uniform then perturb
        h = np.ones(n_points) * 0.5
        h += np.random.normal(0, 0.1, n_points)
        h = enforce_constraints(h, n_points)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    best_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[best_idx]

def crossover(parent1, parent2):
    """Single-point crossover."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate the individual with small random perturbations."""
    n_points = len(h)
    num_mutations = max(1, int(n_points * mutation_rate))
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 45  # Using the given number of samples
    pop_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    elitism_rate = 0.2
    
    # Initialize
    if 'initial_h_values' in globals() and initial_h_values is not None:
        initial_h = np.array(initial_h_values)
    else:
        initial_h = None
        
    population = initialize_population(n_points, pop_size, initial_h)
    dx = 2.0 / n_points
    
    # Track the best solution
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:  # Leave a buffer
        # Evaluate fitness
        fitnesses = []
        for h in population:
            fit = evaluate_overlap(h, dx)
            fitnesses.append(fit)
        
        # Find best in this generation
        min_fitness_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[min_fitness_idx]
        current_best_h = population[min_fitness_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count >= max_no_improvement:
            break
        
        # Tournament selection
        new_population = []
        
        # Elitism: keep top 20%
        num_elite = int(elitism_rate * pop_size)
        elite_indices = np.argsort(fitnesses)[:num_elite]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < crossover_prob:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutate
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            # Enforce constraints
            child1 = enforce_constraints(child1, n_points)
            child2 = enforce_constraints(child2, n_points)
            
            # Add to new population
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
    
    # Final evaluation
    best_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)
