import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    # For each shift k, compute the overlap integral
    max_val = 0
    for k in range(n):
        if k == 0:
            # This is just sum of h(x)(1-h(x)) integrated over [0,2]
            prod = h * (1 - h)
            val = np.sum(prod) * dx
        else:
            # This is convolution-like using circular shift
            shifted = np.roll(h, k)
            prod = h * (1 - shifted)
            val = np.sum(prod) * dx
        if val > max_val:
            max_val = val
    return max_val

    # Ensure that it satisfies the integral constraint (sum should be n/2)
    h = h * (n/2) / np.sum(h)
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    last_improvement = 0
    iteration = 0
