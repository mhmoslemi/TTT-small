import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    learning_rate = 0.1
    patience = 50
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h.copy()
        if abs(np.sum(h) * dx - 1.0) > 1e-10:
            h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5, dtype=float)
        
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based optimization with EM-like updates
    prev_c5 = float('inf')
    no_improvement_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps[k] = overlap * dx
        
        # Update h using gradient descent
        new_h = h.copy()
        for i in range(n_points):
            grad = 0.0
            for k in range(n_points):
                # Gradient of overlap_k w.r.t. h[i]
                j = (i - k) % n_points
                grad += (1 - h[j])
            new_h[i] -= learning_rate * grad * dx
            
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        h = new_h
        c5 = max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if abs(c5 - prev_c5) < 1e-6:
            no_improvement_count += 1
            if no_improvement_count > patience:
                break
        else:
            no_improvement_count = 0
            
        prev_c5 = c5
        
        # Adjust learning rate
        if iteration % 10 == 0 and iteration > 0:
            learning_rate *= 0.9
            
    if best_h is None:
        best_h = h
        
    return (best_h, best_c5, n_points)
