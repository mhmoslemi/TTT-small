def evaluate(h, dx):
    """Compute C5 = max_k integral h(x)(1-h(x+k)) dx"""
    corr = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

    h = np.clip(h, 0, 1)

    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
