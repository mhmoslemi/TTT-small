    iter_count = 0
    no_improve = 0
    max_no_improve = 100

        iter_count += 1
        # Solve convex relaxation
        try:
            # Formulate relaxation using helper function to map product h*(1-h) 
            h_approx = h.copy()

            n = len(h_approx)
            dx = 2.0 / n

            conv = correlate(h_approx, 1 - h_approx, mode='full')

            c5 = np.max(conv) * dx
            
            # Check if this is an improvement
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_approx.copy()
