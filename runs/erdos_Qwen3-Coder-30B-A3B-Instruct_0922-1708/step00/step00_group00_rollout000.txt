import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize configuration
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Get initial solution or create a suitable one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
    else:
        h = np.ones(n_points) * 0.5 * dx

    # Precompute indices for efficiency
    indices = np.arange(n_points)
        
    def compute_overlap(h_vec):
        # Compute correlation for all shifts in one go using conv
        # This computes overlap_k for all valid k
        corr_result = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Take the indices corresponding to valid overlap values
        # overlap_k occurs for k in [-n+1, n-1], so shift index by n-1
        # corr_result has indices from 0 to 2*n-2
        # First n-1 elements are for negative k, last n-1 elements are for positive k
        # Middle element is for k=0
        overlaps = corr_result[n_points-1:]  # Only positive k shifts
        return overlaps * dx

    def smooth_max(overlaps, tau=0.1):
        # a smooth approximation to max
        return np.log(np.sum(np.exp(tau * overlaps))) / tau

    def gradient(h_vec):
        # Compute all overlaps
        overlaps = compute_overlap(h_vec)
        # Softmax weights for the gradient computation
        weights = np.exp(0.1 * overlaps)
        weights /= np.sum(weights)
        # Compute gradient of smooth max 
        # Gradient is weighted sum of gradients d/dh of overlap terms
        # We'll need convolution for each d/dh term
        grad = np.zeros_like(h_vec)
        # Something more efficient:
        # Just compute for forward shifts with correlation
        # d/dh[i] overlap_k = (1-h[i+k]) if k>=0, -h[i-k] if k<0
        # Average of all such gradients weighted by softmax weights
        # Or make loop over k:
        # Let's compute this efficiently:
        # For every sample and every shift, we contribute to its gradient
        for k in range(-n_points+1, n_points):
            if k >= 0:
                # Let's calculate overlap_k efficiently
                shifted_h = np.roll(h_vec, -k)
                shifted_h[:k] = 0
                # Overlap_k = sum( h[j] * (1 - h[j+k]) ) 
                # So gradient wrt h[i] = (1-h[i+k]) when k >= 0
                # We want term in gradient for index i:
                if i+k < n_points:
                    grad[i] += weights[k] * (1 - h_vec[i+k]) 
        return grad

    # Let's implement gradient correctly using convolution
    def compute_grad_smoothmax(h_vec, tau=0.1):
        # Evaluate all overlap values
        overlaps = compute_overlap(h_vec)
        # Softmax weights for smooth max gradient
        exp_overlaps = np.exp(tau * overlaps)
        Z = np.sum(exp_overlaps)
        weights = exp_overlaps / Z
        # Now compute gradient of smooth max
        # d/dh[i] smoothmax = sum_k weights_k * d/dh[i] overlap_k
        # For every shift k and contribution to overlap_i:
        # If overlap_k = sum_j h[j] * (1 - h[j+k]) 
        # Then d/dh[i] overlap_k = (1 - h[i+k]) if k+i < n_points and i >=0
        # and contribution is due to the term h[i] * (1 - h[i+k])
        # This is a complex derivative to compute exactly.
        # Approximate with direct finite differences or symbolic rules.
        # Instead, more efficiently:
        # ∇C₅ ≈ ∑_k weights_k * ∇_h overlap_k
        # We'll simplify and calculate convolution properly:
        # ∇_h overlap_k = h * (1 - h_{-k}) - h_k * (1 - h)  
        # Actually:
        # overlap_k = ∑_i h[i] * (1 - h[i+k]) 
        # ∂overlap_k/∂h[i] = (1 - h[i+k])
        # Take derivatives:
        # ∇(smoothmax_{tau}(overlaps)) = sum_k weights_k * (1 - h[i+k]) for i in k-shift

        # This is better handled directly from corr with weights
        gradient = np.zeros_like(h_vec)
        for k in range(n_points):
            shift = k - (n_points-1)  # k from -44 to 44
            if shift > 0:
                # h[i] contributes (1 - h[i+shift]) if i+shift < n
                gradient[:-shift] += weights[k] * (1 - h_vec[shift:])
            elif shift < 0:
                # h[i-shift] contributes (1 - h[i]) if i-shift >= 0
                gradient[-shift:] += weights[k] * (1 - h_vec[:-shift])
            else:
                # shift == 0
                gradient += weights[k] * (1 - 2 * h_vec)
        
        return gradient

    # Better approach: compute gradient via derivative of correlation 
    def compute_grad(h_vec, tau=0.1):
        overlaps = compute_overlap(h_vec)  # length 2*n-1 ? 
        exps = np.exp(tau * overlaps)
        Z = np.sum(exps)
        # Compute softmax weights
        w = exps / Z
        # For gradient of smoothmax with smoothness parameter tau:
        # d/dh_i smoothmax = (1/Z)* sum_k w_k * d/dh_i overlap_k
        # But d/dh_i overlap_k = (1-h[i+k]) if k >=0 and 0 otherwise, or reverse
        # overlap_k = sum_{j} h[j] * (1 - h[j+k])  where k \in [-n+1, n-1]
        # Then d/dh_i overlap_k = (1-h[i+k]) if i+k < n and i >= 0
        # In other words: d/dh_i ( sum_{j} h[j] * (1 - h[j+k]) ) = (1-h[i+k])  
        # provided i+k < n and i >= 0.
        # So for shift k, the gradient term from d/dh_i is:
        # partial_this_term = 1-h[i+k] for valid k shift
        # But we're computing d/dh_i (sum_k  w_k * overlap_k)
        # The gradient d/dh_i depends on which shifts k involve h[i]
        # For all shifts k where i+k < n and i >= 0
        # Contribution becomes: w_k * (1 - h[i+k])
        # Instead we compute this more directly by considering appropriate corr
        # Let's make a 2D computation:
        n = len(h_vec)
        g = np.zeros_like(h_vec)
        # For each h[i], compute gradient contribution from all related overlaps
        for i in range(n):
            # let k go from -n+1 to n-1
            for shift in range(-n+1, n):
                np_shift = shift + (n-1)  # index in overlap array
                if shift >= 0 and i+shift < n:
                    # h[i] appears in overlap_k (k = shift) as factor h[i] * (1 - h[i+shift])
                    g[i] += w[np_shift] * (1 - h_vec[i+shift])
                elif shift < 0 and i+shift >= 0:
                    # h[i] appears in overlap_k (k = shift) as factor h[i] * (1 - h[i+shift])
                    # Actually when shift < 0, this is h[i] * (1 - h[i-shift])
                    g[i] += w[np_shift] * (1 - h_vec[i-shift])
        return g

    # Optimize with gradient descent and constraints
    best_h = h.copy()
    best_c5 = np.max(compute_overlap(h))
    max_iters = 10000
    cur_time = time.time()
    num_unimproved = 0
    learning_rate = 0.1
    max_lr = 0.5
    min_lr = 0.001
    lr_decay = 0.98
    
    for iter_idx in range(max_iters):
        if time.time() - start_time > budget_s - 1:  # leave 1 sec
            break
            
        # Do gradient descent step
        gradient = compute_grad(h)
        # Check for numerical instability, clip if needed
        gradient = np.clip(gradient, -100, 100)
        h = h - learning_rate * gradient

        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Preserve sum constraint
        h_sum = np.sum(h)
        h = h * (target_sum / h_sum)
        
        # Evaluate objective
        overlaps = compute_overlap(h)
        c5 = np.max(overlaps)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            num_unimproved = 0
        else:
            num_unimproved += 1
            
        # Adaptive learning rate
        if num_unimproved > 10:
            learning_rate *= lr_decay
            if learning_rate < min_lr:
                learning_rate = min_lr
            num_unimproved = 0
        elif num_unimproved == 0:
            learning_rate = min(max_lr, learning_rate * 1.01)  # increase if improving

    # Final check
    overlaps = compute_overlap(best_h)
    c5 = np.max(overlaps)
    return (best_h.tolist(), float(c5), n_points)
