import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except:
        # Default construction
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    patience = 100
    best_c5 = np.inf
    patience_count = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        h_ext = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlaps = correlate(h_ext, h_reflected, mode='full')[:2*n_points-1]
        overlaps = overlaps * dx
        
        # Find maximum overlap and corresponding shift
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            break
            
        # Compute gradient for k_max
        grad = np.zeros(n_points)
        if k_max < n_points:
            # h[i] * (1 - h[i + k_max]) contributes
            for j in range(n_points - k_max):
                grad[j] += (1 - h[j + k_max])
            for j in range(k_max, n_points):
                grad[j] -= h[j - k_max]
        else:
            # h[i] * (1 - h[i - (k_max - n_points)]) contributes
            k_shift = k_max - n_points
            for j in range(n_points - k_shift):
                grad[j] += h[j + k_shift]
            for j in range(k_shift, n_points):
                grad[j] -= (1 - h[j - k_shift])
                
        # Update h
        h_new = h - alpha * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to ensure sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    h_ext = np.concatenate([h, np.zeros(n_points)])
    h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
    overlaps = correlate(h_ext, h_reflected, mode='full')[:2*n_points-1]
    final_c5 = np.max(overlaps) * dx
    
    return (h.tolist(), final_c5, n_points)
