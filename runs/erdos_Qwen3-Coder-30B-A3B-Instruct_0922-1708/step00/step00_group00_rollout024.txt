import numpy as np
import time
from scipy.signal import correlate

def evaluate_solution(h, dx):
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    C5 = np.max(corr) * dx
    return C5

def project_h(h, n_points, dx):
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to ensure sum equals n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5 * dx
    
    h = project_h(h, n_points, dx)
    
    best_h = h.copy()
    best_C5 = evaluate_solution(best_h, dx)
    
    # Optimization parameters
    max_iter = 1000000
    iter_no_improve = 0
    max_no_improve = 1000
    
    # Perturbation parameters
    sigma = 0.1
    decay_rate = 0.99
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate perturbation
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        
        # Project to feasible region
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate new solution
        C5_new = evaluate_solution(h_new, dx)
        
        # Accept improvement
        if C5_new < best_C5:
            best_h = h_new.copy()
            best_C5 = C5_new
            iter_no_improve = 0
        else:
            iter_no_improve += 1
            
        # Update h for next iteration
        h = h_new.copy()
        
        # Decay noise level
        sigma *= decay_rate
        
        # Early stopping if no improvement for a while
        if iter_no_improve > max_no_improve:
            break
    
    return (best_h.tolist(), best_C5, n_points)
