import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import expit

def sigmoid(x):
    return expit(x)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # If initial_h_values available, use it as starting point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * target_sum
    else:
        h0 = np.ones(n_points) * 0.5

    # Basis function setup
    m = int(np.ceil(np.sqrt(n_points) * 2))  # number of basis functions
    basis_centers = np.linspace(0, 2, m)
    basis_scales = 2.0 / m
    basis_functions = lambda x, j: sigmoid((x - basis_centers[j]) / basis_scales)

    def h_from_coeffs(a):
        # compute h(x) value for all x using basis expansion
        x_vals = np.linspace(0, 2, n_points)
        h_vals = a[0] + np.sum([a[j+1] * basis_functions(x_vals, j) for j in range(m)], axis=0)
        return np.clip(h_vals, 0, 1)

    def compute_overlap(h_vals):
        # compute max overlap integral
        h_corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # normalize by dx
        h_corr *= dx
        return np.max(h_corr)

    def obj_and_grad(a):
        # compute objective and gradient (approximate)
        h_vals = h_from_coeffs(a)
        
        # normalize to unit integral
        h_vals_norm = h_vals * target_sum / np.sum(h_vals)
        # enforce bounds
        h_vals_norm = np.clip(h_vals_norm, 0, 1)
        
        # compute overlap
        overlap_val = compute_overlap(h_vals_norm)
        
        # Forward finite difference gradient (simplified)
        eps = 1e-5
        grad = np.zeros_like(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = h_from_coeffs(a_plus)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            h_plus = np.clip(h_plus, 0, 1)
            overlap_plus = compute_overlap(h_plus)
            grad[i] = (overlap_plus - overlap_val) / eps
        
        return overlap_val, grad

    # Initialization
    a_init = np.zeros(m + 1)
    if 'initial_h_values' in globals():
        # fit basis to initial solution
        x_vals = np.linspace(0, 2, n_points)
        y_vals = np.array(initial_h_values) / np.sum(initial_h_values) * target_sum
        # compute coefficients via least-squares
        A = np.column_stack([np.ones(n_points), 
                             np.array([basis_functions(x_vals, j) for j in range(m)]).T])
        if np.linalg.matrix_rank(A) == A.shape[1]:
            a_fit = np.linalg.lstsq(A, y_vals, rcond=None)[0]
            a_init = a_fit
        else:
            # fallback if matrix is singular
            a_init[0] = np.mean(y_vals)
            a_init[1:] = 0

    best_c5 = float('inf')
    best_h = h0.copy()

    # Optimization loop
    current_a = a_init.copy()
    iter_count = 0
    last_improvement = 0
    alpha = 0.01
    max_iter = 500

    while iter_count < max_iter:
        elapsed = time.time() - start_time
        if elapsed >= 0.95 * budget_s:
            break
            
        try:
            # objective and gradient
            c5, grad = obj_and_grad(current_a)
            
            if c5 < best_c5:
                best_c5 = c5
                # Recompute best h
                h_vals = h_from_coeffs(current_a)
                # Normalize to correct sum
                h_vals = h_vals * target_sum / np.sum(h_vals)
                # Keep maximum value 1
                h_vals = np.clip(h_vals, 0, 1)
                best_h = h_vals.copy()
            
            # update
            current_a -= alpha * grad
            # L2 regularization + L1 projection
            l2_reg = 0.01 * current_a
            l1_reg = 0.05 * np.sign(current_a)
            current_a -= l2_reg
            current_a = np.sign(current_a) * np.maximum(np.abs(current_a) - l1_reg, 0)
            
            # Ensure integral constraint
            h_test = h_from_coeffs(current_a)
            h_test = h_test * target_sum / np.sum(h_test)
            h_test = np.clip(h_test, 0, 1)
            
            # Rescale integral
            current_a[0] = current_a[0] + (target_sum - np.sum(h_test)) / n_points
            
            iter_count += 1
            
            if iter_count % 100 == 0:
                alpha *= 0.9
            
        except Exception as e:
            print(f"Error during optimization: {e}")
            break

    return (best_h, best_c5, n_points)
