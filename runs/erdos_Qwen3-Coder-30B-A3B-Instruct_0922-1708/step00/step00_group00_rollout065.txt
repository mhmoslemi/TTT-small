import numpy as np
import scipy.signal as signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    # Compute the correlation of h with (1 - h)
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Apply dx scaling
    overlaps = correlation * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize a population of step functions."""
    population = []
    
    # If initial_h is provided, use it for some individuals
    if initial_h is not None:
        # Normalize the initial_h to satisfy sum constraint
        initial_h = np.array(initial_h)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        # Add initial_h to population if it's provided
        population.append(initial_h.copy())
        
    # Fill the rest of the population with random individuals
    for _ in range(pop_size - len(population)):
        # Generate a random step function
        h = np.random.rand(n_points)
        # Normalize to satisfy the sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        population.append(h)
        
    return population

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    # Select a random crossover point
    point = np.random.randint(1, n)
    
    # Create offspring
    offspring1 = np.concatenate([parent1[:point], parent2[point:]])
    offspring2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return offspring1, offspring2

def mutate(h, mutation_rate=0.05):
    """Mutate a step function."""
    n = len(h)
    mutated = h.copy()
    
    # Mutate each element with given probability
    for i in range(n):
        if np.random.rand() < mutation_rate:
            # Apply small random change
            delta = np.random.normal(0, 0.05)
            mutated[i] += delta
            # Clip to [0, 1]
            mutated[i] = np.clip(mutated[i], 0, 1)
            
    # Normalize to sum constraint
    mutated = mutated * (n / 2) / np.sum(mutated)
    
    return mutated

def tournament_selection(population, fitnesses, tournament_size=5):
    """Perform tournament selection."""
    n = len(population)
    selected = []
    
    for _ in range(n):
        # Select random participants
        indices = np.random.choice(n, tournament_size, replace=False)
        participants = [fitnesses[i] for i in indices]
        # Select the best
        best_idx = indices[np.argmax(participants)]
        selected.append(population[best_idx].copy())
        
    return selected

    best_c5 = fitnesses[best_idx]
    best_generation = 0

    start_time = time.time()
    
    # GA loop
    for generation in range(max_generations):
        # Check time budget
        if time.time() - start_time > budget_s:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = []
        for i in range(0, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < pop_size else selected[0]
            offspring1, offspring2 = crossover(parent1, parent2)
            offspring1 = mutate(offspring1)
            offspring2 = mutate(offspring2)

        # Replace population
        population = new_population[:pop_size]

        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Elitism: keep best
        current_best_idx = np.argmin(fitnesses)
