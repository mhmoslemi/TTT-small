import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        n_points = 70
        h0 = np.array([0.5 * (2.0 / n_points)] * n_points)
    
    # Normalize initial guess
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    # Check feasibility
    if time() - start_time > budget_s * 0.95:
        return h0, evaluate_erdos_solution(h0), n_points
    
    # Setup problem
    x = cp.Variable(n_points)
    Z = cp.Variable((n_points, n_points))
    t = cp.Variable()
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n_points / 2
    ]
    
    # Z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i,j] <= x[i],
                Z[i,j] <= 1 - x[j],
                Z[i,j] >= x[i] + x[j] - 1
            ]
    
    # Objective constraints
    C = []
    for k in range(-n_points+1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >=0) & (idx < n_points)
        sum_z = cp.sum(Z[np.arange(n_points)[valid], idx[valid]])
        C.append(sum_z)
    
    # Constrain all correlations
    for c in C:
        constraints += [c <= t]
    
    # Objective
    objective = cp.Minimize(t)
    
    # Solve
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s - (time() - start_time))
    except:
        try:
            prob.solve(solver=cp.SCS, verbose=False, time_limit=budget_s - (time() - start_time))
        except:
            pass
    
    # Extract result
    if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        h_opt = np.clip(x.value, 0, 1)
        h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
    else:
        h_opt = h0.copy()
    
    # Validate
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2.0) / np.sum(h_opt)
    
    # Final evaluation
    try:
        c5_bound = evaluate_erdos_solution(h_opt)
    except:
        c5_bound = np.inf
    
    return h_opt, c5_bound, n_points
