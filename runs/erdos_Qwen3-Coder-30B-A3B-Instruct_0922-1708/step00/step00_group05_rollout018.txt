import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize with provided construction or make one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h_init = np.ones(n_points) * 0.5
    
    # Ensure integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Make sure it's in [0, 1]
    h_init = np.clip(h_init, 0, 1)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init)
    
    start_time = time.time()
    
    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = c2 = 2.0
    max_iter = int(budget_s / 0.05) if budget_s > 0.05 else 1000
    inertia_decay = 0.99
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Initialize one particle with h_init, others with perturbed version
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for i in range(1, n_particles):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal bests
    pbest = [p.copy() for p in particles]
    pbest_c5 = [evaluate_erdos_solution(p) for p in particles]
    
    # Global best
    gbest_idx = np.argmin(pbest_c5)
    gbest = pbest[gbest_idx].copy()
    gbest_c5 = pbest_c5[gbest_idx]
    
    iteration = 0
    no_improve_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (gbest - particles[i]))
            
            # Keep velocity bounded
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            # Update position
            particles[i] += velocities[i]
            
            # Ensure bounds [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Normalize integral to 1
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new solution
            c5 = evaluate_erdos_solution(particles[i])
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest[i] = particles[i].copy()
                pbest_c5[i] = c5
                
                # Update global best
                if c5 < gbest_c5:
                    gbest = particles[i].copy()
                    gbest_c5 = c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
        
        # Apply inertia decay
        w *= inertia_decay
        
        # Diversity reintroduction after too many iterations with no improvement
        if no_improve_count > 100:
            for i in range(n_particles):
                # Add some noise to restart exploration
                noise = np.random.normal(0, 0.01, n_points)
                particles[i] = np.clip(particles[i] + noise, 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            no_improve_count = 0
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(gbest)
    
    return (gbest, final_c5, n_points)

def evaluate_erdos_solution(h):
    """Evaluate C5 for a given h using the provided optimizer's evaluation."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Using full convolution to compute the maximum overlap
    corr = correlate(h, 1-h, mode="full") * dx
    # Return the maximum correlation value, which is the maximum overlap
    return np.max(corr)
