import numpy as np
import time
from scipy import signal
import math

def evaluate_erdos_solution(h, n_points):
    """Evaluate the Erdős constant C5 for a given step function h."""
    dx = 2.0 / n_points
    # Compute the overlap integral using convolution
    correlation = signal.correlate(h, 1 - h, mode="full")
    # The maximum value of the convolution is the C5 value we want to minimize
    max_correlation = np.max(correlation)
    return max_correlation * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70  # Using the provided starting point size
    dx = 2.0 / n_points
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    best_positions = []
    best_values = []
    
    # Start with the initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a reasonable initial construction
        h_init = np.ones(n_points) * 0.5
        h_init[10:20] = 1.0  # some spikes
        h_init[50:60] = 0.0  # some drops
    
    # Normalize to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    h_init = np.clip(h_init, 0, 1)
    
    # Create initial population
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    best_positions.append(h_init.copy())
    best_values.append(evaluate_erdos_solution(h_init, n_points))
    
    # Fill with perturbed versions
    for _ in range(n_particles - 1):
        h_perturb = h_init + np.random.normal(0, 0.05, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        # Normalize
        h_perturb = h_perturb * (n_points / 2) / np.sum(h_perturb)
        particles.append(h_perturb)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        best_positions.append(h_perturb.copy())
        best_values.append(evaluate_erdos_solution(h_perturb, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = 10000
    iter_per_eval = 50  # roughly 0.05 sec per batch of iterations
    
    current_best = min(best_values)
    stagnation_counter = 0
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Iterate through swarm
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (best_positions[i] - particles[i]) + 
                            c2 * r2 * (best_positions[np.argmin(best_values)] - particles[i]))
            
            # Bound velocity
            velocities[i] = np.clip(velocities[i], -0.1, 0.1)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Clip and normalize after update
            particles[i] = np.clip(particles[i], 0, 1)
            # Normalize sum to n_points / 2
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            current_c = evaluate_erdos_solution(particles[i], n_points)
            
            # Update personal best
            if current_c < best_values[i]:
                best_values[i] = current_c
                best_positions[i] = particles[i].copy()
                
        # Update global best
        current_best_new = min(best_values)
        if current_best_new < current_best:
            current_best = current_best_new
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        # Occasionally introduce diversity
        if stagnation_counter > 100:
            for i in range(n_particles):
                if np.random.random() < 0.5:
                    particles[i] = best_positions[np.argmin(best_values)] + np.random.normal(0, 0.01, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    current_c = evaluate_erdos_solution(particles[i], n_points)
                    best_values[i] = current_c
                    best_positions[i] = particles[i].copy()
            stagnation_counter = 0
            
        # Tournament selection
        if iter_count % iter_per_eval == 0:
            # Select top half
            indices = np.argsort(best_values)[:n_particles//2]
            for i in range(n_particles//2, n_particles):
                # Replace bottom half with new random particles
                h_new = particles[indices[i % (n_particles//2)]].copy() + np.random.normal(0, 0.03, n_points)
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                particles[i] = h_new
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                best_positions[i] = h_new.copy()
                best_values[i] = evaluate_erdos_solution(h_new, n_points)
                
        iter_count += 1
    
    # Final evaluation
    final_idx = np.argmin(best_values)
    final_h = best_positions[final_idx]
    final_c5 = best_values[final_idx]
    
    return (final_h, final_c5, n_points)
