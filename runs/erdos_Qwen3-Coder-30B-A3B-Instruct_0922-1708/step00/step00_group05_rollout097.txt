import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initial guess from parent construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Start with a simple approximation
        h_init = np.ones(n_points) * 0.5
    
    # Fourier representation
    m = 10  # Number of cosine modes (even only to maintain symmetry)
    
    # Variables to optimize are the coefficients of cos(k*pi*x/2), k=1..m
    # a0 is set to 0.5 to satisfy integral constraint
    # Initial coefficients
    a_coeffs = np.random.rand(m) * 0.8 + 0.1  # clip to [0.1, 0.9] for better starting point
    a_coeffs[0] = 0.1  # example nonzero lower value to encourage exploration
    
    def h_from_coeffs(a_coeffs):
        """Convert Fourier coefficients to h(x) samples"""
        h_vals = np.ones(n_points) * 0.5  # a0 = 0.5
        for k in range(1, m+1):
            if k > len(a_coeffs):
                break
            coeff = a_coeffs[k-1]
            # Cosine modes
            x_vals = np.linspace(0, 2, n_points, endpoint=False)
            h_vals += coeff * np.cos(np.pi * k * x_vals / 2.0)
        # Normalize
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals) * dx
        if current_sum > 0:
            h_vals *= target_sum / current_sum
        return h_vals
    
    def compute_C(h):
        """Compute maximum overlap integral (C5)"""
        # correlation with (1 - h) for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        # shifts are indices from -n+1 to n-1 (corresponding to k values)
        max_corr = np.max(np.abs(corr))
        return max_corr * dx
    
    def project_to_feasible(h):
        """Project h to satisfy constraints"""
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust sum to target
        current_sum = np.sum(h) * dx
        if current_sum > 0:
            h *= target_sum / current_sum
        return h
    
    def gradient_descent_step(a_coeffs, step_size=0.03):
        """Perform one gradient step on a_coeffs"""
        h_current = h_from_coeffs(a_coeffs)
        h_current = project_to_feasible(h_current)
        current_C = compute_C(h_current)
        
        # Compute finite difference gradient
        grad = np.zeros_like(a_coeffs)
        eps = 1e-4
        
        for i in range(len(a_coeffs)):
            a_plus = a_coeffs.copy()
            a_minus = a_coeffs.copy()
            a_plus[i] += eps
            a_minus[i] -= eps
            
            h_plus = h_from_coeffs(a_plus)
            h_minus = h_from_coeffs(a_minus)
            
            h_plus = project_to_feasible(h_plus)
            h_minus = project_to_feasible(h_minus)
            
            C_plus = compute_C(h_plus)
            C_minus = compute_C(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update
        new_a = a_coeffs - step_size * grad
        
        # Clip to feasible range
        new_a = np.clip(new_a, 0, 1)
        
        return new_a
    
    best_C = float('inf')
    best_h = None
    
    # Optimization loop
    iter_count = 0
    patience = 0
    last_improvement = 0
    
    # Initial evaluation
    h_start = h_from_coeffs(a_coeffs)
    h_start = project_to_feasible(h_start)
    current_C = compute_C(h_start)
    if current_C < best_C:
        best_C = current_C
        best_h = h_start.copy()
    
    while time.time() - start_time < budget_s - 1.0:  # leave some buffer
        if patience >= 100:
            break
        iter_count += 1
        a_coeffs = gradient_descent_step(a_coeffs)
        # Periodically check
        if iter_count % 10 == 0:
            h_temp = h_from_coeffs(a_coeffs)
            h_temp = project_to_feasible(h_temp)
            new_C = compute_C(h_temp)
            if new_C < best_C - 1e-8:
                best_C = new_C
                best_h = h_temp.copy()
                patience = 0
                last_improvement = iter_count
            else:
                patience += 1
        if iter_count - last_improvement > 1000:
            # Reduce learning rate if stagnating
            step_size *= 0.8
    
    # Final ensures best structure
    if best_h is None:
        best_h = h_from_coeffs(a_coeffs)
        best_h = project_to_feasible(best_h)
        best_C = compute_C(best_h)
        
    return (best_h.tolist(), float(best_C), int(n_points))
