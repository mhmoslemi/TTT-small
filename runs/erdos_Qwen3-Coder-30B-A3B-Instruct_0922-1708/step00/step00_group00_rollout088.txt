import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for a given h"""
    # Compute full correlation with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx and find maximum
    max_overlap = np.max(corr * dx)
    return max_overlap

def ensure_constraints(h, n_points, dx):
    """Ensure h satisfies constraints: 0≤h≤1 and sum(h)*dx=1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        # Linear adjustment
        adjustment = (target_sum - current_sum) / n_points
        h = np.clip(h + adjustment, 0, 1)
    return h

def initialize_population(n_points, population_size, initial_h=None):
    """Initialize population of step functions"""
    population = []
    dx = 2.0 / n_points
    
    if initial_h is not None:
        # Use initial construction as first individual
        h = ensure_constraints(initial_h.copy(), n_points, dx)
        population.append(h)
    
    # Fill rest with random samples
    while len(population) < population_size:
        # Random initialization
        h = np.random.rand(n_points)
        h = ensure_constraints(h, n_points, dx)
        population.append(h)
        
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmax([fitnesses[i] for i in indices])]
    return population[best_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents"""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate an individual"""
    n_points = len(h)
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Settings
    n_points = 45  # As mentioned in pre-import
    population_size = 100
    tournament_size = 5
    crossover_prob = 0.8
    mutation_rate = 0.05
    elitism_size = population_size // 5
    
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, population_size, initial_h_values if 'initial_h_values' in globals() else None)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    try:
        while time.time() - start_time < budget_s - 0.1:
            generation += 1
            
            # Evaluate fitness
            fitnesses = []
            for h in population:
                c5 = evaluate_h(h, dx)
                fitnesses.append(-c5)  # Negative because we minimize
            
            # Check for best solution
            current_best_idx = np.argmax(fitnesses)
            current_best_c5 = -fitnesses[current_best_idx]
            if current_best_c5 < best_c5:
                best_c5 = current_best_c5
                best_h = population[current_best_idx].copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Stopping condition
            if no_improvement_count >= max_no_improvement:
                break
                
            # Selection and reproduction
            new_population = []
            
            # Elitism
            elite_indices = np.argsort(fitnesses)[-elitism_size:]
            for i in elite_indices:
                new_population.append(population[i].copy())
            
            # Generate offspring
            while len(new_population) < population_size:
                # Selection
                parent1 = tournament_selection(population, fitnesses, tournament_size)
                parent2 = tournament_selection(population, fitnesses, tournament_size)
                
                # Crossover
                child1, child2 = crossover(parent1, parent2, crossover_prob)
                
                # Mutation
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                
                # Apply constraints
                child1 = ensure_constraints(child1, n_points, dx)
                child2 = ensure_constraints(child2, n_points, dx)
                
                new_population.extend([child1, child2])
            
            # Trim to population size
            population = new_population[:population_size]
            
    except Exception as e:
        pass
    
    if best_h is None:
        # Fallback to initial solution if nothing found
        best_h = initial_h_values if 'initial_h_values' in globals() else np.full(n_points, 0.5)
        best_c5 = evaluate_h(best_h, dx)
    
    return (best_h, best_c5, n_points)
