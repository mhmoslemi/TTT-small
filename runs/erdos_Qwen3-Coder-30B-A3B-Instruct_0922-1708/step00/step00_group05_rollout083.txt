import numpy as np
from scipy.optimize import minimize
from cvxpy import *
import time
import random

def evaluate_erdos_solution(h):
    # Compute the overlap integral for the given step function h
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (non-negative lags)
    # The maximum overlap is for lags in range [-n+1, n-1]
    max_overlap = 0
    for k in range(-n+1, n):
        idx = k + n - 1
        if 0 <= idx < len(corr):
            overlap = corr[idx] * dx
            max_overlap = max(max_overlap, overlap)
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Default construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.4
        h0[-1] = 0.6
    
    # Normalize initial h to sum to n_points/2
    dx = 2.0 / n_points
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    def optimize_one(h_init):
        # Scale of h is [0,1]
        h = Variable(n_points)
        C = Variable()

        # Constraints
        constraints = [h >= 0, h <= 1]
        # Integral constraint
        constraints.append(sum(h) == n_points / 2)
        # Overlap constraints
        for k in range(-n_points+1, n_points):
            # Compute the overlap integral for a fixed lag k
            if k >= 0:
                # For k >= 0: h[i] * (1 - h[i+k])
                expr = 0
                for i in range(n_points - k):
                    expr += h[i] * (1 - h[i+k])
                expr *= dx
                constraints.append(expr <= C)
            else:
                # For k < 0: h[i+|k|] * (1 - h[i])
                expr = 0
                for i in range(-k):
                    expr += h[i-k] * (1 - h[i])
                expr *= dx
                constraints.append(expr <= C)

        # Objective
        objective = Minimize(C)
        prob = Problem(objective, constraints)
        
        try:
            # Try different solvers in order of preference
            solvers.options['max_iters'] = 10000
            prob.solve(solver=CVXOPT, verbose=False)
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                h_val = np.array([float(h[i].value) for i in range(n_points)])
                c5_val = float(C.value)
                return h_val, c5_val
        except Exception:
            pass
        # If optimization failed, return initial x
        return h_init, evaluate_erdos_solution(h_init)
    
    # Try multiple initializations
    attempts = 0
    max_attemps = 15
    while attempts < max_attemps and (time.time() - start_time) < budget_s - 2:
        attempts += 1
        h_current = h0.copy()
        # Random perturbations
        for i in range(n_points):
            h_current[i] += (random.random() - 0.5) * 0.1
        h_current = np.clip(h_current, 0, 1)
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        h_new, c5_new = optimize_one(h_current)
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new

    # Ensure final solution is close to the bounds and has integral = 1
    dx = 2.0 / n_points
    # Re-normalize to correct sum
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
