import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric initial guess
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Make it symmetric plateau
        center = n_points // 2
        half_width = n_points // 4
        h[max(0, center - half_width):min(n_points, center + half_width)] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iterations = 3000
    convergence_tol = 1e-5
    
    start_time = time.time()
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(-n_points + 1, n_points):
            k_idx = k + n_points - 1
            if k >= 0:
                overlap = np.sum(h[:n_points-k] * (1 - h[k:]))
            else:
                overlap = np.sum(h[-k:] * (1 - h[:n_points+k]))
            overlaps[k_idx] = overlap * dx
        
        # Smooth approximation of max
        C5_approx = np.log(np.sum(np.exp(tau * overlaps))) / tau
        
        # Gradient of log-sum-exp approximation
        exp_overlaps = np.exp(tau * overlaps)
        exp_sum = np.sum(exp_overlaps)
        grad = np.zeros(n_points)
        for i in range(n_points):
            grad_i = 0
            for k in range(-n_points + 1, n_points):
                k_idx = k + n_points - 1
                if k >= 0:
                    if i >= k and i < n_points:
                        grad_i += exp_overlaps[k_idx] * (-1 + h[i-k]) * dx
                else:
                    if i < n_points + k and i >= 0:
                        grad_i += exp_overlaps[k_idx] * (-1 + h[i-k]) * dx
            grad[i] = grad_i / exp_sum
        
        # Gradient descent step
        h_new = h - learning_rate * grad
        
        # Project h to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Anneal temperature and learning rate
        tau *= anneal_factor
        if tau < 1e-8:
            tau = 1e-8
        if learning_rate > 0.001:
            learning_rate *= 0.9999
        
        # Check convergence
        if iteration > 50 and abs(current_c5 - best_c5) < convergence_tol:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute all cross-correlations for k = 0 to 2*n_points-2
    n = len(h)
    conv = correlate(h, 1-h, mode='full')
    # Only consider valid shifts for the overlap
    overlaps = conv[n-1:]  # From k=0 to k=n-1, but we want k=0 to k=n-1
    # But we compute from 0 to 2*n-2 in cross-correlation which includes negative
    # correct evaluation: for k from 0 to n-1, valid range of the overlap structure
    overlaps = conv[n-1:2*n-1]  # This gets the correct overlaps
    return np.max(overlaps) * dx
