import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47  # Use the provided initial construction
    dx = 2.0 / n_points
    
    # Initialize h from provided initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create random initial feasible solution
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure initial h is valid
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    learning_rate = 0.1
    tolerance = 1e-6
    start_time = time.time()
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps using correlate
        # For each lag k, compute sum of h[i] * (1 - h[i+k])
        # This requires computing correlations
        h_padded = np.concatenate([h, np.zeros(n_points - 1)])
        h_mirror = np.concatenate([h, np.zeros(n_points - 1)])
        correlation = correlate(h_padded, 1 - h_mirror, mode='full')
        correlation = correlation[n_points - 1:]  # Trim to valid lags
        correlation = correlation * dx  # Scale by dx

        # Find maximum correlation
        max_k = np.argmax(correlation)
        max_val = correlation[max_k]
        
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Compute gradient for linear subproblem
        # This is a simplified gradient: for k=0, g[i] = 1-h[i+0], but we need to be more precise
        
        # Let's recompute expansion more carefully:
        # For fixed k_max, gradient is partial of overlap w.r.t. h[i]
        # ∂/∂h[i] ∫ h(x)(1 - h(x+k)) dx = (1 - h[i+k])  
        # But we need to make sure we compute this properly with indexing
        
        grad = np.zeros(n_points)
        if max_k >= 0:
            # k_max > 0 means h[i] interacts with h[i+k_max] 
            # Effect on overlap term h[i]*(1-h[i+k_max]) is (-h[i+k_max]) + (-h[i]*h'[i+k_max])
            # i.e. gradient is (-1 * h[i+k_max] if constrained index exists)
            # But really we're looking for direction of steepest ascent, so we solve linear program
            
            # Linear subproblem: minimize <grad, d> subject to 0 <= d_i <= 1, sum d_i = n_points/2
            # In fact, the gradient should be computed using lag k_max:
            # if h[i] contributes to the overlap for lag k_max, then d[i] coeff = 1-h[i+k_max], 
            # but now it's a matter of what we're minimizing. 
            # We want to minimize the overlap by decreasing h[i] when (1-h[i+k_max]) > 0.
            for i in range(n_points):
                if (i + max_k) < n_points:
                    grad[i] = - (1 - h[i + max_k])
                else:
                    # Wrap around
                    grad[i] = 0
        else:
            # k_max < 0: it's equivalent to positive lag but we shift differently
            for i in range(n_points):
                wrapped_index = (i + max_k) % n_points
                grad[i] = - (1 - h[wrapped_index])
        
        # Solve linear subproblem: minimize <grad, d> over feasible region
        # d is the direction to move in, not just the point
        # Intuitive: move toward the direction where gradient is most negative  
        # So a simple strategy: k = argmin grad[i], and set d[k] = 1, others = 0, then rescale
        # Or, cluster points that help to reduce overlap.
        # That's not accurate though.
        
        # A more accurate technique based on Frank-Wolfe's derivation:
        # Minimize <grad, d> such that 0 <= d[i] <= 1, sum d[i] = n_points / 2  
        # Now, what Frank-Wolfe does:
        # 1. Solve: min_{d in feasible} <grad, d>, i.e., minimize linear fit
        # 2. Update: h <- (1 - eta) * h + eta * d  
        # 3. Project back to constraint bundle and rescale
        
        # To minimize <grad, d>, pick most negative entries:
        # But, have sum = n_points/2 constraint.
        # This is an inefficient quadratic solver but let's keep simple:
        d = np.zeros(n_points)
        sort_idx = np.argsort(grad)  # Desc by default, so smaller (more negative) grad at start
        
        # Just pick the first few points that satisfy the sum constraint
        # Strategy: select indices such that sum = n_points/2 and gradient minimized
        # We weight by negative grad, so minimize linear form
        remaining_sum = n_points / 2
        for i in sort_idx:
            d[i] = min(1.0, remaining_sum)
            remaining_sum -= d[i]
            if remaining_sum <= 0:
                break
        if remaining_sum > 0:
            for i in sort_idx[1:]:
                d[i] += remaining_sum / (n_points - 1)
                break
                
        # Project d back to feasible:
        d = np.clip(d, 0, 1)
        # Adjust to make sum correct:
        current_sum = np.sum(d)
        if current_sum > 0:
            d = d * (n_points / 2) / current_sum
        
        # Perform Frank-Wolfe step: update h
        h = (1 - learning_rate) * h + learning_rate * d
        h = np.clip(h, 0, 1)
        
        # Renormalize to maintain sum = n_points/2 constraint
        current_sum = np.sum(h)
        h = h * (n_points / 2) / current_sum
        
        # Check for improvement
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
    # Final evaluation  
    h_padded = np.concatenate([best_h, np.zeros(n_points - 1)])
    h_mirror = np.concatenate([best_h, np.zeros(n_points - 1)])
    correlation = correlate(h_padded, 1 - h_mirror, mode='full')
    correlation = correlation[n_points - 1:] * dx
    final_c5 = np.max(correlation)
    
    return (best_h.tolist(), final_c5, n_points)
