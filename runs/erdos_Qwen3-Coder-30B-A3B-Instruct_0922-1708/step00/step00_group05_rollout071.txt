import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000
    lr = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = None
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h[0] = 0.3
        h[-1] = 0.3
    
    # Project into Fourier domain - symmetric cosine basis
    m = 20
    a = np.zeros(m)
    a[0] = 0.5  # integral constraint
    if 'initial_h_values' in globals():
        # Initial Fourier coefficients
        h0 = h.copy()
        a[1:] = np.fft.fft(h0[:m//2])[1:m//2] / (n_points / 2)
    else:
        a[1:] = np.random.rand(m-1) * 0.5
    
    # Fourier basis functions
    x = np.linspace(0, 2, n_points, endpoint=False)
    cos_basis = np.array([np.cos(np.pi * k * x / 2) for k in range(m)])
    
    def project_h(a_coeffs):
        # Project back to h(x)
        h_proj = np.dot(a_coeffs, cos_basis)
        h_proj = np.clip(h_proj, 0, 1)
        # Rescale to meet integral constraint
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    def compute_overlap(h_vals):
        # Compute the maximum overlap
        h_correlate = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(h_correlate) * dx
    
    # Optimization loop using gradient descent
    start_time = time.time()
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Periodic projection
        if iter_count % 10 == 0:
            h = project_h(a)
        
        # Compute overlaps
        overlap = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(overlap)
        max_overlap = overlap[max_k] * dx
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iter_count
        
        # Finite-difference gradient estimation
        grad = np.zeros_like(a)
        epsilon = 1e-5
        for i in range(1, m):  # avoid a[0] due to constraint
            a_plus = a.copy()
            a_plus[i] += epsilon
            h_plus = project_h(a_plus)
            overlap_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_overlap_plus = overlap_plus[max_k] * dx
            
            a_minus = a.copy()
            a_minus[i] -= epsilon
            h_minus = project_h(a_minus)
            overlap_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            max_overlap_minus = overlap_minus[max_k] * dx
            
            grad[i] = (max_overlap_plus - max_overlap_minus) / (2 * epsilon)
        
        # Update
        a = a - lr * grad
        # Clip
        a = np.clip(a, 0, 1)
        a[0] = 0.5  # keep integral constraint
        
        iter_count += 1

        if iter_count - last_improvement > patience:
            lr *= 0.5
            last_improvement = iter_count
    
    if best_h is None:
        best_h = project_h(a)
    
    # Final validation
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_overlap(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
