import numpy as np
import cvxpy as cp
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        h = np.linspace(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode="full")) * dx
    start_time = time()
    
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s - 2:
        iter_count += 1
        # Compute current max overlap
        corr = np.correlate(h, 1-h, mode="full")
        c5 = np.max(corr) * dx
        max_idx = np.argmax(corr)
        
        # Linearize the objective
        # h(x)(1 - h(x+k)) approximates to h(x) - h(x)*(1-h(x+k)) 
        # or h(x) + h(x)*h(x+k) - h(x)
        # which simplifies to h(x)*h(x+k)  
        # Actually better to do first order Taylor approximation
        # f(x) = h(x) - h(x)*h(x+k) 
        # df/dh(x) = 1 - h(x+k)  
        # So linear approx: f(x) ≈ h(x) + (1 - h(x+k))(h(x) - h_old(x))
        # or better: linearize around current values, target is
        # h_k = sum_i h[i](1 - h[i+k]) = sum_i h[i] - sum_i h[i] * h[i+k]
        # Noting that sum_i h[i] = 1/dx (scaled), so:
        # sum_i h[i] - sum_i h[i] * h[i+k] 
        # What we want to minimize is max_k of this
        
        # Create linearized system (convex outer approximation)
        # For each k, we have sum_i h[i](1-h[i+k]) which is sum_i h[i] - sum_i h[i]*h[i+k]
        # Since h[i] is constant wrt optimization, we linearize h[i+k] term
        # On the current estimate h_old, 
        # h[i+k] ≈ h_old[i+k] -> linearize at h_old, not h

        # Pick the k which gives the highest correlation (maximum conflict)
        # It's the maximum of the full convolution
        # Which over all lags in the integer indices k from 0 to 2*n_points
        # This is the 'toughest' case to handle
        
        # So we focus on minimizing the overlap for that specific k
        # Let's extract this and make it convex:
        # Let z_k be full correlation coefficient for lag k
        # z_k = sum_i h[i] * (1 - h[i+k])
        
        # Approach: try to minimize one of these (the maximum one)
        # The linearization method: 
        # term h[i](1-h[i+k]) = h[i] - h[i]*h[i+k]    
        # Linearizing around h[x], we get:
        # h[i] - h[i]*h[i+k] ≈ h[i] - h[i]*h_old[i+k]
        # turns the objective into:
        # sum_i h[i] - sum_i h[i]*h_old[i+k]
        # which is still nonconvex...
        # better approach: 
        # h[i] - h[i]*h[i+k] 
        # derivative wrt h[i] = 1 - h[i+k]  
        # So linearizing around h_old:  
        # h_new[i] - h_old[i]*h_new[i+k] = h_old[i] - h_old[i]*h_old[i+k] + (1-h_old[i+k])(h_new[i] - h_old[i])
        # But this gets messy. 
        # Better: use the fact that it's \sum_i (h[i] - h[i]*h[i+k]).  
        # Single linearization around h_old: 
        # \sum_i (h[i] - h[i]*h[i+k]) -> 
        # \sum_i (h_old[i] - h_old[i]*h[i+k] + (h[i]-h_old[i])*(1-h_old[i+k])
        # This needs to be done for the appropriate k that makes this maximum.
        # Multiplier for k term: 1-h_old[i+k]
        
        # Let's try: actual gradient-based local optimization using scipy
        
        # Let's do one iteration of a gradient like approach:
        # Slightly more principled, use cvxpy to solve a version of
        # The problem: minimize sum_k w_k * (sum h[i] - sum h[i]*h[i+k]) 
        # where w_k are set appropriately. 
        # Better approach: compute the final correlation with current point,
        # and project down on Lag k which maximizes overlap.
        
        # Another viable approach is an SDP relaxation or Lagrangian decomposition.
        # Let's be smarter: use an iterative optimization heuristic
        # that changes h around the constraint violation direction
        
        # Use subgradient descent in the direction of max gradient  
        # compute gradient of the max term
        # Re-linearize the h[i+k] terms for k that has the peak correlation 
        # project back to constraints
        
        # Simplify and use CVXPY
        # Minimize: sum i in 0,...,n-1 : h[i]*w[i] = sum_i h[i] - sum i h[i]*v[i]
        # subject to h[i] in [0,1], sum(h)*dx = 1
        # Here w[i] = (1 - h[i+k]), v[i] = h[i+k] 

        # However easier is to just sample samples and run a simple algorithm.
        
        # Let's do a different approach:
        # We'll try constrained optimization step-by-step
        
        # Find k which gives the maximum correlation
        # k corresponds to the index of array = 0, 1, ..., n-1 (i+k) 
        # indices of cross-correlation: -n+1 to n-1, with peak at n-1 (gap or offset)
        # so the peak is at k = max_idx, in full mode = 2*n - 2 - max_idx
        # No need to do some tricky search... just optimize that k's connectivity 
        # directly.
        
        # Assuming peak is at index max_idx in corr (full convolved):
        # corr[i] = sum_j h[j] * (1 - h[j+(i-n+1)]) 
        # i=0...2n-2  i-n+1 = -n+1 ... n-1  
        # If peak at i*, then i* - n+1 = k* where we want to optimize
        # so k* = i* - n + 1, index into full array
        # Now relative to h indices (so relative to 0...n-1), it's offset i* - n + 1 from base
        # Say for k the correlation is computed as sum h[i] (1 - h[i+k])
        # So we do it for that k.  
        
        k = max_idx - (n_points - 1)
        # Officially k = -n+1 to n-1 (shift from start)
        # but the convention that corr[0] corresponds to k = 0, 
        # corr[i] corresponds to k = i - (n-1)
        
        # We want to optimize the specific linear term:
        # sum h[i] (1 - h[i+k]) = sum h[i] - sum h[i] * h[i+k]
        
        # Develop a projected gradient approach, but simpler still:
        # Use cvxpy to do one true optimization 
        h_var = cp.Variable(n_points)
        # Objective for this k: minimize sum_i h[i]*(1 - h[i+k])
        # Which is sum_i h[i] - sum h[i]*h[i+k] 
        # Convex approximation: in terms of known h_i, for i-k >=0,  
        # h[i] * h[i+k] linearized by h[i] * h_old[i+k] + h_old[i] * (h[i] - h_old[i])
        # But actually simpler: use maximum of the correlations to estimate the worst case
        
        # Idea: find k that maximizes CV        
        # Subtract off h[i] sum (constant)
        # Restructure: let S[i] = h[i] and T_k[i] = h[i+k] 
        # We want to minimize sum S[i] * (1 - T_k[i]) 
        # So minimize sum S[i] - S[i]*T_k[i] 
        # So reduce to minimizing sum S[i]*T_k[i], since term is fixed
        # where T_k[i] = h[i+k] if valid, else for convolution we take it as temp.
        
        # Compute full correlation, and pick k
        H = np.zeros(n_points)
        for i in range(n_points):
            H[i] = h[i] * (1 - h[(i+k) % n_points]) if abs(k) <= n_points-1 else 0
        corr_val = np.sum(H) * dx
        
        # We now want to solve:
        # minimize sum_i h[i] * (1 - h[i+k])
        # subject to h in [0, 1], sum(h) = 1/dx = n_points/2 
        
        # Method: equate all terms in moment matching?
        # That is, make h[i] another h[i] to satisfy equality in all but the 
        # specific constraint. Let's do CVXPY approach:
        h_cvx = cp.Variable(n_points)
        # Compute linearized value for the given max correlation
        # minimize Lin(h, h_old) where Lin = sum_i h[i] - sum_i h[i]*h_old[i+k]  
        # This is equivalent to minimizing sum_i (h[i] - h[i]*h_old[i+k])
        # Or minimize sum_i h[i] * (1 - h_old[i+k])
        # Here we move h_old to be constant for the choice of linearization
        # sum_i h[i] * (1 - h_old[i+k]) = sum_i h[i] - sum_i h[i] * h_old[i+k]
        # Which is still convex! We're minimizing linear function + constant.  

        # Compute sum_i h[i] * (1 - h_old[i+k])  
        objective = cp.sum(cp.multiply(h_cvx, 1 - h[np.roll(range(n_points), -k)]))
        constraints = [
            h_cvx >= 0,
            h_cvx <= 1,
            cp.sum(h_cvx) == n_points / 2
        ]
        
        prob = cp.Problem(cp.Minimize(objective), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            h_new = h_cvx.value
            if h_new is not None:
                h = h_new
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # fallback to simple heuristic
            if abs(k) == 0:  # if k=0, tackle all terms
                h = np.clip(h - 0.01, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5_current = np.max(np.correlate(h, 1-h, mode="full")) * dx
        if c5_current < best_c5:
            best_h = h.copy()
            best_c5 = c5_current
            no_improvement_count = 0
        else:
            no_improvement_count += 1

        if no_improvement_count > max_no_improvement:
            break
        if best_c5 < 0.3807:
            break
            
    return (best_h.tolist(), best_c5, n_points)
