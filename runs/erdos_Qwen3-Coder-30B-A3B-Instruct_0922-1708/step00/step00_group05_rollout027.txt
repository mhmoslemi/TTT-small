import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 1000
    perturb_fraction = 0.1
    perturb_delta = 0.05
    smoothing_window = 3
    
    # Initialize h from initial_h_values if available, otherwise create a reasonable guess
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Create a simple symmetric initial function
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
    
    # Ensure that h is valid and sum equals target_sum
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum != 0:
        h = h * target_sum / h_sum
    
    # Initial evaluation
    start_time = time.time()
    current_c5 = compute_c5(h, dx)
    best_h = h.copy()
    best_c5 = current_c5
    no_improvement_count = 0
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Generate perturbed version
        h_new = h.copy()
        num_perturb = max(1, int(perturb_fraction * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        
        # Apply small random perturbations
        for i in indices:
            h_new[i] += np.random.uniform(-perturb_delta, perturb_delta)
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Maintain sum constraint via projection
        old_sum = np.sum(h_new)
        if old_sum != 0:
            h_new = h_new * target_sum / old_sum
        
        # Apply smoothing to maintain shape
        if smoothing_window > 1:
            # Simple moving average smoothing
            h_filtered = np.zeros_like(h_new)
            for i in range(n_points):
                start_i = max(0, i - smoothing_window//2)
                end_i = min(n_points, i + smoothing_window//2 + 1)
                h_filtered[i] = np.mean(h_new[start_i:end_i])
            h_new = h_filtered
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new, dx)
        
        if new_c5 <= current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_h = h_new.copy()
                best_c5 = new_c5
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 100:
            break
    
    return (best_h.tolist(), best_c5, len(best_h))

def compute_c5(h, dx):
    """Compute C5 based on the overlap integral."""
    # Compute correlation of h with (1 - h)
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Compute the maximum value and multiply by dx
    max_correlation = np.max(correlation)
    c5 = max_correlation * dx
    return c5
