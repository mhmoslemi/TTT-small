import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Smaller for faster optimization
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset = 20
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial construction as seed
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0.copy())
        velocities.append(np.zeros_like(h0))
        for _ in range(swarm_size - 1):
            # Add noise to initial
            noise = np.random.normal(0, 0.1, n_points)
            h = h0 + noise
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Random initialization
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Convert to arrays
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track best
    personal_best = particles.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    last_improvement = 0
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all particles
        scores = np.zeros(swarm_size)
        for i in range(swarm_size):
            h = particles[i]
            # Cross-correlation of h with (1 - h)
            corr = correlate(h, 1 - h, mode='full')
            # Compute maximum overlap
            max_val = np.max(corr) * dx
            scores[i] = max_val
            
            # Update personal best
            if scores[i] < personal_best_scores[i]:
                personal_best_scores[i] = scores[i]
                personal_best[i] = h.copy()
                
        # Update global best
        min_score = np.min(scores)
        if min_score < global_best_score:
            global_best_idx = np.argmin(scores)
            global_best_score = min_score
            last_improvement = it
            
        # Check stagnation
        if it - last_improvement > stagnation_limit:
            # Inject diversity
            reset_count = max(1, int(swarm_size * diversity_reset / 100))
            indices = np.random.choice(swarm_size, size=reset_count, replace=False)
            for idx in indices:
                h = np.random.rand(n_points)
                h = h / np.sum(h) * (n_points / 2)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            last_improvement = it
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            v_new = (w * velocities[i] + 
                     c1 * r1 * (personal_best[i] - particles[i]) + 
                     c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            velocities[i] = v_new
            new_pos = particles[i] + velocities[i]
            
            # Clip and rescale
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos / np.sum(new_pos) * (n_points / 2)
            
            particles[i] = new_pos
            
    # Return best solution
    best_h = personal_best[global_best_idx]
    return (best_h.tolist(), global_best_score, n_points)
