import numpy as np
import cvxpy as cvx
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Get initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * target_sum
    else:
        h0 = np.full(n_points, 0.5 * dx)
    
    h_best = h0.copy()
    c5_best = evaluate_erdos_solution(h0, n_points)
    
    # Convex relaxation using CVXPY
    def solve_cvx_relaxation(h_init):
        try:
            # Variable definitions
            x = cvx.Variable(n_points)
            # Where we store the correlation values for different k
            max_correlation = cvx.Variable()
            
            # Constraints
            constraints = [
                x >= 0,
                x <= 1,
                cvx.sum(x) == target_sum
            ]
            
            # Compute all correlation terms and add them as constraints
            corrs = []
            for k in range(-n_points+1, n_points):
                # Compute the correlation at lag k
                corr_sum = 0
                for i in range(n_points):
                    j = i - k
                    if 0 <= j < n_points:
                        corr_sum += x[i] * (1 - x[j])
                corrs.append(corr_sum)
            
            # Maximize correlation over all lags
            for corr in corrs:
                constraints.append(corr <= max_correlation)
                
            # Objective
            objective = cvx.Minimize(max_correlation)
            prob = cvx.Problem(objective, constraints)
            
            # Solve with timeout
            prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
            
            if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
                x_val = x.value
                # Clip to [0,1] and rescale
                x_val = np.clip(x_val, 0, 1)
                x_val = x_val / np.sum(x_val) * target_sum
                return x_val
            else:
                return h_init.copy()
        except Exception:
            return h_init.copy()
    
    # Optimization loop with restarts
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h, n_points)
    
    # Limited number of optimizations to allow for multiple restarts
    max_iter = 20
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        h_new = solve_cvx_relaxation(best_h)
        c5_new = evaluate_erdos_solution(h_new, n_points)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # If we converge to a good enough solution, break early
            if best_c5 <= 0.3807:
                break
        else:
            # Random restart
            h_new = np.random.uniform(0, 1, n_points)
            h_new = h_new / np.sum(h_new) * target_sum
            c5_new = evaluate_erdos_solution(h_new, n_points)
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
    # Final evaluation with exact computation
    final_h = best_h.copy()
    final_c5 = evaluate_erdos_solution(final_h, n_points)
    
    return (final_h, final_c5, n_points)
