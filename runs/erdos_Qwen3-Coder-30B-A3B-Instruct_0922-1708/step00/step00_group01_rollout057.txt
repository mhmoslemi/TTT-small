import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Create a symmetric construction with tapering ends
        mid = n_points // 2
        for i in range(mid):
            h[i] = 0.1 + 0.9 * (i / mid)
            h[n_points-1-i] = 0.1 + 0.9 * (i / mid)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # Parameters
    tau = 1.0
    lr = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    tol = 1e-5
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all lags
        # Using correlation to compute overlap at each lag
        # overlap(k) = sum h[i] * (1 - h[i+k]) for all i
        overlap_values = np.correlate(h, 1 - h, mode='full')
        overlap_values = overlap_values[len(overlap_values)//2:] * dx
        
        # Approximate max with log-sum-exp
        log_sum_exp = np.log(np.sum(np.exp(tau * overlap_values)))
        c5_approx = log_sum_exp / tau
        
        # Compute gradient
        # Note: This is not exact gradient of log-sum-exp max,
        # but is a reasonable proxy for optimization
        grad = np.zeros_like(h)
        for k in range(len(overlap_values)):
            k_idx = k - (len(overlap_values) // 2)
            # Gradient contribution from overlap(k)
            if k_idx >= 0:
                for i in range(n_points):
                    j = (i + k_idx) % n_points
                    grad[i] -= (1 - h[j]) * dx * np.exp(tau * overlap_values[k])
            else:
                for i in range(n_points):
                    j = (i - k_idx) % n_points
                    grad[i] -= (1 - h[j]) * dx * np.exp(tau * overlap_values[k])
        
        # Normalize gradient
        norm = np.sum(np.exp(tau * overlap_values))
        if norm > 0:
            grad /= norm
            
        # Update h
        h -= lr * grad
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Enforce constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Anneal temperature
        tau *= anneal_factor
        if tau < 1e-6:
            tau = 1e-6
            
        # Evaluate exact C5
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
            
        # Check convergence
        if iter_num > 10 and abs(c5_approx - best_c5) < tol:
            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points)
    
    return (list(best_h), final_c5, n_points)

def evaluate_erdos_solution(h_vec, n_samples):
    """Return the maximum overlap integral"""
    h = np.array(h_vec)
    dx = 2.0 / n_samples
    # Compute overlap for all lags
    overlaps = np.correlate(h, 1-h, mode='full')
    # Extract positive lags only
    overlaps = overlaps[len(overlaps)//2:] * dx
    return np.max(overlaps)
