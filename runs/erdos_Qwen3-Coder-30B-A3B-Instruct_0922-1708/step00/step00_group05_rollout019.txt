import numpy as np
import time
from scipy import signal
import math

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Take max over all lags
    c5 = np.max(corr) * dx
    return c5

def normalize_h(h, target_sum):
    """Normalize h to have the specified sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (target_sum / current_sum)
    return h

def clip_and_normalize(h, target_sum):
    """Clip h to [0,1] then normalize to target_sum."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, target_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_particles = 20
    max_iter = 12000  # Estimated based on 0.05s per iter
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    vel_max = 0.1
    target_sum = n_points / 2
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = clip_and_normalize(h0, target_sum)
        particles.append(h0.copy())
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = clip_and_normalize(h0, target_sum)
        particles.append(h0.copy())
    
    # Create remaining particles
    for i in range(n_particles - 1):
        h = h0 + np.random.uniform(-0.1, 0.1, n_points)
        h = clip_and_normalize(h, target_sum)
        particles.append(h.copy())
    
    # Initialize velocities
    for i in range(n_particles):
        v = np.random.uniform(-0.1, 0.1, n_points)
        velocities.append(v.copy())
    
    # Evaluate all particles
    scores = []
    for i in range(n_particles):
        s = evaluate_erdos_solution(particles[i])
        scores.append(s)
    
    # Find best particle
    best_idx = np.argmin(scores)
    best_h = particles[best_idx].copy()
    best_score = scores[best_idx]
    
    # Track global best
    global_best_h = best_h.copy()
    global_best_score = best_score
    
    # Optimization loop
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (w * velocities[i] + 
                    c1 * r1 * (particles[i] - particles[i]) + 
                    c2 * r2 * (global_best_h - particles[i]))
            
            # Clip velocity
            v_new = np.clip(v_new, -vel_max, vel_max)
            
            # Update position
            h_new = particles[i] + v_new
            
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to sum=target_sum
            h_new = normalize_h(h_new, target_sum)
            
            # Evaluate new solution
            score_new = evaluate_erdos_solution(h_new)
            
            # Update particle's best if improved
            if score_new < scores[i]:
                scores[i] = score_new
                particles[i] = h_new.copy()
            
            # Update global best if improved
            if score_new < global_best_score:
                global_best_score = score_new
                global_best_h = h_new.copy()
                no_improvement_count = 0
            else:
                no_improvement_count += 1
            
            # Store new velocity
            velocities[i] = v_new.copy()
        
        # Occasionally restart worst particles to maintain diversity
        if no_improvement_count > 100:
            # Replace worst 10% with random solutions
            worst_indices = np.argsort(scores)[-int(0.1 * n_particles):]
            for idx in worst_indices:
                h = global_best_h + np.random.uniform(-0.01, 0.01, n_points)
                h = clip_and_normalize(h, target_sum)
                particles[idx] = h.copy()
                scores[idx] = evaluate_erdos_solution(h)
            
            # Update global best if any in restart improved it
            best_idx = np.argmin(scores)
            if scores[best_idx] < global_best_score:
                global_best_score = scores[best_idx]
                global_best_h = particles[best_idx].copy()
            
            no_improvement_count = 0
            
    # Final normalization check
    global_best_h = clip_and_normalize(global_best_h, target_sum)
    final_score = evaluate_erdos_solution(global_best_h)
    
    return (global_best_h.tolist(), final_score, n_points)
