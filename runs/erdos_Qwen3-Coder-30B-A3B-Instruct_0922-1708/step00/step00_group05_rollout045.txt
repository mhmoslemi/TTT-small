import numpy as np

def evaluate_c5(h_values):
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for shifts k from -n+1 to n-1
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract only the valid shifts for overlap computation
    # The overlap integral is for shifts that result in proper overlap
    # i.e., valid k values from 0 to n-1 in the shifted overlap context
    # However, in full correlation, we get indices from -n+1 to n-1
    # For shift k = i - (n - 1), we have valid range: (n-1) - (n-1) = 0 to 2*(n-1) - (n-1) = n-1
    # For overlap integral, the valid shifts are 0 to n-1, but to calculate C5, it's max of all overlaps
    # So take the correct slice for all relevant overlaps: 
    # We need shifts from -n+1 to n-1, where shift k=0 means no shift, k=1 means left shift by 1, etc.
    # But for correct indexing: 
    # corr contains index from 0 to 2*n-2 corresponding to shifts -n+1 to n-1
    # So max is corr[n-1 : n-1 + n] = corr[n-1 : 2*n-1] but we need max of all these
    # or actually we need max of first n-1 + last n-1 = 2*(n-1) entries after center
    # The overlap corresponds to indices such that after shifting, there is benefit in overlap calculation
    # It's simpler to just do element-wise
    overlaps = np.array([corr[n-1 + k] for k in range(n)]) * dx
    return max(overlaps)
