import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial configuration
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Regularization parameter
    lambda_reg = 1e-4
    
    def compute_C5(h):
        # Compute the full convolution
        conv = np.correlate(h, 1 - h, mode="full")
        # Take the maximum over lags (middle values)
        max_val = np.max(conv[len(h)-1:])
        return max_val * dx
    
    def objective(h_vec):
        # Reshape back to array
        h = h_vec.copy()
        # Add regularization term
        reg_term = lambda_reg * np.sum(h**2)
        # Compute C5
        c5 = compute_C5(h)
        # Return sum of C5 and regularization
        return c5 + reg_term
    
    def gradient(h_vec):
        h = h_vec.copy()
        # Compute the maximum overlap k_index and corresponding contribution
        conv = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(conv[len(h)-1:]) - len(h) + 1  # Convert to actual lag
        # The gradient of C5 with respect to h
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Gradients from the maximum correlation term for this h[i]
            j = i - k_max
            j = j % len(h)
            grad[i] = 1 - 2 * h[j]  # Derived from chain rule
        # Add regularization gradient
        grad += 2 * lambda_reg * h
        return grad
    
    # Keep track of best solution
    best_h = h.copy()
    best_c5 = compute_C5(h)
    
    # Optimization loop
    max_iter = 800
    iter_count = 0
    evaluated = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        iter_count += 1
        
        # compute C5
        current_c5 = compute_C5(h)
        if current_c5 < best_c5:
            best_h = h.copy()
            best_c5 = current_c5
        
        # gradient descent step
        grad = gradient(h)
        # line search step size
        lr = 1e-3
        new_h = h - lr * grad
        # Clip to feasible region
        new_h = np.clip(new_h, 0, 1)
        # Rescale to maintain constraint
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        h = new_h
        
        evaluated += 1
        
    # Local search for shifts
    best_shifted = best_h.copy()
    best_shift_c5 = best_c5
    for _ in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        # Generate shift values
        shift_amount = (np.random.rand() - 0.5) * 0.2
        shifted = np.roll(best_h, int(shift_amount * n_points))
        shifted = np.clip(shifted, 0, 1)
        # Rescale
        shifted = shifted * (n_points / 2) / np.sum(shifted)
        c5_shift = compute_C5(shifted)
        if c5_shift < best_shift_c5:
            best_shifted = shifted.copy()
            best_shift_c5 = c5_shift
    
    # Final validation
    final_c5 = compute_C5(best_shifted)
    return (best_shifted.tolist(), final_c5, n_points)
