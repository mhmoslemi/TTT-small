import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction
        n_points = 45
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project h to satisfy sum constraint
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if abs(current_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / current_sum
        return np.clip(h_vec, 0, 1)
    
    # Evaluate overlap integral
    def compute_c5(h_vec):
        # Compute all overlaps for all shifts
        h_extended = np.concatenate([h_vec, np.zeros(n_points - 1)])
        h_flipped = np.flip(h_vec)
        correlation = signal.correlate(h_extended, h_flipped, mode='full')
        # Extract the valid overlaps
        overlaps = correlation[n_points-1:]
        # Only consider valid k values
        valid_overlaps = []
        for k in range(-n_points+1, n_points):
            if k >= 0:
                idx = k
            else:
                idx = n_points - 1 + k
            if idx < len(overlaps):
                valid_overlaps.append(overlaps[idx])
        # Each overlap needs to be multiplied by dx:
        valid_overlaps = np.array(valid_overlaps) * dx
        return np.max(valid_overlaps)
    
    # Smooth max approximation
    def smooth_max(overlaps, tau=0.1):
        return np.log(np.sum(np.exp(tau * overlaps))) / tau
    
    # Gradient of smooth max loss
    def compute_gradient(h_vec, tau=0.1):
        h_extended = np.concatenate([h_vec, np.zeros(n_points - 1)])
        h_flipped = np.flip(h_vec)
        correlation = signal.correlate(h_extended, h_flipped, mode='full')
        # Extract all overlaps
        overlaps = correlation[n_points-1:]
        # Compute contribution of each position to gradient
        grad = np.zeros_like(h_vec)
        exp_terms = np.exp(tau * overlaps)
        exp_sum = np.sum(exp_terms)
        
        # For each overlap term, shift h_flipped accordingly
        for i in range(n_points):
            # Compute contribution to gradient w.r.t. h[i]
            contrib = 0.0
            for k in range(n_points):
                if i + k < n_points:
                    j = i + k
                    # contribution is -h[j] * exp(τ * overlap) / (exp_sum * h[i] * (1-h[i+k]))
                    contrib += -h_vec[j] * exp_terms[k] / (exp_sum * (1 - h_vec[i]))
            grad[i] = contrib * dx
            
        return grad
    
    def backtracking_line_search(h_vec, grad, alpha_start=0.1, beta=0.5):
        alpha = alpha_start
        loss_old = smooth_max(np.array([compute_c5(h_vec)]))
        while True:
            h_new = project_h(h_vec - alpha * grad)
            loss_new = smooth_max(np.array([compute_c5(h_new)]))
            if loss_new < loss_old:
                return h_new, alpha
            alpha *= beta
            if alpha < 1e-10:
                return h_vec, 0.0
    
    # Main optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    start_time = time.time()
    no_improvement_count = 0
    
    for iteration in range(10000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # compute current value
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > 50:
                break  # Early termination if no improvement

        # Compute gradient
        grad = compute_gradient(h)
        
        # Gradient clipping
        grad = np.clip(grad, -1.0, 1.0)  
        
        # Backtracking line search
        h, alpha = backtracking_line_search(h, grad)
        
        # Ensure sum constraint
        h = project_h(h)
    
    # Final best result
    best_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
