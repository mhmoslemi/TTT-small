import numpy as np
import time
from scipy.signal import correlate

def evaluate_solution(h, dx):
    """Evaluate C5 for given h and dx"""
    # Compute all overlaps
    overlaps = correlate(h, 1 - h, mode='full')
    # Max overlap scaled by dx
    return np.max(overlaps) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradients(h, dx):
    """Compute gradient for current solution"""
    # Cross-correlate h and (1-h)
    corr_h = correlate(h, 1 - h, mode='full')
    # Identify dominant index
    max_idx = np.argmax(corr_h)
    # Compute gradient for each position
    grad = np.zeros_like(h)
    # For lag k at max_idx, we have contribution from h[i] * (1 - h[i+k])
    # Gradient w.r.t. h[i] is (1 - h[i+k]) 
    # Gradient w.r.t. h[i+k] is -h[i]
    # Compute sparse gradients
    for i in range(len(h)):
        # CC is symmetric; just check contributions near max_idx
        start_idx = max(0, max_idx - len(h) + 1)
        end_idx = min(len(h), max_idx + 1)
        for k in range(start_idx, end_idx):
            if max_idx == k + len(h) - 1:  # corresponds to i+k in cross-correlation
                # h[i] contributes (1 - h[i+k]), gradient is 1 - h[i+k] 
                grad[i] += (1 - h[i + k - len(h) + 1]) * dx
    return grad
