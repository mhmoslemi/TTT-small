import numpy as np
from scipy import optimize
import time
import copy

def evaluate_erdos_solution(h_values):
    """From the problem definition: computes C5 using cross-correlation"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # np.correlate with mode='full' gives all shifts, we want max overlap
    # Compute cross-correlation of h with (1-h)
    h = np.array(h_values)
    h_rev = h[::-1]
    ones_minus_h = 1 - h
    correlation = np.correlate(ones_minus_h, h_rev, mode='full')
    # The full convolution includes all lags from -n+1 to n-1
    # The maximum overlap corresponds to the maximum value of the correlation
    max_overlap = np.max(correlation)
    c5_bound = max_overlap * dx
    return c5_bound

def project_h(h, n_points):
    """Ensure h is feasible: all values in [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full(n_points, 1.0 / 2)
    return h

def create_initial_h(n_points):
    """Initialize h from the given initial_h_values, or use a sinusoidal function"""
    try:
        # Try to use the global: initial_h_values
        h = np.array(initial_h_values)
        if len(h) == n_points:
            return project_h(h, n_points)
        else:
            # Interpolate or pad
            x = np.linspace(0, 2, len(initial_h_values))
            x_new = np.linspace(0, 2, n_points)
            h_new = np.interp(x_new, x, initial_h_values)
            return project_h(h_new, n_points)
    except NameError:
        # No initial_h_values, create a sinusoid
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 2.0))
        return project_h(h, n_points)

def ga_optimize(budget_s):
    """Run genetic algorithm to find a good solution"""
    n_points = 45
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism = int(0.2 * pop_size)
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = create_initial_h(n_points)
        h = np.random.rand(n_points) * 0.05 + h * 0.95  # slight noise
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    for gen in range(generations):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append((c5, h))
            
        # Sort by fitness
        fitness.sort()
        
        # Update best
        if fitness[0][0] < best_c5:
            best_c5 = fitness[0][0]
            best_h = fitness[0][1].copy()
        
        # Select top individuals
        selected = [h for _, h in fitness[:elitism]]
        
        # Generate offspring
        new_pop = selected.copy()
        while len(new_pop) < pop_size:
            a, b = np.random.choice(selected, 2, replace=False)
            
            if np.random.rand() < crossover_prob:
                # Uniform crossover
                mask = np.random.rand(n_points) > 0.5
                child = np.where(mask, a, b)
            else:
                child = a.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                idx = np.random.randint(0, n_points)
                child[idx] = np.random.rand()
            
            child = project_h(child, n_points)
            new_pop.append(child)
            
        population = new_pop[:pop_size]
    
    return best_h, best_c5

def gradient_descent_optimize(h_start, budget_s):
    """Use gradient descent on the worst overlap shift"""
    n_points = len(h_start)
    h = h_start.copy()
    learning_rate = 0.01
    max_iter = 10000
    patience = 2000
    
    start_time = time.time()
    last_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    best_c5 = last_c5
    no_improve_count = 0
    
    # Compute the shift with maximum overlap to optimize
    # This is a model: maximize overlap, so we maximize correlate(h, 1-h)
    def compute_overlap_shifts(h):
        # Use cross-correlation
        h_rev = h[::-1]
        ones_minus_h = 1 - h
        correlation = np.correlate(ones_minus_h, h_rev, mode='full')
        max_i = np.argmax(correlation)
        # max_i corresponds to offset from -n+1 to n-1
        shift = max_i - (n_points - 1)
        return shift
    
    # Optimize in a loop
    for i in range(max_iter):
        if time.time() - start_time > 0.3 * budget_s:
            break
            
        # Find the shift that gives maximum overlap
        shift = compute_overlap_shifts(h)
        
        # Compute gradient of C5 w.r.t. h at this shift
        # Small numerical derivative is more stable
        eps = 1e-5
        grad = np.zeros_like(h)
        base_c5 = evaluate_erdos_solution(h)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_erdos_solution(h_plus)
            grad[j] = (c5_plus - base_c5) / eps
        
        # Update with gradient
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_erdos_solution(h_new)
        if new_c5 < last_c5:
            h = h_new
            last_c5 = new_c5
            no_improve_count = 0
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        else:
            no_improve_count += 1
            # Reduce step size
            if no_improve_count % 1000 == 0:
                learning_rate *= 0.5
                if learning_rate < 1e-6:
                    break
                
        if no_improve_count > patience:
            break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Phase 1: Genetic Algorithm
    ga_h, ga_c5 = ga_optimize(budget_s)
    
    # Phase 2: Local Optimization
    gd_h, gd_c5 = gradient_descent_optimize(ga_h, budget_s)
    
    # Final check
    final_c5 = evaluate_erdos_solution(gd_h)
    
    # Return the best one
    if final_c5 < ga_c5 and final_c5 < gd_c5:
        h_return = gd_h
        c5_return = final_c5
    elif ga_c5 < gd_c5:
        h_return = ga_h
        c5_return = ga_c5
    else:
        h_return = gd_h
        c5_return = gd_c5
    
    return (h_return, c5_return, len(h_return))
