import numpy as np
from scipy import signal
import time

def evaluate_c5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h_to_feasible(h, n_points):
    """Project h to satisfy constraints"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm_phase(h0, n_points, pop_size, budget_s, elapsed_time):
    """Genetic Algorithm Phase"""
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time) - elapsed_time
    
    # Population initialization
    population = []
    for _ in range(pop_size):
        h = np.random.rand(n_points)
        h = project_h_to_feasible(h, n_points)
        population.append(h)
    
    # Add initial solution if available
    if h0 is not None:
        population[0] = h0.copy()
    
    best_h = h0.copy() if h0 is not None else population[0].copy()
    best_c5 = evaluate_c5(best_h)
    
    generation = 0
    while time_left() > 5.0:  # Leave time for gradient descent
        # Evaluate population
        fitness = [evaluate_c5(h) for h in population]
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Elitism - keep top 20%
        elite_size = max(1, int(0.2 * pop_size))
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        
        while len(new_population) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(len(elite), 2, replace=False)
            parent1 = elite[idx1]
            parent2 = elite[idx2]
            
            # Crossover
            if np.random.rand() < 0.7:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutate children
                for child in [child1, child2]:
                    for i in range(n_points):
                        if np.random.rand() < 0.1:
                            child[i] = np.random.rand()
                    if len(new_population) < pop_size:
                        child = project_h_to_feasible(child, n_points)
                        new_population.append(child)
            else:
                # Clone without crossover
                child = parent1.copy()
                for i in range(n_points):
                    if np.random.rand() < 0.1:
                        child[i] = np.random.rand()
                child = project_h_to_feasible(child, n_points)
                if len(new_population) < pop_size:
                    new_population.append(child)
        
        population = new_population[:pop_size]
        generation += 1
    
    return best_h, best_c5, time.time() - start_time

def gradient_descent_phase(h0, n_points, budget_s, elapsed_time):
    """Gradient Descent Phase"""
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time) - elapsed_time
    
    # Work with a copy of the starting h
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    # Learning parameters
    learning_rate = 0.01
    lr_decay = 0.5
    patience = 1000
    patience_counter = 0
    
    iteration = 0
    last_best_c5 = best_c5
    
    while time_left() > 1.0 and patience_counter < patience:
        # Compute all overlaps to find worst one
        corr = signal.correlate(h, 1 - h, mode='full')
        # Find which shift is producing the maximum overlap
        max_idx = np.argmax(corr)
        
        # Approximate gradient of overlap with shift = max_idx - n_points + 1
        shift = max_idx - n_points + 1
        if shift >= 0:
            # h[i] affects overlap with (i+shift) through h[i] * (1 - h[i+shift])
            grad = np.zeros(n_points)
            for i in range(n_points):
                if i + shift < n_points:
                    grad[i] += (1 - h[i + shift])  # Positive contribution from h[i]
                    
            # Now for backward pass: this affects overlap with (i-shift) 
            # by (1-h[i]) * h[i-shift], which gives -h[i-shift] contribution to grad[i]
            for i in range(shift, n_points):
                grad[i] += -h[i - shift]
        else:  # shift < 0
            # Index in the other direction
            shift = -shift  # absolute value
            # h[i] affects (i-shift) shift places back
            grad = np.zeros(n_points)
            for i in range(shift, n_points):
                grad[i] += (1 - h[i - shift])
            for i in range(shift):
                grad[i] += -h[i + shift]
        
        # Apply gradient
        h_new = h - learning_rate * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        # Scale to correct sum
        current_sum = np.sum(h_new)
        if current_sum != 0:
            h_new = h_new * (n_points / 2) / current_sum
        
        # Evaluate
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            patience_counter = 0  # Reset
        else:
            patience_counter += 1
            
        h = h_new.copy()
        iteration += 1
        
        # Adjust learning rate if no progress
        if patience_counter % 1000 == 0 and patience_counter > 0:
            learning_rate *= lr_decay
    
    return best_h, best_c5
