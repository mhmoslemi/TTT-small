import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    corr = correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (from lag -(n-1) to n-1)
    # The middle part corresponds to lags from 0 to n-1 and -n+1 to -1
    # For this problem we consider all non-negative lags explicitly
    # But actually, for overlap integral, we consider all nonzero lags.
    # The full correlation is of length 2*n - 1, middle index n-1 corresponds to lag=0
    # So we want for lags k: 0 to n-1, -n+1 to -1, so valid center indices are range n-1 to n-1+2*(n-1)
    # But in our case the useful indices correspond to appropriate lags
    # i-th element corresponds to k = i - (n-1) since center of correlation is at index n-1
    # So valid indices are n-1 to 2*n-2 and 0 to n-2
    # For our purposes max over all non-zero lags, which corresponds to indices 0 to n-2 and n to 2*n-2
    # Get all non-zero lags using slice valid_indices = [0..n-2] + [n..2*n-2]
    # But note that overlap integral requires: k = -(n-1) to -(1) and k=1 to n-1
    # But we can think of our correlation indexed i corresponding to k = i - (n-1)
    valid_indices = list(range(0, n-1)) + list(range(n, 2*n-1))
    overlaps = corr[valid_indices] * dx
    # Return max overlap
    return np.max(overlaps)

def initialize_h(n_points):
    """Initialize h with either provided initial_h_values or create simple construction"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        if len(h) != n_points:
            # Rescale to fit length
            h = np.interp(np.linspace(0, 1, n_points), np.linspace(0, 1, len(h)), h)
    else:
        # Use a simple random construction
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2.0)
    # Clip to valid range
    h = np.clip(h, 0, 1)
    return h

def project_to_feasible(h, dx):
    """Ensure sum=h_sum and all values are in [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if abs(h_sum) < 1e-10:
        h = np.ones_like(h) * dx * 0.5
    else:
        h = h * (0.5 / h_sum)
    return h

def solve_erdos():
    # Parameters
    n_points = 100
    max_iter = 5000
    learning_rate = 0.1
    tol = 1e-6
    start_time = time.time()
    
    # Initialize h
    h = initialize_h(n_points)
    h = project_to_feasible(h, 2.0 / n_points)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > 590:
            break

        # Compute all the overlaps for all lag values
        # h is length n, computing cross-correlation with (1-h) gives series of length 2n-1
        corr = correlate(h, 1 - h, mode='full')
        dx = 2.0 / n_points
        
        # The lag k corresponds to i = k + n-1 in the full correlation array
        # So to get the k=1 is at index n, k=2 at n+1, ..., k=n-1 at 2n-2
        # Similarly, k=-1 at n-2, ..., k=1-n at 0
        # Now only consider lags -n+1 to -1 and 1 to n-1, NOT 0
        # lag k corresponds to index k + n-1
        # Points we want to evaluate: [1..n-1] and [n+1..2n-2] when index 0 is k=-n+1
        # So lags = 1 to n-1, and -n+1 to -1 if indices = 1 to 2n-2  
        # We have full correlation from index 0 to 2n-2
        # Valid lag indices (k = -n+1 to -1 and 1 to n-1) are:
        lags = list(range(1, n_points)) + list(range(n_points + 1, 2*n_points - 1))
        overlaps = corr[lags] * dx
        max_k = lags[np.argmax(overlaps)] - (n_points - 1)  # Convert back to actual lag k

        # Evaluate gradient at max overlap, it is:
        # For x_i, the contribution to maximization is: for lag k_max, we get h_i * (1 - h_{i + k_max})
        # But here we want the derivative of sum h_i * (1 - h_{i+k}) with respect to h_i
        # Which is (1 - h_{i + k_max}) for each i where i + k_max < n
        # But total derivative with respect to h_i includes contributions from both i and i - k_max terms
        # The derivative is: d/dh_i sum_i' h_{i'} * (1 - h_{i'+k}) = (1 - h_{i + k_max}) 
        # But actually we're finding the maximizer, so:
        # We want to compute the gradient of the maximum (for one of the lags) w.r.t h.
        # The specific lag maximising the total overlap is known.
        # For k = k_max, derivative of h[i] term in that version is: 1 - h[(i + k_max) % n] (if k_max > 0)
        # For k = k_max=-1, hav	e contribution to h[i]: -h[i+1] if i+1 < n; and
        # Contribution to h[i+k]: if k=-1, then h[i-1] but considering modulo wrap-around (circularity),
        # Actually the right way to think about is gradient of h_i*(1-h_{i+k}) for some k is:
        # (1 - h_{i+k}) for h_i, and -h_i for h_{i+k}
        # But since we want to know what value of h that gives maximum function value among lagged versions:
        # If we fix a k and take derivative of h_i*(1-h_{i+k}) w.r.t h,
        # we get a derivative vector (k is fixed) of 1-h_{i+k} exactly at i.
        # But we're dealing with the maximum over all k and we are doing Frank Wolfe steps  
        # Let's think more carefully...
        # This is the gradient of the largest overlap term w.r.t h:
        # Since overlap = sum_i h_i*(1 - h_{i+k}) for some fixed k,
        # the gradient of that w.r.t h_i is 1-h_{i+k} if i+k < n, or 1-h_{(i+k)%n} if k<0,
        # or equivalently when index (i+k) wraps around, we use cyclic index:
        # gradient[i] = 1 - h[(i + k_max) % n] for lap k_max        
        # Note this is computing gradient symbolic, unsure if actually differentiating negative of max which is what we want
        gradient = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            gradient[i] = 1 - h[j]
        
        # Solve linear subproblem: minimize gradient^T h subject to h in feasible polytope
        # The linear subproblem is: minimize grad^T h  subject to 0 <= h[i] <= 1, sum h[i] = 0.5
        # Therefore, the solution sets h = argmin grad^T h, which is to choose with h[i]=0 if grad[i] > 0, 1 if < 0
        # But sum must equal 0.5 
        direction = np.zeros(n_points)
        # Choose the directions that reduce gradient, subject to constraints (x[i] in [0,1], sum = 0.5)
        sign_gradients = np.sign(gradient)
        # We want to solve: minimize grad^T h subject to h >= 0, h <= 1, sum h = 0.5
        # This is minimizing linear function with convex polytope.
        # By Frank-Wolfe setting, minimize gradient inner product,
        # We can set h[i] = 0 if gradient[i] > 0, 1 if gradient[i] < 0, clamp and adjust
        temp_h = np.where(gradient < 0, 1.0, 0.0)
        temp_sum = np.sum(temp_h)
        if temp_sum > 0.5:
            # We need to reduce value of h[i] where gradient[i] < 0
            # Find those indices that we only allow to iterate from 1 to less than 0.5 
            # Or in other words, want to reduce sum to equal 0.5
            sort_indices = np.argsort(gradient)
            cum_sum = 0
            for idx in sort_indices:
                if cum_sum + temp_h[idx] > 0.5:
                    # set it so that total = 0.5
                    temp_h[idx] = 0.5 - cum_sum
                    break
                else:
                    cum_sum += temp_h[idx]
        elif temp_sum < 0.5:
            # Increase where possible (gradient[i] > 0)
            sort_indices = np.argsort(-gradient)
            cum_sum = 0
            for idx in sort_indices:
                if cum_sum + temp_h[idx] < 0.5:
                    # toggle it from 0 to 1 or inflate as much as possible
                    temp_h[idx] = 1.0
                    cum_sum += 1.0
                else:
                    # now we are below 0.5, so set it to remainder
                    temp_h[idx] = 0.5 - cum_sum
                    break
                
        # This modified approach doesn't hit the equality constraint exactly.
        # Instead of creating infeasible solution directly, let's tone down the solution
        # Get direction from vertex of feasible set: minimize h^T grad over feasible set.
        direction = temp_h.copy()
        
        # This is actually not the simplest projected Frank-Wolfe:
        # Optimization step is to get h_new that minimizes grad^T h, 
        # subject to 0 <= h <=1 and sum h = 0.5
        # The minimum of such linear optimization is in a vertex of polytope.
        # So set h[i] = 1 if grad[i] < 0, h[i] = 0 if grad[i] > 0.
        # Normalize to meet sum constraint.
        # But for such approach to work, the solution can be found by:
        if np.sum(direction) == 0.5:
            pass
        elif np.sum(direction) > 0.5:
            # we need to reduce values so sum=0.5
            indices = np.argsort(-gradient)  # sort descending by -gradient: -gradient > 0 means location that decreases sum most?
            cur_sum = np.sum(direction)
            for i in indices:
                if cur_sum <= 0.5:
                    break
                diff = min(direction[i], cur_sum - 0.5)
                direction[i] -= diff
                cur_sum -= diff
        else:
            # we need to increase values so sum = 0.5
            indices = np.argsort(gradient)  
            cur_sum = np.sum(direction)
            for i in indices:
                if cur_sum >= 0.5:
                    break
                diff = min(1.0 - direction[i], 0.5 - cur_sum)
                direction[i] += diff
                cur_sum += diff
        # Ensure feasibility  
        direction = np.clip(direction, 0, 1)
        # Rescale to match sum = 0.5
        direction = direction * (0.5 / np.sum(direction))
        
        # Update via Frank-Wolfe step
        h_new = (1 - learning_rate) * h + learning_rate * direction
        h_new = project_to_feasible(h_new, 2.0 / n_points)
        
        # Evaluate current
        c5 = evaluate_h(h_new)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            if best_c5 < 0.3807:
                break
        
        h = h_new

    return best_h, best_c5

# Global definition for harness
def run(seed=42, budget_s=600, **kwargs):
    """Main run function for optimization"""
    np.random.seed(seed)
    h, c5 = solve_erdos()
    return (h.tolist(), c5, len(h))
