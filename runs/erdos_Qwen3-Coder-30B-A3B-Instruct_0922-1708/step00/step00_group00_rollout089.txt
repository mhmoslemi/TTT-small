import numpy as np
import time
from scipy import signal

def evaluate(h, dx):
    # Compute the overlap integral for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral for shift k is corr[n-1+k] * dx
    # We only consider shifts from -n+1 to n-1
    n = len(h)
    overlaps = np.array([corr[n-1+k] * dx for k in range(-n+1, n)])
    return np.max(overlaps)

def enforce_constraints(h, n_points):
    # Ensure h[i] ∈ [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to ensure sum(h) * dx = 1 (i.e. sum(h) = n_points/2)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        # Linear adjustment
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        # Re-clip to [0,1]
        h = np.clip(h, 0, 1)
        # If adjustment pushed some values out, do a more careful adjustment
        final_sum = np.sum(h)
        if abs(final_sum - target_sum) > 1e-6:
            # Simple method: scale and reclip
            scale = target_sum / final_sum
            h = h * scale
            h = np.clip(h, 0, 1)
    return h

def mutate(h, rate=0.05):
    # Mutate by adding small random numbers
    mu = 0.0
    sigma = 0.02
    # Select random indices to mutate
    n = len(h)
    n_mutate = max(1, int(rate * n))
    indices = np.random.choice(n, n_mutate, replace=False)
    delta = np.random.normal(mu, sigma, n_mutate)
    h[indices] += delta
    # Enforce constraints after mutation
    return enforce_constraints(h, n)

def crossover(parent1, parent2):
    # Single point crossover
    n = len(parent1)
    point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def tournament_select(population, fitnesses, tournament_size=5):
    # Select index of best individual in tournament
    indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in indices]
    winner_idx = indices[np.argmin(tournament_fitnesses)]
    return winner_idx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    pop_size = 100
    n_generations = 120
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = pop_size // 5  # Top 20%
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = enforce_constraints(initial_h, n_points)
        population.append(initial_h)
        # Fill remaining with random individuals
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = enforce_constraints(h, n_points)
            population.append(h)
    else:
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = enforce_constraints(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    start_time = time.time()
    fitnesses = [evaluate(h, dx) for h in population]
    
    best_individual = population[np.argmin(fitnesses)]
    best_fitness = min(fitnesses)
    
    # Start optimization
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while generation < n_generations:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        sorted_pop = [population[i] for i in sorted_indices]
        sorted_fit = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep top individuals
        elite = sorted_pop[:elite_size]
        offspring = elite[:]
        
        # Tournament selection and evolutionary operations
        while len(offspring) < pop_size:
            # Parent selection
            parent1_idx = tournament_select(sorted_pop, sorted_fit)
            parent2_idx = tournament_select(sorted_pop, sorted_fit)
            parent1 = sorted_pop[parent1_idx]
            parent2 = sorted_pop[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            offspring.append(child1)
            if len(offspring) < pop_size:
                offspring.append(child2)
        
        # Update population
        population = offspring
        # Re-evaluate fitnesses
        fitnesses = [evaluate(h, dx) for h in population]
        
        # Track best individual
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count >= max_no_improvement:
                break
        
        generation += 1
    
    # Final evaluation
    final_fitness = evaluate(best_individual, dx)
    
    return (best_individual.tolist(), final_fitness, n_points)
