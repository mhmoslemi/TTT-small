import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points, dtype=float) * 0.5
        
    # Ensure sum equals n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_C = evaluate_C(h, dx)
    current_C = best_C
    eta = 0.01
    no_improvement_count = 0
    
    for iteration in range(6000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode="full")
        # Shift to match the actual lags
        lags = np.arange(-n_points + 1, n_points)
        max_k = lags[np.argmax(overlaps)]
        max_over = np.max(overlaps)
        
        # Create gradient for this specific shift
        grad = np.zeros(n_points)
        if max_k >= 0:
            # h[i] * (1 - h[i + max_k])
            grad[:n_points - max_k] = 1 - h[max_k:]
        else:
            # -h[i - max_k] * (1 - h[i])
            grad[-max_k:] = -h[:-max_k]
        
        # Update h
        h_new = h - eta * grad
        
        # Clip and recenter
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        new_C = evaluate_C(h_new, dx)
        
        # Accept or reject update
        if new_C < current_C:
            h = h_new
            current_C = new_C
            if new_C < best_C:
                best_C = new_C
                best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Adaptive learning rate
        if iteration % 100 == 0 or no_improvement_count > 50:
            eta *= 0.95
            
        if no_improvement_count > 100:
            break
            
    return (best_h.tolist(), best_C, n_points)

def evaluate_C(h, dx):
    # Use same method as in the problem description
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx
