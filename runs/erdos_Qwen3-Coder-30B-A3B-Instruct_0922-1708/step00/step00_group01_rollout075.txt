import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction or create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Symmetric initialization
        n = 47
        h = np.ones(n) * 0.5
        mid = n // 2
        h[mid-5:mid+5] = 0.7
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Initialization
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    start_time = time.time()
    
    # SGD parameters
    lr = 0.1
    decay_every = 1000
    max_iter = 5000
    current_time = start_time
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradients
        corr = correlate(h, 1 - h, mode='full')
        lag = np.argmax(corr)
        max_val = corr[lag]
        
        # Compute gradient with respect to h[i]
        grad = np.zeros_like(h)
        for j in range(n_points):
            # derivative of h[i] * (1 - h[i + (lag - n_points + 1)])
            off = lag - (n_points - 1) + j
            if 0 <= off < n_points:
                grad[j] += (1 - h[off]) * dx
            # derivative of h[off] * (1 - h[j])
            if 0 <= j < n_points:
                grad[j] -= h[off] * dx
            
        # SGD update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Project to sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        c5 = compute_c5(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Annealing
        if i % decay_every == 0 and i > 0:
            lr *= 0.5
            
    return (best_h.tolist(), best_c5, n_points)


def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
