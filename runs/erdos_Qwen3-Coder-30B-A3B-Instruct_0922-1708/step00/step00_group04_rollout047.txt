import cvxpy as cp
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    n = len(h_values)
    dx = 2.0 / n
    h = np.array(h_values)
    # Compute overlap integrals for all lags
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral for lag k is corr[n-1+k] * dx
    max_overlap = np.max(corr) * dx
    return max_overlap
