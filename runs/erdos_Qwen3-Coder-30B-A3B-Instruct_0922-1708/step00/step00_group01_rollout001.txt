import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction or create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create symmetric construction with plateau in middle
        n = 47
        h = np.ones(n) * 0.5
        # Make it symmetric
        mid = n // 2
        h[:mid] = 0.3
        h[mid:] = 0.7
        h = h / np.sum(h) * (n / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = None
    
    # Annealing parameters
    lr = 0.1
    anneal_every = 1000
    max_iter = 5000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps with cross-correlation
        conv = signal.correlate(h, 1 - h, mode='full')
        c5_value = np.max(conv) * dx
        
        if best_c5 is None or c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()
        
        # Find dominant lag
        max_idx = np.argmax(conv)
        dominant_lag = max_idx - (n_points - 1)
        
        # Compute gradient in terms of dominant lag
        grad = np.zeros(n_points)
        
        # Compute gradients for overlapping terms
        for j in range(n_points):
            if j + dominant_lag >= 0 and j + dominant_lag < n_points:
                grad[j] = (1 - h[j + dominant_lag]) - h[j]
        
        # Update step
        h_new = h - lr * grad
        
        # Project to feasibility
        h_new = np.clip(h_new, 0, 1)
        # Rescale to integral = 1
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        h = h_new
        
        # Anneal learning rate
        if (i + 1) % anneal_every == 0:
            lr *= 0.5
    
    # Final evaluation
    conv = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(conv) * dx
    
    return (best_h.tolist(), final_c5, n_points)
