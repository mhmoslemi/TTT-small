import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create a baseline
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with constant function
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.3
        h[3] = 0.7
        h[4] = 0.2
        h[5] = 0.8
        h[6] = 0.1
        h[7] = 0.9
        h[8] = 0.05
        h[9] = 0.95

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Force the sum to be correct
    h_sum = np.sum(h)
    if abs(h_sum - target_sum) > 1e-10:
        h = h * target_sum / h_sum

    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    start_time = time.time()
    
    # Gradient descent approach
    eta = 0.01
    patience = 0
    last_improvement = 0
    
    for iteration in range(6000):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations for max lag
        corr = np.correlate(h, 1-h, mode='full')
        k_max = np.argmax(corr)
        max_shift = k_max - (n_points - 1)
        max_val = corr[k_max]
        
        # Compute gradient for this max shift
        grad = np.zeros(n_points)
        h_shifted = np.roll(h, -max_shift)
        if max_shift >= 0:
            grad[:n_points-max_shift] = h_shifted[:n_points-max_shift] * dx
        else:
            grad[-max_shift:] = -(h[:n_points+max_shift]) * dx
        
        # Apply gradient update
        h_new = h - eta * grad
        
        # Clip h to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum
        h_sum_new = np.sum(h_new)
        h_new = h_new * target_sum / h_sum_new
        
        # Evaluate new solution
        c5_new = evaluate_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = iteration
            patience = 0
        else:
            patience += 1
            
        h = h_new.copy()
        
        # Adaptive learning rate
        if iteration % 100 == 0 or patience > 50:
            if patience > 50:
                eta *= 0.95
            patience = 0
            
        if patience > 100:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # This matches the expected evaluation method from the problem description
    corr = np.correlate(h, 1-h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx
