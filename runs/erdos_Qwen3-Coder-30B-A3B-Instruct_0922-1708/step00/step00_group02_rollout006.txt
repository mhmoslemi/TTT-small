import numpy as np
import time
from scipy import optimize

def evaluate_overlap(h):
    """Compute C5 = max_k int h(x)(1 - h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for all shifts
    conv = np.correlate(h, 1 - h, mode='full')
    # Take the maximum among all shifts
    max_overlap = np.max(conv)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial values if available

        h[0] = 0.5
        h[-1] = 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute some quantities
    dx = 2.0 / n_points
    
    # Optimization parameters
    max_iter = 10000
    step_size = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    no_improve_count = 0
    
    # Gradient-based optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        conv = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(conv)
        max_overlap = conv[max_k]
        
        # Compute gradient w.r.t. shift k_max
        grad = np.zeros_like(h)
        k_max = max_k - (n_points - 1)
        for i in range(n_points):
            if 0 <= i + k_max < n_points:
                grad[i] += (1 - h[i + k_max])
            if 0 <= i - k_max < n_points:
                grad[i] -= h[i - k_max]
        
        # Update h
        h = h - step_size * grad
        # Project h into [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to maintain integral constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new C5
        new_c5 = evaluate_overlap(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > patience:
            # Reduce step size and reset patience
            step_size *= 0.5
            no_improve_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
