import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    max_iter = 100000
    alpha = 0.01
    patience = 100
    patience_counter = 0
    best_C5 = float('inf')
    best_h = None
    
    # Get starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to have correct sum
    h = h * target_sum / np.sum(h)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlap_full = correlate(h_padded, h_reflected, mode='full')
        overlaps = overlap_full[n_points:] * dx
        
        # Find the maximum overlap and corresponding shift
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            # Reduce learning rate
            alpha *= 0.5
            patience_counter = 0
            
        # Compute gradient
        # For shift k_max, gradient w.r.t h[i] is:
        #   (1 - h[i+k_max]) - h[i] if i+k_max < n_points and i < n_points
        # But need to be careful with indices
        
        # Create gradient vector: this is modified by considering overlap at k_max only
        grad = np.zeros_like(h)
        k_max = max_idx
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max]) - h[i]
        
        # Update h
        h = h - alpha * grad
        
        # Project h to [0,1]^n
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * target_sum / current_sum
            
        # Check for convergence
        if abs(alpha) < 1e-8:
            break
            
    # Final verification
    if best_h is not None:
        h = best_h
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlap_full = correlate(h_padded, h_reflected, mode='full')
        overlaps = overlap_full[n_points:] * dx
        final_C5 = np.max(overlaps)
    else:
        # Fallback in case nothing was found
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlap_full = correlate(h_padded, h_reflected, mode='full')
        overlaps = overlap_full[n_points:] * dx
        final_C5 = np.max(overlaps)
        
    return (h.tolist(), final_C5, n_points)
