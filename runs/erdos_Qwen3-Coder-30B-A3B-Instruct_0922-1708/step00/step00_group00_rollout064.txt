import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters  
    n_points = 45
    dx = 2.0 / n_points
    population_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    tournament_size = 5
    elite_size = population_size // 5
    max_no_improvement = 100
    
    # Initialize population
    def initialize_individual():
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2) 
        h = np.clip(h, 0, 1)
        return h
    
    def evaluate_fitness(h):
        # Compute overlap for all shifts using correlation
        overlap = correlate(h, 1 - h, mode='full') * dx
        # The overlap values correspond to shifts from -(n-1) to (n-1)
        # We only consider positive shifts (k in [1, n-1]) 
        # but that's not exactly the task, let us find max over all valid shifts
        # actually, task asks max over all k, which relates to cross-correlation
        # the max over full correlation output
        return np.max(overlap)
    
    # If initial solution exists use it
    if 'initial_h_values' in globals():
        population = [np.array(initial_h_values)]
        population.extend([initialize_individual() for _ in range(population_size - 1)])
    else:
        population = [initialize_individual() for _ in range(population_size)]
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Selection with tournament
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size)
            winner_idx = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[winner_idx])
        
        # Elitism (keep best)
        elite_indices = np.argsort(fitnesses)[:elite_size]
        new_population = [population[i] for i in elite_indices]
        
        # Create offspring
        while len(new_population) < population_size:
            if np.random.rand() < crossover_prob and len(selected) >= 2:
                parent1, parent2 = np.random.choice(selected, 2, replace=False)
                
                # Single point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutate
                for child in [child1, child2]:
                    for i in range(n_points):
                        if np.random.rand() < mutation_rate:
                            child[i] += np.random.normal(0, 0.05)
                    # Ensure in [0,1]
                    child = np.clip(child, 0, 1)
                    # Normalize to sum = n_points/2
                    child = child / np.sum(child) * (n_points / 2)
                    new_population.append(child)
            else:
                # Just mutate parent
                parent = np.random.choice(selected)
                child = parent.copy()
                for i in range(n_points):
                    if np.random.rand() < mutation_rate:
                        child[i] += np.random.normal(0, 0.05)
                # Ensure in [0,1]
                child = np.clip(child, 0, 1)
                # Normalize to sum = n_points/2
                child = child / np.sum(child) * (n_points / 2)
                new_population.append(child)
                
        population = new_population[:population_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        actual_c5 = evaluate_fitness(best_h)
        return (best_h.tolist(), actual_c5, n_points)
    else:
        # If no good solution, just return the last best one from initial population
        fitnesses = [evaluate_fitness(h) for h in population[:10]]
        min_fitness_idx = np.argmin(fitnesses)
        actual_c5 = fitnesses[min_fitness_idx]
        return (population[min_fitness_idx].tolist(), actual_c5, n_points)
