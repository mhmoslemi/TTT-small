import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    # Note: This replicates behavior of the harness's version
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlate
    overlaps = correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (from index n-1 to 2*n-2)
    valid_overlaps = overlaps[n-1:]
    C5 = np.max(valid_overlaps) * dx
    return C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial_h_values if available
    try:
        h = initial_h_values.copy()
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    except:
        h = np.ones(n_points) * 0.5
        
    # Ensure sum == n_points / 2 after scaling
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        h = h * (n_points / 2) / current_sum

    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h)
    
    # EM optimization parameters
    learning_rate = 0.1
    patience = 50
    patience_counter = 0
    
    # Iteration loop
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # E-step: compute all overlaps
        overlaps = correlate(h, 1 - h, mode='full')
        valid_overlaps = overlaps[n_points-1:]
        
        # Normalize to form a probability over lags
        total_overlap = np.sum(valid_overlaps)
        if total_overlap > 0:
            p = valid_overlaps / total_overlap
        else:
            p = np.ones_like(valid_overlaps) / len(valid_overlaps)
            
        # Compute gradients (partial derivatives of overlap w.r.t. h[i])
        n = len(h)
        grad_overlaps = np.zeros((n, n))
        for k in range(n):
            # derivative of overlap_k w.r.t. h[i]
            for i in range(n):
                j = (i + k) % n
                grad_overlaps[i, k] = -(1 - h[j])  # derivative via chain rule
        
        # M-step
        new_h = h.copy()
        for i in range(n):
            grad_i = np.sum(p * grad_overlaps[i, :])  # weighted gradient
            new_h[i] -= learning_rate * grad_i
            
        # Clip values to [0, 1]
        new_h = np.clip(new_h, 0, 1)
        
        # Rescale to maintain integral = 1
        new_sum = np.sum(new_h)
        if abs(new_sum) > 1e-10:
            new_h = new_h * (n_points/2) / new_sum
            
        # Update h
        h = new_h
        
        # Evaluate
        C5 = evaluate_erdos_solution(h)
        
        # Check improvement and update best
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            learning_rate *= 0.5
            patience_counter = 0

    return (best_h.tolist(), best_C5, n_points)
