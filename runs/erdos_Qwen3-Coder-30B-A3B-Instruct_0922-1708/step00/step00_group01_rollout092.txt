import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for a given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 100
        h0 = np.random.rand(n_points)
        h0 = project_h(h0, n_points)
    
    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    inertia = w
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Initialize with h0 and perturbed versions
    for i in range(n_particles):
        if i == 0:
            h = h0.copy()
        else:
            h = h0 + np.random.normal(0, 0.05, n_points)
            h = project_h(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, 2.0 / n_points))
    
    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (pbest[i] - h)
            social = c2 * r2 * (gbest - h)
            v = inertia * v + cognitive + social
            
            # Update position
            h_new = h + v
            h_new = project_h(h_new, n_points)
            
            # Evaluate new solution
            new_val = evaluate_h(h_new, 2.0 / n_points)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = h_new.copy()
                pbest_vals[i] = new_val
                if new_val < gbest_val:
                    gbest = h_new.copy()
                    gbest_val = new_val
                    last_improvement = iteration
            
            # Update particle
            particles[i] = h_new
            velocities[i] = v
            
        # Diversity injection if no improvement
        if iteration - last_improvement > 25:
            for i in range(n_particles // 5):
                j = np.random.randint(0, n_particles)
                particles[j] = np.random.rand(n_points)
                particles[j] = project_h(particles[j], n_points)
                velocities[j] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[j] = particles[j].copy()
                pbest_vals[j] = evaluate_h(particles[j], 2.0 / n_points)
                if pbest_vals[j] < gbest_val:
                    gbest = pbest[j].copy()
                    gbest_val = pbest_vals[j]
                    last_improvement = iteration
        
        # Decrease inertia slightly to encourage convergence
        inertia = 0.7 + (0.3 * iteration / max_iter)
    
    return (gbest.tolist(), gbest_val, n_points)
