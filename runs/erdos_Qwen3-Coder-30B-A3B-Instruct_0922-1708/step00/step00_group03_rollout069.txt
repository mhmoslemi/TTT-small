import numpy as np
from scipy import optimize
import time

    start_time = time.time()

    # Normalize to satisfy integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute indices for efficient overlap computation
    indices = np.arange(n_points)
    
    # Helper to compute current C5 bound
    def compute_c5(h):
        # Compute overlaps for all lags
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            shift = k - (n_points - 1)
            if shift >= 0:
                overlaps[k] = np.sum(h * np.roll(1 - h, shift))
            else:
                overlaps[k] = np.sum(h[-shift:] * (1 - h[:n_points + shift]))
        return np.max(overlaps) * dx
    
    # Helper to project h to feasible set
    def project_h(h):
        h = np.clip(h, 0, 1)
        # Maintain sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Optimizer loop
    best_c5 = compute_c5(h)
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    eta = 0.1
    
    while time.time() - start_time < budget_s - 1 and iteration < 5000:
        # Compute current overlaps
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            shift = k - (n_points - 1)
            if shift >= 0:
                overlaps[k] = np.sum(h * np.roll(1 - h, shift))
            else:
                overlaps[k] = np.sum(h[-shift:] * (1 - h[:n_points + shift]))
        
        # Convert to probabilities (normalize)
        prob_overlaps = overlaps / np.sum(overlaps)
        
        # Compute gradient of overlaps w.r.t. h
        # For each point i, gradient is sum over k of prob_overlaps[k] * grad(h[i],k)
        # grad(h[i],k) = (1 - h[(i - k) % n]) if k is positive, -h[(i + k) % n] if k is negative
        
        grad_h = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            shift = k - (n_points - 1)
            prob = prob_overlaps[k]
            if shift == 0:
                # This case doesn't contribute to gradient
                continue
            elif shift > 0:
                # Shift positive: h[i] contributes to overlap with lag k
                # gradient_diff = -h[(i - shift) % n] 
                for i in range(n_points):
                    grad_h[i] += prob * (-h[(i - shift) % n_points])
            else:
                # shift negative: h[(i - shift) % n] contributes to lag k
                # gradient_diff = (1 - h[(i + shift) % n])
                for i in range(n_points):
                    grad_h[i] += prob * (1 - h[(i - shift) % n_points])
        
        # Update h
        h_new = h - eta * grad_h
        h_new = project_h(h_new)
        
        # Update eta
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = iteration
            eta = min(0.1, eta * 1.2)  # Increase slightly
        else:
            eta *= 0.9  # Reduce learning rate
            
        h = h_new
        iteration += 1
        
        # Early stopping
        if iteration - last_improvement > 100:
            break
            
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (list(best_h), final_c5, n_points)
