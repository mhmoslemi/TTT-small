import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all overlaps for lags k = 0, ..., 2*n-2
    # Using correlation between h and (1-h)
    h = np.array(h_values)
    inv_h = 1 - h
    # Compute correlation with full mode
    corr = correlate(h, inv_h, mode='full')
    # The overlap integral for lag k is the correlation at that lag times dx
    # We want max over all lags
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        # Ensure sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 0.5)

    # Ensure constraints
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # EM Optimization
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # learn_rate
    lr = 0.1
    patience = 0
    max_patience = 50
    
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # E-step: compute overlaps for all lags
        overlaps = []
        for k in range(2 * n_points - 1):
            # Compute overlap for this lag
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps.append(overlap * dx)
        
        # Normalize to probability-like distribution
        overlaps = np.array(overlaps)
        if np.sum(overlaps) > 0:
            overlaps = overlaps / np.sum(overlaps)
            
        # M-step: compute gradient and update h
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(2 * n_points - 1):
                j = (i + k) % n_points
                grad[i] += overlaps[k] * (1 - h[j])
                
        # Update h
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Perturb to avoid degeneracy
        noise_std = 0.01 * np.exp(-iteration / 100)
        h_new += np.random.normal(0, noise_std, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_sum = np.sum(h_new)
        h_new = h_new * (n_points / 2) / h_sum
        
        # Update h
        h = h_new
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Update learning rate
        if patience > 10:
            lr *= 0.9 if iteration > 20 else 0.99
            
        if patience >= max_patience:
            break
            
    # Final validation and return
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
