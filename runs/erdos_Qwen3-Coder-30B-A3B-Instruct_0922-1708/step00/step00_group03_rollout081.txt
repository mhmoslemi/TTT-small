import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Problem setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5 * (2 / n_points))
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Try to refine with SDP if time permits
    try:
        # Set a reasonable time limit for the SDP solver
        solver_timeout = min(300, budget_s - (time() - start_time) - 5)
        if solver_timeout > 0:
            # Build SDP problem
            X = cvx.Variable(n_points)
            Z = cvx.Variable((n_points, n_points))
            
            # Constraints
            constraints = [
                X >= 0,
                X <= 1,
                cvx.sum(X) == n_points / 2
            ]
            
            # Add coupling constraints
            for i in range(n_points):
                for j in range(n_points):
                    constraints += [
                        Z[i,j] <= X[i],
                        Z[i,j] <= 1 - X[j],
                        Z[i,j] >= X[i] + X[j] - 1
                    ]
            
            # Objective: minimize max correlation
            C_values = []
            for k in range(-n_points+1, n_points):
                idx = np.arange(n_points) - k
                valid = (idx >= 0) & (idx < n_points)
                if np.any(valid):
                    sum_z = cvx.sum(Z[np.arange(n_points)[valid], idx[valid]])
                    C_values.append(sum_z)
            
            if C_values:
                max_C = cvx.max(cvx.hstack(C_values))  # Use hstack instead of vstack
                objective = cvx.Minimize(max_C)
                prob = cvx.Problem(objective, constraints)
                
                # Solve the problem
                prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=solver_timeout)
                
                if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
                    h_opt = np.clip(X.value, 0, 1)
                    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                    
                    # Evaluate this solution
                    c5_opt = evaluate_erdos_solution(h_opt)
                    if c5_opt < best_c5:
                        best_h = h_opt.copy()
                        best_c5 = c5_opt
    except Exception:
        pass
    
    # Final check
    best_c5 = evaluate_erdos_solution(best_h)
    return (best_h, best_c5, n_points)
