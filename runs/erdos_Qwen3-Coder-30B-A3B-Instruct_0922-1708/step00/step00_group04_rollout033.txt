        # log-sum-exp approximation to max
        return log(np.sum(np.exp(beta * x))) / beta

        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        corr_scaled = corr * dx
        return surrogate_max(corr_scaled, beta)
