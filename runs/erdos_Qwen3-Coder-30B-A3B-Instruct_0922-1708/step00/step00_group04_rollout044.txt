    # Optimization parameters
    max_iter = 600
    iter_count = 0
    no_improvement_count = 0
    learning_rate = 0.1
    tolerance = 1e-6
    
    def compute_overlap(h, k):
        """Compute the overlap at shift k"""
        n = len(h)
        if k >= 0:
            return np.sum(h[:n-k] * (1 - h[k:]))
        else:
            return np.sum(h[-k:] * (1 - h[:-k]))
    
    def compute_c5(h):
        """Compute max overlap with dx normalization"""
        n = len(h)
        dx = 2.0 / n
        max_overlap = 0
        for k in range(1-n, n):  # shifts from -(n-1) to n-1
            overlap = compute_overlap(h, k)
            max_overlap = max(max_overlap, overlap)
        return max_overlap * dx
    
    def project_h(h):
        """Project h to feasible region"""
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Scale to sum = n_points/2
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    # Iterative improvement
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        if no_improvement_count > 100:
            # Random perturbation
            h = project_h(h + 0.01 * np.random.randn(n_points))
            no_improvement_count = 0
            
        # Find worst shift
        max_overlap = 0
        worst_k = 0
        for k in range(1-n_points, n_points):
            overlap = compute_overlap(h, k)
            if overlap > max_overlap:
                max_overlap = overlap
                worst_k = k
                
        # Compute gradient w.r.t. worst shift
        # For overlap at shift k: sum h[i] * (1 - h[i+k])
        grad = np.zeros_like(h)
        if worst_k >= 0:
            grad[:n_points-worst_k] += h[worst_k:] - 1
            grad[worst_k:] += h[:n_points-worst_k] - 1
        else:
            grad[-worst_k:] += h[:-worst_k] - 1
            grad[:-worst_k] += h[-worst_k:] - 1
            
        # Update with gradient descent
        new_h = h - learning_rate * grad
        new_h = project_h(new_h)

        new_c5 = compute_c5(new_h)

            best_h = new_h.copy()

            no_improvement_count = 0

            no_improvement_count += 1
            learning_rate *= 0.5

        iter_count += 1
        
        # Early stopping

            break
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
