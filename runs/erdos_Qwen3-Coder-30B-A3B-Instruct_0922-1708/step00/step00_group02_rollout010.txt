import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    corr = correlate(h, 1 - h, mode='full')
    # Find maximum value
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h: use initial_h_values if available, otherwise create a sinusoidal pattern."""
    try:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
        return h
    except NameError:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 2))  # Sinusoidal pattern
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    crossover_point = np.random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1):
    """Add random noise to h with given mutation rate."""
    mutated = h.copy()
    n = len(h)
    indices = np.random.choice(n, size=int(n * mutation_rate), replace=False)
    for i in indices:
        mutated[i] += np.random.normal(0, 0.05)
    return mutated

def genetic_algorithm(n_points, pop_size=50, max_generations=200, budget_s=600):
    """Genetic Algorithm for optimizing h."""
    start_time = time.time()
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = initialize_h(n_points)
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > 0.7 * budget_s:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Elitism: keep top 20%
        elite_count = max(1, pop_size // 5)
        elites = population[:elite_count]
        
        # Create offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            # Tournament selection
            parent1_idx = np.random.choice(elite_count)
            parent2_idx = np.random.choice(elite_count)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < 0.7:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Project
            child1 = project_h(child1, n_points)
            child2 = project_h(child2, n_points)
            
            offspring.append(child1)
            if len(offspring) < pop_size - elite_count:
                offspring.append(child2)
                
        # Replace worst with offspring
        population = elites + offspring
        
    return best_h, best_c5

def gradient_descent(h, n_points, max_iter=1000, initial_lr=0.01, budget_s=600):
    """Gradient-based optimization."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    lr = initial_lr
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > 0.3 * budget_s:
            break
            
        # Approximate gradient using finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        base_c5 = evaluate_c5(h)
        updated_h = h.copy()
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            grad[i] = (c5_plus - base_c5) / eps
            
        # Update with gradient descent
        new_h = h - lr * grad
        
        # Project to feasible region
        new_h = project_h(new_h, n_points)
        
        new_c5 = evaluate_c5(new_h)
        
        # Accept if better
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = new_h.copy()
            last_improvement = iteration
        elif iteration - last_improvement > 500:
            lr *= 0.5
            last_improvement = iteration
            
        if iteration - last_improvement > 1000:
            break
            
        h = new_h
        
    return best_h, best_c5

def run(seed=42, budget_s=600, n_points=45):
    """Main function to run optimization."""
    np.random.seed(seed)
    start_time = time.time()
    
    # Phase 1: Genetic Algorithm
    ga_start = time.time()
    ga_h, ga_c5 = genetic_algorithm(n_points, budget_s=budget_s)
    
    # Phase 2: Gradient Descent
    gd_start = time.time()
    final_h, final_c5 = gradient_descent(ga_h, n_points, budget_s=budget_s)
    
    # Return best solution found
    if final_c5 < ga_c5:
        return (final_h, final_c5, n_points)
    else:
        return (ga_h, ga_c5, n_points)
