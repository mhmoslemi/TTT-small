import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: uniform distribution
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy sum constraint
    h = h * target_sum / np.sum(h)
    
    # Gradient descent parameters
    alpha = 0.01
    max_iter = 10000
    patience = 100
    last_improvement = 0
    best_C5 = float('inf')
    best_h = h.copy()
    
    def compute_overlap(h, k):
        """Compute overlap for shift k"""
        n = len(h)
        if k >= n or k <= -n:
            return 0.0
        
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i in range(n-k)
            overlap = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            # h[i+k] * (1 - h[i]) for i in range(-k, n)
            overlap = np.sum(h[-k:] * (1 - h[:-k]))
        return overlap
    
    def compute_all_overlaps(h):
        """Compute overlaps for all possible shifts"""
        n = len(h)
        overlaps = []
        for k in range(1 - n, n):
            overlap = compute_overlap(h, k)
            overlaps.append(overlap * dx)
        return np.array(overlaps)
    
    def get_gradient_for_shift(h, k):
        """Compute gradient w.r.t h for a specific shift k"""
        n = len(h)
        grad = np.zeros(n)
        
        if k >= 0:
            # grad[i] = (1 - h[i+k]) - h[i] for valid indices
            for i in range(n - k):
                grad[i] += (1 - h[i + k])
                grad[i + k] -= h[i]
        else:
            # k < 0
            for i in range(-k, n):
                grad[i] += (1 - h[i + k])
                grad[i + k] -= h[i]
        return grad * dx
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        overlaps = compute_all_overlaps(h)
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
        
        if max_overlap < 1e-10:
            break
        
        grad = get_gradient_for_shift(h, max_k - (n_points - 1))
        h = h - alpha * grad
        
        # Project into unit hypercube
        h = np.clip(h, 0, 1)
        
        # Rescale to enforce sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * target_sum / current_sum
        
        # Check for improvement
        if iteration > 0 and iteration % patience == 0:
            if abs(best_C5 - max_overlap) < 1e-6:
                break
        
        iteration += 1

    # One final validation
    final_overlaps = compute_all_overlaps(best_h)
    final_C5 = np.max(final_overlaps)
    
    return (best_h.tolist(), float(final_C5), n_points)
