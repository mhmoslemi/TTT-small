import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values or create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with a smooth periodic function
        n_points = 42
        h = np.ones(n_points)
        # Create a triangle wave
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.cos(np.pi * x))  # Triangle wave
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Precompute correlation
    def compute_C5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Normalize by dx
        corr = corr * dx
        return np.max(corr)
    
    # Gradient descent optimizer
    def optimize_step(h_vals, lr=0.001, max_iter=100, patience=10):
        best_h = h_vals.copy()
        best_C5 = compute_C5(h_vals)
        no_improvement = 0
        
        for _ in range(max_iter):
            # Correlation in frequency domain
            n_full = 2 * n_points - 1
            
            # Compute DFT of both signals
            H = np.fft.fft(h_vals)
            G = np.fft.fft(1 - h_vals)
            
            # Cross-correlation in frequency domain
            corr_freq = H * np.conj(G)
            corr_time = np.real(np.fft.ifft(corr_freq))
            
            # The actual corr_time is of length n_full, with main values in the middle
            corr_time = corr_time[:n_full]
            
            # Loop over all positions and create surrogate max
            # Using log-sum-exp with adjustable β
            beta = 10.0  # Temporarily fixed
            
            # Surrogate gradient approximation
            # The gradient of log(Σ exp(β * x)) is (exp(β * x) / Σ exp(β * x)) * β
            # For max, β -> ∞, so just pick the dominant term
            max_corr_idx = np.argmax(corr_time)
            max_corr_val = corr_time[max_corr_idx]
            
            # Approximate gradient at this position
            # Use finite diff for first order approximation (simplified)
            # Compute gradient of negative correlation with respect to h directly
            # We will compute derivs per element using finite differences
            grad = np.zeros_like(h_vals)
            eps = 1e-5
            
            # Gradient estimate at position giving max corr_val
            # Compute for each element (first order approx)
            for i in range(n_points):
                h_plus = h_vals.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (np.sum(h_vals) / np.sum(h_plus))  # Rescale
                corr_plus = compute_C5(h_plus)
                grad[i] = (corr_plus - best_C5) / eps
            
            # Update with gradient
            update = lr * grad
            h_new = h_vals - update
            
            # Project onto constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (np.sum(h_vals) / np.sum(h_new))  # Rescale
            
            # Check new value
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5 - 1e-10:
                best_h = h_new.copy()
                best_C5 = new_C5
                no_improvement = 0
            else:
                no_improvement += 1
                
            h_vals = h_new
            
            if no_improvement > patience or best_C5 < 0.3807:
                break
                
            if time.time() - start_time > budget_s - 5:
                break
                
        return best_h, best_C5
    
    start_time = time.time()
    
    # Optimize until time available or target achieved
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    iterations = 0
    patience = 20
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and best_C5 > 0.3807:
        
        # Try several strategies to improve
        if iterations > 0 and best_C5 < 0.4:
            lr = 0.001
        else:
            lr = 0.0005
            
        h_best, c5_best = optimize_step(best_h, lr=lr, max_iter=50, patience=patience)
        
        if c5_best < best_C5:
            best_h = h_best
            best_C5 = c5_best
            no_improvement = 0
        else:
            no_improvement += 1
            
        iterations += 1
        
        if no_improvement > patience:
            # Try small random perturbation to escape local minima
            perturbation = np.random.normal(0, 0.001, n_points)
            pert_h = best_h + perturbation
            pert_h = np.clip(pert_h, 0, 1)
            pert_h = pert_h * (np.sum(best_h) / np.sum(pert_h))
            c5_pert = compute_C5(pert_h)
            if c5_pert < best_C5:
                best_h = pert_h.copy()
                best_C5 = c5_pert
            no_improvement = 0
        
        if best_C5 < 0.3807:
            break
        
    # Final validation
    best_C5 = compute_C5(best_h)
    return (best_h.tolist(), best_C5, n_points)
