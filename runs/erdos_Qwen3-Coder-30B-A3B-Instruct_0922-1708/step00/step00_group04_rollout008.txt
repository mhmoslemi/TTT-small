import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if it exists, otherwise create a smooth starting function
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        n_points = len(h)
    else:
        # Initialize with a smooth triangular wave
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))  # triangular wave
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # This computes the maximum of the overlap integral
        # We use the cross-correlation function
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Extract the relevant part of the correlation (original length)
        # At k=0, overlap is from i=0 to n-1, at k=1 from i=0 to n-2, etc.
        # So we take full convolution and pick the max
        return np.max(conv) * dx
    
    def objective(h_vals):
        # Use a smooth approximation to the maximum
        # log-sum-exp with parameter beta
        beta = 100.0
        conv = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Exponentiate with high beta for sharp approximation
        exp_vals = np.exp(beta * conv)
        log_sum_exp = np.log(np.sum(exp_vals)) / beta
        return log_sum_exp * dx
    
    def gradient(h_vals):
        # Analytic gradient for smooth objective
        # We approximate it with finite difference for stability
        eps = 1e-6
        grad = np.zeros_like(h_vals)
        base_obj = objective(h_vals)
        for i in range(len(h_vals)):
            h_pert = h_vals.copy()
            h_pert[i] += eps
            grad[i] = (objective(h_pert) - base_obj) / eps
        return grad
    
    # Optimization
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Simple gradient descent with adaptive steps
    # Use projected gradient descent
    def project_h(h_in):
        # Clip to [0, 1] and rescale
        h_in = np.clip(h_in, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h_in)
        h_in = h_in * scale_factor
        return h_in
    
    max_iter = 1000
    iter_count = 0
    last_improvement = 0
    step_size = 0.01
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        current_C5 = compute_C5(best_h)
        if current_C5 < 0.3807:
            break
            
        # Compute gradient using a finite difference approximation
        eps = 1e-5
        grad = np.zeros_like(best_h)
        base_obj = objective(best_h)
        for i in range(len(best_h)):
            h_pert = best_h.copy()
            h_pert[i] += eps
            grad[i] = (objective(h_pert) - base_obj) / eps
        
        # Update
        updated_h = best_h - step_size * grad
        # Project
        updated_h = project_h(updated_h)
        
        # Evaluate
        new_C5 = compute_C5(updated_h)
        if new_C5 < best_C5:
            best_h = updated_h.copy()
            best_C5 = new_C5
            last_improvement = 0
        else:
            last_improvement += 1
            # Reduce step size on no improvement
            if last_improvement > 10:
                step_size *= 0.95
        
        # Early stop if not improving
        if last_improvement > 100:
            break
    
    # Final check
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
