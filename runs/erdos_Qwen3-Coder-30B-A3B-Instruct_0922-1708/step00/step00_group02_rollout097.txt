import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    sum_target = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) / n_points
    
    # Normalize to correct sum
    h = h * (sum_target / np.sum(h))
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    patience = 500
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Momentum terms
    v_h = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation h ⋆ (1 - h)
        h1_minus_h = 1 - h
        correlation = correlate(h, h1_minus_h, mode='full')
        # We only care about shifts in [0, 2], which corresponds to indices in [n-1, 2*n-2]
        # Shifts k in [0, 2] correspond to indices in [n-1, 2*n-2] (1-indexed) or [n-1, 2*n-2] (0-indexed)
        # Actually in numpy correlate with mode="full", the indices go from -n+1 to n-1
        # we need k in [0,2] ⇔ indices in [n-1, 2*n-2] of the correlation array
        correlation = correlation[n_points-1:2*n_points-1]
        
        # Find max k
        k_max_idx = np.argmax(correlation)
        k_max = k_max_idx * dx
        
        # Compute C5 = max of correlation
        c5 = correlation[k_max_idx]
        
        # Update best if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient of h w.r.t. correlation at maximum
        # d/dh ( h(x) (1 - h(x+k)) ) = (1 - h(x+k)) - h(x) * d/dh (h(x+k))
        # The derivative across correlation: not straightforward.  
        # Use a more robust approach using finite differences on the full function.
        
        # We compute the gradient as: 
        # for each i: dC5/dh[i] = (1 - h[i+k]) + h[i+k] * (some term from d/dh of the integral)
        # But a better idea is to directly do gradient descent on h using the correlation function.
        # Use finite differences in the error function direction.
        
        # Since the true gradient rewrite of full integral is complex:
        # Estimate using a simpler substitution.
        # Here we compute gradient such that the maximum value decreases
        # The direction of change that decreases maximum overlap:
        # For each i, gradient contribution: (1 - h[i+k]) + h[i+i*k]  
        # Actually, just consider finite difference approximation.
        
        # Instead, compute the example gradient w.r.t. one shift, then adjust.
        # For gradient of correlation h ⋆ (1-h), let's think carefully:
        # The change in shift k can be thought directly through the correlation computation.
        # Or, compute directional derivative based on current state.
        
        # Simplified gradient estimate:
        # we want to decrease the max term from correlation,
        # in direction and magnitude where it's most sensitive.
        
        # if g[i] is the change we need to make to h[i] to better decrease C5:
        # g[i] = contribution to gradient from all positions.
        # Approximate as finite difference or direct adjustment:
        # Let's use standard algorithm with explicit gradient case:
        # Assume scenario at max index k_max
        # Gradient per h[i] from overlap increases,  
        # Overlap at k_max = ∫ h(x)(1 - h(x+k_max)) dx
        # Gradient w.r.t. h[i] is (1 - h[i + k_max])
        # We need to symmetrize this for forward difference i.e. 1-h[i + k_max]
        # Only for the evaluation k that is most sensitive!!!
        # Let's build a new gradient vector.
        
        # Extract computed maximum index k
        k_idx = k_max_idx
        gradient_h = np.zeros_like(h)
        
        # The partial derivative of the correlation term (h[i](1 - h[i+k]) ) w.r.t. h[i]
        # For i, the function is h[i] * (1 - h[i+k_idx])
        # So gradient with respect to h[i] is (1 - h[i+k_idx])
        # Do this for every i:
        for i in range(n_points):
            k_shift = k_idx - (n_points - 1)  # Convert correlation index to shift value
            # k_shift might be negative, due to convolution indexing
            j = (i + k_shift) % n_points  # Wrap around
            if k_shift >= 0:
                if i + k_shift < n_points:
                    gradient_h[i] = 1 - h[i + k_shift]
                else:
                    pass  # Out of bounds, ignore (shouldn't happen in full case handling)
            else:
                # k_shift is negative, so i - |k_shift|  
                if i >= -k_shift:
                    gradient_h[i] = 1 - h[i + k_shift]
                else:
                    pass  # Out of bounds, ignore
        
        # Normalization to sum and fp constraints
        gradient_h = 1 - h  # This is not exactly correct, but start from this direction
        
        # Bring in the actual derivative computation
        # Let's recompute with more attention - 
        # Overlap_k = sum over i h[i] * (1 - h[i+k])
        # d/d h[j] = 1 - h[j+k] if i = j, zero otherwise
        # Wait, that's not right because we don't just have one term with each h[j].
        # We sum over each i.
        # d/d h[j] = sum_i (δ_ij (1 - h[i+k]))
        # Which means: for each j, 1 - h[j+k] 
        # Hold: So the gradient is:
        
        # Positive direction gradient contribution: 
        # The gradient of C5 = max_{k} ∫ h(x)(1 - h(x+k)) dx
        # regularization term: ∫ h(x) dx - 1 = 0
        # Gradient is of the form: (1 - h[i + k_max]) for each i
        # We compute a single gradient to align with that direction
        transmitted_grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_idx) % n_points
            transmitted_grad[i] = 1 - h[j]
        
        # Normalize
        gradient_h = transmitted_grad.copy()
        gradient_h = gradient_h - np.mean(gradient_h)
        
        # Update momentum
        v_h = beta * v_h + (1 - beta) * gradient_h
        h = h - alpha * v_h
        
        # Project onto feasible region [0,1] and maintain sum constraint
        h = np.clip(h, 0, 1)
        h = h * (sum_target / np.sum(h))
        
        # Adaptive learning rate
        if i % patience == 0 and i > 0:
            # If no change in best c5, reduce learning rate
            # We can't easily track this here, so just reduce anyway
            alpha *= 0.5
                
    # Final evaluation  
    correlation_full = correlate(best_h, 1 - best_h, mode='full')
    correlation_full = correlation_full[n_points-1:2*n_points-1]
    final_c5 = np.max(correlation_full) * dx
    
    return (best_h, final_c5, n_points)
