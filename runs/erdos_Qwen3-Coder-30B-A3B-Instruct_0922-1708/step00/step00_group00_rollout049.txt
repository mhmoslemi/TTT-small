import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Compute C5 for given step function h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum=1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    n_points = 45
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5 * dx)
        h = project_h(h, n_points)
    
    # Start timing
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    no_improvement_count = 0
    max_no_improvement = 1000
    
    # Perturbation parameters
    initial_sigma = 0.1
    decay_factor = 0.99
    
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        # Generate perturbation
        sigma = initial_sigma * (decay_factor ** iteration)
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        
        # Project to feasible region
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update current h
        h = h_new.copy()
        
        # Early termination
        if no_improvement_count >= max_no_improvement:
            break
    
    return (best_h.tolist(), best_c5, n_points)
