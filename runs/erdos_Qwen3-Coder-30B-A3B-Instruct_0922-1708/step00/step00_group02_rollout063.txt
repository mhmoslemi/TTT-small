import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum is over all possible shifts from -n+1 to n-1
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_population(n_points, pop_size):
    """Initialize a population of feasible h vectors."""
    population = []
    for _ in range(pop_size):
        # Initialize with random values scaled to meet sum constraint
        h = np.random.random(n_points)
        h = project_h(h, n_points)
        population.append(h)
    return population

def crossover(parent1, parent2):
    """Perform uniform crossover between two parents."""
    mask = np.random.random(len(parent1)) > 0.5
    child = np.where(mask, parent1, parent2)
    return child

def mutate(h, mutation_rate=0.1):
    """Mutate h with given probability."""
    mask = np.random.random(len(h)) < mutation_rate
    h = np.where(mask, np.random.random(len(h)), h)
    return h

def genetic_algorithm(h0, n_points, budget_s):
    """Genetic algorithm optimization."""
    pop_size = 50
    max_generations = 1000
    time_left = lambda: budget_s - (time.time() - start_time)
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    population[0] = h0.copy()  # Ensure we start with the initial solution
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    generation = 0
    while time_left() > 2 and generation < max_generations:
        # Evaluate the whole population
        fitness = [evaluate_c5(individual) for individual in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Elitism: keep top 20%
        elite_count = max(1, pop_size // 5)
        elite = population[:elite_count]
        
        # Create offspring
        new_population = elite[:]
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(pop_size // 2, 2, replace=False)
            child = crossover(population[p1], population[p2])
            child = mutate(child)
            new_population.append(child)
            
        population = new_population
        generation += 1
    
    return best_h, best_c5

def gradient_descent(h0, n_points, budget_s):
    """Gradient descent optimization."""
    time_left = lambda: budget_s - (time.time() - start_time)
    start_time = time.time()
    
    h = h0.copy()
    c5 = evaluate_c5(h)
    learning_rate = 0.01
    
    # For gradient descent, we optimize the maximum overlap
    # Let's focus on the worst-performing shift
    
    prev_c5 = 1e10
    no_improvement_count = 0
    max_no_improvement = 5000
    
    while time_left() > 2 and no_improvement_count < max_no_improvement:
        # Compute cross-correlation and find worst shift
        corr = np.correlate(h, 1 - h, mode="full")
        worst_shift_idx = np.argmax(corr)
        worst_corr = corr[worst_shift_idx]
        
        # Compute gradient with respect to h for this shift
        
        # Manually compute finite differences for gradient
        epsilon = 1e-6
        grad = np.zeros(n_points)
        
        for i in range(n_points):
            # Perturb h[i] and compute difference in correlation
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus, n_points)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
            corr_plus_shift = corr_plus[worst_shift_idx]
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_h(h_minus, n_points)
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode="full")
            corr_minus_shift = corr_minus[worst_shift_idx]
            
            grad[i] = (corr_plus_shift - corr_minus_shift) / (2 * epsilon)
        
        # Update h using gradient descent
        h_new = h - learning_rate * grad
        h_new = project_h(h_new, n_points)
        
        c5_new = evaluate_c5(h_new)
        
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Reduce learning rate if no improvement
            if no_improvement_count % 1000 == 0:
                learning_rate *= 0.5
        
        if no_improvement_count % 100 == 0:
            prev_c5 = c5
    
    return h, c5
