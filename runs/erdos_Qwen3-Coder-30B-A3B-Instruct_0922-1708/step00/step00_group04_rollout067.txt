def evaluate_erdos_solution(h, dx):
    # Compute full overlap for all shifts
    corr = np.correlate(h, 1 - h, mode='full')
    mid = len(corr) // 2
    # Extract valid overlaps with shifts from -n+1 to n-1
    overlaps = corr[mid - (len(h) - 1):mid + len(h)]
    return np.max(overlaps) * dx
