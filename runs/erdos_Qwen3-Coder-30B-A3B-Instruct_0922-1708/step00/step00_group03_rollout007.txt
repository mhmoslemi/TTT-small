import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    start_time = time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * target_sum
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Use the initial solution for reference
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization loop with CVXPY
    # Convert to a more manageable form
    x = cvx.Variable(n_points)
    Z = cvx.Variable((n_points, n_points))
    
    # Problem setup
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == target_sum,
    ]
    
    # Add constraints on Z variables
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i,j] <= x[i],
                Z[i,j] <= 1 - x[j],
                Z[i,j] >= x[i] + x[j] - 1
            ]
    
    # Compute C5 value
    max_correlation = 0
    for k in range(-n_points+1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        correlations = cvx.sum(Z[np.arange(n_points)[valid], idx[valid]])
        if correlations > max_correlation:
            max_correlation = correlations
    
    # Objective - minimize max correlation
    objective = cvx.Minimize(max_correlation)
    prob = cvx.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
            x_value = x.value
            h = np.clip(x_value, 0, 1)
            h = h / np.sum(h) * target_sum
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_h = h
                best_c5 = c5
    except:
        pass

    # Final evaluation with a greedy refinement
    def refine_h(h):
        # Simple gradient ascent step
        h = h.copy()
        max_iter = 500
        step_size = 0.01
        for _ in range(max_iter):
            if time() - start_time > budget_s - 1:
                break
            # Compute gradient numerically
            # A simpler policy: small adjustments
            h = np.clip(h + np.random.normal(0, step_size, len(h)), 0, 1)
            h = h / np.sum(h) * target_sum
            new_c5 = evaluate_erdos_solution(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        return best_h, best_c5

    try:
        best_h, best_c5 = refine_h(best_h)
    except:
        pass
    
    return (best_h, best_c5, n_points)
