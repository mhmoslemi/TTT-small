import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / n_points
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum correlation (excluding trivial shifts)
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h onto constraints: [0,1] and sum(h) = n_points/2."""
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to desired sum
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2.0) / current_sum
    return h

def gradient_descent_step(h, n_points, learning_rate=0.01):
    """Perform one gradient descent step."""
    dx = 2.0 / n_points
    # Compute the overlap integral
    corr = np.correlate(h, 1 - h, mode="full")
    # Find the index of maximum correlation
    max_idx = np.argmax(corr)
    # Compute finite differences for gradient approximation
    eps = 1e-6
    grad = np.zeros(n_points)
    for i in range(n_points):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n_points)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode="full")
        grad[i] = (corr_plus[max_idx] - corr[max_idx]) / eps
    
    # Update h
    h_new = h - learning_rate * grad
    h_new = project_h(h_new, n_points)
    return h_new
