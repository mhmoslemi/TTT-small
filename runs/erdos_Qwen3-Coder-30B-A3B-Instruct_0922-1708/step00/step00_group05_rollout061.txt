import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize with the provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: piecewise constant function
        n_points = 70
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure the initial sum is correct
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    
    # Normalize to ensure it's in [0,1]
    h = np.clip(h, 0, 1)
    
    # Initial C5 computation
    def compute_C5(h):
        # Compute all overlaps
        correlation = correlate(h, 1-h, mode='full')
        # Extract the relevant part (overlaps with shifts from 0 to 2)
        # The middle elements correspond to overlaps from shift 0 to 2
        mid = len(correlation) // 2
        overlaps = correlation[mid:mid+n_points]
        # Apply dx scaling
        C5 = np.max(overlaps) * dx
        return C5
    
    # Gradient descent optimization
    best_h = h.copy()
    best_C5 = compute_C5(h)
    learning_rate = 0.01
    last_improvement = 0
    iteration = 0
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute all overlaps
        correlation = correlate(h, 1-h, mode='full')
        # Extract overlaps with proper indexing
        mid = len(correlation) // 2
        overlaps = correlation[mid:mid+n_points]
        # Scale by dx
        overlaps = overlaps * dx
        
        # Find the worst-case shift
        k_worst = np.argmax(overlaps)
        max_overlap = overlaps[k_worst]
        
        # Compute gradient for this k
        grad = np.zeros(n_points)
        if k_worst < n_points:
            # grad[i] = (1 - h[i + k_worst]) if i + k_worst < n_points
            for i in range(n_points):
                if i + k_worst < n_points:
                    grad[i] = 1 - h[i + k_worst]
        else:
            # This shouldn't be the case for a proper shift in [0,2)
            # In case of edge condition, we handle it directly
            pass
        
        # Adjust the gradient to match the correct definition for the problem
        # We compute: integral h(x)(1 - h(x + k)) dx
        # Gradient w.r.t. h[i] is the contribution from h[i]* (1 - h[i+k])  
        # But we process the actual "overlap" that we want to reduce
        
        # We are iterating over lags k from 0 to n_points-1
        # For a fixed k, the gradient w.r.t. h[i] is 
        # 1 - h[i + k], if i+k < n and 0 otherwise.
        # So we fill the gradient in a similar way
        
        # Reset the gradient appropriately
        grad = np.zeros_like(h)
        # Compute the correct gradient (this is the gradient of the overlap integral w.r.t. h)
        # For the overlap k, we have overlap = sum over i of h[i] * (1 - h[i+k]) * dx
        # so d/dh[j] = (if j+k < n then 1 - h[j+k]) * dx
        for i in range(n_points):
            if i + k_worst < n_points:
                grad[i] = 1 - h[i + k_worst]
        
        # Step
        h_new = h - learning_rate * grad
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to keep sum = target_sum
        current_sum_new = np.sum(h_new)
        if abs(current_sum_new - target_sum) > 1e-10:
            # Distribute the difference proportionally
            ratio = target_sum / current_sum_new
            h_new = h_new * ratio
            
        # Ensure final adjustment
        h_new = np.clip(h_new, 0, 1)
        current_sum_new = np.sum(h_new)
        if abs(current_sum_new - target_sum) > 1e-10:
            # Use the difference to adjust
            diff = target_sum - current_sum_new
            # Add diff evenly
            h_new += diff / n_points
            
        # Final normalization to keep sum exact
        h_new = np.clip(h_new, 0, 1)
        current_sum_final = np.sum(h_new)
        if abs(current_sum_final - target_sum) > 1e-10:
            # Adjust last entry to make exactly sum = target_sum
            h_new[-1] = h_new[-1] + (target_sum - current_sum_final)
            h_new = np.clip(h_new, 0, 1)
            
        h = h_new
        
        # Re-compute C5
        new_C5 = compute_C5(h)
        
        # Update best if improved
        if new_C5 < best_C5 - 1e-10:
            best_C5 = new_C5
            best_h = h.copy()
            
        # Adaptive learning rate
        if iteration % 100 == 0 or iteration % 50 == 0:
            if iteration % 100 == 0:
                learning_rate *= 0.95
                
        # Early termination if no improvement for 100 steps
        if iteration > 100 and abs(new_C5 - best_C5) < 1e-10:
            last_improvement += 1
            if last_improvement > 100:
                break
        else:
            last_improvement = 0
            
    return (best_h.tolist(), float(best_C5), int(n_points))
