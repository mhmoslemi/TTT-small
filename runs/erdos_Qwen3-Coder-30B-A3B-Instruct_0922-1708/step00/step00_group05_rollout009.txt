import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    def compute_c5(h):
        # Compute all overlaps
        h_pad = np.concatenate([h, np.zeros(n_points - 1)])
        h_rev = np.concatenate([np.zeros(n_points - 1), h])
        conv = np.convolve(h_pad, h_rev[::-1], mode='full')
        # Overlaps are computed with (1 - h(x+k))
        overlaps = np.convolve(h, 1 - h, mode='full')
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    def evaluate(h):
        return compute_c5(h)
    
    def project(h):
        # Keep bounds
        h = np.clip(h, 0, 1)
        # Maintain the integral constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    def solve_qp(h_start, max_iter=5):
        # Classic QP approach in CVXPY
        h_vars = cp.Variable(n_points)
        C = cp.Variable()
        
        # Objective
        objective = cp.Minimize(C)
        
        # Constraints
        constraints = []
        # Bound constraints
        constraints.extend([h_vars[i] >= 0 for i in range(n_points)])
        constraints.extend([h_vars[i] <= 1 for i in range(n_points)])
        # Integral constraint
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Overlap constraints for different offsets
        for k in range(-n_points + 1, n_points):
            # Convolution-like terms: sum_i h[i] * (1 - h[i + k])
            # If k > 0: h[i] * (1 - h[i+k])
            # If k < 0: h[i] * (1 - h[i+k])  where i+k >= 0
            
            expr = 0
            if k >= 0:
                for i in range(n_points - k):
                    expr += h_vars[i] * (1 - h_vars[i + k])
            else:
                # k < 0
                for i in range(-k, n_points):
                    expr += h_vars[i] * (1 - h_vars[i + k])
            
            constraints.append(expr <= C * dx)
        
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                sol = h_vars.value
                return sol, prob.value
        except:
            pass
        return None, None

    for trial in range(5):

                step_size = 0.01 * dx

                    h_plus = project(h_plus)

                    h_minus[i] -= step_size
                    h_minus = project(h_minus)
                    grad = (loss(h_plus) - loss(h_minus)) / (2 * step_size)
                    h_grad[i] -= grad * 0.005  # small learning rate
                
                h = project(h_grad)

    # Apply fallback optimization if no solution found yet
    if best_h is None:
        h = h0.copy()
        h = project(h)
        
        # Try gradient descent variants manually
        def loss(h_vec):
            h_vec = project(h_vec)
            return evaluate(h_vec)
        
        def safe_grad(h_vec, dx=1e-4):
            grad = np.zeros_like(h_vec)
            h_vec = project(h_vec)
            for i in range(len(h_vec)):
                h_up = h_vec.copy()
                h_up[i] += dx
                h_up = project(h_up)
                h_low = h_vec.copy()
                h_low[i] -= dx
                h_low = project(h_low)
                grad[i] = (loss(h_up) - loss(h_low)) / (2 * dx)
            return grad

        # Gradient descent
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            grad = safe_grad(h)
            h = h - 0.001 * grad
            h = project(h)
            c5_val = evaluate(h)
            if c5_val < best_c5:

                best_h = h.copy()
    
    if best_h is None:
        best_h = h0.copy()
        best_c5 = evaluate(best_h)

    return (best_h, best_c5, n_points)
