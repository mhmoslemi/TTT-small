def evaluate_c5(h_values):
    """Compute C5 from h_values using full correlation."""
    n = len(h_values)
    dx = 2.0 / n
    # use full correlation to get all shifts
    corr = signal.correlate(h_values, 1 - np.array(h_values), mode='full')
    # max over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_and_rescale(h, n_points):
    """Project h to [0,1] and rescale to sum = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    else:
        h = np.full_like(h, 0.5 * (n_points / 2) / n_points)
    return h
