import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0  # Because sum(h) * dx = 1
    
    # Initialize

    # Fourier-based optimization approach
    # Use cosine basis with even symmetry for h(x)
    # We'll use 10 modes (coeffs a_0 to a_9)
    m = 10  # number of cosine modes to use  
    a_coeffs = np.random.rand(m)  # Initialize random a_k's
    a_coeffs[0] = 0.5  # a_0 = 1/2 to satisfy integral constraint
    
    best_C = float('inf')
    best_h = None
    
    def project_h(a_vec):
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points, endpoint=False)
        for k in range(m):
            h += a_vec[k] * np.cos(k * np.pi * x / 2)
        # Clip to [0,1], then rescale to ensure integral = 1
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h = h * target_sum / h_sum
        return h
    
    def compute_C(h_vals):
        h = np.array(h_vals)
        # compute overlap C_5 = max_k int h(x)(1 - h(x+k)) dx 
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(np.abs(corr)) * dx
        return max_corr
    
    # Optimization using gradient-based method
    # We'll optimize a_coeffs to minimize maximum overlap
    def obj_func(a_vec):
        h = project_h(a_vec)
        C = compute_C(h)
        return C
    
    def gradient_estimate(a_vec, k=1):
        eps = 1e-5
        grad = np.zeros_like(a_vec)
        f_base = obj_func(a_vec)
        for i in range(len(a_vec)):
            a_plus = a_vec.copy()
            a_plus[i] += eps
            f_plus = obj_func(a_plus)
            grad[i] = (f_plus - f_base) / eps
        return grad
    
    # Simple gradient descent loop with time tracking
    learning_rate = 0.01
    patience = 0
    max_patience = 100
    
    # Try to optimize for as long as we have time
    while time.time() - start_time < budget_s - 1:
        h = project_h(a_coeffs)
        C = compute_C(h)
        
        if C < best_C:
            best_C = C
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        if patience >= max_patience:
            break
            
        grad = gradient_estimate(a_coeffs)
        # update
        a_coeffs -= learning_rate * grad
        
        # project a_coeffs back to [0,1]
        a_coeffs = np.clip(a_coeffs, 0, 1)
        
        # If learning rate gets too small, lower it
        if np.linalg.norm(grad) < 1e-8:
            learning_rate *= 0.5
            
        if learning_rate < 1e-8:
            break
    
    if best_h is None:
        # fallback to initial if no improvement found
        h = project_h(a_coeffs)
        best_h = h.copy()
        best_C = compute_C(best_h)
    
    # Final check for feasibility
    assert 0 <= np.min(best_h) <= 1
    assert 0 <= np.max(best_h) <= 1
    assert abs(np.sum(best_h) * dx - 1.0) < 1e-10
    
    # For exact match we want sum(h)*dx = 1, so sum(h) = target_sum
    assert abs(np.sum(best_h) - target_sum) < 1e-10
    
    return (best_h.tolist(), best_C, n_points)
