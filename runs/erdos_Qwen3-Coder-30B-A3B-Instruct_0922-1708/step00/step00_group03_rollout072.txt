import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def initial_population(n_points, pop_size, initial_h):
    """Create initial population."""
    pop = [initial_h]
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        # Adjust to have correct integral
        h = h * (n_points / 2) / np.sum(h)
        pop.append(h)
    return pop

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    # Random crossover point
    cp = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:cp], parent2[cp:]])
    child2 = np.concatenate([parent2[:cp], parent1[cp:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, mutation_strength=0.1):
    """Mutate a step function."""
    h = h.copy()
    n = len(h)
    if np.random.random() > mutation_rate:
        return h
    # Select random indices to mutate
    n_mutate = max(1, int(0.1 * n))
    indices = np.random.choice(n, size=n_mutate, replace=False)
    for i in indices:
        h[i] += np.random.uniform(-mutation_strength, mutation_strength) * (2.0 / n)
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to desired sum
    h = h * (n / 2) / np.sum(h)
    return h

def tournament_selection(pop, fitnesses, tournament_size=0.2):
    """Perform tournament selection."""
    n = len(pop)
    k = max(1, int(tournament_size * n))
    selected_indices = np.random.choice(n, size=k, replace=False)
    best_idx = selected_indices[np.argmin([fitnesses[i] for i in selected_indices])]
    return pop[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    pop_size = 100
    max_generations = 300
    mutation_rate = 0.2
    mutation_strength = 0.1

    # Initialize
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.uniform(0, 1, n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    # Population
    pop = initial_population(n_points, pop_size, initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    
    for gen in range(max_generations):
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h) for h in pop]
        
        # Update best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = pop[min_idx].copy()
        
        # Check if time is running out
        if time.time() - start_time > budget_s - 30:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        pop = [pop[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top half
        pop = pop[:pop_size//2]
        fitnesses = fitnesses[:pop_size//2]
        
        # Generate new population with crossover and mutation
        new_pop = [best_h]  # Elitism
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(pop, fitnesses)
            parent2 = tournament_selection(pop, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate, mutation_strength)
            child2 = mutate(child2, mutation_rate, mutation_strength)
            new_pop.extend([child1, child2])
        
        # Trim to exact population size
        pop = new_pop[:pop_size]
    
    return (best_h.tolist(), best_c5, n_points)
