import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric plateau with tapering
        n_points = 47
        h = np.ones(n_points)
        # Create symmetric tapering
        mid = n_points // 2
        h[:mid] = np.linspace(1.0, 0.1, mid)
        h[mid:] = np.linspace(0.1, 1.0, n_points - mid)
        # Normalize to sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h *= (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    max_iter = 3000
    tau = 1.0
    anneal_factor = 0.999
    learning_rate = 0.05
    convergence_tol = 1e-5
    
    start_time = time.time()
    best_h = h.copy()
    best_C5 = compute_C5(h, n_points)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps
        overlaps = compute_overlaps(h, n_points)
        
        # Smooth max using log-sum-exp
        C5_approx = np.log(np.sum(np.exp(tau * overlaps))) / tau
        
        # Compute gradients
        grad = compute_gradient(h, n_points, overlaps, tau)
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        
        # Evaluate true C5
        current_C5 = compute_C5(h, n_points)
        
        # Update best solution
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Anneal temperature
        tau *= anneal_factor
        
        # Convergence check
        if i > 10 and abs(C5_approx - np.max(overlaps)) < convergence_tol:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def compute_overlaps(h, n_points):
    dx = 2.0 / n_points
    # Use cross-correlation with flipped h to compute overlaps
    h_flipped = np.flip(1 - h)
    corr = correlate(h, h_flipped, mode='full')
    # Adjust indices to match problem definition
    overlaps = corr[n_points-1:] * dx
    return overlaps

def compute_C5(h, n_points):
    overlaps = compute_overlaps(h, n_points)
    return np.max(overlaps)

def compute_gradient(h, n_points, overlaps, tau):
    dx = 2.0 / n_points
    # Compute the full overlap values with respect to all shifts
    # This could be optimized but for now we recalculate the full overlap
    h_expanded = np.concatenate([h, h, h])  # Helps with indexing
    # Compute the gradient components
    grad = np.zeros_like(h)
    for i in range(n_points):
        # For shift k, we have overlap = sum_{j} h[j] * (1 - h[(j+k)%n])
        # so d/dh[i] = sum_{j: (j+k)%n = i} (1 - h[j]) * dx
        # We compute for all k shifts
        temp_grad = np.zeros_like(h)
        for k in range(n_points):
            if (i - k) % n_points < n_points:
                j = (i - k) % n_points
                temp_grad[i] += -(1 - h[j]) * dx
        grad += temp_grad * np.exp(tau * overlaps[k])
    
    # Normalize by the denominator
    denom = np.sum(np.exp(tau * overlaps))
    if denom > 0:
        grad /= denom
    else:
        grad = np.zeros_like(grad)
    return grad
