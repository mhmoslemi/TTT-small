import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral for given h."""
    conv = correlate(h, 1 - h, mode='full')
    max_lag = len(conv) // 2
    return np.max(conv[:max_lag]) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up problem parameters
    n_points = 70
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05, 0.05, n_points) * dx
        h = np.clip(h, 0, 1)
        h = h * (1.0 / (np.sum(h) * dx))  # Rescale to sum to 1
    
    # Convex surrogate approach using smooth max
    alpha = 0.1

    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h * (1.0 / (np.sum(h) * dx))  # Rescale to meet constraint
        h = h * dx  # Convert back to dx scaled
        
        conv = correlate(h, 1-h, mode='full')
        max_lag = len(conv) // 2
        overlaps = conv[:max_lag]
        # Use log-sum-exp for smooth max approximation
        # Scaled version to avoid overflow
        scaled_overlaps = alpha * overlaps
        max_scaled = np.max(scaled_overlaps)
        log_sum = max_scaled + np.log(np.sum(np.exp(scaled_overlaps - max_scaled)))
        return log_sum / alpha
    
    def jac(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (1.0 / (np.sum(h) * dx))
        h = h * dx  # Convert back to dx scaled
        
        conv = correlate(h, 1-h, mode='full')
        max_lag = len(conv) // 2
        overlaps = conv[:max_lag]
        # Gradient considerations: This is a rough approximation
        # For optimization, we'll use numerics with bounds enforcement
        
        # Simulated gradient via small perturbation
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_pert = h.copy()
            h_pert[i] += eps
            # Normalization for this perturbed h
            h_pert = h_pert * (1.0 / (np.sum(h_pert) * dx))
            h_pert = h_pert * dx
            conv_pert = correlate(h_pert, 1-h_pert, mode='full')
            overlaps_pert = conv_pert[:max_lag]
            scaled_pert = alpha * overlaps_pert
            max_scaled_pert = np.max(scaled_pert)
            log_sum_pert = max_scaled_pert + np.log(np.sum(np.exp(scaled_pert - max_scaled_pert)))
            grad[i] = (log_sum_pert - objective(h_vec)) / eps
        return grad

    # Optimization loop with progress tracking
    best_h = h.copy()
    best_c5 = float('inf')
    last_improvement = 0
    iteration = 0
    
    # Memoize the function evals to prevent recomputation
    last_h = None
    last_c5_val = float('inf')

        iteration += 1
        # For stability, use a sampled version
        try:
            h_flat = h.flatten()
            h_init = h_flat.copy()

            # Apply bfgs-style descent with bounds

            conv = correlate(h_opt, 1-h_opt, mode='full')

    final_max_lag = len(final_conv) // 2
    final_c5 = np.max(final_conv[:final_max_lag]) * dx

    return (final_h.tolist(), final_c5, n_points)
