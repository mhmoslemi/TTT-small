import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    lambda_reg = 1e-4
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vec):
        # Compute full convolution of h with (1-h)
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Return maximum value scaled by dx
        return np.max(conv) * dx
    
    def objective(h_vec):
        # Add regularization and return max overlap
        return compute_c5(h_vec) + lambda_reg * np.sum(h_vec**2)
    
    def gradient(h_vec):
        # Approximate gradient via finite difference
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (compute_c5(h_plus) - compute_c5(h_minus)) / (2 * eps)
        return grad + 2 * lambda_reg * h_vec
    
    def project_and_rescale(h_vec):
        # Clip to [0,1] and rescale to sum = n_points / 2
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def line_search(h_vec, grad, alpha=1.0, max_iter=10):
        for _ in range(max_iter):
            h_new = h_vec - alpha * grad
            h_new = project_and_rescale(h_new)
            if compute_c5(h_new) < compute_c5(h_vec):
                return h_new
            alpha *= 0.5
        return h_vec
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Optimization iterations
    max_iter = 800
    iter_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute gradient
        grad = gradient(best_h)
        
        # Perform line search
        h_new = line_search(best_h, grad)
        
        # Update
        best_h = h_new
        current_c5 = compute_c5(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
        
        iter_count += 1
        
        # Occasionally perform a local shift search
        if iter_count % 50 == 0:
            # Try shifts for shift directions that might reduce overlap
            for k in range(1, n_points // 2):
                shifted_h = np.roll(best_h, k)
                c5_shifted = compute_c5(shifted_h)
                if c5_shifted < best_c5:
                    best_c5 = c5_shifted
                    best_h = shifted_h
    return best_h.tolist(), best_c5, n_points
