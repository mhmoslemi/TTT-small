def compute_c5(h, dx):
    # Compute overlapping integrals for all lag values and find maximum
    n = len(h)
    max_over = 0.0
    for k in range(n):
        # Compute integral of h(x)(1 - h(x+k)), where h(x+k) is circularly shifted
        integral = 0.0
        for i in range(n):
            j = (i + k) % n
            integral += h[i] * (1 - h[j])
        integral *= dx
        if integral > max_over:
            max_over = integral
    return max_over
