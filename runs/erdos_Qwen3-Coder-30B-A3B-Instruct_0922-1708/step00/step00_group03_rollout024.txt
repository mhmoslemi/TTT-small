import numpy as np
import time
from scipy import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    # Normalize to ensure sum(h) * dx = 1
    dx = 2.0 / len(h)
    h = np.array(h)
    h = h * (len(h) / 2) / np.sum(h)
    
    # Compute overlap integral for all shifts
    overlaps = correlate(h, 1 - h, mode="full") * dx
    C5 = np.max(overlaps)
    return C5

def initial_population(n_points, pop_size, initial_h_values=None):
    """Create initial population of step functions."""
    population = []
    
    # Add initial solution if available
    if initial_h_values is not None:
        population.append(np.array(initial_h_values))
    
    # Fill with random samples that satisfy constraints
    while len(population) < pop_size:
        h = np.random.random(n_points)
        # Adjust to make sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def crossover(parent1, parent2):
    """Crossover two parents to produce offspring."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Rescale to satisfy constraints
    dx = 2.0 / n
    child1 = child1 * (n / 2) / np.sum(child1)
    child1 = np.clip(child1, 0, 1)
    child2 = child2 * (n / 2) / np.sum(child2)
    child2 = np.clip(child2, 0, 1)
    
    return child1, child2

def mutate(h, mutation_rate=0.2):
    """Mutate a step function."""
    if np.random.random() > mutation_rate:
        return h
    
    n = len(h)
    dx = 2.0 / n
    h = h.copy()
    
    # Choose points to mutate (10% of points)
    n_mutate = max(1, n // 10)
    indices = np.random.choice(n, n_mutate, replace=False)
    for i in indices:
        delta = np.random.uniform(-0.2, 0.2) * dx
        h[i] += delta
    
    # Clip values
    h = np.clip(h, 0, 1)
    
    # Rescale to maintain constraints
    h = h * (n / 2) / np.sum(h)
    
    return h

def tournament_selection(population, fitnesses, k=5):
    """Select a parent using tournament selection."""
    indices = np.random.choice(len(population), k, replace=False)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    # Initialize
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    pop_size = 100
    max_generations = 300
    
    # Get initial population
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    population = initial_population(n_points, pop_size, initial_h)
    
    # Evaluate initial population
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_C5 = fitnesses[best_idx]
    
    generation = 0
    no_improvement_count = 0
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 30:
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 50% as parents
        num_parents = max(2, pop_size // 2)
        parents = population[:num_parents]
        parent_fitnesses = fitnesses[:num_parents]
        
        # Create new population
        new_population = parents.copy()
        
        # Crossover and mutate to create children
        while len(new_population) < pop_size:
            parent1 = tournament_selection(parents, parent_fitnesses)
            parent2 = tournament_selection(parents, parent_fitnesses)
            
            if np.random.random() < 0.5:
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1)
                child2 = mutate(child2)
                new_population.extend([child1, child2])
            else:
                child1 = mutate(parent1)
                new_population.append(child1)
        
        # Keep only the top population
        new_population = new_population[:pop_size]
        
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h) for h in new_population]
        
        # Update best solution
        min_fitness = min(new_fitnesses)
        if min_fitness < best_C5:
            best_C5 = min_fitness
            best_idx = np.argmin(new_fitnesses)
            best_h = new_population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        population = new_population
        fitnesses = new_fitnesses
        
        generation += 1
        
        # Terminate if no improvement for too many generations
        if no_improvement_count > 50:
            break
    
    return (list(best_h), best_C5, n_points)
