import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        h[-n_points//3:] = 0.1
        h[n_points//3:-n_points//3] = 1.0
        # Normalize to correct sum
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 3000
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    convergence_tol = 1e-5
    
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps efficiently using FFT
        # overlap[k] = \int h(x) * (1 - h(x+k)) dx
        h_extended = np.concatenate([h, h])  # double the array to handle circular
        overlap = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            overlap[k] = np.sum(h * (1 - h_extended[k:k + n_points])) * dx
        
        # Compute smoothed max
        C5_smooth = np.log(np.sum(np.exp(tau * overlap))) / tau
        # Compute gradient of smooth approximation
        grad = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            k_offset = k - (n_points - 1)
            if k_offset < 0:
                # k_offset is negative, so we shift index
                grad += np.exp(tau * overlap[k]) * (-1 + h_extended[k:k + n_points]) * dx
            else:
                # k_offset is positive, so we shift backwards
                grad += np.exp(tau * overlap[k]) * (-1 + h_extended[n_points - 1 - k_offset:n_points - 1 - k_offset + n_points]) * dx
        
        # Normalize gradient
        grad_sum = np.sum(np.exp(tau * overlap))
        if grad_sum > 1e-12:
            grad = grad / grad_sum
            
        # Gradient step
        h = h - learning_rate * grad
        # Clamp to feasible range
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate current solution
        C5 = evaluate_erdos_solution(h, n_points)
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Anneal temperature
        tau *= anneal_factor
        if tau < 1e-6:
            tau = 1e-6
            
        # Early stopping condition
        if i > 10 and abs(C5 - best_C5) < convergence_tol:
            break
            
    return (best_h.tolist(), float(best_C5), n_points)

def evaluate_erdos_solution(h_vals, n_points):
    """Evaluate the overlap integral using the exact computation."""
    h = np.array(h_vals)
    dx = 2.0 / n_points
    
    # Begin with a "full" convolution with 1-h
    h_extended = np.concatenate([h, h])  # To avoid boundary effect
    # But for overlap, we should do it differently
    # Because the overlap is defined with circular shift
    full_correlation = np.correlate(h, 1 - h, mode='full')
    # We want to keep only the non-neg shifts, which are from 0 to 2*n_points-2
    # Actually, let's recompute directly with loops to ensure correctness
    # For the overlap we have:
    # \int_0^2 dx h(x)(1 - h(x+k))
    # But we're really looking at k from 0 to 2 (we cut off to 0,0001) but for different lags
    max_val = 0
    for k_idx in range(2 * n_points - 1):
        # k_idx goes from 0 to 2*n_points - 2
        # Corresponds to k = -1 to 1 in actual shift (since we're circular)
        k = (k_idx - (n_points - 1)) * dx  # Convert index to shift k in [0,2]
        if k < 0:
            k = -k  # Due to symmetry, it's `k` with opposite sign
        if k > 2:
            continue
            
        # This way causes overlapping indices in our discrete case
        # Simpler loop over shifts directly
        # For dshift in [-n_points+1, ..., n_points-1], we get valid cross-correlations
        # But steps are pretty small due to dx = 2/n_points
        # So we compute the ones that match or are close to primary
        # Simpler approach similar to manual cross-correlation
        prod = 0
        for i in range(n_points):
            j = (i + k_idx) % n_points
            prod += h[i] * (1 - h[j])
        val = prod * dx
        max_val = max(val, max_val)
    
    # Re-scale approach:
    # Hence, re-implement the steady approach more carefully:
    h_shift = np.zeros(2*n_points - 1)
    for i in range(n_points):
        h_shift[i] = h[i]
    h_shift = np.tile(h_shift, 3)  # Make it periodic in a loop-like way
    
    # But faster way using the correlation directly
    # Simple loop over lags like in filtering
    max_val = 0.0
    for k_idx in range(2*n_points - 1):
        k = (k_idx - (n_points - 1)) * dx  # So k = -1 to 1
        if k < 0:
            k = -k  # Bring to positive shift
        elif k > 1:
            continue  # Outside range for meaningful results
            
        val = 0.
        for i in range(n_points):
            j = (i + k_idx) % n_points
            val += h[i] * (1 - h[j]) * dx
        max_val = max(val, max_val)

    return max_val
