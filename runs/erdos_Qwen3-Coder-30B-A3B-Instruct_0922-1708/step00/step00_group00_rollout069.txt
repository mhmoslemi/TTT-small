import numpy as np
import time
from scipy import signal

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h"""
    dx = 2.0 / len(h)
    # Compute overlap integral using correlation
    correlation = signal.correlate(h, 1 - h, mode='full')
    # Apply dx scaling
    overlaps = correlation * dx
    # Return maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with diverse step functions"""
    population = []
    
    if initial_h is not None:
        # Use initial solution
        population.append(initial_h.copy())
        # Add scaled versions to ensure diversity
        for i in range(pop_size//5):
            scaled_h = initial_h + np.random.normal(0, 0.05, n_points)
            scaled_h = np.clip(scaled_h, 0, 1)
            scaled_h = scaled_h * (len(h) / np.sum(scaled_h))  # normalize sum
            population.append(scaled_h)
    
    # Add random individuals to complete the population
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (len(h) / np.sum(h))  # normalize sum
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select parent with tournament selection"""
    selected = []
    indices = list(range(len(population)))
    np.random.shuffle(indices)
    
    for i in range(0, len(indices), tournament_size):
        group_indices = indices[i:i+tournament_size]
        group_indices = [idx for idx in group_indices if idx < len(population)]
        
        if not group_indices:
            continue
            
        best_idx = min(group_indices, key=lambda idx: fitnesses[idx])
        selected.append(population[best_idx].copy())
    
    return selected

def crossover(parent1, parent2):
    """Perform single-point crossover"""
    n_points = len(parent1)
    if n_points <= 1:
        return parent1.copy(), parent2.copy()
        
    point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate step function with small perturbations"""
    n_points = len(h)
    mutated = h.copy()
    
    # Randomly mutate some elements
    num_mutations = max(1, int(n_points * mutation_rate))
    to_mutate = np.random.choice(n_points, num_mutations, replace=False)
    
    for idx in to_mutate:
        delta = np.random.normal(0, 0.05)
        mutated[idx] += delta
        
    # Ensure bounds [0,1]
    mutated = np.clip(mutated, 0, 1)
    # Normalize sum to be correct
    mutated = mutated * (len(h) / np.sum(mutated))
    
    return mutated

    n_points = 50  # Convenient size, small for speed
    pop_size = 100
    max_generations = 120

    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_c5 = float('inf')
    
    # Store early best
    best_in_run = float('inf')
    
    for gen in range(max_generations):
        if time.time() >= time_limit:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h)
            fitnesses.append(c5)
            
            if c5 < best_in_run:
                best_in_run = c5
                best_h = h.copy()
                
        # Update global best
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create next generation
        new_population = []
        
        # Elitism: keep top 20%
        elite_size = pop_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            if len(selected) < 2:
                break
                
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Crossover
            if np.random.rand() < 0.8:  # 80% crossover probability
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation

            child2 = mutate(child2)

        return (initial_h.tolist() if initial_h is not None else np.random.rand(n_points).tolist(),
