import numpy as np
from scipy.signal import correlate
import time
import math

def evaluate_c5(h):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h and (1-h)
    # This gives us the overlap integral for all shifts
    corr = correlate(h, 1-h, mode="full")
    # The max overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def init_h(n_points):
    """Initialize a step function h based off initial_h_values or create one"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h[:n_points]  # Trim to required size
        h = project_h(h, n_points)
    else:
        # Create starting function as sin wave scaled to meet constraint
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x))
        h = project_h(h, n_points)
    return h

def genetic_algorithm_phase(h0, n_points, budget_s):
    """Run genetic algorithm to escape local minima"""
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.2
    max_generations = 1000
    elapsed = 0
    start_time = time.time()
    
    # Initialize population
    population = [h0.copy()]
    for i in range(pop_size - 1):
        h = init_h(n_points)
        population.append(h)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    for gen in range(max_generations):
        if elapsed > 0.7 * budget_s:
            break
            
        # Evaluate all individuals
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for improvement
        current_best = population[0]
        current_c5 = fitness[0]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Elitism: keep top 20%
        n_elite = int(elitism_rate * pop_size)
        elite = population[:n_elite]
        
        # Create offspring
        offspring = []
        for _ in range(pop_size - n_elite):
            # Tournament selection
            if len(population) < 2:
                a, b = population[0], population[0]
            else:
                idx_a, idx_b = np.random.choice(len(population), 2, replace=False)
                a, b = population[idx_a], population[idx_b]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points - 1)
                child = np.concatenate([a[:crossover_point], b[crossover_point:]])
            else:
                child = a.copy()
            
            # Mutation
            if np.random.rand() < mutation_rate:
                mutation_idx = np.random.randint(n_points)
                child[mutation_idx] = np.random.rand()
            
            # Ensure feasibility
            child = project_h(child, n_points)
            offspring.append(child)
        
        # Replace worst with offspring
        population = elite + offspring
        
        elapsed = time.time() - start_time
        
    return best_h, best_c5, elapsed

def gradient_descent_phase(best_h, n_points, budget_s):
    """Refine using gradient descent on the worst-case overlap"""
    start_time = time.time()
    elapsed = 0
    current_h = best_h.copy()
    best_h = current_h.copy()
    best_c5 = evaluate_c5(current_h)
    
    # Learning rate
    lr = 0.01
    current_lr = lr
    patience = 0
    patience_limit = 1000
    
    converged = False
    while not converged and elapsed < 0.3 * budget_s:
        # Evaluate current C5 and identify the worst shift
        c5 = evaluate_c5(current_h)
        if c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = c5
            
        # For gradient computation, we compute correlation to get overlap at each shift
        # We keep track of the maximum overlap (and its shift)
        
        # Use all points to find gradient more intelligently
        # Alternative approach: we compute the gradient w.r.t to all shifts,
        # and optimize w.r.t. the shift that gave the max overlap
        corr = correlate(current_h, 1 - current_h, mode='full')
        max_shift_idx = np.argmax(corr)
        # Simplified gradient: we approximate using finite differences over the one problematic shift
        # Somewhat hacky, but works in this scenario
        
        # Compute approximate gradient numerically
        eps = 1e-4
        # Try to find worst case shift e.g., find k that maximizes overlap
        # This involves evaluating 2*n_points - 1 correlations at each shift
        # For large-scale computing, approximate:
        
        # Re-evaluate at several small perturbations of current h
        h_plus = current_h + eps
        h_minus = current_h - eps
        h_plus = project_h(h_plus, n_points)
        h_minus = project_h(h_minus, n_points)
        
        c5_plus = evaluate_c5(h_plus)
        c5_minus = evaluate_c5(h_minus)
        
        # Approximate gradient w.r.t. each component
        # by finite difference of the max overlap
        grad = (c5_plus - c5_minus) / (2 * eps) * (h_plus - h_minus) / np.linalg.norm(h_plus - h_minus) if np.linalg.norm(h_plus - h_minus) > 1e-12 else np.zeros(n_points)
        
        # Update with step
        next_h = current_h - current_lr * grad
        next_h = project_h(next_h, n_points)
        
        # Check for convergence by tracking C5
        if abs(evaluate_c5(next_h) - c5) < 1e-6:
            patience += 1
            if patience >= patience_limit:
                converged = True
        else:
            patience = 0
        
        # Update learning rate if no improvement for some iterations
        if evaluate_c5(next_h) < c5:
            current_h = next_h.copy()
        else:
            current_lr *= 0.5
            if current_lr < 1e-6:
                converged = True
                
        elapsed = time.time() - start_time
    
    # Final check
    final_c5 = evaluate_c5(best_h)
    return best_h, final_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    
    # Initialize
    n_points = 45
    h0 = init_h(n_points)
    
    # Phase 1: Genetic Algorithm
    ga_h, ga_c5, elapsed = genetic_algorithm_phase(h0, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    final_h, final_c5 = gradient_descent_phase(ga_h, n_points, budget_s)
    
    # Final evaluation
    final_c5 = evaluate_c5(final_h)
    
    # Ensure feasibility of final result
    final_h = project_h(final_h, n_points)
    
    elapsed = time.time() - start_time
    if final_c5 > 0.38085:
        # Fall back to more conservative approach
        final_h = init_h(n_points)
        final_c5 = evaluate_c5(final_h)
        
    return (final_h, final_c5, n_points)
