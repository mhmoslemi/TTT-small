import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = int(budget_s / 0.05)  # Rough estimate of iterations
    
    # Initialize population
    m = 20  # number of particles
    particles = []
    velocities = []
    
    # Initialize from provided construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Fallback construction
        h0 = np.ones(n_points) * 0.5
        h0[10:20] = 1.0  # Add a few high regions
        h0[40:50] = 0.0  # Add some low regions
    
    # Normalize to meet constraint
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    h0 = np.clip(h0, 0, 1)
    
    for i in range(m):
        if i == 0:
            h = h0.copy()
        else:
            # Perturb the initial solution
            h = h0 + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2.0) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track global best
    global_best_h = None
    global_best_C = float('inf')
    no_improvement_count = 0
    
    start_time = time.time()
    iteration = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Evaluate all particles
        particle_Cs = []
        for i, h in enumerate(particles):
            # Compute the overlap integral
            corr = signal.correlate(h, 1 - h, mode='full')
            C_val = np.max(corr) * dx
            particle_Cs.append(C_val)
            
            # Update personal best
            if C_val < particle_Cs[i]:
                particle_Cs[i] = C_val
                
        # Find global best
        min_C_idx = np.argmin(particle_Cs)
        if particle_Cs[min_C_idx] < global_best_C:
            global_best_C = particle_Cs[min_C_idx]
            global_best_h = particles[min_C_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update particles
        w = 0.7
        c1 = c2 = 2.0
        
        for i in range(m):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            personal_best = particles[i] if particle_Cs[i] < particle_Cs[i] else h
            v = (w * v + 
                 c1 * r1 * (personal_best - h) + 
                 c2 * r2 * (global_best_h - h))
            
            # Limit velocity
            v = np.clip(v, -0.1, 0.1)
            velocities[i] = v
            
            # Update position
            new_h = h + v
            new_h = np.clip(new_h, 0, 1)
            
            # Normalize
            new_h = new_h * (n_points / 2.0) / np.sum(new_h)
            
            particles[i] = new_h.copy()
        
        # Tournament selection and diversity maintenance
        if iteration % 10 == 0 and iteration > 0:
            # Sort by C value
            sorted_indices = np.argsort(particle_Cs)
            keep_indices = sorted_indices[:m // 2]
            # Replace worst with random
            for i in range(m // 2, m):
                particles[i] = np.random.rand(n_points)
                particles[i] = particles[i] * (n_points / 2.0) / np.sum(particles[i])
                particles[i] = np.clip(particles[i], 0, 1)
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Periodic diversity reintroduction
        if no_improvement_count >= 100:
            for i in range(m):
                particles[i] += np.random.normal(0, 0.01, n_points)
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2.0) / np.sum(particles[i])
            no_improvement_count = 0
            
        iteration += 1
    
    # Final validation of global best
    if global_best_h is not None:
        corr = signal.correlate(global_best_h, 1 - global_best_h, mode='full')
        final_C = np.max(corr) * dx
        return (global_best_h.tolist(), final_C, n_points)
    
    # Fallback if no improvement found
    return (h0.tolist(), 0.38085, n_points)
