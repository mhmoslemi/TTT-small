        # Compute overlaps for all lags
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlap = 0
            for j in range(n_points):
                idx = (j + k) % n_points
                overlap += h[j] * (1 - h[idx])
            overlaps[k] = overlap * dx
        
        # E-step: compute current probabilities (normalized)
        p = overlaps / np.sum(overlaps)
        
        # M-step: update h using gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            for j in range(n_points):
                idx = (j - k) % n_points
                grad[j] += p[k] * (1 - h[idx])
        
        # Update h
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to enforce constraint
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Perturb slightly to avoid stagnation
        h_new += np.random.normal(0, 0.0001, n_points)
        
        # Check new C5 value
        h = h_new
        c5 = evaluate_erdos_solution(h)[1]
        
        # Update best solution if improved
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Adjust learning rate
        if no_improvement_count % 10 == 0 and no_improvement_count > 0:
            eta *= 0.5
            
        # Early stopping
        if no_improvement_count >= patience:
            break
