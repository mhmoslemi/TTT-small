import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the overlap integral for a given h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap C5 is the maximum value
    c5 = np.max(corr) * dx
    return c5

def enforce_constraints(h):
    """Enforce that 0 <= h[i] <= 1 and sum(h) = n/2."""
    h = np.clip(h, 0, 1)
    target_sum = len(h) / 2
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        # Adjust by scaling
        if current_sum > 0:
            h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.05):
    """Mutate an individual by small random changes."""
    h = h.copy()
    n_points = len(h)
    # Apply mutation
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            # Add random noise
            delta = np.random.normal(0, 0.05)
            h[i] += delta
    # Enforce constraints
    h = enforce_constraints(h)
    return h

def crossover_individuals(parent1, parent2):
    """Perform crossover between two parents."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    # Apply constraint enforcement
    child1 = enforce_constraints(child1)
    child2 = enforce_constraints(child2)
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def initialize_population(n_points, pop_size, initial_h_values=None):
    """Initialize a population of individuals."""
    population = []
    if initial_h_values is not None and len(initial_h_values) == n_points:
        # Use the initial construction
        h = enforce_constraints(np.array(initial_h_values, dtype=float))
        population.append(h)
        pop_size -= 1

    # Fill the rest with random individuals
    for _ in range(pop_size):
        # Generate random h, then scale to meet sum constraint
        h = np.random.rand(n_points)
        h = enforce_constraints(h)
        population.append(h)
    return population

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    pop_size = 100
    mutation_rate = 0.05
    elite_size = int(0.2 * pop_size)
    max_generations = 120
    tournament_size = 5
    crossover_prob = 0.8
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = np.min(fitnesses)
    
    generation = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        generation += 1
        
        # Selection
        new_population = []
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses, tournament_size)
            parent2 = tournament_selection(population, fitnesses, tournament_size)
            
            if np.random.rand() < crossover_prob:
                child1, child2 = crossover_individuals(parent1, parent2)
                new_population.extend([child1, child2])
            else:
                # Just copy parents
                new_population.extend([parent1.copy(), parent2.copy()])
        
        # Trim to correct size
        new_population = new_population[:pop_size]
        population = new_population[:]
        
        # Mutation
        for i in range(len(population)):
            population[i] = mutate_individual(population[i], mutation_rate)
        
        # Evaluate
        fitnesses = [evaluate_h(h) for h in population]
        
        # Check for best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
    
    return (best_h.tolist(), best_c5, n_points)
