from scipy.optimize import minimize
import time

    target_sum = n_points / 2.0

    def compute_C5(h):
        h = np.array(h)
        # Compute max overlap
        corr = np.correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx

        return np.sum(h) - target_sum

    def smooth_objective(h, reg=1e-4):
        h = np.array(h)
        C5 = compute_C5(h)
        # Add regularization for stability
        reg_term = reg * np.sum(h**2)
        return C5 + reg_term

    def gradient(h):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = smooth_objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            grad[i] = (smooth_objective(h_plus) - base_val) / eps
        return grad

    # Project onto feasible set
    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h *= target_sum / current_sum
        return h

    # Optimization loop
    start_time = time.time()
    max_iter = 800
    lr = 0.01
    tolerance = 1e-6
    prev_C5 = float('inf')

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient (simplified version using finite differences)
        grad = gradient(h)
        
        # Update step with projection
        h_new = h - lr * grad
        h_new = project(h_new)
        
        # Line search using backtracking
        alpha = 1.0
        while alpha > 1e-10:
            h_trial = h - alpha * grad
            h_trial = project(h_trial)
            if smooth_objective(h_trial) < smooth_objective(h) - 0.5 * alpha * np.sum(grad**2):
                h = h_trial
                break
            alpha *= 0.5
        
        # Check for convergence
        current_C5 = compute_C5(h)
        if abs(current_C5 - prev_C5) < tolerance:
            break
        prev_C5 = current_C5

    # Perform discrete shift search to lower the value
    best_h = h.copy()
    best_C5 = compute_C5(h)

    # Exhaustive search over small shifts
    for shift in range(-5, 6):

        shifted_C5 = compute_C5(shifted_h)
        if shifted_C5 < best_C5:
            best_C5 = shifted_C5
            best_h = shifted_h.copy()
    
    # Final validation
    final_C5 = compute_C5(best_h)
