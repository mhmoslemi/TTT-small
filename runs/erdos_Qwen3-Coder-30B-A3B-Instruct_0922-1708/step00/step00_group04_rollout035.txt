import numpy as np
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Model as mixed integer problem
    # Use a fixed number of segments for simplicity
    n_points = 42
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Try to improve initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Initialize with some smooth construction (model as 0-1 segments)
        h0 = np.ones(n_points) * 0.5
    
    # Optimize with scipy's differential evolution to minimize max correlation
    def compute_C5(h_vals):
        # Rescale to match integral constraint
        h_vals = np.array(h_vals)
        h_vals = h_vals * (1.0 / np.sum(h_vals)) * (2.0 / dx)
        # Ensure in range
        h_vals = np.clip(h_vals, 0, 1)
        # Evaluate overlap integral
        h = np.array(h_vals)
        conv = np.correlate(h, 1 - h, mode='full')
        # The overlap is at lags from -n+1 to n-1. We want max of the positive lags.
        # Symmetric: use just the second half (positive lags, including zero)
        conv = conv[len(conv)//2:]  # Take from zero to end (positive lags)
        res = np.max(conv) * dx
        return res

    # Define bounds [0,1] for all h values
    bounds = [(0,1) for _ in range(n_points)]
    
    # Use differential evolution to minimize C5
    from scipy.optimize import differential_evolution
    
    def objective(h_vals):
        return compute_C5(h_vals)
    
    # Initial guess
    start_h = h0.copy()
    
    try:
        # Set the time limit for differential evolution
        sol = differential_evolution(
            objective, 
            bounds,
            seed=seed,
            maxiter=500,
            popsize=15,
            strategy='best1bin',
            recombination=0.8,
            atol=1e-6,
            tol=1e-6
        )
        
        # Extract the solution
        h_final = sol.x
        c5_val = compute_C5(h_final)
    except Exception as e:
        # Fallback to initial solution if anything goes wrong
        h_final = start_h
        c5_val = compute_C5(h_final)
    
    # Ensure the solution meets the integral constraint
    # Check if integral is near 1
    integral_check = np.sum(h_final) * dx
    if not abs(integral_check - 1.0) < 1e-4:
        # Rescale h_final to enforce integral = 1
        h_final = h_final * (1.0 / integral_check)
    
    # Ensure final h values within bounds [0,1]
    h_final = np.clip(h_final, 0, 1)
    
    # Re-evaluate once more with scaled values
    final_c5 = compute_C5(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
