import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.full(n_points, 0.5)
    
    # Ensure sum constraint
    h *= target_sum / np.sum(h)
    
    # Function to compute C5
    def compute_c5(h):
        # Compute all overlaps
        overlaps = []
        for k in range(2 * n_points - 1):
            # Compute convolution manually
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlaps.append(overlap * dx)
        return max(overlaps)
    
    # Gradient-based optimization step
    def optimize_h(h, iterations=50, lr=0.1):
        for i in range(iterations):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Compute overlaps and gradients
            overlap_vals = []
            grad = np.zeros(n_points)
            
            for k in range(2 * n_points - 1):
                overlap = 0.0
                for i in range(n_points):
                    j = (i + k) % n_points
                    overlap += h[i] * (1.0 - h[j])
                    grad[i] += (1.0 - h[j])  # Simple gradient approximation
                overlap_vals.append(overlap * dx)
            
            # Gradient descent update
            grad /= len(overlap_vals)  # Normalize
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new *= target_sum / np.sum(h_new)
            
            h = h_new
            lr *= 0.995  # Decrease learning rate
            
        return h
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Local search attempts
    attempts = 0
    max_attempts = 50
    no_improve_count = 0
    max_no_improve = 50
    
    while attempts < max_attempts and time.time() - start_time < budget_s * 0.95:
        # Random perturb
        perturbation = np.random.normal(0, 0.01, n_points)
        h_trial = h + perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial *= target_sum / np.sum(h_trial)
        
        # Optimize
        h_opt = optimize_h(h_trial, iterations=20)
        c5_trial = compute_c5(h_opt)
        
        if c5_trial < best_c5:
            best_c5 = c5_trial
            best_h = h_opt.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        attempts += 1
        
        if no_improve_count > max_no_improve:
            break
    
    # Final optimization
    if time.time() - start_time < budget_s * 0.95:
        best_h = optimize_h(best_h, iterations=100)
        best_c5 = compute_c5(best_h)
        
    return (best_h, best_c5, n_points)
