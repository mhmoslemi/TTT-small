import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure initial sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Regularization parameter
    reg_lambda = 1e-4
    
    # Optimization parameters
    max_iter = 1000
    line_search_iter = 20
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i - k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps[k] = overlap * dx
            
        max_overlap = np.max(overlaps)
        
        # Compute gradient using finite differences
        grad = np.zeros(n_points)
        eps = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            overlap_plus = 0.0
            for k in range(n_points):
                overlap_k = 0.0
                for j in range(n_points):
                    l = (j - k) % n_points
                    overlap_k += h_plus[j] * (1 - h_plus[l])
                overlap_plus = max(overlap_plus, overlap_k * dx)
            
            grad[i] = (overlap_plus - max_overlap) / eps
        
        # Add regularization term
        grad += 2 * reg_lambda * h
        
        # Line search
        alpha = 1.0
        for _ in range(line_search_iter):
            h_new = h - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            overlap_new = 0.0
            for k in range(n_points):
                overlap_k = 0.0
                for j in range(n_points):
                    l = (j - k) % n_points
                    overlap_k += h_new[j] * (1 - h_new[l])
                overlap_new = max(overlap_new, overlap_k * dx)
                
            if overlap_new < max_overlap:
                h = h_new
                break
                
            alpha *= 0.5
            
        # Shift search improvement
        if iteration % 5 == 0:
            best_h = h.copy()
            best_overlap = max_overlap
            for shift in range(-5, 6):
                h_shift = np.roll(h, shift)
                h_shift = np.clip(h_shift, 0, 1)
                h_shift = h_shift * (n_points / 2) / np.sum(h_shift)
                
                shift_overlap = 0.0
                for k in range(n_points):
                    overlap_k = 0.0
                    for j in range(n_points):
                        l = (j - k) % n_points
                        overlap_k += h_shift[j] * (1 - h_shift[l])
                    shift_overlap = max(shift_overlap, overlap_k * dx)
                
                if shift_overlap < best_overlap:
                    best_overlap = shift_overlap
                    best_h = h_shift.copy()
                    
            h = best_h
            max_overlap = best_overlap
    
    # Final evaluation
    overlap_values = []
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            j = (i - k) % n_points
            overlap += h[i] * (1 - h[j])
        overlap_values.append(overlap * dx)
        
    c5_bound = max(overlap_values)
    
    return (h.tolist(), c5_bound, n_points)
