import numpy as np

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    else:
        n_points = 42

        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h.copy()

        h_vec = np.clip(h_vec, 0, 1)

        return h_vec
    
    # Optimization loop
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        iteration += 1
        # Compute current overlaps
        correlation = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation)

        M = cp.Variable(n_points)
        objective = cp.Minimize(M[0])
        constraints = [M[0] >= cp.sum(cp.multiply(h, 1 - h)) * dx]
        constraints += [0 <= h[i] for i in range(n_points)]
        constraints += [h[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h) == n_points / 2]
        
        # Use a projection approach instead of full optimization

        h_new = h.copy()

def evaluate_erdos_solution(h, dx):
    # Evaluate the overlap integral directly using np.correlate
    corr = np.correlate(h, 1 - h, mode="full") * dx
    return np.max(corr)
