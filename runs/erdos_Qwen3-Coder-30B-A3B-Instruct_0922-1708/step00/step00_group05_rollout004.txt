import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h):
    """Evaluate the maximum overlap for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Maximize over all possible lags
    max_corr = np.max(corr)
    return max_corr * dx

def validate_and_normalize(h, n_points):
    """Ensure h is valid and normalized."""
    h = np.array(h, dtype=float)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Scale to meet integral constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def perturb_h(h, n_points, delta=0.05):
    """Generate a perturbed version of h."""
    h_new = h.copy()
    # Randomly select some indices to perturb
    n_perturb = max(1, n_points // 10)
    indices = np.random.choice(n_points, size=n_perturb, replace=False)
    for i in indices:
        # Random perturbation
        h_new[i] += np.random.uniform(-delta, delta)
    # Ensure constraints
    h_new = validate_and_normalize(h_new, n_points)
    # Apply smoothing
    window = min(5, n_points // 10)
    if window > 1:
        h_smooth = np.zeros_like(h_new)
        for i in range(n_points):
            start = max(0, i - window//2)
            end = min(n_points, i + window//2 + 1)
            h_smooth[i] = np.mean(h_new[start:end])
        h_new = h_smooth
    return h_new
