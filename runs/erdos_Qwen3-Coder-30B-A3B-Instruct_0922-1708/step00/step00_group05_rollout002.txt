from scipy import signal
import time

def evaluate_erdos_solution(h):
    """Compute C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1-h, mode="full")
    # The maximum overlap
    max_corr = np.max(corr)
    # Scale by dx to get C5
    c5 = max_corr * dx
    return c5
