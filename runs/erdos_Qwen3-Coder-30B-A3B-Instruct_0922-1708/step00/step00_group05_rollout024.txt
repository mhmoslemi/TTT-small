import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    max_time = start_time + budget_s
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        # Default construction: 70 points
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Parameters for optimization
    m = min(20, n_points // 2)  # number of cosine modes
    max_iter = 10000
    lr = 0.01
    patience = 100
    
    # Initialize Fourier coefficients
    a = np.random.rand(m) * 0.5  # [0, 0.5] range
    a[0] = 0.5  # fixed for integral constraint
    
    # Precompute grid for smooth evaluation
    dx = 2.0 / n_points
    x = np.linspace(0, 2, n_points, endpoint=False)
    
    # Precompute cosine matrix
    cos_matrix = np.cos(np.arange(m)[:, None] * np.pi * x / 2)
    
    best_h = h0.copy()
    best_C = evaluate_erdos_solution(best_h)
    
    iter_count = 0
    no_improvement = 0
    
    while iter_count < max_iter and time.time() < max_time:
        iter_count += 1
        # ========= Reconstruct h from coefficients =========
        h = np.dot(cos_matrix.T, a)  # shape (n_points,)
        h = np.clip(h, 0, 1)
        
        # ========= Enforce integral constraint =========
        h_sum = np.sum(h) * dx
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        
        # ========= Check best =========
        C = evaluate_erdos_solution(h)
        if C < best_C:
            best_C = C
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Reset learning rate and randomize
            lr *= 0.5
            a = np.random.rand(m) * 0.5
            a[0] = 0.5
            no_improvement = 0
            
        # ========= Compute gradient =========
        # For optimization, we compute finite diff gradient for worst-case shift
        # Since computing full gradient on overlap matrix for all k is expensive,
        # we use greedy approach: pick k that maximizes overlap and diff it
        corr = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(corr)
        max_corr = corr[max_k]
        
        # Compute finite difference approx for gradient of max overlap w.r.t. a_k
        eps = 1e-4
        grad = np.zeros(m)
        for j in range(m):
            a_plus = a.copy()
            a_plus[j] += eps
            h_plus = np.dot(cos_matrix.T, a_plus)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus_sum = np.sum(h_plus) * dx
            if h_plus_sum > 0:
                h_plus = h_plus * (n_points / 2) / h_plus_sum
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            grad[j] = (corr_plus[max_k] - max_corr) / eps
            
        # Update coefficients
        a -= lr * grad
        a = np.clip(a, 0, 1)
        
    return (best_h, best_C, n_points)

def evaluate_erdos_solution(h):
    """Helper to compute C5 for given h."""
    n = len(h)
    # Since we use numpy complete convolution, we convert to proper overlap
    # correlate(h, 1-h, mode='full') computes all overlaps with shifts k=-n+1 to n-1
    # overlap at shift k is sum(h[i] * (1 - h[i+k])) for valid indices
    corr = np.correlate(h, 1 - h, mode='full')
    # Subtract the zero index, because it's a convolution not overlap (it's different from sum in definition)
    # But definition uses correlate, so let's be precise:
    dx = 2.0 / n
    corr = corr * dx
    # Violates math, correct idea is we consider all shifts k=1 to n-1 or radians
    # taking correlation of h with 1-h with full overlap, we consider largest value
    C5 = np.max(np.abs(corr))
    return C5
