import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    lr = 0.1
    patience = 50
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if abs(np.sum(h) * dx - 1.0) > 1e-10:
            h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
    
    # Clip and normalize
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    c5 = evaluate_erdos_solution(h)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # EM iteration loop
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # E-step: Compute overlaps for all lags
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1 - h[j])
            overlaps[k] = overlap * dx
        
        # Normalize overlaps to form probabilities
        if np.sum(overlaps) > 0:
            p = overlaps / np.sum(overlaps)
        else:
            p = np.ones(n_points) / n_points
        
        # M-step: Update h using gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            for i in range(n_points):
                j = (i - k) % n_points
                grad[i] += p[k] * (1 - h[j])
        
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Perturb with noise
        noise_std = max(0.01, 0.1 * np.exp(-iteration / 100))
        h_new += np.random.normal(0, noise_std, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        elif iteration > 0 and iteration % patience == 0:
            lr *= 0.5
            if lr < 1e-6:
                break
    
    # Final validation
    if best_h is not None:
        h = best_h
        c5 = evaluate_erdos_solution(h)
        # Ensure final normalization
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_erdos_solution(h)
    
    return (h, c5, n_points)
