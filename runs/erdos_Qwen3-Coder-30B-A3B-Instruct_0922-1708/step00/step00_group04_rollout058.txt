    best_h = None

    def compute_correlation(h_arr):
        # This computes all cross-correlations with 1-h
        h_rev = h_arr[::-1]
        corr = np.correlate(h_arr, h_rev, mode='full')
        # Note: correlate computes correlation with 1-h, which means we want
        # to compute h(x) * (1-h(x+k)) -> swap h with 1-h for correlate
        # So we compute np.correlate(h, 1-h, mode='full'), equivalent to
        # np.correlate(h, (1-h), mode='full') → (1-h) is flipped in correlate
        one_minus_h = 1 - h_arr
        one_minus_h_rev = one_minus_h[::-1]
        corr = np.correlate(h_arr, one_minus_h_rev, mode='full')
        return corr * dx
    
    def compute_c5(h_arr):
        corr = compute_correlation(h_arr)
        return np.max(corr)
    
    def soft_max(x, beta=100):
        # Approximation of max with log-sum-exp for differentiable max
        # as beta -> inf, log(sum(exp(beta * x))) / beta -> max(x)
        return np.log(np.sum(np.exp(beta * x))) / beta
    
    # Gradient of soft max approximation
    def soft_max_grad(x, beta=100):
        exp_x = np.exp(beta * x)
        return exp_x / np.sum(exp_x)
    
    def compute_objective(h_arr):
        corr = compute_correlation(h_arr)
        return soft_max(corr, beta=50)
    
    def compute_grad(h_arr):
        # This is not precise but provides good direction
        corr = compute_correlation(h_arr)
        max_idx = np.argmax(corr)
        # Find gradient w.r.t. positive and negative contributions
        # Use soft_max_grad at max
        # For the gradient, let's compute the finite differences:
        # Backward difference using FFT for efficiency
        h_copy = h_arr.copy()
        eps = 1e-5
        # Estimate gradient via finite diff
        grad = np.zeros_like(h_arr)
        for i in range(n_points):
            h_plus = h_copy.copy()
            h_minus = h_copy.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Compute finite diff manually using properties for now
            corr_plus = compute_correlation(h_plus)
            corr_minus = compute_correlation(h_minus)
            # Slightly squeeze the difference
            grad[i] = (corr_plus[max_idx] - corr_minus[max_idx]) / (2 * eps)
        return grad

        h_arr = np.clip(h_arr, 0, 1)
        # Rescale to sum to n_points/2

            h_arr = h_arr * (n_points / 2) / current_sum
        return h_arr

        h_new = h - step_size * grad
