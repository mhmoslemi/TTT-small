import numpy as np
import time
from scipy import correlate

def evaluate_c5(h_values):
    """Compute C5 value for given h_values"""
    dx = 2.0 / len(h_values)
    # Compute full correlation of h with (1-h)
    corr = correlate(h_values, 1 - h_values, mode="full")
    # Take maximum value and multiply by dx
    return np.max(corr) * dx

def project_h(h_vals):
    """Project h to feasible set: [0,1] and sum = n/2"""
    h = np.clip(h_vals, 0, 1)
    n = len(h)
    target_sum = n / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    else:
        h = np.full(n, 0.5)
    return h
