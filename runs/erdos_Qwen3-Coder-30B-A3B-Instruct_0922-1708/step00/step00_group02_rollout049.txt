import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with uniform distribution
        h = np.ones(n_points) * 0.5
    
    # Ensure constraint sum(h) * dx = 1
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Track momentum
    v_h = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlap for all shifts using cross-correlation
        # h(x)(1 - h(x+k)) is equivalent to conv(h, 1-h)
        h_reflect = 1 - h
        overlap = np.correlate(h, h_reflect, mode='full')
        # Adjust for the periodic boundary
        # With mode='full', overlap has length 2*n_points - 1
        # The maximum overlap corresponds to shifts in [0, 2] 
        # which maps to indices [n_points-1, n_points-1+2*n_points-1]
        # but we need only the shift k in [0, 2] which corresponds to 
        # a normalized index in [0, n_points-1] (using the periodic nature)
        
        # Compute the actual shifts
        step_idx = np.arange(2 * n_points - 1)
        shifts = (step_idx - (n_points - 1)) * dx
        
        # Focus on relevant range: 0 to 2
        valid_mask = (shifts >= 0) & (shifts <= 2)
        valid_shifts = shifts[valid_mask]
        valid_overlap = overlap[valid_mask]
        
        # Find k that maximizes overlap
        max_idx = np.argmax(valid_overlap)
        k_best = valid_shifts[max_idx]
        c5_current = valid_overlap[max_idx] * dx
        
        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter > patience:
                alpha *= 0.5
                patience_counter = 0
                
        # Compute gradient with respect to h.
        # The overlap integral is ∫ h(x) * (1 - h(x+k)) dx
        # Gradient ≈ 1 - h(x+k) - h(x) * ∂h(x+k)/∂h(x)
        # But for our optimization we take a more direct approach
        # Apply a gradient step based on cross correlation
        # We shift h by k and correlate with 1 - h
        # The gradient in this direction is more complex, 
        # but one good approximation is:
        # ∇h = (1 - h) - (gradient from correlation)
        
        # Compute the sensitivity of the max overlap to changes in h
        # By finite difference
        gradient = np.zeros(n_points)
        eps = 1e-5
        for j in range(n_points):
            # Create perturbed h
            h_pert = h.copy()
            h_pert[j] += eps
            h_pert = np.clip(h_pert, 0, 1)
            # Renormalize
            h_pert = h_pert * (n_points / 2.0) / np.sum(h_pert)
            
            # Compute new overlap
            h_reflect_pert = 1 - h_pert
            overlap_pert = np.correlate(h_pert, h_reflect_pert, mode='full')
            overlap_pert = overlap_pert[(overlap_pert >= 0) & (overlap_pert <= 2)]
            c5_pert = np.max(overlap_pert) * dx
            
            gradient[j] = (c5_pert - c5_current) / eps
            
        # Update with momentum
        v_h = beta * v_h + (1 - beta) * gradient
        h = h - alpha * v_h
        
        # Clip h
        h = np.clip(h, 0, 1)
        # Renormalize if needed
        h = h * (n_points / 2.0) / np.sum(h)
        
    if best_h is None:
        best_h = h
        
    # Final computation
    h_reflect = 1 - best_h
    overlap = np.correlate(best_h, h_reflect, mode='full')
    step_idx = np.arange(2 * n_points - 1)
    shifts = (step_idx - (n_points - 1)) * dx
    # Restrict to k in [0, 2] 
    valid_mask = (shifts >= 0) & (shifts <= 2)
    valid_shifts = shifts[valid_mask]
    valid_overlap = overlap[valid_mask]
    final_c5 = np.max(valid_overlap) * dx
    
    return (best_h, final_c5, n_points)
