import numpy as np
import time
from scipy.signal import correlate

def evaluate(h):
    # Compute C5 from the given h values
    n = len(h)
    dx = 2.0 / n
    # Compute all cross-correlations for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral is the maximum correlation (scaled)
    C5 = np.max(corr) * dx
    return C5

def project(h, n_points):
    # Ensure h remains feasible: 0 <= h[i] <= 1 and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    if np.sum(h) != n_points / 2:
        h = h * (n_points / 2) / np.sum(h)
    return h

def genetic_algorithm(h, n_points, pop_size=50, n_generations=100, time_limit=420):
    start_time = time.time()
    # Initialize population
    population = []
    for _ in range(pop_size):
        h_candidate = np.random.rand(n_points)
        h_candidate = project(h_candidate, n_points)
        population.append(h_candidate)
    
    # If we have an initial solution, use it as first individual
    if h is not None:
        population[0] = h.copy()
    
    best_h = None
    best_C5 = float('inf')
    
    for g in range(n_generations):
        if time.time() - start_time > time_limit:
            break
            
        # Evaluate fitness
        fitness = []
        for ind in population:
            C5 = evaluate(ind)
            fitness.append(C5)
            
        # Sort population
        idxs = np.argsort(fitness)
        population = [population[i] for i in idxs]
        fitness = [fitness[i] for i in idxs]
        
        # Update best solution
        if fitness[0] < best_C5:
            best_C5 = fitness[0]
            best_h = population[0].copy()
        
        # Elitism: keep top 20%
        elite_size = pop_size // 5
        new_pop = population[:elite_size]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            idx1, idx2 = np.random.choice(elite_size, 2, replace=False)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < 0.7:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < 0.1:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project to feasible region
            child1 = project(child1, n_points)
            child2 = project(child2, n_points)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        population = new_pop
        #print(f"GA Generation {g} best: {best_C5}")
    
    return best_h, best_C5

def gradient_descent(h, n_points, max_iter=1000, lr=0.01, tol=1e-6, time_limit=180):
    start_time = time.time()
    h = h.copy()
    best_h = h.copy()
    best_C5 = evaluate(h)
    no_improvement = 0
    patience = 200
    
    # Precompute 1-h for efficiency
    h_complement = 1 - h
    
    for i in range(max_iter):
        if time.time() - start_time > time_limit:
            break
            
        # Compute the full cross-correlation (using FFT for efficiency)
        # But we also need to efficiently get gradients for the shift where we have maximum correlation
        corr = correlate(h, h_complement, mode="full")
        max_shift_idx = np.argmax(corr)
        max_C5 = np.max(corr) * (2.0 / n_points)
        
        # Compute gradient of max correlation with respect to h
        # Simplification: assume the maximum shift is at the middle
        # This is an approximation for gradient computation
        # Note: We use numerical gradients for cycling through all positions

        # Approach: Compute gradient by finite differences for the maximum shift
        epsilon = 1e-4
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project(h_plus, n_points)

            corr_plus = correlate(h_plus, h_complement, mode="full")
            C5_plus = np.max(corr_plus) * (2.0 / n_points)
            
            grad[j] = (C5_plus - max_C5) / epsilon
            
        # Apply gradient descent update
        h = h - lr * grad
        
        # Project to feasible set
        h = project(h, n_points)
        
        new_C5 = evaluate(h)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        if no_improvement > patience:
            lr *= 0.5
            no_improvement = 0
            
        if abs(new_C5 - max_C5) < tol:
            break
            
        #print(f"GD Iter {i} C5: {new_C5}")
        
    return best_h, best_C5
