import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for the given step function h."""
    # Compute the full cross-correlation with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx to get actual overlaps
    overlaps = corr * dx
    # Return the maximum overlap
    return np.max(overlaps)

def enforce_constraints(h, n_points, dx):
    """Ensure h satisfies constraints: 0 <= h[i] <= 1 and sum(h) * dx = 1."""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-6:
        # Uniform adjustment to meet constraint
        adjustment = (target_sum - current_sum) / n_points
        h += adjustment
        h = np.clip(h, 0, 1)
    return h

def generate_individual(n_points, initial=None):
    """Generate a valid random individual."""
    if initial is not None:
        h = initial.copy()
        # Slight random perturbation
        noise = np.random.normal(0, 0.05, n_points)
        h += noise
        h = enforce_constraints(h, n_points, 2.0 / n_points)
    else:
        h = np.random.uniform(0, 1, n_points)
        h = enforce_constraints(h, n_points, 2.0 / n_points)
    return h

def crossover(parent1, parent2):
    """Segment-wise crossover of two parents."""
    n_points = len(parent1)
    # Randomly select crossover point
    cp = np.random.randint(1, n_points)
    offspring1 = np.concatenate([parent1[:cp], parent2[cp:]])
    offspring2 = np.concatenate([parent2[:cp], parent1[cp:]])
    return offspring1, offspring2

def mutate(individual, mut_rate=0.05):
    """Apply mutation to an individual."""
    n_points = len(individual)
    for i in range(n_points):
        if np.random.random() < mut_rate:
            individual[i] += np.random.normal(0, 0.05)
    return enforce_constraints(individual, len(individual), 2.0 / len(individual))

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    # Sample tournament_size individuals
    indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in indices]
    # Select the best from tournament
    best_idx = indices[np.argmin(tournament_fitnesses)]
    return population[best_idx]
