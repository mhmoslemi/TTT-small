import numpy as np
import time
import random
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for a given h."""
    # Compute correlations for all shifts
    conv = correlate(h, 1 - h, mode='full')
    # Convert to actual overlap values by multiplying with dx
    overlaps = conv * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h):
    """Initialize a population of step functions."""
    population = []
    # Add the initial solution
    if initial_h is not None:
        population.append(initial_h.copy())
    # Generate random individuals
    for i in range(pop_size - len(population)):
        # Start with uniform distribution scaled to desired sum
        h = np.full(n_points, 1.0 / n_points * n_points / 2)
        # Slightly perturb
        for j in range(n_points):
            h[j] += random.uniform(-0.1, 0.1)
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Normalize to satisfy constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        population.append(h)
    return population

def selection(population, fitnesses, tournament_size=5):
    """Tournament selection."""
    selected = []
    pop_size = len(population)
    for _ in range(pop_size):
        tournament_indices = random.sample(range(pop_size), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_index].copy())
    return selected

def crossover(parent1, parent2, crossover_prob=0.8):
    """Single-point crossover with probability."""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    # Select a crossover point
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate: add small random changes and normalize."""
    for i in range(len(h)):
        if random.random() < mutation_rate:
            h[i] += random.gauss(0, 0.05)
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize to sum to n_points/2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (len(h) / 2) / h_sum
    return h

def genetic_algorithm(n_points, pop_size, generations, initial_h, dx, budget_s):
    """Run genetic algorithm."""
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h)
    best_h = None
    best_fitness = float('inf')
    generations_without_improvement = 0

    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness for the whole population
        fitnesses = []
        for h in population:
            fit = evaluate_h(h, dx)
            fitnesses.append(fit)
            if fit < best_fitness:
                best_fitness = fit
                best_h = h.copy()
                generations_without_improvement = 0
        generations_without_improvement += 1

        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population through crossover and mutation
        new_population = []
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:pop_size//5]
        for i in elite_indices:
            new_population.append(population[i].copy())
            
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        # Early stopping if no improvement
        if generations_without_improvement > 100:
            break
    
    # Final evaluation of best solution
    if best_h is not None:
        final_fitness = evaluate_h(best_h, dx)
        return best_h, final_fitness
    else:
        return population[0], evaluate_h(population[0], dx)
