import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values):
    """Evaluate the solution using the provided evaluation function."""
    # Create a copy to avoid modifying the input
    h = np.array(h_values, dtype=float)
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral for each shift k
    overlap = corr * dx
    # C5 is the maximum of all overlaps over all shifts
    c5 = np.max(overlap)
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1, sum(h) = n_points/2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from provided initial_h_values if available, else create one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert len(h) == n_points
    else:
        h = np.full(n_points, 0.5 * dx)
    
    # Project to satisfy constraints
    h = project_h(h, n_points)
    
    # Evaluate initial solution
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # Per iteration settings
    sigma = 0.1
    decay = 0.999
    max_no_improvement = 1000
    no_improvement_count = 0
    
    # Start optimization loop
    while True:
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Generate perturbation
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        
        # Project new h
        h_new = project_h(h_new, n_points)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update h for next iteration
        h = h_new.copy()
        
        # Reduce variance for next iteration
        sigma *= decay
        
        # Early stopping condition for no improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Double-check final result
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), float(final_c5), n_points)
