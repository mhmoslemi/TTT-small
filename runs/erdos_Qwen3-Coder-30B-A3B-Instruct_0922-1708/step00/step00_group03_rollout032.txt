import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default construction: step function with 0.5 values
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        
    # Normalize initial values
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    # Variables
    x = cp.Variable(n_points)
    z = cp.Variable((n_points, n_points))
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cp.sum(x) == n_points / 2.0
    ]
    
    # z_{i,j} constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints.extend([
                z[i,j] <= x[i],
                z[i,j] <= 1 - x[j],
                z[i,j] >= x[i] + x[j] - 1
            ])
    
    # Objective
    C = []
    for k in range(-n_points+1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        corr_sum = cp.sum(z[np.arange(n_points)[valid], idx[valid]])
        C.append(corr_sum)
    
    t = cp.Variable()
    for i in range(len(C)):
        constraints.append(C[i] <= t)
    
    objective = cp.Minimize(t)
    prob = cp.Problem(objective, constraints)
    
    # Initialize with known solution
    x.value = h0
    z.value = np.outer(h0, 1-h0)
    
    try:
        # Solve the problem
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s)
        
        # Extract solution
        h_sol = np.array(x.value).flatten()
        h_sol = np.clip(h_sol, 0, 1)
        h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
        
        # Evaluate the actual C5
        C5_val = evaluate_erdos_solution(h_sol)
        
        # Ensure exact sum
        h_sol = h_sol * (n_points / 2.0) / np.sum(h_sol)
        
        return (h_sol, C5_val, n_points)
        
    except Exception as e:
        # If solver fails, return initial construction
        C5_val = evaluate_erdos_solution(h0)
        return (h0, C5_val, n_points)
