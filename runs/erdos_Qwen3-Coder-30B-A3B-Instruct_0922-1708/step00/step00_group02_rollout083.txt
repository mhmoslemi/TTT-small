import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Fallback to uniform construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # Ensure the sum is correct
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Optimization parameters
    step_size = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = None
    
    h = h0.copy()
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        dx = 2.0 / n_points
        h_aug = np.concatenate([h, np.zeros(n_points)])
        overlaps = np.correlate(h_aug, 1 - h, mode="full") * dx
        k_max = np.argmax(overlaps)
        c5 = overlaps[k_max]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
        
        # Gradient w.r.t. h for the worst shift k_max
        grad = np.zeros_like(h)
        for i in range(n_points):
            if i + k_max < n_points:
                grad[i] += (1 - h[i + k_max]) - h[i]
            else:
                # Handle the case where i + k_max >= n_points
                # This overlap is partial, need to be handled properly
                # approximation: ignore for this case, estimating negative impact
                continue
                
        # Update h
        h = h - step_size * grad
        # Project into [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to keep sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Decrease step size if no improvement for a while
        if iteration - last_improvement > patience:
            step_size *= 0.5
            last_improvement = iteration

    # Final evaluation to make sure we have the true value
    h_final = best_h if best_h is not None else h0
    h_aug_final = np.concatenate([h_final, np.zeros(n_points)])
    final_overlaps = np.correlate(h_aug_final, 1 - h_final, mode="full") * (2.0 / n_points)
    final_c5 = np.max(final_overlaps)

    return (h_final.tolist(), final_c5, n_points)
