import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    tau = 0.1  # temperature for smooth max
    max_iter = 1000
    patience = 50
    step_size_init = 0.1
    backtracking_factor = 0.5
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * target_sum
    else:
        h = np.full(n_points, 0.5 * dx)
        h = h / np.sum(h) * target_sum
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # Compute overlaps and gradients
        overlaps = compute_overlaps(h, dx)
        c5_value = np.max(overlaps)
        
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()
            last_improvement = iteration
            
        # Smooth max gradient computation
        exp_overlaps = np.exp(tau * overlaps)
        smooth_max = np.sum(exp_overlaps) / tau
        grad = np.zeros_like(h)
        
        # Compute gradient with respect to h
        for k in range(-n_points + 1, n_points):
            # Compute overlap for shift k
            if k >= 0:
                valid_h = h[:-k] if k > 0 else h[:]
                valid_h_shifted = h[k:] if k > 0 else h[:]
                overlap_k = np.sum(valid_h * (1 - valid_h_shifted))
            else:
                valid_h = h[-k:] if k < 0 else h[:]
                valid_h_shifted = h[:k] if k < 0 else h[:]
                overlap_k = np.sum(valid_h * (1 - valid_h_shifted))
            
            overlap_idx = k + n_points - 1
            grad_term = exp_overlaps[overlap_idx] / smooth_max
            # Add contribution to gradient
            if k >= 0:
                for i in range(len(valid_h)):
                    if i + k < len(h):
                        grad[i] += grad_term * (1 - h[i + k])
            else:
                for i in range(len(valid_h_shifted)):
                    if i < len(h) + k:
                        grad[i - k] += grad_term * (1 - h[i])
        
        # Normalize gradient
        grad = grad / np.linalg.norm(grad) * step_size_init if np.linalg.norm(grad) > 1e-8 else grad
        
        # Backtracking line search
        alpha = 1.0
        old_h = h.copy()
        while alpha > 1e-6:
            h_try = old_h - alpha * grad
            h_try = np.clip(h_try, 0, 1)
            h_try = h_try / np.sum(h_try) * target_sum
            
            try_c5 = evaluate_c5(h_try, dx)
            if try_c5 < c5_value:
                h = h_try
                break
            else:
                alpha *= backtracking_factor

        iteration += 1
        if iteration - last_improvement >= patience:
            break
            
    # Final validation
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    best_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), best_c5, n_points)


def compute_overlaps(h, dx):
    # Compute all overlaps for different shifts
    n_points = len(h)
    overlaps = np.zeros(2 * n_points - 1)
    
    # Use correlation to compute overlaps
    # h(x)(1 - h(x+k)) corresponds to correlation of h with (1 - h) with shift
    convolved = correlate(h, 1 - h, mode='full')
    # Extract the correct shifts
    overlaps = convolved[n_points-1:] * dx  # drops the negative shifts, only keep valid positive shifts
    return -overlaps  # make negative to match indexing


def evaluate_c5(h, dx):
    n = len(h)
    overlaps = np.zeros(2 * n - 1)
    
    # Use scipy correlate to compute overlaps efficiently
    convolved = correlate(h, 1 - h, mode='full')
    # The result of correlate with 'full' has size 2*n-1, 
    # and the valid range for the overlap (for non-negative shifts) is from index n-1 to 2*n-2.
    valid_overlaps = convolved[n-1:] * dx  # Valid overlaps for shifts [0, n-1]
    return np.max(valid_overlaps)
