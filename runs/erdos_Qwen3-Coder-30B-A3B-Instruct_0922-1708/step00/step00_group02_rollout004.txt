import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback to uniform distribution
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_h(h_vec):
        # Ensure the constraint ∫h(x)dx = 1 is satisfied
        current_sum = np.sum(h_vec) * dx
        h_scaled = h_vec * (target_sum / current_sum)
        
        # Compute overlaps using correlation
        h_folded = np.concatenate([h_scaled, np.zeros(n_points)])
        h_shifted = np.concatenate([np.zeros(n_points), 1 - h_scaled])
        
        correlation = np.correlate(h_folded, h_shifted, mode='valid')
        max_correlation = np.max(correlation) * dx
        
        return max_correlation, h_scaled
    
    # Gradient descent optimization
    def objective(h_vec):
        max_overlap, _ = evaluate_h(h_vec)
        return max_overlap
    
    def gradient(h_vec):
        # This is a simplified gradient estimation
        epsilon = 1e-6
        grad = np.zeros_like(h_vec)
        base_obj, _ = evaluate_h(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += epsilon
            obj_plus, _ = evaluate_h(h_plus)
            grad[i] = (obj_plus - base_obj) / epsilon
        return grad
    
    # Line search
    def line_search(h_vec, grad_vec, alpha_max=0.1):
        alpha = alpha_max
        current_obj = objective(h_vec)
        while alpha > 1e-8:
            h_new = h_vec - alpha * grad_vec
            # Project onto bounds
            h_new = np.clip(h_new, 0, 1)
            # Project onto constraint
            current_sum = np.sum(h_new) * dx
            h_new = h_new * (target_sum / current_sum)
            new_obj = objective(h_new)
            if new_obj <= current_obj:
                return h_new
            alpha *= 0.5
        return h_vec
    
    best_h = h.copy()
    best_c5 = objective(h)
    learning_rate = 0.01
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        grad = gradient(best_h)
        updated_h = line_search(best_h, grad, learning_rate)
        
        # Project back to valid space
        updated_h = np.clip(updated_h, 0, 1)
        current_sum = np.sum(updated_h) * dx
        updated_h = updated_h * (target_sum / current_sum)
        
        new_c5 = objective(updated_h)
        
        if new_c5 < best_c5:
            best_h = updated_h
            best_c5 = new_c5
        else:
            # If no improvement, reduce learning rate
            learning_rate *= 0.95
            
            # Occasionally add noise to escape local minima
            if iteration % 50 == 0:
                noise_scale = 0.001
                noise = np.random.normal(0, noise_scale, len(best_h))
                noisy_h = best_h + noise
                noisy_h = np.clip(noisy_h, 0, 1)
                noisy_sum = np.sum(noisy_h) * dx
                noisy_h = noisy_h * (target_sum / noisy_sum)
                noisy_c5 = objective(noisy_h)
                if noisy_c5 < best_c5:
                    best_h = noisy_h
                    best_c5 = noisy_c5
                    
        if learning_rate < 1e-8:
            break
    
    # Final check
    final_c5, final_h = evaluate_h(best_h)
    
    return (final_h.tolist(), final_c5, n_points)
