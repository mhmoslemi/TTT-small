import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial construction if available, otherwise create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
        
    # Objective function: C_5 bound
    def compute_c5(h):
        h = np.asarray(h)
        # Correlation with flipped function
        conv = np.correlate(h, 1 - h, mode='full')
        # Take max and scale by dx
        max_val = np.max(conv) * dx
        return max_val
    
    # Constraint functions
    def integral_constraint(h):
        return np.sum(h) - target_sum
    
    def bounds_constraints(h):
        return np.concatenate([h, 1 - h])
    
    def optimize_step(h, max_iter=1000):
        # Fast iterative optimization using Projected Gradient Descent
        best_h = h.copy()
        best_c5 = compute_c5(h)
        last_improvement = 0
        for iter in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Approximate gradient via finite differences
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1: h_plus[i] = 1
                if h_plus[i] < 0: h_plus[i] = 0
                c5_plus = compute_c5(h_plus)
                
                h_minus = h.copy()
                h_minus[i] -= eps
                if h_minus[i] > 1: h_minus[i] = 1
                if h_minus[i] < 0: h_minus[i] = 0
                c5_minus = compute_c5(h_minus)
                
                grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
            # Update with gradient step
            step_size = 0.01
            h_new = h - step_size * grad
            
            # Project to valid bounds [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Project to correct integral
            current_sum = np.sum(h_new)
            if abs(current_sum - target_sum) > 1e-6:
                scale = target_sum / current_sum
                h_new = h_new * scale
                
            # Compute new C5
            c5_new = compute_c5(h_new)
            
            # Accept move if improvement
            if c5_new < best_c5:
                h = h_new.copy()
                best_c5 = c5_new
                last_improvement = 0
            else:
                last_improvement += 1
                if last_improvement > 50:
                    break
                    
            # Try different strategies if stuck
            if last_improvement > 20 and iter % 20 == 0:
                # Random perturbation
                h = np.clip(h + (np.random.rand(n_points) - 0.5) * 0.05, 0, 1)
                # Rescale
                h = h / np.sum(h) * target_sum
                
            if iter % 500 == 0:
                # Occasionally recompute best
                current_c5 = compute_c5(h)
                if current_c5 < best_c5:
                    best_c5 = current_c5
                    best_h = h.copy()
                if time.time() - start_time > budget_s - 1:
                    break
                    
        # Final check
        final_c5 = compute_c5(best_h)
        return best_h, final_c5
    
    # Initial optimization
    h, c5 = optimize_step(h.copy())
    
    # Try a few restarts with different starting points
    for i in range(3):
        if time.time() - start_time > budget_s - 1:
            break
        # Perturb current best
        perturbed = h + (np.random.rand(n_points) - 0.5) * 0.02
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed / np.sum(perturbed) * target_sum
        h_new, c5_new = optimize_step(perturbed)
        if c5_new < c5:
            h, c5 = h_new, c5_new
    
    return (list(h), c5, n_points)
