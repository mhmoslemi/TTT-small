import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2

    # Basis function parameters
    m = int(2 * np.sqrt(n_points))
    centers = np.linspace(0, 2, m, endpoint=False)
    scales = np.full(m, 2.0 / m)

    def sigmoid(x, center, scale):
        return expit((x - center) / scale)

    def basis_functions(x):
        return np.array([sigmoid(x, center, scale) for center, scale in zip(centers, scales)])

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = h_init / np.sum(h_init) * target_sum
    else:
        h_init = np.full(n_points, 0.5)

    # Optimization loop
    best_h = None
    best_c5 = np.inf

    def overlap_objective(a):
        # Reconstruct h from basis coefficients
        x = np.linspace(0, 2, n_points)
        h = a[0] + np.dot(a[1:], basis_functions(x))
        h = np.clip(h, 0, 1)
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode='full')
        return -np.max(overlaps) * dx

    def overlap_gradient(a):
        x = np.linspace(0, 2, n_points)
        h = a[0] + np.dot(a[1:], basis_functions(x))
        h = np.clip(h, 0, 1)
        # Compute overlap integrals efficiently
        overlaps_full = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(overlaps_full)
        # Extract k corresponding to max overlap
        k = max_k - (n_points - 1)
        # For this lag k, compute gradient
        grad = np.zeros_like(a)
        # Gradient w.r.t. a_0
        grad[0] = -2 * np.sum(1 - h) * dx  # constant term
        # Gradient w.r.t. a_j for j=1..m
        if k >= 0:
            h_lag = np.concatenate([np.zeros(k), h[:-k]])
        else:
            h_lag = np.concatenate([h[-k:], np.zeros(-k)])
        for j in range(m):
            phi_j = basis_functions(x)[j]
            grad[1+j] = -2 * np.sum(phi_j * (1 - h_lag)) * dx
        # Elastic net regularization
        gamma = 0.8
        grad[1:] -= gamma * np.sign(a[1:])
        return grad

    # Optimization with time control
    a_cur = np.random.uniform(-0.1, 0.1, size=m+1)
    a_cur[0] = 0.5  # initial guess for constant part
    
    # Scale to meet the integral constraint
    x = np.linspace(0, 2, n_points)
    h_temp = a_cur[0] + np.dot(a_cur[1:], basis_functions(x))
    h_temp = np.clip(h_temp, 0, 1)
    integral = np.sum(h_temp) * dx
    a_cur[0] += (target_sum - integral) / dx

    max_iter = 500
    iter_count = 0
    
    # Gradient descent with regularization
    learning_rate = 0.01
    for i in range(max_iter):
        if time.time() - start_time > 0.95 * budget_s:
            break

        old_h = a_cur[0] + np.dot(a_cur[1:], basis_functions(x))
        old_h = np.clip(old_h, 0, 1)
        old_c5 = -np.max(np.correlate(old_h, 1 - old_h, mode='full')) * dx

        grad = overlap_gradient(a_cur)
        a_cur -= learning_rate * grad
        a_cur[1:] -= 0.8 * np.sign(a_cur[1:])  # Elastic net L1

        # Project to feasible region
        h_new = a_cur[0] + np.dot(a_cur[1:], basis_functions(x))
        h_new = np.clip(h_new, 0, 1)
        integral = np.sum(h_new) * dx
        a_cur[0] += (target_sum - integral) / dx

        c5 = -np.max(np.correlate(h_new, 1 - h_new, mode='full')) * dx
        if c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = c5

        iter_count += 1
        if i % 100 == 0 and i > 0:
            learning_rate *= 0.5

    if best_h is None:
        x = np.linspace(0, 2, n_points)
        h_final = np.full(n_points, 0.5)
        h_final = np.clip(h_final, 0, 1)
        best_h = h_final
        best_c5 = -np.max(np.correlate(h_final, 1 - h_final, mode='full')) * dx
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    final_sum = np.sum(h_final) * dx
    h_final = h_final * target_sum / final_sum

    return (h_final, best_c5, n_points)
