import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given step function h."""
    full_corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(full_corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy sum(h) = n_points / 2 and 0 <= h <= 1."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using either initial_h_values or create sinusoidal."""
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to fit n_points
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 2))
        h = np.clip(h, 0, 1)
    h = project_h(h, n_points)
    return h

def cross_correlate_gradient(h, dx):
    """Compute the gradient of max correlation with respect to h."""
    n = len(h)
    # Compute all correlations for shifts k = 0, 1, ..., 2*n-2
    corr = correlate(h, 1 - h, mode="full")
    max_idx = np.argmax(corr)
    max_shift = max_idx - (n - 1)
    
    # For max_shift, compute gradient via finite differences
    grad = np.zeros_like(h)
    eps = 1e-6
    base_corr = evaluate_h(h, dx)
    
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, len(h))
        corr_plus = evaluate_h(h_plus, dx)
        grad[i] = (corr_plus - base_corr) / eps
    
    return grad

def gradient_descent(h, n_points, dx, max_iter=2000, lr=0.01):
    """Improve h with gradient descent."""
    h = h.copy()
    prev_c5 = evaluate_h(h, dx)
    converged = False
    improvement_count = 0
    lr_factor = 1.0
    
    for i in range(max_iter):
        if i % 1000 == 0 and i > 0:
            lr_factor *= 0.5
        grad = cross_correlate_gradient(h, dx)
        h -= lr * lr_factor * grad
        h = project_h(h, n_points)
        c5 = evaluate_h(h, dx)
        if c5 < prev_c5:
            prev_c5 = c5
            improvement_count = 0
        else:
            improvement_count += 1
        if improvement_count >= 50:
            break
    return h, prev_c5

def genetic_algorithm(n_points, dx, pop_size=50, num_generations=100):
    """Run genetic algorithm to find good starting point."""
    def create_individual():
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        return h
    
    def evaluate_individual(h):
        return evaluate_h(h, dx)
    
    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    fitness = [evaluate_individual(h) for h in population]
    
    best_h = population[np.argmin(fitness)]
    best_c5 = min(fitness)
    
    for generation in range(num_generations):
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        elite_count = pop_size // 5
        elite = population[:elite_count]
        
        # Generate offspring
        offspring = []
        for _ in range(pop_size - elite_count):
            parent1, parent2 = np.random.choice(population[:pop_size//2], 2, replace=False)
            # Crossover
            crossover_point = np.random.randint(1, n_points-1)
            child = np.concatenate((parent1[:crossover_point], parent2[crossover_point:]))
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] += np.random.normal(0, 0.05)
                    child[i] = np.clip(child[i], 0, 1)
            child = project_h(child, n_points)
            offspring.append(child)
        
        # Replace population
        population = elite + offspring
        fitness = [evaluate_individual(h) for h in population]
        
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)]
    
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    
    # Phase 1: Genetic Algorithm
    h, c5 = genetic_algorithm(n_points, dx, pop_size=50, num_generations=50)
    
    # Phase 2: Gradient Descent
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 60:
        h, c5 = gradient_descent(h, n_points, dx, max_iter=1000, lr=0.01)
    
    return (h, c5, n_points)
