import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a simple starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
        h[2] = 0.7
        h[3] = 0.3
        h[4] = 0.8
        h[5] = 0.2
        h[6] = 0.9
        h[7] = 0.1
        h[8] = 0.5
        h[9] = 0.5

    # Normalize to sum to n_points/2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Parameters
    eta = 0.01
    max_iter = 6000
    patience = 100
    patience_count = 0
    last_improvement = 0
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Find the maximum overlap
        c5_vals = np.correlate(h, 1-h, mode="full") * dx
        max_k = np.argmax(c5_vals)
        max_val = c5_vals[max_k]
        
        # Compute gradient for this max_k
        grad = np.zeros(n_points)
        if max_k < n_points:
            # h[i] * (1 - h[i + k]) contribution
            for j in range(n_points - max_k):
                grad[j] += (1 - h[j + max_k]) * dx
        else:
            # h[i] * (1 - h[i - k]) contribution
            k = max_k - n_points + 1
            for j in range(k, n_points):
                grad[j] += -h[j - k] * dx
                
        # Update h
        h = h - eta * grad
        
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Adjust to maintain sum constraint
        h_sum = np.sum(h)
        adjustment = (n_points / 2 - h_sum) / n_points
        h = h + adjustment
        
        # Ensure h stays in [0, 1]
        h = np.clip(h, 0, 1)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = i
            patience_count = 0
        else:
            patience_count += 1
            
        # Adaptive learning rate
        if i % 100 == 0 or patience_count > 50:
            eta *= 0.95
            
        if patience_count > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute C5 as maximum of all overlaps from correlation
    c5_vals = np.correlate(h, 1-h, mode="full") * dx
    return np.max(c5_vals)
