import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    lr = 0.01
    beta = 0.9
    improvement_threshold = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        
    h = h / np.sum(h) * (n_points / 2)
    h = np.clip(h, 0, 1)
    
    # Momentum terms
    v_h = np.zeros_like(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute cross-correlation: h(x)(1 - h(x+k))
        # Using full mode to get all possible shifts
        corr = correlate(h, 1 - h, mode='full')
        # The correlation array is doubled in length, but we're interested in the overlap values
        # The mid part represents k from 0 to 2 in steps of dx
        corr_len = len(corr)
        half_len = corr_len // 2
        overlap_values = corr[half_len:]
        
        # Find maximum overlap
        k_max = np.argmax(overlap_values)
        max_overlap = overlap_values[k_max]
        
        # Compute gradient w.r.t h
        # Gradient of the overlap integral w.r.t h(x) is (1 - h(x+k)) - h(x+k)
        # We compute this over all x, considering the shift
        grad_h = np.zeros_like(h)
        
        # Approximate gradient using finite differences or direct correlation
        # The gradient of \int h(x)(1 - h(x+k)) dx w.r.t h(x) is (1 - h(x+k)) on the lagged array
        # We compute h(x+k) for different k
        # Let's compute it element-wise:
        # For fixed k, gradient of overlap w.r.t h_i = (1 - h_{i+k}) if we're looking at x_i,
        # But we need to handle the wrap carefully, here we use convolution approach
        # Gradient via finite differences: 
        # Let's compute the mapping: d/dh_i of integral = 1-h_{i+k} [if considering the shift valid]
        # For every x_i, we contribute 1-h_{i+k} to gradient of h_i when considering shift k_max
        # But g(x_i) = \int h(x)(1 - h(x+k)) dx. 
        # Optimize with respect to h: let's build Jacobian more carefully.
        # Actually, for fixed k_max:
        # grad_{h_i} = \int_{x} d/dh_i [h(x)(1 - h(x+k))] dx 
        # = \int_{x} [1_{x=x_i}(1 - h(x+k)) - h(x) \delta_{x+k = x_i}] dx
        # For x_i, we contribute (1 - h_{i+k_max}) + h_{i-k_max} - h_i * (\delta_{x_i = x_i+k_max})
        # Turn out correct: each overlap term is h(x)(1- h(x+k)), gradient w.r.t h_i:
        # If x=k_max+i, then term h(x)(1 - h(x+k)) = h(i)(1 - h(k_max+i))
        # Grad = (1 - h_{i+k_max}) + (since it's only non-zero for h_i multiplied)
        # Actually, we can build gradient directly from cross-correlation involving h
        # We already have relabeling with 1-h, so directly use:
        h_shifted = np.zeros_like(h)
        for i in range(n_points):
            j = (i + k_max) % n_points
            h_shifted[i] = h[j]
            
        grad_h = 1 - h_shifted
        
        # Gradient descent step with momentum
        v_h = beta * v_h + (1 - beta) * grad_h
        h = h - lr * v_h
        
        # Projection onto constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        
        # Check if this iteration improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            last_improvement = iter_count
            
        # Update learning rate
        if iter_count - last_improvement > improvement_threshold:
            lr *= 0.5
            last_improvement = iter_count
            
        iter_count += 1
    
    # Final verification of best solution
    corr_best = correlate(best_h, 1 - best_h, mode='full')
    half_len = len(corr_best) // 2
    overlap_vals = corr_best[half_len:]
    final_c5 = np.max(overlap_vals)
    
    # Make sure final_c5 is computed correctly via the evaluator
    # use the same approach that the test harness does:
    final_h = best_h.copy()
    final_c5 = np.max(np.correlate(final_h, 1 - final_h, mode='full')) * dx
    
    return (final_h, final_c5, n_points)
