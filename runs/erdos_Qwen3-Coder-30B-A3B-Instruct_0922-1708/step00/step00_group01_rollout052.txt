import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
    except NameError:
        # Create symmetric plateau initial h
        n_points = 47
        h = np.ones(n_points)
        mid = n_points // 2
        h[:mid] = 0.5
        h[mid:] = 0.5
        h = h * 2 / n_points  # scale to sum = 1

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # SGD params
    learning_rate = 0.1
    anneal_interval = 1000
    max_iter = 5000
    iteration = 0

    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        # Compute all overlaps
        h_rev = h[::-1]
        overlaps = correlate(h, 1 - h_rev, mode='full')
        # Adjust for dx and find max
        overlaps *= dx
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # Find gradient w.r.t. h for this max overlap lag
        # For correlation at lag L, gradient d/dh_i of h[i] * (1 - h[i+L]) 
        # is (1 - h[i+L]) if i+L < n_points, 0 otherwise
        # Since we compute full correlation, lag=0 means h[i] * (1 - h[i])
        # Lag L means h[i] * (1 - h[i+L]) (when i+L < n_points)
        # To compute finite diff gradient: We use only dominant contribution 
        # (i.e., i where max overlap is obtained)
        
        # Consider maximizing for a single dominant lag
        lag = max_idx - (n_points - 1)  # convert from full index to lag
        gradient = np.zeros_like(h)
        if lag == 0:
            gradient = 1 - 2 * h
        elif lag > 0:
            # gradient[i] += 1 - h[i + lag] if i + lag < n_points
            for i in range(n_points - lag):
                gradient[i] += 1 - h[i + lag]
        elif lag < 0:
            # gradient[i] += 1 - h[i + lag] if i + lag >= 0
            for i in range(-lag, n_points):
                gradient[i] += 1 - h[i + lag]
        
        # Update
        h = h - learning_rate * gradient
        
        # Project to feasible
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum  # Rescale so sum equals n/2

        # Check improvement
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Annealing
        if iteration > 0 and iteration % anneal_interval == 0:
            learning_rate *= 0.5
            
        iteration += 1

    # Final check and return
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    h_rev = h[::-1]
    overlaps = correlate(h, 1 - h_rev, mode='full')
    overlaps *= dx
    return np.max(overlaps)
