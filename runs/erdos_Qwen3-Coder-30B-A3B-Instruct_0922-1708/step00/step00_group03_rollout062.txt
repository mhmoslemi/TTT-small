import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

# Global variables
n_points = 70
dx = 2.0 / n_points

def evaluate_overlap(h):
    """Evaluate the maximum overlap for the given h"""
    h = np.array(h)
    # Compute convolution of h and (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    # Extract the relevant part (k from -n+1 to n-1)
    mid = len(conv) // 2
    max_overlap = np.max(conv[mid - n_points + 1:mid + n_points])
    # Scale by dx
    return max_overlap * dx

def initial_h():
    """Initialize h from initial_h_values or create a simple construction"""
    try:
        return np.array(initial_h_values)
    except NameError:
        # Simple initial construction
        h = np.ones(n_points) * 0.5
        return h

def project_to_feasible(h):
    """Project h to satisfy constraints (0 <= h <= 1, sum = n/2)"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    # Scale to have the correct sum
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def eval_objective_and_gradient(h, reg_lambda=1e-4):
    """Evaluate objective and gradient using proximal method"""
    h = np.array(h)
    h = project_to_feasible(h)
    # Compute overlap for each shift
    conv = np.correlate(h, 1 - h, mode='full')
    mid = len(conv) // 2
    overlaps = conv[mid - n_points + 1:mid + n_points]
    max_k = np.argmax(overlaps)
    max_overlap = overlaps[max_k]
    
    # Gradient of max_overlap w.r.t. h (approximated)
    # For k=0, the gradient at position i is (1 - h[i]) 
    # But since we're looking at max, use a simplified approach
    # We will compute an approximated gradient with a small perturbation
    eps = 1e-5
    # Construct gradient of the maximum term only
    gradient = np.zeros_like(h)
    # Gradient contribution from current maximum value
    if max_k < n_points - 1:
        # Backward shift: [0,1,2,...,k]
        k = max_k
        for i in range(n_points):
            j = (i - k) % n_points
            if 0 <= j < n_points:
                gradient[i] += 1 - h[j]
    else:
        # Forward shift: [k, k+1, k+2,..., n-1]
        k = max_k - (n_points - 1)
        k = -k  # convert back to positive shift
        for i in range(n_points):
            j = (i + k) % n_points
            if 0 <= j < n_points:
                gradient[i] += 1 - h[j]
    
    # Add regularization
    gradient += 2 * reg_lambda * h
    
    return max_overlap, gradient

def line_search(f_and_grad, h, direction, alpha_max=1.0, max_iter=10):
    """Perform line search along the given direction"""
    best_f = float('inf')
    best_h = h.copy()
    for i in range(max_iter):
        alpha = alpha_max * (0.5 ** i)
        new_h = h + alpha * direction  
        new_h = project_to_feasible(new_h)
        f, _ = f_and_grad(new_h)
        if f < best_f:
            best_f = f
            best_h = new_h.copy()
        else:
            break
    return best_h

def run(seed=42, budget_s=600):
    """Main optimization loop"""
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    h = initial_h()
    h = project_to_feasible(h)
    
    # Parameters
    reg_lambda = 1e-4
    max_iter = 5000
    tolerance = 1e-6
    last_improvement = 0
    
    # Iterative optimization
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate current state
        current_val, gradient = eval_objective_and_gradient(h, reg_lambda)
        
        # Stop if gradient norm is small
        grad_norm = np.linalg.norm(gradient)
        if grad_norm < tolerance:
            break
            
        # Determine search direction
        direction = -gradient
            
        # Line search
        new_h = line_search(eval_objective_and_gradient, h, direction, alpha_max=0.1)
        
        # Update
        h = project_to_feasible(new_h)
        
        # Local shift search (occasionally, every 20 iterations)
        if iteration % 20 == 0:
            best_h = h.copy()
            best_val = current_val
            for shift in range(-5, 6):
                h_shifted = np.roll(h, shift)
                h_shifted = project_to_feasible(h_shifted)
                val = evaluate_overlap(h_shifted)
                if val < best_val:
                    best_val = val
                    best_h = h_shifted.copy()
            h = best_h.copy()
    
    # Final evaluation
    final_val = evaluate_overlap(h)
    
    return (h.tolist(), final_val, n_points)
