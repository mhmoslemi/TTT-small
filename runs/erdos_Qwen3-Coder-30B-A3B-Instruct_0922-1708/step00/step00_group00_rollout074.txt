import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h values."""
    # Compute correlation: h * (1 - h) with full mode
    corr = correlate(h, 1 - h, mode="full")
    # Scale by dx and find maximum
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points / 2"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Perturbation parameters
    sigma = 0.1
    decay_factor = 0.99
    patience = 1000
    patience_counter = 0

            best_c5 = c5_new

            patience_counter += 1
        
        # Decrease sigma
        sigma *= decay_factor
        
        # Early termination
        if patience_counter >= patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
