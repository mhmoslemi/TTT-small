import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral
    conv = correlate(h, 1 - h, mode="full")
    max_val = np.max(conv) * dx
    return max_val

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction or create a uniform one
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5 * (2.0 / n_points))
    
    # Ensure integral constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 0.1
    max_iter = 1000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    # Objective function using convex surrogate
    def surrogate_obj(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode="full")
        # Use log-sum-exp to approximate max
        max_vals = conv * dx
        return np.log(np.sum(np.exp(alpha * max_vals))) / alpha
    
    def surrogate_grad(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode="full")
        max_vals = conv * dx
        # Gradient of log-sum-exp approximation
        exp_vals = np.exp(alpha * max_vals)
        sum_exp_vals = np.sum(exp_vals)
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Compute derivatives of the convolution
            grad_conv = np.zeros_like(h_vec)
            for j in range(len(h_vec)):
                if 0 <= i + j < len(h_vec):
                    grad_conv[j] = -1.0 if i + j >= 0 else 0.0
                if 0 <= i - j < len(h_vec):
                    grad_conv[j] += 1.0 if i - j >= 0 else 0.0
            # Note: this is a simplified version for performance
            # A more accurate gradient would involve complex derivatives
            grad += (exp_vals * grad_conv * dx / sum_exp_vals)
        # Final normalization and adjust for scaling constraint
        # This is partially estimating gradient for scaling constraint
        scale_factor = np.sum(h_vec) / (n_points / 2)
        grad = grad / scale_factor + np.sum(grad) * (h_vec - np.mean(h_vec)) / (n_points * scale_factor)
        return grad
    
    # Initial evaluation
    current_c5 = evaluate_c5(h, dx)
    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = h.copy()
    
    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Optimize using scipy with bounds and trust-constr
        bounds = [(0, 1) for _ in range(n_points)]
        res = minimize(
            surrogate_obj,
            h,
            method='L-BFGS-B',
            bounds=bounds,
            jac=surrogate_grad,
            options={'maxiter': 10}
        )
        
        if not res.success:
            break
            
        h = np.clip(res.x, 0, 1)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate new solution
        new_c5 = evaluate_c5(h, dx)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Early stopping if no improvement
        if iteration - last_improvement > patience:
            break
            
        # Occasionally perturb the solution to aid escape from local optima
        if iteration % 50 == 0 and iteration > 0:
            noise = np.random.uniform(-0.05, 0.05, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
