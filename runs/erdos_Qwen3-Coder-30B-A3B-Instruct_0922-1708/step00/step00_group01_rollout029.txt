import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute all overlaps using cross-correlation
    # h is a step function: h[i] gives height at position i*dx
    # 1-h is also step function with 1-h[i] at position i*dx
    # Cross-correlation computes integrals like int h(x)(1 - h(x+k)) dx
    # for various lags k
    # corrs = cross-correlation of h and (1-h) 
    corrs = correlate(h, 1 - h, mode='full')
    # Note that we're computing sum_{i} h[i]* (1-h[i+k]) * dx
    # This is exactly the cross-correlation scaled by dx
    max_corr = np.max(corrs) * dx
    return max_corr

def project_h(h):
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet integral constraint: sum(h)*dx = 1 
    # => sum(h) = n_points / 2
    n_points = len(h)
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def compute_gradient(h, dx, k_max):
    # Compute gradient of max overlap w.r.t. h
    # We compute the dominant max overlap first
    corrs = correlate(h, 1 - h, mode='full')
    k_star = np.argmax(corrs)
    # For simplicity compute full gradient based on k_star
    # This approximates the gradient towards k_star
    # gradient terms: for h[i], it affects terms with k_star = j - i
    # so we look for exact indices and accumulate
    # More precisely, for lag k=n_points-1-k_star
    # d/dh[i] (int h(x) (1-h(x+k)) dx) = (1-h[i+k]) * dx (if we don't care for boundary)
    # But we use finite differences or simpler method:
    # This is indeed approximated as just extended h of size 2*n - 1
    # and we take the contribution from the shifted version (1-h shifted) 
    # gradient_i = d/dh_i ( sum_j h[j] (1-h[j+k_star]) ) = (1-h[j_max]) * dx where j_max = i + k_star
    # Actually, more careful:
    # Effect of change in h[i] on overlap with lag k=0:
    # h[x] * (1-h[x+0]) => d/dh[i] * (h[i] * (1-h[i])) = 1-2*h[i]
    # For overlap with lag k, that effect is: 
    # Sum of contributions where h[j] (1-h[j+k]) is an entry 
    # i.e. for all k, what is the effect of changing h[i]?
    # the j term gives h[j] (1-h[j+k]), so derivative wrt h[i] is 0 if i != j-k
    # if i = j-k, then j = i+k, so contribution to overlap is h[i+k] (1-h[i])  
    # so d/dh[i] (h[j] (1-h[j+k])) = (1-h[i]) if j = i+k else 0
    # So for overlap with lag k_star, the gradient w.r.t h[i] is:
    # This is tricky to compute precisely w.r.t. general k, so let's simplify
    # We'll compute from that term directly: for h[0], we get the d/dh[0] 
    # ex: h[0] (1-h[k_star]) <= depends on h[0] if we have a term like h[i] (1-h[i+k])
    # So derivative of term h[j] (1-h[j+k]) with respect to h[i] is:
    # if i = j, then derivative of h[j] (1-h[j+k]) = (1-h[j+k])  (if j==i)  
    # if i = j+k, then derivative of h[j] (1-h[j+k]) = -h[j] = -h[i-k] 
    # For lag k_star = n_points - 1 - k_idx, we examine indeed those:
    # At h[i], it only affects om of lag k_star's cross term iff i = j + k_star, i.e. j= i-k_star
    # so the weight of h[i] in the overlap with k_star is: 
    # Contribution is h[j] * (1-h[j+k_star]) = h[i-k_star] * (1-h[i]), 
    #   => the derivative of this is d/dh[i] = - (1-h[i]) if i-k_star >= 0, but also 0 if i-k_star < 0 
    # But we also compute gradient from the first term:
    # For h[j] (1-h[j+k]), wrt h[i], the result is (1-h[j+k]) if i=j, else 0
    # => so term h[i] (1-h[i+k_star]) has derivative -h[i+k_star] (if i+k_star < n) 
    # The gradient w.r.t h[i] of max overlap is from max term only:
    # d/dh[i] (max_{k} int h(x)(1-h(x+k)) dx) is the contribution of that max lag.
    # But let's use a different approach. Compute for any i, what the total effect is using finite diff.
    # gradient applied as - learning_rate * (all grad terms for each i where effect occurs)
    # So computing: d/dh_i [sum_k h[j] (1-h[j+k]) * dx] over all possible k 
    # For k=0: sum_j h[j] (1-h[j]) => d/dh_i: (1-2h[i])  
    # For k=1: sum_j h[j] (1-h[j+1]) => d/dh_i = -h[i+1] if i+1 < n, 
    #   0 if not, (or -h[i-1] if k=-1?)
    # More robust way using fixed finite diff:  
    # Compute gradient based on finite differences  
    # We will compute Jacobian of overlaps w.r.t h-vector directly and get a vector.  
    # Instead we compute approximate gradient. 
    # Approx But we'll essentially focus on most dominant term.  
    # For overlap integral over a fixed k, when we vary h[i], 
    # this is h[i] * (1-h[i+k]) and maybe also hold at indices depending on whether h[i] appears in the product.
    # If h[i] is the middle of integral h[j] (1-h[j+k]), and we fix h,  
    # the only such it's a product is for h[i] and h[i+k], so d/dh[i] ( h[i] (1-h[i+k]) ) = 1 - h[i+k]
    # Similarly for h[i+k] term as (h[i+k] - h[i] h[i+k])  
    # d/dh[i] (h[i] (1-h[i+k])) = 1 - h[i+k]
    # But when considering the Cross-correlation, it is sum over j of h[j] (1-h[j+k]) 
    # so d/dh[i] (sum_j h[j] (1-h[j+k])) = \sum_{j} (1-h[j+k]) * delta_{i,j} = (1-h[i+k])  
    # Ideal gradient: if k = k_star fixed for maximum overlap 
    # d/dh[i] of overlap_lag_k = sum_j h[j] (1-h[j+k]) = 
    # derivative: (1-h[i+k]) if k is fixed
    n = len(h)
    k = k_star - n + 1
    # This needs correction. The cross-correlation computes: 
    # [ ... ] [k: 0 to 2*n-2]
    # sum_j h[j] (1-h[j+k_star]) 
    # computed with indices j in [0:n-1], j+k_star in [k_star, n+k_star-1]
    # so a valid j is in [0:n-1] AND j+k_star in [0:n-1]  
    # So j in [0:n-1] AND j in [0:n-1-k_star] => j in [0:min(n-1, n-1-k_star)]  
    # For overlap lag k_star, we want d/dh[i] of sum of all  h[j] (1-h[j+k_star]) 
    # = sum_{j : j, j+k_star in bounds} (1-h[j+k_star]) * delta_{i,j}  
    # = (1-h[i+k_star]) under assumption that i, i+k_star are both valid indices in [0,n-1]
    # If k_star < 0, then j+k_star < j, so for j such that j+k_star >=0, this is:  
    # The overlap integral (with lag k) is  sum_{j} h[j] (1-h[j+k]) 
    # For h[k] to be involved, it requires j_k + k = i, i.e. j = i-k  
    # => 
    # To compute gradient of overlap over given lag k:
    # Let sum of such products = sum_j h[j] (1-h[j+k]) 
    # d/dh[i] (sum_j h[j] (1-h[j+k])) =  
    # if j=i: (1-h[i+k])  
    # else 0  
    # But we are worried that i+k may be beyond constraint.  
    # Keep the indices in range.
    # Make r = range(n) 
    # Compute 1-h[i+k] for valid i,k such that i+k < n and -k <= i 
    # this is j = i reduced so that i+k < n => i < n-k, and i >= 0,  
    # So valid range for i, given fixed k, is [0, n-1-k],  
    # but also j=-k is valid => i = j+k >= 0 => k >= -j => k >= - (n-1) = 1-n  
    # so valid k is in [1-n, n-1]
    # The grad is then simple if k = k_star:
    grad = np.zeros_like(h)
    for i in range(n):
        # effect of h[i] on overlap at lag k_star
        j = i + k_star - n + 1  # Back to real correlation index.
        # The real lag k= k_star - (n-1) = k_star - n + 1
        # Compute for h[i], amount of contribution to self-correlation:
        # h[i] (1-h[i+k_star])  
        # so d/dh[i] = 1 - h[i+k_star] 
        # this is: if i+k_star < n, then gradient += (1 - h[i+k_star]) 
        # these are the gradient terms contributing to overlap with lag k_star
        k_true = k_star - (n - 1)  # inverse of correlation lag formula  
        if i + k_true < n and i + k_true >= 0:
            grad[i] = 1 - h[i + k_true]
    return grad

    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    current_h = h.copy()

    for k in range(num_iters):
        current_time = time.time()
        if current_time - start_time > budget_s - 5: # leave 5s margin 
            break

        # Evaluate current solution
        c5 = evaluate_h(current_h, dx)
        if c5 < best_c5:
            best_h = current_h.copy()
            best_c5 = c5

        if k % 1000 == 0 and k > 0:
            # Anneal learning rate
            learning_rate *= 0.5

        # Compute gradient with respect to dominant lag
        corr = correlate(current_h, 1 - current_h, mode='full')
        k_best = np.argmax(corr)
        k_true = k_best - n_points + 1  # back to core lag from correlation index
        
        grad = np.zeros(n_points) 
        # Subtract the effective gradient for dominant lag k_best  
        # i.e., for terms h[i] (1-h[i+k_true]),  
        # gradient of overlap w.r.t h[i] is 1 - h[i+k_true]  
        for i in range(n_points):
            j = i + k_true  # pointed term
            if 0 <= j < n_points:
                grad[i] = 1 - current_h[j]
            else:
                grad[i] = 0

        # Update - gradient descent step
        current_h = current_h - learning_rate * grad
        # Project back to feasible space
        current_h = project_h(current_h)

    # Final round-up
    best_c5 = evaluate_h(best_h, dx)
    return (best_h.tolist(), best_c5, n_points)
