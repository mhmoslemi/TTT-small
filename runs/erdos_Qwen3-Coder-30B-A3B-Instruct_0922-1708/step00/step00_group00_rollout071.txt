import numpy as np
import time
from scipy import correlate

def evaluate_h(h_values, dx):
    """Evaluate the maximum overlap integral for given h values."""
    # Compute correlation of h with (1-h)
    corr = correlate(h_values, 1 - h_values, mode='full')
    # Extract the relevant part (shifts k from -n+1 to n-1)
    n = len(h_values)
    corr = corr[n-1:2*n-1]  # This corresponds to k from -n+1 to n-1
    # Multiply by dx to get the actual integral values
    overlap_values = corr * dx
    # Return the maximum overlap
    return np.max(overlap_values)

def project_h(h, target_sum, dx):
    """Project h to satisfy sum constraint and bound constraints."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        # Scale h to match target sum
        h = h * (target_sum / current_sum)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    else:

    h = project_h(h, target_sum, dx)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    no_improvement_count = 0
    
    # Decreasing variance for perturbation
    sigma_start = 0.1
    sigma_decay = 0.99
    sigma = sigma_start
    
    iteration = 0
    while time.time() - start_time < budget_s and no_improvement_count < 1000:
        # Generate perturbed h
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        # Project to feasible region
        h_new = project_h(h_new, target_sum, dx)
        
        # Evaluate new solution
        c5_new = evaluate_h(h_new, dx)
        
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Update h for next iteration
        h = h_new.copy()
        # Decay sigma
        sigma *= sigma_decay
        iteration += 1
    
    return (best_h.tolist(), best_c5, n_points)
