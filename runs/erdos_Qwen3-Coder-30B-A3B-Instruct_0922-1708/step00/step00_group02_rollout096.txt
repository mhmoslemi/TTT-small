import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    improvement_threshold = 500
    
    # Initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) / n_points
    
    # Normalize to sum to 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Momentum
    v_h = np.zeros_like(h)
    
    for i in range(max_iter):
        # Check time budget
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlap integral for all shifts
        # h(x)(1 - h(x+k)) corresponds to correlation of h with (1-h)
        h_reflected = 1 - h
        c = correlate(h_reflected, h, mode='full')
        c = c * dx
        
        # Find k that maximizes overlap (i.e., maximum correlation value)
        max_idx = np.argmax(c)
        # The shift k = max_idx - (n_points - 1) ranges from -1 to 1
        # But we want k in [0,2], so k = max_idx - (n_points - 1) + 1 = max_idx - n_points + 2
        k_opt = max_idx - n_points + 2
        if k_opt < 0:
            k_opt += 2
        elif k_opt > 2:
            k_opt -= 2
            
        # Compute current C5
        current_c5 = c[max_idx]
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Approximate gradient of overlap w.r.t. h
        # ∂/∂h[i] ∫ h(x)(1 - h(x+k)) dx = (1 - h[i+k]) - h[i] * (d/dh[i] h[i+k])
        # For simplicity, we use finite differences based on the maximum correlation
        # It's an approximation but it guides towards improvement
    
        # Compute gradient of C_5 w.r.t. h
        # We want to minimize max_k ∫ h(x)(1 - h(x+k)) dx
        # The gradient involves only direct influence, so for a fixed k that maximizes this,
        # We compute: d/dh ∫ h(x)(1 - h(x+k)) dx = (1 - h_k)
        # We apply this in a full convolution-like form.
        # Compute gradient analytically:
        # Let c = correlation(h, 1-h), then:
        # grad_h[i] propto (1 - h[i+k]) for the k that maximizes c
        
        gradient = np.zeros(n_points)
        # Forward shift (considering circular or proper index)
        k_shift = (max_idx - (n_points - 1))  # [-n_points + 1, n_points - 1]
        
        # Position of peak k in [0, 2)
        k_shift_float = k_shift
        
        # Apply the gradient update
        for j in range(n_points):
            j_new = (j + k_shift) % n_points
            if j_new < n_points:
                gradient[j] = 1 - h[j_new]
            else:
                gradient[j] = 1 - h[j_new - n_points]
        
        # Update h with gradient and momentum
        v_h = beta * v_h + (1 - beta) * gradient
        h = h - alpha * v_h
        
        # Project h back to valid domain
        h = np.clip(h, 0, 1)
        
        # Normalize h to the correct sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive learning rate (simplified)
        if i % improvement_threshold == 0 and i > 0:
            # Check if we're improving
            if best_c5 == float('inf') or i > 1000:
                alpha *= 0.5
                
    # Final evaluation
    h_final = best_h.copy()
    h_reflected = 1 - h_final
    c_final = correlate(h_reflected, h_final, mode='full')
    c_final = c_final * dx
    final_c5 = np.max(c_final)
    
    return (h_final, final_c5, n_points)
