import numpy as np
from scipy.optimize import minimize
from math import exp, log
import time

def evaluate_c5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full cross-correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap integral C5 is the maximum of the cross-correlation
    # scaled by dx
    c5 = np.max(corr) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):

        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    # Ensure h_init is feasible: clip and rescale
    h_init = np.clip(h_init, 0, 1)
    h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    best_h = h_init.copy()
    best_c5 = evaluate_c5(best_h)
    
    # Adaptive surrogate function to smooth maximum
    def surrogate_max(arr, beta=100.0):
        return (1.0 / beta) * np.log(np.sum(np.exp(beta * arr)))

        return exp_arr / np.sum(exp_arr)
    
    # Optimizer parameters
    lr = 0.01
    max_iter = 1000
    patience = 100
    no_improvement_count = 0

        if time.time() - start_time > budget_s - 1:

        # Compute the cross-correlation
        corr = np.correlate(h, 1 - h, mode="full")
        c5 = np.max(corr)
        
        # Compute gradient w.r.t. h
        # Note: gradient of np.correlate w.r.t. h is tricky
        # We will use finite difference for simplicity
        eps = 1e-5
        grad = np.zeros_like(h)

        # Update h using gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Compute new c5
        c5_new = evaluate_c5(h_new)
        
        # Accept or reject update
        if c5_new < best_c5:

            best_c5 = c5_new
            no_improvement_count = 0
        else:
