    def smooth_max_overlap(h_vals):
        h_vals = np.array(h_vals)
        lagged_convs = correlate(h_vals, 1 - h_vals, mode='full')
        # Cut off the valid overlap lags (excluding padding)
        valid_lags = lagged_convs[n_points-1:]
        max_overlaps = np.abs(valid_lags)
        # Smooth approximation of max: log(sum(exp(alpha * x)))
        alpha = 0.1
        return np.log(np.sum(np.exp(alpha * max_overlaps))) / alpha

        # For overlap at index max_idx, x_i = h_i * (1 - h_{i+1}) + (1-h_i) * h_{i-1}
        # For simplicity, approximate gradient numerically inside the optimization loop

        eps = 1e-6

        max_overlaps = np.abs(valid_lags)

            best_h = h_new.copy()

        h = h_new.copy()
        
    # Final evaluation
    final_val = eval_func(best_h)
    return (best_h.tolist(), final_val, n_points)
