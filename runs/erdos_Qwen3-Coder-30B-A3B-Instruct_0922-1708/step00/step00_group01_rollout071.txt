import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Reasonable size for optimization
    max_iter = 1000
    learning_rate = 0.1
    tol = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        # Use full cross-correlation in the proper way
        corr = correlate(h, 1 - h, mode='full')
        # corr has length 2*n_points - 1
        # The overlap for lag k is at index k + n_points - 1
        # We need to compute the overlaps with respect to the actual range of lags
        overlaps = []
        dx = 2.0 / n_points
        for k in range(1 - n_points, n_points):
            if k >= 0:
                idx = k + n_points - 1
            else:
                idx = n_points - 1 + k
                
            overlap = corr[idx] * dx
            overlaps.append(overlap)
        
        max_overlap = max(overlaps)
        max_idx = np.argmax(overlaps)
        
        # Get k from max_idx
        # In full correlation, index k + n_points - 1 maps to lag k
        k_max = max_idx - n_points + 1
        
        # Compute gradient (approximate via finite differences)
        # For each i, we want to compute how the overlap changes with h[i]
        # giving an approximation of partial derivative.
        # Approximate it using finite difference of one-sided shifts
        
        grad = np.zeros_like(h)
        eps = 1e-6  # Small epsilon for numerical gradient
        
        h_plus = h.copy()
        h_minus = h.copy()
        
        # We approximate the j-th partial derivative of max overlap
        # by using h[i + shift] to change large interaction terms
        
        # Simple finite difference gradient for k_max:
        # Try shifting h[i] by eps in positive direction
        for i in range(len(h)):
            h_plus[i] = min(1.0, h[i] + eps)
            h_minus[i] = max(0.0, h[i] - eps)
            
            # Recompute overlap for both shifted versions
            # We only need to recompute the part where it affects
            old_overlap = 0.0
            new_overlap_plus = 0.0
            new_overlap_minus = 0.0
            
            # For efficient computation do full correlation only once more
            # find out which terms change with small shift from h[i]
            for j in range(len(h)):
                k_current = k_max
                if j + k_current >= 0 and j + k_current < len(h):
                    # overlap term is h[j]*(1 - h[j+k])
                    old_overlap += h[j] * (1 - h[j + k_current]) * dx
                    new_overlap_plus += h_plus[j] * (1 - h_plus[j + k_current]) * dx
                    new_overlap_minus += h_minus[j] * (1 - h_minus[j + k_current]) * dx
            
            # Centered difference: (f(x+eps) - f(x-eps)) / (2*eps)  
            grad[i] = (new_overlap_plus - new_overlap_minus) / (2 * eps)
            
            # Restore h
            h_plus[i] = h[i]
            h_minus[i] = h[i]
            
        # Solve linear subproblem: minimize grad^T * x subject to bounds
        # and constraint sum(x) = n_points / 2
        
        # Simplified: gradient descent with projection
        # Simple and fast method: just project to box [0,1] and scale
        # a better approach: solve projected gradient problem
        
        # Frank-Wolfe: use projected gradient descent with a simple update:
        # d = argmin_{x: 0 <= x[i] <= 1, sum(x) = n_points/2} <grad, x>
        # This is a linear program; due to constraints we can use coordinate descent
        
        # Simple solution here: move in direction of -grad and project to feasible set
        
        d = h - learning_rate * grad
        
        # Projection to [0,1] box
        d = np.clip(d, 0, 1)
        
        # Rescale to satisfy sum(h) = n_points / 2
        current_sum = np.sum(d)
        if current_sum > 0:
            d = d * (n_points / 2) / current_sum
        
        # Update h
        h = (1 - learning_rate) * h + learning_rate * d
        
        # Re-project to keep within bounds
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint again
        h = h / np.sum(h) * (n_points / 2)
        
        # Check improvement
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
    
    # Final check
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_overlaps = []
    dx = 2.0 / n_points
    for k in range(1 - n_points, n_points):
        if k >= 0:
            idx = k + n_points - 1
        else:
            idx = n_points - 1 + k
        overlap = corr_final[idx] * dx
        final_overlaps.append(overlap)
    
    final_c5 = max(final_overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
