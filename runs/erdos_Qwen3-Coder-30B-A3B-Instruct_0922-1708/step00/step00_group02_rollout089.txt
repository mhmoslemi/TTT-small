import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar

def evaluate_c5(h):
    """Evaluate the C5 constant for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute cross-correlation with 1-h
    corr = correlate(h, 1 - h, mode="full")
    # Maximum overlap over all shifts
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm_phase(h0, n_points, budget_s):
    """Run genetic algorithm to find good initial solution."""
    pop_size = 50
    generations = 100
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.2
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = project_h(h, n_points)
        population.append(h.copy())
    
    # Evaluate initial population
    fitness = [evaluate_c5(h) for h in population]
    
    best_h = population[np.argmin(fitness)].copy()
    best_c5 = min(fitness)
    
    gen_time = budget_s * 0.7 / generations
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > budget_s * 0.7:
            break
            
        # Elitism
        elite_indices = np.argsort(fitness)[:int(elitism_rate * pop_size)]
        elite = [population[i] for i in elite_indices]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - len(elite):
            # Select two parents (tournament selection)
            a, b = np.random.choice(pop_size, size=2, replace=False)
            pa, pb = population[a], population[b]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([pa[:crossover_point], pb[crossover_point:]])
                child2 = np.concatenate([pb[:crossover_point], pa[crossover_point:]])
            else:
                child1, child2 = pa.copy(), pb.copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Project and keep
            child1 = project_h(child1, n_points)
            child2 = project_h(child2, n_points)
            
            offspring.extend([child1, child2])
            
        # Replace worst with offspring
        worst_indices = np.argsort(fitness)[-len(offspring):]
        for i, idx in enumerate(worst_indices):
            if i < len(offspring):
                population[idx] = offspring[i]
        
        # Re-evaluate
        fitness = [evaluate_c5(h) for h in population]
        current_best = min(fitness)
        if current_best < best_c5:
            best_c5 = current_best
            best_h = population[np.argmin(fitness)].copy()
            
    return best_h, best_c5

def gradient_descent_phase(h0, n_points, budget_s):

            c5_plus = evaluate_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Simple gradient descent with line search for a few steps 
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.3:
            break
            
        # Assume any shift can be the winner for now (just do a few simplest steps with gradient descent)
        # The precise gradient can be approximated; for simplicity:
        # We compute C5 directly and use that as a rough indicator of descent direction
        current_c5 = evaluate_c5(best_h)
        
        h_candidate = best_h.copy()

        # Simple line search

        else:
            patience_count = 0
        if new_c5 < best_c5:
            best_c5 = new_c5
    
    return best_h, best_c5
