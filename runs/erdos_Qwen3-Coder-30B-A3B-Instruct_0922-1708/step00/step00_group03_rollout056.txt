import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial_h_values if provided, otherwise create a simple starting point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.full(n_points, 1.0 / n_points)
        
    # Normalize to sum = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Initialize variables
    h = cvx.Variable(n_points)
    # Objective variable
    t = cvx.Variable()
    
    # Set up constraints
    constraints = [
        h >= 0,
        h <= 1,
        cvx.sum(h) == n_points / 2,
        t >= 0
    ]
    
    # Create correlation terms
    # For each lag k, compute sum h[i] * (1 - h[i-k])
    # This will be represented as sum of z[i, i-k] = h[i] * (1 - h[i-k])
    # We'll linearize with constraints
    
    # Create coupling variables
    z = cvx.Variable((n_points, n_points))
    
    # Relaxation constraints for z[i,j] <= h[i] * (1 - h[j])
    # equivalent to: z <= h[i], z <= 1 - h[j], z >= h[i] + h[j] - 1
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                z[i,j] <= h[i],
                z[i,j] <= 1 - h[j],
                z[i,j] >= h[i] + h[j] - 1
            ]
    
    # Compute all correlation sums
    max_corr = 0
    C_vals = []
    for k in range(-n_points+1, n_points):
        corr_sum = 0
        for i in range(n_points):
            j = i - k
            if 0 <= j < n_points:
                corr_sum += z[i, j]
        C_vals.append(corr_sum)
        max_corr = max(max_corr, corr_sum)
    
    # Objective
    objective = cvx.Minimize(max_corr)
    
    # Build problem
    prob = cvx.Problem(objective, constraints)
    
    try:
        # Solve 
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            h_opt = h.value
        else:
            h_opt = h0
    except:
        # Fall back to initialization
        h_opt = h0
    
    # Ensure bounds
    h_opt = np.clip(h_opt, 0, 1)
