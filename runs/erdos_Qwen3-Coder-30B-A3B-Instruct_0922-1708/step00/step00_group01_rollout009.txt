import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 47  # Use the initial construction size
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create initial symmetric plateau
        h = np.ones(n_points) * 0.5
        # Make it taper at ends
        for i in range(n_points):
            if i < n_points // 4:
                h[i] = 0.1 + 0.4 * i / (n_points // 4)
            elif i >= 3 * n_points // 4:
                h[i] = 0.1 + 0.4 * (n_points - 1 - i) / (n_points // 4)
        # Normalize to meet sum constraint
        h = h / np.sum(h) * (n_points / 2)
    
    # Parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iterations = 3000
    convergence_tol = 1e-5
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute overlaps for all lags
        # Using the full cross-correlation (equivalent to correlation with flipped kernel)
        overlaps = correlate(h, 1 - h, mode='full')
        # Adjust for the fact that we're using the wrong kernel indexing (this accounts for proper offset alignment)
        # The overlap for lag k is at index k + (n_points - 1)
        # But iterate over all lags from -(n_points-1) to (n_points-1)
        overlaps_full = np.zeros(2 * n_points - 1)
        overlaps_full[:] = overlaps
        # Extract the actual overlaps by shifting indices
        # overlap(k) = sum_{i} h[i] * (1 - h[i+k])
        # In the convolution result, this is at index (n_points-1) + k
        # We shift to get this directly
        shift = n_points - 1
        all_overlaps = np.array([overlaps_full[shift + k] * dx for k in range(-n_points + 1, n_points)])
        all_overlaps = all_overlaps / dx  # undo the scaling in convolution for now
        all_overlaps = all_overlaps * dx
        
        # Log-sum-exp smooth approximation
        tmp = np.exp(tau * all_overlaps)
        log_sum_tmp = np.log(np.sum(tmp))
        C5_smooth = log_sum_tmp / tau
        
        # Gradient of log-sum-exp approximation
        # d/dh[i] of C5_smooth 
        # Using chain rule:
        # dC5_smooth / dh[i] = sum_k (exp(tau*overlap_k) * tau * doverlap_k/dh[i]) / sum_k(exp(tau*overlap_k)) 
        #   = (sum_k (exp(tau*overlap_k) * tau * - (1-h[i-k]))) / sum_k (exp(tau*overlap_k))
        # Let's do a direct derivation using einsum or standard instructions
        
        grad = np.zeros(n_points)
        exp_overlaps = np.exp(tau * all_overlaps)
        for i_idx in range(n_points):
            for k in range(-n_points+1, n_points):
                # overlap_k contributes to gradient on h[i] through - h[i-k] due to
                # grad in h[i] of sum_j h[j] * (1 - h[j+k]) = -(1-h[i+k])
                # non-loss for negative indices due to wrapping
                j = (i_idx - k) % n_points
                if j != i_idx:
                    grad[i_idx] += exp_overlaps[k + n_points - 1] * tau * (- (1.0 - h[j])) * dx
                else:
                    grad[i_idx] += exp_overlaps[k + n_points - 1] * tau * (- (1.0 - h[j])) * dx
                    
        # Avoid gradients from the aprtial terms of h[i] itself
        # Just sum over the above calc: for all points with positive contributions
        # Actually better to compute manually gradient via the convolution derivative:
        # Gradient of overlap(k) w.r.t. h[i] is (1 - h[i-k]) * dx
        # Main computation
        grad_sum = np.zeros(n_points)
        for k in range(-n_points+1, n_points):
            lag = k % n_points if k >= 0 else (n_points + k) % n_points
            # Compute x_k[i] = h[i] * (1 - h[i+k]) 
            # So + d/dh[i] x_k[i] = 1 - h[i+k], if i is associated with that term
            # So  dC5 / dh[i] = quadratic w.r.t. h:
            # Accumulate the influence of constraint that each term contributes 1-h[i-k]
            tmp_grad = np.zeros(n_points)
            for j in range(n_points):
                i_idx = j
                j_idx = (j - k) % n_points
                if j_idx < n_points:
                    tmp_grad[i_idx] -= exp_overlaps[k + n_points - 1] * (1 - h[j_idx]) * dx
            # Or easier, group by the h[i] term:
            # Gradients from all h[i-k]: it's - (1-h[i-k]) * dx term of overlap(i-k), 
            # from overlap(k) contribition to grad_h[i]:
            # rewrite indices with a bit more clarity:
            for i_idx in range(n_points):
                # all terms affecting contribution at i_idx from overlap vector:
                # Each overlap_k contributes - (1-h[i_idx-k]) * dx for h[i_idx]
                # But overlap index offset: it's indexed at k_n = k + n_points - 1
                if (k >= 0):
                    j_idx = (i_idx - k) % n_points
                    grad_sum[i_idx] -= exp_overlaps[k + n_points - 1] * (1 - h[j_idx]) * dx
                else:
                    # k < 0 this is offset
                    j_idx = (i_idx - k) % n_points
                    grad_sum[i_idx] -= exp_overlaps[k + n_points - 1] * (1 - h[j_idx]) * dx
                    
        # Final correction to normalize gradients:
        total_exp = np.sum(exp_overlaps) 
        if total_exp > 0:
            grad_sum /= total_exp
            
        # Update h
        h -= learning_rate * grad_sum
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Enforce constraint sum(h) = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        
        # Check improvement
        current_C5 = evaluate_C5(h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Anneal temperature and learning rate
        tau *= anneal_factor
        if tau < 1e-6:
            tau = 1e-6
        # Decaying learning rate
        learning_rate *= 0.999
        
        # Early stop if no significant improvement
        if i > 100 and np.abs(np.mean(grad_sum**2)) < 1e-10:
            break

    return (best_h.tolist(), best_C5, n_points)

def evaluate_C5(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps manually using correlation
    # h convolution with (1-h) and take the maximum of all lags
    h_rev = h[::-1]
    h_flipped = h_rev  # reversal for convolution-like cross-correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # Extract the full lags from - (n_points-1) to + (n_points-1)
    full_overlaps = np.array([overlaps[i] for i in range(len(overlaps))]) * dx
    # Extract lags from -(n-1) to n-1
    # We care about full objective: max_{k} int h(x) * (1-h(x+k)) dx
    # Which equals max_{k} overlap(k) * dx
    # So indices from zero to 2*(n-1) represent all lags from -(n-1) to n-1
    # We take the max normed overlaps among all of these, but also account for correct 0 offset
    # Actually, better approach:
    result_lags = []
    
    shift = n_points - 1  # Index where k=0 overlaps
    for k in range(-n_points+1, n_points):
        idx = shift + k
        if 0 <= idx < len(overlaps):
            result_lags.append(overlaps[idx] * dx)
            
    return max(result_lags)
