import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with one known good solution and random ones."""
    pop = []
    pop.append(initial_h.copy())
    for _ in range(pop_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        pop.append(h)
    return pop

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select the best individual from a random subset."""
    indices = np.random.choice(len(population), tournament_size)
    local_fitnesses = [fitnesses[i] for i in indices]
    winner_index = indices[np.argmin(local_fitnesses)]
    return population[winner_index]

def crossover(parent1, parent2):
    """Perform crossover of two parents by swapping segments."""
    n = len(parent1)
    if n < 2:
        return parent1.copy(), parent2.copy()
    split_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, mutation_strength=0.1):
    """Mutate a solution."""
    h = h.copy()
    n = len(h)
    if np.random.rand() > mutation_rate:
        return h
    indices = np.random.choice(n, size=int(0.1 * n), replace=False)
    for i in indices:
        h[i] += np.random.uniform(-mutation_strength, mutation_strength) * (2.0 / n)
    h = np.clip(h, 0, 1)
    # Rescale to meet integral constraint
    h_sum = np.sum(h)
    h = h * (n / 2) / h_sum
    return h

def genetic_algorithm(initial_h_values, n_points=70, pop_size=100, budget_s=600):
    """Optimize step function using genetic algorithm."""
    # Initialize
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    # Fitness computation
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_fitness = min(fitnesses)
    
    start_time = time.time()
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 30:
        # Selection and evolution
        new_population = []
        for _ in range(pop_size // 2):
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Evaluate new population
        population = new_population
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_fitness:
            best_fitness = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Early stopping if no improvement in last 50 generations
        if no_improvement >= 50:
            break
            
        generation += 1
    
    return best_h, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Default initialization
    
    # Run genetic algorithm
    h_opt, c5_bound = genetic_algorithm(h0, n_points, pop_size=100, budget_s=budget_s)
    return h_opt, c5_bound, n_points
