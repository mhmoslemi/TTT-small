import numpy as np
from scipy import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for given h and dx."""
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum=1."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 0.5 / n_points)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Setup
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    patience = 1000
    patience_count = 0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5 * dx)
        h = project_h(h, n_points)
    
    # Evaluate initial solution
    c5_best = evaluate_c5(h, dx)
    h_best = h.copy()
    
    # Parameters
    sigma = 0.1  # Initial noise level
    decay = 0.999  # Decaying noise
    start_time = time.time()
    
    # Optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate perturbation
        perturb = np.random.normal(0, sigma, n_points)
        h_new = h + perturb
        
        # Project to feasible set
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_best:
            h = h_new
            c5_best = c5_new
            h_best = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        # Decay noise
        sigma *= decay
        
        # Early termination if no improvement
        if patience_count > patience:
            break
    
    return (list(h_best), c5_best, n_points)
