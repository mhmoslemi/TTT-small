        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation for all lags
        corr = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        max_corr = corr[max_lag_idx]
        
        # Gradient approximation via finite differences for simplicity
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)

            h_minus[i] -= eps

def compute_c5(h, dx):
    # Compute all correlations and return max
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
