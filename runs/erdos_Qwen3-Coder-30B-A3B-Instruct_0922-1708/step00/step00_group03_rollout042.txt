import numpy as np
from scipy.signal import correlate
import time

def evaluate_erdos_solution(h):
    """Evaluate the current h and return the C5 constant."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps for all lags
    overlaps = correlate(h, 1 - h, mode='full')
    # The maximum overlap is at the middle positions
    max_overlap = np.max(overlaps[len(overlaps)//2:])
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h.copy()
        # Normalize to ensure sum(h) == n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Learning rate and related parameters
    eta = 0.1
    patience = 0
    max_patience = 50

    # EM iterations
    for iteration in range(5000):
        if time.time() - start_time > budget_s - 1:
            break

        # E-step: compute overlaps and normalize them as probabilities
        overlap_vals = correlate(h, 1 - h, mode='full')
        # Extract only the valid lags (positive lags)
        overlap_vals = overlap_vals[len(overlap_vals)//2:]
        overlap_probs = overlap_vals / np.sum(overlap_vals)
        
        # M-step: update h based on gradient
        grad = np.zeros_like(h)
        for k in range(len(overlap_vals)):
            # Gradient wrt h[i] equals (1 - h[i - k])
            shifted_h = np.roll(h, k)
            grad += overlap_probs[k] * (1 - shifted_h)

        h_new = h - eta * grad
        
        # Clip h to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale h to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Evaluating objective
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break
            
        # Adaptive learning rate
        if iteration % 10 == 0 and iteration > 0:
            eta *= 0.9
                
    return (best_h.tolist(), best_c5, n_points)
