import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    # Only consider the overlapping parts
    full_corr = corr[len(h)-1:]
    C5 = np.max(full_corr) * dx
    return C5

def project_h(h, n_points):
    """Project h to satisfy constraints"""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction or create one
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        n_points = len(h_base)
    else:
        n_points = 45
        h_base = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    h_best = project_h(h_base.copy(), n_points)
    c5_best = evaluate_h(h_best, dx)
    
    # Optimization parameters
    max_iter = 1000000
    patience = 1000
    patience_counter = 0
    noise_scale = 0.1
    decay_rate = 0.9999
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate new candidate
        h_new = h_best + np.random.normal(0, noise_scale, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update noise scale
        noise_scale *= decay_rate
        
        # Early stopping
        if patience_counter >= patience:
            break
            
    return (h_best.tolist(), c5_best, n_points)
