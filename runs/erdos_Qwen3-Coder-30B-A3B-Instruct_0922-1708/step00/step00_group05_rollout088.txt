        corr = np.correlate(h, 1 - h, mode="full")

        max_shift = max_idx - (len(h) - 1)
        
        # Compute gradient for the maximal shift
        grad = np.zeros(n_points)
        if max_shift >= 0:
            # Forward shift case
            for i in range(n_points - max_shift):
                grad[i] += (1 - h[i + max_shift]) * dx
        else:
            # Backward shift case
            abs_shift = -max_shift
            for i in range(abs_shift, n_points):
                grad[i] += -h[i - abs_shift] * dx
                
        # Update h with gradient
        h_new = h - learning_rate * grad
        
        # Clip h to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust sum to exactly target_sum
        h_sum_new = np.sum(h_new)
        if abs(h_sum_new - target_sum) > 1e-10:
            # Distribute the difference proportionally
            diff = target_sum - h_sum_new
            h_new += diff / n_points
            
        h = h_new.copy()

        c5 = np.max(np.correlate(h, 1 - h, mode="full")) * dx
        
        if c5 < best_c5:

            best_h = h.copy()
            last_improved = iteration
            
        iteration += 1
