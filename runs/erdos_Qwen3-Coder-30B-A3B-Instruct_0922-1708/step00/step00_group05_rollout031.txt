import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Fallback: construct a simple initial guess
        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 0.0
        h0 /= np.sum(h0) * 2.0 / n_points  # normalize so sum = n_points / 2
    
    # Compute dx
    dx = 2.0 / n_points
    
    # Constraint and objective function for CVXPY
    def compute_C(h_vals):
        # Compute all correlations for all lags
        h = np.array(h_vals)
        corr = correlate(h, 1 - h, mode='full')
        # The lags are from -n+1 to n-1
        # We want to extract the max for real lags in [-n+1, n-1] excluding 0
        max_val = 0.0
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue  # skip zero lag
            idx = k + n_points - 1  # The index of the correlation corresponding to lag k
            if 0 <= idx < len(corr):
                val = corr[idx] * dx
                if val > max_val:
                    max_val = val
        return max_val

    def solve_qp(h_init, max_iters=10):
        # Define optimization variables
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Create constraints
        constraints = []
        constraints.append(h >= 0)
        constraints.append(h <= 1)
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Add overlap constraints
        for k in range(-n_points + 1, n_points):
            if k == 0:
                continue
            # Compute the correlation for lag k
            # This approach uses a direct quadratic constraint
            if k > 0:
                lagged_sum = cp.sum(cp.multiply(h[k:], 1 - h[:n_points - k]))
            elif k < 0:
                lagged_sum = cp.sum(cp.multiply(h[:n_points + k], 1 - h[-k:]))
            else:
                lagged_sum = 0
            constraints.append(lagged_sum <= C * dx)
        
        # Objective
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Try solving
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                if not np.isnan(C.value):
                    return h.value, C.value
        except:
            pass
        return None, None

    # Main optimization loop
    best_h = h0.copy()
    best_C = compute_C(best_h)
    n_iter = 0
    
    while time.time() - start_time < budget_s - 1:  # Leave some grace time
        n_iter += 1
        # Random perturbation
        h_perturbed = best_h + np.random.normal(0, 0.02, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        # Renormalize
        h_perturbed *= (n_points / 2) / np.sum(h_perturbed)
        
        # Try solving with CVXPY if we are allowed to
        if n_iter % 3 == 0 and n_points <= 100:  # Avoid too heavy QP for large problems
            h_sol, C_sol = solve_qp(h_perturbed)
            if h_sol is not None and C_sol < best_C:
                best_h = h_sol
                best_C = C_sol
        else:
            # Evaluate current value
            C_current = compute_C(h_perturbed)
            # Accept if better
            if C_current < best_C:
                best_h = h_perturbed
                best_C = C_current
    
    # Final validation
    h_final = np.clip(best_h, 0, 1)
    h_final *= (n_points / 2) / np.sum(h_final)  # Ensure sum = n_points/2
    final_C = compute_C(h_final)
    
    return (h_final, final_C, n_points)
