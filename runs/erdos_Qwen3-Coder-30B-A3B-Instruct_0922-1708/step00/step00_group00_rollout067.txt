import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx and find maximum
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize a population of step functions."""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        pop_size -= 1
    
    # Generate diverse random individuals
    for _ in range(pop_size):
        # Start with uniform distribution
        h = np.full(n_points, 0.5)
        # Add small random perturbations
        perturb = np.random.normal(0, 0.1, n_points)
        h += perturb
        # Clip values
        h = np.clip(h, 0, 1)
        # Adjust to meet sum constraint
        h_sum = np.sum(h)
        h *= (n_points / 2) / h_sum
        population.append(h.copy())
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform single-point crossover."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a step function."""
    h_new = h.copy()
    n_points = len(h)
    
    # Apply mutations
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            # Add small random noise
            noise = np.random.normal(0, 0.05)
            h_new[i] += noise
            # Clip to [0,1]
            h_new[i] = np.clip(h_new[i], 0, 1)
    
    # Renormalize to meet sum constraint
    h_sum = np.sum(h_new)
    h_new *= (n_points / 2) / h_sum
    
    return h_new

def genetic_algorithm(n_points, pop_size=100, generations=120, 
                      mutation_rate=0.05, initial_h=None, timeout=600):
    """Genetic algorithm to optimize the step function h."""
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    dx = 2.0 / n_points
    
    best_solution = None
    best_fitness = float('inf')
    
    start_time = time.time()
    
    for gen in range(generations):
        if time.time() - start_time > timeout - 1:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track the best solution
        for i, fit in enumerate(fitnesses):
            if fit < best_fitness:
                best_fitness = fit
                best_solution = population[i].copy()
        
        # Elitism: keep top 20%
        elite_count = pop_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elites = [population[i].copy() for i in elite_indices]
        
        # Create new population
        new_population = elites[:]
        
        # Fill rest of population
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
    
    return best_solution, best_fitness

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run optimization."""
    np.random.seed(seed)
    
    # Get the initial construction if available
    initial_h = None
    if 'initial_h_values' in globals():
        # Scale initial_h to match expected format
        initial_h = np.array(initial_h_values)
    
    # Use reasonable parameters based on the problem
    n_points = 45  # Use original size
    
    # Run the genetic algorithm
    best_h, best_c5 = genetic_algorithm(
        n_points=n_points,
        pop_size=100,
        generations=120,
        mutation_rate=0.05,
        initial_h=initial_h,
        timeout=budget_s
    )
    
    # Final check and return
    dx = 2.0 / n_points
    computed_c5 = evaluate_h(best_h, dx)
    
    # Ensure sum constraint
    assert abs(np.sum(best_h) - n_points / 2) < 1e-6
    
    return (best_h, computed_c5, n_points)
