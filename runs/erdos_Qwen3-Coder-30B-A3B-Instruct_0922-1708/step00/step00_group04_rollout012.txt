import numpy as np
from scipy import correlate
import cvxpy as cp
import time

def evaluate_erdos_solution(h):
    """Compute the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps for lags k in [-n+1, n-1]
    corr = correlate(h, 1 - h, mode='full')
    # Rescale by dx
    corr *= dx
    # Extract the maximum overlap for k in [0, 2*n-2]
    max_overlap = np.max(corr[n-1:])
    return max_overlap

    # Try different optimization approaches, starting simple
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Simple gradient-like refinement using hill climbing
    dx = 2.0 / n
    step_size = 0.01
    
    for i in range(10000):  # reasonable iteration limit

        if c5_new < best_c5:
            best_h = h_new
            best_c5 = c5_new
    
    # Final check and return
    final_c5 = evaluate_erdos_solution(best_h)
