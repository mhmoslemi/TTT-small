import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum==n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Default construction
        n_points = 100
        h0 = np.ones(n_points) * 0.5
    
    # Initialize swarm
    n_particles = 50
    particles = []
    velocities = []
    
    # Start with initial point
    h0 = project_h(h0, n_points)
    particles.append(h0.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        particles.append(h.copy())
        velocities.append((np.random.rand(n_points) - 0.5) * 0.1)
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 2.0
    max_iter = 500
    stagnation_limit = 50
    diversity_reset = 25
    
    # Track best
    pbest = [p.copy() for p in particles]
    pbest_scores = [evaluate_h(p, 2.0/n_points) for p in particles]
    gbest_idx = np.argmin(pbest_scores)
    gbest = pbest[gbest_idx].copy()
    gbest_score = pbest_scores[gbest_idx]
    
    # Iterations
    iter_count = 0
    since_improvement = 0
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        iter_count += 1
        since_improvement += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            score = evaluate_h(particles[i], 2.0/n_points)
            if score < pbest_scores[i]:
                pbest[i] = particles[i].copy()
                pbest_scores[i] = score
                if score < gbest_score:
                    gbest = particles[i].copy()
                    gbest_score = score
                    since_improvement = 0
        
        # Update swarm
        for i in range(n_particles):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            v = w * velocities[i] + c1 * r1 * (pbest[i] - particles[i]) + c2 * r2 * (gbest - particles[i])
            velocities[i] = v
            particles[i] += v
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points)
        
        # Diversity injection
        if since_improvement > diversity_reset and iter_count > 10:
            for i in range(int(0.2 * n_particles)):
                j = np.random.randint(n_particles)
                particles[j] = np.random.rand(n_points)
                particles[j] = project_h(particles[j], n_points)
                velocities[j] = (np.random.rand(n_points) - 0.5) * 0.1
            since_improvement = 0
    
    # Final check
    final_scores = [evaluate_h(p, 2.0/n_points) for p in particles]
    final_best_idx = np.argmin(final_scores)
    if final_scores[final_best_idx] < gbest_score:
        gbest_score = final_scores[final_best_idx]
        gbest = particles[final_best_idx].copy()
    
    return (gbest.tolist(), gbest_score, n_points)
