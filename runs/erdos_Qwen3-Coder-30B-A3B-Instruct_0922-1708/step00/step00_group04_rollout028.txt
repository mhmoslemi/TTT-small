def run(seed=42, budget_s=600, **kwargs):

    # Optimization loop
    iteration = 0
    last_improvement = 0
    max_iter = 600
    
    while time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Formulate and solve convex approximation problem
        h = cp.Variable(n_points)
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        
        # Integral constraint

        # Evaluate convolution-like sums
        # Create a linear approximation of h*(1-h)
        # We hope that for smooth h, C5 ~ (h*1) - (h^2) or similar
        # Here we directly approximate overlap integral.
        
        # Create full-length convolution with flipped 1-h for each k
        # Since we are not forming explicit CVX problem for C5 directly,

        # Grid search to evaluate C5 efficiently
        h_vals = current_h.copy()

        # Make a small perturbation to find better solution
        h_new = h_vals.copy()

    return (best_h.tolist(), final_c5, n_points)


def evaluate_c5(h_values, dx):
    # Compute overlap integral C_5 = max_k int h(x)(1 - h(x+k)) dx
    h = np.array(h_values)
    
    # Convolution of h with (1-h) via correlate
    # Note: correlate returns full convolution
    # Our kernel is 1-h, padded to match full length
    h_rev = h[::-1]  # reverse h
    kernel = 1 - h
    
    corr = np.correlate(h, kernel, mode="full")
    # Note: convolution definition used here is not exactly matching
    # to make sure we get aligned. Use convolution with reverse and lag-k.
    
    # The overlap integral is defined via correlations with shifted h
    # So we compute for all valid shifts: h(x) * (1 - h(x+k))
    # Which we represent by the full convolution
    
    # Take maximum over all lag values
    lags = np.arange(-h.shape[0] + 1, h.shape[0])
    # extract up to h.shape[0] - 1
    max_corr = np.max(corr)
    return max_corr * dx
