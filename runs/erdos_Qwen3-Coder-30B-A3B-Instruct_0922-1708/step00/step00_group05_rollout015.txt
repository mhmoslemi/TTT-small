import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_erdos_solution(h, dx=2.0):
    """Evaluate the Erdos constant C5 for given step function h."""
    # Compute full correlation
    corr = correlate(h, 1 - h, mode="full")
    # Extract the relevant part (k from -n+1 to n-1)
    n = len(h)
    corr = corr[n-1:2*n-1]
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    """Normalize h to have target_sum."""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def project_h(h):
    """Project h to [0,1] and normalize."""
    h = np.clip(h, 0, 1)
    h = normalize_h(h, 1.0)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Starting point
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = 1.0
    
    start_time = time.time()
    
    # PSO parameters
    m = 20  # number of particles
    w = 0.7  # inertia weight
    c1, c2 = 2.0, 2.0  # acceleration coefficients
    v_max = 0.1  # max velocity
    
    # Initialize particles
    particles = []
    velocities = []
    
    # One particle is the initial solution
    h_proj = project_h(h0.copy())
    particles.append(h_proj.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate other particles
    for _ in range(m - 1):
        pert = np.random.normal(scale=0.05, size=n_points)
        h_perturb = h0 + pert
        h_new = project_h(h_perturb)
        particles.append(h_new.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Compute initial fitness
    fitness = [evaluate_erdos_solution(p, dx) for p in particles]
    best_idx = np.argmin(fitness)
    g_best = particles[best_idx].copy()
    g_best_fitness = fitness[best_idx]
    
    # Track improvement
    last_improvement = 0
    iteration = 0
    
    # PSO iterations
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        
        # Update each particle
        for i in range(m):
            # Update velocity
            r1, r2 = np.random.random(2)
            velocities[i] = (w * velocities[i] +
                             c1 * r1 * (particles[i] - particles[i]) +
                             c2 * r2 * (g_best - particles[i]))
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] += velocities[i]
            particles[i] = project_h(particles[i])
            
            # Evaluate new fitness
            new_fitness = evaluate_erdos_solution(particles[i], dx)
            
            if new_fitness < fitness[i]:
                fitness[i] = new_fitness
                
            if new_fitness < g_best_fitness:
                g_best = particles[i].copy()
                g_best_fitness = new_fitness
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Tournament selection and diversity
        if iteration % 50 == 0:
            # Sort by fitness
            sorted_idx = np.argsort(fitness)
            # Keep top 50%
            top_idx = sorted_idx[:m//2]
            # Replace bottom 50%
            for i in range(m//2, m):
                mutated_idx = np.random.choice(top_idx)
                mutation = np.random.normal(scale=0.05, size=n_points)
                h_mutated = particles[mutated_idx] + mutation
                particles[i] = project_h(h_mutated)
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                fitness[i] = evaluate_erdos_solution(particles[i], dx)
        
        # Add diversity if no improvement in a while
        if last_improvement > 50:
            for i in range(m):
                mutation = np.random.normal(scale=0.02, size=n_points)
                particles[i] += mutation
                particles[i] = project_h(particles[i])
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                fitness[i] = evaluate_erdos_solution(particles[i], dx)
            last_improvement = 0
            
    return (g_best, g_best_fitness, n_points)
