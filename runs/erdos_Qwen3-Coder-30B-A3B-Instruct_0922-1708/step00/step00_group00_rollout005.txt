    max_iter = 500

    inner_iter = 10
    last_improvement = 0

            # Compute gradient and update
            overlap_vals = np.correlate(h_new, 1-h_new, mode='full')
            grad = np.zeros_like(h_new)
            
            # Construct gradient from correlations
            for k in range(len(overlap_vals)):
                if k < n_points:
                    grad[:n_points-k] += overlap_vals[k] * (-1 + 2 * h_new[k:])
                else:
                    grad[k-n_points+1:] += overlap_vals[k] * (-1 + 2 * h_new[:n_points-(k-n_points+1)])
            
            grad *= dx
            
            h_new = h_new - 0.01 * grad
            # Project onto [0,1]

            # Enforce integral constraint
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
        # Update dual variables
        h = h_new

        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:

    return best_h.tolist(), best_C5, n_points

def evaluate_C5(h, dx):
    # Compute overlaps for all shifts and take maximum
    overlaps = np.correlate(h, 1 - h, mode='full') * dx
    return np.max(overlaps)
