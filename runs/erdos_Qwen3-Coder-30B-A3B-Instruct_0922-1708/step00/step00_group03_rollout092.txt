import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    try:
        h = initial_h_values.copy()
        n_points = len(h)
        dx = 2.0 / n_points
    except:
        # Default initialization
        n_points = 70
        dx = 2.0 / n_points
        h = np.full(n_points, 0.5)
    
    # Ensure the sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Compute initial C5
    def compute_c5(h_vals):
        h_vals = np.array(h_vals)
        overlaps = signal.correlate(h_vals, 1 - h_vals, mode='full')
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # EM loop
    start_time = time.time()
    iterations = 0
    patience = 0
    max_patience = 50
    learning_rate = 0.1
    last_c5 = best_c5
    
    while time.time() - start_time < budget_s - 1 and patience < max_patience:
        # E-step - compute overlaps for all lags
        h_array = np.array(h)
        overlaps = signal.correlate(h_array, 1 - h_array, mode='full')
        overlaps = overlaps * dx
        # Normalize overlaps to get probabilities
        p = np.abs(overlaps) / np.sum(np.abs(overlaps))
        
        # M-step - update h
        # Gradient computation
        grad = np.zeros_like(h_array)
        for k in range(len(overlaps)):
            # Circuitous way to compute gradient
            k_offset = k - (len(h_array) - 1)
            if k_offset < 0:
                shifted_indices = np.arange(len(h_array) + k_offset, len(h_array))
                h_shifted = np.concatenate([h_array[0:k_offset], np.zeros(-k_offset)])
            else:
                shifted_indices = np.arange(0, len(h_array) - k_offset)
                h_shifted = np.concatenate([np.zeros(k_offset), h_array[k_offset:]])
            p_k = p[k]
            grad += p_k * (1 - h_shifted)
        
        # Update h
        h_new = h_array - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check improvement and update
        new_c5 = compute_c5(h_new)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        h = h_new.copy()
        
        # Adjust learning rate
        if new_c5 >= last_c5:
            learning_rate *= 0.5
        else:
            learning_rate = min(0.1, learning_rate * 1.05)
        last_c5 = new_c5
        
        iterations += 1
        if iterations % 50 == 0:
            print(f"iter {iterations}: C5 = {best_c5:.6f}")
    
    return (best_h.tolist(), best_c5, n_points)
