import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Compute C5 for given h values."""
    # Compute overlap integral: max_k int h(x)(1 - h(x+k)) dx
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral is scaled by dx, and we need the maximum among all shifts
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points, dx):
    """Project h to feasible region: [0,1] and sum = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum = n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    # Initialize h with the given initial construction or create a simple one
    if 'initial_h_values' in globals():

    else:
        # Simple initialization
        h = np.full(n_points, 0.5 * dx)
    
    # Ensure sum constraint
    h = project_h(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Randomized hill climbing parameters
    sigma = 0.1
    decay = 0.99
    patience = 1000
    patience_counter = 0
    
    iteration = 0
    while time.time() - start_time < budget_s and patience_counter < patience:
        iteration += 1
        # Generate perturbation
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        
        # Project to feasible region
        h_new = project_h(h_new, n_points, dx)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update h for next iteration
        h = h_new
        
        # Decay noise
        sigma *= decay
        
    return (list(best_h), best_c5, n_points)
