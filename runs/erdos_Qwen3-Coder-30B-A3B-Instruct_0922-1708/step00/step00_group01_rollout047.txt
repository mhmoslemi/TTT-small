def compute_c5(h, dx):
    # Compute the maximum overlap across all lags
    h = np.asarray(h)
    n = len(h)
    c5_values = []
    for k in range(-n+1, n):
        if k >= 0:
            # h[:n-k] * (1 - h[k:]) 
            val = np.sum(h[:n-k] * (1 - h[k:])) * dx
        else:
            # h[-k:] * (1 - h[:-k])  
            val = np.sum(h[-k:] * (1 - h[:-k])) * dx
        c5_values.append(val)
    return max(c5_values)
