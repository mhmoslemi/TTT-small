import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Fallback initialization
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Optimization loop
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps for all shifts
        corr = correlate(h, 1 - h, mode='full') * dx
        max_overlap = np.max(corr)
        
        # Find which shift causes maximum overlap
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        # Compute gradient for the worst shift
        grad = np.zeros_like(h)
        if max_k >= 0:
            # h[i] contributes to overlap at shift max_k
            grad[:n_points - max_k] += 1 - h[max_k:]
            grad[max_k:] += -h[:-max_k]
        else:
            # max_k < 0, so h[i + max_k] contributes to overlap at shift max_k from h[i]
            grad[-max_k:] += 1 - h[:n_points + max_k]
            grad[:n_points + max_k] += -h[-max_k:]
            
        # Update h
        step_size = 0.01
        h_new = h - step_size * grad
        
        # Project onto feasible set
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Accept new solution if it improves C5
        h = h_new
        last_improvement = time.time()
    
    # Final evaluation
    corr = correlate(h, 1 - h, mode='full') * dx
    final_c5 = np.max(corr)
    
    return (h, final_c5, n_points)
