import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Use initial construction if available, otherwise create random feasible h

    # Ensure feasible constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    # Frank-Wolfe optimizer
    max_iter = 1000
    learning_rate = 0.1
    tolerance = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()
    time_limit = start_time + budget_s

    for iteration in range(max_iter):
        if time.time() > time_limit:
            break
            
        # Compute overlaps for all lags using cross-correlation
        # h * (1 - h[k:]) for all k
        expanded_h = np.concatenate([h, np.zeros(n_points - 1)])
        expanded_h_rev = np.concatenate([np.zeros(n_points - 1), h])
        overlaps = np.zeros(2 * n_points - 1)
        for i in range(2 * n_points - 1):
            if i < n_points:
                # For positive lags
                overlaps[i] = np.sum(h[:n_points-i] * (1 - h[i:]))
            else:
                # For negative lags
                overlaps[i] = np.sum(h[i-n_points+1:] * (1 - h[:n_points-(i-n_points+1)]))
        
        # Evaluate overlaps to get C5
        dx = 2.0 / n_points
        c5 = np.max(overlaps) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find maximum overlap lag
        max_k = np.argmax(overlaps)
        
        # Compute gradient
        # Gradient at position i is -h[(i + k) % n] if k >= 0 or h[(i - k) % n] if k < 0
        # For k = max_k, 
        grad = np.zeros(n_points)
        if max_k < n_points:
            # Positive lag
            for i in range(n_points - max_k):
                grad[i] -= h[i + max_k]
        else:
            # Negative lag
            k_neg = max_k - n_points + 1
            for i in range(n_points - k_neg):
                grad[i + k_neg] -= h[i]
        
        # Add 1 for each component
        grad += 1

        # Solve linear subproblem: minimize grad^T * d, subject to 0 <= d <= 1, sum(d) = n_points/2
        # Best direction is d = argmin d^T * grad s.t. 0 <= d <= 1, sum(d) = n_points/2
        # This can be solved with simple strategy: sort by gradient, assign smallest values where grad is largest
        
        # Create indices and sort by decreasing gradient values
        indices = np.argsort(-grad)
        d = np.zeros(n_points)
        sum_d = 0
        
        # Initialize d with minimum and maximum values to satisfy sum
        target = n_points / 2.0
        for idx in indices:
            if sum_d + 1.0 <= target:
                d[idx] = 1.0
                sum_d += 1.0
            elif sum_d < target:
                d[idx] = target - sum_d
                sum_d = target
            if sum_d >= target:
                break

        # Update h
        h = (1 - learning_rate) * h + learning_rate * d
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)

        # Check convergence
        if iteration > 0 and abs(c5 - best_c5) < tolerance:
            break

    # Final computation of best solution
    expanded_h = np.concatenate([best_h, np.zeros(n_points - 1)])
    expanded_h_rev = np.concatenate([np.zeros(n_points - 1), best_h])
    overlaps = np.zeros(2 * n_points - 1)
    for i in range(2 * n_points - 1):
        if i < n_points:
            # For positive lags
            overlaps[i] = np.sum(best_h[:n_points-i] * (1 - best_h[i:]))
        else:
            # For negative lags
            overlaps[i] = np.sum(best_h[i-n_points+1:] * (1 - best_h[:n_points-(i-n_points+1)]))
    dx = 2.0 / n_points
    final_c5 = np.max(overlaps) * dx
    return (best_h.tolist(), final_c5, n_points)
