import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Take maximum value and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    # Clip
    h = np.clip(h, 0, 1)
    # Rescale
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 100  # Reasonable size for optimization
    dx = 2.0 / n_points
    max_iter = 500
    swarm_size = 50
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset = 20
    time_start = time.time()
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use parent construction
        h_init = np.array(initial_h_values)
        h_init = project_h(h_init, n_points)
        particles.append(h_init.copy())
        velocities.append(np.zeros_like(h_init))
        
        # Add noise particles
        for _ in range(swarm_size - 1):
            h = h_init + np.random.normal(0, 0.05, n_points)
            h = project_h(h, n_points)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Random initialization
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solutions
    pbest = [p.copy() for p in particles]
    pbest_vals = [evaluate_h(p, dx) for p in particles]
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    iter_no_improve = 0
    for iter_num in range(max_iter):
        # Check time budget
        if time.time() - time_start > budget_s - 1.0:
            break
            
        # Evaluate all particles
        current_vals = []
        for i, p in enumerate(particles):
            val = evaluate_h(p, dx)
            current_vals.append(val)
            
            if val < pbest_vals[i]:
                pbest_vals[i] = val
                pbest[i] = p.copy()
                
                if val < gbest_val:
                    gbest_val = val
                    gbest = p.copy()
                    iter_no_improve = 0  # Reset

        iter_no_improve += 1
        if iter_no_improve > stagnation_limit:
            # Inject diversity - reset some particles
            reset_count = int(swarm_size * diversity_reset / 100)
            indices = np.random.choice(swarm_size, reset_count, replace=False)
            for i in indices:
                if 'initial_h_values' in globals():
                    h = np.array(initial_h_values)
                    h = project_h(h, n_points)
                else:
                    h = np.random.rand(n_points)
                    h = project_h(h, n_points)
                particles[i] = h
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                pbest[i] = h.copy()
                pbest_vals[i] = evaluate_h(h, dx)
            
            # Reset stagnation counter
            iter_no_improve = 0

        # Update particles
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_h(particles[i], n_points)
    
    # Final evaluation of gbest
    final_c5 = evaluate_h(gbest, dx)
    
    # Return best solution found
    return (gbest.tolist(), final_c5, n_points)
