import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute correlations for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # Convert to actual overlap values
    overlaps = corr * dx
    # Return maximum overlap
    return np.max(overlaps)

def normalize_h(h, target_sum):
    # Normalize h to have the target sum
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        return h
    h = h * target_sum / current_sum
    return h

def enforce_constraints(h, n_points):
    # Ensure values are in [0, 1]
    h = np.clip(h, 0, 1)
    # Ensure sum constraint
    h = normalize_h(h, n_points / 2)
    return h

def mutate_h(h, mut_rate=0.05):
    # Mutation: add random noise and normalize
    noise = np.random.normal(0, 0.1, size=h.shape)
    h = h + noise
    h = enforce_constraints(h, len(h))
    return h

def crossover_h(h1, h2):
    # Single-point crossover
    n = len(h1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([h1[:point], h2[point:]])
    child2 = np.concatenate([h2[:point], h1[point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    # Select parent via tournament
    idxs = np.random.choice(len(population), size=tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def create_initial_population(n_points, pop_size, initial_h=None):
    population = []
    # Use initial_h if provided, otherwise create random
    if initial_h is not None:
        population.append(enforce_constraints(initial_h.copy(), n_points))
        pop_size -= 1
    else:
        # Start with a uniform distribution
        pop = np.ones(n_points) * 0.5
        pop = enforce_constraints(pop, n_points)
        population.append(pop)
        pop_size -= 1

    for _ in range(pop_size):
        # Create a random valid individual
        h = np.random.random(n_points)
        h = enforce_constraints(h, n_points)
        population.append(h)

    return population

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 45  # starting with standard size
    pop_size = 100
    generations = 120
    mut_rate = 0.05
    crossover_prob = 0.8
    elitism_rate = 0.2
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = None
    
    # Setup
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    population = create_initial_population(n_points, pop_size, initial_h)
    
    for gen in range(generations):
        # Check time budget
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            
        # Update best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = int(elitism_rate * pop_size)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite[:]
        
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child1, child2 = crossover_h(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            child1 = mutate_h(child1, mut_rate)
            child2 = mutate_h(child2, mut_rate)
            
            new_population.extend([child1, child2])
            
        # Trim to exact population size
        population = new_population[:pop_size]
    
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h, final_c5, n_points)
    else:
        # Try one final evaluation with the last population
        final_h = population[0]
        final_c5 = evaluate_h(final_h, dx)
        return (final_h, final_c5, n_points)
