import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from provided values or create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 47
        h = np.ones(n) * 0.5
        h[10:37] = 1.0  # Plateau in the middle
        h[0:10] = 0.0
        h[37:47] = 0.0
        h = h * 2.0 / np.sum(h)  # Normalize to sum = 2
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Project h to meet constraints: 0 <= h <= 1 and sum = n/2
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Objective function: compute overlap integral (C5)
    def compute_C5(h_vec):
        # Cross-correlation of h with (1 - h)
        corr = signal.correlate(h_vec, 1 - h_vec, mode="full")
        # Take the maximum value from the correlation
        return np.max(corr) * dx
    
    # Gradient computation: approximate with finite differences
    # of the maximum overlap term only
    def compute_gradient(h_vec, use_lag=None):
        h_copy = h_vec.copy()
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        
        # Center differences for gradient calculation
        for i in range(len(h_vec)):
            h_plus = h_copy.copy()
            h_minus = h_copy.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            # Clamp to [0,1] and rescale
            h_plus = project_h(h_plus)
            h_minus = project_h(h_minus)
            
            # Compute overlap at this point
            max_plus = compute_C5(h_plus)
            max_minus = compute_C5(h_minus)
            
            # Gradient estimate based on finite difference
            grad[i] = (max_plus - max_minus) / (2 * eps)
        
        return grad
    
    # Optimizer: stochastic gradient descent with projection
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    initial_C5 = best_C5
    max_iter = 5000
    eta = 0.1
    last_decay = 0
    
    start_time = time.time()
    
    for iter_num in range(max_iter):
        # Time check
        if time.time() - start_time > budget_s - 2:
            break

        # Compute current gradient
        grad = compute_gradient(best_h)
        
        # Update using gradient descent
        updated_h = best_h - eta * grad
        
        # Project to feasible region
        updated_h = project_h(updated_h)
        
        # Evaluate new solution
        new_C5 = compute_C5(updated_h)
        
        if new_C5 < best_C5:
            best_h = updated_h
            best_C5 = new_C5
            
        # Annealing
        if iter_num - last_decay > 1000 and eta > 1e-6:
            eta *= 0.5
            last_decay = iter_num
            
    # Final evaluation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), n_points)
