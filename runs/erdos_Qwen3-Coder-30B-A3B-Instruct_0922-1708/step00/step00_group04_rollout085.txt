import numpy as np
from scipy.signal import correlate
from scipy.optimize import differential_evolution
import time

def evaluate_erdos_solution(h):
    """Computes the true C5 value for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    h_full = np.concatenate([h, np.zeros(n-1)])
    h_rev = np.concatenate([np.zeros(n-1), h])
    corr = correlate(h_full, 1 - h_rev, mode='valid')
    max_corr = np.max(corr) * dx
    return max_corr

        # Normalize to integral = 1
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points

    # Optimization parameters
    max_iter = 1000
    current_iter = 0
    
    def objective(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        # Normalize
        integral = np.sum(h) * dx
        h = h * (n_points / 2) / integral
        return evaluate_erdos_solution(h)

        integral = np.sum(h) * dx

    # Clip bounds to [0,1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply initial solution
    current_h = h.copy()

    if current_c5 < best_c5:
        best_c5 = current_c5
        best_h = current_h.copy()

    # Use differential evolution
    result = differential_evolution(
