import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    """
    Evaluates the Erdős minimum overlap constant C_5 for a given step function h.
    """
    dx = 2.0 / len(h)
    # Compute the overlap integral for all possible shifts k
    correlation = correlate(h, 1 - h, mode="full")
    # Take only the relevant part (shifts from -len(h)+1 to len(h)-1)
    # We want shifts from 0 to 2*len(h)-2 (full overlap window)
    # But due to mode='full', it's from -(len(h)-1) to (len(h)-1)
    # So we extract from len(h)-1 to 3*len(h)-3 (which is 2*(len(h)-1) + 2*(len(h)-1))
    # Actually, since h is defined on [0,2], the meaningful shifts are from 0 to 2
    # The correlation return a full length of 2*len(h) - 1, centered at the middle
    # The previous logic had an error - we want to extract the shifted parts properly:
    # For the overlap the non-negative shifts (k=0 to len(h)-1) correspond to:
    # corr[len(h)-1 + k], which is index 0 to len(h)-1
    # But we consider k from 0 to len(h)-1 (not all possible, but for the meaningful numbers that define this integral)
    # But actually, we are returning max over all k, so we consider the range of correlation also
    # The indexes for correlation are from 0 to 2*len(h)-2 and center is len(h)-1
    # We want the max of integral values, which on the expanded array corresponds to:
    # shift k = 0 to len(h)-1, should use indices len(h)-1 to 2*len(h)-2
    # but we go only up to len(h)-1 for first k=0 (i.e. corr[len(h)-1]), second k=1 (corr[len(h)]), ...
    # and that's n=len(h) terms (i.e., corr[len(h)-1], ..., corr[2*len(h)-2])
    # But we also have negative shifts --        in reverse  -> [0, ..., len(h)-2] corresponding to k=[len(h)-1, ..., 1] 
    # But all shifts from k=0 to len(h)-1 are positive or zero shifts (like k in [0, 2] interval)
    # From our problem: h(x) is defined on [0,2], we want the overlap integral
    # so the relevant time shifts k are those such that the overlap is meaningful ...
    # For h(x) and h(x+k), using periodic interplay: dt = 2.0 
    # This is confusing due to how correlate works; let's think differently
    
    # By definition, for fixed k: integral = ∫ h(x) (1 - h(x+k)) dx
    # When we use np.correlate with 1-h as filter:
    # correlate(h, 1-h, mode="full") returns corr[i] = ∑ h[j] * (1 - h[j+i]) for all valid j
    # where the indices follow the correlation convention.
    
    # If h has len n, then the correlation returns 2*n - 1 entries
    # In correlation format:
    #    corr[0]   = sum_{j=0}^{n-1} h[j] * (1 - h[j- (n-1)])  
    #    corr[1]   = sum_{j=0}^{n-1} h[j] * (1 - h[j- (n-2)])  
    # ...
    #    corr[n-1] = sum_{j=0}^{n-1} h[j] * (1 - h[j- 0])
    # ...
    #    corr[2*n-2]= sum_{j=0}^{n-1} h[j] * (1 - h[j- (n-1)])
    # These are correlations with lag and are redundant for our purpose.
    
    # The actual correct overlaps (i.e., integral shifts) appear in the middle:
    # The consecutive overlapping ideals:
    #   
    # For k=0 to n-1, shift from j=0 to n-1, we eval:
    # ∫ h(x) (1 - h(x+k)) dx  = (in discrete: time=0 to n-1) sum over j h[j] * (1 - h[(j+k)%n])  
    # This corresponds to the overlap terms that we want.
    # But convolution with a flipped kernel:
    # np.correlate(h, (1-h)) in standard format is equivalent to:
    # sum h[m] * (1 - h[m - k]) for all m, the variable substitutions and definitions[down to ] is complex...
    
    # Instead, let's re-use the script's call convention.
    # Indeed, np.correlate(h, 1-h, mode="full")  
    # Results in an array of size 2*len(h) - 1
    # Let's extract it_directly  
    # The valid non-trivial overlap values (for k from 0 to 2*dx)  -- in full simulation
    # corresponds to index range [len(h)-1, 2*len(h)-2]
    
    # Actually, we'll loop over k=0 to n-1 compute integral:
    # integral for k = sum_{j=0}^{n-1} h[j] * (1 - h[(j+k)%n])  
    # Tricky but simpler is one iteration over all possible shifts.
    # Revisit faster sufficient: 
    # Use correlation at the location of interest
    # Use:  
    # corr = [0:2n-1] -> The portion relevant for positive k's:
    # for shift k = 0 to n-1, the equivalent index in corr is (n-1 + k) 
    # so: corr[n-1] to corr[2n-2] gives us last n elements, which are:
    # indices:  n-1 + k  for k in [0, n-1], so it covers k=0 to n-1
    
    # x = [0, ..., 2] with n equally spaced steps. 
    # slice corr from correlation index [n-1:2n-1] with slice [n-1:2*n-1]
    # i includes for k=0 to k=(n-1)
    # Let's use full method correctly: 
    # ANSWER - we need to properly extract all computed correlation values and find the max.
    # np.correlate(h, 1-h, mode='full') has 2*n -1 elements (e.g., if n=3):
    #   correlation index 0 to 5 (if n=3)
    #   terms are like sum h[j] * (1 - h[j + 0]), sum h[j] *(1- h[j+ 1]), ...
    # What we want: ∫ h(x) (1 - h(x + k)) dx    
    # That's same as 
    # ∑_{j=0}^{n-1} h[j] (1 - h[(j+k)%n]) 
    # So we look for correlations for shifts 0 to n-1:
    # i corresponds, for result c_k = correlation(n-1 + k)
    # e.g., for n=3: 
    # c_0 = corr[2], c_1 = corr[3], c_2 = corr[4]  
    # hence take corr[n-1:n-1+n] = corr[n-1:2*n-1]
    # This corresponds to all overlaps [k=0 to k=n-1] which is what we want
    
    overlaps = correlation[n-1:2*n-1] * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):

    start_time = time.time()
    
    # Parameters:
    n_points = 70
    n_particles = 20
    max_iterations = int(budget_s / 0.05)  # Rough estimate of iterations in the budget
    
    # Start from the initial solution if available else generate a new one
    if 'initial_h_values' in globals():

    else:
        # Create a random initial solution and normalize it
        h_base = np.random.rand(n_points)
        h_base = h_base / np.sum(h_base) * (n_points / 2)
    
    # Initialize particles in a population-based approach inspired by PSO
    particles = []
    velocities = []
    
    # One particle based on the initial h values
    h_corrected = np.clip(h_base, 0, 1)
    sum_h = np.sum(h_corrected)
    if sum_h > 0:
        h_corrected = h_corrected * (n_points/2) / sum_h
    
    particles.append(h_corrected)
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Additional particles: slightly perturb the base
    for i in range(n_particles - 1):
        p = h_base + np.random.normal(0, 0.02, n_points)
        p = np.clip(p, 0, 1)
        sum_p = np.sum(p)
        if sum_p > 0:
            p = p * (n_points/2) / sum_p
        particles.append(p)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO hyperparameters
    inertia_weight = 0.7
    c1, c2 = 2.0, 2.0
    
    # Initialize best solutions
    personal_best_particles = particles.copy()
    personal_best_costs = [evaluate_erdos_solution(p) for p in particles]
    global_best_index = np.argmin(personal_best_costs)
    global_best_cost = personal_best_costs[global_best_index]
    global_best_particle = personal_best_particles[global_best_index].copy()
    
    iteration = 0
    
    # Main optimization loop
    while iteration < max_iterations and (time.time() - start_time) < budget_s:
        for i in range(n_particles):
            p = particles[i]
            v = velocities[i]
            pb = personal_best_particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v = inertia_weight * v + c1 * r1 * (pb - p) + c2 * r2 * (global_best_particle - p)
            
            # Update position
            p = p + v
            p = np.clip(p, 0, 1)
            
            # Project to satisfy constraint sum(h) * dx = 1 --> sum(h) = n/2
            # dx = 2/n
            sum_p = np.sum(p)
            if sum_p > 0:
                p = p * (n_points / 2) / sum_p
            
            # Store new particle and compute its cost
            particles[i] = p
            cost = evaluate_erdos_solution(p)
            
            # Update personal best
            if cost < personal_best_costs[i]:
                personal_best_costs[i] = cost
                personal_best_particles[i] = p.copy()
                
                # Update global best if needed
                if cost < global_best_cost:
                    global_best_cost = cost
                    global_best_particle = p.copy()
                    
        iteration += 1

        # Tournaments to maintain diversity
        if iteration % 50 == 0:
            # Sort particles by cost
            indices = np.argsort(personal_best_costs)
            top_half = indices[:n_particles // 2]
            for i in range(n_particles // 2, n_particles):
                # Replace worst performing with new random particles
                particles[i] = np.random.rand(n_points)
                particles[i] = np.clip(particles[i], 0, 1)
                # Normalize it
                sum_p = np.sum(particles[i])
                if sum_p > 0:
                    particles[i] = particles[i] * (n_points / 2) / sum_p
                personal_best_particles[i] = particles[i].copy()
                personal_best_costs[i] = evaluate_erdos_solution(particles[i])
                
            # Update the global best
            global_best_index = np.argmin(personal_best_costs)
            global_best_cost = personal_best_costs[global_best_index]
            global_best_particle = personal_best_particles[global_best_index].copy()
                
    # Final evaluation of global best
    final_cost = global_best_cost
    
    return (global_best_particle.tolist(), final_cost, n_points)
