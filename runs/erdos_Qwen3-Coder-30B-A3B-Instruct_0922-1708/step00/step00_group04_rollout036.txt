import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default construction
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        h0[:n_points//2] = 1.0
        h0[n_points//2:] = 0.0
        h0 = h0 * 2.0 / n_points  # Normalize integral to 1

    # Discretization
    dx = 2.0 / n_points
    t_start = time.time()

    # Method: We optimize step function with fixed number of transitions
    # use a simpler approach with fewer parameters
    
    # Let's try gradient-based local search from initial point
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Optimization parameters
    max_iter = 10000
    learning_rate = 0.01
    momentum = 0.9
    
    # For gradient updates, we can try simple finite differences
    # or modify a few parameters
    v = np.zeros_like(h0)
    
    for i in range(max_iter):
        if time.time() - t_start > budget_s - 1:
            break
            
        current_c5 = evaluate_erdos_solution(best_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
        
        # Simple gradient ascent
        # We'll sample small random changes to approximate gradient
        h_trial = best_h.copy()
        # Perturb a few indices
        perturb_indices = np.random.choice(len(h0), size=min(5, len(h0)), replace=False)
        for idx in perturb_indices:
            # Randomly perturb value
            delta = np.random.normal(0, 0.01*learning_rate)
            h_trial[idx] = np.clip(h_trial[idx] + delta, 0, 1)
            
        # Force integral constraint
        integral = np.sum(h_trial) * dx
        h_trial = h_trial * (1.0 / integral)
        h_trial = np.clip(h_trial, 0, 1)
        
        # Check constraint
        trial_c5 = evaluate_erdos_solution(h_trial)
        if trial_c5 < best_c5:
            best_c5 = trial_c5
            best_h = h_trial.copy()
    
    # Ensure final check of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation at all lags
    corr = correlate(h, 1-h, mode='full')
    # Only consider the relevant lags [0, 2] 
    # corresponding to k in [0, 2]
    # For a full correlation of length 2*n-1, 
    # the central part corresponds to lags in [0, 2]
    # that's when indices from n-1 to n-1+2-1 = n is valid
    # so k's range is [0,2] corresponds to indices [n-1, n+1] 
    # which are len(corr)=2n-1 elements
    # Lag k = 0 corresponds to index n-1, k=2 corresponds to n+1 
    # But need to be careful here.
    # Full correlation is index 0 to 2*n-2. 
    # We want lags in [0,2] but due to the way correlate works 
    # For k from 0 to 2, we want indices from n-1 to n+1
    # so we take the central portion: corr[n-1 : n+1+1] = corr[n-1 : n+2] 
    # But that assumes k can go up to 2. 
    # Actually: for each k in [0,2], we want it  
    # But we actually don't know exactly what's being requested
    # The function call is in a fixed form.
    # Evaluate the full overlap over all lags
    # i.e., compute for each k in [0,2], the integral of h(x) * (1 - h(x+k))
    # The maximum over all these integrals is what we want.
    # Instead, if computing with finite differences:
    # It computes perform cross-correlation of h and 1-h
    # and returns max of all correlations * dx
    max_corr = np.max(corr) * dx
    return max_corr
