import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h"""
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to match target sum
        h = h * target_sum / current_sum
    return h

def perturb_h(h, sigma):
    """Perturb h with Gaussian noise and project back to feasible set"""
    h_new = h + np.random.normal(0, sigma, len(h))
    return project_h(h_new, len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    sigma = 0.1
    decay = 0.99
    patience = 1000
    patience_counter = 0
    
    # Initialize h from initial_h_values if available or create simple one
    if 'initial_h_values' in globals():

    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    current_c5 = best_c5
    
    for i in range(max_iter):

        # Decrease noise variance over time
        sigma *= decay

            break
            
    return (best_h.tolist(), best_c5, n_points)
