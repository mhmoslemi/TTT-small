import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial solution if available
    h0 = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 47 if h0 is not None else 100
    
    # Adjust n_points to ensure it's appropriate
    if h0 is not None:
        n_points = len(h0)
    else:
        h0 = np.ones(n_points) * 0.5
    
    # Normalize initial solution to satisfy constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Ensure bounds
    h0 = np.clip(h0, 0, 1)
    
    def evaluate_h(h):
        h = np.array(h)
        h = h * (n_points / 2) / np.sum(h)  # Renormalize to sum = n_points/2
        h = np.clip(h, 0, 1)  # Maintain bounds
        # Compute all cross-correlations
        corr = correlate(h, 1 - h, mode='full')
        # Extract upper half (positive lags)
        c5 = np.max(corr[len(corr)//2:]) * (2.0 / n_points)
        return c5
    
    # Direct optimization using scipy's minimize
    bounds = [(0, 1) for _ in range(n_points)]
    options = {'maxiter': 1000, 'disp': False}
    
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h)
    
    # Use differential evolution as global optimizer
    def objective(h):
        return evaluate_h(h)
    
    # Wrapper to track best solution
    best_solution = [best_h, best_c5]
    
    def callback(xk, convergence):
        c5 = evaluate_h(xk)
        if c5 < best_solution[1]:
            best_solution[0] = xk.copy()
            best_solution[1] = c5
    
    # Initial optimization with different algorithms
    for _ in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Use scipy.optimize.minimize with different methods for variety
        try:
            method_list = ['Nelder-Mead', 'Powell', 'CG', 'BFGS']
            method = method_list[np.random.randint(len(method_list))]
            res = minimize(objective, best_h.copy(), method=method, 
                           bounds=bounds, options=options, callback=callback)
            if not res.success:
                continue
            c5 = evaluate_h(res.x)
            if c5 < best_solution[1]:
                best_solution[0] = res.x.copy()
                best_solution[1] = c5
        except:
            pass
            
        # Occasionally try differential evolution
        try:
            res = differential_evolution(objective, bounds, maxiter=100, 
                                         popsize=10, mutation=(0.5, 1), 
                                         recombination=0.7, seed=seed)
            if not res.success:
                continue
            c5 = evaluate_h(res.x)
            if c5 < best_solution[1]:
                best_solution[0] = res.x.copy()
                best_solution[1] = c5
        except:
            pass
    
    # Final verification of the best solution
    final_h = np.clip(best_solution[0], 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = evaluate_h(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
