import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up problem
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 2000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Prepare correlation operations
    def compute_C5(h):
        h = np.array(h)
        # Compute overlaps for all shifts
        overlaps = []
        for k in range(2 * n_points - 1):
            # Compute integral for shift k
            if k < n_points:
                # h[0:n_points-k] * (1 - h[k:n_points]) 
                integrand = h[:n_points-k] * (1 - h[k:])
            else:
                # h[k-n_points+1:n_points] * (1 - h[0:n_points-(k-n_points+1)])
                integrand = h[k-n_points+1:] * (1 - h[:n_points-(k-n_points+1)])
            overlaps.append(np.sum(integrand) * dx)
        return max(overlaps)
    
    # Optimization wrapper
    def objective(h):
        return compute_C5(h)
    
    # Constraints
    def constraint_integral(h):
        return np.sum(h) - n_points / 2
    
    def constraint_bounds(h):
        return np.concatenate([h, 1 - h])
    
    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [
        {'type': 'eq', 'fun': constraint_integral},
    ]
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    last_improvement = 0
    
    # Iterative optimization steps
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Use scipy minimize with L-BFGS-B method
        result = minimize(objective, h, method='L-BFGS-B', 
                         bounds=bounds, constraints=constraints,
                         options={'maxiter': 10})
        if result.success:
            h = result.x
            C5 = compute_C5(h)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
                last_improvement = i
        else:
            # Reinitialize if optimization fails
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h / np.sum(h) * (n_points / 2)
        
        # No improvement for a while, reinitialize
        if i - last_improvement > 50:
            h = np.random.rand(n_points) * 0.5 + 0.25
            h = h / np.sum(h) * (n_points / 2)
            last_improvement = i
            
    # Final validation
    final_C5 = compute_C5(best_h)
    
    return (list(best_h), final_C5, n_points)
