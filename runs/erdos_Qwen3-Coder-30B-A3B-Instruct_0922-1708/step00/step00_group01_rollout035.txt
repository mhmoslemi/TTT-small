from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given h"""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraint sum(h) = n_points / 2"""
    h = np.array(h)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    # Normalize and clip
    h = np.clip(h * target_sum / current_sum, 0, 1)
    return h
