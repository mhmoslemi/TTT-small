    learning_rate = 0.1
    no_improve_count = 0
    best_c5 = float('inf')
    best_h = h.copy()

        # E-step: compute overlaps
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlaps[k] = overlap * dx
            
        # Normalize overlaps to form probability distribution
        p = overlaps / np.sum(overlaps)
        
        # M-step: update h
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            for k in range(2 * n_points - 1):
                j = (i + k) % n_points
                grad_h[i] += p[k] * (1.0 - h[j])
                
        h_new = h - learning_rate * grad_h
        
        # Clamp h to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new = h_new * (target_sum / np.sum(h_new))
        
        h = h_new
        
        # Decaying learning rate
        if iteration % 10 == 0 and iteration > 0:
            learning_rate *= 0.5
            
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
