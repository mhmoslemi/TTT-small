import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def sigmoid(x):
    return 1.0 / (1.0 + np.exp(-x))

def initial_h():
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Create uniform distribution
        h = np.ones(45) * (2.0 / 45.0)
        return h

def build_basis(n_points, m=None):
    if m is None:
        m = int(2 * sqrt(n_points))
    mu = np.linspace(0, 2, m+2)[1:-1]
    sigma = 2.0 / m
    def phi(x, mu, sigma):
        return sigmoid((x - mu) / sigma)
    return phi, mu, sigma

def optimize_h(budget_s):
    time_start = time.time()
    n_points = 100
    dx = 2.0 / n_points
    basis_fn, mu, sigma = build_basis(n_points)
    
    # Initialize h
    h = initial_h()
    if len(h) != n_points:
        # Interpolate h to n_points
        x_original = np.linspace(0, 2, len(h))
        x_new = np.linspace(0, 2, n_points)
        h = np.interp(x_new, x_original, h)
    
    # Use basis function expansion to represent h
    # We use sigmoid basis functions
    phi_matrix = np.zeros((n_points, len(mu)))
    x_vals = np.linspace(0, 2, n_points)
    for i, m in enumerate(mu):
        phi_matrix[:, i] = basis_fn(x_vals, m, sigma)
    
    a_coef = np.linalg.lstsq(phi_matrix, h, rcond=None)[0]
    
    # Elastic net regularization parameters
    gamma = 0.8
    lambda_reg = 0.01
    
    alpha = 0.01
    
    def compute_overlap(a):
        h_vals = phi_matrix @ a
        h_vals = np.clip(h_vals, 0, 1)
        # compute correlation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        corr = corr * dx
        return np.max(corr)
    
    def gradient(a, k):
        # derivative w.r.t. a of overlap for fixed k
        k_ind = k + n_points - 1
        h_vals = phi_matrix @ a
        h_vals = np.clip(h_vals, 0, 1)
        # This derivative is a bit tricky because of the max. 
        # We will look at the derivative with respect to a of the full correlation
        # But we want gradient of the maximum.
        # Evaluate gradient step by step. 
        # For overlapping regions
        grad = np.zeros_like(a)
        # This is a complex operation, for illustration and simplicity
        # We compute it numerically and do not overthink the derivation
        return grad
        
    # Gradient descent for 500 iterations or time limit
    max_iter = 500
    best_c5 = float('inf')
    best_a = None
    last_improvement = 0
    iteration = 0
    
    while iteration < max_iter:
        if time.time() - time_start > 0.95 * budget_s:
            break
            
        h_vals = phi_matrix @ a_coef
        h_vals = np.clip(h_vals, 0, 1)
        c5_val = compute_overlap(a_coef)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_a = a_coef.copy()
            last_improvement = iteration
            
        # Approximate gradient using finite differences for simplicity
        # Note: This is a simplified version of a potential exact gradient
        eps = 1e-5
        grad = np.zeros_like(a_coef)
        
        for i in range(len(a_coef)):
            a_plus = a_coef.copy()
            a_plus[i] += eps
            h_plus = np.clip(phi_matrix @ a_plus, 0, 1)
            c5_plus = compute_overlap(a_plus)
            
            a_minus = a_coef.copy()
            a_minus[i] -= eps
            h_minus = np.clip(phi_matrix @ a_minus, 0, 1)
            c5_minus = compute_overlap(a_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update
        a_coef = a_coef - alpha * grad
        # Elastic net penalty
        sign_a = np.sign(a_coef)
        a_coef = a_coef - alpha * gamma * sign_a - alpha * (1 - gamma) * a_coef
        
        # Project to feasible region
        h_vals = phi_matrix @ a_coef
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale
        if np.sum(h_vals) != 0:
            h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
            # Update coefficients
            a_coef = np.linalg.lstsq(phi_matrix, h_vals, rcond=None)[0]
        
        # Adaptive learning rate
        if iteration - last_improvement > 20:
            alpha *= 0.9
            
        iteration += 1
        
        # Early stopping: no improvement in 50 iterations
        if iteration - last_improvement > 50:
            break
    
    if best_a is None:
        best_a = a_coef.copy()
    
    h_final = phi_matrix @ best_a
    h_final = np.clip(h_final, 0, 1)
    # Adjust for integral constraint
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    
    # final evaluation
    c5_final = compute_overlap(best_a)
    
    return h_final, c5_final, n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    h, c5, n = optimize_h(budget_s)
    return (h, c5, n)
