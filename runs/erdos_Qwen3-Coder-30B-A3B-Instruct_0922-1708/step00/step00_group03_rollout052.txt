import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Initial sum constraint violated"
    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05 * dx, 0.05 * dx, n_points)
    
    # Ensure the sum constraint
    h *= target_sum / np.sum(h)
    
    def evaluate_c5(h_vals):
        h_vals = np.array(h_vals)
        # Compute overlap integral
        conv = correlate(h_vals, 1 - h_vals, mode="full")
        max_lag = len(conv) // 2
        result = np.max(conv[:max_lag]) * dx  # Only consider non-negative lags
        return result
    
    def objective(h_vals):
        # Convex surrogate: log sum of exponentials of overlap values
        h_vals = np.array(h_vals)
        conv = correlate(h_vals, 1 - h_vals, mode="full")
        max_lag = len(conv) // 2
        overlaps = conv[:max_lag]
        alpha = 0.1  # Temperature parameter
        return np.log(np.sum(np.exp(alpha * overlaps))) * dx
    
    def gradient(h_vals):
        h_vals = np.array(h_vals)
        conv = correlate(h_vals, 1 - h_vals, mode="full")
        max_lag = len(conv) // 2
        overlaps = conv[:max_lag]
        alpha = 0.1
        
        # Gradient calculation (simplified for performance)
        # Approximate gradient with finite differences
        eps = 1e-8
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_minus = h_vals.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Bound constraints
    bounds = [(0.0, 1.0) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    
    best_c5 = evaluate_c5(h)
    best_h = h.copy()
    
    # Optimization using L-BFGS-B
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < 500:
        iter_count += 1
        try:
            # Use scipy's L-BFGS-B with a simple approximation of gradient
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10}
            )
            if not result.success:
                break
            h = result.x.copy()
            
            # Update constraints
            h_sum = np.sum(h) * dx
            h *= target_sum / h_sum
            
            # Clamp and rescale
            h = np.clip(h, 0, 1)
            h_sum = np.sum(h) * dx
            h *= target_sum / h_sum
            
            current_c5 = evaluate_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()
                last_improvement = 0
            else:
                last_improvement += 1
            
            # Early stopping condition
            if last_improvement > 100:
                break
                
        except Exception as e:
            break
            
    return (best_h.tolist(), best_c5, n_points)
