import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    sum_target = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * sum_target
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * sum_target
    
    # Optimization parameters
    alpha = 0.01
    beta = 0.9
    min_alpha = 1e-6
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute arrays for efficiency
    ones = np.ones(n_points)
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute correlation: h * (1 - h) with full mode
        corr = signal.correlate(h, ones - h, mode='full')
        # Integrate over the valid range [0, 2] 
        # The valid correlations are from index n_points-1 to 2*n_points-2
        valid_corr = corr[n_points-1:2*n_points-1]
        k_max = np.argmax(valid_corr)
        c5 = valid_corr[k_max] * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
        
        # Update learning rate if no improvement for patience steps
        if patience_counter >= patience and alpha > min_alpha:
            alpha *= 0.5
            patience_counter = 0
        
        # Compute gradient with respect to h
        # Gradient of sum h[i]*(1-h[i+k]) w.r.t. h[i]
        # = (1-h[i+k]) - h[i] * d(h[i+k])/dh[i] = (1-h[i+k]) - h[i] * 0
        # But we actually need the full effect of shifting h
        # Let's use finite differences for robustness
        # d/dh[i] ∫ h(x)(1-h(x+k)) dx = (1-h[i+k])  
        # But this depends on the shift k
        # Approximate gradient using cross-correlation derivative
        
        # Instead, let's do gradient descent directly:
        # Since c5 = sum(h[i] * (1 - h[i+k_max])), 
        # ∂c5/∂h[i] = (1 - h[i+k_max]) for the indices that are valid
        
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad_h[i] = (1 - h[j])
        
        # Update h with momentum
        if iteration == 1:
            velocity = np.zeros(n_points)
        velocity = beta * velocity - alpha * grad_h
        h = h + velocity
        
        # Project h back to valid range and sum constraint
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h)
        if sum_h > 0:
            h = h * (sum_target / sum_h)

    return (best_h.tolist(), best_c5, n_points)
