import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with smaller size for faster evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize with the initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    else:
        # Create a symmetric plateau-like initial solution
        h = np.ones(n_points) * 0.5
        # Create a piecewise constant function
        mid = n_points // 2
        h[:mid] = 0.7
        h[mid:] = 0.3
        # Normalize to sum to target_sum
        h = h * target_sum / np.sum(h)
        
    best_h = h.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    # Simplex size for local refinement
    simplex_size = 10
    
    # Time monitoring
    max_iter = 3000
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Genetic Algorithm step
        # Create offspring population
        population_size = 100
        population = []
        
        # Start with current best
        population.append(best_h.copy())
        
        # Generate rest from mutation of current best or random
        for _ in range(population_size - 1):
            mut_h = best_h.copy()
            # Mutation: add random noise
            noise = np.random.normal(0, 0.05, n_points)
            mut_h += noise
            # Clip to [0,1]
            mut_h = np.clip(mut_h, 0, 1)
            # Rescale to correct sum
            mut_h = mut_h * target_sum / np.sum(mut_h)
            population.append(mut_h)
        
        # Evaluate population
        fitness = []
        for ind in population:
            c5 = evaluate_h(ind, dx)
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Select top individuals
        top_indices = sorted_indices[:max(10, population_size // 3)]
        top_individuals = [population[i] for i in top_indices]
        
        # Nelder-Mead refinement on top individuals
        for ind in top_individuals[:simplex_size]:
            if time.time() - start_time > budget_s - 1:
                break
                
            # Nelder-Mead using scipy
            x0 = ind.copy()
            bounds = [(0, 1)] * n_points
            try:
                # Use a simple local optimizer, but constrained in domain
                def opt_func(x):
                    # Project x to feasible region
                    x = np.clip(x, 0, 1)
                    # Rescale to target sum
                    x = x * target_sum / np.sum(x)
                    return evaluate_h(x, dx)
                
                result = minimize(opt_func, x0, method='Nelder-Mead', options={'maxiter': 100})
                if result.success:
                    refined = result.x.copy()
                    refined = np.clip(refined, 0, 1)
                    refined = refined * target_sum / np.sum(refined)
                    new_c5 = evaluate_h(refined, dx)
                    if new_c5 < best_c5:
                        best_h = refined.copy()
                        best_c5 = new_c5
            except:
                pass  # Skip if optimization fails
        
        # Update best if better found
        if fitness[0] < best_c5:
            best_h = sorted_population[0].copy()
            best_c5 = fitness[0]
            
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    # Compute the overlap integral
    # This is a cross-correlation of h with (1 - h)
    h = np.array(h)
    reflected = 1 - h
    # Use full mode for correlation to cover all lags
    correlation = correlate(h, reflected, mode='full')
    # The maximum value over all lags times dx
    max_overlap = np.max(correlation)
    c5 = max_overlap * dx
    return c5
