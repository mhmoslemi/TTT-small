import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Reasonable balance of accuracy and speed
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available, else random
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure bounds and sum constraint
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    # Frank-Wolfe optimization
    max_iter = 5000
    learning_rate = 0.1
    tolerance = 1e-6
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        conv = signal.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        overlaps = conv * dx
        
        # Find k that maximizes overlap
        k_max = k_indices[np.argmax(overlaps)]
        
        # Compute gradient (approximate)
        grad = np.zeros(n_points)
        if k_max >= 0:
            # h[i] * (1 - h[i + k_max])
            for i in range(n_points - k_max):
                grad[i] += 1 - h[i + k_max]
        else:
            # h[i] * (1 - h[i + k_max]) = h[i] * (1 - h[i - |k_max|])
            for i in range(-k_max, n_points):
                grad[i] += 1 - h[i - (-k_max)]
        
        # Solve linear subproblem: minimize grad^T * d subject to 0 <= d <= 1, sum(d) = target_sum
        # This is a simple optimization - select elements with most negative gradients
        d = np.zeros(n_points)
        sorted_indices = np.argsort(-grad)  # descending order
        remaining_sum = target_sum
        for i in sorted_indices:
            if remaining_sum <= 0:
                break
            d[i] = min(1.0, remaining_sum)
            remaining_sum -= d[i]
        if remaining_sum > 0:
            # Distribute remaining equally to satisfy sum
            for i in range(n_points):
                if d[i] < 1.0:
                    add = min(1.0 - d[i], remaining_sum)
                    d[i] += add
                    remaining_sum -= add
                    if remaining_sum <= 0:
                        break
        
        # Update h
        h = (1 - learning_rate) * h + learning_rate * d
        
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h / h_sum * target_sum
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Check convergence
        if iteration > 0 and abs(current_c5 - best_c5) < tolerance:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    h = np.array(h)
    conv = signal.correlate(h, 1 - h, mode='full')
    overlaps = conv * dx
    return np.max(overlaps)
