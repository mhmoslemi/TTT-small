import numpy as np
from scipy.optimize import minimize
from scipy.fft import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values or create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        h = np.ones(n_points) * 0.5  # Simple initialization
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Adjust initial h to meet constraint sum(h) = target_sum
    h = h * target_sum / np.sum(h)
    
    # Precompute FFTs for efficient correlation
    def compute_overlap(h_vals):
        h_ext = np.concatenate([h_vals, np.zeros(n_points)])
        h_fft = fft(h_ext)
        one_minus_h = 1 - h_vals
        one_minus_h_ext = np.concatenate([one_minus_h, np.zeros(n_points)])
        one_minus_h_fft = fft(one_minus_h_ext)
        corr = ifft(h_fft * np.conj(one_minus_h_fft)).real
        corr = corr[:n_points]  # Only the valid correlations
        return np.max(corr) * dx
    
    # Objective function - minimize maximum overlap
    def objective(h_vals):
        return compute_overlap(h_vals)
    
    # Gradient estimation (central differences)
    def grad_objective(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_minus = h_vals.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Optimization loop
    start_time = time.time()
    best_h = h.copy()
    best_c5 = objective(h)
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        iteration += 1
        g = grad_objective(best_h)
        
        # Gradient descent step (adaptive learning rate)
        lr = 1e-3
        h_new = best_h - lr * g
        
        # Project onto feasible region [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * target_sum / np.sum(h_new)
        
        # Update if improvement
        new_c5 = objective(h_new)
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = 0
        else:
            last_improvement += 1
            
        # Early termination if no improvement in 100 iterations
        if last_improvement > 100:
            break
            
    # Final computation of C5
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
