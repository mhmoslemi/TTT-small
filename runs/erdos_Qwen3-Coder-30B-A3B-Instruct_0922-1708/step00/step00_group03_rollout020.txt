import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h, dx=2.0/70):
    """Evaluate the C5 bound for a given step function h."""
    h = np.array(h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, population_size, initial_h_values):
    """Initialize population with one known solution and random samples."""
    population = [initial_h_values]
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select an individual using tournament selection."""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmin([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def crossover(parent1, parent2):
    """Perform segment swapping crossover."""
    n = len(parent1)
    split_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:split_point], parent2[split_point:]])
    child2 = np.concatenate([parent2[:split_point], parent1[split_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, dx=2.0/70):
    """Mutate solution h."""
    h = h.copy()
    if np.random.rand() < mutation_rate:
        n_points = len(h)
        mutate_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        for i in mutate_indices:
            h[i] += np.random.uniform(-0.1, 0.1) * dx
        h = np.clip(h, 0, 1)
        # Rescale to maintain integral constraint
        h = h * (n_points / 2) / np.sum(h)
    return h

def rescale_h(h, n_points):
    """Rescale h to ensure integral constraint."""
    h = np.array(h)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    initial_h_values = np.array(initial_h_values)
    
    population_size = 100
    max_generations = 300
    tournament_size = int(0.2 * population_size)
    mutation_rate = 0.2
    
    # Initialize population
    population = initialize_population(n_points, population_size, initial_h_values)
    fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
    
    best_h = population[np.argmin(fitnesses)]
    best_C5 = min(fitnesses)
    
    start_time = time.time()
    generation = 0
    
    # GA loop
    while time.time() - start_time < budget_s - 10 and generation < max_generations:
        generation += 1
        new_population = []
        
        # Selection and reproduction
        parents = []
        for _ in range(population_size // 2):
            p1 = tournament_selection(population, fitnesses, tournament_size)
            p2 = tournament_selection(population, fitnesses, tournament_size)
            children = crossover(p1, p2)
            new_population.extend(children)
            
        # Mutation
        mutated_population = [mutate(h, mutation_rate, dx) for h in new_population]
        
        # Keep best population
        population = mutated_population
        fitnesses = [evaluate_erdos_solution(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_C5:
            best_C5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx]
        
        # Check for stopping condition
        if generation % 50 == 0:
            if np.std(fitnesses) < 1e-6:
                break
    
    # Ensure the final solution is valid
    best_h = rescale_h(best_h, n_points)
    final_C5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)
