    target_sum = n_points / 2.0
    
    # Helper to compute C5
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        # Cross-correlation of h with (1-h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Find maximum correlation value
        max_val = np.max(conv)
        return max_val * dx
    
    # Helper to project h onto feasible set
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Project to constraint sum = target_sum
    h = project_h(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # Gradient-based optimization with SGD
    learning_rate = 0.1
    iterations = 0
    max_iterations = 5000
    
    while iterations < max_iterations and (time.time() - start_time) < budget_s - 1:
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # Find dominant lag(s)
        conv = correlate(h_current, 1 - h_current, mode='full')
        # Identify dominant lag indices

        dominant_lags = [max_idx]

            eps = 1e-5
            h_plus = h_current.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps

        h = h_current - learning_rate * grad
