import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 42
        # Initialize with a triangular wave
        h = np.linspace(0, 1, n_points//2)
        h = np.append(h, np.linspace(1, 0, n_points//2))
        h = h / np.sum(h) * n_points / 2
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.max(np.correlate(h, 1-h, mode='full')) * dx
    
    start_time = time()
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time() - start_time < budget_s - 5:
        iter_count += 1
    
        # Compute current overlap values (approximate with full convolution)
        conv = np.correlate(h, 1-h, mode='full')
        current_c5 = np.max(conv) * dx
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Stop if target achieved or no improvement
        if best_c5 < 0.3807 or no_improvement_count > max_no_improvement:
            break
            
        # Find the most constraining lag
        max_idx = np.argmax(conv)
        max_k = max_idx - (len(h) - 1)  # convert to actual lag
        
        # Construct linearized convex program
        # We want to minimize: sum_i h[i] * (1 - h[i + k]) * dx
        # For a fixed k, we approximate it as a linear function
        # Using first-order Taylor expansion around h:
        # h[i] * (1 - h[i+k])  ≈ h[i] * (1 - h[i+k]) + (1 - h[i+k]) * (h[i] - h[i]) = h[i] * (1 - h[i+k])
        # But the gradient w.r.t. h[i] is (1 - h[i+k]) - h[i] * (1 - h[i+k]) = 1 - h[i+k] - h[i] + h[i]*h[i+k]
        # This is complex, let's directly compute and solve a convex approximation
                
        # Create an approximate linear function at current h
        # Linearize f(x) = h[i]*(1 - h[i+k]) around h:
        # If we treat h[i] as variable and h[i+k] as a constant:
        # f = h[i] + (1 - h[i+k]) - h[i] = 1 - h[i+k] (constant w.r.t. h[i], does not depend on h[i])
        # That does not work. 
        # What we want is to solve max_k sum h[i] (1 - h[i+k]) * dx subject to sum h = n_points/2, and 0 <= h <= 1
        
        # Approximate the objective: minimize the maximum h[i] * (1 - h[i+k]) over all k
        # To do this via convex programming, we linearize around current h:
        # h[i] * (1 - h[i + k]) ≈ h[i] * (1 - h[i+k]) + (1 - h[i+k]) * (h[i] - h[i]) = h[i] - h[i]*h[i+k]
        # We're minimizing sum(h[i] - h[i]*h[i+k])
        # A better approach:
        # Use the fact that h[i] * (1 - h[i+k]) = h[i] - h[i]*h[i+k], 
        # So the rate of change w.r.t. h[i] = 1 - h[i+k] (minus h[i] times derivative of h[i+k] which is 0, since fixed)
        # That's not enough, just approximate h[i] * h[i+k] by its linearization
        # h[i] * h[i+k] ≈ h[i] * h[i+k]^0 + h[i]^0 * h[i+k] + ... but h[i]^0, h[i+k]^0 are just constants.
        # So: h[i] * h[i+k] ≈ h[i] * (fixed h[i+k]) -> Too inaccurate and nonlinear.
        
        # Simpler approach:
        # For each k, we linearize the term over h:
        # Let th = h[i]*(1-h[i+k])  
        # Then d/dh[i] = (1-h[i+k])
        # So we linearize:
        # The gradient at h: 1 - h[i+k]  
        # But no, we’re minimizing: sum_{i,k} h[i] * (1 - h[i+k])
        # subj to bounds on h[i], sum h = 1/dx
        # Maybe using cvxpy's nonlinear programming for the full problem is more accurate.
        
        # Let's restart with projected gradient approach. 
        # Or use projected subgradient as a quick way to take a nearly feasible step:
        # Find largest h[i](1 - h[i+k]) 
        # Also, recent experiments show that previously the full H matrix is computed via correlate,
        # So just try to improve the direction.
        
        # Try a local descent using gradient of largest term.
        max_k_idx = np.argmax(conv)  
        max_k = max_k_idx - (len(h) - 1)  
    
        # Compute gradient for the max_k term
        # Element-wise gradient of h[i] * (1 - h[i + k]) with respect to h[i] = 1 - h[i + k]  
        # But we compute gradient of the sum (regardless of k) with respect to h[i]
        # = sum_{k} (1 - h[i + k]) for i - k in valid range
        # This is messy. Instead perform a simple descent step.
        
        # Compute gradient per component for max_cost term 
        gradient = np.zeros_like(h) 
        for i in range(len(h)):
            if i + max_k < len(h) and i + max_k >= 0:
                gradient[i] = 1.0 - h[i + max_k]
            else:
                gradient[i] = 0.0
        
        # Take a small step in the direction of gradient descent (but inverted sign since we're minimizing)  
        step_size = 0.01
        h_new = h + step_size * gradient
            
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to sum to n_points/2
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new = h_new * scale_factor
        
        h = h_new
        
    return (best_h.tolist(), best_c5, n_points)
