import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize with parent solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Basis function setup: sigmoidal basis
    m = max(20, int(2 * np.sqrt(n_points)))  # Number of basis functions
    centers = np.linspace(0, 2, m+2)[1:-1]  # Avoid endpoints
    scales = np.ones(m) * (2.0 / m) * 0.5
    
    def basis_func(x, center, scale):
        return expit((x - center) / scale)
    
    def get_basis_matrix(x_vals):
        # Compute all basis functions for given x values
        B = np.zeros((len(x_vals), m))
        for j in range(m):
            B[:, j] = basis_func(x_vals, centers[j], scales[j])
        return B
    
    # Optimization loop
    best_c5 = float('inf')
    best_h = None
    
    def objective_and_gradient(a):
        # Reconstruct h from coefficients
        x_vals = np.linspace(0, 2, n_points)
        B = get_basis_matrix(x_vals)
        h_vals = B @ a
        h_vals = np.clip(h_vals, 0, 1)
        
        # Ensure integral constraint
        current_sum = np.sum(h_vals) * dx
        scale_factor = target_sum / current_sum
        h_vals = h_vals * scale_factor
        h_vals = np.clip(h_vals, 0, 1)
        
        # Compute overlap integrals
        # we need all correlations for lags k in [0, 2] with step size dx
        h_full = np.concatenate([h_vals, np.zeros(n_points)])
        overlaps = np.correlate(h_full, 1 - h_full, mode="full")
        # extract the valid lags [0,2] -> [n_points, 3*n_points-1]
        # overlapping regions are the ones with valid ranges
        corr_results = []
        for k_idx in range(n_points):  # k in [0, dx, ..., 2-dx]
            k_off = k_idx
            corr_value = overlaps[n_points + k_off] * dx  # scaled by dx
            corr_results.append(corr_value)
        c5_val = max(corr_results)
        
        # If time's up, early stop
        if time.time() - start_time > 0.95 * budget_s:
            return float('inf'), np.zeros_like(a)
        
        # Approximate gradient (simplified finite difference for now)
        grad = np.zeros_like(a)
        
        # Simple finite difference gradient approximation
        eps = 1e-6
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus_vals = B @ a_plus
            h_plus_vals = np.clip(h_plus_vals, 0, 1)
            current_sum_plus = np.sum(h_plus_vals) * dx
            scale_factor_plus = target_sum / current_sum_plus
            h_plus_vals *= scale_factor_plus
            h_plus_vals = np.clip(h_plus_vals, 0, 1)
            
            h_full_plus = np.concatenate([h_plus_vals, np.zeros(n_points)])
            overlaps_plus = np.correlate(h_full_plus, 1 - h_full_plus, mode="full")
            corr_plus = max(overlaps_plus[n_points:n_points*2] * dx)
            
            a_minus = a.copy()
            a_minus[i] -= eps
            h_minus_vals = B @ a_minus
            h_minus_vals = np.clip(h_minus_vals, 0, 1)
            current_sum_minus = np.sum(h_minus_vals) * dx
            scale_factor_minus = target_sum / current_sum_minus
            h_minus_vals *= scale_factor_minus
            h_minus_vals = np.clip(h_minus_vals, 0, 1)
            
            h_full_minus = np.concatenate([h_minus_vals, np.zeros(n_points)])
            overlaps_minus = np.correlate(h_full_minus, 1 - h_full_minus, mode="full")
            corr_minus = max(overlaps_minus[n_points:n_points*2] * dx)
            
            grad[i] = (corr_plus - corr_minus) / (2 * eps)
        
        return c5_val, grad
    
    def callback(a):
        nonlocal best_c5, best_h
        h_vals = get_basis_matrix(np.linspace(0, 2, n_points)) @ a
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals) * dx
        scale_factor = target_sum / current_sum
        h_vals = h_vals * scale_factor
        h_vals = np.clip(h_vals, 0, 1)
        
        # Evaluate current solution
        h_full = np.concatenate([h_vals, np.zeros(n_points)])
        overlaps = np.correlate(h_full, 1 - h_full, mode="full")
        corr_results = []
        for k_idx in range(n_points):
            corr_value = overlaps[n_points + k_idx] * dx
            corr_results.append(corr_value)
        c5_val = max(corr_results)
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h_vals.copy()
        
        if time.time() - start_time > 0.95 * budget_s:
            return True
        return False
    
    # Run optimization
    # Initial guess
    B_init = get_basis_matrix(np.linspace(0, 2, n_points))
    try:
        a_init = np.linalg.lstsq(B_init, h_init, rcond=None)[0]
        # Ensure satisfying integral
    except:
        a_init = np.random.normal(0, 0.1, m)
    
    # Minimize objective function
    try:
        # Simple gradient descent
        a = a_init.copy()
        alpha = 0.005
        for it in range(1000):
            if time.time() - start_time > 0.95 * budget_s:
                break
            cost, grad = objective_and_gradient(a)
            if np.isnan(cost) or np.all(np.isnan(grad)):
                break
            a -= alpha * grad
            if it % 20 == 0:
                # Project a to feasible region
                a = np.clip(a, -1, 1)
                # Project to satisfy constraints?
                # For now just clip coefficients
                # Simple projection to L1 ball
                # elastic net, with some radius?
                # This is a simplified version
                a = a * 0.99  # reduce magnitude slowly (semi-elastic net)
                
                # Update if necessary
                if callback(a):
                    break
    except Exception as e:
        pass

    # Final projection to get h
    B_final = get_basis_matrix(np.linspace(0, 2, n_points))
    h_final = B_final @ a
    h_final = np.clip(h_final, 0, 1)
    current_sum = np.sum(h_final) * dx
    h_final = h_final * (target_sum / current_sum)
    h_final = np.clip(h_final, 0, 1)
    
    # Re-validate best
    if best_h is not None:
        h_final = best_h
    
    h_final = np.clip(h_final, 0, 1)  
    current_sum = np.sum(h_final) * dx
    h_final = h_final * (target_sum / current_sum)
    h_final = np.clip(h_final, 0, 1)
    
    # One more final validation
    h_full = np.concatenate([h_final, np.zeros(n_points)])
    overlaps = np.correlate(h_full, 1 - h_full, mode="full")
    corr_results = []
    for k_idx in range(n_points):
        corr_value = overlaps[n_points + k_idx] * dx
        corr_results.append(corr_value)
    final_c5 = max(corr_results)
    
    # Return final results
    return (h_final, final_c5, n_points)
