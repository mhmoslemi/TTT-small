import numpy as np
import cvxpy as cp
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except NameError:
        # Generate a smooth initial guess
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h_init = 0.5 + 0.3 * np.sin(np.pi * x) * np.cos(0.5 * np.pi * x)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    h_best = h_init.copy()
    c5_best = evaluate_erdos_solution(h_best)
    
    # Optimization loop
    iter_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Convex optimization approach
        h_current = h_best.copy()
        
        # Define optimization variables
        h = cp.Variable(n_points)
        
        # Define the constraints
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        # Objective: approximate the maximum overlap
        # Using a linearized version for convexity
        # For each k, we want to minimize sum(h[i] * (1 - h[i+k]))
        # Since h[i] * (1 - h[i+k]) is nonlinear, we linearize
        # For current h_current, approximate
        linearized = np.convolve(h_current, 1 - h_current, mode='full')
        # Find the maximum value and use it as constraint
        max_val = np.max(linearized)
        
        # Use a conservative upper bound for our linear model
        objective = cp.Minimize(cp.sum(h))
        
        # Since we want to minimize the maximum correlation, 
        # we'll instead try with the full convolution rounded to a reasonable bound
        # Linearization: approximate h * (1-h) as a linear term
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CBC, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_new = h.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new.copy()
                    last_improvement = iter_count
            else:
                # Fallback if optimization fails
                h_new = h_current
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                c5_new = evaluate_erdos_solution(h_new)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_new.copy()
                    last_improvement = iter_count
        except:
            # Fallback to simple repair
            h_new = h_current
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5_new = evaluate_erdos_solution(h_new)
            if c5_new < c5_best:
                c5_best = c5_new
                h_best = h_new.copy()
                last_improvement = iter_count

            break
        if iter_count - last_improvement > 100:
            break
            
    return (h_best, c5_best, n_points)

def evaluate_erdos_solution(h):
    # This function evaluates the true objective value for given h
    # It computes the maximum of the overlap integral
    n = len(h)
    h_ext = np.concatenate([h, np.zeros(n)])  # Extend for full convolution
    h_complement = 1 - h
    h_complement_ext = np.concatenate([h_complement, np.zeros(n)])
    conv_result = signal.correlate(h_ext, h_complement_ext, mode='full')
    max_overlap = np.max(conv_result[:2*n-1])
    return max_overlap / 2.0  # normalize by dx = 2/n
