import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    """Evaluate C5 for a given h"""
    # Compute cross-correlation of h and (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1, sum(h) = n_points/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
