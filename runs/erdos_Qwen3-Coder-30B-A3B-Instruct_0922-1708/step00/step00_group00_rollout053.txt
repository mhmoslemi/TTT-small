import numpy as np
from scipy.signal import correlate
import time

def evaluate(h, dx):
    # Compute overlaps for all shifts
    h_rev = h[::-1]
    overlaps = correlate(h, h_rev, mode="full")  # this is sum(h[i] * h[i+k])
    # The actual overlap we want is sum(h[i] * (1 - h[i+k]))
    # so we compute: sum(h[i] * 1) - sum(h[i] * h[i+k]) = sum(h) - overlaps
    # But sum(h) * dx = 1, so sum(h) = n_points / 2, dx = 2 / n_points => sum(h) = 1/dx
    n = len(h)
    # But actually, for overlap_k = sum_{i} h[i] * (1 - h[i+k]), we do:
    # compute sum(h[i] * 1) = sum(h) = 1/dx for each k
    # subtract sum(h[i] * h[i+k]) to get the actual overlap.
    # We compute all overlaps in one go:
    h_sum = np.sum(h)
    # Flip h to compute cross-correlations
    ones = np.ones_like(h)
    h_ones = correlate(ones, h_rev, mode="full")
    h_h = correlate(h, h_rev, mode="full")
    overlaps = h_ones - h_h
    # Now there's a mismatch of indexing, overlapping terms, etc.
    # Use a cleaner approach: compute each term separately

    # For k = 0, overlap = sum(h[i] * (1 - h[i])) = sum(h) - sum(h^2)
    # For k = 1, overlap = sum(h[i] * (1 - h[i+1]))
    # [0, 1, 2, ..., n-1], and we look at h[i] * h[i+1] with appropriate indexing.
    # We compute: sum on k of max_{shifts} of overlap_k
    # overlap_k = sum(h[i] * (1 - h[i+k])) for valid i
    # So overlap_k = sum(h) - sum(h[i] * h[i+k])
    # But `sum(h)` is fixed, and we move k in [-n+1, n-1]

    # Full supports: full length of cross-correlation
    full_len = 2 * n - 1
    # actual overlap_k for k in range(-n+1, n) (k=0 means no shift)
    overlaps = np.full(full_len, h_sum, dtype=float)
    # now subtract the cross-correlations
    # correlate(h, h_rev, mode="full") is sum over j of h[i] * h[i-j] = h[i]*h[i], h[i]*h[i-1], ...
    # it's actually the correlation at each k with proper indexing...
    # Let me carefully calculate in double loop
    overlaps = np.full(2 * n - 1, 0.0)
    for k in range(-n+1, n):
        total = 0.0
        for i in range(n):
            j = i + k
            if 0 <= j < n:
                total += h[i] * (1.0 - h[j])
        overlaps[k + n - 1] = total
    # We return the maximum overlap normalized by dx
    return np.max(overlaps) * dx

def project_and_rescale(h, n_points, dx):
    # project onto [0,1]
    h = np.clip(h, 0.0, 1.0)
    # rescale to match constraint sum(h) = 1/dx = n_points/2
    h_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(h_sum) > 1e-10:
        h = h * target_sum / h_sum
    return h

def loss_and_grad(h, n_points, dx, temperature=0.1):
    # Compute overlaps
    h_rev = h[::-1]
    h_one = np.ones_like(h)
    # cross-corelations
    h_h = correlate(h, h_rev, mode="full")
    h_ones = correlate(h_one, h_rev, mode="full")
    # Now apply: overlap_k = sum(h[i] * (1 - h[i+k])) = sum(h) - sum(h[i]*h[i+k])
    full_len = 2 * n_points - 1
    h_sum = np.sum(h)
    overlaps = h_sum - h_h
    # Now we approximate max with soft-max: log(sum(exp(τ * overlap_k))) / τ
    # Apply temperature to soften the max
    soft_max = np.sum(np.exp(temperature * overlaps))
    loss = np.log(soft_max) / temperature
    
    # Compute gradient
    exp_overlaps = np.exp(temperature * overlaps)
    weights = exp_overlaps / soft_max
    
    # Gradient using chain rule
    # dL/dh = (1/τ) sum_k w_k * d/dh (overlap_k) = (1/τ) sum_k w_k * (1 - d/dh h[i+k]) = ...
    # Actually, we're taking d/dh (sum h[i] (1-h[i+k]))
    # Let's do this more carefully:
    # overlap_k = sum_{i} h[i] * (1 - h[i+k]) = sum_{i} (h[i] - h[i]*h[i+k])
    # d/dh_m (overlap_k) = d/dh_m h[i] (1 - h[i+k]) = (1 - h[i+k]) * (i==m) - h[i]* (i+k==m)
    # Specifically:
    # d/dh_m overlap_k = - (h[m+k] if (m+k) is valid) if (m+k) is valid
    #                  = - (h_{m+k}) if m+k in [0,n)
    # d/dh_m overlap_k = 1 if m == m-k, should consider the h[i]*h[i+k] term too
    # No, better:
    # d/dh_m overlap_k = 1 if m == i, -h[i+k] if m == i+k?
    # Really: overlap_k = sum h[i] (1 - h[i+k]) 
    # d/dh_m overlap_k = (1 - h[m+k]) if m is valid for i=m-k, 0 else
    # So: d/dh_m (overlap_k) = -h_{m+k} if m+k is valid and k is valid

    # Actually I compute the gradient cumulatively:
    # df/dh_m
    # overlap_k = sum h[i] (1 - h[i+k])
    # So for fixed k, d/dh_m overlap_k = (1 - h[m+k]) if m = i, -h[i] if m = i+k (if valid)
    # Since there's a summation over k, the full gradient is:
    # df/dh_m = sum_k w_k * (1 - h[m+k]) [-1 if -k, i=m] + sum_k w_k * (-h[i]) if (m=i+k)

    # Better yet let's do elementwise gradient computation
    # The keras-like approach: create A_matrix such that
    # overlap_k = A_k @ h, where A_k is the coeff matrix for shift k.
    # However, since we compute the max over all shifts, it's not a single matrix but a full gradient.

    # Indirect: we compute for each k, what is d/dh overlap_k, then sum weighted.
    # loss = log(sum(exp(τ * overlap_k))) / τ

    # Let's do through finite differences for correct row d/dh_m overlap_k
    # Bruke følgende redaksjon:
    # For each h[i], we compute how it contributes to different overlaps...
    # d/dh_m overlap_k = sum_i d/dh_m h[i] (1 - h[i+k]) = (i==m) (1-h[i+k]) - (i+k==m) h[i]
    # if m != i and m != i+k, then d/dh_m overlap_k = 0
    # elif m == i, then d/dh_m overlap_k = 1 - h[i+k]
    # elif m == i+k, then d/dh_m overlap_k = - h[i]
    # The k indexing is a bit complex: it shifts opposite to array indexing...

    # The key loop:s
    # We build gradient matrix that represents d/dh overlap_k.
    # Loop over k from -n+1 to n-1
    # For each k, compute g_k = d/dh (overlap_k) = [gradient vector]
    # Let's build: For all possible k, compute contribution.
    # Element-wise of gradient: 
    # Also, the sign of overlap_k might be implicitly sign?
    # The original loss gradient should be derived as:
    # d/dh_m (log sum exp τ overlap_k) = 1/τ  sum_k w_k d/dh_m overlap_k

    # This is too inventive.
    # Something simpler: 
    # gradient conceptually is sum_k w_k d/dh overlap_k
    # Let's compute by loop for over del/overlaps:
    # Since we have overlaps for each shift k over 2*n-1 indices
    # Center (index n-1) to be deleted
    # For each m k, the contribution d/dh_m overlap_k is:
    # If m = i , the contribution is (1 - h[i+k]) 
    # If m = i+k, the contribution is -h[i]
    # So it's mix of constructions involving shifts
    # We have overlaps as a full vector of 2n-1 elements.
    # The elements correspond to shifts k in -n+1 to +n-1
    # Consider all ingredient terms where we find h[i] in sum h[i] * (1 - h[i+k])

    # Condense to computational loops:
    grad = np.zeros_like(h)
    for k_idx, overlap in enumerate(overlaps):
        k = k_idx - (n_points - 1)
        # Apply gradient: sum_k w_k d/dh overlap_k
        # For shift k, contributions to h[m]:
        # - if m = i then derivative is (1-h[i+k])  
        # - if m = i+k then derivative is -h[i]
        # These must be properly indexed:
        # We want phor all valid i, h[i] contributes to gradient:
        # d/dh[i] (overlap_k) = (1-h[i+k]) + (-h[i]) where shift i+k is valid
        # Actually separating into term definitions:
        # For each i, let's compute how it contributes to gradient

        # We include 1-term:
        for i in range(n_points):
            if i+k >= 0 and i+k < n_points:  # valid shift
                # The index is such that i += (i+k) => i goes 1-indexed, but it's ...
                # Imagine: overlap_k involves h[i] * (1 - h[i+k])
                # d/dh[i] = 1 - h[i+k]
                # d/dh[i+k] = -h[i]
                # The index: shift index 0 means i=0, i+k=0 => i=0, k=0; 
                # Shift index 1 means i=0, i+k=1 => k=1
                # General: There's 2*n-1 elements centered at k=0. Our indices go from k=-n+1 to n-1
                # So how to apply the chain rule in order?

                # Simpler, ergo, we registered that in line above we are associated k
                # Gradient for m-th term:
                # laplace-like:
                # Contribution of d/dh[i] overlap_k needs (1-h[i+k]) for m=i
                # Contribution of d/dh[i+k] overlap_k also needs -h[i] for m=(i+k)  
                # Start loop from min(n_points, n_points+k), max(0, k) ...
                # Let me construct the final loop
                # Total grad[m] = sum_k w_k d/dh_m overlap_k across all valid k

                # For shift k index k_idx in overlaps, let me compute the gradient according to loop
                # This is a bit odd due to issue of indexing. May be better to loop manually
                # But let's just assign gradients directly
                # shift variable: k = k_idx - (n_points-1)
                # Resulting expression is 
                # If m = i: (1 - h[i+k]) 
                # If m = i+k: -h[i]

                if 0 <= i + k < n_points:
                    # gradient from d/dh[i] (overlap_k) = (1 - h[i+k])
                    grad[i] += weights[k_idx] * (1 - h[i+k])
                    # gradient from d/dh[i+k] (overlap_k) = -h[i]
                    grad[i + k] += weights[k_idx] * (-h[i])
                else:
                    # invalid indexing
                    pass

    # Just take normal rows and average or proper connections 
    # simpler approach:
    # compute by convolutions or finite difference
    # Let's create 2n-1 envelops  - one shifting left, one right 
    # to get convolutions equivalent to d/dh overlap_k 
    # What about building more directly:
    # For a given k-index, we compute the gradient from d/dh overlap_k: 
    # It's a matrix):
    #        [ 1 - h[0+k], -h[0], 0, 0, ... ]
    # d/d h[] overlap_k = [ -h[1], 1-h[1+k], -h[1], 0, ... ]
    # etc. 

    # .. to avoid confusion I calculate by simple reuse of existing library:
    # centroids and simple matrix traces...
    # Let's workaround:okusáeVec!
    grad = np.zeros_like(h)
    for k in range(-n_points+1, n_points):
        k_idx = k + (n_points - 1)
        # Compute the correct physicalizing map:
        for i in range(n_points):
            if 0 <= i+k < n_points:
                grad[i] += weights[k_idx] * (1 - h[i+k])
                grad[i+k] -= weights[k_idx] * h[i]
    return loss, grad

    start_time = time.time()
    time_budget = budget_s - 0.01
    max_iter = 10000
    alpha = 0.1  
    beta = 0.5
    temperature = 0.1
    
    # Initialize h from `initial_h_values` if available

        # Check sum constraint
        h_sum = np.sum(h)
        if abs(h_sum - n_points / 2.0) > 1e-6:

    except NameError:
        # None provided: create simple one
        h0 = np.zeros(n_points)
        h0[:n_points//2] = 1.0
        h = h0.copy()
        h = project_and_rescale(h, n_points, dx)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Restart counter
    last_improve = 0
    patience = 50
    
    # Iteration loop
    for j in range(max_iter):
        if time.time() - start_time > time_budget:
            break
        loss, grad = loss_and_grad(h, n_points, dx, temperature)
        # Project gradients
        grad = np.clip(grad, -1, 1)
        # Backtracking line search for step size α 
        # Try reducing α until loss improves
        step = alpha
        h_new = h - step * grad
        h_new = project_and_rescale(h_new, n_points, dx)
        new_loss, _ = loss_and_grad(h_new, n_points, dx, temperature)
        # Avoid overshooting
        while new_loss > loss:
            step *= beta
            if step < 1e-8:
                break
            h_new = h - step * grad
            h_new = project_and_rescale(h_new, n_points, dx)
            new_loss, _ = loss_and_grad(h_new, n_points, dx, temperature)
        
        h = h_new
        # evaluate C5
        c5 = evaluate(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improve = j
            
        if j - last_improve > patience:
            # Restart with new random point
            h = np.random.rand(n_points) * 0.5
            h = project_and_rescale(h, n_points, dx)
            last_improve = j
    return best_h.tolist(), best_c5, n_points
