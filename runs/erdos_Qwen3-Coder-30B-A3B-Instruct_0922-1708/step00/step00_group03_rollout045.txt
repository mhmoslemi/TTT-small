import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Maximize over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with a mix of initial solution and random solutions."""
    population = []
    # Add initial solution
    population.append(initial_h)
    # Add random solutions
    for _ in range(pop_size - 1):
        # Generate random h with sum = n_points/2
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=10):
    """Select a parent using tournament selection."""
    selected = []
    for _ in range(len(population) // 2):
        tournament_indices = random.sample(range(len(population)), tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def crossover(parent1, parent2):
    """Perform crossover (segment swap) between two parents."""
    n = len(parent1)
    # Choose crossover point
    point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, sigma=0.1):
    """Mutate the solution."""
    n = len(h)
    if random.random() > mutation_rate:
        return h
    # Randomly select points to mutate
    num_mutate = max(1, int(0.1 * n))
    indices = random.sample(range(n), num_mutate)
    for i in indices:
        h[i] += random.uniform(-sigma, sigma) * (2.0 / n)
    # Clip and rescale
    h = np.clip(h, 0, 1)
    # Rescale to satisfy constraint
    sum_h = np.sum(h)
    if abs(sum_h - n/2) > 1e-6:
        h *= (n / 2) / sum_h
    return h

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 70
    
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
    else:
        init_h = np.random.rand(n_points)
        init_h = init_h * (n_points / 2) / np.sum(init_h)
        init_h = np.clip(init_h, 0, 1)
    
    pop_size = 100
    population = initialize_population(n_points, pop_size, init_h)
    
    # Calculate initial fitnesses
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    min_fitness = min(fitnesses)
    best_h = population[fitnesses.index(min_fitness)]
    
    generation = 0
    no_improvement = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1.0 and no_improvement < max_no_improvement:
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Generate new population through crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            if i + 1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
            else:
                child1 = selected[i]
                child2 = selected[i]  # Just copy if odd number
                
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Keep population size same
        population = new_population[:pop_size]
        
        # Evaluate new population
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        current_min_fitness = min(fitnesses)
        if current_min_fitness < min_fitness:
            min_fitness = current_min_fitness
            best_h = population[fitnesses.index(min_fitness)]
            no_improvement = 0
        else:
            no_improvement += 1

        generation += 1
        
    return (best_h.tolist(), min_fitness, n_points)
