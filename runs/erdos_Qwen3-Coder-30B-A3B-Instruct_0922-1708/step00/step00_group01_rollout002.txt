import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a symmetric construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        h[mid-5:mid+5] = 0.8
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Project to valid sum
    h = h * target_sum / np.sum(h)
    
    # Learning parameters
    learning_rate = 0.1
    anneal_every = 1000
    max_iter = 5000
    best_c5 = float('inf')
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integral for all lags
        cross_corr = signal.correlate(h, 1 - h, mode='full')
        c5_val = np.max(cross_corr) * dx
        
        if c5_val < best_c5:
            best_c5 = c5_val
            best_h = h.copy()
        
        # Find dominant lag(s) that contribute most to C5
        max_idx = np.argmax(cross_corr)
        # We compute gradient for this dominant lag
        # Gradient from h[i] term in the overlap integral
        # For overlap at lag k, gradient w.r.t. h[i] is (1 - h[i+k]) * dx
        # But we need to be more careful here
        # To optimize C₅ = max_k (h * (1-h))[k], we can approximate:
        # We take gradient of the dominant term
        
        # Construct gradient for the dominant overlap
        gradient = np.zeros_like(h)
        # The cross-correlation with 1-h at lag k = max_idx - (n_points-1)
        # Actually, full correlation indices go from -(n_points-1) to (n_points-1)
        k = max_idx - (n_points - 1)
        
        if k >= 0:
            # h[i] contributes to overlap at lag k iff i+k < n_points
            # i from 0 to n_points-1-k
            for i in range(n_points - k):
                gradient[i] += -dx  # Unlike normal derivation, we're 
        else:
            # k < 0, so i+k in [0, n_points-1] => i in [-k, n_points-1]
            k = -k
            # h[i] contributes to overlap at lag -k, so we look at h[i+k] 
            for i in range(n_points - k):
                gradient[i+k] += -dx
                
        # Never adjust if we're out of bounds - just use a safer approach
        # More accurately handling:
        # Evaluate the gradient as: ∂(sum over k of h[i] * (1-h[i+k])) / ∂h[i]
        # If we are interested in C₅ = max_k of overlap integral, 
        # and we know dominant lag k is responsible for maximum,
        # The gradient w.r.t. h[i] is just: -dx * (1 if i+k<n else 0) = -dx * (i+k < n)
        
        # So across all valid i:
        gradient = np.zeros_like(h)
        # As per cross-correlation indices:
        # For a cross-correlation at lag = max_idx - (n_points-1) = offset,
        # We need to sum contribution at this lag, i.e., h[i] * (1 - h[i+offset]) 
        # The derivative at h[i] would be - (1 - h[i+offset]) in the sum for this overlap lag
        # Which is: -dx * (1 - h[i+offset]) (if i+offset is valid)
        # Competition: for each i, backpropagate to see which one affects max overlap
        
        # Less exact but effective:
        # Get indices of the dominant lag (full correlation)
        offset = max_idx - (n_points - 1)
        if offset >= 0:
            for i in range(n_points - offset):
                gradient[i] -= dx * (1 - h[i + offset])
        else:
            offset = -offset
            for i in range(n_points - offset):
                gradient[i + offset] -= dx * (1 - h[i])
        
        # SGD update
        h -= learning_rate * gradient
        
        # Projection: h in [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to enforce sum = target_sum
        h = h * target_sum / np.sum(h)
        
        # Anneal learning rate
        if (i + 1) % anneal_every == 0:
            learning_rate *= 0.5
            
    # Final evaluation
    final_cross_corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_cross_corr) * dx
    return (best_h.tolist(), float(final_c5), len(best_h))
