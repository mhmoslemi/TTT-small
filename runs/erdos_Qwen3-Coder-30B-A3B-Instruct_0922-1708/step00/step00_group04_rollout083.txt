import numpy as np
from scipy.optimize import minimize

import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available, else create a smooth one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * 1.0  # normalize to sum to 1
    else:
        # Create a simple smooth starting vector
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h = np.maximum(0, 1 - np.abs(x - 1))
        h = h / np.sum(h) * 1.0

    n_points = len(h)
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Optimization loop
    iteration = 0
    last_improvement = 0
    
    # Surrogate function parameters
    beta = 1.0
    
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 100:
        iteration += 1
        
        # Compute gradient using finite differences for a smoother approach
        # Evaluate objective
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # If we're close to target, try to carefully refine
        if best_c5 < 0.3807:
            break
            
        # Perturb h slightly and assess direction
        h_new = h.copy()
        # Perturbation
        delta = 0.01
        # Create a direction vector for gradient estimation
        perturb = np.random.normal(0, 0.01, n_points)
        perturb = perturb - np.mean(perturb)
        h_perturbed = h + delta * perturb
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed / np.sum(h_perturbed) * 1.0
        
        # Try adjustment
        if compute_c5(h_perturbed, dx) < current_c5:
            h = h_perturbed
        else:
            # Try gradient descent with current best gradient estimation
            h = h - 0.001 * perturb
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * 1.0
            
        # Adaptive surrogate parameter
        if iteration % 10 == 0:
            beta *= 1.05

    # Final evaluation
    final_c5 = compute_c5(best_h, dx)
    return (best_h.tolist(), final_c5, len(best_h))

def compute_c5(h, dx):
    # Maximum cross-correlation with 1-h
    h1 = np.array(h)
    h2 = 1 - h1
    # Use full correlation to capture all lags
    corr = np.correlate(h1, h2, mode='full')
    max_corr = np.max(corr)
    # Normalize by dx
    return max_corr * dx
