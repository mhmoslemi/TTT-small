    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    try:

        if len(h) != n_points:
            h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(initial_h_values)), initial_h_values)
    except:
        h = np.ones(n_points) * 0.5
    
    # Ensure constraint ∫ h(x) dx = 1, i.e., sum(h) * dx = 1
    h = h * (1.0 / (sum(h) * dx))
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vec):
        h_vec = h_vec.copy()
        h_vec = h_vec / (sum(h_vec) * dx) * 1.0  # Normalize to get ∫ h dx = 1
        # Compute correlation-like scores for all shifts k
        score = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = max(score) * dx
        return c5
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (1.0 / (sum(h_vec) * dx))
        return h_vec

    # Simple gradient descent with constraints
    max_iter = 10000
    lr = 0.01
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        #Approximate gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        for j in range(len(h)):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[j] = (c5_plus - c5_minus) / (2 * eps)
        
        h = h - lr * grad
        h = project_h(h)
    
    return (best_h.tolist(), best_c5, n_points)
