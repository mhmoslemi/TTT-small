import numpy as np
from scipy import signal
import time
import random

def evaluate_h(h, dx):
    """Evaluate C5 for a given h"""
    corr = signal.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, n_points):
    """Rescale h to meet constraint sum(h) * dx = 1"""
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * (target_sum / current_sum)
    return h

def mutate_h(h, n_points, mutation_rate=0.2, delta=0.1):
    """Mutate h with probability mutation_rate"""
    if random.random() > mutation_rate:
        return h
    dx = 2.0 / n_points
    h = h.copy()
    # Select random indices to mutate
    n_mutate = max(1, int(0.1 * n_points))
    indices = random.sample(range(n_points), n_mutate)
    for i in indices:
        h[i] += random.uniform(-delta * dx, delta * dx)
        h[i] = max(0.0, min(1.0, h[i]))
    return rescale_h(h, n_points)

def crossover_h(h1, h2, n_points):
    """Perform crossover of two h vectors"""
    point = random.randint(1, n_points - 1)
    child = np.concatenate([h1[:point], h2[point:]])
    return rescale_h(child, n_points)

def initialize_population(n_points, pop_size, initial_h):
    """Initialize the GA population"""
    pop = [initial_h]
    while len(pop) < pop_size:
        h = np.random.rand(n_points)
        h = rescale_h(h, n_points)
        pop.append(h)
    return pop

def tournament_selection(pop, fitnesses, k=10):
    """Select individuals using tournament selection"""
    selected = []
    for _ in range(len(pop) // 2):
        tournament_indices = random.sample(range(len(pop)), k)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(pop[winner_index])
    return selected

    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available, otherwise generate random

    except NameError:

        h0 = rescale_h(h0, 70)
        
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 100
    max_generations = 500
    patience = 50
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h, dx)
    no_improvement = 0
    elapsed = 0
    
    pop = initialize_population(n_points, pop_size, h0)
    
    for gen in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed > budget_s - 30:
            break
            
        # Evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in pop]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = pop[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            break
            
        # Selection
        selected = tournament_selection(pop, fitnesses)
        
        # Crossover and mutation
        children = []
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                child1 = crossover_h(selected[i], selected[i+1], n_points)
                child2 = crossover_h(selected[i+1], selected[i], n_points)
                children.extend([mutate_h(child1, n_points), mutate_h(child2, n_points)])
            else:
                children.append(mutate_h(selected[i], n_points))
                
        # Replacement
        pop.extend(children)
        # Sort by fitness and keep top pop_size
        combined = list(zip(pop, fitnesses))
        combined.sort(key=lambda x: x[1])

    # Final evaluation
