import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default construction: 70 points
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[1] = 0.0
        h[2] = 0.0
        h[3] = 0.0
        h[4] = 0.0
        h[5] = 0.0
        h[6] = 0.0
        h[-1] = 0.0
        h[-2] = 0.0
        h[-3] = 0.0
        h[-4] = 0.0
        h[-5] = 0.0
        h[-6] = 0.0
        h[-7] = 0.0

    # Ensure sum(h) == n_points / 2
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    dx = 2.0 / n_points
    c5_bound = None
    best_c5 = float('inf')
    best_h = None
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 60000
    patience = 100
    patience_counter = 0
    last_improvement = 0
    
    start_time = time.time()

    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags using full correlation
        # This is: integral of h(x)(1 - h(x+k)) dx for all valid k
        # The correlate can be used with unflipped h (since one is 1-h)
        h_shifted = np.concatenate([h, np.zeros(n_points)])
        h_one_minus = 1 - h
        correlate_result = signal.correlate(h_shifted, h_one_minus, mode='full')
        # Extract the overlap values for different lags: from -n+1 to n-1
        overlaps = correlate_result[n_points-1:] * dx
        max_k = np.argmax(overlaps)
        max_overlap = overlaps[max_k]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
            last_improvement = i
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            break
            
        # Gradient for the max_k
        # gradient[i] = (1 - h[i+max_k])  if i+max_k < n_points
        #              = -h[i-max_k]        if i-max_k >= 0
        gradient = np.zeros(n_points)
        
        # Compute the gradient for the worst-case lag
        if max_k < n_points:
            # case: max_k is positive shift, overlapping forward
            # gradient[i] = (1 - h[i + max_k])
            for j in range(n_points - max_k):
                gradient[j] += (1 - h[j + max_k]) * dx
        else:
            # case: max_k is negative shift
            # gradient[i] = -h[i - max_k]
            for j in range(max_k - n_points + 1, n_points):
                k = j - max_k + n_points - 1
                if k < n_points:
                    gradient[j] -= h[k] * dx

        # Update h
        h = h - learning_rate * gradient
        
        # Clip h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Subtract from all entries the necessary amount to enforce sum(h) == n_points/2
        target_sum = n_points / 2
        current_sum = np.sum(h)
        diff = target_sum - current_sum
        if abs(diff) > 1e-10:
            # distribute diff uniformly over all entries
            h = h + diff / n_points
            
        # Ensure h again lies in [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust for sum constraint by scaling
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            h = h * (target_sum / current_sum)
    
    if best_h is not None:
        c5_bound = best_c5
        h_values = best_h
    else:
        # fallback to the initial h
        h_values = h
        corr = signal.correlate(np.concatenate([h, np.zeros(n_points)]), 1-h, mode='full')
        final_overlaps = corr[n_points-1:] * dx
        c5_bound = np.max(final_overlaps)
    
    return (h_values.tolist(), c5_bound, n_points)
