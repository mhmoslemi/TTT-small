import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47  # Use initial construction size if available
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

        assert len(h) == n_points
    else:

        h = np.ones(n_points) * 0.5
        # Taper at ends
        taper_width = n_points // 4
        for i in range(taper_width):
            factor = float(i) / taper_width
            h[i] = 0.1 + 0.4 * factor
            h[-1-i] = 0.1 + 0.4 * factor
    
    # Normalize to required sum
    h *= target_sum / np.sum(h)
    
    # Optimization parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    convergence_tol = 1e-5
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Precompute shifts for efficient overlap computation
    shifts = np.arange(n_points)
    shifted_h = np.array([np.roll(h, i) for i in range(n_points)])
    
    start_time = time.time()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s * 0.99:
            break
            
        # Compute overlaps with all shifts using FFT for efficiency
        h_ext = np.concatenate([h, np.zeros(n_points)])
        h_rev = np.flip(h)
        # Correlation equivalent to overlap integral
        overlaps = signal.fftconvolve(h_ext, h_rev, mode='full')[:n_points]
        # Correct for convolution convention (we want h(x) * (1-h(x+k)), so reflect one side)
        # Need to compute for each k: sum(h[i] * (1 - h[(i+k)%n]):
        # This is equivalent to correlation of h with 1-h, with periodic padding
        # Use convolution: h * (1 - h) to compute overlaps (shifted with signs corrected)
        # We manually compute it:
        all_overlaps = np.zeros(n_points)
        for k in range(n_points):
            # For overlap at lag k: sum(h[i]*(1-h[(i+k)%n]) ) 
            hshift = np.roll(h, k)
            dotp = np.dot(h, 1 - hshift)
            all_overlaps[k] = dotp

        # Smooth approximation to max
        max_it = np.max(all_overlaps)
        c5_approx = np.log(np.sum(np.exp(tau * (all_overlaps - max_it)))) / tau + max_it
        c5_approx *= dx
        
        # Check best
        if c5_approx < best_c5:
            best_c5 = c5_approx
            best_h = h.copy()
        
        # Compute gradients
        # Gradient of sum(exp(tau * overlap)) / sum(exp(tau * overlap)) * (tau * d/dh overlap)
        # Let S = sum(exp(tau * overlap)) 
        # Grad = (sum(exp(tau*overlap) * tau * d/dh overlap) / S) - (S' / S^2) * S 
        # Simplify: derivative of log sum exp
        # Forward pass
        exp_terms = np.exp(tau * all_overlaps)
        sum_exp = np.sum(exp_terms)
        weights = exp_terms / sum_exp
        
        # Gradient w.r.t. each h[i] 
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            # This is sum over k of: weights[k] * d/dh[i] overlap[k]
            # For overlap[k] = sum h[j] * (1-h[(j+k)%n]), 
            # d/dh[i] overlap[k] = -(1-h[(i-k)%n]) if i==j else 0
            # But j is different from i in sum, so just of the form
            # sum over j of h[j] * d/dh[i] (1-h[(j+k)%n]) = - h[i] * d/dh[i] h[(i+k)%n]
            # This is unclear, let's simplify naive approach:
            # For each k, overlap[k] = sum_i h[i] * (1 - h[(i+k)%n])
            # So d/dh[i] overlap[k] = (1 - h[(i+k)%n]) if i is not involved 
            # This is wrong again. If we're looking at a specific h[i], then 
            # derivative only if that h[i] is in the i-th term!
            # overlap[k] = sum_j h[j] * (1 - h[(j+k)%n])
            # d/dh[i] = (1 - h[(i+k)%n])
            # and it's 0 if there's no i-term.
            # So this needs careful sum loop 
            # For h[i], the sum is over k, where d/dh[i] overlap[k] = (1 - h[(i+k)%n])
            # but this is very long derivative chain divergence. Need different way
            
            for k in range(n_points):
                if (i - k) % n_points == i:  # backward shift, must be same index (that's not possible for sum)
                    # Better approach:
                    # In overlap[k] = sum over j of h[j] * (1 - h[(j+k)%n])
                    # Gradient wrt h[i] = (1 - h[(i+k)%n]) 
                    pass  # let it be through indexing
                    
            # Better yet: use FFT loop:
            # Simpler and implicit way:
            # Use centered difference for gradient
            # d/dh[i]overlap[k] = (1 - h[(i + k)%n]) 
            
        # Compute d/dh[i] of overlap[k] = sum_j h[j] * (1 - h[(j+k)%n]) = h[i] * (1 - h[(i+k)%n]) when j==i 
        grad = np.zeros(n_points)
        # More accurate way: 
        # For i-th component, overlap[lag k] derivative = 1 - h[(i-k)%n]
        # Let d_k = (1 - h[(i-k)%n])
        # Then Grad_i = sum_k weights[k] * d_k
        # Loop over all shifts:
        for i in range(n_points):
            proj_grad = 0.0
            for k in range(n_points):
                j = (i - k) % n_points
                proj_grad += weights[k] * (1 - h[j])
            grad[i] = proj_grad * dx

        # Update step
        h = h - learning_rate * grad
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        # Scale to correct sum
        h = h * (target_sum / np.sum(h))
        
        # Anneal temperature
        tau *= anneal_factor
        if tau < 1e-8:  # Prevent underflow
            tau = 1e-8
        
        # Check for convergence
        if it > 10 and np.allclose(h, best_h, rtol=convergence_tol):
            break
            
    # Final evaluation with exact metric
    final_overlaps = np.zeros(n_points)
    h_ext = np.concatenate([best_h, np.zeros(n_points)])
    h_rev = np.flip(best_h)
    conv = signal.fftconvolve(h_ext, h_rev, mode='full')[:n_points]
    for k in range(n_points):
        hshift = np.roll(best_h, k)
        final_overlaps[k] = np.dot(best_h, 1 - hshift) * dx
        
    final_c5 = np.max(final_overlaps)
    
    return (best_h.tolist(), final_c5, n_points)
