import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant from the step function h"""
    # Compute overlap integral for all lags
    conv = correlate(h, 1 - h, mode="full")
    # Scale by dx for continuous approximation
    conv *= dx
    # Return maximum overlap value
    return np.max(conv)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start time
    start_time = time.time()
    
    # Settings
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float).copy()
        # Ensure it satisfies constraint
        h_sum = np.sum(h) * dx
        h = h * (n_points / 2) / h_sum
    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05, 0.05, n_points) * dx
        h = np.clip(h, 0, 1)

    # Target value for sum
    target_sum = n_points / 2.0

    # Optimization function
    def objective(h_vec):
        # Reshape for easier handling
        h = h_vec.copy()
        # Rescale so that sum = target_sum
        current_sum = np.sum(h) * dx
        h *= target_sum / current_sum
        # Compute C5
        c5 = evaluate_c5(h, dx)
        return c5

    # Constraints
    def constraint_sum(h_vec):
        h = h_vec.copy()
        current_sum = np.sum(h) * dx
        return current_sum - target_sum

    # Bounds (0 to 1 for all entries)
    bounds = [(0.0, 1.0) for _ in range(n_points)]

    # Initial objective
    current_c5 = objective(h)
    best_c5 = current_c5
    best_h = h.copy()

    # Optimization loop
    max_iter = 1000
    iter_count = 0
    last_improvement = 0
    last_c5 = current_c5

    # Get constraint object
    con = {'type': 'eq', 'fun': constraint_sum}

    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        # Optimize with L-BFGS-B
        result = minimize(
            objective,
            h,
            method="L-BFGS-B",
            bounds=bounds,
            constraints=con,
            options={"maxiter": 10, "ftol": 1e-8},
        )
        
        if not result.success:
            break
        
        h = result.x
        # Ensure h is within [0,1]
        h = np.clip(h, 0, 1)
        # Rescale h so that sum = target_sum
        h_sum = np.sum(h) * dx
        h *= target_sum / h_sum
        # Evaluate
        new_c5 = objective(h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improvement = iter_count
        
        # Early stopping
        if abs(new_c5 - last_c5) < 1e-8:
            last_improvement += 1
        last_c5 = new_c5
        
        if last_improvement >= 50:
            break
        
        # Check time budget
        if (time.time() - start_time) > budget_s - 1:
            break

    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    # Ensure final h is valid
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    h_sum = np.sum(final_h) * dx
    final_h *= target_sum / h_sum

    return (final_h.tolist(), final_c5, n_points)
