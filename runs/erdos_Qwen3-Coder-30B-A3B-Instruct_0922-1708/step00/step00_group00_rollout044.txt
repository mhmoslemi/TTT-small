import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    pop_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    tournament_size = 5
    elite_size = pop_size // 5
    max_generations = 120
    
    # Generate initial population
    if 'initial_h_values' in globals():
        initial_pop = [np.array(initial_h_values)]
    else:
        initial_pop = []
        
    population = initial_pop.copy()
    
    # Generate diverse initial solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    def evaluate_fitness(h):
        # Compute overlaps
        overlaps = correlate(h, 1 - h, mode="full") * dx
        # Find maximum overlap (from shifts k = -n+1 to n-1)
        max_overlap = np.max(overlaps)
        return max_overlap
    
    def mutate(h):
        h_mut = h.copy()
        num_mutations = int(mutation_rate * len(h))
        indices = np.random.choice(len(h), num_mutations, replace=False)
        for i in indices:
            # Add small random perturbation
            delta = np.random.normal(0, 0.05)
            h_mut[i] += delta
        # Clip to [0, 1]
        h_mut = np.clip(h_mut, 0, 1)
        # Renormalize to sum to n_points/2
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return h_mut
    
    def crossover(p1, p2):
        if np.random.rand() > crossover_prob:
            return p1.copy(), p2.copy()
        crossover_point = np.random.randint(1, len(p1) - 1)
        child1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
        child2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
        # Renormalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        return child1, child2
    
    # Main GA loop
    generation = 0
    no_improvement_count = 0
    last_best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 0.1 and generation < max_generations:
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        min_fitness = fitnesses[min_fitness_idx]
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > 100:
            # Early stop
            break
            
        # Selection (tournament)
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            selected.append(population[winner_index].copy())
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        new_population = [population[i].copy() for i in elite_indices]
        
        # Generate offspring
        for i in range(elite_size, pop_size, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim population to exact size
        population = new_population[:pop_size]
        generation += 1
        
    if time.time() - start_time >= budget_s - 0.1:
        # Ensure we have a valid result
        if best_h is None:
            # Return last best found among the population
            fitnesses = [evaluate_fitness(h) for h in population]
            best_idx = np.argmin(fitnesses)
            best_h = population[best_idx].copy()
            best_c5 = fitnesses[best_idx]
    
    # Final check
    final_fitness = evaluate_fitness(best_h)
    return (best_h, final_fitness, n_points)
