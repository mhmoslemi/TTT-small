import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):

        n_points = len(h_init)
    except:
        # Fallback: uniform distribution with sum = n/2
        n_points = 45
        h_init = np.ones(n_points) * 0.5
    
    # Ensure h_init is properly normalized
    h = h_init.copy()
    h *= (n_points / 2.0) / np.sum(h)
    
    # Initialize optimization parameters
    dx = 2.0 / n_points
    max_iter = 20000
    alpha = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    start_time = time.time()
    iter_count = 0
    
    # Main optimization loop
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute all overlaps
        h_rev = np.flip(h)
        correlation = np.correlate(h, h_rev, mode='full')
        overlap_values = correlation * dx
        
        # Find maximum overlap and its shift
        max_idx = np.argmax(overlap_values)
        c5 = overlap_values[max_idx]
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient for the max overlap shift
        max_shift = max_idx - (n_points - 1)
        grad = np.zeros(n_points)
        for k in range(n_points):
            if 0 <= k + max_shift < n_points:
                grad[k] = h[k] - (1.0 - h[k + max_shift])
            elif 0 <= k < n_points and k + max_shift < 0:
                # Roll shift to the right
                grad[k] = h[k] - (1.0 - h[k + max_shift + n_points])
        
        # Update h
        h = h - alpha * grad
        
        # Project h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to keep sum(h) = n/2
        h *= (n_points / 2.0) / np.sum(h)
            
        iter_count += 1
        
        # Adaptive learning rate
        if i % patience == 0 and i > 0:
            if c5 >= best_c5:
                alpha *= 0.5
                
    # Final evaluation
    h_final = best_h
    h_rev = np.flip(h_final)
    correlation = np.correlate(h_final, h_rev, mode='full')
    overlap_values = correlation * dx
    final_c5 = np.max(overlap_values)
    
    return (h_final, final_c5, n_points)
