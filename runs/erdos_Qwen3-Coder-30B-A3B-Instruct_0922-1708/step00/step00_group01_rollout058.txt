import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        # Create symmetric plateau centered at 1.0 with tapering ends
        h = np.ones(n_points)
        half_width = n_points // 3
        for i in range(half_width):
            h[i] = 0.1 + 0.9 * (i / half_width)
            h[-1-i] = 0.1 + 0.9 * (i / half_width)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    # Discretization parameters
    dx = 2.0 / n_points
    
    # Optimization parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iterations = 3000
    convergence_tol = 1e-5
    
    # Precompute all lags for efficiency
    lags = np.arange(n_points)
    
    # Store best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps using correlation
        # We compute overlaps for all shifts k
        # overlap(k) = integral h(x) * (1 - h(x+k)) dx
        # We compute this efficiently using correlation
        
        # Compute the full correlation (or overlap values)
        # overlap[k] = sum h[i] * (1 - h[(i+k) % n_points]) * dx
        # This is equivalent to correlate(h, 1-h, mode="full")
        # But we're computing directly
        
        h_rev = 1 - h
        # For overlap(k), it's convolution of h with flipped h_rev
        # But we can compute this efficiently with FFT or direct computation
        overlaps = []
        for k in range(n_points):
            # overlap = sum h[i] * h_rev[(i+k) % n_points] * dx
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * h_rev[j]
            overlaps.append(overlap * dx)
        
        # Use log-sum-exp smooth approximation of max
        overlaps = np.array(overlaps)
        log_sum_exp = np.log(np.sum(np.exp(tau * overlaps)))
        c5_approx = log_sum_exp / tau
        
        # Check if this is the best solution so far
        if c5_approx < best_c5:
            best_c5 = c5_approx
            best_h = h.copy()
            
        # Compute gradient
        # d/dh[i] [log(sum(exp(tau * overlap[k])))]
        # = (sum_k exp(tau * overlap[k]) * d/dh[i] overlap[k]) / sum_k exp(tau * overlap[k])
        # d/dh[i] overlap[k] = -dx * h_rev[(i-k) % n_points]
        # Since correlation is symmetric in the discretized case:
        #   We compute gradient as: -dx * correlate(h_rev, one_hot_vector at position i with shift k)
        
        # Rephrase: we have overlap_k = sum h[i] * h_rev[(i+k) % n_points]
        # Thus d/overlap_k/dh[i] = h_rev[(i-k) % n_points] (when k corresponds to shift)
        
        # But we build the exact gradient for the full sum using d/dh[i] log(sum exp(tau*overlap))
        # Controls the gradient based on the shape with d/dh[i] exp(tau * overlap[k]) = tau * exp(tau * overlap[k]) * d_overlap[k]/d_h[i]
        # That is, 1/sum exp(tau * overlap[k]) * sum_k tau * exp(tau * overlap[k]) * d_overlap[k]/d_h[i]
        
        # Actually for each k, we have overlap[k] = \sum_i h[i] * h_rev[(i+k)%n]
        # d_overlap[k]/d_h[i] = h_rev[(i-k)%n]
        # d/dh[i] log(sum exp(tau*overlap[k])) = 1/Z * sum_k tau * exp(tau*overlap[k]) * h_rev[(i-k)%n]
        # = sum_k exp(tau * overlap[k]) * h_rev[(i-k)%n] / sum_k exp(tau * overlap[k]) 
        # = sum_k p_k * h_rev[(i-k)%n] where  p_k = exp(tau * overlap[k]) / Z
        
        # But we actually can do that more directly!
        # Since we precomputed overlaps, we can do:
        exp_tau_overlaps = np.exp(tau * overlaps)
        Z = np.sum(exp_tau_overlaps)
        weights = exp_tau_overlaps / Z
        
        # Now compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(n_points):
                j = (i - k) % n_points
                grad[i] += weights[k] * h_rev[j]
        
        # Apply gradient step
        h = h - learning_rate * grad
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust h to enforce constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Anneal temperature
        tau = max(1e-6, tau * anneal_factor)
        
        # Check convergence (not critical but good for early exit)
        if iteration > 100 and np.allclose(h, h.copy(), atol=convergence_tol):
            break
    
    # Evaluate exact C5 at best solution for final result
    # To compute exact C5 we need overlap values for all lags
    overlaps_final = []
    h_rev = 1 - best_h
    for k in range(n_points):
        overlap = 0.0
        for i in range(n_points):
            j = (i + k) % n_points
            overlap += best_h[i] * h_rev[j]
        overlaps_final.append(overlap * dx)

    # Final C5 is the max of all overlaps
    exact_c5 = np.max(overlaps_final)
    
    return (best_h.tolist(), exact_c5, n_points)
