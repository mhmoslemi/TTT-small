import numpy as np
from time import time
import math

def run(seed=42, budget_s=600, **kwargs):
    # Set random seed
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    timeout = time() + budget_s
    
    # Initialize h from initial_h_values if available, else create a valid starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)  # Force sum to be n_points / 2
    
    # Constraint: sum(h) == n_points / 2, and 0 <= h[i] <= 1
    def project_h(h):
        h = np.clip(h, 0, 1)
        # Rescale to ensure sum constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h

    # Fast evaluation for C5
    def evaluate_C5(h):
        # compute overlap with 1-h (this is the convolution)
        corr = np.correlate(h, 1 - h, mode="full")
        # shift corr back to get the appropriate indices
        max_k = len(corr) - 1
        mid = max_k // 2
        # only consider the non-trivial overlaps:
        overlaps = corr[mid:mid+n_points]
        return np.max(overlaps) * dx

    # Start with current best
    h_best = h.copy()
    c5_best = evaluate_C5(h_best)
    
    # Control variables for adaptive perturbation
    sigma = 0.1
    decay = 0.99
    
    # Iteration counter
    iteration = 0
    no_improvement = 0

    # Optimization loop
    while iteration < max_iter and time() < timeout:
        # Generate perturbed h
        h_new = h_best + np.random.normal(0, sigma, n_points)
        h_new = project_h(h_new)
        
        # Evaluate the new solution
        c5_new = evaluate_C5(h_new)
        
        # Accept the new solution if it's better
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Adaptive perturbation
        sigma *= decay
        iteration += 1
        
        # Terminate if no improvement for long
        if no_improvement > 1000:
            break

    # Final validation
    h_best = project_h(h_best)
    c5_final = evaluate_C5(h_best)
    
    # Round to expected precision to fulfill contract
    c5_final = round(c5_final, 5)
    
    return (h_best.tolist(), c5_final, n_points)
