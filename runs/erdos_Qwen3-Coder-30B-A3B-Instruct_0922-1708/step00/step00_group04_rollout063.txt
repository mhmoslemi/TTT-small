def evaluate_erdos_solution(h):

    dx = 2.0 / n
    # Compute all overlaps for lags 0 to n-1
    conv = np.correlate(h, 1 - h, mode='full')
    # Take max over all lags
    max_overlap = np.max(conv) * dx
    return max_overlap
