import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Maximum over all shifts
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, crossover_prob=0.7):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.1):
    """Mutate a solution by adding noise."""
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    noise = np.random.normal(0, 0.05, n)
    h = h + noise * mask
    return h

def genetic_algorithm(initial_h, n_points, n_generations=1000, pop_size=50, 
                       time_limit=420):
    """Perform genetic algorithm search."""
    start_time = time.time()
    population = []
    # Initialize population
    for _ in range(pop_size):
        h = initial_h + np.random.normal(0, 0.01, n_points)
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h)
    
    for generation in range(n_generations):
        if time.time() - start_time > time_limit:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Elitism: keep top 20%
        elite_size = pop_size // 5
        elite = population[:elite_size]
        
        # Generate new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Select parents (tournament selection)
            parent1 = population[np.random.randint(0, pop_size // 2)]
            parent2 = population[np.random.randint(0, pop_size // 2)]
            
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            child1 = project_h(child1, n_points)
            child2 = project_h(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
    
    return best_h, best_c5

def gradient_descent(initial_h, n_points, time_limit=180):
    """Perform gradient descent exploitation phase."""
    start_time = time.time()
    h = initial_h.copy()
    
    # Precompute reverse of h for faster convolution
    rev_h = 1 - h[::-1]
    
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    
    learning_rate = 0.01
    no_improve_count = 0
    max_no_improve = 100
    
    for iteration in range(10000):
        if time.time() - start_time > time_limit:
            break
            
        # Compute correlation to find worst shift
        corr = correlate(h, rev_h, mode='full')
        shift_idx = np.argmax(corr)
        # The actual shift is shift_idx - (len(h) - 1), i.e., in [-len(h)+1, len(h)-1]
        shift = shift_idx - (len(h) - 1)
        
        # Compute gradient of correlation w.r.t h, at worst shift
        # We only compute gradient for the shift that gives maximum overlap
        # For shift k, Cross-correlation contribution = sum over i of h[i] * (1 - h[i+k])
        # For the particular worst shift:
        grad = np.zeros(n_points)
        
        # This computation is complex for cross-correlation, so we simplify:
        # Approximate gradient using finite differences
        eps = 1e-5
        base_c5 = evaluate_c5(h)
        
        temp_h = h.copy()
        best_grad = 0
        best_idx = 0
        
        for i in range(n_points):
            temp_h = h.copy()
            temp_h[i] += eps
            temp_h = project_h(temp_h, n_points)
            
            new_c5 = evaluate_c5(temp_h)
            gradient_at_i = (new_c5 - base_c5) / eps
            
            if gradient_at_i < best_grad:
                best_grad = gradient_at_i
                best_idx = i
        
        # Descend in direction of steepest descent
        if abs(best_grad) > 1e-10:
            try:
                new_h = h - learning_rate * best_grad
                new_h = project_h(new_h, n_points)
                new_c5 = evaluate_c5(new_h)
                
                if new_c5 < best_c5:
                    h = new_h
                    best_c5 = new_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
                    if no_improve_count > max_no_improve:
                        learning_rate *= 0.5
                        no_improve_count = 0
            except:
                pass
    
    return h, best_c5
