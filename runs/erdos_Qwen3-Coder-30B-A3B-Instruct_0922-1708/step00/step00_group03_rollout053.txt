import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h"""
    conv = correlate(h, 1 - h, mode='full')
    # Only consider the overlapping parts (not the full correlation)
    # The valid range of lags is from -n+1 to n-1, 
    # but we only consider lags from 0 to n-1 for the overlap integral.
    # The effective max lag is 2*n-2, but we only care about the first n elements
    # which correspond to lags 0 ... n-1
    # In the full correlation output, the first n-1 elements are negative lags, 
    # and the n elements from index n-1 onwards are positive lags up to n-1.
    # The last n-1 elements specify lags from 1 to n-1, so the first element
    # corresponds to lag n-1, and the last one corresponds to lag 0.
    # we need lags from 0 to n-1, that is elements from index n-1 to 2n-3.
    # number of lags we actually care about is n elements (from lag 0 to n-1)
    # the first element after n-1 is lag 0, and we need n elements of lags 0 to n-1  
    # i.e., indices n-1 to 2*n-3 (inclusive) = 2*n-3 - (n-1) + 1 = n elements
    # directly compute the integral of convolution over overlapping domain
    # At lag k, we consider the integral over (0, 2-k) which is an interval of length 2-k
    # The number of overlapping intervals for lags k=0..n-1
    # The convolution at lag k includes the contribution of h(x)*h(x+k) while h(x+k) is
    # in [0,2]. This gives a contribution of \int_{max(0, k)}^{min(2, 2+k)} h(x)*h(x+k) dx
    # An alternative simpler approach: 
    # for python correlate r[k] = sum_{x=0}^{N-1} h[x]* (1-h[x+k]) for lags k that 
    # correspond to valid pairs of values: h[x] is valid and h[x+k] is valid 
    # So the max overlap max_k r[k] * dx = max_k integral_{max(0,k)}^{min(2,2+k)} h[x] * (1-h[x+k]) dx
    # Could compute max_k on correlation array - but more natural mathematically to work on 
    # the full correlation and then mask.
    # index 0 of full convolution has lag -n+1, up to 2*n-3 = n-1+n-1
    # So indices n-1 through 2*n-3 correspond to lags 0 to n-1 (or the other way)
    # The full convolution mirrors the kind of symmetric lags as in full cross-correlation
    # of h with (1-h) - 
    return np.max(conv[n-1:2*n-2]) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # define the problem
    n = 70  # standard resolution
    dx = 2.0 / n
    
    # initialization
    global initial_h_values
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = initial_h_values.copy()
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Sum constraint violated"
    else:
        h = np.full(n, 0.5 * dx)  # uniform start
        h += np.random.uniform(-0.05, 0.05, size=n) * dx
        h = np.clip(h, 0, 1)

    # re-scale h to sum to 1/dx
    target_sum = n / 2.0 
    current_sum = np.sum(h)
    h *= (target_sum / current_sum) 

    # Pre-calculate convolution indices and shift
    full_conv_indices = np.arange(2*n - 1) - (n - 1)

    # Define surrogate objective (smooth approximation to max)
    def surrogate_obj(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Convolve h with (1-h) and select correct lags
        h = h_vec
        h_minus_1 = 1 - h_vec
        corr = correlate(h, h_minus_1, mode='full')  # full correlation
        # extract lags from 0 to n-1 
        # first index of corr is lag -n+1
        # we want lag from 0 to n-1: so from indices n-1 to 2n-3
        # corr indices n-1 through 2n-2 -> lags 0 to n-1
        # so we consider elements corr[n-1 : 2*n-1]
        corr_vals = corr[n-1 : 2*n-1]
        # We now want to optimize a convex smooth surrogate
        # One option is to use the log-sum-exp with a parameter alpha to make it smooth
        # Let C_k = corr_vals[k] for k=0...n-1. 
        # Return -log(sum(exp(alpha * C_k) for k)) which minimizes the max
        alpha = 0.1
        # Guarantees that exp(alpha * C_k) is well behaved for even large C_k
        # with alpha=0.1, if max_C_k is around order 100, then max(alpha * C_k) ~ 10
        # exp(10) ~ 22000, not too big
        return -np.log(np.sum(np.exp(alpha * corr_vals)))
    
    def surrogate_grad(h_vec):
        # Formal derivative of log-sum-exp
        # \frac{d}{dh} \log(\sum_k \exp(alpha * C_k)) where C_k = corr_k(h) 
        # So we need to compute the gradient of C_k with respect to h_i for all k
        # The gradient is (exp(alpha*C_k) * dC_k/dh_i) / (sum_k exp(alpha*C_k))
        # Use the fact that for C_k = sum_x h[x] * (1 - h[x+k]) with h periodicity
        # dC_k / dh_j = (1-h[j+k]) if x=j and x+k = x+j, i.e. j = x, and same x+k = x+j
        # i.e. when h_j is part of that term, ... not straightforward 
        # better to compute it numerically or symbolically
        h = np.clip(h_vec, 0, 1)
        # compute full correlation matrix of h with 1-h
        h_minus_1 = 1 - h
        corr = correlate(h, h_minus_1, mode='full')
        corr_vals = corr[n-1 : 2*n-1]
        
        # For surrogate obj = -log(sum(exp(alpha * corr_vals)))
        # Gradient w.r.t. h:  - (exp(alpha * corr_vals) * alpha) / sum(exp(alpha * corr_vals)) * dr/dh
        alpha = 0.1
        exp_term = np.exp(alpha * corr_vals)
        sum_exp = np.sum(exp_term)
        # Extract gradient with respect to h
        # For each term exp(alpha * corr_k), we need d/dh -> we need the gradient of corr_k wrt h
        # The correlation is: r_k = sum_{x=0}^{n-1} h[x] * (1 - h[x+k])
        # d r_k / d h_i = (1 - h[i+k]) if x=i and x+k = i -> 1 if k=0, 0 otherwise
        # More generally: i. kron +/- k = x => d r_k / d h_i = (1 - h[x]) 
        # Actually no. need mod arithmetics carefully: r_k = sum_{x} h[x] * (1-h[x+k]) 
        # d r_k / d h_j = sum_x (delta_{x, j} * (1 - h[x+k]) ) = 1-h[j+k] (returned at 0)
        # for local x+k>=n we rotate x+k as (x+k)%n (since arrays are periodic)
        # gradient of r_k w.r.t. h_j -> nonzero case when x=j, that is when x+k=j => x = j -k
        # So the d r_k / d h_j = 1-h[j-k] if j>=k else 1-h[(j-k)%n]
        # The permuted version is a convolution of 1-h with identity
        one_minus_h = 1 - h
        # compute gradient of r_k w.r.t. h, for k=0..n-1
        # i.e., for each k, grad_k = [d r_k / d h_0, ..., d r_k / d h_{n-1}]
        # grad_k[j] = 1 - h[(j-k)%n] if k <= j else 1 - h[(j-k)%n] 
        # The above answers both non-periodic and periodic extension cases correctly
        weight_matrix = np.zeros((n, n))
        for i in range(n):
            for k in range(n):
                j = (i - k) % n  # which position in h contributes to the partial derivative of the kth element of corr
                weight_matrix[k, i] = one_minus_h[j]
        # The derivative w.r.t. the full h: 
        # We want grad = sum_k ( exp(alpha*corr[k]) * alpha / sum_k exp(alpha*corr_k) ) * weight_matrix[k,:] 
        #=> for each i, sum_k (exp(alpha*corr_k) * alpha / sum_k exp(alpha*corr_k) ) * weight_matrix[k,i]
        #=> (alpha / sum_k exp(alpha*corr_k)) * sum_k ( exp(alpha*corr_k) * weight_matrix[k,i] )
        # But weight_matrix is not very useful dimensionally, should be correct for indexing equiv constraints
        # Let's approach differently: be more exact in computation
        # For low-level derivative of correlation part:
        # For h_a with varied = j:
        # We form 1-h, and compute corr_k = sum x h[x] (1-h[x+k]) 
        # d corr_k / d h_j = (1-h[j+k]) 
        # only 1 non-zero value in the gradient at position j for that particular corr_k
        # for k fixed:
        # gradient is (1-h[j+k]) but j could run from 0 to n-1 but we only include j + k from 0 to n-1
        # = 1 - h[(j - k) % n] if j-k >= 0 else 1 - h[(j-k)%n]
        # Or better: 
        # For k from 0 to n-1, d corr_k / d h_i = (1 - h[(i+k)%n])
        # So the total gradient is: - alpha * sum_k exp(alpha * corr_k) * (1-h[(i+k)%n]) / sum_k exp(alpha * corr_k)
        # for h_i, where i is a fixed index
        exp_corr = np.exp(alpha * corr_vals)
        grad = np.zeros(n)
        for i in range(n):
            for k in range(n):
                j = (i + k) % n
                grad[i] += exp_corr[k] * (1 - h[j])
        grad *= - alpha / sum_exp
        return grad
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    start_time = time.time()
    
    # optimizer loop
    iteration = 0
    last_improvement = 0
    tol = 1e-5
    
    # Method: Limited-memory BFGS with tight control
    opt_result = minimize(
        fun=surrogate_obj,
        x0=h,
        method='L-BFGS-B',
        jac=surrogate_grad,
        bounds=[(0, 1)] * n,
        options={'maxiter': 500, 'ftol': 1e-9, 'gtol': 1e-6},
        callback=lambda x: None  # place holder to just monitor
    )
    
    h_opt = opt_result.x
    h_opt = np.clip(h_opt, 0, 1)
    # Rescale
    current_sum = np.sum(h_opt)
    h_opt *= (target_sum / current_sum)
    c5_opt = evaluate_c5(h_opt, dx)
    
    return (h_opt, c5_opt, n)
