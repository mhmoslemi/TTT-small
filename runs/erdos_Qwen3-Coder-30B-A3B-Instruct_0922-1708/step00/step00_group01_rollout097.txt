def evaluate_c5(h_values, dx):
    h = np.array(h_values)
    # Compute all correlations as with signal.correlate for symmetry
    h_rev = np.flip(h)
    correlations = np.correlate(h, 1 - h_rev, mode='full')
    # Take valid cross-correlations
    overlaps = correlations[len(h)-1:] * dx  # Only forward lags
    return np.max(overlaps)
