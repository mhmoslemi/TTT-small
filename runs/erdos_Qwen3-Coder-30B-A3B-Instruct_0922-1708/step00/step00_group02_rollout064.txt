import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation for all shifts
    correlation = np.correlate(h, 1 - h, mode="full")
    # The max is taken over all possible overlaps
    c5 = np.max(correlation) * dx
    return c5

def project_h(h, n_points):
    """Project h to ensure it's in [0,1] and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    if np.sum(h) != n_points / 2:
        h = h * (n_points / 2) / np.sum(h)
    return h

def init_h(n_points):
    """Initialize h using the provided initial_h_values or fallback to a sinusoidal."""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Rescale if needed
            h = np.interp(np.linspace(0, 1, n_points), 
                          np.linspace(0, 1, len(h)), h)
    else:
        # Fallback to a sinusoidal shape
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 2.0))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(parent1, parent2):
    """Perform single-point crossover between two individuals."""
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    # Pick a random crossover point
    cx_point = np.random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    return child1, child2

def mutate(h, mutation_rate):
    """Mutate the individual with given probability."""
    n = len(h)
    for i in range(n):
        if np.random.random() < mutation_rate:
            # Add small random noise
            h[i] += np.random.normal(0, 0.1)
    return h

def ga_phase(h_init, n_points, time_budget):
    """Genetic algorithm phase to explore solution space."""
    pop_size = 50
    mutation_rate = 0.1
    crossover_rate = 0.7
    generations = 500  # Limited based on time budget
    
    # Initialize population
    population = [h_init.copy() for _ in range(pop_size)]
    for i in range(pop_size):
        population[i] = project_h(population[i], n_points)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    elapsed_time = 0
    start_time = time.time()
    
    for gen in range(generations):
        if elapsed_time > time_budget * 0.8:
            break
        # Evaluate fitness
        fitness = [evaluate_erdos_solution(p) for p in population]
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Check for better solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Elitism: keep top 20%
        new_population = population[:pop_size // 5]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents using tournament selection
            idx1, idx2 = np.random.choice(pop_size // 2, 2, replace=False)
            p1, p2 = population[idx1], population[idx2]
            
            # Crossover
            if np.random.random() < crossover_rate:
                c1, c2 = crossover(p1, p2)
            else:
                c1, c2 = p1.copy(), p2.copy()
                
            # Mutate
            c1 = mutate(c1, mutation_rate)
            c2 = mutate(c2, mutation_rate)
            
            # Project to feasible region
            c1 = project_h(c1, n_points)
            c2 = project_h(c2, n_points)
            
            new_population.append(c1)
            if len(new_population) < pop_size:
                new_population.append(c2)
                
        population = new_population
        elapsed_time = time.time() - start_time
    
    return best_h, best_c5

def gd_phase(h_init, n_points, time_budget):
    """Gradient descent phase to fine-tune."""
    h = h_init.copy()
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    start_time = time.time()
    
    # Simple gradient descent iterations

    lr = 0.01
    iterations = 1000
    no_improvement = 0

        # Instead of full optimization, we'll do small steps with a fixed policy
        # For this we'll just randomly perturb the function but make sure we

        h_new = h + np.random.normal(0, 0.001, size=len(h))
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_erdos_solution(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        
        if i % 100 == 0:

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved bound for C5."""
    np.random.seed(seed)
    n_points = 45  # Start with smaller size
    h = init_h(n_points)
    elapsed_time = 0
    start_time = time.time()
    
    # GA phase
    h_ga, c5_ga = ga_phase(h, n_points, budget_s)
    best_h = h_ga
    best_c5 = c5_ga
    
    # GD phase
    h_gd, c5_gd = gd_phase(h_ga, n_points, budget_s)
    
    if c5_gd < best_c5:
        best_h = h_gd
        best_c5 = c5_gd
    
    return (best_h, best_c5, n_points)
