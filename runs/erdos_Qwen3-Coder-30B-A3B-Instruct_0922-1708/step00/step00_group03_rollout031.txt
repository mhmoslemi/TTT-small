import cvxpy as cvx
from time import time

    target_sum = n_points / 2

    if 'initial_h_values' in globals():

    else:

    h_init = h_init * target_sum / np.sum(h_init)
    
    # Optimization with SDP relaxation using CVXPY
    x = cvx.Variable(n_points)
    z = cvx.Variable((n_points, n_points))
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == target_sum,
    ]
    
    # Add z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                z[i,j] <= x[i],
                z[i,j] <= 1 - x[j],
                z[i,j] >= x[i] + x[j] - 1
            ]
    
    # Objective setup: Maximize the sum over all shifts
    C_values = []
    for k in range(-(n_points-1), n_points):
        # Compute corresponding indices for a shift of k
        i_indices = np.arange(n_points)
        j_indices = i_indices - k
        valid = (j_indices >= 0) & (j_indices < n_points)
        valid_i = i_indices[valid]
        valid_j = j_indices[valid]
        sum_z = cvx.sum(z[valid_i, valid_j])
        # Use cvx scalar to sum
        C_values.append(sum_z)
    
    # Our goal is to minimize the maximum value of all C_values
    # So we use a variable t that must be greater than or equal to all C_values
    t = cvx.Variable()
    for c_val in C_values:
        constraints.append(t >= c_val)
    
    objective = cvx.Minimize(t)
    prob = cvx.Problem(objective, constraints)
    
    # Solve with timeout
    try:
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
    except:

            prob.solve(solver=cvx.SCS, verbose=False, max_iters=10000)
        except:
