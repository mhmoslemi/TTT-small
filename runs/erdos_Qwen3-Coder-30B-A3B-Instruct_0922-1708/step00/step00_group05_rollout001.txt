import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    """Evaluate the C_5 bound for a given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations with 1-h
    corr = correlate(h, 1 - h, mode='full')
    # Take max over all lags (this is the C_5 value)
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from initial_h_values if available

        n_points = 70
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.4
        h0[1] = 0.6
    
    n_points = len(h0)
    dx = 2.0 / n_points
    
    # Ensure sum(h) = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Ensure bounds
    h0 = np.clip(h0, 0, 1)
    
    # Compute initial value
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h)
    
    # Some constants
    perturb_fraction = 0.1
    n_perturb = max(1, int(perturb_fraction * n_points))
    delta = 0.05
    smoothing_window = 3
    max_no_improvement = 100
    no_improvement_count = 0
    
    start_time = time.time()
    iter_count = 0
    
    while time.time() - start_time < budget_s - 0.1 and no_improvement_count < max_no_improvement:
        iter_count += 1
        improved = False
        
        for trial in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Create a perturbed copy
            hp = best_h.copy()
            
            # Randomly select indices to perturb
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            for i in indices:
                # Add small random change
                change = np.random.uniform(-delta, delta)
                hp[i] += change
                
            # Ensure valid bounds
            hp = np.clip(hp, 0, 1)
            
            # Maintain sum constraint
            old_sum = np.sum(best_h)
            new_sum = np.sum(hp)
            adjustment = (old_sum - new_sum) / n_points
            hp += adjustment  # Distribute the adjustment
            hp = np.clip(hp, 0, 1)
            
            # Smooth the function to maintain some structure
            if smoothing_window > 0:
                # Moving average smoothing
                smoothed = np.zeros_like(hp)
                for i in range(n_points):
                    start_idx = max(0, i - smoothing_window)
                    end_idx = min(n_points, i + smoothing_window + 1)
                    smoothed[i] = np.mean(hp[start_idx:end_idx])
                hp = smoothed
            
            # Evaluate
            new_c5 = evaluate_h(hp)
            
            if new_c5 < best_c5:
                best_h = hp.copy()
                best_c5 = new_c5
                improved = True
                no_improvement_count = 0
            elif not improved:
                no_improvement_count += 1
            else:
                no_improvement_count = 0
                
    return (best_h.tolist(), best_c5, n_points)
