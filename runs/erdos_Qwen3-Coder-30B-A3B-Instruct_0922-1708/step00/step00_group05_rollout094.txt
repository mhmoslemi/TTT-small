import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given h values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h_values, 1 - np.array(h_values), mode='full')
    # Take max over all lags
    max_corr = np.max(corr)
    return max_corr * dx

    best_h = h.copy()

    velocities = []

    for i in range(n_particles):

        particles.append(particle_h.copy())

            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)

            particles[i] += velocities[i]
