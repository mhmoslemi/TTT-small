import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial solution if available
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except:
        # Default construction if no initial_h_values
        n_points = 70
        h_init = np.ones(n_points) * 0.5
    
    # Normalize if sum is not correct
    dx = 2.0 / n_points
    h = np.array(h_init, dtype=float)
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        h = h * (n_points/2) / h_sum
    
    # Initialize
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        iter_count += 1
        improved = False
        
        # Try multiple perturbations
        for _ in range(100):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Create perturbed version
            h_new = best_h.copy()
            
            # Select random points to perturb
            num_perturb = max(1, n_points // 20)
            indices = np.random.choice(n_points, num_perturb, replace=False)
            
            # Apply small random changes
            for i in indices:
                delta = np.random.uniform(-0.05, 0.05)
                h_new[i] += delta
                
            # Enforce bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain sum constraint
            current_sum = np.sum(h_new)
            target_sum = n_points / 2
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # Distribute the change uniformly
                h_new += diff / n_points
                
                # Re-clip
                h_new = np.clip(h_new, 0, 1)
                
            # Smooth the function
            window_size = min(5, n_points // 10 + 1)
            if window_size > 1:
                h_smooth = np.convolve(h_new, np.ones(window_size)/window_size, mode='same')
                h_new = h_smooth
            
            # Evaluate new solution
            c5_new = evaluate_erdos_solution(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                improved = True
                no_improvement_count = 0
                break
                
        if not improved:
            no_improvement_count += 1
    
    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (list(best_h), final_c5, n_points)

def evaluate_erdos_solution(h_values):
    # Convert to numpy array
    h = np.array(h_values)
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute 1-h
    one_minus_h = 1.0 - h
    
    # Full correlation
    corr = correlate(h, one_minus_h, mode='full')
    
    # The maximum overlap
    max_overlap = np.max(corr)
    
    # Apply dx scaling
    c5 = max_overlap * dx
    
    return c5
