import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Use the provided initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Ensure sum constraint is satisfied
        h *= target_sum / np.sum(h)
    else:
        # Create random feasible h
        h = np.random.rand(n_points)
        h *= target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Frank-Wolfe optimization
    learning_rate = 0.1
    max_iterations = 500
    tolerance = 1e-5
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all lags
        # The overlap integral is: ∫ h(x)(1 - h(x+k)) dx
        # This corresponds to correlation of h with (1-h) 
        h_rev = 1 - h
        corr = correlate(h, h_rev, mode='full') * dx
        
        # The valid lags for auto-correlation
        # Indices go from -n+1 to n-1
        lags = np.arange(-n_points + 1, n_points)
        # Find the maximum overlap and corresponding lag
        max_idx = np.argmax(corr)
        k_max = lags[max_idx]
        
        # Compute gradient of overlap w.r.t h for the maximizing lag
        # For lag k_max, the overlap is ∑ h[i] * (1 - h[i+k_max])
        # Gradient w.r.t h[i] is (1 - h[i+k_max]) for i < n - |k_max|
        grad = np.zeros(n_points)
        
        # Shifted h_rev
        # For h_rev[i], we take h_rev[i+k_max] which is h_rev[ik]
        # We need to account for how values are aligned
        if k_max >= 0:
            # Forward shift: h[i] * (1 - h[i+k_max])
            # derivative w.r.t h[i] is 1 - h[i+k_max]
            for i in range(n_points - k_max):
                grad[i] += 1 - h[i + k_max]
        else:
            # Backward shift: h[i] * (1 - h[i+k_max])
            # derivative w.r.t h[i] is 1 - h[i+k_max] 
            for i in range(-k_max, n_points):
                grad[i] += 1 - h[i + k_max]
        
        # Solve linear subproblem: minimize grad^T * d subject to 
        # 0 <= d[i] <= 1, sum d[i] = target_sum
        # This is a simple LP with box constraints and a sum constraint  
        # Using coordinate descent on the reduced space (degenerate case with sum constraint)
        # In practice, this should be solved more efficiently, but for simplicity:
        # The solution is half of the negative gradient clipped to [0,1], 
        # then normalized to desired sum.
        d = -grad
        d = np.clip(d, 0, 1)
        # Adjust to match target sum    
        d_sum = np.sum(d)
        if d_sum > 0:
            d *= target_sum / d_sum
        else:
            d = np.full(n_points, target_sum / n_points)
        
        # Update h using Frank-Wolfe rule
        h = (1 - learning_rate) * h + learning_rate * d
        
        # Maintain bounds and sum constraint
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h *= target_sum / h_sum
        else:
            h.fill(target_sum / n_points)
            
        # Evaluate new C5
        current_c5 = evaluate_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral for a given function h"""
    h_rev = 1 - np.array(h)
    corr = correlate(h, h_rev, mode='full') * dx
    # Extract valid correlation indices for lags -n+1 to n-1
    lags = np.arange(-len(h)+1, len(h))
    max_overlap = np.max(corr)
    return max_overlap
