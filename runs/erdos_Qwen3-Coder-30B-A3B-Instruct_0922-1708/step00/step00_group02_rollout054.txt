import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to 1
    h = h * n_points / (2 * np.sum(h))
    
    dx = 2.0 / n_points
    
    def compute_C5(h):
        # Compute correlation of h with (1-h)
        corr = np.correlate(h, 1 - h, mode='full')
        # The maximum value is the C5 bound
        return np.max(corr) * dx
    
    def objective(h):
        return compute_C5(h)
    
    def project(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to correct integral
        h = h * n_points / (2 * np.sum(h))
        return h
    
    # Initial evaluation
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradient descent with line search
    iter_count = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        
        # Simple gradient approximation (finite differences)
        eps = 1e-6
        grad = np.zeros_like(h)
        base_val = compute_C5(h)
        
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            h_eps = project(h_eps)
            val_eps = compute_C5(h_eps)
            grad[i] = (val_eps - base_val) / eps
        
        # Apply gradient step
        h_new = h - lr * grad
        h_new = project(h_new)
        
        new_C5 = compute_C5(h_new)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            h = h_new.copy()
        else:
            # Reduce learning rate if no improvement
            lr *= 0.95
            
        # Occasionally reinitialize with slightly perturbed best
        if iter_count % 100 == 0 and iter_count > 100:
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = project(h)
            
        if iter_count % 1000 == 0:
            # Early stopping if solution is improving slowly
            if iter_count > 10000:
                break
    
    return (best_h.tolist(), best_C5, n_points)
