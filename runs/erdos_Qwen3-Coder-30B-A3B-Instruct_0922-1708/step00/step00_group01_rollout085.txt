```python
import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    # Cross-correlation h * (1 - h) with full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # Take the maximum value and scale by dx
    return np.max(corr) * dx

def project_h(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def initialize_h(n_points):
    """Initialize h from the parent construction or use a symmetric plateau"""
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to correct length
            x_old = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_old, h)
    else:
        # Create a symmetric plateau centered around 1.0, tapering to 0.1
        h = np.ones(n_points)
        half = n_points // 2
        # Tapering on both sides
        taper_len = min(half, n_points // 4)
        h[:taper_len] = np.linspace(1.0, 0.1, taper_len)
        h[-taper_len:] = np.linspace(0.1, 1.0, taper_len)
    
    h = project_h(h, n_points, 2.0 / n_points)
    return h

def mutate_h(h, noise_scale=0.05):
    """Apply Gaussian noise mutation"""
    h = h + np.random.normal(0, noise_scale, len(h))
    return project_h(h, len(h), 2.0 / len(h))

def crossover_h(parent1, parent2, crossover_rate=0.8):
    """Perform segment-wise averaging crossover"""
    if np.random.rand() > crossover_rate:
        return parent1.copy()
    
    # Simple average of segments
    n = len(parent1)
    split = np.random.randint(1, n-1)
    offspring = np.concatenate([parent1[:split], parent2[split:]])
    return project_h(offspring, n, 2.0 / n)

def nelder_mead_refine(h, dx, maxiter=100):
    """Refine h using Nelder-Mead"""
    def objective(h_flat):
        h = h_flat.reshape(-1)
        return evaluate_c5(h, dx)
    
    # Flatten the input
    h_flat = h.flatten()
    
    # Use Nelder-Mead to minimize the C5 value
    result = minimize(objective, h_flat, method='Nelder-Mead', 
                     options={'maxiter': maxiter, 'adaptive': True})
    
    if result.success:
        refined_h = result.x.reshape(h.shape)
        return project_h(refined_h, len(h), dx)
    else:
        return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 47
    dx = 2.0 / n_points
    max_generations = 1000
    pop_size = 100
    tournament_size = 20
    simplex_size = 10
    mutation_rate = 0.1
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = initialize_h(n_points)
        population.append(h)
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    # Main optimization loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all candidates
        fitness = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        current_best_c5 = fitness[sorted_indices[0]]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[sorted_indices[0]].copy()
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(sorted_indices[:tournament_size], 
                                          size=2, replace=False)
            winner = tournament[0] if fitness[tournament[0]] < fitness[tournament[1]] else tournament[1]
            selected.append(winner)
        
        # Create new population
        new_population = []
        for i in range(0, pop_size, 2):
            parent1_idx = selected[i]
            parent2_idx = selected[(i + 1) % pop_size]
            
            child1 = crossover_h(population[parent1_idx], population[parent2_idx])
            child2 = crossover_h(population[parent1_idx], population[parent2_idx])
            
            if np.random.rand() < mutation_rate:
                child1 = mutate_h(child1)
            if np.random.rand() < mutation_rate:
                child2 = mutate_h(child2)
                
            new_population.extend([child1, child2])
        
        # Keep top of population only if needed
        if len(new_population) > pop_size:
            new_fitness = [evaluate_c5(h, dx) for h in new_population]
            top_indices = np.argsort(new_fitness)[:pop_size]
            population = [new_population[i] for i in top_indices]
        else:
            population = new_population
            
        # Apply local search on top solutions
        if gen % 10 == 0:  # Every 10 generations
            top_candidates = [population[i] for i in sorted_indices[:simplex_size]]
            for i, h in enumerate(top_candidates):
                refined_h = nelder_mead_refine(h, dx) 
                idx = sorted_indices[i]
                population[idx] = refined_h
        
    # Final check against the best solution
    if best_h is None:
        # If no progress, use initial solution
        best_h = initialize_h(n_points)
        best_c5 = evaluate_c5(best_h, dx)
    
    # Final refinement step
    final_h = nelder_mead_refine(best_h, dx)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
```
