import numpy as np
from scipy.signal import correlate
import time

def evaluate_h(h, n_points):
    """Evaluate C_5 for given h"""
    dx = 2.0 / n_points
    # Compute cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # The maximum happens at the center of the full correlation
    max_shift = len(corr) // 2
    max_val = np.max(corr)
    return max_val * dx

def project_h(h, n_points):
    """Project h to feasible region"""
    h = np.clip(h, 0, 1)
    # Rescale to enforce sum(h) = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_descent_phase(h, n_points, budget_s, start_time):
    """Gradient descent to refine solution"""
    dx = 2.0 / n_points
    max_iter = 10000
    lr = 0.01
    patience = 1000
    best_h = h.copy()
    best_C5 = evaluate_h(h, n_points)
    no_improvement = 0
    iter_count = 0
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s * 0.9:
        # Compute correlation matrix
        corr = correlate(h, 1 - h, mode="full")
        max_shift = len(corr) // 2
        worst_shift_idx = np.argmax(corr)
        worst_shift = worst_shift_idx - max_shift
        
        # Gradient approximation using finite differences
        eps = 1e-6
        g = np.zeros_like(h)
        for i in range(n_points):
            # Perturbation
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            C5_plus = evaluate_h(h_plus, n_points)
            g[i] = (C5_plus - best_C5) / eps
        
        # Update step
        h_new = h - lr * g
        h_new = project_h(h_new, n_points)
        
        # Check if better
        new_C5 = evaluate_h(h_new, n_points)
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            no_improvement = 0
            best_h = h.copy()
        else:
            no_improvement += 1
            if no_improvement > patience:
                lr *= 0.5
                no_improvement = 0
                if lr < 1e-6:
                    break
                    
        iter_count += 1
        
    return best_h, best_C5

def genetic_algorithm_phase(n_points, budget_s, start_time):
    """Genetic algorithm to explore solution space"""
    pop_size = 50
    generations = 500
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_ratio = 0.2
    
    # Initialize population
    population = []
    for i in range(pop_size):
        if 'initial_h_values' in globals():
            h = initial_h_values.copy()
        else:
            # Sinusoidal initialization
            x = np.linspace(0, 2, n_points)
            h = 0.5 * (1 + np.sin(2 * np.pi * x / 0.5))  # Period 0.5
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = None
    best_C5 = float('inf')
    
    for gen in range(generations):
        if (time.time() - start_time) > budget_s * 0.7:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            C5 = evaluate_h(h, n_points)
            fitness.append(C5)
            if C5 < best_C5:
                best_C5 = C5
                best_h = h.copy()
        
        # Elitism
        sorted_indices = np.argsort(fitness)
        elite_count = int(pop_size * elitism_ratio)
        elite = [population[i] for i in sorted_indices[:elite_count]]
        
        # Create new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = population[sorted_indices[np.random.randint(0, elite_count)]]
            parent2 = population[sorted_indices[np.random.randint(0, elite_count)]]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            child1 = project_h(child1, n_points)
            child2 = project_h(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
        
    return best_h, best_C5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    # Try a larger size if we have time
    if budget_s > 500:
        n_points = 60
    
    # Phase 1: Genetic Algorithm
    best_h_ga, best_C5_ga = genetic_algorithm_phase(n_points, budget_s, start_time)
    
    # Phase 2: Gradient Descent
    h_refined, C5_refined = gradient_descent_phase(best_h_ga, n_points, budget_s, start_time)
    
    # Final evaluation
    final_C5 = evaluate_h(h_refined, n_points)
    # Final projection to be safe
    h_final = project_h(h_refined, n_points)
    
    return (h_final, final_C5, n_points)
