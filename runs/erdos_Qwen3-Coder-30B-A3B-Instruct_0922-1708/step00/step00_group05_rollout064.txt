import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default construction
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h *= (n_points / 2) / np.sum(h)
    
    # Initialize variables
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    max_iter = 6000
    eta = 0.01
    patience = 0
    last_improvement = 0
    
    def project_and_normalize(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Adjust sum to exactly n_points/2
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Proportional redistribution
            scale = (n_points/2) / current_sum
            h *= scale
        return h
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Compute all overlaps
        all_overlaps = np.correlate(h, 1 - h, mode="full")
        # Scale by dx
        all_overlaps *= dx
        # Extract the valid overlaps
        k_offset = n_points - 1  # shift indices start from -n+1 to n-1
        valid_overlaps = all_overlaps[k_offset:]  # only positive shifts
        # Find max overlap and its shift
        max_idx = np.argmax(valid_overlaps)
        max_k = max_idx - (n_points - 1)
        
        # Compute gradient for max_k
        grad = np.zeros_like(h)
        if max_k >= 0:
            # h[i] * (1 - h[i + k]) term contributes
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k]) * dx
            # But we only have valid entries
        else:
            # h[i] * (1 - h[i + k]) where k is negative -> has to be h[i - abs(k)]
            abs_k = -max_k
            for i in range(abs_k, n_points):
                grad[i] += -h[i - abs_k] * dx
        
        # Update h
        h = h - eta * grad
        # Project and normalize
        h = project_and_normalize(h)
        
        # Check improvement
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
            last_improvement = iteration
        else:
            patience += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0 or patience > 50:
            if patience > 50:
                eta *= 0.95
            patience = 0
        
        # Early stopping
        if iteration - last_improvement > 100:
            break
    
    # Final prediction
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)
