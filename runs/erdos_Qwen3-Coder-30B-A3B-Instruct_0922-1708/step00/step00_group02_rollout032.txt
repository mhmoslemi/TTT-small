import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, n_points):
    dx = 2.0 / n_points
    # Compute all overlaps for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    # The maximum overlap (C5) is the maximum value
    c5 = np.max(overlap) * dx
    return c5

def project_h(h, n_points):
    # Ensure values are in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to match sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    # We compute the gradient for the overlap at shift k_max
    # For each i: gradient[i] = (1 - h[i + k_max]) - h[i]
    # Boundary conditions are handled by indexing
    grad = np.zeros_like(h)
    h_shifted = np.roll(h, k_max)
    grad = (1 - h_shifted) - h
    return grad
