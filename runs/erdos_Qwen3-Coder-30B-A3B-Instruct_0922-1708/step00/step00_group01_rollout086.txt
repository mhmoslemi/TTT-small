import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints"""
    h = np.clip(h, 0, 1)
    target_sum = n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def nelder_mead_refine(h0, dx, maxiter=100):
    """Refine h using Nelder-Mead"""
    def obj(h):
        return -evaluate_h(h, dx)  # Negative because we minimize
    
    h0 = project_h(h0, len(h0))
    res = minimize(obj, h0, method='Nelder-Mead', options={'maxiter': maxiter, 'disp': False})
    return project_h(res.x, len(h0))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47
    dx = 2.0 / n_points
    pop_size = 100
    max_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 20
    simplex_size = 10
    noise_scale = 0.05
    
    # Initialize population
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        # Create starting point
        h0 = np.ones(n_points) * 0.5
        h0[0] = h0[-1] = 0.1
        h0[n_points//2] = 1.0
    
    h0 = project_h(h0, n_points)
    population = [h0.copy()]
    
    for i in range(1, pop_size):
        h = h0 + np.random.normal(0, noise_scale, n_points)
        h = project_h(h, n_points)
        population.append(h)
    
    best_h = h0.copy()
    best_c5 = evaluate_h(best_h, dx)
    
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (lower is better)
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        elites = [population[i] for i in sorted_indices[:tournament_size]]
        
        # Select parents
        parents = []
        for _ in range(pop_size):
            if random.random() < crossover_rate:
                parent1, parent2 = random.sample(elites, 2)
                # Blend crossover
                alpha = random.random()
                child = alpha * parent1 + (1 - alpha) * parent2
                child = project_h(child, n_points)
            else:
                child = random.choice(elites)
                # Mutation
                if random.random() < mutation_rate:
                    noise = np.random.normal(0, noise_scale, n_points)
                    child = child + noise
                    child = project_h(child, n_points)
            parents.append(child)
        
        # Refine top solutions with Nelder-Mead
        top_solutions = [population[i] for i in sorted_indices[:simplex_size]]
        for i, sol in enumerate(top_solutions):
            refined = nelder_mead_refine(sol, dx, maxiter=20)
            parents[i] = refined
        
        population = parents
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
