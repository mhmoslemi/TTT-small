import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 47  # Use the initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Ensure it sum to target_sum
        h = h * target_sum / np.sum(h)
    else:
        # Create a random feasible initial solution
        h = np.random.rand(n_points)
        h = h * target_sum / np.sum(h)
    
    # Ensure h in [0, 1]
    h = np.clip(h, 0, 1)
    
    # Optimization parameters
    max_iter = 1000
    learning_rate = 0.1
    tol = 1e-6
    start_time = time.time()
    
    best_c5 = evaluate_c5(h, dx)
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        # The full cross-correlation with 1-h
        h_rev = 1 - h
        cross_corr = signal.correlate(h, h_rev, mode='full')
        # Adjust indices for proper lags - we need to focus on lags from -n+1 to n-1
        # e.g., for size 3: lag -2, -1, 0, 1, 2
        lag_indices = np.arange(-n_points + 1, n_points)
        lag_corr = cross_corr[n_points - 1:]  # Adjust indices
        c5_vals = lag_corr * dx
        
        # Find the maximum
        k_max = np.argmax(c5_vals)
        max_val = c5_vals[k_max]
        
        # If we have a new best, update it
        if max_val < best_c5:
            best_c5 = max_val
            best_h = h.copy()
        
        # Compute gradient of the objective w.r.t h for the max overlap lag
        # Gradient: d/dh_i of sum over j of h[j] * (1 - h[j + k_max])
        # For i-th element, it's (1 - h[(i + k_max) % n]) if k_max >= 0
        # or (1 - h[(i - |k_max| ) % n]) if k_max < 0
        grad = np.zeros(n_points)
        if k_max == 0:  # This case is not needed here as we want max k
            continue
        else:
            k_val = k_max - (n_points - 1)  # Map (k_max) back to -n+1 to n-1
            for i in range(n_points):
                j = (i + k_val) % n_points
                grad[i] = 1.0 - h[j]
        
        # Solve linear subproblem: minimize grad^T h subject to h in feasible region
        # This just picks the direction of the negative gradient, clamped to [0,1], and rescaled
        
        # Frank-Wolfe step: use d as the gradient direction, project to feasible region
        d = -grad
        # Make sure to keep feasible - clip and rescale
        d = np.clip(d, 0, 1)
        d_sum = np.sum(d)
        if d_sum > 0:
            d = d * target_sum / d_sum
        
        # Update h
        h = (1 - learning_rate) * h + learning_rate * d
        # Ensure sum constraint is still satisfied and values in [0,1]
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        # Check for convergence - we don't need strict convergence
        if iteration % 50 == 0:
            current_c5 = evaluate_c5(h, dx)
            if abs(current_c5 - best_c5) < tol:
                break
    
    final_c5 = evaluate_c5(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral C5"""
    h_arr = np.array(h)
    h_rev = 1 - h_arr
    cross_corr = signal.correlate(h_arr, h_rev, mode='full')
    # Correlation indices correspond to lags from -n+1 to n-1
    n = len(h_arr)
    lag_indices = np.arange(-n + 1, n)
    # Extract correlation values for each lag
    # The middle elements correspond to zero lag, and the array is symmetric
    # But we need the overlap over the correct range of lag indices
    # full mode gives us 2n-1 elements
    lag_corr = cross_corr[n - 1:]  # This gives us lags from -n+1 to n-1
    c5_vals = lag_corr * dx
    max_over = np.max(c5_vals)
    return max_over
