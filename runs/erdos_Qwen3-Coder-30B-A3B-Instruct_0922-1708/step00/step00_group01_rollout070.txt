import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for the given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using cross-correlation
    # We want: int h(x)(1 - h(x+k)) dx
    # This translates to: correlate(h, 1-h, mode="full") * dx
    overlaps = correlate(h, 1 - h, mode="full")
    # The maximum overlap corresponds to the maximum in the center of the cross-correlation
    # The full correlation has 2*n - 1 elements
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def project_to_feasible(h, n_points):
    """Project h to feasible region: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def solve_frank_wolfe(seed=42, budget_s=600, n_points=47):
    np.random.seed(seed)
    time_start = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.random.rand(n_points)
        h = project_to_feasible(h, n_points)
    
    # Parameters
    max_iter = 5000
    eta = 0.1
    tol = 1e-6
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    for it in range(max_iter):
        if time.time() - time_start > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        overlaps = correlate(h, 1 - h, mode="full")
        max_k = np.argmax(overlaps) - (n_points - 1)
        
        # Compute gradient: for each i, grad[i] = 1 - h[(i + k) % n]
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[:n_points - max_k] = 1 - h[max_k:]
        else:
            grad[-max_k:] = 1 - h[:n_points + max_k]
        
        # Solve linear subproblem: minimize <d, grad> subject to:
        # 0 <= d[i] <= 1, sum(d) = n_points / 2
        # This problem is separable in each component, so we can solve:
        # min sum_i d[i] * grad[i] subject to 0 <= d[i] <= 1, sum(d) = n_points / 2
        # This can be solved by sorting gradients, and setting variables to either 0 or 1
        # or to their "cut" value to satisfy the sum constraint.
        
        # Sort indices by gradients (ascending)
        idx_sorted = np.argsort(grad)
        d = np.zeros(n_points)
        remaining_sum = n_points / 2.0
        
        for i in range(n_points):
            j = idx_sorted[i]
            if remaining_sum >= 1:
                d[j] = 1.0
                remaining_sum -= 1
            elif remaining_sum > 0:
                d[j] = remaining_sum
                remaining_sum = 0
            else:
                d[j] = 0
            if remaining_sum <= 0:
                break
        
        # Update h
        h = (1 - eta) * h + eta * d
        h = project_to_feasible(h, n_points)
        
        # Evaluate and keep best
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    return solve_frank_wolfe(seed, budget_s, **kwargs)
