import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for a given h."""
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints."""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    patience = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h(h, n_points)
    else:
        h = np.full(n_points, 0.5 * dx)
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    patience_count = 0
    
    # Perturbation schedule
    sigma = 0.1
    decay = 0.99
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate new candidate
        h_new = h + np.random.normal(0, sigma, n_points)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            h = h_new
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_count = 0
        else:
            patience_count += 1
            
        # Update sigma
        sigma *= decay
        
        # Early stopping if no improvement
        if patience_count > patience:
            break
    
    return (best_h.tolist(), best_c5, n_points)
