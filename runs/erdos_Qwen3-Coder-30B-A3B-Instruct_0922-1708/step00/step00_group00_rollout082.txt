    def compute_overlap(h_val, k):

        return np.dot(h_val, (1 - h_shifted)) * dx
    
    def compute_C5(h_val):
        max_val = -float('inf')
        for k in range(n_points):
            val = compute_overlap(h_val, k)
            max_val = max(max_val, val)
        return max_val
    
    # Function to project h to feasible set
    def project_h(h_val, target_sum, dx):
        # Ensure bounds
        h_val = np.clip(h_val, 0, 1)
        # Adjust to satisfy integral constraint
        current_sum = np.sum(h_val) * dx
        scale_factor = target_sum / current_sum
        h_val = h_val * scale_factor
        return h_val

            h_minus[i] -= eps
            h_minus = project_h(h_minus, target_sum, dx)

        # Update h using gradient descent + projection
        learning_rate = 0.1
