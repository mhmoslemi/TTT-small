import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5  # Start with uniform distribution
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum

    # Start timing
    start_time = time.time()
    
    # Parameters
    max_iter = 6000
    eta = 0.01
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    def evaluate_c5(h_vec):
        # Evaluate all overlaps
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # We only need the valid overlaps, not the full correlation
        # The valid range is [0, n_points-1] for the actual overlap integrals
        # The middle part of the correlation corresponds to the true overlaps
        valid_corr = corr[n_points-1:]  # Take from the middle of correlation
        return np.max(valid_corr) * dx
    
    def get_worst_lag(h_vec):
        # Compute correlation
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        valid_corr = corr[n_points-1:]   # Take valid overlaps  
        max_corr_idx = np.argmax(valid_corr)
        return max_corr_idx
    
    def compute_gradient(h_vec, k):
        grad = np.zeros_like(h_vec)
        # For lag k, compute gradient
        # For i < n_points-k, moving forward
        # For i >= k, moving backward
        for i in range(n_points):
            if i + k < n_points:
                grad[i] += (1 - h_vec[i + k]) * dx
            if i >= k:
                grad[i] -= h_vec[i - k] * dx
        return grad
    
    converged = False
    iter_count = 0
    no_improvement_count = 0
    
    while iter_count < max_iter:
        if time.time() - start_time > budget_s - 1.0:
            break
            
        current_c5 = evaluate_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            break
            
        # Get the worst overlap lag
        worst_k = get_worst_lag(h)
        
        # Compute gradient for this lag
        grad = compute_gradient(h, worst_k)
        
        # Update with gradient descent
        h = h - eta * grad
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust to maintain sum
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) > 1e-10:
            # Adjust proportionally
            adjustment = (target_sum - current_sum) / n_points
            h = h + adjustment
            # Re-clip just in case
            h = np.clip(h, 0, 1)
            
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % 100 == 0 or no_improvement_count <= 50:
            eta *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
