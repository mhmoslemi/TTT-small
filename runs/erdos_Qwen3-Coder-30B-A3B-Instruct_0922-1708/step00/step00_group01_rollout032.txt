import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available, otherwise create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 47
        # Create symmetric starting point
        h = np.ones(n_points) * 0.5
        mid = n_points // 2
        for i in range(mid):
            h[i] = 0.1 + 0.4 * (1 - i / mid)
            h[n_points - 1 - i] = h[i]
        # Adjust sum to meet constraint exactly
        h = h * (n_points / 2) / np.sum(h)
    
    # Set up optimization parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iterations = 3000
    convergence_tol = 1e-5
    dx = 2.0 / n_points
    
    # Store best solution
    best_h = h.copy()
    best_C5 = float('inf')
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags - use correlate for efficiency
        # h(x) * (1 - h(x+k)) for all k
        h_rev = np.flip(h)
        overlaps = correlate(h, h_rev, mode='full')
        # Adjust indices to match problem definition
        overlap_vals = np.zeros(n_points * 2 - 1)
        # Shift to match positive lags first
        overlap_vals[n_points-1:] = overlaps[:n_points]
        overlap_vals[:n_points-1] = overlaps[n_points:][::-1]
        # Multiply by dx
        overlap_vals = overlap_vals * dx
        
        # Use log-sum-exp to approximate maximum
        C5_approx = np.log(np.sum(np.exp(tau * overlap_vals))) / tau
        
        if C5_approx < best_C5:
            best_C5 = C5_approx
            best_h = h.copy()
            
        # Compute gradient of the log-sum-exp approximation w.r.t h
        exp_overlaps = np.exp(tau * overlap_vals)
        exp_sum = np.sum(exp_overlaps)
        if exp_sum == 0:
            break
            
        # Gradient computation for each k, sum over all k:
        grad = np.zeros(n_points)
        for k in range(len(overlap_vals)):
            # This is the key analogy to the projection step:
            # The gradient is weighted sum of contributions from each overlap
            # One major simplification we make is by using the convolution property
            # and leveraging the fact that d/dh_i of overlap_k is very constrained
            # We will compute the gradient element-wise manually
            # For each overlap term (with index k in overlap_vals), compute gradient impact on h[i]
            # Each h[i] contributes through the part h[i] * (1 - h[i+k]) in overlap_k
            # This contributes -dx * (1 - h[i+k]) to d/dh[i] of overlap_k for k = i+shift
            # That's complex, so we'll compute it properly with correlate
            
            # Let's instead use a derivative approximation by perturbation
            # Compute gradient with respect to h
            k_shift = k - (n_points - 1)  # -n_points+1 to n_points-1
            r = np.roll(h_rev, -k_shift)  # rolling to match alignment
            r = r[:n_points]  # trim to valid indices instead of full roll
            if k_shift >= 0:
                # For negative k_shift, shift h to the left (simulate i+k shifts)
                r = np.roll(h, -k_shift)  # roll forward
                r = r[:n_points]
            else:
                r = np.roll(h, -k_shift)  # roll forward but could also be negative lead
                # To avoid over-complication with shifting, just iterate correctly
                # Actually, we can derive analytical gradients correctly:
                # overlap_k = sum_i h[i] * (1 - h[i+k])
                # d/dh[i] = 1 * (1 - h[i + k]) for i+k valid
                # d/dh[i] = 0 else (or 1 * (1 - h[i - k])) if shifting term is negative
                # But for maximum over k's:
                # We'll compute "tensor-wise" derivative of each term to the current h
                h_shifted = np.zeros(n_points)
                if k_shift >= 0:
                    h_shifted[k_shift:] = h[:-k_shift] if k_shift < n_points else []
                else:
                    h_shifted[:k_shift] = h[-k_shift:] if -k_shift < n_points else []
                # In fact, let's be more careful:
                # For overlap_k = sum h[i] * (1 - h[i + shift])
                # Gradient w.r.t. h[i]:
                # d/dh[i] [ overlapping with h[i+shift] ] = 1 * (1 - h[i + shift]) 
                # This is more efficiently computed by constructing a difference matrix
                # But we proceed with basic diff using correlate of binary (1-h)
                binary_vec = 1 - h
                result = correlate(h, np.flip(binary_vec), mode='full')
                # Restore the positive overlap shift (move it to the middle)
                result = np.concatenate([result[n_points-1:], result[:n_points-1]])
                
            # Simulation: the complex individual loop is simplified below.
            # The idea:
            # ∂ overlap_k / ∂ h[i] changes based on k and the difference position
            # The updated "realistic" gradient computation yields:
            local_grad = np.zeros(n_points)
            for i in range(n_points):
                sl = (i - k_shift) % n_points
                local_grad[i] = -dx * (1 - h[sl])
            # Total gradient:
            grad += (exp_overlaps[k] / exp_sum) * local_grad
                
        # Gradient descent update
        h = h - learning_rate * grad
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Anneal tau
        tau *= anneal_factor
        if tau < 1e-5:
            tau = 1e-5
            
        # Check for convergence
        if iteration > 10 and np.abs(C5_approx - best_C5) < convergence_tol:
            break
            
    # Final evaluation using exact overlap (cross-correlation)
    overlap_vals = np.zeros(2 * n_points - 1)
    h_rev = np.flip(h)
    over = correlate(h, h_rev, mode='full')
    overlap_vals[n_points-1:] = over[:n_points]
    overlap_vals[:n_points-1] = over[n_points:][::-1]
    C5_exact = np.max(overlap_vals * dx)
    
    if C5_exact < best_C5:
        return (h.tolist(), C5_exact, n_points)
    else:
        return (best_h.tolist(), best_C5, n_points)
