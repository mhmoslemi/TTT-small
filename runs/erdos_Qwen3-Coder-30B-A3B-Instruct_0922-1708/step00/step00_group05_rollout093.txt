import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h_values):
    """Evaluate the C5 constant for given step function h"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    # Compute all correlations for different lags
    correlations = correlate(h_values, 1 - h_values, mode="full")
    # Extract the relevant part (positive lags) and multiply by dx
    max_corr = np.max(correlations) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):

    # Start with initial construction if available

        h0[0] = 0.2
        h0[1] = 0.8
        h0[2] = 0.3
        h0[3] = 0.7
        # Normalize to sum to n_points / 2
        h0 *= (n_points / 2) / np.sum(h0)
    
    # Initialize parameters
    n_points = len(h0)
    dx = 2.0 / n_points
    max_iter = 12000 # Estimated based on 0.05s per iteration
    n_particles = 20
    w = 0.7
    c1 = c2 = 2.0
    v_max = 0.1
    
    # Generate swarm
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    
    # Add initial solution as one particle
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    personal_best.append(h0.copy())
    personal_best_vals.append(evaluate_erdos_solution(h0))
    
    # Generate other particles
    for _ in range(n_particles - 1):
        # Perturb initial solution
        pert = np.random.normal(0, 0.1, n_points)
        h = h0 + pert
        h = np.clip(h, 0, 1)
        # Normalize
        h *= (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        personal_best.append(h.copy())
        personal_best_vals.append(evaluate_erdos_solution(h))
    
    # Find global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = personal_best[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]

    last_improvement_iter = 0
    start_time = time.time()

        if time.time() - start_time > budget_s:
            break
            
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            pbest = personal_best[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            v_new = (w * v + 
                     c1 * r1 * (pbest - h) + 
                     c2 * r2 * (global_best - h))
            
            # Limit velocity
            v_new = np.clip(v_new, -v_max, v_max)
            
            # Update position
            h_new = h + v_new
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            
            # Normalize to correct sum
            h_new *= (n_points / 2) / np.sum(h_new)
            
            # Evaluate new solution
            new_val = evaluate_erdos_solution(h_new)
            
            # Update personal best
            if new_val < personal_best_vals[i]:

                personal_best_vals[i] = new_val

                global_best_val = new_val

            # Store new particle and velocity
            particles[i] = h_new
            velocities[i] = v_new
            
        # Tournament selection and diversity maintenance
        if iteration % 100 == 0 and iteration > 0:
            # Reintroduce diversity if no improvement
            if iteration - last_improvement_iter > 100:
                # Perturb all particles slightly
                for i in range(n_particles):

                    h = particles[i] + pert

                    particles[i] = h

                    h_rand *= (n_points / 2) / np.sum(h_rand)
                    particles[i] = h_rand
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                    personal_best[i] = h_rand.copy()
                    personal_best_vals[i] = evaluate_erdos_solution(h_rand)
    
    return (global_best, global_best_val, n_points)
