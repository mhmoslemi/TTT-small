import numpy as np
from scipy.optimize import minimize
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # Return maximum value scaled by dx
    return np.max(conv) * dx

def project_h(h, dx):
    """Project h to satisfy constraints: h[i] in [0,1] and sum(h)*dx = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to 1/dx
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (1.0 / (h_sum * dx))
    return h

    start_time = time.time()
    learning_rate = 0.01

        eps = 1e-6
        grad = np.zeros_like(h)

        h_new = h - learning_rate * grad
