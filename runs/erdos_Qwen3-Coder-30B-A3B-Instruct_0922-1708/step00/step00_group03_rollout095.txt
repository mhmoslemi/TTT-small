import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the overlap integral for the given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The maximum value of the correlation gives the overlap integral
    max_overlap = np.max(corr) * dx
    return max_overlap

def initialize_population(n_points, initial_h_values, pop_size):
    """Initialize the population for genetic algorithm"""
    population = []
    # Add the initial solution
    if initial_h_values is not None:
        population.append(initial_h_values.copy())
    # Add random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        # Scale to meet constraint sum(h) * dx = 1
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def mutate(h, mutation_rate, dx):
    """Mutate a solution"""
    h = h.copy()
    if random.random() < mutation_rate:
        # Select indices to modify
        n_modify = max(1, int(len(h) * 0.1))
        indices = random.sample(range(len(h)), n_modify)
        for i in indices:
            # Add small random change
            delta = random.uniform(-0.1, 0.1) * dx
            h[i] += delta
            # Clip to [0,1]
            h[i] = max(0, min(1, h[i]))
        # Rescale to maintain constraint
        h = h * (len(h) / 2) / np.sum(h)
    return h

def crossover(h1, h2):
    """Perform crossover between two solutions"""
    n_points = len(h1)
    if random.random() < 0.5:
        # Swap segments
        cut_point = random.randint(1, n_points - 1)
        h_child1 = np.concatenate([h1[:cut_point], h2[cut_point:]])
        h_child2 = np.concatenate([h2[:cut_point], h1[cut_point:]])
        return h_child1, h_child2
    else:
        return h1.copy(), h2.copy()

def select_parents(population, fitnesses, num_parents):
    """Tournament selection"""
    parents = []
    for _ in range(num_parents):
        # Tournament size
        tournament_size = max(2, len(population) // 10)
        tournament_indices = random.sample(range(len(population)), tournament_size)
        # Select the best in tournament
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        parents.append(population[winner_idx])
    return parents

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run the optimization"""
    random.seed(seed)
    np.random.seed(seed)
    
    # Timer
    start_time = time.time()
    remaining_budget = budget_s
    
    # Parameters
    n_points = 70
    pop_size = 100
    generations = 300  # estimated
    mutation_rate = 0.2
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, initial_h_values, pop_size)
    
    # Evaluate initial fitness
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_C5 = fitnesses[best_idx]
    
    # GA loop
    last_improvement_gen = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 30 and generation < generations:
        # Evaluate current fitness
        current_fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Select parents
        num_parents = min(len(population) // 2, pop_size // 2)
        parents = select_parents(population, current_fitnesses, num_parents)
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - len(parents):
            parent1, parent2 = random.sample(parents, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_rate, dx)
            child2 = mutate(child2, mutation_rate, dx)
            offspring.extend([child1, child2])
        
        # Combine parents and offspring
        new_population = (parents + offspring)[:pop_size]
        # Evaluate new population
        new_fitnesses = [evaluate_erdos_solution(h) for h in new_population]
        
        # Update global best
        min_fitness_idx = np.argmin(new_fitnesses)
        if new_fitnesses[min_fitness_idx] < best_C5:
            best_C5 = new_fitnesses[min_fitness_idx]
            best_h = new_population[min_fitness_idx].copy()
            last_improvement_gen = generation
            
        # Replace population
        population = new_population
        fitnesses = new_fitnesses
        
        generation += 1
        
        # Early stopping based on no improvement
        if generation - last_improvement_gen > 50:
            break
    
    # Ensure final result is valid
    best_h = np.maximum(0, np.minimum(1, best_h))
    final_sum = np.sum(best_h)
    best_h = best_h * (n_points / 2) / final_sum
    
    return (best_h, best_C5, n_points)
