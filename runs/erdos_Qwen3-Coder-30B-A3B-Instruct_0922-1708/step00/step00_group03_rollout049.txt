import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    dx = 2.0 / len(h)
    # Compute correlation of h with (1-h) in full mode
    corr = correlate(h, 1 - h, mode="full")
    # Maximum value scaled by dx
    return np.max(corr) * dx

def normalize_h(h, n_points):
    """Ensure h sums to n_points / 2."""
    h = np.array(h)
    sum_h = np.sum(h)
    if abs(sum_h - n_points / 2) > 1e-6:
        h = h * (n_points / 2) / sum_h
    return np.clip(h, 0, 1)

def mutate_h(h, mutation_rate=0.2, n_points=70):
    """Mutate the step function h."""
    h = np.array(h)
    if np.random.random() < mutation_rate:
        # Select random indices
        num_mutate = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, num_mutate, replace=False)
        # Apply random changes
        delta = np.random.uniform(-0.1, 0.1, num_mutate) * 2.0 / n_points
        h[indices] += delta
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = normalize_h(h, n_points)
    return h.tolist()

def crossover_h(h1, h2):
    """Swap segments of two step functions."""
    n_points = len(h1)
    if n_points != len(h2):
        return h1, h2
    point = np.random.randint(1, n_points-1)
    new_h1 = np.concatenate([h1[:point], h2[point:]])
    new_h2 = np.concatenate([h2[:point], h1[point:]])
    return new_h1.tolist(), new_h2.tolist()

def tournament_selection(population, fitnesses, tournament_size=0.2):
    """Select best individuals based on tournament selection."""
    selected_indices = []
    num_select = int(len(population) * tournament_size)
    for _ in range(num_select):
        tournament_indices = np.random.choice(len(population), 5, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
        selected_indices.append(winner_index)
    return [population[i] for i in selected_indices]

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    population_size = 100
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        initial = normalize_h(initial_h_values, n_points)
        population.append(initial)
    else:
        # Fallback: generate random step functions
        initial = np.random.random(n_points)
        initial = normalize_h(initial, n_points)
        population.append(initial)
    
    # Fill rest of population
    for _ in range(population_size - 1):
        h = np.random.random(n_points)
        h = normalize_h(h, n_points)
        population.append(h.tolist())
    
    # Evaluate initial fitnesses
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    
    best_h = population[np.argmin(fitnesses)]
    best_c5 = min(fitnesses)
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 10:
        generation += 1
        # Tournament selection
        selected = tournament_selection(population, fitnesses, 0.2)
        
        # Create offspring
        offspring = []
        num_offspring = len(selected) - 1
        for i in range(0, num_offspring, 2):
            parent1 = selected[i]
            parent2 = selected[i+1] if i+1 < len(selected) else selected[0]
            child1, child2 = crossover_h(parent1, parent2)
            child1 = mutate_h(child1)
            child2 = mutate_h(child2)
            offspring.extend([child1, child2])
        
        # Add offspring to population
        population.extend(offspring[:population_size - len(population)])
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Select top individuals
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices[:population_size]]
        fitnesses = [fitnesses[i] for i in sorted_indices[:population_size]]
        
        # Track best solution
        current_best_c5 = min(fitnesses)
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break

    return best_h, best_c5, n_points
