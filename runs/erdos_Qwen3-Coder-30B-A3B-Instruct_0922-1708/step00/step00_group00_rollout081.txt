import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Store best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    # ADMM parameters
    rho = 1.0
    max_iter = 500
    tol = 1e-6
    penalty = 1.0
    
    # Get target integral
    target_integral = n_points / 2
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s * 0.95:
            break
            
        # Evaluate current C5
        current_c5 = evaluate_erdos_solution(h, n_points)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute overlap terms for all shifts
        overlap_vals = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            shift = k - n_points + 1
            if shift >= 0:
                overlap_vals[k] = np.sum(h[shift:] * (1 - h[:n_points-shift]))
            else:
                overlap_vals[k] = np.sum(h[:n_points+shift] * (1 - h[-shift:]))
        
        # Find maximum overlap
        max_overlap_idx = np.argmax(overlap_vals)
        max_overlap = overlap_vals[max_overlap_idx]
        
        # Compute gradient and update h
        grad = np.zeros(n_points)
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                if 0 <= i+k < n_points and 0 <= i < n_points:
                    if k >= 0:
                        grad[i] += (1 - h[i+k]) * dx
                    else:
                        grad[i] += h[i+k] * dx
        
        # ADMM update
        h_old = h.copy()
        h = h - penalty * grad
        
        # Project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to enforce integral constraint
        integral = np.sum(h) * dx
        h = h * target_integral / integral
        
        # Check change in h
        if np.linalg.norm(h - h_old) < tol:
            penalty *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, n_points):
    """Evaluate C5 using numpy correlate"""
    dx = 2.0 / n_points
    h = np.array(h)
    # Compute overlaps
    correlate = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part
    c5_val = np.max(correlate) * dx
    return c5_val
