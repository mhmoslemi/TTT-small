import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialization
    n_points = 70
    dx = 2.0 / n_points
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    
    # Optimization parameters
    max_iter = 1000
    lambda_reg = 1e-4
    tolerance = 1e-6
    
    def compute_overlap(h_vec):
        # Compute correlations for all shifts and return max
        conv = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only use the relevant parts (excluding padding)
        n = len(h_vec)
        full_conv = conv[n-1:2*n-1]
        return np.max(full_conv) * dx
    
    def objective_and_gradient(h_vec):
        # Regularized objective
        overlap = compute_overlap(h_vec)
        # Simple gradient approximation using finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (compute_overlap(h_plus) - compute_overlap(h_minus)) / (2 * eps)
        
        # Add regularization gradient
        reg_grad = 2 * lambda_reg * h_vec
        grad += reg_grad
        
        return overlap, grad
    
    # Gradient descent with line search to update h
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        overlap, grad = objective_and_gradient(h)
        # Simple gradient descent step
        alpha = 0.01 / (1.0 + iteration / 100.0)
        h_new = h - alpha * grad
        
        # Clip to [0, 1] and rescale to sum to n_points / 2
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        h = h_new
        
        # Check termination
        if iteration > 10 and abs(overlap - compute_overlap(h)) < tolerance:
            break
    
    # Local shift refinement
    best_h = h
    best_c5 = compute_overlap(h)
    
    for shift in range(-10, 11):
        if time.time() - start_time > budget_s - 1:
            break
        h_shifted = np.roll(h, shift)
        h_shifted = np.clip(h_shifted, 0, 1)
        h_shifted = h_shifted * (n_points / 2.0) / np.sum(h_shifted)
        c5_shifted = compute_overlap(h_shifted)
        if c5_shifted < best_c5:
            best_h = h_shifted
            best_c5 = c5_shifted
    
    return (best_h.tolist(), best_c5, n_points)
