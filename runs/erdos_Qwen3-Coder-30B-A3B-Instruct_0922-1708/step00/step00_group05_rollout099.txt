    def fourier_to_h(coeffs):
        x = np.linspace(0, 2, n_points)
        h = coeffs[0] * np.ones(n_points)  # a_0
        for k in range(1, len(coeffs)):  # a_1, a_2, ...
            h += coeffs[k] * np.cos(k * np.pi * x / 2)
        return np.clip(h, 0, 1)

        # Ensure sum = n_points/2
        h = h * (n_points/2) / np.sum(h)

        c5 = max(np.correlate(h, 1-h, mode="full")) * dx

        return max(np.correlate(h, 1-h, mode="full")) * dx

        else:
            patience += 1
            
        # Scale learning rate

            lr *= 0.95
        lr = max(lr, 1e-6)
        
        iter_count += 1
        if iter_count % 100 == 0:
            lr *= decay
    
    return (best_h, best_c5, n_points)
