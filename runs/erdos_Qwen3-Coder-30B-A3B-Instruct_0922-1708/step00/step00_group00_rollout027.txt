import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    """Compute the overlap integral C_5 for given h_values"""
    n_points = len(h_values)
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all overlaps for different shifts
    overlaps = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap index is at the center of the full correlation
    max_overlap_idx = len(overlaps) // 2
    max_overlap = overlaps[max_overlap_idx]
    # The other overlaps correspond to shifts k from -n_points+1 to n_points-1
    # But we only care about those that make sense (in range)
    for i in range(len(overlaps)):
        if i != max_overlap_idx:
            if overlaps[i] > max_overlap:
                max_overlap = overlaps[i]
    return max_overlap * dx
