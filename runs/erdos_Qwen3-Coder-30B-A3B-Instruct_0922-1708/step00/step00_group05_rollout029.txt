import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.full(n_points, 0.5)
    
    # Normalize if needed to satisfy integral constraint
    integral = np.sum(h) * 2.0 / n_points
    h = h / integral * (n_points / 2.0)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    def evaluate(h_vector):
        # Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx
        # This is equivalent to max_k correlation(h, 1-h, mode='full') * dx
        dx = 2.0 / n_points
        h1 = 1 - h_vector
        correlation = signal.correlate(h_vector, h1, mode='full')
        max_corr = np.max(correlation)
        return max_corr * dx
    
    best_h = h.copy()
    best_c5 = evaluate(best_h)
    
    # Parameters
    max_perturb = 0.05
    smoothing_window = 3
    iterations_no_improvement = 0
    max_no_improvement = 100
    trial_per_iter = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 0.1:
        if iterations_no_improvement >= max_no_improvement:
            break
            
        improved = False
        
        for _ in range(trial_per_iter):
            # Create a perturbed version
            h_new = h.copy()
            
            # Randomly select some points to perturb
            n_perturb = np.random.randint(1, n_points // 4)
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            # Perturb values
            for i in indices:
                delta = np.random.uniform(-max_perturb, max_perturb)
                h_new[i] += delta
            
            # Maintain the integral constraint
            current_sum = np.sum(h_new)
            target_sum = n_points / 2.0
            adjustment = (target_sum - current_sum) / n_points
            h_new += adjustment
            
            # Keep values within bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Apply smoothing to maintain structure
            if n_points > smoothing_window:
                h_smooth = np.convolve(h_new, np.ones(smoothing_window)/smoothing_window, mode='same')
                h_new = h_smooth
            
            # Evaluate new solution
            c5_new = evaluate(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
                improved = True
                
        if improved:
            h = best_h.copy()
            iterations_no_improvement = 0
        else:
            iterations_no_improvement += 1
    
    return (best_h.tolist(), best_c5, n_points)
