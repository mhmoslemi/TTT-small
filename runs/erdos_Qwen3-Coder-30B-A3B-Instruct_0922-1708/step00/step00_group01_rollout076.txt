import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Fallback: symmetric construction
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Add some variation
        h[10:37] = 0.7
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Optimization parameters
    lr = 0.1
    decay_every = 1000
    max_iter = 5000
    
    def compute_c5(h_vec):
        # Cross-correlation with 1-h
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Only consider non-negative lags (the valid ones)
        # Center is at len(h)-1, we want indices from 0 to 2*len(h)-2
        # But we only care about lag >= 0
        mid = len(h_vec) - 1
        max_val = np.max(corr[mid:])
        return max_val * dx
    
    def project_h(h_vec):
        # Clip to [0,1] and rescale to target sum
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if abs(current_sum) > 1e-10:
            h_vec *= target_sum / current_sum
        return h_vec
    
    def compute_gradients(h_vec, k_max):
        # Compute gradient w.r.t. h[i] for lag k_max
        # d/dh[i] [h[i] * (1 - h[i+k_max])]
        # = (1 - h[i+k_max]) - h[i] * dh[i+k_max]/dh[i]
        # For correlation h * (1-h), gradient for i-th entry is
        # 1 - h[i+k_max] for i and 1 - h[i-k_max] for i+k_max
        # This is an approximation - we're doing directional gradients
        
        # Instead we'll compute finite differences for gradient check
        grad = np.zeros_like(h_vec)
        # Only compute gradient for the specific dominating lag
        # For k=k_max, h[i] contributes to (i+k)th element of cross-correlation
        # And vice versa
        # We'll compute symbolically the gradient for the maximum correlation
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        mid = len(h_vec) - 1
        k_max_idx = mid + k_max
        corr_val = corr[k_max_idx]
        # Each element contributes to the maximum correlation via its neighbors.
        # Exact symbolic gradient is hard, so let's compute a approximated one:
        for i in range(len(h_vec)):
            if i + k_max < len(h_vec):
                grad[i] += 1 - h_vec[i + k_max]
            if i >= k_max:
                grad[i] += -(1 - h_vec[i - k_max])
        return grad
    
    iteration = 0
    while iteration < max_iter and (time.time() - start_time) < budget_s:
        if iteration % decay_every == 0 and iteration > 0:
            lr *= 0.5
            
        # Compute overlap values
        corr = signal.correlate(h, 1 - h, mode='full')
        mid = len(h) - 1
        corr_vals = corr[mid:]  # Valid lags from 0 to n-1
        k_max = np.argmax(corr_vals)
        max_overlap = corr_vals[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        g = compute_gradients(h, k_max)
        
        # Stochastic gradient descent update
        h = h - lr * g
        
        # Project back
        h = project_h(h)
        
        iteration += 1
        if (time.time() - start_time) > 0.8 * budget_s:
            break
            
    # Final check
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    mid = len(best_h) - 1
    corr_vals = corr[mid:]  # Valid lags from 0 to n-1
    final_c5 = np.max(corr_vals) * dx
    
    return (best_h, final_c5, len(best_h))
