import numpy as np
import time
from scipy import signal
import math

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    conv = signal.correlate(h, 1 - h, mode="full")
    # Take the maximum value (properly scaled)
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_particles = 20
    max_iter = int(budget_s / 0.05)  # roughly 12000 iterations
    inertia_w = 0.7
    c1, c2 = 2.0, 2.0
    velocity_bounds = 0.1
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():

    else:
        # Create a basic initial guess
        h_init = np.ones(n_points) * 0.5
        # Distribute more evenly
        h_init = np.clip(h_init, 0.0, 1.0)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
    # Normalize to sum to n_points/2 = 35
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial population
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))
    
    for _ in range(n_particles - 1):
        h_pert = h_init + np.random.normal(0, 0.1, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-velocity_bounds, velocity_bounds, n_points))

    # Store personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_vals = [evaluate_erdos_solution(p) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(personal_best_vals)
    global_best = particles[global_best_idx].copy()
    global_best_val = personal_best_vals[global_best_idx]
    
    last_improvement = 0
    iteration = 0
    early_stop_count = 0
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        iteration += 1
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = (inertia_w * velocities[i] + 
                    c1 * r1 * (personal_bests[i] - particles[i]) + 
                    c2 * r2 * (global_best - particles[i]))
            # Clip velocity
            v_new = np.clip(v_new, -velocity_bounds, velocity_bounds)
            velocities[i] = v_new
            
            # Update position
            h_new = particles[i] + velocities[i]
            # Bound h_new in [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Normalize to sum to n_points / 2
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            
            # Evaluate and update personal best
            new_val = evaluate_erdos_solution(h_new)
            if new_val < personal_best_vals[i]:
                personal_bests[i] = h_new.copy()
                personal_best_vals[i] = new_val
                
                # Check if this is a new global best
                if new_val < global_best_val:
                    global_best = h_new.copy()
                    global_best_val = new_val
                    last_improvement = iteration
                    early_stop_count = 0
        # Tournament selection
        if iteration % 20 == 0:
            # Sort particles by their fitness
            fitness = np.array(personal_best_vals)
            sorted_indices = np.argsort(fitness)
            # Keep top 50% and replace bottom 50%
            keep_indices = sorted_indices[:n_particles // 2]
            replace_indices = sorted_indices[n_particles // 2:]
            
            for replace_idx in replace_indices:
                # Re-initialize with small random perturbation
                h_new = global_best.copy()

                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                particles[replace_idx] = h_new
                velocities[replace_idx] = np.random.uniform(-velocity_bounds, velocity_bounds, n_points)
                personal_bests[replace_idx] = h_new.copy()
                personal_best_vals[replace_idx] = evaluate_erdos_solution(h_new)
        
        # Early stopping if no improvement for many iterations
        early_stop_count += 1
        if early_stop_count > 100:
            early_stop_count = 0
            # Introduce some novelty
            for i in range(n_particles):
                if np.random.rand() > 0.75:
                    h_new = particles[i] + np.random.normal(0, 0.01, n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    particles[i] = h_new
                    velocities[i] = np.random.uniform(-velocity_bounds, velocity_bounds, n_points)
    
    # Final re-evaluation of best solution
    final_c5 = evaluate_erdos_solution(global_best)
    
    return (global_best, final_c5, n_points)
