def compute_c5(h, dx):
    # compute all correlations of h with (1-h) and find max
    cross = correlate(h, 1-h, mode='full')
    # The cross-correlation returns indices from -n+1 to n-1
    # We want valid k in [0, 2]
    # k corresponds to shift: k = (shift + n-1) * dx = shift_idx * dx 
    shift = np.arange(-len(h)+1, len(h))
    # else: convert to shift values:
    k_vals = shift * dx    
    valid_k = (k_vals >= 0) & (k_vals <= 2)
    valid_corrs = cross[valid_k]
    return np.max(valid_corrs) * dx
