import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Default construction: uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    # dx for discretization
    dx = 2.0 / n_points
    
    # Helper to compute overlap
    def compute_overlap(h):
        # Correlate h(x) and 1 - h(x+k) for all shifts k
        conv = correlate(h, 1 - h, mode='full')
        # Extract the valid overlaps (shifts from -n+1 to n-1)
        overlaps = conv[n_points - 1:] * dx
        return np.max(overlaps)
    
    def project_h(h):
        # Clip values to [0, 1]
        h = np.clip(h, 0, 1)
        # Adjust to satisfy sum(h) = n_points / 2
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Initial evaluation
    c5_bound = compute_overlap(h)
    best_c5 = c5_bound
    best_h = h.copy()
    
    # Learning rate and params
    eta = 0.01
    patience = 0
    last_improvement = 0
    
    # Maximum iterations (projected to fit in budget)
    max_iter = 6000
    iter_count = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 0.1:
        iter_count += 1
        
        # Compute overlaps for all lags
        conv = correlate(h, 1 - h, mode='full')
        overlaps = conv[n_points - 1:] * dx
        
        # Find the worst lag
        k_worst = np.argmax(overlaps)
        k = k_worst - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros(n_points)
        for i in range(n_points):
            if i + k >= 0 and i + k < n_points:
                grad[i] = (1 - h[i + k]) * dx
            if i - k >= 0 and i - k < n_points:
                grad[i] -= h[i - k] * dx
        
        # Update h
        h_new = h - eta * grad
        
        # Project into feasible set
        h_new = project_h(h_new)
        
        # Compute new overlap
        new_c5 = compute_overlap(h_new)
        
        # Accept if better
        if new_c5 < c5_bound:
            h = h_new
            c5_bound = new_c5
            if c5_bound < best_c5:
                best_c5 = c5_bound
                best_h = h.copy()
            patience = 0
            last_improvement = iter_count
        else:
            patience += 1
        
        # Adaptive learning rate
        if iter_count % 100 == 0 or patience > 50:
            eta *= 0.95
            
        if patience >= 100:
            break
            
    # Final projection
    best_h = project_h(best_h)
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
