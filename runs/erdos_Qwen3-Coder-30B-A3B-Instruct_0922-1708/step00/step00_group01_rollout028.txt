import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 47
        h = np.ones(n) * 0.5
        # Slightly asymmetric to avoid symmetry issues
        h[10:25] = 0.6
        h[25:35] = 0.4
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Optimization parameters
    eta = 0.1
    decay_every = 1000
    max_iter = 5000
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        corr = correlate(h, 1 - h, mode='full')
        # Only consider meaningful lags
        mid = len(corr) // 2
        lags = list(range(-mid, mid+1))
        # Only take positive lags for our problem
        valid_lags = [lag for lag in lags if 0 <= lag < len(corr)]
        
        # Find dominant lags (max values)
        max_val = -1
        dominant_lags = []
        for lag in valid_lags:
            val = corr[lag + mid] * dx
            if val > max_val:
                max_val = val
                dominant_lags = [lag]
            elif val == max_val:
                dominant_lags.append(lag)
        
        # Sample a dominant lag
        if not dominant_lags:
            continue
        chosen_lag = np.random.choice(dominant_lags)
        
        # Compute gradient approximation via finite differences
        eps = 1e-4
        grad = np.zeros(n_points)
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] = min(1.0, h_plus[j] + eps)
            h_minus = h.copy()
            h_minus[j] = max(0.0, h_minus[j] - eps)
            
            # Recompute overlaps with modified h[j]
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
            
            # Find max overlap for both
            max_plus = max([corr_plus[lag + mid] * dx for lag in valid_lags])
            max_minus = max([corr_minus[lag + mid] * dx for lag in valid_lags])
            
            grad[j] = (max_plus - max_minus) / (2 * eps)
        
        # Update h
        h = h - eta * grad
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * target_sum / h_sum
            
        # Track best solution
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Anneal learning rate
        if (i + 1) % decay_every == 0:
            eta *= 0.5
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = correlate(h, 1 - h, mode='full')
    mid = len(corr) // 2
    # Only consider valid lag indices (non-negative)
    result = max([corr[i] * dx for i in range(len(corr))])
    return result
