import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for lags 0 to n-1
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap for lag k is corr[n-1+k]
    max_overlap = np.max(corr[n-1:])
    return max_overlap * dx
