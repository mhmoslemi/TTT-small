import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for k from -n+1 to n-1
    corr = correlate(h, 1 - h, mode='full')
    # Only consider valid k values, i.e., k from 0 to n-1
    # That corresponds to indices n-1 to 2*n-2
    corr = corr[n-1:2*n-1]
    return np.max(corr) * dx

def normalize_h(h, target_sum):
    """Normalize h to have sum equal to target_sum"""
    current_sum = np.sum(h)
    if current_sum == 0:
        return np.full_like(h, target_sum / len(h))
    return h * (target_sum / current_sum)

def project_h(h):
    """Project h to [0,1] and normalize to sum to 1"""
    h = np.clip(h, 0, 1)
    return normalize_h(h, 1.0)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 70
    m = 20  # number of particles
    w = 0.7
    c1 = c2 = 2.0
    max_iter = 12000  # estimated based on 0.05s per iteration
    dx = 2.0 / n_points
    target_sum = 1.0
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = project_h(h0)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Create random initial particle
        h0 = np.random.rand(n_points)
        h0 = project_h(h0)
        particles.append(h0.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))

    # Generate remaining particles
    for _ in range(m-1):
        h = h0 + np.random.normal(0, 0.1, n_points)
        h = project_h(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate initial population
    particle_fitness = []
    for h in particles:
        fitness = evaluate_erdos_solution(h)
        particle_fitness.append(fitness)
    
    # Find global best
    best_idx = np.argmin(particle_fitness)
    global_best = particles[best_idx].copy()
    global_best_fitness = particle_fitness[best_idx]
    
    # Iteration tracking
    start_time = time.time()
    it = 0
    no_improve_iter = 0
    best_found_at = 0
    
    while it < max_iter and time.time() - start_time < budget_s:
        for p in range(m):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[p] = (
                w * velocities[p] +
                c1 * r1 * (particles[p] - particles[p]) +
                c2 * r2 * (global_best - particles[p])
            )
            
            # Clip velocity
            velocities[p] = np.clip(velocities[p], -0.1, 0.1)
            
            # Update position
            particles[p] += velocities[p]
            
            # Clip position to [0,1]
            particles[p] = np.clip(particles[p], 0, 1)
            
            # Normalize to have sum = target_sum
            particles[p] = normalize_h(particles[p], target_sum)
            
            # Evaluate fitness
            fitness = evaluate_erdos_solution(particles[p])
            particle_fitness[p] = fitness
            
            # Update personal best
            if fitness < particle_fitness[p]:  # Assumes self is already the best
                pass  # Already set when initialized
            
            # Check if this is better than global best
            if fitness < global_best_fitness:
                global_best = particles[p].copy()
                global_best_fitness = fitness
                no_improve_iter = 0
                best_found_at = it
            else:
                no_improve_iter += 1
        
        # Tournament selection and diversity maintenance
        if it % 100 == 0 and it > 0:
            # Reintroduce diversity if no improvement for a while
            if no_improve_iter > 100:
                for p in range(m):
                    perturbation = np.random.normal(0, 0.01, n_points)
                    particles[p] += perturbation
                    particles[p] = project_h(particles[p])
        
        # Tournament selection
        fitness_array = np.array(particle_fitness)
        sort_indices = np.argsort(fitness_array)
        top_indices = sort_indices[:m//2]
        new_particles = [particles[i] for i in top_indices]
        new_velocities = [velocities[i] for i in top_indices]
        
        # Fill remaining with random solutions
        for i in range(len(new_particles), m):
            h = np.random.rand(n_points)
            h = project_h(h)
            new_particles.append(h)
            new_velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        
        particles = new_particles
        velocities = new_velocities
        
        # Recompute fitness for all
        particle_fitness = []
        for h in particles:
            fitness = evaluate_erdos_solution(h)
            particle_fitness.append(fitness)
        
        best_idx = np.argmin(particle_fitness)
        if particle_fitness[best_idx] < global_best_fitness:
            global_best = particles[best_idx].copy()
            global_best_fitness = particle_fitness[best_idx]
            
        it += 1
    
    # Record final result
    h_recovered = global_best.copy()
    c5_bound = evaluate_erdos_solution(h_recovered)
    return (h_recovered.tolist(), c5_bound, n_points)
