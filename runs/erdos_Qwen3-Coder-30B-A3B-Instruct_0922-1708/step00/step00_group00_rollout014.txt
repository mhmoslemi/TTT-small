import numpy as np

from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for a given step function h"""
    # Compute overlap integral for all shifts
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, n_pop, initial_h):
    """Initialize population with variation around initial solution"""
    population = []
    
    # Start with the initial solution
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Add random variations
    while len(population) < n_pop:
        # Generate random point with correct sum
        h = np.random.rand(n_points)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.mean(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def mutate_individual(h, mutation_rate):
    """Mutate a single individual"""
    h = h.copy()
    n_points = len(h)
    
    # Mutate random elements
    n_mutations = int(mutation_rate * n_points)
    indices = np.random.choice(n_points, n_mutations, replace=False)
    
    for i in indices:
        # Add small random perturbation
        h[i] += np.random.normal(0, 0.05)
        # Clip to [0,1]
        h[i] = np.clip(h[i], 0, 1)
    
    # Renormalize to maintain sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    
    return h

def crossover(parent1, parent2):

    crossover_point = np.random.randint(1, n_points - 1)
    
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    # Renormalize both offspring
    for offspring in [offspring1, offspring2]:
        current_sum = np.sum(offspring)
        target_sum = n_points / 2
        if current_sum > 0:
            offspring *= target_sum / current_sum
            offspring = np.clip(offspring, 0, 1)
    
    return offspring1, offspring2

        competitor_index = np.random.randint(len(population))
