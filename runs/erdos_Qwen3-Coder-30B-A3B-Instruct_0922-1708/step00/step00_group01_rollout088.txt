import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 47
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert len(h) == n_points
    else:
        # Initialize with symmetric plateau
        h = np.ones(n_points) * 0.5
        # Ensure it sums to 1
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # Cross-correlation with (1 - h)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Take maximum and scale by dx
        return np.max(corr) * dx
    
    def feasibility(h_vec):
        return np.all((h_vec >= 0) & (h_vec <= 1)) and abs(np.sum(h_vec) * dx - 1.0) < 1e-10
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        total = np.sum(h_vec)
        if abs(total * dx - 1.0) > 1e-10:
            h_vec = h_vec * (n_points / 2) / total
        return np.clip(h_vec, 0, 1)
    
    current_h = h.copy()
    current_C5 = compute_C5(current_h)
    
    best_h = current_h.copy()
    best_C5 = current_C5
    
    # GA parameters
    pop_size = 100
    tournament_size = 20
    crossover_rate = 0.8
    mutation_rate = 0.1
    max_generations = 1000
    noise_scale = 0.05
    simplex_size = 10
    
    def create_population():
        pop = []
        for _ in range(pop_size):
            if np.random.rand() < 0.5 and 'initial_h_values' in globals():
                # Start near initial solution
                new_h = current_h + np.random.normal(0, noise_scale, n_points)
            else:
                # Random starting point
                new_h = np.random.rand(n_points) * 0.5 + 0.25
            new_h = project_to_feasible(new_h)
            pop.append(new_h)
        return pop
    
    def selection(pop, fitnesses):
        indices = np.argsort(fitnesses)
        selected = []
        for _ in range(tournament_size):
            contestant = np.random.choice(indices[:pop_size // 2])
            selected.append(pop[contestant])
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        # Uniform crossover
        mask = np.random.rand(n_points) < 0.5
        child1 = np.where(mask, parent1, parent2)
        child2 = np.where(mask, parent2, parent1)
        return child1, child2
    
    def mutate(h_vec, rate):
        if np.random.rand() > rate:
            return h_vec
        noise = np.random.normal(0, noise_scale, n_points)
        mutated = h_vec + noise
        return project_to_feasible(mutated)
    
    def refine_with_nelder_mead(pop, fitnesses):
        indices = np.argsort(fitnesses)[:simplex_size]
        simplex_points = [pop[i] for i in indices]
        # Convert to list of column vectors
        def obj_h(h_vec):
            return compute_C5(h_vec)
        # Use nelder-mead on a subset of good candidates
        best_point = simplex_points[0]
        best_val = fitnesses[indices[0]]
        for point in simplex_points[:2]:
            # Semi-random nelder-mead; we just accept points that improve
            new_point = point + np.random.normal(0, 0.01, n_points)
            new_point = project_to_feasible(new_point)
            new_val = compute_C5(new_point)
            if new_val < best_val:
                best_val = new_val
                best_point = new_point
        return best_point, best_val
    
    # Main loop
    for gen in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        population = create_population()
        fitnesses = [compute_C5(p) for p in population]
        
        # Selection and reproduction
        selected = selection(population, fitnesses)
        offspring = []
        for i in range(0, len(selected), 2):
            if i+1 < len(selected):
                child1, child2 = crossover(selected[i], selected[i+1])
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                offspring.extend([child1, child2])
            else:
                off = mutate(selected[i], mutation_rate)
                offspring.append(off)
        
        # Evaluate offspring
        offspring_fitness = [compute_C5(p) for p in offspring]
        
        # Replace worst in population with offspring
        pop_with_offspring = population + offspring
        pop_fitness = fitnesses + offspring_fitness
        sorted_idx = np.argsort(pop_fitness)
        population = [pop_with_offspring[i] for i in sorted_idx[:pop_size]]
        fitnesses = [pop_fitness[i] for i in sorted_idx[:pop_size]]
        
        # Refinement with Nelder-Mead
        best_point, best_val = refine_with_nelder_mead(population, fitnesses)
        if best_val < best_C5:
            best_C5 = best_val
            best_h = best_point.copy()
        
        # Early termination by budget
        if time.time() - start_time >= budget_s:
            break
    
    # Final check of best solution
    final_C5 = compute_C5(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
        best_h = best_h.copy()
        
    # Ensure it satisfies constraints
    best_h = project_to_feasible(best_h)
    assert feasibility(best_h)
    
    return (best_h.tolist(), best_C5, n_points)
