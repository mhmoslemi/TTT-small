import numpy as np
from scipy import signal
import time
import random

def evaluate_solution(h, dx):
    """Evaluate the maximum overlap for given h"""
    # Compute correlation with 1-h
    corr = signal.correlate(h, 1 - h, mode='full')
    # Convert to actual overlap values
    overlaps = corr * dx
    # The overlap is at indices from -n+1 to n-1, but we only care about the maximum
    # The values are indexed starting from n-1 (the middle section)
    max_overlap = np.max(overlaps)
    return max_overlap

def enforce_constraints(h, n_points, dx):
    """Ensure h stays within [0,1] and sum(h) * dx = 1"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to make sum equal to n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-6:
        # Simple approach: scale or shift
        scale = target_sum / current_sum
        h = h * scale
        # Ensure bounds again after scaling
        h = np.clip(h, 0, 1)
    return h

def create_initial_population(n_points, population_size, dx, initial_h=None):
    """Create initial population making sure constraints are held"""
    population = []
    
    # If initial values given, use them
    if initial_h is not None and len(initial_h) == n_points:
        h0 = enforce_constraints(np.array(initial_h), n_points, dx)
        population.append(h0)
    
    # Generate random individuals 
    remaining = population_size - len(population)
    for _ in range(remaining):
        h = np.random.random(n_points)
        h = enforce_constraints(h, n_points, dx)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Perform tournament selection"""
    selected = []
    n_pop = len(population)
    for _ in range(n_pop):
        # Select tournament participants
        idxs = np.random.choice(n_pop, tournament_size)
        # Find the best one among them
        best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
        selected.append(population[best_idx])
    return selected

def crossover(parent1, parent2):
    """Perform single-point crossover"""
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.05):
    """Mutate an individual"""
    n = len(individual)
    mutated = individual.copy()
    
    for i in range(n):
        if random.random() < mutation_rate:
            # Add small random change
            mutated[i] += (np.random.random() - 0.5) * 0.2
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    total_time_start = time.time()
    
    # Parameters
    n_points = 45  # Matches the initial construction size
    dx = 2.0 / n_points
    population_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = population_size // 5  # Top 20%
    max_generations = 120  # Ensuring budget is respected
            
    # Create initial population
    initial_h = initial_h_values if 'initial_h_values' in globals() else None
    population = create_initial_population(n_points, population_size, dx, initial_h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Main GA loop
    for generation in range(max_generations):
        if time.time() - total_time_start > budget_s - 1.0:
            break
        
        # Evaluate fitness
        fitnesses = [evaluate_solution(h, dx) for h in population]
        
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elite = [population[i] for i in elite_indices]
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Crossover and mutation
        new_population = elite[:]  # Start with elites
        
        while len(new_population) < population_size:
            # Select parents
            parent1 = random.choice(selected)
            parent2 = random.choice(selected)
            
            # Crossover
            if random.random() < crossover_prob:
                offspring1, offspring2 = crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
                
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Enforce constraints
            offspring1 = enforce_constraints(offspring1, n_points, dx)
            offspring2 = enforce_constraints(offspring2, n_points, dx)
            
            new_population.append(offspring1)
            if len(new_population) < population_size:
                new_population.append(offspring2)
                
        population = new_population
        
        # Brute-force re-evaluate the best inside this generation
        # to ensure it's still the best
        try:
            current_best_eval = evaluate_solution(best_h, dx)
            if current_best_eval < best_c5:
                best_c5 = current_best_eval
        except:
            pass
    
    # Final evaluation of best solution found
    if best_h is not None:
        final_c5 = evaluate_solution(best_h, dx)
        # Make sure it's properly constrained
        best_h = enforce_constraints(best_h, n_points, dx)
        best_c5 = evaluate_solution(best_h, dx)
        final_c5 = best_c5
    else:
        # Fallback to original method if something went awry
        final_c5 = float('inf')
        
    # Return optimal h and c5
    return (best_h, final_c5, n_points)
