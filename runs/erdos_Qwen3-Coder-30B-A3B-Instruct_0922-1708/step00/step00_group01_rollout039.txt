import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral for all possible lags
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_to_feasible(h, n_points, dx):
    # Ensure h is within [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h) * dx
    if current_sum != 1.0:
        h = h * (1.0 / current_sum)
    return h

def fitness_function(h, n_points, dx):
    h = project_to_feasible(h, n_points, dx)
    return evaluate_c5(h, dx)
