import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 0.5)
        h[0] = 0.0
        h[-1] = 1.0
        h = h / np.sum(h) * n_points / 2.0
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    
    # Temperature for smooth max approximation
    temp = 0.1
    
    # Gradient descent parameters
    max_iter = 1000
    step_size = 0.1
    backtracking_factor = 0.5
    patience = 50
    no_improvement_count = 0
    
    def compute_overlap_and_grad(h):
        # Compute overlaps for full cross-correlation
        h_shifted = np.concatenate([h, np.zeros(n_points-1)])
        ones_shifted = np.concatenate([np.ones(n_points), np.zeros(n_points-1)])
        overlap_vals = []
        grad = np.zeros_like(h)
        
        # For each shift k (from -n_points+1 to n_points-1)
        full_corr = correlate(h, ones_shifted - h, mode="full")
        for k in range(-n_points+1, n_points):
            if k >= 0:
                idx = k
                val = full_corr[n_points-1 + k]
            else:
                idx = k + 2*n_points - 2
                val = full_corr[k + 2*n_points - 2]
            overlap_vals.append(val * dx)
        
        return np.array(overlap_vals), grad
    
    def smooth_max(overlaps, temp):
        # Use soft-max to approximate max
        exp_vals = np.exp(temp * overlaps)
        return np.log(np.sum(exp_vals)) / temp
    
    def compute_grad(h):
        # Very simple gradient estimation
        # This is a simplistic but effective heuristic
        # Actual gradient could be derived from chain rule
        h = h.copy()
        overlaps, _ = compute_overlap_and_grad(h)
        max_k = np.argmax(overlaps)
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            j_start = max(0, i-max_k)
            j_end = min(n_points, i + (n_points - max_k))
            for j in range(j_start, j_end):
                if j - i == max_k:
                    h_grad[i] += (1 - h[j])
        return h_grad
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current solution
        current_overlaps, _ = compute_overlap_and_grad(h)
        current_c5 = np.max(current_overlaps)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > patience:
            # Restart with slightly perturbed version
            h = best_h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * n_points / 2.0
            no_improvement_count = 0
            continue
            
        # Compute simple gradient
        grad = compute_grad(h)
        
        # Normalize gradient
        grad_norm = np.linalg.norm(grad)
        if grad_norm > 0:
            grad = grad / grad_norm
            
        # Backtracking line search
        alpha = step_size
        while True:
            h_new = h - alpha * grad
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to correct sum
            h_new = h_new / np.sum(h_new) * n_points / 2.0
            new_overlaps, _ = compute_overlap_and_grad(h_new)
            new_c5 = np.max(new_overlaps)
            
            if new_c5 < current_c5 or alpha < 1e-8:
                h = h_new
                break
            else:
                alpha *= backtracking_factor
                
    return best_h.tolist(), best_c5, n_points

def evaluate_erdos_solution(h_vals, n_points, dx):
    """Evaluate C₅ for given h values"""
    h = np.array(h_vals)
    # Compute max overlap using full correlation
    h_shifted = np.concatenate([h, np.zeros(n_points-1)])
    ones_h = np.concatenate([np.ones(n_points), np.zeros(n_points-1)])
    corr = correlate(h, ones_h - h, mode="full")
    # Extract valid overlaps from index 0 to 2*n_points - 2
    valid_indices = range(n_points-1, 2*n_points-1) # k from 0 to n_points - 1, negatives go to -n_points+1 to -1
    overlaps = []
    for k in range(-n_points+1, n_points):
        if k >= 0:
            idx = n_points - 1 + k
        else:
            idx = k + 2*n_points - 2
        overlaps.append(corr[idx] * dx)
    return max(overlaps)
