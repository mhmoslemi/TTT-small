import numpy as np
from cvxpy import *
import time
import math

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    overlaps = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        init_h = initial_h_values
        n_points = len(init_h)
    except:
        # Default construction
        n_points = 70
        init_h = np.ones(n_points) * 0.5
        init_h[0] = 0.4
        init_h[1] = 0.6
        init_h[2] = 0.5
        init_h[3] = 0.5
        init_h[4] = 0.5

    best_h = init_h.copy()
    best_C = evaluate_erdos_solution(best_h, n_points)
    
    start_time = time.time()
    max_time = start_time + budget_s
    
    # Try multiple random restarts
    for restart in range(5):
        if time.time() > max_time:
            break
            
        # Initialize h with small random perturbation
        h = init_h.copy()
        if restart > 0:
            pert = np.random.normal(0, 0.01, n_points)
            h += pert
            # Clip and normalize
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Set up the optimization
        h_var = Variable(n_points)
        C = Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1]
        constraints.append(sum(h_var) == n_points / 2)
        
        # Overlap constraints
        dx = 2.0 / n_points
        # All overlaps with up to n_points lags
        max_k = n_points - 1
        for k in range(-max_k, max_k + 1):
            if k == 0:
                continue
            # Compute the lagged convolution manually
            # i from 0 to n_points-1
            # For lag k, sum h[i] * (1 - h[i+k]) where valid
            total = 0
            if k > 0:
                # lag is positive: h[i+k] is to the right of h[i]
                for i in range(n_points - k):
                    total += h_var[i] * (1 - h_var[i + k])
                total *= dx
            else:
                # lag is negative: h[i+k] is to the left of h[i]
                for i in range(-k, n_points):
                    total += h_var[i] * (1 - h_var[i + k])
                total *= dx
            constraints.append(total <= C)
        
        # Objective
        prob = Problem(Minimize(C), constraints)
        
        try:
            # Attempt to solve with multiple solvers
            prob.solve(solver=ECOS, verbose=False, max_iters=500)
            
            if prob.status in [OPTIMAL, OPTIMAL_INACCURATE]:
                C_val = float(C.value)
                h_sol = h_var.value
                
                # Ensure feasibility
                if np.all(h_sol >= 0) and np.all(h_sol <= 1):
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    C_eval = evaluate_erdos_solution(h_sol, n_points)
                    if C_eval < best_C:
                        best_C = C_eval
                        best_h = h_sol.copy()
                        
        except Exception:
            # If optimization fails, proceed with current h if still viable
            pass

    
    # Final perturbation & check
    final_h = best_h.copy()
    if len(final_h) != n_points:
        n_points = len(final_h)
    final_C = evaluate_erdos_solution(final_h, n_points)
    
    # Validation
    if np.min(final_h) < 0 or np.max(final_h) > 1:
        # Reinitialize with proper bounds
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        final_C = evaluate_erdos_solution(final_h, n_points)
        
    return (final_h.tolist(), float(final_C), n_points)
