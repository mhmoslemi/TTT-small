import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Fourier-based approach with m modes
    m = min(10, n_points//2)  # Number of cosine modes
    a = np.random.rand(m) * 0.5 + 0.25 # Initialize coefficients
    a0 = 0.5  # Integral constraint ensures a0 = 0.5
    
    # Convert from a to h(x)
    def a_to_h(a_vec):
        x = np.linspace(0, 2, n_points)
        result = a0 * np.ones_like(x)
        for i in range(m):
            result += a_vec[i] * np.cos(np.pi * i * x / 2)
        return np.clip(result, 0, 1)
    
    # Normalize to integral constraint
    def normalize_h(h):
        integral = np.sum(h) * 2.0 / n_points
        if integral > 0:
            h = h * (n_points / 2) / integral
        return np.clip(h, 0, 1)
    
    # Compute C5 value
    def compute_C5(h):
        # Use numpy correlate for overlap computation
        h_flip = np.flip(h)
        correlations = np.correlate(h, 1-h_flip, mode="full")
        # Correlations are indexed from -n+1 to n-1, but we need 0 to 2*n-2.
        # Take the first n points (shifts from 0 to n-1)
        shifts = correlations[n_points-1:] 
        # Multiply by dx = 2/n_points
        dx = 2.0 / n_points
        C5 = np.max(shifts) * dx
        return C5
    
    # Gradient approximation
    def finite_diff_grad(a_vec, h_func, f_func, eps=1e-4):
        grad = np.zeros_like(a_vec)
        f0 = f_func(h_func(a_vec))
        for i in range(len(a_vec)):
            a_plus = a_vec.copy()
            a_plus[i] += eps
            f_plus = f_func(h_func(a_plus))
            grad[i] = (f_plus - f0) / eps
        return grad
    
    best_C5 = float('inf')
    best_h = None
    
    # Optimization loop
    max_iter = 100000
    iter_count = 0
    last_improvement = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Construct h from current a
        h = a_to_h(a)
        h = normalize_h(h)
        
        # Compute current C5
        C5 = compute_C5(h)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improvement = iter_count
            
        # Compute gradients
        grad = finite_diff_grad(a, a_to_h, compute_C5, eps=1e-3)
        
        # Update coefficients with gradient descent
        a -= lr * grad
        # Clip coefficients to [0, 1]
        a = np.clip(a, 0, 1)
        
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count - last_improvement > 100:
            lr *= 0.9
            last_improvement = iter_count
            
    if best_h is None:
        best_h = h0
        best_C5 = compute_C5(best_h)
        
    return (best_h, best_C5, n_points)
