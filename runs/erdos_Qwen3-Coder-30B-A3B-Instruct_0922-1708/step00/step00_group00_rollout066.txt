import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute the full cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # Apply dx to convert from discrete sum to integral
    overlaps = corr * dx
    # Return the maximum overlap
    return np.max(overlaps)

def ensure_constraints(h, n_points, dx):
    """Ensure h satisfies constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    # Clip h to [0, 1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-6:
        # Scale h to meet target sum
        h = h * target_sum / current_sum
    return h

def generate_initial_population(n_points, pop_size, initial_h=None):
    """Generate initial population for genetic algorithm."""
    population = []
    if initial_h is not None:
        # Add the initial solution
        population.append(ensure_constraints(initial_h.copy(), n_points, 2.0 / n_points))
        # Fill the rest with random individuals
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = ensure_constraints(h, n_points, 2.0 / n_points)
            population.append(h)
    else:
        # All random individuals
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = ensure_constraints(h, n_points, 2.0 / n_points)
            population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate an individual."""
    h = h.copy()
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    for i in indices:
        # Add small random change
        delta = np.random.normal(0, 0.05)
        h[i] += delta
    return ensure_constraints(h, len(h), 2.0 / len(h))

    """Run the genetic algorithm to find h minimizing C₅."""
    pop_size = 100
    max_generations = 120
    dx = 2.0 / n_points
    
    # Initialize population
    population = generate_initial_population(n_points, pop_size, initial_h)
    
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness of each individual
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Create new population
        new_population = []
        
        # Elitism: keep best 20%
        elite_size = max(1, pop_size // 5)
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Selection, crossover, and mutation for rest
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
    return best_h, best_c5
