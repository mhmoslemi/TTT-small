import numpy as np
from scipy import optimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale so sum equals n_points / 2
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def create_initial_h(n_points):
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    # Create sinusoidal shape
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(2 * np.pi * x))
    h = project_h(h, n_points)
    return h

def crossover(parent1, parent2):
    # Simple single-point crossover
    n = len(parent1)
    if n <= 1:
        return parent1.copy(), parent2.copy()
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate):
    # Randomly mutate some elements
    n = len(h)
    mask = np.random.rand(n) < mutation_rate
    # For mutation, perturb values slightly
    delta = np.random.normal(0, 0.1, n) * mask
    h = h + delta
    return project_h(h, n)

def genetic_algorithm(h0, n_points, budget_s):
    # Parameters
    pop_size = 50
    generations = 10000
    mutation_rate = 0.1
    crossover_rate = 0.7
    elite_ratio = 0.2
    
    # Start timing
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        h = mutate(h0.copy(), mutation_rate)
        population.append(h)
    
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h)
    
    for gen in range(generations):
        if elapsed() > 0.7 * budget_s:
            break
            
        # Evaluate all individuals
        fitness = [evaluate_c5(h) for h in population]
        
        # Elitism
        sorted_indices = np.argsort(fitness)
        elite_count = int(elite_ratio * pop_size)
        elite = [population[i] for i in sorted_indices[:elite_count]]
        
        # Keep track of best so far
        current_best = population[sorted_indices[0]]
        current_c5 = fitness[sorted_indices[0]]
        if current_c5 < best_c5:
            best_h = current_best.copy()
            best_c5 = current_c5
        
        # Create new population
        new_population = elite.copy()
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(population, 2, replace=False)
            
            if np.random.rand() < crossover_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            child1 = mutate(child1, mutation_rate)
            child2 = mutate(child2, mutation_rate)
            
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:pop_size]
        
    return best_h, best_c5

def gradient_descent(h0, n_points, budget_s):
    start_time = time.time()
    elapsed = lambda: time.time() - start_time
    
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    max_iter = 10000
    lr = 0.01
    patience = 1000
    no_improvement_count = 0
    
    for i in range(max_iter):
        if elapsed() > 0.9 * budget_s:
            break
            
        # Compute correlation
        corr_full = np.correlate(h, 1 - h, mode="full")
        
        # Find kernel that maximizes the overlap, i.e., max of correlation 
        shift_idx = np.argmax(corr_full)
        max_shift = shift_idx - len(h) + 1  # shift in range [-len+1, len-1]
        
        # In the correlation function, for shift k (when shift is negative)
        # the correlation contributes terms from h[i] * (1 - h[i-k])
        # We compute the gradient of corr[k] w.r.t. h[i]
        # Let's compute the gradient over vector h analytically:
        # grad_corr_k = [d/dh[i] corr_k] = patterns of 1s and -1s in indices
        
        grad_corr = np.zeros_like(h)
        for i in range(len(h)):
            if 0 <= i + max_shift < len(h):
                grad_corr[i] += -1  # derivative of h[i] with respect to corr_k
            if 0 <= i - max_shift < len(h):
                grad_corr[i] += 1   # derivative of (1-h[i]) w.rt. corr_k
        
        # Gradient descent step
        h_new = h - lr * grad_corr
        h_new = project_h(h_new, n_points)
        
        new_c5 = evaluate_c5(h_new)
        if new_c5 < best_c5:
            h = h_new
            best_h = h.copy()
            best_c5 = new_c5
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            if no_improvement_count > patience:
                lr *= 0.5
                no_improvement_count = 0
                
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial h if available, otherwise create it
    n_points = 45
    h0 = create_initial_h(n_points)
    
    # Phase 1: Genetic Algorithm
    h_ga, c5_ga = genetic_algorithm(h0, n_points, budget_s)
    
    # Phase 2: Gradient Descent refinement
    h_final, c5_final = gradient_descent(h_ga, n_points, budget_s)
    
    # Final check for best
    final_c5 = evaluate_c5(h_final)
    if final_c5 < c5_final:
        c5_final = final_c5
    
    return (h_final, c5_final, n_points)
