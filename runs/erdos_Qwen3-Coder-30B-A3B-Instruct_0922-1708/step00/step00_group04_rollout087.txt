def evaluate_erdos_solution(h):
    # Evaluates C5 = max_k ∫ h(x)(1 - h(x+k)) dx
    n = len(h)
    dx = 2.0 / n
    # Use cross correlation
    # Transform to desired format
    h = np.asarray(h)
    conv = np.correlate(h, 1 - h, mode="full")
    # We care about the max, but the calculation is a bit tricky
    # hook to python's Signal processing
    # shift conv it to middle
    max_val = np.max(conv) * dx
    return max_val
