import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    try:
        h_init = initial_h_values
    except NameError:
        # Create initial step function
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
    
    # Normalize to correct integral
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier optimization parameters
    m = 10  # number of cosine modes (<= m/2 used for symmetry)
    
    # Initialize coefficients
    a = np.random.rand(m) * 0.5  # coefficients in [0, 0.5]
    a[0] = 0.5  # ensures integral = 1 after normalization
    
    best_h = h_init.copy()
    best_C = evaluate_erdos_solution(best_h, n_points, dx)
    
    ITERATIONS_WITHOUT_IMPROVEMENT = 0
    MAX_NO_IMPROVEMENT = 100
    learning_rate = 0.01
    
    def h_from_a(a_coeffs):
        # create h from cosine coefficients
        x = np.linspace(0, 2, n_points)
        h = np.full(n_points, a_coeffs[0])  # a0
        for k in range(1, min(m, len(a_coeffs))):
            h += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        # Ensure bounds
        h = np.clip(h, 0, 1)
        # Re-scale to correct integral
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_C(h):
        # compute overlap
        h_rev = np.flip(h)
        correlations = np.correlate(h, h_rev, mode="full")
        result = np.max(correlations) * dx
        # Also compute correlation with 1-h
        one_minus_h = 1.0 - h
        corr2 = np.correlate(h, one_minus_h, mode="full")
        return max(np.max(np.abs(corr2)) * dx, result)
    
    # Gradient step
    def compute_gradient(a_coeffs, k_star):
        eps = 1e-5
        g = np.zeros_like(a_coeffs)
        base_h = h_from_a(a_coeffs)
        base_C = compute_C(base_h)
        for i in range(len(a_coeffs)):
            a_pert = a_coeffs.copy()
            a_pert[i] += eps
            h_pert = h_from_a(a_pert)
            C_pert = compute_C(h_pert)
            g[i] = (C_pert - base_C) / eps
        return g

    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and ITERATIONS_WITHOUT_IMPROVEMENT < MAX_NO_IMPROVEMENT:
        iter_count += 1
        # Project a into h:
        h = h_from_a(a)
        
        # Compute overlap for all shifts
        h_rev = np.flip(h)
        corr = np.correlate(h, h_rev, mode="full")
        corr2 = np.correlate(h, 1.0 - h, mode="full")
        corr_all = np.abs(np.concatenate([corr, corr2]))
        
        # Find max correlation shift
        k_star = np.argmax(corr_all)
        max_corr = corr_all[k_star]
        
        # Update coefficients
        grad = compute_gradient(a, k_star)
        a = a - learning_rate * grad
        
        # Clamp to bounds
        a = np.clip(a, 0., 1.)
        
        # Re-check solution quality
        new_h = h_from_a(a)
        new_C = compute_C(new_h)
        
        if new_C < best_C:
            best_C = new_C
            best_h = new_h.copy()
            ITERATIONS_WITHOUT_IMPROVEMENT = 0
        else:
            ITERATIONS_WITHOUT_IMPROVEMENT += 1
            
        if iter_count % 10 == 0:
            # Reduce learning rate
            learning_rate *= 0.99
        
    final_h = best_h
    final_C = evaluate_erdos_solution(final_h, n_points, dx)
    
    return (final_h, final_C, n_points)
