import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 70
    dx = 2.0 / n_points
    initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.ones(n_points) * 0.5
    initial_h = np.clip(initial_h, 0, 1)
    
    # Normalize to sum to n_points / 2
    h = initial_h.copy()
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Evaluate initial C5
    def compute_C5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Gradually perturb and improve
    last_improvement = 0
    iterations_without_improvement = 0
    max_iter_without_improvement = 100
    num_trials_per_iter = 100
    
    while time.time() - start_time < budget_s - 0.1 and iterations_without_improvement < max_iter_without_improvement:
        for _ in range(num_trials_per_iter):
            if time.time() - start_time >= budget_s - 0.1:
                break
                
            # Make small random perturbations
            h_trial = h.copy()
            num_to_modify = max(1, n_points // 20)  # Modify ~5% of points
            indices = np.random.choice(n_points, size=num_to_modify, replace=False)
            delta = 0.05
            
            for idx in indices:
                # Random step
                step = np.random.uniform(-delta, delta)
                h_trial[idx] += step
                
            # Ensure bounds
            h_trial = np.clip(h_trial, 0, 1)
            
            # Adjust to maintain integral constraint
            current_sum = np.sum(h_trial)
            adjustment_factor = (n_points / 2) / current_sum
            h_trial = h_trial * adjustment_factor
            h_trial = np.clip(h_trial, 0, 1)
            
            # Smooth slightly to retain function structure
            window_size = min(5, n_points // 10)
            if window_size > 1:
                kernel = np.ones(window_size) / window_size
                h_trial = np.convolve(h_trial, kernel, mode='same')
                # Re-adjust to maintain constraint after smoothing
                h_trial = np.clip(h_trial, 0, 1)
                h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
                
            # Compute new C5
            new_C5 = compute_C5(h_trial)
            
            if new_C5 < best_C5:
                best_h = h_trial.copy()
                best_C5 = new_C5
                iterations_without_improvement = 0
            else:
                iterations_without_improvement += 1
                
        h = best_h.copy()
        
    return (best_h.tolist(), best_C5, n_points)
