    target_sum = n_points / 2.0

    h = h * target_sum / np.sum(h)
    
    # Hyperparameters
    learning_rate = 0.1
    anneal_period = 1000
    max_iter = 5000
    
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:

        # Compute all overlaps
        overlaps = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_lag_idx]
        
        # Gradient w.r.t. h
        # Only compute for positions that affect max overlap
        grad = np.zeros(n_points)
        max_lag = max_lag_idx - (n_points - 1)
        
        for j in range(n_points):
            # Derivative of h[i] * (1 - h[i+k]) w.r.t. h[i]
            # If i + k >= 0 and i + k < n_points
            k = max_lag
            if 0 <= j + k < n_points:
                grad[j] += 1 - h[j + k]
            # The derivative of h[i] * (1 - h[i+k]) is 1-h[i+k] for i when k = i-j
            # In a correlation, we're looking at h[i] * (1-h[i+k]) where k 
            # is the lag. We compute how much increasing h[i] affects the
            # overlap at that specific lag.
            
        # Gradient trick to focus on max_lag only
        # Use finite differences per dimensional update
        grad = np.zeros(n_points)
        for j in range(n_points):
            k = max_lag
            if 0 <= j + k < n_points:
                grad[j] = 1 - h[j + k]
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to [0,1] constraints
        h = np.clip(h, 0, 1)
        
        # Rescale to sum = target_sum
        h = h * target_sum / np.sum(h)
        
        # Update best solution
        current_C5 = compute_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            
        # Anneal learning rate
        if i > 0 and i % anneal_period == 0:
            learning_rate *= 0.5
            
    # Final check
    final_C5 = compute_C5(best_h, dx)
    
    return (best_h.tolist(), final_C5, n_points)

def compute_C5(h, dx):
    # Compute overlaps
    overlaps = np.correlate(h, 1 - h, mode='full')
    return np.max(overlaps) * dx
