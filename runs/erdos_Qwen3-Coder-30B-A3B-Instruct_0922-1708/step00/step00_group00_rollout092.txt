import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx to get the actual overlap values
    overlaps = corr * dx
    # Return the maximum overlap
    return np.max(overlaps)

def enforce_constraints(h, n_points, dx):
    """Enforce constraints: h[i] in [0,1] and sum(h)*dx = 1."""
    # Clip values between 0 and 1
    h = np.clip(h, 0, 1)
    # Normalize so that sum(h)*dx = 1
    h_sum = np.sum(h)
    target_sum = n_points / 2.0
    if h_sum > 0:
        h = h * target_sum / h_sum
    return h

def initialize_population(n_points, population_size, initial_h=None):
    """Initialize population with diverse candidates."""
    population = []
    if initial_h is not None:
        # Use initial_h as one individual
        h = enforce_constraints(initial_h.copy(), n_points, 2.0 / n_points)
        population.append(h)
    
    # Generate the rest randomly
    while len(population) < population_size:
        # Generate random values, normalize, and clip
        h = np.random.rand(n_points)
        h = enforce_constraints(h, n_points, 2.0 / n_points)
        population.append(h)
    
    return population

def selection(population, fitnesses, tournament_size=5):
    """Tournament selection."""
    selected = []
    n = len(population)
    for _ in range(n):
        tournament_indices = np.random.choice(n, tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_rate=0.8):
    """Single-point crossover."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Add small random perturbations."""
    n = len(individual)
    num_mutations = int(n * mutation_rate)
    indices = np.random.choice(n, num_mutations, replace=False)
    for i in indices:
        individual[i] += np.random.normal(0, 0.05)
    # Enforce constraints
    dx = 2.0 / n
    individual = enforce_constraints(individual, n, dx)
    return individual

            child2 = mutate(child2)
