import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # A simple symmetric construction
        mid = n_points // 2
        h[:mid] = np.linspace(0.1, 1.0, mid)
        h[mid:] = np.linspace(1.0, 0.1, n_points - mid)
        h = h * (n_points / 2) / np.sum(h)
    
    # Precompute dx
    dx = 2.0 / n_points
    
    # Optimization parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    tol = 1e-5
    
    start_time = time.time()
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h, dx)
    
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all lags
        overlaps = []
        for k in range(n_points):
            overlap = np.sum(h * np.roll(1-h, k))
            overlaps.append(overlap * dx)
            
        # Smooth approximation of max
        exp_overlaps = np.exp(tau * np.array(overlaps))
        smooth_max = np.log(np.sum(exp_overlaps)) / tau
        
        # Compute gradient
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Gradient contribution from overlap(k)
            contrib = exp_overlaps[k] * (-np.roll(1-h, k)) * dx
            grad += contrib
        
        grad = grad / np.sum(exp_overlaps)
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Update tau and learning rate
        tau *= anneal_factor
        if tau < 1e-6:
            tau = 1e-6
            
        # Check for improved solution
        current_C5 = evaluate_C5(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
    
    return (best_h, best_C5, n_points)

def evaluate_C5(h, dx):
    # Compute all overlaps with full correlation
    corr = correlate(h, 1-h, mode="full")
    # Index of maximum overlap
    max_idx = np.argmax(corr)
    # Max overlap value
    max_overlap = corr[max_idx]
    return max_overlap * dx
