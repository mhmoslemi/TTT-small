import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, noise_scale=0.05):
    """Mutate h with Gaussian noise and project"""
    h_new = h + np.random.normal(0, noise_scale, len(h))
    return project_h(h_new, len(h))

def crossover_h(h1, h2):
    """Blend two h vectors"""
    alpha = np.random.random()
    h_new = alpha * h1 + (1 - alpha) * h2
    return project_h(h_new, len(h1))

def evaluate_fitness(h, dx, n_points):
    """Evaluate fitness (negative C5) with constraint check"""
    h = project_h(h, n_points)
    c5 = evaluate_c5(h, dx)
    return -c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        n_points = len(initial_h)
    else:
        # Fallback to symmetric plateau
        n_points = 47
        x = np.linspace(0, 2, n_points)
        initial_h = np.ones(n_points) * 0.5
        # Create a smooth taper
        mid = n_points // 2
        for i in range(n_points):
            dist = abs(i - mid) / (n_points / 2)
            initial_h[i] = max(0.1, 0.9 * (1 - dist))
        initial_h = project_h(initial_h, n_points)
        
    dx = 2.0 / n_points
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    start_time = time.time()
    iteration = 0
    
    # Run evolution
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        # Generate population 
        pop_size = 100
        population = [best_h.copy()]
        for _ in range(pop_size - 1):
            if np.random.random() < 0.5:
                # Mutate best
                new_h = mutate_h(best_h.copy())
            else:
                # Crossover with random individual
                parent1 = population[np.random.randint(0, len(population))]
                parent2 = population[np.random.randint(0, len(population))]
                new_h = crossover_h(parent1, parent2)
            population.append(new_h)
        
        # Evaluate population (vectorized)
        fitnesses = []
        for h in population:
            fitness = evaluate_fitness(h, dx, n_points)
            fitnesses.append(fitness)
        
        # Select top individuals
        idxs = np.argsort(fitnesses)[:20]
        top_individuals = [population[i] for i in idxs]
        
        # Refine top with Nelder-Mead
        for i, h0 in enumerate(top_individuals[:10]):
            # Use a smaller simplex for exploration
            h_refined = h0.copy()
            res = minimize(lambda x: evaluate_fitness(x, dx, n_points), 
                          h_refined, method='Nelder-Mead', 
                          options={'maxiter': 50, 'disp': False})
            if res.success:
                h_candidate = project_h(res.x, n_points)
                c5_candidate = evaluate_c5(h_candidate, dx)
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
        
        # Update best
        if evaluate_c5(best_h, dx) < best_c5:
            best_c5 = evaluate_c5(best_h, dx)
            
    return (best_h.tolist(), best_c5, n_points)
