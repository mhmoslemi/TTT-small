import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Default setup
        n_points = 70
        h0 = np.array([0.5 * (2.0 / n_points)] * n_points)
    
    # Normalize to sum = n_points / 2
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Define CVXPY variables
    x = cvx.Variable(n_points)
    t = cvx.Variable(1)
    
    # Initialize constraints
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == n_points / 2,
        t >= 0
    ]
    
    # Construct matrix of z_ij = x_i(1 - x_j) constraints
    Z = cvx.Variable((n_points, n_points))
    
    # Add the z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i, j] <= x[i],
                Z[i, j] <= 1 - x[j],
                Z[i, j] >= x[i] + x[j] - 1
            ]
    
    # Compute correlations
    C = []
    for k in range(-n_points+1, n_points):
        indices = np.arange(n_points) - k
        valid = (indices >= 0) & (indices < n_points)
        sum_z = cvx.sum(Z[np.arange(n_points)[valid], indices[valid]])
        C.append(sum_z)
    
    # Add max constraints
    for c in C:
        constraints.append(c <= t)
    
    # Objective
    objective = cvx.Minimize(t)
    
    # Create and solve problem
    prob = cvx.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
    except:
        try:
            prob.solve(solver=cvx.SCS, verbose=False, time_limit=budget_s)
        except:
            pass
    
    # Extract solution
    if x.value is not None:
        h_opt = np.clip(x.value, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    else:
        h_opt = h0.copy()

    c5_bound = evaluate_erdos_solution(h_opt)

    return (h_opt, c5_bound, n_points)
