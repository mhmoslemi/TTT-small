import numpy as np
from scipy.optimize import minimize
import time
import math

def evaluate_c5(h, dx):
    # Cross-correlation of h with (1-h)
    conv = np.correlate(h, 1 - h, mode='full')
    return np.max(conv) * dx

def project_h(h, n_points):
    # Ensure h is feasible: 0 <= h[i] <= 1 and sum(h) = n_points/2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, noise_scale):
    # Add Gaussian noise and project
    h_new = h + np.random.normal(0, noise_scale, len(h))
    return project_h(h_new, len(h))

def crossover_h(h1, h2):
    # Simple arithmetic crossover
    alpha = np.random.random()
    h_new = alpha * h1 + (1 - alpha) * h2
    return project_h(h_new, len(h1))

def genetic_algorithm(initial_h, n_points, dx, population_size=100, 
                     max_generations=1000, mutation_rate=0.1, 
                     crossover_rate=0.8, tournament_size=20,
                     simplex_size=10):
    # Initialize population
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = mutate_h(initial_h, 0.05)
        population.append(h)
    
    best_h = initial_h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    start_time = time.time()
    
    for generation in range(max_generations):
        if time.time() - start_time > 580:  # 20s buffer
            break
        
        # Evaluate all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_c5(h, dx)
            fitnesses.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(population_size, tournament_size)
            tournament_fits = [fitnesses[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fits)]
            selected.append(population[winner_idx])
        
        # Nelder-Mead refinement for top simplex_size individuals
        top_indices = np.argsort(fitnesses)[:simplex_size]
        for idx in top_indices:
            h = population[idx]
            # Refine with Nelder-Mead
            res = minimize(lambda x: evaluate_c5(x, dx), h, method='Nelder-Mead', 
                          options={'maxiter': 100, 'disp': False})
            refined_h = res.x
            refined_h = project_h(refined_h, n_points)
            # Test refined solution
            refined_c5 = evaluate_c5(refined_h, dx)
            if refined_c5 < fitnesses[idx]:
                population[idx] = refined_h
                fitnesses[idx] = refined_c5
                if refined_c5 < best_c5:
                    best_c5 = refined_c5
                    best_h = refined_h.copy()
        
        # Generate new population
        new_population = []
        for _ in range(population_size // 2):
            parent1 = selected[np.random.randint(population_size)]
            parent2 = selected[np.random.randint(population_size)]
            if np.random.random() < crossover_rate:
                child = crossover_h(parent1, parent2)
            else:
                child = parent1.copy()
            if np.random.random() < mutation_rate:
                child = mutate_h(child, 0.05)
            new_population.append(child)
        
        # Replace worst in population with new offspring
        worst_indices = np.argsort(fitnesses)[::-1][:len(new_population)]
        for i, idx in enumerate(worst_indices):
            population[idx] = new_population[i]
    
    return best_h, best_c5
