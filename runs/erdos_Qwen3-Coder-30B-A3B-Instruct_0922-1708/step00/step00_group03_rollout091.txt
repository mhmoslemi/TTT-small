import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using full convolution
    overlaps = np.correlate(h, 1 - h, mode='full')
    # Only take the relevant lags (the ones with valid overlaps)
    # The valid lags are from -n+1 to n-1, but we're only interested in
    # the ones corresponding to the [0,2] interval
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial construction if available

        h = np.ones(n_points) * 0.5
        # Perturb slightly to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        # Ensure it's feasible
        h = np.clip(h, 0, 1)
    
    # Scale to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Precision: we want to go below 0.3807
    target = 0.3807
    
    # Hyperparams
    learning_rate = 0.1
    max_iter = 500
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_overlaps(h):
        # Using direct computation for overlap
        n = len(h)
        dx = 2.0 / n
        overlaps = []
        for k in range(2 * n - 1):
            # Compute the overlap integral for lag k
            # Offset for k is from -n+1 to n-1
            lag = k - (n - 1)
            if lag >= 0:
                # h[i] * (1 - h[i + lag]) for valid i
                overlap = 0.0
                for i in range(n - lag):
                    overlap += h[i] * (1 - h[i + lag])
                # Add dx factor
                overlaps.append(overlap * dx)
            else:
                # Same but shifted
                overlap = 0.0
                for i in range(-lag, n):
                    overlap += h[i] * (1 - h[i + lag])
                overlaps.append(overlap * dx)
        return overlaps
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Now this is rather trivial due to being continuous:
        # we will use scipy.optimize.minimize to get the gradient of C5
        # w.r.t h, so we minimize C5. However directly minimizing C5
        # is hard because of the max.

        # For computational reasons, use above 'compute_overlaps' to get empirical values
        
        # Fast Estimation:
        # compute overlap values using current h
        overlaps = compute_overlaps(h)
        c5 = np.max(overlaps)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        if best_c5 <= target:
            break
            
        # Update gradient w.r.t overlaps:
        # Each overlap_k = sum_i h[i] * (1 - h[i + k]) 
        # Gradient of overlap_k w.r.t. h[j] is:
        # if j == i: (1 - h[i + k]) if i + k < n else 0
        # if j == i + k: -h[i] if i >= 0 else 0
        # and so on...
        # More direct: active gradient update
        n = len(h)
        dx = 2.0 / n
        
        # Get the largest overlap
        max_idx = np.argmax(overlaps)
        max_lag = max_idx - (n-1)
        
        # Compute gradient for this particular overlap
        gradient = np.zeros(n)
        
        # Compute only gradient for one term:
        # For overlap_k, the i-th term is h[i] * (1 - h[i+k]) so:
        # d/dh[i] = 1 - h[i+k] 
        # d/dh[i+k] = -h[i] 
        if max_lag >= 0:
            for i in range(n - max_lag):
                gradient[i] += (1 - h[i + max_lag])
                if i + max_lag < n:
                    gradient[i + max_lag] -= h[i]
        else:
            # max_lag < 0
            for i in range(-max_lag, n):
                gradient[i] += (1 - h[i + max_lag])
                if i + max_lag >= 0:
                    gradient[i + max_lag] -= h[i]

        # Update step using gradient descent
        h = h - learning_rate * gradient * dx
        h = np.clip(h, 0, 1)
        
        # Scale again
        h = h * (n_points / 2) / np.sum(h)
        
        # Drop learning rate every 10 steps without better value
        if i % 10 == 0:
            if i > 0 and np.allclose(best_h, h, atol=1e-5):
                learning_rate = learning_rate * 0.5
                if learning_rate < 1e-6:
                    break

    # Final check with most precise computation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return best_h, final_c5, n_points
