import numpy as np
import time
from scipy.signal import correlate

def evaluate_overlap(h, dx):
    # Compute the overlap integral C5
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset = 20
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = project_h(h_init, n_points)
        particles = [h_init.copy() for _ in range(swarm_size)]
        # Add some noise to create diversity
        for i in range(1, swarm_size):
            noise = np.random.normal(0, 0.05, n_points)
            particles[i] = project_h(particles[0] + noise, n_points)
    else:
        particles = []
        for _ in range(swarm_size):
            h = np.random.random(n_points)
            h = project_h(h, n_points)
            particles.append(h)
    
    # Initialize velocities
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    # Initialize personal bests
    personal_best = [particle.copy() for particle in particles]
    personal_best_values = [evaluate_overlap(particle, dx) for particle in particles]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO loop
    start_time = time.time()
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        iter_count += 1
        
        # Evaluate all particles
        current_values = [evaluate_overlap(particle, dx) for particle in particles]
        
        # Update personal best
        for i in range(swarm_size):
            if current_values[i] < personal_best_values[i]:
                personal_best[i] = particles[i].copy()
                personal_best_values[i] = current_values[i]
                
                # Check for global best
                if current_values[i] < global_best_value:
                    global_best = particles[i].copy()
                    global_best_value = current_values[i]
                    last_improvement = iter_count

        # Update global best if not updated in this iteration
        if iter_count - last_improvement > stagnation_limit:
            # Inject diversity
            reset_count = int(swarm_size * diversity_reset / 100)
            indices = np.random.choice(swarm_size, reset_count, replace=False)
            for idx in indices:
                # Reset with a slightly modified version of global best
                h = global_best + np.random.normal(0, 0.05, n_points)
                particles[idx] = project_h(h, n_points)
                personal_best[idx] = particles[idx].copy()
                personal_best_values[idx] = evaluate_overlap(particles[idx], dx)
                if personal_best_values[idx] < global_best_value:
                    global_best = personal_best[idx].copy()
                    global_best_value = personal_best_values[idx]
                    last_improvement = iter_count

        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.random(2)
            velocities[i] = (
                w * velocities[i] +
                c1 * r1 * (personal_best[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            particles[i] = particles[i] + velocities[i]
            particles[i] = project_h(particles[i], n_points)
            
    return (global_best.tolist(), global_best_value, n_points)
