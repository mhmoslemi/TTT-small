import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        # Create triangular pattern
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2 * x / 2 - 1)
        # Scale to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    max_iter = 1000
    lr = 0.1
    patience = 100
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find worst overlap shift
        overlaps = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(overlaps)
        k_shift = k_max - (n_points - 1)
        
        # Compute current gradient for that shift
        grad = np.zeros_like(h)
        if k_shift >= 0:
            grad[k_shift:] -= h[:-k_shift] if k_shift < n_points else 0
            grad[:-k_shift] += h[k_shift:] if k_shift < n_points else 0
        else:
            k_shift = -k_shift
            grad[k_shift:] += h[:-k_shift] if k_shift < n_points else 0
            grad[:-k_shift] -= h[k_shift:] if k_shift < n_points else 0
        
        # Gradient descent
        h_new = h - lr * grad
        
        # Project h_new to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to satisfy constraint sum(h) = n_points / 2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_C5 = evaluate_erdos_solution(h_new, dx)
        
        if new_C5 < best_C5:
            h = h_new
            best_C5 = new_C5
            no_improve_count = 0
        else:
            no_improve_count += 1
            # Reduce learning rate if no improvement
            lr *= 0.95
            
        if no_improve_count >= patience:
            # Perturb slightly and restart
            h = h + 0.001 * np.random.randn(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            no_improve_count = 0
            lr = 0.1
            
        if best_C5 < 0.3807:
            break
    
    return (best_h.tolist(), best_C5, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute overlap integral for all shifts
    overlaps = np.correlate(h, 1 - h, mode='full')
    # Return maximum overlap scaled by dx
    return np.max(overlaps) * dx
