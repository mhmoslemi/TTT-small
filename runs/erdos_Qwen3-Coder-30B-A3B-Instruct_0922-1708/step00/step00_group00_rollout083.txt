import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Helper functions
    def compute_C5(h):
        h = np.array(h)
        conv = np.correlate(h, 1 - h, mode='full')
        return np.max(conv) * dx
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h = h * scale
        return h
    
    def admm_step(h, rho=1.0):
        # Use gradient-based approach to solve subproblem
        def objective(h_sol):
            overlap = np.correlate(h_sol, 1 - h_sol, mode='full')
            return -np.max(overlap) * dx
        
        # Simple gradient descent or trust-region
        result = minimize(objective, h, method='L-BFGS-B', bounds=[(0, 1)] * n_points)
        h_new = result.x
        h_new = project_h(h_new)
        return h_new
    
    # Main optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(h)
    last_improvement = 0
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        h = admm_step(h)
        C5 = compute_C5(h)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
        
        if last_improvement > 50:
            break
            
    return (best_h.tolist(), best_C5, n_points)
