import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric initial guess
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Create a plateau with tapering ends
        center = n_points // 2
        width = n_points // 3
        for i in range(n_points):
            if i < center - width // 2:
                h[i] = 0.1 + 0.4 * (i / (center - width // 2))
            elif i > center + width // 2:
                h[i] = 0.1 + 0.4 * ((n_points - 1 - i) / (center - width // 2))
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iterations = 3000
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    convergence_tol = 1e-5
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        # Use correlation with flipped (1-h) 
        overlap_values = np.zeros(2 * n_points - 1)
        h1_minus_h = 1.0 - h
        corr = correlate(h, h1_minus_h, mode='full')
        overlap_values = corr * dx
        
        # Extract the central values (actual overlaps)
        central_start = n_points - 1
        overlaps = overlap_values[central_start:central_start + n_points]
        
        # Smooth approximation of max
        c5_smooth = np.log(np.sum(np.exp(tau * overlaps))) / tau
        
        # Compute gradient of smooth approximation
        exp_overlaps = np.exp(tau * overlaps)
        exp_sum = np.sum(exp_overlaps)
        grad = np.zeros(n_points)
        
        # Compute gradient via correlation with shifted versions
        for i in range(n_points):
            g = 0.0
            for k in range(n_points):
                shift = (i - k) % n_points
                g += exp_overlaps[k] * (-h1_minus_h[shift]) * dx
            grad[i] = g / exp_sum
        
        # Update step
        h = h - learning_rate * grad
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Anneal temperature
        tau *= anneal_factor
        
        # Evaluate exact C5
        overlap_exact = overlap_values[central_start:central_start + n_points]
        c5_exact = np.max(overlap_exact)
        
        if c5_exact < best_c5:
            best_c5 = c5_exact
            best_h = h.copy()
        
        # Check for convergence - if improvement is small
        if iteration > 10 and np.abs(c5_smooth - c5_exact) < convergence_tol:
            break
    
    return (best_h.tolist(), float(best_c5), n_points)
