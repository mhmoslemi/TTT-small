import numpy as np
from scipy.optimize import minimize
import time

def compute_overlap(h, dx):
    # Compute the overlap integral using correlation
    corr = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum correlation value
    return np.max(corr) * dx

def project_h(h, dx, n_points):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet constraint sum(h) * dx = 1
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2.0) / h_sum
    return h

def optimize_step(h, dx, n_points, lr=0.01):
    # Evaluate current overlap
    current_overlap = compute_overlap(h, dx)
    
    # Compute finite difference approximation of gradient
    grad = np.zeros_like(h)
    eps = 1e-5
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, dx, n_points)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, dx, n_points)
        
        overlap_plus = compute_overlap(h_plus, dx)
        overlap_minus = compute_overlap(h_minus, dx)
        grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
    
    # Update h using gradient descent
    h_new = h - lr * grad
    
    # Project back to feasible set
    h_new = project_h(h_new, dx, n_points)
    
    return h_new
