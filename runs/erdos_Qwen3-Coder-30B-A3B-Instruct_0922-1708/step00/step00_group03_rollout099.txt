import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h"""
    n_points = len(h)
    dx = 2.0 / n_points
    c5_vals = correlate(h, 1-h, mode='full') * dx
    return np.max(c5_vals)

def initialize_population(n_points, pop_size, initial_h_values):
    """Initialize population with given initial solution"""
    population = []
    if initial_h_values is not None:
        population.append(initial_h_values.copy())
    else:
        population.append(np.full(n_points, 0.5))
    
    while len(population) < pop_size:
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    return population

def mutate(h, mutation_rate, dx):
    """Apply mutation to a candidate solution"""
    if np.random.random() > mutation_rate:
        return h
    
    h = h.copy()
    n_points = len(h)
    # Select indices to mutate
    n_mutate = max(1, int(0.1 * n_points))
    indices = np.random.choice(n_points, n_mutate, replace=False)
    
    # Apply small random changes
    for i in indices:
        delta = np.random.uniform(-0.1, 0.1) * dx
        h[i] += delta
    
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    
    # Rescale to maintain integral constraint
    h = h / np.sum(h) * (n_points / 2)
    
    return h

def crossover(h1, h2):
    """Perform crossover between two solutions"""
    n_points = len(h1)
    if n_points != len(h2):
        raise ValueError("Vectors must have same length")
    
    # Choose crossover point
    crossover_point = np.random.randint(1, n_points)
    
    # Create child
    child = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    
    return child

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select best individual from random tournament"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    pop_size = 100
    max_generations = 300
    mutation_rate = 0.2
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h_values)
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    
    best_h = population[np.argmin(fitnesses)].copy()
    best_c5 = min(fitnesses)
    
    no_improvement_count = 0
    generation = 0
    
    while generation < max_generations:
        if time.time() - start_time > budget_s - 30:
            # Time limit approaching, stop early
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep top 50% of population
        keep_size = pop_size // 2
        population = population[:keep_size]
        fitnesses = fitnesses[:keep_size]
        
        # Add new children until pop_size is restored
        while len(population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child = crossover(parent1, parent2)
            
            # Mutation
            child = mutate(child, mutation_rate, dx)
            
            population.append(child)
            fitnesses.append(evaluate_erdos_solution(child))
        
        # Check if we found a better solution
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 50:
            break
            
        generation += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h, final_c5, n_points)
