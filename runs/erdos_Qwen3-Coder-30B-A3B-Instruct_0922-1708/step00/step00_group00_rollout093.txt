import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given step function h."""
    # Compute overlap integral for all shifts
    overlaps = correlate(h, 1 - h, mode='full') * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with random step functions satisfying constraints."""
    population = []
    
    # If initial solution provided, use it
    if initial_h is not None:
        population.append(initial_h.copy())
        pop_size -= 1
    
    # Generate random individuals
    for _ in range(pop_size):
        # Start with uniform distribution scaled to sum = n_points/2
        h = np.full(n_points, 0.5)
        # Perturb slightly (but keep within [0,1])
        perturbation = np.random.normal(0, 0.1, n_points)
        h += perturbation
        h = np.clip(h, 0, 1)
        # Normalize to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    return population

def selection(population, fitnesses, tournament_size=5):
    """Tournament selection."""
    selected = []
    pop_size = len(population)
    
    for _ in range(pop_size):
        # Select k random individuals for tournament
        tournament_indices = np.random.choice(pop_size, tournament_size)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx].copy())
    
    return selected

def crossover(parent1, parent2, crossover_prob=0.8):
    """Single-point crossover."""
    if np.random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Basic mutation: add small random noise and clip to [0,1]."""
    n_points = len(h)
    num_mutations = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, num_mutations, replace=False)
    
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
        h[i] = np.clip(h[i], 0, 1)
    
    # Renormalize to ensure sum = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    return h

def genetic_algorithm(n_points, pop_size=100, max_generations=120, initial_h=None, budget_s=600):
    """Run genetic algorithm to optimize h for minimal C5."""
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h)
    dx = 2.0 / n_points
    
    best_h = None
    best_c5 = float('inf')
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 1:  # Leave 1s margin
            break
            
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_h(individual, dx) for individual in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Selection
        selected = selection(population, fitnesses)
        
        # Create new population via crossover and mutation
        new_population = []
        # Elitism: keep best 20%
        elite_size = pop_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:pop_size]
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Run optimization to find improved upper bound for C5."""
    np.random.seed(seed)
    
    # Use initial construction if available
    n_points = 45 if 'initial_h_values' in globals() else 60
    initial_h = globals().get('initial_h_values', None)
    
    best_h, best_c5 = genetic_algorithm(
        n_points, pop_size=100, max_generations=120,
        initial_h=initial_h, budget_s=budget_s
    )
    
    return (best_h, best_c5, n_points)
