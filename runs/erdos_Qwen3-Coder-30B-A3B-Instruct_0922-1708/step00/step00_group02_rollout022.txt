import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    patience = 500
    patience_count = 0
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5
    
    # Scale h to meet constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Previous values for momentum
    h_prev = np.zeros_like(h)
    k_max_prev = 0
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute cross-correlation
        corr = signal.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(corr)
        
        # Compute C5
        c5 = corr[k_max] * dx
        
        # Update best solution if current is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_count = 0
        else:
            patience_count += 1
            
        if patience_count > patience:
            alpha *= 0.5
            patience_count = 0
            
        # Gradient of the overlap w.r.t h
        # ∂/∂h_i ( ∫ h(x)(1 - h(x+k)) dx ) = 1 - h[i+k] - h[i] * (1-h[i+k])'
        # Actual gradients for each position i:
        # Gradient is approximated by differentiating the convolution
        # For the current overlap at k_max, compute gradient for h
        
        # Approximate directional derivative for gradient direction
        # We do one step of gradient descent using the current maximum cross-correlation
        grad = np.zeros_like(h)
        # Compute gradient to maximize overlapping integral
        # ∫ h(x)(1-h(x+k)) dx = h*(1-h) convolution
        # Gradient w.r.t h(i) = (1 - h(i+k)) + (1 - h(i)) * delta(k) 
        # Approximate gradient using neighboring values
        for i in range(n_points):
            # Approximate gradient for h[i]
            # This is approximate: computing orthogonal gradient w.r.t. k
            grad[i] = 0
            # Easiest case: compute gradient explicitly 
            # for j in range(n_points):
            #   grad[i] += (1 - h[j]) * (1 if i == j else 0) 
            # Actually, from longer derivation:
            # d/dh[i] sum h[i] (1 - h[i+k]) = 1 - h[i+k]  (for the term h[i] * 1)  
            # But also need to include d/dh[i] sum h[i] (1 - h[i+k])  
            # But for driving up max correlation, we compute how we should move h
            # ...
            # Simpler gradient approach:
            # Use a correction from the current correlation point using finite differences
            # Vectorized safe implementation:
            pass
        # Use a more direct method:
        # Use a gradient that maximizes the correlation
            
        # Direct gradient update:
        # At index k_max, the array in the convolution is full of h[i+k] values
        # If we look at convolution result (at shift k_max), it has terms:
        # sum_{i} h[i] * (1 - h[i+k_max])
        # d/dh[j] of this would be 1 - h[j+k_max] if j < N-k_max
        # But because of cross-correlation (signal.convolve with full mode), 
        # the i-th element of result shows h[i] * (1-h[i+k_max])
        # BUT wait, it's actually expansive correlation: h *^ (1-h)
        # Signal.convolve(h, 1-h, mode='full') gives array of correlations for all shifts
        # Del j of k_max term = 1 - h[j+k_max]
        
        # Compute gradient based on other items in our approximated line
        # At k_max shift,  \sum_i h[i] (1 - h[i+k_max])
        # So d/dh[i] = (1 - h[i+k_max]), but only for some i
        
        # Actually implement: gradient along k dimension of convolution
        ones_minus_h = 1 - h
        # Compute single gradient step from the specific k_max where we are max
        # GRADIENT_wrt_h_along_correlation_direction = D_h_kmax = ones_minus_h * delta + etc
        # But: line 1: h[i] accounts to term 1-h[i+k_max] for terms up to j where j-k_max = i
        # That would be j=i+k_max. 
        # In full convolution result k_max, i-th term is hy[i]*(1-h[i+k_max])
        # But only one of these terms belongs to i
        # So we want to overtile wrong ones from all positions
        t = np.zeros_like(h)
        # Selector based on where overlap comes from
        for i in range(n_points):
            if i + k_max < n_points:
                t[i] = ones_minus_h[i + k_max]  # then h[i] influences term (1-h[i+k_max])
            else:
                # والله! shift overflow (just let it ignore for now)
                pass
        # Use this gradient to update
        # But also we use the total gradient w.r.t. h
        new_h = h + alpha * (t - h)  # Update based on gradient from "best" shift
        # Apply constraints
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        
        # Momentum
        velocity = beta * (new_h - h) + (1 - beta) * (new_h - h_prev)
        h = h + velocity
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        h_prev = new_h.copy()
    
    # Final evaluation
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h, final_c5, n_points)
