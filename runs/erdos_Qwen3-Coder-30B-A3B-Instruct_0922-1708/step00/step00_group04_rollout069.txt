        n = len(h)

        else:
            return np.sum(h[-k:] * (1 - h[:-k])) * dx

        n = len(h)
        overlaps = []
        for k in range(-n+1, n):
            overlaps.append(compute_overlap(h, k, dx))
        return np.array(overlaps)
    
    def eval_c5(h, dx):
        return np.max(compute_all_overlaps(h, dx))
    
    # Local optimization helpers
    def local_update_h(h, k_star, dx, lr=0.05):
        n = len(h)
        h_new = h.copy()

        # approx estimate based on influence of nearby terms
        # A simplified approach: vary h in overlapping region for shift k_star
        start_idx = max(0, k_star)
        end_idx = min(n, n - abs(k_star))
        if k_star >= 0:
            # h[i] * (1 - h[i+k_star])
            # Derivative would be (1 - h[i+k_star]) - h[i] (when i+k_star < n)

            grad = np.zeros_like(h)

                grad[i] = (1 - h[i+k_star]) - h[i]

            h_new = h_new - lr * grad

            h_new = project_h(h_new, dx, n/2)

def evaluate_erdos_solution(h_values, n_points):
    """Simple correlator version to compute max overlap."""
    h = np.array(h_values, dtype=np.float64)
    dx = 2.0 / n_points
    # use full correlation to get all overlaps
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract overlapping parts
    overlap_values = corr[len(h)-1:] * dx
    # max of all overlaps
    return np.max(overlap_values)
