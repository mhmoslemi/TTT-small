import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Initialize with a simple construction
        h = np.ones(n_points) * 0.5
    
    # Normalize to satisfy constraint
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    # Time tracking
    start_time = time.time()
    best_h = h.copy()
    best_C = float('inf')
    
    # Parameters
    eta = 0.01
    patience = 0
    max_patience = 100
    adaptive_interval = 100
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s:
        iteration += 1
        
        # Compute overlaps for all shifts
        # Use full correlation for all lags
        overlap = signal.correlate(h, 1 - h, mode='full')
        # Extract valid overlaps (corresponding to shifts 0 to n_points-1)
        valid_overlap = overlap[n_points-1:]
        k_max = np.argmax(valid_overlap)
        max_overlap = valid_overlap[k_max] * dx
        
        if max_overlap < best_C:
            best_C = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > max_patience:
            break
            
        # Compute gradient for the worst-case shift k_max
        grad = np.zeros_like(h)
        if k_max < n_points:
            # For shift k_max, gradient is (1 - h[i + k_max]) if i + k_max < n
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] = (1 - h[i + k_max]) * dx
                if i - k_max >= 0:
                    grad[i] -= h[i - k_max] * dx
        else:
            # For negative shifts, adjust accordingly
            k = k_max - n_points + 1
            # Gradient is (1 - h[i - k]) for i >= k
            for i in range(n_points):
                if i >= k:
                    grad[i] = (1 - h[i - k]) * dx
                if i + k < n_points:
                    grad[i] -= h[i + k] * dx
                    
        # Update h with gradient descent
        h = h - eta * grad
        
        # Clip h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Adjust to satisfy sum constraint exactly
        current_sum = np.sum(h)
        if abs(current_sum - n_points/2) > 1e-10:
            # Proportional adjustment to maintain sum
            adjustment = (n_points/2 - current_sum) / n_points
            h += adjustment
        
        # Ensure bound constraints
        h = np.clip(h, 0, 1)
        
        # Adaptive learning rate
        if iteration % adaptive_interval == 0 or patience == 0:
            if patience > 0:
                eta *= 0.95
            else:
                eta = 0.01  # reset
                
    # Final evaluation of best solution
    final_overlap = signal.correlate(best_h, 1 - best_h, mode='full')
    final_valid_overlap = final_overlap[n_points-1:]
    final_C = np.max(final_valid_overlap) * dx
    
    return (best_h.tolist(), final_C, len(best_h))
