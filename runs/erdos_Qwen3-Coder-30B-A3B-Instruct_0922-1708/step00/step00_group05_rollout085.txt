import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.0
        h[-1] = 1.0
        h = h / np.sum(h) * (n_points / 2.0)
    
    dx = 2.0 / n_points
    
    # Helper to evaluate C5
    def evaluate_c5(h_vec):
        # Use correlation to compute maximum overlap
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # idx for the middle (max shift)
        max_idx = len(corr) // 2
        return np.max(corr) * dx
    
    # Constraint enforcement
    def enforce_constraints(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) > 1e-10:
            # Scale to target sum
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    # Initial evaluation
    current_c5 = evaluate_c5(h)
    best_h = h.copy()
    best_c5 = current_c5
    
    # Gradient descent loop
    eta = 0.01
    iterations = 0
    no_improvement_count = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        iterations += 1
        
        # Find max overlap lag via correlation
        corr = np.correlate(h, 1 - h, mode="full")
        # Find the maximum lag that contributes most to the error
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (len(corr) // 2)
        
        # Compute gradient for max_k
        grad = np.zeros_like(h)
        if max_k >= 0:
            # h[i] * (1 - h[i + max_k]) for i + max_k < n_points
            for i in range(n_points - max_k):
                grad[i] += (1 - h[i + max_k]) * dx
        else:
            # h[i] * (1 - h[i - max_k]) for i - max_k >= 0
            for i in range(-max_k, n_points):
                grad[i] += -h[i + max_k] * dx
        
        # Update h
        h_new = h - eta * grad
        h_new = enforce_constraints(h_new)
        
        # Evaluate new C5
        new_c5 = evaluate_c5(h_new)
        if new_c5 < current_c5:
            h = h_new
            current_c5 = new_c5
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
            no_improvement_count = 0
            last_improvement = iterations
        else:
            no_improvement_count += 1
        
        # Adaptive learning rate
        if iterations % 100 == 0 or no_improvement_count > 50:
            eta *= 0.95
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
