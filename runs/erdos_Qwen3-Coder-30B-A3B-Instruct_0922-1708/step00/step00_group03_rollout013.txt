import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:

        n_points = len(h)
    except NameError:
        # Initialize with uniform distribution
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[1] = 0.6
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        # Compute all correlations
        correlation = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Take the maximum (excluding the trivial zero shift)
        max_corr = np.max(correlation[len(h_vals)-1:])
        return max_corr * dx

    def constraint_sum(h_vals):
        return np.sum(h_vals) - target_sum

    def constraint_bounds(h_vals):
        return np.concatenate([h_vals, 1 - h_vals])

    # Regularized objective for gradient-based optimization
    def regularized_objective(h_vals):
        C5 = compute_C5(h_vals)
        # Regularization term to keep solution stable
        reg_term = 1e-4 * np.sum(h_vals**2)
        return C5 + reg_term

    def gradient_regularized(h_vals):
        # Simple finite differences for gradient approximation
        eps = 1e-5
        grad = np.zeros_like(h_vals)
        base_C5 = compute_C5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (target_sum / np.sum(h_plus))
            C5_plus = compute_C5(h_plus)
            grad[i] = (C5_plus - base_C5) / eps
        return grad

    # Optimization loop
    best_h = h.copy()
    best_C5 = compute_C5(h)
    max_iter = 1000
    step_size = 0.01

    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        grad = gradient_regularized(best_h)
        new_h = best_h - step_size * grad
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (target_sum / np.sum(new_h))
        
        new_C5 = compute_C5(new_h)
        if new_C5 < best_C5:
            best_h = new_h
            best_C5 = new_C5

    # Post-optimization shift search
    best_h = np.array(best_h)
    best_C5 = compute_C5(best_h)
    
    # Local search over small shifts
    for _ in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        shift_amount = np.random.randint(-n_points//10, n_points//10)
        shifted_h = np.roll(best_h, shift_amount)
        shifted_h = shifted_h * (target_sum / np.sum(shifted_h))
        shifted_C5 = compute_C5(shifted_h)
        if shifted_C5 < best_C5:
            best_h = shifted_h
            best_C5 = shifted_C5

    return (best_h.tolist(), best_C5, n_points)
