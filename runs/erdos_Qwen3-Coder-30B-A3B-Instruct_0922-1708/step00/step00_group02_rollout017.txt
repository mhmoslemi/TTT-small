import numpy as np
from scipy.optimize import minimize
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available

        h0 = initial_h_values.copy()
        n_points = len(h0)
    except:
        # Default construction
        n_points = 45
        h0 = np.ones(n_points) * 0.5
    
    # Optimization parameters
    max_iter = 500
    lr = 0.01
    elastic_net = 0.8  # L1 weight
    update_freq = 100
    patience = 20
    
    def evaluate_overlap(h):
        # Compute overlaps for all lags
        h_ext = np.concatenate([h, np.zeros(n_points)])
        overlaps = np.array([
            np.sum(h_ext[:n_points] * (1 - h_ext[i:i+n_points])) 
            for i in range(n_points)
        ])
        return np.max(overlaps) * (2.0 / n_points)
    
    def project_h(h):
        # Ensure h ∈ [0,1] and ∫h dx = 1
        h = np.clip(h, 0, 1)
        sum_h = np.sum(h) * (2.0 / n_points)
        if sum_h > 0:
            h = h * (n_points / 2) / sum_h
        return h
    
    # Precompute sigmoid basis functions
    def compute_basis(x, m=20):
        centers = np.linspace(0, 2, m+1)[1:-1]
        s = 0.2
        basis = np.zeros((len(x), m))
        for i, c in enumerate(centers):
            basis[:, i] = 1 / (1 + np.exp(-(x - c) / s))
        return basis
    
    # Use a smaller basis for efficiency
    basis_m = int(2 * sqrt(n_points))
    x = np.linspace(0, 2, n_points)
    basis = compute_basis(x, basis_m)
    
    # Initialize coefficients
    try:
        # Estimate coefficients from h0
        a = np.linalg.lstsq(basis, h0, rcond=None)[0]
    except:
        a = np.random.uniform(-0.1, 0.1, basis_m)
    
    best_c5 = float('inf')
    best_h = h0.copy()
    last_improvement = 0
    
    # Optimization loop
    for it in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > 0.95 * budget_s:
            break
            
        # Compute h from coefficients (transformed to match [0,1] and integral)
        h = basis @ a
        h = project_h(h)
        
        # Update coefficients using gradient ascent
        # Gradient approximation via finite difference
        eps = 1e-5
        grad = np.zeros_like(a)
        base_c5 = evaluate_overlap(h)
        
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = project_h(basis @ a_plus)
            c5_plus = evaluate_overlap(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        # Elastic net regularization
        reg_grad = elastic_net * np.sign(a) + (1 - elastic_net) * a
        a = a - lr * (grad + reg_grad)
        
        # Ensure a remains valid
        a = np.clip(a, -2.0, 2.0)
        
        # Check current solution
        h_final = project_h(basis @ a)
        c5 = evaluate_overlap(h_final)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_final.copy()
            last_improvement = it
            
        # Adaptive learning rate
        if it > 0 and it % update_freq == 0 and it - last_improvement > patience:
            lr *= 0.5
            if lr < 1e-6:
                break
    
    # Final validation
    final_h = project_h(basis @ a)
    final_c5 = evaluate_overlap(final_h)
    
    return (final_h, final_c5, n_points)
