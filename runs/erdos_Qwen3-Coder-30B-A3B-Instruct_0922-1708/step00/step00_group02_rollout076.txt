import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral."""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Take maximum value and scale by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, dx):
    """Project h onto feasible set with sum(h) = 1/dx."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * (1/dx)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    learning_rate = 0.01
    current_time = time.time()
    timeout = current_time + budget_s
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Uniform distribution
        h = np.ones(n_points) * 0.5
    
    # Adjust to ensure sum equals 1/dx (i.e., n_points/2)
    h = project_h(h, dx)
    
    best_c5 = evaluate_overlap(h, dx)
    best_h = h.copy()
    
    for i in range(max_iter):
        if time.time() > timeout:
            break
            
        # Evaluate overlap for current h
        current_c5 = evaluate_overlap(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient (finite difference approximation)
        grad = np.zeros(n_points)
        epsilon = 1e-5
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += epsilon
            h_plus = project_h(h_plus, dx)
            c5_plus = evaluate_overlap(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[j] -= epsilon
            h_minus = project_h(h_minus, dx)
            c5_minus = evaluate_overlap(h_minus, dx)
            
            grad[j] = (c5_plus - c5_minus) / (2 * epsilon)
        
        # Update h with gradient descent
        h = h - learning_rate * grad
        
        # Project onto constraints
        h = project_h(h, dx)
    
    # Final evaluation
    final_c5 = evaluate_overlap(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), n_points)
