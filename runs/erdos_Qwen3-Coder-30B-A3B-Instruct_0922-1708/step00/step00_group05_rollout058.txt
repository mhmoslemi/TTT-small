import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    h = np.array(h_values)
    # Compute all overlaps for different lags
    overlap = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlap) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    # Initialize with the provided initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a random construction normalized to sum to n_points/2
        np.random.seed(seed)
        n_points = 70
        h_init = np.random.rand(n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    best_h = h_init.copy()
    best_c5 = evaluate_erdos_solution(h_init, n_points)
    
    # Each solve iteration
    start_time = time.time()
    iter_count = 0
    
    # Use multiple random initializations
    random_seeds = [seed + i for i in range(5)]
    
    for s in random_seeds:
        np.random.seed(s)
        current_h = h_init.copy()
        
        # Perturb slightly
        current_h += np.random.normal(0, 0.01, n_points)
        current_h = np.clip(current_h, 0, 1)
        current_h = current_h * (n_points / 2) / np.sum(current_h)
        
        # Iterative refinement with QCQP
        for it in range(5):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Define QCQP in CVXPY
            h = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                h >= 0,
                h <= 1,
                cp.sum(h) == n_points / 2,
                C >= 0
            ]
            
            # Build overlap constraints for all lags k
            dx = 2.0 / n_points
            for k in range(-n_points + 1, n_points):
                # Head of the convolution for lag k
                if k >= 0:
                    start_idx = k
                    end_idx = n_points
                    corr_idx = k
                else:
                    start_idx = 0
                    end_idx = n_points + k
                    corr_idx = 0
                
                if end_idx > start_idx:
                    # sum(h[i] * (1 - h[i + k])) <= C * dx
                    lhs = cp.sum(
                        cp.multiply(
                            h[start_idx:end_idx],
                            1 - h[corr_idx:corr_idx + (end_idx - start_idx)]
                        )
                    )
                    constraints.append(lhs <= C * dx)
            
            # Objective
            objective = cp.Minimize(C)
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    h_sol = h.value
                    c5_val = C.value
                    
                    # Validate solution is feasible
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    evaluated_c5 = evaluate_erdos_solution(h_sol, n_points)
                    
                    if evaluated_c5 < best_c5:
                        best_h = h_sol.copy()
                        best_c5 = evaluated_c5
            except:
                pass
                
            # Perturb and retry if needed or continue
            current_h += np.random.normal(0, 0.005, n_points)
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            h_init = current_h.copy()
            
            iter_count += 1
            if time.time() - start_time > budget_s - 1:
                break
    
    return (list(best_h), best_c5, n_points)
