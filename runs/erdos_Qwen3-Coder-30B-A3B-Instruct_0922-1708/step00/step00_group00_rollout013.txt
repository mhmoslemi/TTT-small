import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for a given h"""
    # Compute all overlaps using correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # Scale by dx to get actual integral values
    overlaps = overlaps * dx
    # Return maximum overlap
    return np.max(overlaps)

def enforce_constraints(h, n_points, dx):
    """Ensure h is in [0,1] and sums to 1 (n_points/2 * dx)"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Correct the sum to be exactly n_points/2
    current_sum = np.sum(h)
    if abs(current_sum - n_points/2) > 1e-10:
        # Normalize to the target sum
        h = h * (n_points/2) / current_sum
    return h

def initialize_population(pop_size, n_points, seed=42):
    """Create an initial population"""
    np.random.seed(seed)
    population = []
    
    # If initial_h_values exists (from global scope), use it for 20% of population
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        init_h = enforce_constraints(init_h, n_points, 2.0/n_points)
        population.append(init_h)
        pop_size -= 1
    
    # Fill the rest with random initializations
    for _ in range(pop_size):
        # Start with uniform distribution scaled appropriately
        h = np.ones(n_points) * (n_points/2) / n_points
        # Add small random noise
        h += np.random.normal(0, 0.1, n_points)
        h = enforce_constraints(h, n_points, 2.0/n_points)
        population.append(h)
        
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select a parent using tournament selection"""
    selected_idx = np.random.randint(0, len(population))
    for _ in range(tournament_size - 1):
        candidate_idx = np.random.randint(0, len(population))
        if fitnesses[candidate_idx] < fitnesses[selected_idx]:
            selected_idx = candidate_idx
    return population[selected_idx]

def crossover(parent1, parent2):
    """Perform single-point crossover between parents"""
    n_points = len(parent1)
    # Select crossover point
    crossover_point = np.random.randint(1, n_points)
    # Create offspring
    offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return offspring1, offspring2

def mutate(h, mutation_rate=0.05):
    """Mutate an individual"""
    n_points = len(h)
    for i in range(n_points):
        if np.random.random() < mutation_rate:
            # Add small random change
            h[i] += np.random.normal(0, 0.1)
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Fixed parameters
    n_points = 45
    pop_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = int(0.2 * pop_size)
    max_generations = 120
    dx = 2.0 / n_points
    
    # Initialize population
    population = initialize_population(pop_size, n_points, seed)
    
    start_time = time.time()
    generation = 0
    
    # Track best solution so far
    best_h = None
    best_c5 = float('inf')
    
    while time.time() - start_time < budget_s and generation < max_generations:
        # Evaluate fitness for the current population
        fitnesses = []
        for h in population:
            fitness = evaluate_h(h, dx)
            fitnesses.append(fitness)
            if fitness < best_c5:
                best_c5 = fitness
                best_h = h.copy()
        
        # Select parents and generate offspring
        new_population = []
        
        # Elitism: keep best individuals
        sorted_indices = np.argsort(fitnesses)
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate rest of population
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.random() < crossover_prob:
                offspring1, offspring2 = crossover(parent1, parent2)
            else:
                offspring1, offspring2 = parent1.copy(), parent2.copy()
            
            # Mutation
            offspring1 = mutate(offspring1, mutation_rate)
            offspring2 = mutate(offspring2, mutation_rate)
            
            # Enforce constraints
            offspring1 = enforce_constraints(offspring1, n_points, dx)
            offspring2 = enforce_constraints(offspring2, n_points, dx)
            
            new_population.extend([offspring1, offspring2])
        
        # Trim to exact pop_size
        population = new_population[:pop_size]
        
        generation += 1
        
        # Occasionally print progress
        if generation % 20 == 0:
            elapsed = time.time() - start_time
            if elapsed > 0.95 * budget_s:
                break
        
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to a simple valid construction
        h = np.ones(n_points) * 0.5
        h = enforce_constraints(h, n_points, dx)
        c5 = evaluate_h(h, dx)
        return (h.tolist(), c5, n_points)
