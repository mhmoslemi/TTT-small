import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a simple starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to meet constraint
    current_sum = np.sum(h)
    h = h * (target_sum / current_sum)
    
    def evaluate(h):
        # Compute overlap integral
        h_flip = 1 - h
        correlation = signal.correlate(h, h_flip, mode='full')
        # The correlation is for all shifts, we want the maximum
        # Since this is a valid correlation with 1-h, we take the max
        # We need to extract the valid correlations and normalize by dx
        # For fast evaluation, just do the max of the correlation values
        max_overlap = np.max(correlation)
        return max_overlap * dx
    
    # Initial evaluation
    current_c5 = evaluate(h)
    best_h = h.copy()
    best_c5 = current_c5
    no_improvement_count = 0
    start_time = time.time()
    max_trials = 1000
    trial_count = 0
    
    while trial_count < max_trials and (time.time() - start_time) < budget_s - 1:
        trial_count += 1
        # Create a new candidate h by perturbing some points
        h_candidate = h.copy()
        # Choose a number of points to adjust (random percentage)
        n_perturb = np.random.randint(1, max(1, n_points // 10))
        indices = np.random.choice(n_points, size=n_perturb, replace=False)
        
        for i in indices:
            # Add a small random change to h[i]
            delta = np.random.uniform(-0.05, 0.05)
            h_candidate[i] += delta
            
        # Ensure values stay in [0, 1]
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Adjust to maintain sum constraint
        current_sum = np.sum(h_candidate)
        adjustment = (target_sum - current_sum) / n_points
        h_candidate += adjustment
        
        # Apply smoothing to reduce noise
        # Simple moving average filter
        window = min(5, n_points // 5)
        if window % 2 == 0:
            window += 1
        if window > 1:
            h_smooth = np.convolve(h_candidate, np.ones(window)/window, mode='same')
            h_candidate = h_smooth
        
        # Ensure still in bounds
        h_candidate = np.clip(h_candidate, 0, 1)
        
        # Re-adjust to meet sum constraint after filtering
        current_sum = np.sum(h_candidate)
        h_candidate = h_candidate * (target_sum / current_sum)
        
        # Evaluate new solution
        c5_candidate = evaluate(h_candidate)
        
        if c5_candidate <= current_c5:
            h = h_candidate
            current_c5 = c5_candidate
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
                no_improvement_count = 0
            else:
                no_improvement_count += 1
        else:
            no_improvement_count += 1

        if no_improvement_count > 100:
            break
    
    # Final re-evaluation
    final_c5 = evaluate(best_h)
    
    return (best_h.tolist(), final_c5, len(best_h))
