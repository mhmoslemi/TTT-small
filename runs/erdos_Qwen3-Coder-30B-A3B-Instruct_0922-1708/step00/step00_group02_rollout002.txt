import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    """Evaluate the C5 constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    conv = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is scaled by dx
    c5 = np.max(conv) * dx
    return c5

def project_h(h, n):
    """Project h to satisfy constraints: h in [0,1], sum(h) = n/2"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to satisfy sum condition
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n / 2) / current_sum
    return h

            best_c5 = c5
            best_h = h.copy()
        
        # Simple gradient estimation via finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
