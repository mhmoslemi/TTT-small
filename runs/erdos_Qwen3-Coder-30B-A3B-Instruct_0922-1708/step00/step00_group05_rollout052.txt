import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum correlation scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

            for i in indices:
                delta = np.random.uniform(-0.05, 0.05)
                h_new[i] += delta

            h_new = np.clip(h_new, 0, 1)

            sum_h = np.sum(h_new) * dx
            if sum_h != 1.0:

                idx = np.random.randint(0, n_points)

    final_c5 = evaluate_erdos_solution(best_h)
