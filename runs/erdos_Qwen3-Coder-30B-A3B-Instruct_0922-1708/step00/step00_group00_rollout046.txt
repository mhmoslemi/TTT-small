import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h_values, n_points):
    dx = 2.0 / n_points
    # Compute the full convolution of h with (1-h)
    corr = signal.correlate(h_values, 1 - h_values, mode='full')
    # Maximum overlap value
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy sum(h) = n_points/2 and 0 <= h[i] <= 1"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to target sum
        h = h * target_sum / current_sum
        h = np.clip(h, 0, 1)
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    timeout = time.time() + budget_s
    
    # Initialize
    if 'initial_h_values' in globals():
        h_current = np.array(initial_h_values)
    else:
        h_current = np.full(n_points, 0.5 * dx)
    
    h_current = project_h(h_current, n_points)
    
    c5_current = evaluate_erdos_solution(h_current, n_points)
    
    best_h = h_current.copy()
    best_c5 = c5_current
    
    # Parameters
    sigma = 0.1
    decay = 0.99
    iter_no_improve = 0
    max_no_improve = 1000
    
    for i in range(max_iter):
        if time.time() > timeout:
            break

        # Generate new candidate
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h_current + perturbation
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new, n_points)
        
        # Accept new solution if it's better
        if c5_new < c5_current:
            h_current = h_new
            c5_current = c5_new
            iter_no_improve = 0
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        else:
            iter_no_improve += 1
            
        # Decrease sigma over time
        sigma *= decay
        
        # Stop if no improvement for a while
        if iter_no_improve > max_no_improve:
            break
    
    return (best_h.tolist(), best_c5, n_points)
