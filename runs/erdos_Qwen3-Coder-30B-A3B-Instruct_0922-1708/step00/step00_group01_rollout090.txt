import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    correlation = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(correlation) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    personal_best_positions = [particle.copy() for particle in swarm]
    
    global_best_idx = np.argmin(personal_bests)
    global_best_value = personal_bests[global_best_idx]
    global_best_h = personal_best_positions[global_best_idx].copy()

    max_iter = 500
    stagnation_count = 0
    last_improvement = 0

            particle = project_h(particle, n_points)
            
            # Store updated particle
            swarm[i] = particle
            
            # Evaluate and update personal best
            current_value = evaluate_h(particle, 2.0 / n_points)
            if current_value < personal_bests[i]:
                personal_bests[i] = current_value
                personal_best_positions[i] = particle.copy()
                
                # Update global best
                if current_value < global_best_value:
