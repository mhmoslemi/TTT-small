import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for a given step function h."""
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        # Start with a simple construction
        n_points = 100
        h0 = np.ones(n_points) * 0.5
        h0[0] = 0.0
        h0[-1] = 1.0
        h0 = project_h(h0, n_points)

    # PSO parameters
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    
    # Initialize particles
    particles = []
    velocities = []
    p_best = []
    p_best_vals = []
    
    # Seed with initial solution
    particles.append(h0.copy())
    velocities.append(np.zeros_like(h0))
    p_best.append(h0.copy())
    p_best_vals.append(evaluate_h(h0, 2.0 / n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        p_best.append(h.copy())
        p_best_vals.append(evaluate_h(h, 2.0 / n_points))
    
    # Global best
    gb_idx = np.argmin(p_best_vals)
    g_best = p_best[gb_idx].copy()
    g_best_val = p_best_vals[gb_idx]
    
    # PSO iterations
    start_time = time.time()
    stagnation_count = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            vel_update = (w * velocities[i] +
                          c1 * r1 * (p_best[i] - particles[i]) +
                          c2 * r2 * (g_best - particles[i]))
            velocities[i] = vel_update
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points)
            
            # Evaluate
            val = evaluate_h(particles[i], 2.0 / n_points)
            
            # Update personal best
            if val < p_best_vals[i]:
                p_best[i] = particles[i].copy()
                p_best_vals[i] = val
                
                # Update global best
                if val < g_best_val:
                    g_best = particles[i].copy()
                    g_best_val = val
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Diversity injection to avoid local optima
        if stagnation_count > 25:
            for _ in range(n_particles // 5):
                idx = np.random.randint(n_particles)
                h = np.random.rand(n_points)
                h = project_h(h, n_points)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                val = evaluate_h(h, 2.0 / n_points)
                p_best_vals[idx] = val
                p_best[idx] = h.copy()
            stagnation_count = 0
            
    return (g_best.tolist(), g_best_val, n_points)
