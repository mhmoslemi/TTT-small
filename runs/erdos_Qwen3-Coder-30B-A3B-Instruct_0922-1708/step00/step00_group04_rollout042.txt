import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial h values based on parent construction or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 42
        # Create triangular h function
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2*x - 1)
        # Scale to ensure sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_bound = 0.3807
    
    def evaluate_C5(h_vals):
        # Compute all overlaps
        h_plus = h_vals
        h_minus = 1 - h_vals
        # Use full cross-correlation
        overlaps = correlate(h_plus, h_minus, mode='full')
        # The overlap integrals are from correlation scaled by dx
        max_overlap = np.max(overlaps) * dx
        return max_overlap

    def project_h(h_vals):
        # Project h to [0,1] and maintain sum constraint
        h_vals = np.clip(h_vals, 0, 1)
        current_sum = np.sum(h_vals)
        target_sum = n_points / 2
        if not np.isclose(current_sum, target_sum, rtol=1e-10):
            scale = target_sum / current_sum
            h_vals = h_vals * scale
        return h_vals
    
    def objective_and_gradient(h_vals, k_star):
        # Objective: minimize overlap for given k
        # Gradient: derivative of overlap with respect to h
        
        # Compute overlap related to shift k_star
        h_plus = h_vals
        h_minus = 1 - h_vals
        
        # Check if we compute needed indices
        k = k_star
        if k > 0:
            # h[i] * h_minus[i+k], i from 0 to n-k-1
            indices1 = np.arange(n_points - k)
            indices2 = np.arange(k, n_points)
        elif k < 0:
            # h[i] * h_minus[i+k], i from -k to n-1
            indices1 = np.arange(-k, n_points)
            indices2 = np.arange(n_points + k)
        else:
            # k = 0
            indices1 = np.arange(n_points)
            indices2 = np.arange(n_points)
            # overlap = sum(h[i] * (1 - h[i]))
        
        # Compute overlap value and gradient (partial derivatives)
        if k == 0:
            # As derived: overlap = sum(h[i] * (1 - h[i]))
            # Grad = 1 - 2 * h[i] 
            overlap = np.sum(h_plus * h_minus) * dx
            grad = (1 - 2 * h_vals) * dx
        else:
            # Handle case where k != 0
            # overlap = sum(h[i] * h_minus[i+k]) 
            # Derivative: for i, partial w.r.t. h[i] = h_minus[i+k]
            # and for j where j+k=i, partial w.r.t. h[j] = h[i]
            # Effectively: grad[i] = h_minus[i+k];
            # But for all indices, construct full gradient in logic-friendly form
            
            # Actually compute difference via convolution logic
            # For j-th element: Prev approach wasn't clear enough
            
            # Begin with full convolution
            overlap = 0
            grad = np.zeros_like(h_vals)
            if k > 0:
                # For i from 0 to n-k-1
                for i in range(n_points - k):
                    overlap += h_plus[i] * h_minus[i + k]
                    grad[i] += h_minus[i + k]
            elif k < 0:
                # For i from -k to n-1
                k_abs = -k
                for i in range(k_abs, n_points):
                    overlap += h_plus[i] * h_minus[i - k_abs]
                    grad[i - k_abs] += h_minus[i]
            else:
                for i in range(n_points):
                    overlap += h_plus[i] * h_minus[i]
                    grad[i] = 1 - 2 * h_vals[i]  # Simplified for k=0
            
            overlap *= dx
            grad *= dx
            
        return overlap, grad
    
    best_h = h.copy()
    best_C5 = evaluate_C5(best_h)
    
    # Alternating optimization loop
    iter_count = 0
    update_threshold = 100
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 5 and best_C5 > target_bound:
        iter_count += 1
        # Compute all overlaps for all shifts
        overlaps_full = np.zeros(2 * n_points - 1)
        h_plus = best_h
        h_minus = 1 - best_h
        
        # Compute overlap for all shifts using full correlation
        # We need to do this separately in a loop due to how correlate works
        max_overlap = -1
        best_k = 0
        # We will loop manually to compute for all k
        for k in range(-n_points + 1, n_points):
            if k >= 0:
                offset = k
                vals = h_plus[:n_points - k] * h_minus[offset:]
            else:
                offset = -k
                vals = h_plus[offset:] * h_minus[:n_points - offset]
            o = np.sum(vals) * dx
            overlaps_full[k + n_points - 1] = o
            if o > max_overlap:
                max_overlap = o
                best_k = k

        # Now optimize for the worst shift
        h_new = best_h.copy()
        # Loop for local update: gradient descent on specific shift
        tolerance = 1e-5
        max_iter = 100
        learning_rate = 0.1
        for epoch in range(max_iter):
            if time.time() - start_time > budget_s - 5:
                break
            # Compute loss and gradient for best_k
            overlap, grad = objective_and_gradient(h_new, best_k)
            # Update (with a small step to try to improve)
            h_new = h_new - learning_rate * grad
            # Project back to bounds
            h_new = project_h(h_new)
            # Check convergence or revision
            if epoch % 10 == 0:
                new_score = evaluate_C5(h_new)
                if new_score < best_C5:
                    best_h = h_new.copy()
                    best_C5 = new_score
                    last_improvement = iter_count
                elif new_score < best_C5 * 1.001:
                    # Consider not moving forward if no gain
                    pass
                elif learning_rate > 1e-6:
                    learning_rate *= 0.8  # reduce LR on stall
            if epoch == max_iter - 1 and abs(overlap) < tolerance:
                break
            if overlap < 1e-12:
                break
        
        # Final check after GD
        current_C5 = evaluate_C5(best_h)
        if current_C5 < best_C5:
            best_C5 = current_C5
            last_improvement = iter_count
        elif iter_count - last_improvement > update_threshold:
            # We are not improving much, perturb a bit
            noise_magnitude = 0.01
            noise = np.random.normal(0, noise_magnitude, n_points)
            temp_h = best_h + noise
            temp_h = project_h(temp_h)
            temp_C5 = evaluate_C5(temp_h)
            if temp_C5 < best_C5:
                best_h = temp_h
                best_C5 = temp_C5
                last_improvement = iter_count
                
        if best_C5 <= target_bound:
            break 
            
    # Final cleanup
    final_h = project_h(best_h)
    final_C5 = evaluate_C5(final_h)
    
    # Return the best solution found
    return (final_h.tolist(), float(final_C5), int(n_points))
