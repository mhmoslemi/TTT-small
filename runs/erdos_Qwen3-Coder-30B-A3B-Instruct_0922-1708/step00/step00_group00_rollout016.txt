import numpy as np
import time
import random
from scipy import correlate

def evaluate_h(h, dx):
    # Compute overlaps for all shifts
    overlaps = correlate(h, 1 - h, mode="full") * dx
    # Return maximum overlap
    return np.max(overlaps)

def normalize_h(h, target_sum):
    # Ensure sum is correct
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    # Keep within bounds
    h = np.clip(h, 0, 1)
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    # Select one individual using tournament selection
    selected_idx = random.randint(0, len(population)-1)
    for _ in range(tournament_size - 1):
        idx = random.randint(0, len(population)-1)
        if fitnesses[idx] < fitnesses[selected_idx]:
            selected_idx = idx
    return population[selected_idx].copy()

def crossover(parent1, parent2, crossover_prob=0.8):
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = random.randint(1, n-1)
    
    offspring1 = np.concatenate([parent1[:point], parent2[point:]])
    offspring2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return offspring1, offspring2

def mutate(individual, mutation_rate=0.05):
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            individual[i] = random.random()
    return individual

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Set parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize population
    population_size = 100
    population = []
    
    # Include initial solution if available
    if 'initial_h_values' in globals():
        population.append(np.array(initial_h_values, dtype=float))
    
    # Generate random valid individuals
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, target_sum)
        population.append(h)
    
    # Track best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Run genetic algorithm
    generation = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 2 and no_improvement < max_no_improvement:
        # Evaluate fitness of all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
        
        # Update best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
            no_improvement = 0
        
        # Create new population
        new_population = []
        
        # Elitism: keep best individuals
        sorted_indices = np.argsort(fitnesses)
        for i in range(int(0.2 * population_size)):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < population_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            offspring1, offspring2 = crossover(parent1, parent2)
            offspring1 = mutate(offspring1)
            offspring2 = mutate(offspring2)
            # Ensure constraints
            offspring1 = normalize_h(offspring1, target_sum)
            offspring2 = normalize_h(offspring2, target_sum)
            new_population.extend([offspring1, offspring2])
        
        # Trim to exact size
        population = new_population[:population_size]
        
        generation += 1
        no_improvement += 1
        
    # Final evaluation of best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (list(best_h), float(final_c5), n_points)
    else:
        # Return the first individual if no improvement found
        return (list(population[0]), float(evaluate_h(population[0], dx)), n_points)
