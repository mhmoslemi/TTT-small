import numpy as np
import time
from scipy.signal import correlate

def evaluate_c5(h, dx):
    """Evaluate the C5 bound for given step function h"""
    # Compute overlap integral for all shifts
    overlap = correlate(h, 1 - h, mode='full')
    # Scale by dx
    overlap *= dx
    # Return maximum overlap value
    return np.max(overlap)

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h
