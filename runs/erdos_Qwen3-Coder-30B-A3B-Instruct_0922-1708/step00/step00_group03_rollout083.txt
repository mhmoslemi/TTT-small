import numpy as np
import cvxpy as cp
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    # Setup
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * target_sum / np.sum(h)  # Normalize
    else:
        h = np.full(n_points, 0.5)
    
    # Optimization variables
    X = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        X >= 0,
        X <= 1,
        cp.sum(X) == target_sum
    ]
    
    # Add coupling constraints using auxiliary variables
    # z[i,j] := X[i] * (1 - X[j])
    Z = cp.Variable((n_points, n_points), symmetric=False)
    
    # Add coupling constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i, j] <= X[i],
                Z[i, j] <= 1 - X[j],
                Z[i, j] >= X[i] + X[j] - 1
            ]
    
    # Objective: minimize max correlation
    C_values = []
    for k in range(-n_points+1, n_points):
        if k >= 0:
            # Left shift: indices i, i+k, valid range
            valid = np.arange(n_points - k)
            c_val = cp.sum(Z[valid, valid + k])
        else:
            # Right shift: indices |k| + i, i
            valid = np.arange(-k, n_points)
            c_val = cp.sum(Z[valid + k, valid])
        C_values.append(c_val)
    
    # Add constraints for each correlation value
    t = cp.Variable()
    for c_val in C_values:
        constraints.append(c_val <= t)
    
    # Define objective
    objective = cp.Minimize(t)
    
    # Try to solve with different solvers
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
            h_sol = X.value
        else:
            h_sol = h
    except:
        h_sol = h

    # Final validation and rescaling
    h_sol = np.clip(h_sol, 0, 1)
    h_sol = h_sol * target_sum / np.sum(h_sol)
    
    # Calculate actual C5 value
    c5_bound = evaluate_erdos_solution(h_sol)
    
    return (h_sol.tolist(), c5_bound, n_points)
