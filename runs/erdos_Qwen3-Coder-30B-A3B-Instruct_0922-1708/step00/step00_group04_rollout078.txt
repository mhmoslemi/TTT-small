def evaluate_erdos_solution(h):
    # Evaluate overlap manually using convolution
    # We want C_5 = max_k int h(x)(1 - h(x+k)) dx 
    # Compute full convolution h * (1 - h) over full domain
    n = len(h)
    # Slightly more efficient: 
    # For : (h * (1 - h))[k] = sum_{x} h[x] * (1 - h[x + k])
    # The explicit form is:
    # h = [h[0], h[1], ..., h[n-1]]
    # We are computing over k = 0, 1, ..., 2*n-2
    # We efficiently compute using numpy correlate
    
    # But note that this has k in [0, 2] range.  x runs from [0,2),
    # so for integer indexing in [0, n),   k = x + u index (recall integer index k)
    
    # Convert to a full convolution over [0, 2] (n_points)
    h_full = np.concatenate([h, np.zeros(n-1)])
    h1 = np.concatenate([np.ones(n), np.zeros(n-1)])
    conv_result = np.correlate(h_full, h1 - h_full[:n], mode="full")
    
    # But we only want k from 0 to 2 with appropriate sampling
    # x in [0, 2) with spacing dx = 2/n  
    # So overlapping k = 0 to 1 is from 0 to n_1 = 2n/2 = n  
    # We know convolution index k goes from 0 to 2n-2
    # But eval integral at x, the value of h(x+k) relies on x+k in [0, 2) 
    # since h outside [0, 2) is considered 0.
    # So we have actual kernel size = 2n 
    # ... Actually, it's convolution of h with (1-h) in [0,2]
    # We reconstruct full spanning convolution from samples, use valid segment
    
    dx = 2.0 / len(h)
    k_range_values = []
    # Compute overlap for each shift k = 0 to 2 (in discrete steps simulate)
    for k_idx in range(2 * len(h) - 1):
        # This is x in [0,2] , and we want for shift k, (h(x)(1-h(x+k)))  
        # Reference to kernel: in x range, for k, we have (x+k) in [k, 2+k]
        # limit intersection with x in [0,2] to get valid range:
        # This loop essentially does int h(x) (1-h(x+k)) dx  
        # The integral blends into the n-point piecewise areas of h
        # So it's just about running all x (steps in h), weighted by 2/n (dx)
        
        # for simplicity, compute correlation of h with (1-h) in full mode  
        # but limit to relevant values that really contribute based on 0<=x,x+k<=2
        # Rebuild using custom logic for each k:
        kronk = 0
        for i in range(len(h)):
            x = i * dx  # x in [0, 2) 
            x_plus_k = x + k_idx * dx / 2  # Use step size similar to dx
            if 0 <= x_plus_k <= 2:                
                # In the continuous version, the integral is:
                # integral_{x \in [0,2]} h(x) * (1 - h(x + k)) dx
                # The discrete sampling is weighted by dx. So the total integral        
                # = sum_h h[i] * (1 - h[i+k]) * dx
                
                j = int(x_plus_k / dx)  # integer represents index in valid range
                if 0 <= j < len(h):
                    kronk += h[i] * (1 - h[j]) * dx
                    
        k_range_values.append(kronk)
    
    return max(k_range_values) if k_range_values else 0.0
