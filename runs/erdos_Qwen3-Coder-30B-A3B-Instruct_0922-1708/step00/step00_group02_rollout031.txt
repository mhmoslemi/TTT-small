def run(seed=42, budget_s=600, **kwargs):

    best_c5 = float('inf')

        # C_5 = max_k ∫ h(x)(1 - h(x+k)) dx = max_k d * (1 - h[i] * h[i+k])
        # So overlaps from convolution need to be adjusted
        # Taking advantage of the fact that we do h(x)(1 - h(x+k)) which is:
        # for each k, sum over i of h[i] * (1-h[i+k])
        # This is equivalent to: sum(1*h) - sum(h[i]*h[i+k])
        # But in our case we use sum(h) instead of conv_hh
        
        # Manually compute the overlap integral for the current h
        # C_5 = max_k integral h(x) * (1 - h(x+k)) dx 
        # This is equivalent to
        # max_k ( sum(h[i] * (1-h[i+k])) for i = 0 to n-1, where i+k is within range )
        # For now lets simply compute the maximum across shifts using convolution

        c5_candidates = []
        max_realizable_shift = n_points - 1
        for shift in range(-max_realizable_shift, max_realizable_shift + 1):
            if shift >= 0:
                # i from 0 to min(n-1, n-1-shift) 
                i_start = 0
                i_end = min(n_points, n_points - shift)
                if i_start < i_end:
                    overlap_sum = np.sum(h[i_start:i_end] * (1 - h[i_start + shift:i_end + shift]))
                    c5_candidates.append(overlap_sum * (2 / n_points))
            else: # shift < 0 
                # i from max(0, -shift) to n-1  
                i_start = max(0, -shift)
                i_end = n_points
                if i_start < i_end:
                    overlap_sum = np.sum(h[i_start:i_end] * (1 - h[i_start - shift:i_end - shift]))
                    c5_candidates.append(overlap_sum * (2 / n_points))
        
        if len(c5_candidates) == 0:
            # fallback
            current_c5 = 0.5
        else:
            current_c5 = max(c5_candidates)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration

            alpha *= 0.5
            last_improvement = iteration

        max_shift_idx = -1
        max_overlap_val = -1

            if shift >= 0:
                i_start = 0
                i_end = min(n_points, n_points - shift)

                        max_shift_idx = shift
            else:
                i_start = max(0, -shift)
                i_end = n_points
                if i_start < i_end:
                    overlap_sum = np.sum(h[i_start:i_end] * (1 - h[i_start - shift:i_end - shift]))
                    if overlap_sum > max_overlap_val:
                        max_overlap_val = overlap_sum
                        max_shift_idx = shift

        # Compute approximate gradient for that specific shift
        grad = np.zeros_like(h)

        if max_shift_idx >= 0:
            # For shift k, increase h[i] if h[i+k] is small to reduce overlap 
            # Overlap is sum over i of h[i] * (1 - h[i+k])
            # So gradient wrt h[i] = (1 - h[i+k]) when k == shift.
            # But we must also account for d in the integral and possible boundaries
            # Let's write just the first approximation:
            for i in range(n_points):
                # We compute gradient of overlap wrt h_i
                # Overlap = sum_j h_j * (1 - h_{j+shift}) 
                # Grad = (1 - h_{i+shift}) if i+shift is in range, else 0
                if i + max_shift_idx < n_points and i + max_shift_idx >= 0:
                    grad[i] = 1 - h[i + max_shift_idx]
                else:
                    grad[i] = 0
                # Additional term from dependency on h_i
                if i + max_shift_idx < n_points and i + max_shift_idx >= 0:
                    grad[i] -= h[i]  # From term h_i * (1 - h_{i+shift})
        else:
            # max_shift_idx < 0: shift it backward
            # We want to reduce ~ sum_{i} h[i] * (1 - h[i-shift]) = sum_{i} h[i] - h[i] * h[i-shift]
            # Gradient of overlap wrt h[i]: 1 - h[i-shift] if i-shift in range
            for i in range(n_points):
                if i - (-max_shift_idx) < n_points and i - (-max_shift_idx) >= 0:
                    grad[i] = 1 - h[i - (-max_shift_idx)]
                else:
                    grad[i] = 0
                if i - (-max_shift_idx) < n_points and i - (-max_shift_idx) >= 0:
                    grad[i] -= h[i]

        # Apply update
        h = h - alpha * grad
        
        # Project back into [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to make sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    # Compute final C5 value 
    c5_final = best_c5
    
    return (best_h.tolist(), c5_final, n_points)
