import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap integral for any k
    max_overlap = np.max(corr)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 70
    n_particles = 20
    max_iter = int(budget_s / 0.05)  # estimate
    w = 0.7
    c1 = c2 = 2.0
    v_max = 0.1
    tournament_size = 4  # 20% of particles

    # Initialize particles
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = np.clip(h_start, 0, 1)
        h_start = h_start * (n_points / 2) / np.sum(h_start)  # Normalize
        particles = [h_start]
        for _ in range(n_particles - 1):
            h_p = h_start + np.random.normal(0, 0.1, n_points)
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (n_points / 2) / np.sum(h_p)
            particles.append(h_p)
    else:
        # Create a simple step function if no initial construction
        particles = []
        for _ in range(n_particles):
            h_p = np.random.rand(n_points)
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (n_points / 2) / np.sum(h_p)
            particles.append(h_p)

    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    personal_bests = [particles[i].copy() for i in range(n_particles)]
    personal_best_scores = [evaluate_erdos_solution(particles[i]) for i in range(n_particles)]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_bests[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]

    # For diversity
    last_improvement = 0
    start_time = time.time()

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break

        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            v_new = w * velocities[i] + \
                    c1 * r1 * (personal_bests[i] - particles[i]) + \
                    c2 * r2 * (global_best - particles[i])
            v_new = np.clip(v_new, -v_max, v_max)
            velocities[i] = v_new

            # Update position
            h_new = particles[i] + velocities[i]
            h_new = np.clip(h_new, 0, 1)

            # Normalize
            h_new = h_new * (n_points / 2) / np.sum(h_new)

            # Evaluate
            new_score = evaluate_erdos_solution(h_new)

            # Update personal best
            if new_score < personal_best_scores[i]:
                personal_bests[i] = h_new.copy()
                personal_best_scores[i] = new_score

            # Update global best
            if new_score < global_best_score:
                global_best = h_new.copy()
                global_best_score = new_score
                last_improvement = iteration

            particles[i] = h_new

        # Tournament selection
        if iteration % 20 == 0:
            indices = np.argsort(personal_best_scores)
            keep_indices = indices[:n_particles//2]
            for i in range(n_particles):
                if i not in keep_indices:
                    # Replace by new random solution
                    h_p = np.random.rand(n_points)
                    h_p = np.clip(h_p, 0, 1)
                    h_p = h_p * (n_points / 2) / np.sum(h_p)
                    particles[i] = h_p
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
                    personal_bests[i] = h_p.copy()
                    personal_best_scores[i] = evaluate_erdos_solution(h_p)

        # Reintroduce diversity if no improvement in 100 iterations
        if iteration - last_improvement >= 100:
            for i in range(n_particles):
                # Slight perturbation
                particles[i] += np.random.normal(0, 0.01, n_points)
                particles[i] = np.clip(particles[i], 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                personal_bests[i] = particles[i].copy()
                personal_best_scores[i] = evaluate_erdos_solution(particles[i])

    # Final verification of global best
    final_score = evaluate_erdos_solution(global_best)
    
    return (global_best.tolist(), final_score, n_points)
