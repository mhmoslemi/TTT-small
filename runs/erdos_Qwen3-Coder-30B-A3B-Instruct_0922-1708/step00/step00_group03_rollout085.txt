import numpy as np
from scipy.optimize import minimize
import time


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h_vals):
        # Evaluate correlation matrix
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Only keep the relevant parts (shifts from -n+1 to n-1)
        n = len(h_vals)
        corr = corr[n-1:2*n-1]
        return np.max(corr) * dx
    
    def proj_h(h_vals):
        # Clip and rescale
        h_vals = np.clip(h_vals, 0, 1)
        # Ensure sum = n_points / 2
        h_vals = h_vals * (n_points / 2) / np.sum(h_vals)
        return h_vals
    
    def descend_step(h_vals, lr=1e-3, reg=1e-4):
        # Compute C5 and gradient approximation
        c5 = compute_C5(h_vals)
        # Gradient approximation via finite differences
        grad = np.zeros_like(h_vals)
        eps = 1e-4
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            c5_plus = compute_C5(proj_h(h_plus))
            grad[i] = (c5_plus - c5) / eps
        # Regularization
        grad += 2 * reg * h_vals
        # Update
        h_new = h_vals - lr * grad
        return proj_h(h_new)
    
    def shift_search(h_vals, num_shifts=20):
        # Local search over shifts
        best_h = h_vals.copy()
        best_c5 = compute_C5(h_vals)
        for _ in range(num_shifts):
            if time.time() - start_time > budget_s * 0.95:
                break
            shift = np.random.uniform(-0.1, 0.1, len(h_vals))
            h_trial = h_vals + shift
            h_trial = proj_h(h_trial)
            c5_trial = compute_C5(h_trial)
            if c5_trial < best_c5:
                best_h = h_trial
                best_c5 = c5_trial
        return best_h, best_c5
    
    # Main loop
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Gradient descent step
        h = descend_step(h, lr=5e-2, reg=1e-4)
        # Local shift search
        h, c5 = shift_search(h)
        # Early stopping if improvement is small
        if i > 10 and i % 5 == 0:
            h = proj_h(h)
            current_c5 = compute_C5(h)
            if current_c5 < 0.3807:
                break

    h = proj_h(h)
    final_c5 = compute_C5(h)
    # Ensure we don't exceed budget
    if time.time() - start_time > budget_s * 0.98:
        pass  # Already at limit
        
    return (h.tolist(), final_c5, len(h))
