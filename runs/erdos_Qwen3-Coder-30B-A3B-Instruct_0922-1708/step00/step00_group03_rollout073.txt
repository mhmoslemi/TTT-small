import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    dx = 2.0 / len(h)
    # Compute the correlation of h with (1 - h)
    corr = correlate(h, 1 - h, mode="full")
    max_corr = np.max(corr)
    return max_corr * dx

def initialize_population(n_points, population_size, initial_h):
    """Initialize the population with one known good solution and random ones."""
    population = [initial_h.copy()]
    for _ in range(population_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def mutate_h(h, mutation_rate=0.2, step_size=0.1):
    """Mutate the step function h with given probability and step size."""
    h = h.copy()
    if np.random.rand() < mutation_rate:
        n_points = len(h)
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h[i] += np.random.uniform(-step_size, step_size) * (2.0 / n_points)
        h = np.clip(h, 0, 1)
        # Rescale to maintain constraint
        h = h * (n_points / 2) / np.sum(h)
    return h

def crossover_h(h1, h2):
    """Perform crossover between two step functions."""
    n_points = len(h1)
    if np.random.rand() < 0.5:
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([h1[:point], h2[point:]])
        child2 = np.concatenate([h2[:point], h1[point:]])
        return child1, child2
    else:
        child1 = h1.copy()
        child2 = h2.copy()
        return child1, child2

def tournament_selection(population, fitnesses, tournament_size=0.2):
    """Select parents using tournament selection."""
    n_parents = int(len(population) * tournament_size)
    selected_indices = np.random.choice(len(population), size=n_parents, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in selected_indices]
    best_indices = np.argsort(tournament_fitnesses)[:len(population)//2]
    return [population[selected_indices[i]] for i in best_indices]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    population_size = 100
    max_generations = 300
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.random.rand(n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        
    population = initialize_population(n_points, population_size, initial_h)
    best_h = initial_h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Evolution loop
    no_improvement_count = 0
    for gen in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        # Find best so far
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_h = population[current_best_idx].copy()
            best_c5 = fitnesses[current_best_idx]
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        if no_improvement_count >= 50:
            break
        # Selection
        selected_parents = tournament_selection(population, fitnesses)
        # Create new population
        new_population = [best_h.copy()]
        while len(new_population) < population_size:
            if len(selected_parents) < 2:
                break
            parent1, parent2 = np.random.choice(selected_parents, 2, replace=False)
            child1, child2 = crossover_h(parent1, parent2)
            child1 = mutate_h(child1)
            child2 = mutate_h(child2)
            new_population.extend([child1, child2])
        population = new_population[:population_size]
        
    return (best_h.tolist(), best_c5, n_points)
