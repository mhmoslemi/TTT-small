import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral for given h
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100  # Reasonable balance
    dx = 2.0 / n_points
    max_iter = 500
    swarm_size = 50
    w = 0.7
    c1, c2 = 2.0, 2.0
    
    # Initialize swarm
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = project_h(h_start, n_points)
        swarm = [h_start.copy() for _ in range(swarm_size)]
        for i in range(1, swarm_size):
            noise = np.random.normal(0, 0.05, n_points)
            swarm[i] = project_h(swarm[0] + noise, n_points)
    else:
        swarm = []
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            swarm.append(h)
    
    # Initialize personal bests
    personal_best = [swarm[i].copy() for i in range(swarm_size)]
    personal_best_values = [evaluate_h(h, dx) for h in personal_best]
    
    # Initialize global best
    global_best_idx = np.argmin(personal_best_values)
    global_best = personal_best[global_best_idx].copy()
    global_best_value = personal_best_values[global_best_idx]
    
    # PSO parameters
    velocities = [np.zeros(n_points) for _ in range(swarm_size)]
    
    start_time = time.time()
    stagnation = 0
    diversity_reset = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break

        # Update all particles
        for i in range(swarm_size):
            # Compute current overlap
            current_value = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
            
            # Update global best
            if current_value < global_best_value:
                global_best = swarm[i].copy()
                global_best_value = current_value
                stagnation = 0
            else:
                stagnation += 1
            
            # Velocity update
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Position update
            swarm[i] = swarm[i] + velocities[i]
            
            # Projection back to feasible region
            swarm[i] = project_h(swarm[i], n_points)
        
        # Diversity injection if stuck
        if stagnation > 25:
            diversity_reset += 1
            for i in range(int(0.2 * swarm_size)):
                idx = np.random.randint(0, swarm_size)
                noise = np.random.normal(0, 0.1, n_points)
                swarm[idx] = project_h(swarm[idx] + noise, n_points)
            stagnation = 0
            
            if diversity_reset > 5:
                break
    
    # Final evaluation
    final_value = evaluate_h(global_best, dx)
    return (global_best.tolist(), final_value, n_points)
