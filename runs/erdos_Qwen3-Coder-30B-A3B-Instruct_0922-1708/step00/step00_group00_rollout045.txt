import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute overlap integral for all shifts
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of these values
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to match the target sum
        h *= target_sum / current_sum
    return h

    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iters = 1000000
    max_no_improvement = 1000

    else:

    h_current = project_h(h_current, n_points)
    c5_current = evaluate_erdos_solution(h_current)
    best_h = h_current.copy()
    best_c5 = c5_current
    
    start_time = time.time()
    no_improvement_count = 0
    iteration = 0
    
    # Decreasing variance for perturbation
    sigma_start = 0.1
    sigma_decay = 0.99
    
    while time.time() - start_time < budget_s and iteration < max_iters:
        iteration += 1
        
        # Generate noise
        noise = np.random.normal(0, sigma_start * (sigma_decay ** iteration), n_points)
        h_new = h_current + noise
        
        # Project the new h to feasible set
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        
        # Accept if better
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            # Accept with some probability based on how bad it is
            # Not part of the algorithm due to simplicity
            if no_improvement_count > max_no_improvement:
                # Restart with a new random perturbation if stuck
                h_current = best_h + np.random.normal(0, 0.01, n_points)
                h_current = project_h(h_current, n_points)
                no_improvement_count = 0
            else:
                h_current = h_new
