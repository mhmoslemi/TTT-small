import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Normalize to satisfy sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    # Parameters
    max_iter = 10000
    tau = 0.1  # temperature for smooth max
    alpha_init = 0.1
    beta = 0.5
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vals):
        dx = 2.0 / len(h_vals)
        # Compute all overlaps for all shifts
        shifted = 1 - h_vals
        overlaps = np.correlate(h_vals, shifted, mode='full')
        # The overlaps are indexed from -n+1 to n-1
        # We're interested in max k, which is from 0 to n-1 (positive shifts)
        # subset = overlaps[n-1:]  # For shifts k=0 to n-1
        # Actually, full correlation gives us all valid shifts
        # But for the overlap integral, we want shifts from 0 to n-1
        # Since we consider h(x) * (1 - h(x+k)), we look at right-aligned correlation
        # The full correlation center is at n-1
        # Positive shifts k=0..n-1 correspond to indices n-1 .. 2*n-2
        # This means the actual max is from indices n-1 to 2*n-2
        # But this includes both positive and negative, we only want positive
        # However, actually, we compute for ALL shifts and take the max
        
        # Just ensure we get positive shifts according to problem definition
        # The actual max in the discrete case for shift 0 to n-1
        # You need to take transposed view of the correlation correctly
        # Let's redo this properly:
        # Due to the nature of convolution in scipy, with mode='full':
        # The indices correspond to shift -n+1 to n-1
        # The shift k goes from 0 to n-1 corresponds to index range [2*n-2, n-1]
        # But that's backwards, let us be more careful:

        # For h(x) * (1 - h(x + k)), if h has n points:
        # for shift k = 0, overlap = sum(h[i] * (1 - h[i])) = sum(h[i] - h[i]^2)
        # for shift k = 1, overlap = sum(h[i] * (1 - h[i+1]))
        # The correlation gives us:
        # indices 0..n-1 for k = -n+1..0
        # indices n..2n-2 for k = 1..n-1
        # So k from 0..n-1 is indices n-1..2n-2
        # But we're only interested in k=0 to n-1, shifts for overlapping regions
        # Or more correctly: all valid starts of both functions
        
        # Let's compute how many times each k is valid
        # if we have h[n] and shift it k positions, the overlap is:
        # sum for i = 0 to n-1-k  of h[i] * (1 - h[i+k]) 
        # which if k starts at 0, maximum is k = n-1
        # Also considering k ranges from 0 to n-1 (beyond that not much overlap)
        
        # So compute the actual overlap values for valid k = 0..n-1
        # Need indices n-1 to 2*n-2 from full correlation (i.e., shift k=0 to n-1)
        max_overlap = np.max(overlaps[n_points-1:2*n_points-1])
        return max_overlap * (2.0 / n_points)  # dx = 2/n

    def compute_loss_and_grad(h_vals):
        # Compute C5 and its gradient
        dx = 2.0 / len(h_vals)
        # Correlation h(x) * (1 - h(x+k)) for all shifts k
        # This effectively computes sums for t = 0 to n-1:
        # sum_i h[i] * (1 - h[i+t]) = sum_i h[i] - sum_i h[i] * h[i+t]
        shifted = 1 - h_vals
        overlaps = correlate(h_vals, shifted, mode="full")
        # The indices go from -n+1 to n-1
        # We need overlaps for shifts from 0 to n-1 which are indices 
        # n-1 (k=0) to 2*n-2 (k=n-1)
        valid_overlaps = overlaps[n_points-1:2*n_points-1]  # shift 0 to n-1
        
        # Smoothed max of overlaps with temperature tau
        # L = (1/tau) * log(sum(exp(tau * overlap)))
        max_o = np.max(valid_overlaps)
        exp_term = np.exp(tau * (valid_overlaps - max_o))
        Z = np.sum(exp_term)
        loss = (max_o + np.log(Z) / tau) * dx
        
        # Gradient computation
        # Take: d/dh_i of (1/tau) * log(sum_k exp(tau * sum_j h_j * (1-h_{j+k}))) 
        # The partial w.r.t. h_i:
        # Let inner = sum_j h_j * (1-h_{j+k}) for one shift k
        # d(h_i (1-h_{i+k}))/h_i = (1-h_{i+k}) (for k=0), or -h_{i-k} for k>0  
        # But we have to be very careful about indexing
        
        # Let me simplify:
        # L = (1/tau) * log(sum_{k=0}^{n-1} exp(tau * o_k))
        # dL/dh_i = (1/tau) * (sum_k exp(tau*o_k) * d_o_k / dh_i) / (sum_k exp(tau*o_k))
        # d_o_k / dh_i = d/dh_i (sum_j h_j * (1-h_{j+k})) 
        #  = (1-h_{i+k}) 
        # Thus we get:
        # dL/dh_i = (1/tau) * (sum_k exp(tau*o_k) * (1-h_{i+k}) )  / Z
        
        grad = np.zeros_like(h_vals)
        for k in range(n_points):
            # core of overlap_k is sum_i h_i (1 - h_{i+k})  
            # d_o_k / dh_i = (1 - h_{i+k})  if i+k < n
            # d_o_k / dh_i = 0 otherwise  
            # we need to compute d/dh_i sum_{k} exp(tau * o_k)
            # 
            # Just compute df/du * du/dh_i for f = exp(tau * o_k)
            # Consider: (grad[i] += 1/tau * normalizer * tau * exp(tau*o_k) * (1-h_{i+k}))
            # 
            # A simpler implementation is to compute d_o_k / dh_i directly:
            # For all k, overlap_k = sum_{i=0}^{n-1-k} h[i] * (1-h[i+k])
            # df/dh_i for k from 0 to n-1:
            # If we're outside valid range, 0
            # Otherwise, (1-h_{i+k})
            # 
            # So for i-th element:
            # Let T_i = sum_k exp(tau*o_k) * (1-h_{i+k}) / (Z * tau)  
            # Where terms are only positive when i+k < n 
            # This corresponds to sum over k where k <= n-1-i
            
            # This technique combines all contributions:
            
            # This computation needs to be carefully done for each element
            # Let's rewrite cleanly:
            # Contribution of shift k to gradient of element i:
            # We must compute d (h_i * (1-h_{i+k})) / d h_i = (1-h_{i+k})
            # To build the complete gradient for all i:
            # grad[i] += (1/tau) * (exp(tau * overlap_k) * (1-h_{i+k})) / Z
            
            # Now how to do fast vectorized:
            # For shift k = 0 to n-1:
            # The contribution of term (1 - h_{i+k}) for valid k of the overlap_sum_k to all h_i
            # Element w_i of gradient for overlap_k = sum_{j: j+k < n} h_j * (1-h_{j+k})
            # Grad of i-th element w.r.t. h_i:
            # Only the terms H_i * (1 - H_{i+k}) contribute via chain rule where k was chosen appropriately
            
            # So, loop among k:
            # set img to zeros
            # For specific k, img[i] += exp(tau*overlap_k)/(Z*tau) * (1 - h[i+k]) if valid
            img_idx = n_points - 1 + k  # For correlation index of overlap k 
            if img_idx >= len(overlaps): 
                continue
            norm_factor = np.exp(tau * valid_overlaps[k]) / (Z * tau)
            if k == 0:  # offset in array
                # The term x(i)*1 of overlap_k is d/dh_i when k=0
                grad += norm_factor * (1 - h_vals)
            else:  # k > 0
                # overlap_k = h[i]*1 + h[i+1]*(1-h[i+k]) + ...
                # We're looking at the contribution of (1-h[i+k]) term (in terms of h[i] term)
                # Natural to think of convolution in the chain rule
                # But easier simple labeling
                # For contribution for k, we add: 
                # (1 - h[i + k]) if i + k < n and i bounds
                # So: take 1-h_vals[k:] and make it the "k" component, then shuffle
                right_shift_plus_k = np.concatenate((np.zeros(k), 1 - h_vals[:-k]))
                grad += norm_factor * right_shift_plus_k
        
        # You see, my matrix-vector calculation needs to be smarter 
        # Better approach:
        # Since we have all overlaps computed, and we compute grad via:
        # For each k: gradient of h[i] w.r.t. o_k leads to:
        # Contribution = (1 - h[i + k]) for valid indexes
        # It's better:
        # For each k: 1-overlap-k, 1,h[0], h[1],... then you slide for k shift
        # OR
        # You compute W = [0, ..., 0, 1-h[0], 1-h[1], ..., 1-h[n-1]] for shift k=0,
        # and shift downward by k steps, then replicate and sum
        
        # Compute as 1 against shift
        # Effective for all k=0 to n-1:
        # X_k = [1-h[i+k]] if valid, 0 else
        # so compute matrix M_{i,k} of these, and sum over K 
        grad = np.zeros_like(h_vals)
        for k in range(n_points):
            if k == 0:
                grad += (1 - h_vals) * exp_term[k] / (Z * tau)
            else:
                temp = np.concatenate((np.full(k, 0), 1 - h_vals[:-k]))
                grad += temp * exp_term[k] / (Z * tau)
        
        return loss, grad

    def backtracking_line_search(h_vals, grad, alpha0=alpha_init):
        # Simple backtracking
        alpha = alpha0
        loss_now = compute_c5(h_vals) 
        while True:
            h_new = h_vals - alpha * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            if compute_c5(h_new) < loss_now or alpha < 1e-8:
                return alpha
            alpha *= beta

    # Optimization loop
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        loss, grad = compute_loss_and_grad(h)
        grad = np.clip(grad, -100, 100)  # Clip gradients
        
        # Optimized step selection:
        alpha = backtracking_line_search(h, grad)
        
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        # Normalize to sum constraint:
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = i
        
        h = h_new
        # Early stopping
        if i - last_improvement > patience:
            break

    # Final return
    return (best_h.tolist(), best_c5, n_points)
