import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create basic initial configuration
        n_points = 47
        h = np.zeros(n_points)
        # Create symmetric plateau
        mid = n_points // 2
        width = n_points // 3
        h[mid - width//2:mid + width//2] = 1.0
        # Normalize to sum = n_points/2
        h *= (n_points / 2) / np.sum(h)

    # Discretization details
    dx = 2.0 / n_points
    
    # Parameters
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    conv_tol = 1e-5
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags (as in evaluation)
        overlaps = np.correlate(h, 1 - h, mode="full")
        overlaps = overlaps * dx  # Scale by dx
        
        # Smooth approximation to max
        c5_approx = np.log(np.sum(np.exp(tau * overlaps))) / tau
        
        # Compute gradient of smooth C5 w.r.t. h
        exp_terms = np.exp(tau * overlaps)
        denom = np.sum(exp_terms)
        
        # For each h[i], compute dC5/dh[i] using chain rule
        grad = np.zeros_like(h)
        for i in range(n_points):
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    j = (i - k) % n_points
                else:
                    j = (i + abs(k)) % n_points
                if 0 <= j < n_points:
                    # Derivative of overlap(k) w.r.t h[i] is - (1 - h[j])
                    # But here we compute gradient of the entire sum
                    shift = k + n_points - 1
                    grad[i] += exp_terms[shift] * (- (1 - h[j])) * dx / denom
        
        # Update h in direction of gradient
        h -= learning_rate * grad
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Enforce sum constraint
        h *= (n_points / 2) / np.sum(h)
        
        # Compute actual C5
        c5_actual = compute_c5(h, dx)
        
        if c5_actual < best_c5:
            best_c5 = c5_actual
            best_h = h.copy()
        
        # Anneal temperature
        tau *= anneal_factor
        
        # Adaptive adjustments
        if c5_actual > best_c5 * 1.01:
            learning_rate *= 0.9
        if np.linalg.norm(grad) < conv_tol:
            break
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    """Compute exact C5 value."""
    n_points = len(h)
    # Compute all overlaps for all lags
    overlaps = np.correlate(h, 1-h, mode="full")
    overlaps = overlaps * dx
    return np.max(overlaps)
