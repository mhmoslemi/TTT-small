import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Constants
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize from initial_h_values if available, otherwise random
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.random.rand(n_points)
    
    # Project initial h to satisfy constraint sum(h)*dx = 1
    h_init = h_init * target_sum / np.sum(h_init)
    
    # Fourier-based optimization parameters
    n_modes = 10  # Keep low for efficiency
    
    # Initial coefficients: a_k for k=0 to n_modes-1
    # a_0 required to ensure integral constraint
    # We optimize a_1, ..., a_{n_modes-1}  
    # Set a_0 = 0.5 to satisfy integral constraint when sum(h) = target_sum
    
    # Initialize from projected h
    # Use cosine transform with even symmetry
    # Represent h(x) = a_0 + sum_{k=1}^{n_modes-1} a_k*cos(k*pi*x/2)
    a = np.zeros(n_modes)
    # Ensure integral constraint by setting a_0 = 0.5
    a[0] = 0.5
    
    # Optimize using scipy
    def func(coeffs):
        # Reshape for computation
        a_coeffs = coeffs.copy()
        a_coeffs[0] = 0.5  # Fix integral constraint
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points, endpoint=False)
        for k in range(n_modes):
            h += a_coeffs[k] * np.cos(k * np.pi * x / 2)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        h = h * (target_sum / h_sum)  # rescale to sum = target_sum
        
        # Compute overlap
        C = 0.0
        for k in range(1, 2 * n_points):  # domain of correlation
            corr = np.correlate(h, 1 - h, mode='full')
            C = max(C, corr[k] * dx)
        return C

    def jac(coeffs):
        # Simple finite difference
        eps = 1e-6
        grad = np.zeros_like(coeffs)
        base_val = func(coeffs)
        for i in range(len(coeffs)):
            coeffs_idx = coeffs.copy()
            coeffs_idx[i] += eps
            grad[i] = (func(coeffs_idx) - base_val) / eps
        return grad

    # Optimization
    best_coeff = a.copy()
    best_C = float('inf')
    learning_rate = 0.01
    patience = 0
    max_patience = 100

    # Start from the smoothed initial h projection
    coeffs = a.copy()[1:]  # Remove a[0] since it's fixed
    
    while time.time() - start_time < budget_s:
        # Evaluate current
        current_C = func(np.concatenate([[0.5], coeffs]))
        if current_C < best_C:
            best_C = current_C
            best_coeff = coeffs.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                break

        # step size for descent
        step = learning_rate * jac(coeffs)
        
        # update
        coeffs -= step
        
        # clip
        coeffs = np.clip(coeffs, 0, 1)
        
        # decay learning rate
        if patience > 50:
            learning_rate *= 0.95

    # Final h construction
    a_final = np.concatenate([[0.5], best_coeff])
    h_final = np.zeros(n_points)
    x = np.linspace(0, 2, n_points, endpoint=False)
    for k in range(n_modes):
        h_final += a_final[k] * np.cos(k * np.pi * x / 2)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (target_sum / np.sum(h_final))
    
    return (h_final.tolist(), best_C, n_points)
