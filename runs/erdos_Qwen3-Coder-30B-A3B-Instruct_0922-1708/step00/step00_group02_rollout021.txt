import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
    
    # Optimization parameters
    max_iter = 3000
    lr = 0.01
    beta = 0.9
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute
    h_old = np.zeros_like(h)
    momentum = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute correlation: h * (1 - h) shifted
        corr = signal.correlate(h, 1 - h, mode='full')
        # Only look at valid shifts [0, 2] = [0, n_points]
        corr = corr[n_points-1:2*n_points-1]
        
        # Find k that maximizes the overlap
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]
        
        # Update best if improved
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early stopping
        if patience_counter > patience:
            break
            
        # Compute gradient w.r.t h
        # grad = (1 - h) - h * (shifted gradient computed via convolution)
        # We compute gradient as simplified expression
        h_grad = (1 - h) - h * (1 - h)
        
        # Apply momentum
        momentum = beta * momentum + (1 - beta) * h_grad
        h = h - lr * momentum
        
        # Project h to feasible set
        h = np.clip(h, 0, 1)
        h = h * (target_sum / np.sum(h))
        
        # Small step size decay
        if i % 100 == 0 and i > 0:
            lr *= 0.95
    
    # Final evaluation of best solution
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    corr = corr[n_points-1:2*n_points-1]
    final_c5 = np.max(corr) * dx
    
    return (best_h, final_c5, n_points)
