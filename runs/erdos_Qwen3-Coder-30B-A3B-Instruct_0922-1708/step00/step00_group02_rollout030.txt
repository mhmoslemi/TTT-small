import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    h = h * (n_points / 2) / np.sum(h)  # Normalize sum to n_points/2
    
    # Optimization parameters
    alpha = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlap = correlate(h, 1 - h, mode='full')
        overlap = overlap * dx
        k_max = np.argmax(overlap)
        c5_current = overlap[k_max]
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Gradient computation for the worst case k_max
        grad = np.zeros(n_points)
        k = k_max - (n_points - 1)  # Offset to get actual shift
        for j in range(n_points):
            if 0 <= j + k < n_points:
                grad[j] += (1 - h[j + k]) - h[j]
        
        # Update
        h = h - alpha * grad
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if i % patience == 0 and i > 0:
            alpha *= 0.5
    
    return (best_h.tolist(), best_c5, n_points)
