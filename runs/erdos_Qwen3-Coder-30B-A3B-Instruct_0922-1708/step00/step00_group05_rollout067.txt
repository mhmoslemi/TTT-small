import numpy as np
import time
import scipy.signal
from scipy.optimize import minimize
import math


def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C_5 for given step function h"""
    dx = 2.0 / len(h)
    # Using full convolution to compute max overlap
    correlation = scipy.signal.correlate(h, 1 - h, mode="full")
    # Extract the valid range of correlations
    mid = len(h) - 1
    correlations = correlation[mid - len(h) + 1:mid + len(h)]
    return np.max(correlations) * dx


def project_and_normalize(h, target_sum):
    """Enforce constraint that sum(h) = target_sum"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    # Linear rescaling to hit target
    if current_sum != 0:
        h = h * target_sum / current_sum
    return h


def clip_to_bounds(h):
    """Clamp h to [0, 1]"""
    return np.clip(h, 0, 1)


def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Get initial vector or construct one
    if 'initial_h_values' in globals():

        n_points = len(h_base)
    else:
        n_points = 70
        h_base = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points * 0.5  # sum should be n/2
    
    # Normalize base vector
    h_base = project_and_normalize(h_base, target_sum)
    h_base = clip_to_bounds(h_base)
    
    # Initialize PSO parameters
    n_particles = 20
    max_iter = int(budget_s / 0.05)  # approx 12000 iterations
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    
    # Particle positions and velocities
    particles = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Initialize particles
    particles.append(h_base.copy())
    personal_best.append(h_base.copy())
    personal_best_values.append(evaluate_erdos_solution(h_base))
    
    for i in range(1, n_particles):
        # Small random perturbation
        h_p = h_base + np.random.normal(0, 0.1, n_points)
        h_p = clip_to_bounds(h_p)
        h_p = project_and_normalize(h_p, target_sum)
        particles.append(h_p)
        velocity = np.random.uniform(-0.1, 0.1, n_points)
        velocities.append(velocity)
        value = evaluate_erdos_solution(h_p)
        personal_best.append(h_p.copy())
        personal_best_values.append(value)
    
    # Find global best
    best_idx = np.argmin(personal_best_values)
    global_best = personal_best[best_idx].copy()
    global_best_value = personal_best_values[best_idx]
    
    # PSO iterations
    iter_count = 0
    no_improve_iter = 0
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        iter_count += 1
        no_improve_iter += 1
        
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            p_best = personal_best[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (w * v +
                 c1 * r1 * (p_best - h) +
                 c2 * r2 * (global_best - h))
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            
            # Ensure bounds
            h_new = clip_to_bounds(h_new)
            
            # Re-normalize to meet constraint
            h_new = project_and_normalize(h_new, target_sum)
            
            particles[i] = h_new
            
            # Evaluate new position
            new_value = evaluate_erdos_solution(h_new)
            
            # Update personal best
            if new_value < personal_best_values[i]:
                personal_best_values[i] = new_value
                personal_best[i] = h_new.copy()
                # Update global best if needed
                if new_value < global_best_value:
                    global_best = h_new.copy()
                    global_best_value = new_value
                    no_improve_iter = 0
        
        # Tournament selection - keep top 50%
        top_indices = np.argsort(personal_best_values)[:n_particles//2]
        new_particles = [personal_best[i] for i in top_indices]
        new_velocities = [velocities[i] for i in top_indices]
        new_pbest_values = [personal_best_values[i] for i in top_indices]
        
        # Fill the rest with new random particles
        while len(new_particles) < n_particles:
            h_p = global_best + np.random.normal(0, 0.05, n_points)
            h_p = clip_to_bounds(h_p)
            h_p = project_and_normalize(h_p, target_sum)
            new_particles.append(h_p)
            new_velocities.append(np.random.uniform(-0.05, 0.05, n_points))
            new_pbest_values.append(evaluate_erdos_solution(h_p))
        
        particles = new_particles
        velocities = new_velocities
        personal_best_values = new_pbest_values
        personal_best = [particles[i] for i in range(n_particles)]
        
        # Occasionally reintroduce diversity if no improvement
        if no_improve_iter > 100 and iter_count < max_iter - 100:
            for i in range(n_particles):
                h_p = particles[i] + np.random.normal(0, 0.01, n_points)
                h_p = clip_to_bounds(h_p)
                h_p = project_and_normalize(h_p, target_sum)
                particles[i] = h_p
                velocities[i] = np.clip(velocities[i], -0.1, 0.1)
                personal_best[i] = particles[i]
                personal_best_values[i] = evaluate_erdos_solution(particles[i])
            no_improve_iter = 0
    
    # Final check of the global best
    final_h = global_best
    final_c5 = evaluate_erdos_solution(final_h)
    
    return (final_h.tolist(), float(final_c5), int(n_points))
