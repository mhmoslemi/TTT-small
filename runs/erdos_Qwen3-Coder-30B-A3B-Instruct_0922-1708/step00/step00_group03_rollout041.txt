def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    correlations = np.correlate(h, 1-h, mode="full")
    # Maximum value is the peak overlap (since we want max_k)
    max_overlap = np.max(correlations) * dx
    return max_overlap

        for k in range(1, n_points):
            # Move h by k positions (circular)
            h_shifted = np.roll(h, k)
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
        
        if len(overlaps) == 0:
            break

        pr = np.array(overlaps)
        pr = np.exp(pr - np.max(pr))  # avoid numerical issues

        grad = np.zeros_like(h)

            h_backward = np.roll(h, k)  # shifted back by k
            grad += pr[k-1] * (1 - h_backward)  # note index adjustment

        # Update h using gradient descent
        h_new = h - eta * grad
        
        # Clip h to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to sum = target_sum
        current_sum_new = np.sum(h_new)
