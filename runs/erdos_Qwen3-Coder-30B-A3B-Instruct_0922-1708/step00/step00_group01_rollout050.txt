import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from provided values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create symmetric plateau initialization
        n_points = 47
        h = np.ones(n_points) * 0.5
        midpoint = n_points // 2
        h[midpoint-5:midpoint+5] = 0.8  # Slightly higher in middle
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to satisfy sum constraint
    h = h * target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # SGD parameters
    learning_rate = 0.1
    anneal_every = 1000
    max_iter = 5000
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlaps = correlate(h, 1 - h, mode='full')
        # Normalized overlap values for each lag (ignore zero lag)
        lags = np.arange(-len(h)+1, len(h))
        max_idx = np.argmax(overlaps)
        dominant_lag = lags[max_idx]
        
        # Compute gradient for dominant lag
        grad = np.zeros_like(h)
        if dominant_lag >= 0:
            grad[dominant_lag:] += -(1 - h[:-dominant_lag])
        else:
            grad[:dominant_lag] += -(1 - h[-dominant_lag:])
        
        # Update h
        h = h - learning_rate * grad
        h = np.clip(h, 0, 1)
        
        # Rescale to preserve sum
        h_sum = np.sum(h)
        h = h * target_sum / h_sum
        
        # Evaluate and store best
        current_c5 = compute_c5(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Anneal learning rate
        if i % anneal_every == 0 and i > 0:
            learning_rate *= 0.5
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Compute the overlap integral using cross-correlation
    overlaps = correlate(h, 1 - h, mode='full')
    # Extract valid overlaps (excluding the zero-lag which isn't considered)
    # We want max of positive lags
    max_overlap = np.max(overlaps)
    return max_overlap * dx
