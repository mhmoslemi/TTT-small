import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45  # Use the initial construction size if available
    dx = 2.0 / n_points
    max_generations = 120
    population_size = 100
    elite_size = population_size // 5
    mutation_rate = 0.05
    crossover_prob = 0.8
    tournament_size = 5
    
    # Initialize population
    def initialize_individual():
        h = np.random.rand(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    population = []
    
    # Add initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Fill remaining population with random individuals
    while len(population) < population_size:
        h = initialize_individual()
        population.append(h)
    
    def evaluate_fitness(h):
        # Compute overlaps with 1-h via correlation
        corr = correlate(h, 1 - h, mode='full')
        # Take only relevant correlations at k=0 to n_points-1
        overlaps = corr[n_points-1:] * dx
        score = np.max(overlaps)
        return score
    
    def select_parents(population, fitnesses):
        parents = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
            parents.append(population[winner_index].copy())
        return parents
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_prob:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Ensure constraints
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    def mutate(h):
        # Apply small random perturbations to each element
        mutated = h + np.random.normal(0, 0.05, n_points)
        mutated = np.clip(mutated, 0, 1)
        # Re-normalize
        mutated = mutated * (n_points / 2) / np.sum(mutated)
        return mutated
    
    generation = 0
    best_fitness = float('inf')
    best_individual = None
    
    # Main evolution loop
    while generation < max_generations:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 1:  # Reserve 1s for cleanup
            break
            
        # Evaluate fitness for entire population
        fitnesses = [evaluate_fitness(h) for h in population]
        
        # Find best in current generation
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_individual = population[current_best_idx].copy()
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elite = [population[i].copy() for i in elite_indices]
        
        # Selection
        parents = select_parents(population, fitnesses)
        
        # Create new population
        new_population = elite.copy()
        
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(parents, 2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            
            if np.random.rand() < mutation_rate:
                child1 = mutate(child1)
            if np.random.rand() < mutation_rate:
                child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        population = new_population[:population_size]
        generation += 1
    
    # Final check on best individual
    if best_individual is None:
        # Fallback: evaluate all individuals one more time
        final_fitnesses = [evaluate_fitness(h) for h in population]
        final_best_idx = np.argmin(final_fitnesses)
        best_individual = population[final_best_idx].copy()
        best_fitness = final_fitnesses[final_best_idx]
    
    return (best_individual, best_fitness, n_points)
