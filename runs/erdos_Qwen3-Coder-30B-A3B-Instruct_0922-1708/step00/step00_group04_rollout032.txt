    def smooth_max(arr, beta=100.0):
        return np.log(np.sum(np.exp(beta * arr))) / beta

    # Define surrogate objective and its gradient
    def surrogate_objective(h):
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        corr = corr[len(corr)//2:]  # Only positive lags
        return smooth_max(corr)

    def surrogate_gradient(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')

        # This is simplified: actually use finite differences or explicit formulas

        eps = 1e-6
        grad = np.zeros_like(h)

            h_plus = h.copy()
            h_plus[i] += eps
            h_minus = h.copy()
            h_minus[i] -= eps

        grad = surrogate_gradient(h)

def compute_c5(h, dx):
    # Compute overlap integral using FFT for efficiency
    corr = np.correlate(h, 1 - h, mode='full')
    # Take only the positive lags
    corr = corr[len(corr)//2:]
    return np.max(corr) * dx
