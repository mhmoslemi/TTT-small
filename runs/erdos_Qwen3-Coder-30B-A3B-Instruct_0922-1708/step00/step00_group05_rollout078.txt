import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = len(initial_h_values) if 'initial_h_values' in globals() else 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h from initial_h_values if available, otherwise create a simple one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to target_sum
    h = h * target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Compute initial C5
    def compute_C5(h_vals):
        h_vals = np.array(h_vals)
        corr = signal.correlate(h_vals, 1 - h_vals, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    c5_bound = compute_C5(h)
    best_h = h.copy()
    best_c5 = c5_bound
    
    # Perturbation settings
    max_perturb = 0.05
    num_trials = 100
    smoothing_window = 3
    
    # Track time
    start_time = time.time()
    
    iterations = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s * 0.95:
        iterations += 1
        if no_improvement_count > 100:
            break
            
        improved = False
        
        for _ in range(num_trials):
            # Make a copy
            h_new = h.copy()
            
            # Randomly select indices to perturb
            num_perturbed = np.random.randint(1, max(1, n_points // 5))
            indices = np.random.choice(n_points, num_perturbed, replace=False)
            
            # Perturb values
            for idx in indices:
                perturb = np.random.uniform(-max_perturb, max_perturb)
                h_new[idx] += perturb
            
            # Ensure bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Adjust to maintain sum constraint
            current_sum = np.sum(h_new)
            diff = target_sum - current_sum
            if abs(diff) > 1e-10:
                # Distribute the difference
                avg_change = diff / n_points
                h_new += avg_change
                h_new = np.clip(h_new, 0, 1)
            
            # Smoothing
            if smoothing_window > 0 and n_points > smoothing_window:
                h_smooth = np.convolve(h_new, np.ones(smoothing_window)/smoothing_window, mode='same')
                h_new = h_smooth
            
            # Evaluate
            new_c5 = compute_C5(h_new)
            
            if new_c5 <= c5_bound:
                h = h_new
                c5_bound = new_c5
                improved = True
                no_improvement_count = 0
                if new_c5 < best_c5:
                    best_c5 = new_c5
                    best_h = h.copy()
            else:
                no_improvement_count += 1
        
        if not improved:
            no_improvement_count += 1
    
    return (list(best_h), best_c5, n_points)
