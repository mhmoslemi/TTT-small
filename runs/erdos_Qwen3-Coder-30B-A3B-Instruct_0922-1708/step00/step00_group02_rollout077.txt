def evaluate_c5(h_values):
    """Evaluate the C5 constant for given h_values."""
    n = len(h_values)
    dx = 2.0 / n
    # Compute the correlation of h with (1-h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # The maximum of the correlation is the C5 value
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h[i] <= 1 and sum(h) = n_points/2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to correct sum
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
