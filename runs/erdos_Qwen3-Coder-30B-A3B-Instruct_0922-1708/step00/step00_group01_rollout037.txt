import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given h values."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    # The maximum value of the correlation corresponds to the maximum overlap
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to ensure sum constraint and bounds."""
    # Ensure bounds
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        target_sum = n_points / 2.0
        h = h * target_sum / current_sum
    return h

def init_h(n_points):
    """Initialize h values with a relatively flat profile."""
    h = np.ones(n_points) * 0.5
    # Add slight tapering at the ends
    for i in range(n_points):
        taper = min(i, n_points - 1 - i)
        h[i] *= (1.0 - 0.4 * taper / (n_points / 2.0))
    return project_h(h, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial if available and valid
    if 'initial_h_values' in globals():

        n_points = len(h)
        if n_points == 47:
            # Validate constraints
            if not (0 <= np.all(h) <= 1 and abs(np.sum(h) - n_points / 2.0) < 1e-10):
                h = init_h(n_points)
        else:
            h = init_h(n_points)
    else:
        n_points = 47
        h = init_h(n_points)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Use a combination of gradient-free methods for better exploration
    def objective(h_vec):
        h = project_h(h_vec, n_points)
        return evaluate_c5(h, dx)
    
    # First, try to optimize with scipy's differential evolution
    try:
        # Limit iterations for faster runs
        de_res = differential_evolution(
            objective,
            bounds,
            maxiter=200,
            popsize=30,
            seed=seed,
            disp=False,
            timeout=budget_s - (time.time() - start_time) - 5,
        )
        if de_res.success:
            h_opt = project_h(de_res.x, n_points)
            c5_opt = evaluate_c5(h_opt, dx)
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
    except Exception:
        pass

    # Local refinement with Nelder-Mead if time remains
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 10:
        try:
            # Sample few good candidates from population if needed
            candidates = []
            for _ in range(20):
                h_trial = h + np.random.normal(0, 0.05, n_points)
                h_trial = project_h(h_trial, n_points)
                candidates.append((h_trial, evaluate_c5(h_trial, dx)))
            
            # Sort and get top performers
            candidates.sort(key=lambda x: x[1])
            top_h = candidates[0][0]
            
            # Refine locally
            res = minimize(objective, top_h, method='Nelder-Mead', options={'maxiter': 200, 'disp': False})
            if res.success:
                h_refined = project_h(res.x, n_points)
                c5_refined = evaluate_c5(h_refined, dx)
                if c5_refined < best_c5:
                    best_h = h_refined
                    best_c5 = c5_refined
        except Exception:
            pass

    # Final check
    final_h = project_h(best_h, n_points)
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
