import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize with initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.random.rand(n_points)
    
    h = np.clip(h, 0, 1)
    h = h * (target_sum / np.sum(h))
    
    # Optimization parameters
    max_modes = 10
    learning_rate = 0.01
    patience = 100
    no_improvement_count = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Fourier coefficients (symmetric)
    a = np.random.rand(max_modes) * 0.5  # a_1 to a_{max_modes}
    a[0] = 0.5  # a_0 = 1/2 to satisfy integral
    
    def project_to_h(a_coeffs):
        """Project Fourier coefficients back to h(x)"""
        x = np.linspace(0, 2, n_points, endpoint=False)
        h_vals = np.full(n_points, a_coeffs[0])
        for k in range(1, len(a_coeffs)):
            h_vals += a_coeffs[k] * np.cos(np.pi * k * x / 2)
        return np.clip(h_vals, 0, 1)
    
    def evaluate_c5(h_vals):
        """Evaluate maximum overlap using numpy correlate"""
        c5_vals = correlate(h_vals, 1 - h_vals, mode='full')
        max_c5 = np.max(c5_vals) * dx
        return max_c5
    
    def update_coefficients(a_coeffs, h_vals, current_c5):
        """Gradient-based update for coefficients"""
        # Compute gradient for one shift direction
        # Use finite differences
        eps = 1e-4
        grad = np.zeros_like(a_coeffs)
        for i in range(len(a_coeffs)):
            a_plus = a_coeffs.copy()
            a_plus[i] += eps
            h_plus = project_to_h(a_plus)
            c_plus = evaluate_c5(h_plus)
            
            a_minus = a_coeffs.copy()
            a_minus[i] -= eps
            h_minus = project_to_h(a_minus)
            c_minus = evaluate_c5(h_minus)
            
            grad[i] = (c_plus - c_minus) / (2 * eps)
            
        # Update with learning rate
        a_new = a_coeffs - learning_rate * grad
        # Ensure bounded [0,1]
        a_new = np.clip(a_new, 0, 1)
        return a_new
    
    # Iteration loop
    iteration = 0
    while time.time() - start_time < budget_s:
        h_vals = project_to_h(a)
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        
        # Evaluate C5
        c5 = evaluate_c5(h_vals)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vals.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= patience:
            break
            
        # Update coefficients
        a = update_coefficients(a, h_vals, c5)
        iteration += 1
        
        # Adjust learning rate if needed
        if iteration % 100 == 0 and no_improvement_count > 50:
            learning_rate *= 0.5
            
    return (best_h, best_c5, n_points)
