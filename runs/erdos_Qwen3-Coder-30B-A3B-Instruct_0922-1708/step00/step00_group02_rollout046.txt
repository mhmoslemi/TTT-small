from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):

    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points

    alpha = 0.01
    beta = 0.9

    patience_counter = 0

        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation: h * (1 - h)
        h1 = 1 - h
        corr = correlate(h, h1, mode='full')
        # Extract valid correlations for k in [0, 2]
        corr = corr[len(h1)-1:]
        k_max = np.argmax(corr)
        max_overlap = corr[k_max]

            best_c5 = max_overlap
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            alpha *= 0.5
            patience_counter = 0

        # We want to minimize max_k ∫ h(x)(1 - h(x+k)) dx
        # Gradient wrt h: d/dh [h(x)(1 - h(x+k))] = (1 - h(x+k)) - h(x) * (d/dh h(x+k))
        # Since h(x+k) is shifted, we get the derivative as shifted version
        # So grad = (1 - h) - h * (derivative of h decay)
        # But in fact, let's recompute by simulating exact gradient processing
        
        # === Refined Gradient Computation ===
        # h(x) * (1 - h(x+k)) contributes to overlap for each k
        # Let's find which k gave max overlap and compute gradient of that
        
        # If we fix k = k_max, we can approximate derivative in h space
        # df/dx = (1 - h(x+k)) - h(x) * (1 - h(x+k)) * (computed correctly)

        # df/dh[x] = (1 - h[x+k]) - h[x]

        # Let's compute exact gradient correctly
        # For joint function i = max_k integral h(x)(1-h(x+k)) dx:

        # Better approach:
        # Manually compute influence of h[i] on correlations
        # df/dh[i] = sum_j (1 - h[j]) * (1 if j = i + k_max else 0) - h[i] * (1 - h[i + k_max])
        # i.e., arbitrarily directional:
        
        # Let's use the shift k_max that maximizes overlap
        # We compute derivative via full correlation gradient exploration
        
        # Compute finite difference gradient for h[i] w.r.t. k_max shift
        shift = k_max
        grad_h = np.zeros_like(h)

        # We manually compute ∂/∂h[i] (h(x)(1-h(x+k_max)) dx)

        # We want ∫ h(x) (1 - h(x + k)) dx

        # Let's re-derive

        # This should be = (1 - h[i + k])

            gap = shift

            # we are convolving h with (1-h)

            # Let's implement in a more numerically stable way
            # Our primary purpose is to ensure h is valid, improve overlap
            # Let's iteratively compute and update
            
        # Simpler working approach:

        grad_h = 1 - h
        
        # With h from left to right, shift effect is captured now.

        v_h = beta * v_h + (1 - beta) * grad_h

        # Project h back

        h = h * (n_points / 2) / np.sum(h)
    
    if best_h is not None:
        h = best_h
    
    # Final check
    c5_computed = max(np.correlate(h, 1-h, mode="full")) * dx
    return (h, c5_computed, n_points)
