import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for a given step function h"""
    # Compute correlation of h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode="full")
    # Apply dx scaling
    corr *= dx
    # Return maximum overlap
    return np.max(corr)

def is_valid_h(h, n_points, dx):
    """Check if h satisfies constraints"""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) * dx - 1.0) < 1e-6)

def normalize_h(h, n_points, dx):
    """Ensure h satisfies sum constraint"""
    target_sum = n_points / 2.0
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) > 1e-6:
        h = h * target_sum / current_sum
    return h

def mutate_h(h, mutation_rate=0.05):
    """Apply small random mutations to h"""
    h = h.copy()
    indices = np.random.choice(len(h), size=int(len(h) * mutation_rate), replace=False)
    for i in indices:
        delta = np.random.normal(0, 0.05)
        h[i] += delta
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    return h

def crossover_h(h1, h2):
    """Crossover two step functions"""
    h1 = h1.copy()
    h2 = h2.copy()
    crossover_point = np.random.randint(1, len(h1))
    new_h1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
    new_h2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
    return new_h1, new_h2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select best individual from a tournament"""
    selected_idx = np.random.choice(len(population), tournament_size)
    best_idx = selected_idx[np.argmax([fitnesses[i] for i in selected_idx])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45  # Starting with the provided number of points
    dx = 2.0 / n_points
    population_size = 100
    generations = 0
    max_generations = 600
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = population_size // 5
    
    # Initialize population
    population = []
    fitnesses = []
    
    # Include initial solution if available
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = normalize_h(initial_h, n_points, dx)
        population.append(initial_h)
    else:
        # Initialize randomly
        initial_h = np.random.rand(n_points)
        initial_h = normalize_h(initial_h, n_points, dx)
        population.append(initial_h)
    
    # Fill up the rest of population
    for _ in range(population_size - len(population)):
        h = np.random.rand(n_points)
        h = normalize_h(h, n_points, dx)
        population.append(h)
    
    start_time = time.time()
    
    # Main GA loop
    best_h = None
    best_c5 = float('inf')
    
    while time.time() - start_time < budget_s and generations < max_generations:
        generations += 1
        
        # Evaluate fitness for all in population
        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
        
        # Selection, crossover, mutation
        new_population = []
        
        # Elitism: keep best individuals
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate rest of population
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            if np.random.rand() < crossover_prob:
                child1, child2 = crossover_h(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate_h(child1, mutation_rate)
            child2 = mutate_h(child2, mutation_rate)
            
            # Normalize
            child1 = normalize_h(child1, n_points, dx)
            child2 = normalize_h(child2, n_points, dx)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    if best_h is None:
        # Fallback to the last evaluated if no better found
        current_best_idx = np.argmin(fitnesses)
        best_h = population[current_best_idx].copy()
        best_c5 = fitnesses[current_best_idx]
    
    return (best_h, best_c5, n_points)
