import numpy as np
import time
from scipy import correlate

def evaluate_h(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    corr = correlate(h, 1 - h, mode="full")
    # The full correlation includes lags from -n+1 to n-1
    # We only need the positive lags in [0, n-1] because of symmetry
    # So we extract the relevant part (the middle part of full correlation)
    n = len(h)
    full_corr = corr
    mid_idx = len(full_corr) // 2
    # Extract the correlations for k = 0 to n-1, corresponding to negative shifts to positive shifts
    # Since correlate does k from -n+1 to n-1, and the value at k-th element corresponds to shift k
    # So we want from index (mid_idx) to (mid_idx + n - 1), which gives valid k from 0 to n-1
    # Actually: for corr of len 2n-1, element at index i corresponds to shift i - (n-1)
    # So corr has index 0 to 2n-2 and shift -n+1 to n-1
    # We need k from 0 to n-1 which means indices from (n-1) to (2n-2)
    valid_corr = full_corr[(n-1):]
    return np.max(valid_corr) * dx

def project_h(h, n_points):
    """Perform projection to ensure h is in [0,1] and sum equals n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if np.abs(current_sum - target_sum) > 1e-10:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h from the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5 * dx)
    
    h = project_h(h, n_points)
    
    best_h = h.copy()
    best_C5 = evaluate_h(best_h, dx)
    
    # Perturbation parameters
    initial_sigma = 0.1
    decay_factor = 0.99
    max_iter = 1000000
    patience = 1000
    patience_counter = 0
    current_sigma = initial_sigma
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate perturbation
        perturbation = np.random.normal(0, current_sigma, n_points)
        h_new = h + perturbation
        h_new = project_h(h_new, n_points)
        
        # Evaluate new C5
        C5_new = evaluate_h(h_new, dx)
        
        if C5_new < best_C5:
            best_h = h_new.copy()
            best_C5 = C5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        h = h_new
        current_sigma *= decay_factor
        
        if patience_counter >= patience:
            break
    
    return (best_h.tolist(), best_C5, n_points)
