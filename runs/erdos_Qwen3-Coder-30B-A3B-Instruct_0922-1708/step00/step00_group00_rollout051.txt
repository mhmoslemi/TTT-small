import numpy as np

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    # Compute all overlaps for shifts k in [-n+1, n-1]
    n = len(h)
    overlaps = correlate(h, 1 - h, mode='full')
    # Take only the valid overlaps (when both h[i] and h[i+k] are defined)
    # The full correlation has 2*n - 1 elements, indexed from -(n-1) to n-1
    # The valid ones are k = -(n-1) to n-1 with at least one element
    # For shift k, the overlap is sum(h[i] * (1 - h[i+k])) for valid i
    # The middle index of full correlation is n-1
    valid_overlaps = overlaps[n-1:]  # This gives us k = 0 to n-1
    # We also need negative k values, so let's reverse and append
    # But because of symmetry: overlap of k == overlap of -k (up to boundary)
    # However, the cross-correlation returns full range, we need the actual overlap
    # For k=-n+1 to 0, do this:
    full_overlaps = overlaps.copy()
    # Need overlaps for k = -n+1 to n-1
    # The values are at indices n-1-k to n-1+k, but we must be careful about indexing
    # Actually, the overlap for shift k in correlate(h, 1-h) is at index k + (n-1)
    # When k is negative, it's at k + (n-1) == (n-1) + k
    # So full overlaps[0] = overlap for k = 1-n
    # full overlaps[n-1] = overlap for k = 0
    # full overlaps[2*n-2] = overlap for k = n-1
    # We need all k from -(n-1) to n-1
    # Let's figure out the correct indexing
    max_overlap = 0
    # For k from -n+1 to n-1, the overlap is lowered from the full cross-correlation
    # We know positive shifts are full_overlaps[n-1:] so the ones for negative k are full_overlaps[:n-1]
    # Let O_k be overlap for shift k. For k >= 0: O_k = full_overlaps[n-1+k]
    # For k < 0: O_k = full_overlaps[n-1+k] but we're going backwards with k
    # Actually, full_overlaps[0] = overlap for k = -(n-1)
    # full_overlaps[n-1] = overlap for k = 0
    # full_overlaps[2*n-2] = overlap for k = n-1
    # So for k=0 to n-1 it's full_overlaps[n-1 + k]
    # For k from -(n-1) to 0  
    # So full_overlaps also holds the negative indices but not fully
    # Let's extract the corresponding indices better:
    all_overlaps = np.zeros(2*n - 1) 
    all_overlaps[0] = full_overlaps[n-1]
    for i in range(1, n):
        all_overlaps[i] = full_overlaps[n-1+i]
        all_overlaps[2*n-2-i] = full_overlaps[n-1-i]
    # Now we compute the maximum
    max_overlap = np.max(all_overlaps)
    return max_overlap * dx

def project_h_to_constraints(h, n_points):
    """Project h to [0,1] and normalize to sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum == 0:
        h = 0.5 * np.ones_like(h)
        current_sum = n_points / 2
    else:
        h = h * (n_points / 2) / current_sum
    return h

def smooth_max(x, tau=0.1):
    """Compute smooth approximation to max(x)"""
    return np.log(np.sum(np.exp(tau * x))) / tau

def evaluate_loss_gradient(h, n_points, dx, tau=0.1):
    """Evaluate gradient in a direct way for teleportation"""
    # Recompute overlap for each k gracefully.
    # Assume h is valid in [0,1]. Just compute overlaps.

    # Core idea:
    # We want to generate a set of unsafe (but axis-aligned)
    # gradient components per shift.
    # Correct f(h) = max_k overlap_k
    # Take smooth version: loss = log(sum exp(tau*overlap_k)) / tau
    
    # Derivative: d/dx_i of loss (with step) 
    # Let f_k = sum_i h[i] * (1 - h[i+k]) = sum_h[i] - sum_h[i]*h[i+k]
    # Then d f_k / d h[i] = 1 - h[i+k] if i+k valid
    # Let's loop over all k
    overlaps = []
    d_overlaps = []
    all_k = []
    for k in range(-n_points+1, n_points):
        if k == 0:
            val = np.dot(h, 1-h)  # Manually compute
        else:
            # Shift h by -k and compute dot product with (1-h)
            if k > 0:
                shift_h = np.concatenate([h[k:], np.zeros(k)])
            else:
                shift_h = np.concatenate([np.zeros(-k), h[:k]])
            val = np.dot(h, 1-shift_h)
        overlaps.append(val)
        all_k.append(k)

    overlap_values = np.array(overlaps) * dx  
    # Now get full gradient  
    grad = np.zeros_like(h)
    # Apply chain rule on max smoothing:
    # loss = smooth_max(overlap_values) 
    # let exps = exp(tau * overlap_values) 
    # loss = log(sum(exp(tau * overlap_values)) / tau, so
    # dloss = 1/tau * (1/sum exps) * sum(d/dh_i exp(tau * overlap_values))
    # Let's compute d/dh[i] overlap_k
    for i in range(n_points):
        # Loop over valid k values where (i+k) is valid range [0, n_points)
        for k in range(-n_points+1, n_points):
            if 0 <= (i+k) < n_points:
                # This contributes to the derivative of overlap_k w.r.t h[i]
                # d/ h[i] overlap_k = (1 - h[i+k])
                # So accumulate this: this is a valid contribution
                grad[i] += -exp(tau * overlap_values[k + n_points -1]) * (1 - h[i+k])  # shift to match baseline
    # Now normalize by the denominator:
    sum_exp = np.sum(np.exp(tau * overlap_values))
    grad = grad / (sum_exp * tau)
    return smooth_max(overlap_values, tau), grad

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    # Parameters
    n_points = 45 if 'initial_h_values' not in globals() else len(initial_h_values)
    dx = 2.0 / n_points
    max_iter = 10000
    tau = 0.1
    learning_rate = 0.01
    max_stall = 50
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = project_h_to_constraints(h, n_points)
    else:
        h = np.ones(n_points) * 0.5
        h = project_h_to_constraints(h, n_points)
    
    # Store best
    best_C5 = evaluate_h(h, dx)
    best_h = h.copy()
    
    # Iteration loop
    iter_count = 0
    stall_count = 0
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        iter_count += 1
        current_C5 = evaluate_h(h, dx)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
            stall_count = 0
        else:
            stall_count += 1
        
        # Compute gradient
        loss, grad = evaluate_loss_gradient(h, n_points, dx, tau)
        # Update step
        h = h - learning_rate * grad
        # Project to [0,1] and normalize
        h = project_h_to_constraints(h, n_points)
        if stall_count > max_stall:
            # Restart with random#
            # slightly perturbed:
            h = np.random.rand(n_points) 
            h = project_h_to_constraints(h, n_points)
            stall_count = 0
        if time.time() - start_time > budget_s - 0.5:
            break
        
    # Final evaluation
    final_C5 = evaluate_h(best_h, dx)
    
    return (list(best_h), final_C5, n_points)
