import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2"""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

    swarm_size = 50
    max_iter = 500
    inertia_w = 0.7
    c1, c2 = 2.0, 2.0
    max_stagnation = 50
    diversity_injection = 20  # percentage of particles to reset
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add the initial solution
    h_proj = project_h(initial_h.copy(), n_points)
    particles.append(h_proj)
    velocities.append(np.zeros(n_points))
    
    # Add random particles with same sum constraint
    for _ in range(swarm_size - 1):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Best solutions tracking
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [evaluate_h(p, 2.0 / n_points) for p in particles]
    global_best_idx = np.argmin(personal_best_scores)
    global_best = personal_best[global_best_idx].copy()
    global_best_score = personal_best_scores[global_best_idx]
    
    # Optimization iteration
    iteration = 0
    stagnation_count = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        scores = []
        for i, p in enumerate(particles):
            score = evaluate_h(p, 2.0 / n_points)
            scores.append(score)
            
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = p.copy()
                
                if score < global_best_score:
                    global_best_score = score
                    global_best = p.copy()
                    stagnation_count = 0  # reset stagnation counter
                else:
                    stagnation_count += 1
        
        # Update swarm positions using PSO
        for i in range(swarm_size):
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            # Update velocity
            velocities[i] = (inertia_w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            # Update position
            particles[i] += velocities[i]
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points)

        if stagnation_count >= max_stagnation:
            # Inject diversity by resetting some particles

                h = project_h(h, n_points)

                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0  # reset counter
            
    # Return best solution found
    return (global_best, global_best_score, n_points)
