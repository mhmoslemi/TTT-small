import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate C5 for a given h array."""
    # Compute correlation between h and (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum value of the correlation
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Set seed for reproducibility
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        n_points = len(h_start)
    else:
        # Use default construction with 100 points
        n_points = 100
        h_start = np.ones(n_points) * 0.5  # Uniform distribution as fallback
        
    # Initialize parameters
    max_iter = 500
    swarm_size = 50
    w = 0.7
    c1 = c2 = 2.0
    n_evaluations = 0
    start_time = time.time()
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial solution as first particle
    h_projected = project_h(h_start.copy(), n_points)
    particles.append(h_projected)
    velocities.append(np.zeros_like(h_projected))
    
    # Generate the rest of the swarm
    for i in range(1, swarm_size):
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        particles.append(h)
        velocities.append(np.random.rand(n_points) * 0.1)
    
    # Initialize personal bests
    p_best = [p.copy() for p in particles]
    p_best_vals = [evaluate_h(p, 2.0 / n_points) for p in particles]
    
    # Initialize global best
    best_idx = np.argmin(p_best_vals)
    g_best = p_best[best_idx].copy()
    g_best_val = p_best_vals[best_idx]
    
    # PSO iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        for i in range(swarm_size):
            # Compute C5 for particle i
            current_val = evaluate_h(particles[i], 2.0 / n_points)
            n_evaluations += 1
            
            # Update personal best if better
            if current_val < p_best_vals[i]:
                p_best_vals[i] = current_val
                p_best[i] = particles[i].copy()
                
            # Update global best if better
            if current_val < g_best_val:
                g_best_val = current_val
                g_best = particles[i].copy()
                
        # Update velocities and positions
        for i in range(swarm_size):
            # Random coefficients
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project back to valid region
            particles[i] = project_h(particles[i], n_points)
    
    # Return the best solution found
    return (g_best.tolist(), g_best_val, n_points)
