import numpy as np
import time
from scipy import signal

def evaluate_solution(h, dx):
    """Evaluate the overlap integral for a given step function h."""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Scale by dx to get actual overlap values
    overlaps = corr * dx
    # Return maximum overlap
    return np.max(overlaps)

def validate_h(h, n_points):
    """Validate constraints on h."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2) < 1e-6)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population of step functions."""
    population = []
    
    # Use initial solution if available
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Add random individuals
    while len(population) < pop_size:
        # Start with uniform distribution
        h = np.full(n_points, 0.5)
        # Random perturbations
        for i in range(n_points):
            h[i] += np.random.normal(0, 0.1)
            h[i] = np.clip(h[i], 0, 1)
        
        # Normalize to satisfy sum constraint
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        if validate_h(h, n_points):
            population.append(h.copy())
    
    return population

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a step function."""
    h = h.copy()
    n_points = len(h)
    
    for i in range(n_points):
        if np.random.rand() < mutation_rate:
            h[i] = np.clip(h[i] + np.random.normal(0, 0.1), 0, 1)
    
    return h

def re_normalize(h, n_points):
    """Ensure h sums to n_points/2 while keeping values within [0,1]."""
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) < 1e-6:
        return h
    
    # Scale to meet constraint
    h = h * (n_points / 2) / h_sum
    # Clip values
    h = np.clip(h, 0, 1)
    return h

            c5 = evaluate_solution(h, dx)
            fitness_scores.append(c5)
            
        # Update best solution

        if min_fitness < best_c5:

        new_population = []
