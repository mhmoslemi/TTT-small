import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Load initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a simple initial construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[0] = 0.4
        h[-1] = 0.6
    
    # Normalize to satisfy the constraint sum(h) * dx = 1
    dx = 2.0 / n_points
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    
    # Validate initial h
    assert np.all(h >= 0) and np.all(h <= 1), "Initial h must be in [0,1]"
    assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Integral constraint not satisfied"
    
    # Function to compute C5
    def compute_C5(h):
        corr = correlate(h, 1-h, mode='full')
        # The full convolution length is 2*n_points - 1
        # We want the maximum among the overlapping parts
        # The valid lags range from -(n_points-1) to (n_points-1)
        # The maximum is at indices n_points-1 to 2*n_points-2 (positive lags)
        # The minimum is at indices 0 to n_points-2 (negative lags)
        # Maximum overlapping part corresponds to lags from 0 to n_points-1
        # We take the maximum value in the valid range
        max_val = np.max(corr[n_points-1:])
        return max_val * dx

    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Parameters
    max_trials = 1000
    trials_since_improvement = 0
    max_no_improvement = 100

    start_time = time.time()
    elapsed = 0
    
    for trial in range(max_trials):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        # Perturb h
        new_h = h.copy()
        
        # Select random points to modify (10% of total points)
        num_modify = max(1, n_points // 10)
        indices = np.random.choice(n_points, size=num_modify, replace=False)
        
        # Modify values
        for i in indices:
            delta = np.random.uniform(-0.05, 0.05)
            new_h[i] += delta
            
        # Renormalize to maintain sum(h)*dx=1
        old_sum = np.sum(h)
        new_sum = np.sum(new_h)
        
        # Adjust all values to sum to old_sum (maintains target integral)
        if abs(new_sum) > 1e-10:
            # Scale so that sum matches
            new_h = new_h * (old_sum / new_sum)
            # Ensure all values are in [0,1]
            new_h = np.clip(new_h, 0, 1)
        
        # Apply smoothing
        window_size = min(5, n_points // 3 + 1)
        if window_size > 1:
            # Apply a moving average filter (simple)
            smoothed = np.zeros_like(new_h)
            for i in range(n_points):
                start = max(0, i - window_size//2)
                end = min(n_points, i + window_size//2 + 1)
                smoothed[i] = np.mean(new_h[start:end])
            new_h = smoothed
            
        # Re-normalize again
        new_sum = np.sum(new_h)
        if abs(new_sum) > 1e-10:
            new_h = new_h * (old_sum / new_sum)
        
        # Check constraints
        new_h = np.clip(new_h, 0, 1)
        
        # Compute new C5
        new_C5 = compute_C5(new_h)
        
        # Update if improved
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = new_h.copy()
            trials_since_improvement = 0
        else:
            trials_since_improvement += 1
            
        # Early exit if no improvement
        if trials_since_improvement >= max_no_improvement:
            break

    # Final check and recompute
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
