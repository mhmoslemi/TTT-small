import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values).copy()
        # Ensure it satisfies the constraint
        current_sum = np.sum(h) * dx
        h *= (target_sum / current_sum)
    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05, 0.05, n_points) * dx
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h) * dx
        h *= (target_sum / current_sum)
    
    # Objective function to minimize (smooth approximation of max overlap)
    def objective(h_flattened):
        h = h_flattened.reshape(-1)
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        current_sum = np.sum(h) * dx
        h *= (target_sum / current_sum)
        # Compute overlap
        c5_val = evaluate_c5(h, dx)
        return c5_val
    
    # Constraint for sum(h) * dx = 1
    def constraint_sum(h_flattened):
        h = h_flattened.reshape(-1)
        return np.sum(h) * dx - target_sum
    
    # Bounds for h
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization using L-BFGS-B
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimizer parameters
    max_iter = 2000
    iteration = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Initialize with current best
    current_h = h.copy()
    current_c5 = objective(current_h.flatten())
    best_c5 = min(best_c5, current_c5)
    if current_c5 == best_c5:
        best_h = current_h.copy()
    
    # Timing and early stopping
    while time.time() - start_time < budget_s - 1.0 and iteration < max_iter:
        iteration += 1
        
        # Small random perturbation to avoid local minima
        perturbation = np.random.normal(0, 0.01, n_points) * dx
        proposed_h = current_h + perturbation
        proposed_h = np.clip(proposed_h, 0, 1)
        current_sum = np.sum(proposed_h) * dx
        proposed_h *= (target_sum / current_sum)
        
        # Evaluate
        new_c5 = objective(proposed_h.flatten())
        if new_c5 < current_c5:
            current_h = proposed_h
            current_c5 = new_c5
        
        # Update best
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
            
        # Check for time
        if time.time() - start_time > budget_s - 0.5:
            break
    
    # Final evaluation
    final_h = best_h.copy()
    final_c5 = evaluate_c5(final_h, dx)
    
    return (final_h.tolist(), final_c5, n_points)
