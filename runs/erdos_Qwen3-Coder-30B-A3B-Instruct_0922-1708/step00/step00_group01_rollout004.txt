def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral C_5"""
    # h is a numpy array with values in [0,1]
    # dx = 2.0 / len(h)
    # Cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Each entry corresponds to a lag from -len(h)+1 to len(h)-1
    # We're interested in the maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    """Project h to ensure it's in [0,1] and sum(h) = n_points/2"""
    # Clip values to [0,1]

    # Rescale to enforce the constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
