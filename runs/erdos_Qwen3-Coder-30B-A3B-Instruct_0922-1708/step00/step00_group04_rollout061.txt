import numpy as np
from scipy import signal
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Compute the C5 bound for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    corr = signal.correlate(h, 1 - h, mode="full")
    # Maximize over all lags
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
    else:
        # Create simple initial construction
        n_points = 42
        h_start = np.ones(n_points)
        h_start[:n_points//2] = 0
        h_start = h_start * 2.0 / n_points

    # Discretization
    n_points = len(h_start)
    dx = 2.0 / n_points

    # Preliminary C5 value
    c5_initial = evaluate_erdos_solution(h_start)
    best_h = h_start.copy()
    best_c5 = c5_initial

    # Parameters
    max_iter = 1000
    iter_count = 0
    
    # Simple optimization using gradient-based method
    # We'll modify h by small random perturbations and accept improvements
    
    def perturb_h(h, step_size=0.05):
        # Create a new h by perturbing current one
        h_new = h.copy()
        # Pick random indices to modify
        indices = np.random.choice(n_points, size=max(1, n_points//10), replace=False)
        for i in indices:
            h_new[i] = np.clip(h_new[i] + np.random.normal(0, step_size), 0, 1)
        
        # Renormalize to integral = 1
        h_new = h_new * (1.0 / np.sum(h_new) * dx)
        return h_new

    # Gradient descent-like optimization
    h_current = h_start.copy()
    c5_current = c5_initial

    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        h_candidate = perturb_h(h_current, step_size=0.05)
        
        c5_candidate = evaluate_erdos_solution(h_candidate)
        
        if c5_candidate < c5_current:
            h_current = h_candidate
            c5_current = c5_candidate
            
            if c5_candidate < best_c5:
                best_h = h_candidate.copy()
                best_c5 = c5_candidate
        
        iter_count += 1
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), float(final_c5), n_points)
