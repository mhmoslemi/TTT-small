import numpy as np
import time
from scipy.signal import correlate

def evaluate_C5(h, dx):
    # Compute all overlaps for all lags
    overlap_vals = correlate(h, 1 - h, mode='full') * dx
    # Return the maximum overlap
    return np.max(overlap_vals)

def project_h(h, target_sum):
    """Project h to satisfy sum constraint"""
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * target_sum / current_sum
    return np.clip(h, 0, 1)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a simple initial configuration
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Make it symmetric and taper to 0 at edges
        half = n_points // 2
        for i in range(half):
            factor = (half - i) / half
            h[i] = 0.1 + 0.4 * factor
            h[n_points - 1 - i] = 0.1 + 0.4 * factor
        h = project_h(h, n_points / 2.0)
        
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Smoothed max approximation
    tau = 1.0
    learning_rate = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    tol = 1e-5
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all lags
        h_padded = np.concatenate([h, h])  # Double length for easier circular indexing
        overlap_vals = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # Compute overlap for lag k
            if k < n_points:
                # Forward overlap
                prod = h * (1 - h_padded[k:k+n_points])
            else:
                # Backward overlap
                k_rev = 2 * n_points - 2 - k
                prod = h * (1 - h_padded[k_rev:k_rev+n_points])
            overlap_vals[k] = np.sum(prod) * dx
        
        # Smooth approximation to maximum
        C5_smooth = np.log(np.sum(np.exp(tau * overlap_vals))) / tau
        
        # Compute gradients
        grad = np.zeros(n_points)
        exp_terms = np.exp(tau * overlap_vals)
        exp_sum = np.sum(exp_terms)
        
        # Each element of the gradient
        for i in range(n_points):
            grad_term = 0.0
            for k in range(2 * n_points - 1):
                # The kth component contributes to the gradient of the ith element
                if k < n_points:
                    # Forward
                    j = (i - k) % n_points
                else:
                    # Backward
                    k_rev = 2 * n_points - 2 - k
                    j = (i + k_rev) % n_points
                if j >= 0 and j < n_points:
                    # Gradient of the k-th overlap with respect to h[i]
                    grad_term += -exp_terms[k] * (1 - h[j]) * dx
            grad[i] = grad_term / exp_sum
        
        # Update h using gradient descent
        h = h - learning_rate * grad
        
        # Project to feasible region
        h = project_h(h, target_sum)
        
        # Evaluate actual C5
        actual_C5 = evaluate_C5(h, dx)
        
        if actual_C5 < best_C5:
            best_C5 = actual_C5
            best_h = h.copy()
        
        # Anneal temperature and learning rate
        tau *= anneal_factor
        if iteration > 500:
            learning_rate *= 0.9999
            
        # Convergence check
        if iteration > 10 and abs(C5_smooth - evaluate_C5(h, dx)) < tol:
            learning_rate *= 0.5  # Slow down if stalled
        
        # Occasionally print current status
        if iteration % 500 == 0:
            pass
        
    # Final evaluation
    final_C5 = evaluate_C5(best_h, dx)
    return (best_h.tolist(), final_C5, len(best_h))
