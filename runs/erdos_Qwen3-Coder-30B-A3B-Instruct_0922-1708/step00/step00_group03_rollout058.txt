import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Define variables
    X = cvx.Variable(n_points)
    Z = cvx.Variable((n_points, n_points))
    
    # Constraints
    constraints = [
        X >= 0,
        X <= 1,
        cvx.sum(X) == n_points / 2,
    ]
    
    # Add Z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i,j] <= X[i],
                Z[i,j] <= 1 - X[j],
                Z[i,j] >= X[i] + X[j] - 1
            ]
    
    # Objective
    C = []
    for k in range(-n_points+1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        sum_z = cvx.sum(Z[np.arange(n_points)[valid], idx[valid]])
        C.append(sum_z)
    
    t = cvx.Variable()
    for c in C:
        constraints.append(c <= t)
    
    objective = cvx.Minimize(t)
    prob = cvx.Problem(objective, constraints)
    
    # Optimization loop
    max_iter = 100
    for i in range(max_iter):
        if time() - start_time > budget_s - 1:
            break
            
        # Initialize with current best
        X.value = best_h.copy()
        
        try:
            prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s - (time() - start_time) - 1)
            
            if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
                new_h = np.clip(X.value, 0, 1)
                new_h = new_h / np.sum(new_h) * (n_points / 2)
                new_c5 = evaluate_erdos_solution(new_h, dx)
                
                if new_c5 < best_c5:
                    best_h = new_h.copy()
                    best_c5 = new_c5
                    
        except Exception:
            pass
            
        # Random restart
        h = np.random.rand(n_points) * 0.2 + 0.4
        h = h / np.sum(h) * (n_points / 2)
        c5 = evaluate_erdos_solution(h, dx)
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
    
    return (best_h, best_c5, n_points)
