import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Create a reasonable starting point
        h = np.ones(n_points) * 0.5
        h[0] = 0.1
        h[-1] = 0.1
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient descent with adaptive learning rate
    eta = 0.01
    last_improvement = 0
    iteration = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Compute overlaps for all shifts
        corr = signal.correlate(h, 1 - h, mode='full')
        lags = np.arange(-n_points + 1, n_points)
        overlaps = corr * dx
        
        # Find the maximum overlap (worst case shift)
        max_idx = np.argmax(overlaps)
        k = lags[max_idx]
        
        # Compute gradient for this shift
        grad = np.zeros(n_points)
        if k >= 0:
            # h[i] contributes to overlap with h[i+k] 
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k]) * dx
            # For i+k >= n_points, contribution is from h[i] to 0 (no more)
        else:
            # h[i] contributes to overlap with h[i+k]
            for i in range(-k, n_points):
                grad[i] += -h[i + k] * dx
        
        # Update h
        h = h - eta * grad
        
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        
        # Ensure sum = n_points / 2
        h_sum = np.sum(h)
        if h_sum > 0:
            # Rescale to fix sum and preserve shape
            h = h * (n_points / 2) / h_sum
        
        # Evaluate new C5
        current_c5 = compute_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
        
        iteration += 1
        
        # Adaptive learning rate
        if iteration % 100 == 0 or iteration - last_improvement > 50:
            eta *= 0.95
            if eta < 1e-6:
                eta = 1e-6
        
        # Early termination check
        if iteration - last_improvement > 100:
            break
            
    return (best_h.tolist(), float(best_c5), int(n_points))

def compute_c5(h, dx):
    # Use the same approach as evaluate_erdos_solution
    corr = np.correlate(h, 1 - h, mode='full')
    overlaps = corr * dx
    return np.max(overlaps)
