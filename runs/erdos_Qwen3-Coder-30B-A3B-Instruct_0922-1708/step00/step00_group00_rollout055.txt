def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part
    n = len(h)
    # The valid range is from -n+1 to n-1
    # We want the cases where k = 1 to n-1 (shifts)
    # But the correlate returns full convolution so we extract the right part
    # The overlap integral for shift k is corr[n-1+k] * dx
    max_val = 0
    for k in range(1, n):
        val = corr[n-1+k] * dx
        max_val = max(max_val, val)
    return max_val
