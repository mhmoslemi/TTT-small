import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize from provided construction or create symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # create symmetric construction with plateau
        mid = n_points // 2
        h[mid-5:mid+5] = 1.0
    
    # Validate initial construction
    h = np.clip(h, 0, 1)
    dx = 2.0 / n_points
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor

    # Optimization parameters
    max_iter = 5000
    learning_rate = 0.1
    decay_every = 1000
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Timing info
    start_time = time.time()
    
    def compute_c5(h_vec, dx):
        # Compute cross-correlation of h with (1 - h)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # Take the maximum value (excluding trivial zero-lag)
        max_val = np.max(corr)
        return max_val * dx

    def compute_gradient(h_vec, dx, dominant_lag):
        n = len(h_vec)
        grad = np.zeros(n)
        # Gradient computation for specific lag dominates our C5
        # We only compute the gradient for the dominant overlap lag
        # Analogous to d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} 
        # with focus on h_i term
        for i in range(n):
            j = i + dominant_lag
            if 0 <= j < n:
                grad[i] = -h_vec[j]  
        return grad * dx

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        corr = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(corr)
        # The actual max lag from cross-correlation
        lag = max_lag_idx - (len(h) - 1)
        
        # Compute gradient w.r.t. this dominant lag
        grad = compute_gradient(h, dx, lag)
        
        # Update with gradient ascent
        h = h + learning_rate * grad
        
        # Clip to feasible domain [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to meet constraint sum(h) * dx = 1
        sum_h = np.sum(h)
        if sum_h > 0:
            scale = (n_points / 2.0) / sum_h
            h = h * scale
            
        # Compute new C5
        new_c5 = compute_c5(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Anneal learning rate
        if iteration % decay_every == 0 and iteration > 0:
            learning_rate *= 0.5

        if iteration % 50 == 0:
            elapsed = time.time() - start_time
            if elapsed > budget_s - 0.5:
                break
    
    return (best_h.tolist(), best_c5, n_points)
