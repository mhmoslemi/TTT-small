import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h
    h = h * (target_sum / np.sum(h))
    
    # Optimization parameters
    alpha = 0.01
    beta = 0.9
    min_alpha = 1e-6
    patience = 500
    patience_counter = 0
    
    # Previous best
    best_h = h.copy()
    best_C5 = float('inf')
    
    # Momentum
    v_h = np.zeros(n_points)
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        if iter_count % 1000 == 0:
            # Recheck best solution
            current_C5 = evaluate_erdos_solution(h)
            if current_C5 < best_C5:
                best_C5 = current_C5
                best_h = h.copy()
        
        # Compute correlation for all shifts
        h_padded = np.pad(h, (0, n_points), mode='constant')
        one_minus_h = 1.0 - h
        one_minus_h_padded = np.pad(one_minus_h, (0, n_points), mode='constant')
        
        # Full cross-correlation
        corr = correlate(h_padded, one_minus_h_padded, mode='full')
        # We actually want shift from 0 to 2, which is equivalent to indices from n_points to 2*n_points-1
        # but only 0 to 2 means indices 0 to n_points
        corr = corr[n_points:]  # Only valid shifts [0,2]
        k_max_idx = np.argmax(corr)
        C5_current = corr[k_max_idx]
        
        if C5_current < best_C5:
            best_C5 = C5_current
            best_h = h.copy()
        
        # Compute gradient with respect to h
        # Gradient of integral h(x)(1-h(x+k)) dx
        # ∂/∂h_i = (1 - h_{i+k}) - h_{i+k} * ∂h_{i+k}/∂h_i
        # We'll approximate this by finite differences or compute it directly using convolution
        
        # Use the fact that d/dh_i [h_i * (1 - h_{i+k})] = 1 - h_{i+k} 
        # And we are effectively doing some gradient descent here with: 
        # update_h = h + alpha * gradient
        
        # Gradient estimate of the correlation operation
        grad_h = np.zeros(n_points)
        
        # Manually compute finite difference gradient
        # d/dh_i [integral h(x)(1-h(x+k_max))] = 1 - h_{i+k_max} 
        # but we need to properly index this when h is circularly shifted
        
        # Compute actual gradient
        # essentially for every i, gradient = sum over all k of:
        # d/dh_i [h(x)(1-h(x+k))] = (1 - h(x+k)) * delta_{x, i}  
        # This is equivalent to: 1 - h[i + k_max] for (i, k_max)
        # This is a sliding sum.
        # Estimate this using linearization:
        h_shifted = np.zeros(2 * n_points)
        h_shifted[:n_points] = h
        h_shifted[n_points:] = h  # This is just making it periodic for indexing
        
        # Compute gradient for each i
        for i in range(n_points):
            grad_h[i] = 1.0 - h_shifted[i + k_max_idx]
        
        # Apply momentum
        v_h = beta * v_h + (1 - beta) * grad_h
        h = h - alpha * v_h
        
        # Project to feasible region
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if abs(h_sum) > 1e-10:
            h = h * (target_sum / h_sum)
        
        # Adaptive learning rate
        patience_counter += 1
        if patience_counter > patience:
            alpha *= 0.5
            if alpha < min_alpha:
                alpha = min_alpha
            patience_counter = 0
    
    # Final evaluation
    final_C5 = evaluate_erdos_solution(best_h)
    if final_C5 < best_C5:
        best_C5 = final_C5
    
    return (best_h, best_C5, n_points)

def evaluate_erdos_solution(h):
    # Pad h to make it periodic
    h_extend = np.pad(h, (0, len(h)), mode='constant')
    one_minus_h = 1 - h
    one_minus_h_extend = np.pad(one_minus_h, (0, len(h)), mode='constant')
    
    # correlation_full = np.correlate(h_extend, one_minus_h_extend, mode='full')
    # Extract valid shifts (expected in [0,2])
    # But they are offset based on size
    
    correlation_full = correlate(h_extend, one_minus_h_extend, mode='full')
    # Valid shift indices correspond to [0, n_points)
    correlation_valid = correlation_full[len(h):len(h)*2]
    
    dx = 2.0 / len(h)
    max_overlap = np.max(correlation_valid) * dx
    return max_overlap
