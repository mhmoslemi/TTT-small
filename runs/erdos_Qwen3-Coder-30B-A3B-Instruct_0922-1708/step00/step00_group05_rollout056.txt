def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    max_lag = n_points - 1
    
    # Initialize h
    if 'initial_h_values' in globals():

    else:

        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # Alternative optimization approach: gradient-based with constraints
    def objective(h_flat):
        h = h_flat.reshape(n_points)
        # Reshape for correlation calculation
        overlap = np.correlate(h, 1 - h, mode="full")
        # Extract valid overlaps (excluding padding)
        valid_overlap = overlap[n_points-1:2*n_points-1]
        return np.max(valid_overlap) * dx
    
    def constraint_eq(h_flat):
        h = h_flat.reshape(n_points)
        return np.sum(h) * dx - 1.0
    
    def constraint_bounds(h_flat):
        h = h_flat.reshape(n_points)
        return np.concatenate([h, 1 - h])
    
    # Define constraints and bounds
    cons = [{'type': 'eq', 'fun': constraint_eq}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Random perturbation
        delta = np.random.normal(0, 0.01, n_points)
        h_new = h + delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate
        c5_new = evaluate_erdos_solution(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        if iter_count % 20 == 0:
            # Reoptimize with scipy if we're making progress
            try:
                res = minimize(objective, h_new, method='SLSQP', 
                               bounds=bounds, constraints=cons,
                               options={'maxiter': 10})
                if res.success:
                    h_test = res.x.reshape(n_points)
                    c5_test = evaluate_erdos_solution(h_test)
                    if c5_test < best_c5:
                        best_c5 = c5_test
                        best_h = h_test.copy()
            except:
                pass
                
        h = h_new.copy()
        
    return (best_h.tolist(), best_c5, n_points)
