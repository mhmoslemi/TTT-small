import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    """Evaluate the C5 constant for a given step function h."""
    # Normalize h so that sum(h) * dx = 1
    dx = 2.0 / len(h)
    h = np.array(h)
    h = h * (len(h) / 2.0) / np.sum(h)
    
    # Compute overlap integral for all k values
    overlaps = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(overlaps) * dx
    return max_overlap

def initialize_population(n_points, initial_h, pop_size=100):
    """Initialize a population of step functions."""
    population = []
    
    # Add the initial solution
    population.append(initial_h.copy())
    
    # Fill up rest of population with random valid solutions
    for i in range(pop_size - 1):
        h = np.random.rand(n_points)
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2.0) / np.sum(h)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select parents using tournament selection."""
    selected = []
    for _ in range(len(population)):
        # Select tournament_size individuals randomly
        indices = np.random.choice(len(population), tournament_size, replace=False)
        # Select the best individual from the tournament
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    # Randomly select a crossover point
    cx_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, perturb_scale=0.1):
    """Mutate a step function."""
    h = h.copy()
    n = len(h)
    # Determine which elements to mutate
    mutate_idx = np.random.choice(n, size=int(n * 0.1), replace=False)
    
    # Apply mutation
    for i in mutate_idx:
        if np.random.rand() < mutation_rate:
            h[i] += np.random.uniform(-perturb_scale, perturb_scale) * (2.0 / n)
    
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    
    # Rescale to satisfy integral constraint
    h = h * (n / 2.0) / np.sum(h)
    return h

def genetic_algorithm(initial_h_values, budget_s, pop_size=100, max_gen=1000):
    """Run genetic algorithm to find improved C5 constant."""
    n_points = len(initial_h_values)
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, initial_h_values, pop_size)
    
    # Compute fitness for initial population
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    
    # Track the best solution found
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    generations_no_improvement = 0
    
    for generation in range(max_gen):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Selection
        selected_parents = tournament_selection(population, fitnesses)
        
        # Crossover and Mutation to generate children
        new_population = [best_h.copy()]  # Elitism: carry forward best solution
        for i in range(0, len(selected_parents), 2):
            parent1 = selected_parents[i]
            parent2 = selected_parents[i+1] if i+1 < len(selected_parents) else selected_parents[0]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        # Update population
        population = new_population[:pop_size]
        
        # Evaluate fitness
        fitnesses = [evaluate_erdos_solution(h) for h in population]
        
        # Update best solution
        current_best_idx = np.argmin(fitnesses)
        current_best_fitness = fitnesses[current_best_idx]
        
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = population[current_best_idx].copy()
            generations_no_improvement = 0
        else:
            generations_no_improvement += 1
            
        # Early stopping if no improvement for a while
        if generations_no_improvement > 50:
            break
            
    return best_h, best_fitness, n_points

def run(seed=42, budget_s=600, **kwargs):
    """Main function to run the optimization."""
    np.random.seed(seed)
    
    # Use the initial h values if available
    if 'initial_h_values' in globals():
        h_values = initial_h_values
    else:
        # Start with a basic construction
        n_points = 70
        h_values = np.ones(n_points) * 0.5
    
    # Run the genetic algorithm
    h_opt, c5_bound, n_points = genetic_algorithm(h_values, budget_s)
    
    return (list(h_opt), c5_bound, n_points)
