```python
import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_objective(h, dx):
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points, dx):
    # Ensure h is in [0, 1]
    h = np.clip(h, 0, 1)
    # Scale to meet constraint sum(h) * dx = 1
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def initial_h():
    # If initial_h_values exists, use it; otherwise create one
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create starting solution: symmetric plateau
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Smooth initial shape
        h[:n_points//4] = np.linspace(0.1, 0.5, n_points//4)
        h[3*n_points//4:] = np.linspace(0.5, 0.1, n_points//4)
    return h, n_points

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    best_h, n_points = initial_h()
    dx = 2.0 / n_points
    best_c5 = evaluate_objective(best_h, dx)
    current_time = time.time()

    # GA parameters
    pop_size = 100
    max_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 20
    simplex_size = 10
    noise_scale = 0.05
    
    # Population initialization
    population = [project_to_feasible(best_h + np.random.normal(0, noise_scale, n_points), n_points, dx)
                  for _ in range(pop_size)]

    for generation in range(max_generations):
        if current_time - start_time > budget_s * 0.95:
            break

        # Evaluate fitness
        fitnesses = []
        for ind in population:
            c5 = evaluate_objective(ind, dx)
            fitnesses.append(c5)

        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]

        # Update best solution
        if fitnesses[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitnesses[0]

        # Selection using tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size, replace=False)
            winner = min(tournament, key=lambda x: fitnesses[x])
            selected.append(population[winner])

        # Reproduction
        new_population = []
        for i in range(0, pop_size, 2):
            if np.random.rand() < crossover_rate and i + 1 < pop_size:
                parent1, parent2 = selected[i], selected[i+1]
                # Uniform crossover of segments
                child1 = np.copy(parent1)
                child2 = np.copy(parent2)
                crossover_point = np.random.randint(1, n_points)
                child1[crossover_point:] = parent2[crossover_point:]
                child2[crossover_point:] = parent1[crossover_point:]
                child1 = project_to_feasible(child1, n_points, dx)
                child2 = project_to_feasible(child2, n_points, dx)
                new_population.extend([child1, child2])
            else:
                new_population.append(selected[i])
                if i + 1 < pop_size:
                    new_population.append(selected[i+1])

        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                noise = np.random.normal(0, noise_scale, n_points)
                new_population[i] = project_to_feasible(new_population[i] + noise, n_points, dx)

        # Local refinement with Nelder-Mead
        if generation % 20 == 0 and len(new_population) >= simplex_size:
            simplex_candidates = new_population[:simplex_size]
            for i in range(simplex_size):
                h_refine = simplex_candidates[i]
                # Build a simple local optimization
                def obj_func(x):
                    # Reshape x back to h
                    h_trial = project_to_feasible(x, n_points, dx)
                    return evaluate_objective(h_trial, dx)
                # Nelder-Mead refine
                try:
                    result = minimize(obj_func, h_refine, method='Nelder-Mead', options={'maxiter': 50})
                    if result.success:
                        new_h = project_to_feasible(result.x, n_points, dx)
                        new_c5 = evaluate_objective(new_h, dx)
                        if new_c5 < best_c5:
                            best_h = new_h
                            best_c5 = new_c5
                except:
                    pass
                # Update population
                new_population[i] = new_h if 'new_h' in locals() else simplex_candidates[i]

        population = new_population
        current_time = time.time()

    # Final evaluation
    final_c5 = evaluate_objective(best_h, dx)
    return (list(best_h), final_c5, n_points)
```
