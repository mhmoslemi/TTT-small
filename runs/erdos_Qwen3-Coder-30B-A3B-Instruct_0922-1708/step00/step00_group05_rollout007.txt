import numpy as np
from scipy.optimize import minimize
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.random.rand(n_points)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    
    start_time = time.time()
    best_h = h_init.copy()
    best_C = float('inf')
    
    # Define solver function
    def solve_qp(h0, n_points):
        dx = 2.0 / n_points
        h = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints
        constraints = [h >= 0, h <= 1]
        constraints.append(cp.sum(h) == n_points / 2)
        
        # Max overlap constraints for all lags
        for k in range(-n_points+1, n_points):
            # Compute indices for convolution
            if k >= 0:
                h_k = h[k:]
                h_shifted = h0[:n_points-k]
            else:
                h_k = h0[-k:]
                h_shifted = h[:n_points+k]
                
            # Compute overlap integral (just symbolic representation)
            # We'll evaluate it in the objective function in the actual solve
            # Placeholder - will be computed below in separate checks
            temp_h = h_k if k >= 0 else h_shifted
            temp_h0 = h_shifted if k >= 0 else h_k
            
            expr = cp.sum(cp.multiply(h[:len(temp_h)], 1 - temp_h0)) * dx
            constraints.append(expr <= C)

        # Objective 
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)

        try:
            prob.solve(solver=cp.CVXOPT, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                return h.value, C.value
        except:
            pass
        return None, None
    
    # Grid search fallback that does not use CVXPY
    def grid_search_fallback(start_h, max_iter=1000):
        dx = 2.0 / n_points
        h = start_h.copy()
        best_h = h.copy()
        best_C = float('inf')
        
        # Gradient descent roughly
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C
            C = 0.0
            # Use direct evaluation per overlap lag
            # Overlap for all lags k
            for k in range(-n_points+1, n_points):
                if k >= 0:
                    h_k = h[k:]
                    h_shifted = h[:-k] if k < n_points else np.array([])
                else:
                    h_k = h[:k] if k != 0 else np.array([])
                    h_shifted = h[-k:] if k != 0 else h
                if len(h_k) > 0 and len(h_shifted) > 0:
                    overlap = np.sum(h_k * (1.0 - h_shifted)) * dx
                    C = max(C, overlap)
            
            if C < best_C:
                best_C = C
                best_h = h.copy()
            
            # Simple gradient estimation
            eps = 0.001
            grads = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                if h_plus[i] > 1:
                    h_plus[i] = 1
                elif h_plus[i] < 0:
                    h_plus[i] = 0
                C_plus = 0.0
                for k in range(-n_points+1, n_points):
                    if k >= 0:
                        h_k = h_plus[k:]
                        h_shifted = h_plus[:-k] if k < n_points else np.array([])
                    else:
                        h_k = h_plus[:k] if k != 0 else np.array([])
                        h_shifted = h_plus[-k:] if k != 0 else h_plus
                    if len(h_k) > 0 and len(h_shifted) > 0:
                        overlap = np.sum(h_k * (1.0 - h_shifted)) * dx
                        C_plus = max(C_plus, overlap)
                grads[i] = (C_plus - C) / eps
            
            # Update step
            step_size = 0.01
            h = h - step_size * grads
            h = np.clip(h, 0, 1)
            # Maintain integral constraint
            h = h * (n_points / 2) / np.sum(h)
            
        return best_h, best_C
    
    # Try several local optimizations from different restarts
    num_restarts = 5
    for trial in range(num_restarts):
        if time.time() - start_time > budget_s - 1:
            break
            
        if trial == 0:
            h_current = h_init.copy()
        else:
            # Perturb slightly from current best
            h_current = best_h + np.random.normal(0, 0.01, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            
        # Run grid search fallback or QCQP solver
        h_sol, C_sol = grid_search_fallback(h_current, max_iter=200)
        
        if C_sol is not None and C_sol < best_C:
            best_h = h_sol
            best_C = C_sol
            
    # Final evaluation with actual test
    dx = 2.0 / n_points
    C_final = 0.0
    for k in range(-n_points+1, n_points):
        if k >= 0:
            h_k = best_h[k:]
            h_shifted = best_h[:n_points-k]
        else:
            h_k = best_h[:k]
            h_shifted = best_h[-k:]
            
        if len(h_k) > 0 and len(h_shifted) > 0:
            overlap = np.sum(h_k * (1.0 - h_shifted)) * dx
            C_final = max(C_final, overlap)
            
    return (list(best_h), C_final, n_points)
