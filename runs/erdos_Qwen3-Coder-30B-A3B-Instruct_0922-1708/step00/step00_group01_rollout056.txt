import numpy as np
from math import exp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction if it exists
    try:
        h0 = initial_h_values.copy()
        n_points = len(h0)
    except NameError:
        # Fallback: create a simple symmetric construction
        n_points = 47
        h0 = np.ones(n_points) * 0.5
        # Make it symmetric and taper to edges
        half = n_points // 2
        taper = np.linspace(0.1, 0.5, half)
        h0[:half] = taper
        h0[-half:] = taper[::-1]
        # Normalize to sum to n_points/2
        h0 *= (n_points / 2) / np.sum(h0)
    
    # Ensure h0 is in [0, 1] and sum constraint
    h0 = np.clip(h0, 0, 1)
    h0 *= (n_points / 2) / np.sum(h0)

    # Define helper functions
    def evaluate_C5(h):
        # Compute cross-correlation for all lags using FFT
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_conj = np.concatenate([h[::-1], np.zeros(n_points)])
        # After convolution, we want indices 0 through n_points-1 (non-negative lags)
        correlation = np.convolve(h_padded, h_conj, mode='valid')
        # Now compute overlap for all k = 0 to 2*n_points - 2
        overlaps = np.array([np.sum(h * np.roll(1 - h, k)) for k in range(n_points)])
        return np.max(overlaps) * (2.0 / n_points)  # dx = 2/n_points
    
    def log_sum_exp(x, tau):
        # Sum of exp(tau * x_i) divided by tau
        return np.log(np.sum(np.exp(tau * x))) / tau
    
    def smooth_C5(h, tau):
        # Compute all overlaps systematically
        overlaps = np.zeros(n_points)
        dx = 2.0 / n_points
        # Direct computation
        for k in range(n_points):
            # Slide h against (1 - h) with k shift
            shifted = np.roll(1 - h, k)
            overlaps[k] = np.sum(h * shifted) * dx
        return log_sum_exp(overlaps, tau)
    
    def project_h(h, n_points):
        h = np.clip(h, 0, 1)
        h *= (n_points / 2) / np.sum(h)
        return h
    
    def compute_grad(h, tau):
        dx = 2.0 / n_points
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            overlaps[k] = np.sum(h * shifted) * dx
        exp_overlaps = np.exp(tau * overlaps)
        total_exp = np.sum(exp_overlaps)
        # Gradient w.r.t. h
        grad = np.zeros(n_points)
        # Derivative is sum over k of exp(tau*overlap_k) * d/df_k
        # where f(k) = sum h[i] * (1 - h[(i+k)%n_points])
        # So df_k/dh[i] = (1 - h[(i-k)%n_points])
        for i in range(n_points):
            d_term = 0.0
            for k in range(n_points):
                d_term += exp_overlaps[k] * (1 - h[(i - k) % n_points]) * dx
            grad[i] = d_term / total_exp
        return grad
    
    best_h = h0.copy()
    best_C5 = evaluate_C5(best_h)
    
    h = h0.copy()
    tau = 1.0
    lr = 0.05
    t0 = time.time()
    
    max_iter = 3000
    conv_tol = 1e-5
    
    for iteration in range(max_iter):
        if time.time() - t0 > budget_s - 1:
            break
            
        current_C5 = smooth_C5(h, tau)
        grad = compute_grad(h, tau)
        h -= lr * grad
        h = project_h(h, n_points)
        
        new_C5 = evaluate_C5(h)
        if new_C5 < best_C5:
            best_h = h.copy()
            best_C5 = new_C5
            
        # Annealing
        tau *= 0.999
        if tau < 1e-5:
            tau = 1e-5
            
        if iteration > 100 and abs(current_C5 - new_C5) < conv_tol:
            lr *= 0.5
            if lr < 1e-6:
                break
    
    # Final validation
    final_C5 = evaluate_C5(best_h)
    return (best_h, final_C5, n_points)
