import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlations for all lags
    corr = signal.correlate(h, 1 - h, mode='full')
    # The overlap integral is the negative of the cross-correlation
    # but we're looking for maximum of h(x)(1-h(x+k))
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100  # Default
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Ensure constraints are met
    dx = 2.0 / n_points
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    current_time = time.time()
    
    # Parameters for optimization
    learning_rate = 0.1
    max_iter = 5000
    tolerance = 1e-6
    iteration = 0
    
    while current_time - start_time < budget_s - 1 and iteration < max_iter:
        # Compute all cross-correlations
        corr = signal.correlate(h, 1 - h, mode='full')
        # Find the maximum lag
        max_idx = np.argmax(corr)
        k_max = max_idx - (n_points - 1)
        
        # Compute the gradient for this k
        grad = np.zeros_like(h)
        
        # This could be made more efficient but for now let's replicate mechanism
        # Gradient at i is approximately 1 - h[(i + k_max) % n]
        for i in range(n_points):
            j = (i + k_max) % n_points
            if k_max >= 0:
                grad[i] = 1 - h[j]
            else:
                # If k_max < 0, (i + k_max) wraps, we shift to positive
                j = (i - (-k_max)) % n_points
                grad[i] = 1 - h[j]
        
        # Linear minimization step - find d that minimizes grad^T * d 
        # subject to constraints
        # This is something like :
        # minimze sum(grad[i] * d[i])
        # s.t. 0 <= d[i] <= 1 
        #      sum(d) = n_points / 2
        # We can solve this with coordinate descent or project gradient descent:
        # Try to assign the smallest grad elements to 1 and largest to 0 
        # while respecting the sum constraint.
        
        d = np.zeros_like(h)
        sorted_indices = np.argsort(grad)
        remaining_weight = n_points / 2
        i = 0
        
        while remaining_weight > 0 and i < n_points:
            idx = sorted_indices[i]
            if remaining_weight >= 1:
                d[idx] = 1
                remaining_weight -= 1
            else:
                d[idx] = remaining_weight
                remaining_weight = 0
            i += 1
            
        # Now we have a feasible solution, update:
        h_new = (1 - learning_rate) * h + learning_rate * d
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Quick convergence check using small step change
        if np.linalg.norm(h_new - h) < 1e-6:
            break
            
        h = h_new
        c5 = evaluate_erdos_solution(h)
        
        if c5 < best_c5:
            best_h = h.copy()
            best_c5 = c5
            
        iteration += 1
        current_time = time.time()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    return (best_h.tolist(), final_c5, len(best_h))
