import numpy as np
import time
from scipy.optimize import minimize
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize using initial_h_values if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except NameError:
        n_points = 70
        h0 = np.ones(n_points) * 0.5  # Simple initial guess
    
    # Ensure integral constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Fourier basis representation
    m = min(10, n_points // 2)
    
    # Use cosine basis for even symmetry
    def h_from_coeffs(a):
        x = np.linspace(0, 2, n_points, endpoint=False)
        h = a[0] * np.ones_like(x)
        for k in range(1, len(a)):
            h += a[k] * np.cos(k * np.pi * x / 2)
        return np.clip(h, 0, 1)
    
    def eval_C(a):
        h = h_from_coeffs(a)
        # Adjust for shift in correlation calculation
        corr = signal.correlate(h, 1 - h, mode='full')
        # Normalize by dx = 2/n_points
        dx = 2.0 / n_points
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    def project_coeffs(a):
        # Clip to [0,1] and preserve sum
        h = h_from_coeffs(a)
        h = np.clip(h, 0, 1)
        # Rescale to desired integral
        h = h * (n_points / 2) / np.sum(h)
        # Project back to coefficients
        x = np.linspace(0, 2, n_points, endpoint=False)
        a_new = np.zeros_like(a)
        a_new[0] = np.sum(h) * 0.5 / n_points
        for k in range(1, len(a)):
            a_new[k] = 2.0 / n_points * np.sum(h * np.cos(k * np.pi * x / 2))
        return a_new
    
    a = np.array([0.5] + [np.random.rand() for _ in range(m)])
    
    best_C = float('inf')
    best_h = h0.copy()
    
    gradient_step = 0.01
    patience = 0
    max_patience = 100
    
    while time.time() - start_time < budget_s - 0.1:
        h = h_from_coeffs(a)
        C = eval_C(a)
        
        if C < best_C:
            best_C = C
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience >= max_patience:
            break

        # Gradient approximation
        eps = 1e-4
        grad = np.zeros_like(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            a_minus = a.copy()
            a_minus[i] -= eps
            grad[i] = (eval_C(a_plus) - eval_C(a_minus)) / (2 * eps)
        
        # Gradient descent step
        a -= gradient_step * grad
        # Clip coefficients to valid range [0,1]
        a = np.clip(a, 0, 1)
        a = project_coeffs(a)
        
        # Decay learning rate
        if patience % 10 == 0:
            gradient_step *= 0.95

    # Final evaluation and validation
    h = h_from_coeffs(a)
    final_C = eval_C(a)
    
    if final_C < best_C:
        best_C = final_C
        best_h = h.copy()
    
    return (best_h.tolist(), best_C, n_points)
