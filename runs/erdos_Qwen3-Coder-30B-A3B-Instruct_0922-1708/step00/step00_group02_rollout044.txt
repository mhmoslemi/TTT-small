import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    time_limit = budget_s * 0.95
    
    # Use initial construction if available
    h0 = initial_h_values if 'initial_h_values' in globals() else None
    n_points = 45 if h0 is not None else 90
    dx = 2.0 / n_points
    
    # Set up basis functions
    m = int(2 * np.sqrt(n_points))  # number of basis functions
    centers = np.linspace(0, 2, m, endpoint=False)
    scales = 2.0 / m
    
    def basis(x, center, scale):
        return expit((x - center) / scale)
    
    # Generate basis matrix
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    Phi = np.array([[basis(x, c, scales) for c in centers] for x in x_vals])
    
    # Initial coefficients
    if h0 is not None:
        # Fit initial h0 to basis functions
        a = np.linalg.lstsq(Phi, h0, rcond=None)[0]
    else:
        a = np.random.uniform(-0.1, 0.1, m)
        # Scale to match integral constraint
        h_init = Phi @ a
        integral = np.sum(h_init) * dx
        a = a * (1.0 / integral)
    
    # Optimization parameters
    gamma = 0.8  # L1 penalty weight
    alpha = 0.01
    max_iter = 500
    patience = 20
    best_c5 = float('inf')
    best_a = a.copy()
    
    def compute_overlap(a_vec):
        h = Phi @ a_vec
        h = np.clip(h, 0, 1)
        # Compute all overlaps for different lags
        h_extended = np.concatenate([h, np.zeros(n_points)])
        overlaps = []
        for k in range(n_points):
            overlap = np.sum(h * h_extended[k:k+n_points]) * dx
            overlaps.append(overlap)
        return max(overlaps)
    
    def compute_gradient(a_vec):
        h = Phi @ a_vec
        h = np.clip(h, 1e-6, 1 - 1e-6)
        # For each k, we compute d/dx of h(x)(1-h(x+k))
        # This is: (1-2*h(x))*dh(x)/dx - (1-h(x+k))*dh(x+k)/dx
        # This needs a complex interaction.
        # Let's compute overlap for each k and take derivative of max k:
        # Worst case lag
        h_ext = np.concatenate([h, np.zeros(n_points)])
        max_overlap = 0
        max_k = 0
        for k in range(n_points):
            overlap = np.sum(h * h_ext[k:k+n_points]) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                max_k = k
        # Gradient of this overlap w.r.t. a
        grad = np.zeros_like(a_vec)
        for i in range(n_points):
            grad += Phi[i, :] * (h[i] - h_ext[max_k + i]) * dx
        return grad
    
    for it in range(max_iter):
        if time.time() - start_time > time_limit:
            break
        c5_val = compute_overlap(a)
        if c5_val < best_c5:
            best_c5 = c5_val
            best_a = a.copy()
        
        gradient = compute_gradient(a)
        # Elastic net regularization
        reg_grad = gamma * np.sign(a) + (1 - gamma) * a
        a = a - alpha * gradient - alpha * reg_grad
        
        # Project to feasible region
        h = Phi @ a
        h = np.clip(h, 0, 1)
        integral = np.sum(h) * dx
        a = a * (1.0 / integral)
        
        # Ensure bounds
        h = Phi @ a
        h = np.clip(h, 0, 1)
        a = np.linalg.lstsq(Phi, h, rcond=None)[0]
        
        # Adjust learning rate
        if it % 100 == 0 and it > 0:
            alpha *= 0.5
            if alpha < 1e-6:
                break
    
    # Final adjustment to ensure constraints
    final_h = np.clip(Phi @ best_a, 0, 1)
    final_h *= (n_points / 2) / np.sum(final_h)
    final_h = np.clip(final_h, 0, 1)
    
    # Final evaluation
    final_overlap = compute_overlap(best_a)
    
    h_result = final_h
    
    return (h_result, final_overlap, n_points)
