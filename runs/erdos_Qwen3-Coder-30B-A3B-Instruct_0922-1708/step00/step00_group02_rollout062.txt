import numpy as np
import time
from scipy import optimize
from scipy.signal import correlate

def evaluate_C5(h):
    """Evaluate C5 for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    corr = correlate(h, 1 - h, mode="full")
    # Extract the overlapping parts (for shifts from -n+1 to n-1)
    # We only want the overlap for shifts -n+1 to n-1, i.e., middle n elements
    middle_start = n - 1
    middle_end = 2 * n - 1
    corr_values = corr[middle_start:middle_end]
    return np.max(corr_values) * dx

def project_h(h, n_points):
    """Project h to meet constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2"""
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def genetic_algorithm_phase(start_h, n_points, time_budget):
    """Perform genetic algorithm phase to explore solution space"""
    pop_size = 50
    max_generations = 1000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.2
    pop = []
    
    # Initialize population with variations of start_h
    pop.append(start_h)
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, 0.1, n_points)
        h = start_h + noise
        h = project_h(h, n_points)
        pop.append(h)
    
    best_h = start_h.copy()
    best_C5 = evaluate_C5(best_h)
    
    start_time = time.time()
    time_left = time_budget * 0.7
    
    for generation in range(max_generations):
        if time.time() - start_time > time_left:
            break
            
        # Evaluate all individuals
        fitness_scores = []
        for h in pop:
            C5 = evaluate_C5(h)
            fitness_scores.append(C5)
        
        # Sort population by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness_scores)
        pop = [pop[i] for i in sorted_indices]
        fitness_scores.sort()
        
        # Keep elite individuals
        num_elite = int(elitism_rate * pop_size)
        elite = pop[:num_elite]
        new_pop = elite.copy()
        
        # Generate offspring
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(range(num_elite))
            parent2_idx = np.random.choice(range(num_elite))
            parent1 = pop[parent1_idx]
            parent2 = pop[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                children = [child1, child2]
            else:
                children = [parent1.copy(), parent2.copy()]
            
            # Mutation
            for child in children:
                if np.random.rand() < mutation_rate:
                    # Add small Gaussian noise
                    noise = np.random.normal(0, 0.05, n_points)
                    child += noise
                    child = project_h(child, n_points)
                new_pop.append(child)
        
        # Ensure we have exactly pop_size individuals
        pop = new_pop[:pop_size]
        
        # Update best solution
        current_best = pop[0]
        current_C5 = evaluate_C5(current_best)
        if current_C5 < best_C5:
            best_h = current_best.copy()
            best_C5 = current_C5
            
    return best_h, best_C5

def gradient_descent_phase(start_h, n_points, time_budget):
    """Perform gradient descent optimization to refine solution"""
    h = start_h.copy()
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    
    start_time = time.time()
    time_left = time_budget * 0.3
    
    # Set learning rate and parameters
    learning_rate = 0.01
    max_iter = 5000
    patience = 100
    patience_counter = 0
    prev_C5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > time_left:
            break
            
        # Compute C5 with current h
        current_C5 = evaluate_C5(h)
        
        # Check if solution improved
        if current_C5 < best_C5:
            best_h = h.copy()
            best_C5 = current_C5
            patience_counter = 0
        else:
            patience_counter += 1
            if patience_counter >= patience:
                # Reduce learning rate
                learning_rate *= 0.5
                patience_counter = 0
                if learning_rate < 1e-6:
                    break
                
        # Compute gradient by finite difference approximation
        grad = np.zeros(n_points)
        epsilon = 1e-6
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = project_h(h_plus, n_points)
            C5_plus = evaluate_C5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= epsilon
            h_minus = project_h(h_minus, n_points)
            C5_minus = evaluate_C5(h_minus)
            
            grad[i] = (C5_plus - C5_minus) / (2 * epsilon)
        
        # Update h
        h = h - learning_rate * grad
        h = project_h(h, n_points)
        
    return best_h, best_C5

        start_h = project_h(h, n_points)
    
    # Phase 1: Genetic Algorithm
    best_h_ga, best_C5_ga = genetic_algorithm_phase(start_h, n_points, budget_s)
