import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        # Initialize with a smooth function
        n_points = 42
        h = np.linspace(0, 1, n_points)
        h = 0.5 * (1 + np.sin(np.linspace(0, 4*np.pi, n_points)))
        h = h / np.sum(h) * (n_points / 2)
    
    n_points = len(h)
    dx = 2.0 / n_points
    current_best = float('inf')
    best_h = h.copy()
    last_improvement = 0
    
    def objective(h):
        # Compute C5
        corr = np.correlate(h, 1 - h, mode="full")
        max_corr = np.max(corr) * dx
        return max_corr
    
    def gradient(h):
        # Approximate gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h)
        orig_obj = objective(h)
        for i in range(len(h)):
            h_plus = h.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - orig_obj) / eps
        return grad
    
    def project(h):
        # Clip to [0,1] and rescale
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    current_h = h.copy()
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        grad = gradient(current_h)
        step_size = 0.01 / (0.1 + np.linalg.norm(grad))
        current_h = current_h - step_size * grad
        current_h = project(current_h)
        
        current_obj = objective(current_h)
        if current_obj < current_best:
            current_best = current_obj
            best_h = current_h.copy()
            last_improvement = iteration
            
        if current_obj < 0.3807:
            break
            
        if iteration - last_improvement > 100:
            break
            
    return (best_h.tolist(), current_best, n_points)
