import numpy as np
import time
from scipy.signal import correlate
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    def evaluate(h):
        # Ensure h is a valid vector
        h = np.array(h)
        # Compute overlap integral for each k
        corr = correlate(h, 1 - h, mode='full')
        max_overlap = np.max(corr) * dx
        return max_overlap if not np.isnan(max_overlap) else float('inf')
    
    def rescale(h):
        # Rescale h to satisfy sum constraint
        current_sum = np.sum(h)
        if abs(current_sum - target_sum) < 1e-10:
            return h
        alpha = target_sum / current_sum
        h_scaled = h * alpha
        return h_scaled
    
    def mutate(h, rate=0.2, strength=0.1):
        h = np.array(h)
        if np.random.random() > rate:
            return h
        # Select random points to modify
        n_mutate = max(1, int(len(h) * 0.1))
        indices = np.random.choice(len(h), size=n_mutate, replace=False)
        for i in indices:
            delta = np.random.uniform(-strength * dx, strength * dx)
            h[i] += delta
        h = np.clip(h, 0, 1)
        h = rescale(h)
        return h
    
    def crossover(p1, p2):
        # Segment swap crossover
        if np.random.random() < 0.5:
            return p1, p2
        # Random segment swap
        cut_point = np.random.randint(1, len(p1) - 1)
        child1 = np.concatenate([p1[:cut_point], p2[cut_point:]])
        child2 = np.concatenate([p2[:cut_point], p1[cut_point:]])
        return child1, child2
    
    def initialize_population(size, initial_h):
        population = [initial_h]
        for _ in range(size - 1):
            h = np.random.rand(n_points)
            h = rescale(h)
            population.append(h)
        return population
    
    def tournament_selection(population, k=5):
        selected = []
        for _ in range(len(population)):
            candidates = random.sample(population, k)
            best = min(candidates, key=evaluate)
            selected.append(best)
        return selected
    
    # Initialize
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
    else:
        initial_h = np.ones(n_points) * 0.5
    initial_h = rescale(initial_h)
    population_size = 100
    population = initialize_population(population_size, initial_h)
    best_h = min(population, key=evaluate)
    best_c5 = evaluate(best_h)
    
    start_time = time.time()
    generations_without_improvement = 0
    max_generations_without_improvement = 50
    
    while time.time() - start_time < budget_s - 30:
        # Evaluate all
        population = sorted(population, key=evaluate)
        current_best = evaluate(population[0])
        if current_best < best_c5:
            best_h = population[0]
            best_c5 = current_best
            generations_without_improvement = 0
        else:
            generations_without_improvement += 1
        if generations_without_improvement > max_generations_without_improvement:
            break
            
        # Selection
        selected = tournament_selection(population)
        # Crossover and mutation
        new_population = []
        for i in range(0, len(selected), 2):
            p1 = selected[i]
            p2 = selected[(i + 1) % len(selected)]
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            new_population.extend([c1, c2])
            
        # Keep top half of new population plus top half of old population
        all_candidates = population[:len(population)//2] + new_population
        population = sorted(all_candidates, key=evaluate)[:population_size]
        
        if time.time() - start_time > budget_s - 5:
            break
    
    return (best_h.tolist(), best_c5, n_points)
