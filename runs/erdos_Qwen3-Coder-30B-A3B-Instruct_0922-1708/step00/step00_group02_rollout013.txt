import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 45
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 2.0))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_C5(h):
        # Compute correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Scale by dx
        dx = 2.0 / n_points
        return np.max(corr) * dx
    
    def project_h(h):
        # Clip values
        h = np.clip(h, 0, 1)
        # Rescale to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    # Phase 1: Genetic Algorithm
    ga_time_limit = 0.7 * budget_s
    pop_size = 50
    gen = 0
    while time.time() - start_time < ga_time_limit and gen < 1000:
        gen += 1
        # Create population
        population = []
        for _ in range(pop_size):
            # Mutate a copy of h
            mut_h = h + np.random.normal(0, 0.1, n_points)
            mut_h = project_h(mut_h)
            population.append(mut_h)
        
        # Evaluate fitness
        fitness = [compute_C5(p) for p in population]
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        new_pop = population[:pop_size // 5]
        best_h_in_gen = population[0]
        best_C5_in_gen = fitness[0]
        if best_C5_in_gen < best_C5:
            best_h = best_h_in_gen.copy()
            best_C5 = best_C5_in_gen
            
        # Crossover and mutation
        while len(new_pop) < pop_size:
            parent1, parent2 = np.random.choice(population[:pop_size//2], 2, replace=False)
            # Crossover
            cross_point = np.random.randint(1, n_points - 1)
            child = np.concatenate([parent1[:cross_point], parent2[cross_point:]])
            # Mutation
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    child[i] += np.random.normal(0, 0.1)
            child = project_h(child)
            new_pop.append(child)
        
        population = new_pop
    
    # Phase 2: Gradient Descent
    h = best_h.copy()
    current_C5 = best_C5
    
    def gradient_descent_step(h, max_iter=1000):
        learning_rate = 0.01
        patience = 100
        patience_counter = 0
        prev_C5 = np.inf
        
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 10:
                break
                
            # Compute the maximum overlap and the corresponding shift
            corr = np.correlate(h, 1 - h, mode='full')
            shift_idx = np.argmax(corr) - (len(h) - 1)
            max_corr = np.max(corr)
            
            # Compute gradient of the maximized correlation w.r.t. h
            grad = np.zeros_like(h)
            # This is a rough gradient approximation
            for j in range(len(h)):
                shifted_h = np.roll(h, j - shift_idx)
                grad[j] = 1 - 2 * shifted_h[j]
            
            # Update h
            h_new = h - learning_rate * grad
            h_new = project_h(h_new)
            
            new_C5 = compute_C5(h_new)
            if new_C5 < current_C5:
                h = h_new
                current_C5 = new_C5
                patience_counter = 0
            else:
                patience_counter += 1
                if patience_counter > patience and learning_rate > 1e-6:
                    learning_rate *= 0.5
                    patience_counter = 0
            
            if np.abs(current_C5 - prev_C5) < 1e-8:
                break
            prev_C5 = current_C5
            
        return h, current_C5
            
    try:
        h, current_C5 = gradient_descent_step(h)
    except:
        pass
        
    if current_C5 < best_C5:
        best_h = h
        best_C5 = current_C5
    
    return (best_h, best_C5, n_points)
