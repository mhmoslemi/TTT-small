import numpy as np
import scipy.optimize as opt
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Get initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.full(n_points, 0.5 * dx)
    
    # Normalize h0 to satisfy constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Helper function to compute C5
    def compute_c5(h):
        h = np.array(h)
        corr = np.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx

    # Gradient-based optimization using L-BFGS-B
    def objective(h):
        return compute_c5(h)
    
    def constraint_fun(h):
        return np.sum(h) - n_points / 2
    
    # Bounds: 0 <= h[i] <= 1
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints
    cons = {'type': 'eq', 'fun': constraint_fun}
    
    # Optimization parameters
    opt_result = None
    best_h = h0.copy()
    best_c5 = compute_c5(h0)
    
    # Try multiple random restarts
    for i in range(5):
        # Random perturbation of h0
        h_init = h0 + np.random.normal(0, 0.01, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        try:
            opt_result = opt.minimize(objective, h_init, method='L-BFGS-B', 
                                      bounds=bounds, constraints=cons,
                                      options={'maxiter': 1000})
            if opt_result.success:
                h_opt = opt_result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                c5_current = compute_c5(h_opt)
                if c5_current < best_c5:
                    best_h = h_opt.copy()
                    best_c5 = c5_current
        except:
            pass
            
        # Early stopping
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final check
    final_h = best_h.copy()
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    final_c5 = compute_c5(final_h)
    
    return (final_h.tolist(), final_c5, n_points)
