import numpy as np
from scipy.optimize import differential_evolution, minimize
from scipy import signal
import time

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for h"""
    # Compute cross-correlation: h * (1 - h)
    corr = signal.correlate(h, 1 - h, mode='full')
    # Find maximum value and scale by dx
    max_overlap = np.max(corr)
    return max_overlap * dx

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h)*dx = 1"""
    h = np.clip(h, 0, 1)
    # Now adjust the sum to be exactly n_points/2 (with dx = 2/n_points)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h
