def evaluate_erdos_solution(h):
    """
    Evaluate the maximum overlap integral C5
    """
    n = len(h)
    dx = 2.0 / n
    # Compute by convolution via numpy
    # h(x) * (1 - h(x+k)) over all k
    # Full convolution gives matrix that captures all lags
    conv = np.correlate(h, 1 - h, mode="full") * dx
    # Max of the full convolution
    return np.max(conv)
