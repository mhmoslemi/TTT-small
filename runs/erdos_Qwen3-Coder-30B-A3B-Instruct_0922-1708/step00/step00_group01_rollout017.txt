import numpy as np
import time
from scipy.signal import correlate
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset = 20
    
    # Initialize
    dx = 2.0 / n_points
    
    # Use initial if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
        
    # Create swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Initialize particles
    for i in range(swarm_size):
        if i == 0:
            # First particle is the initial solution
            h = h0.copy()
        else:
            # Perturb the initial solution
            h = h0 + np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            # Rescale to meet constraint
            h = h / np.sum(h) * (n_points / 2)
            
        swarm.append(h)
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_values.append(float('inf'))
        
    # Global best tracking
    global_best = None
    global_best_value = float('inf')
    
    start_time = time.time()
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Evaluate current swarm
        for i in range(swarm_size):
            h = swarm[i]
            # Compute correlation for all lags
            corr = correlate(h, 1 - h, mode='full')
            # The maximum index corresponds to k=0 to k=2*len(h)-2
            # We need only lags from 0 to 2*n_points-2, so we take middle part
            # Actually, for overlap integral we are interested in all lags
            max_corr = np.max(corr) * dx
            if max_corr < personal_best_values[i]:
                personal_best[i] = h.copy()
                personal_best_values[i] = max_corr
                if max_corr < global_best_value:
                    global_best = h.copy()
                    global_best_value = max_corr
                    last_improvement = iter_count
        
        # Update swarm
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # PSO update
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Update position
            new_h = h + v
            
            # Projection onto feasible set
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h / np.sum(new_h) * (n_points / 2)
            
            swarm[i] = new_h
            velocities[i] = v
            
        iter_count += 1
        
        # Check for stagnation and diversity reset
        if iter_count - last_improvement > stagnation_limit:
            # Inject diversity
            reset_count = int(swarm_size * diversity_reset / 100)
            for i in range(reset_count):
                idx = np.random.randint(0, swarm_size)
                h = np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                swarm[idx] = h
                velocities[idx] = np.zeros(n_points)
        
    # Return best solution found
    if global_best is None:
        # Fallback to initial if needed
        global_best = h0
        corr = correlate(global_best, 1 - global_best, mode='full')
        global_best_value = np.max(corr) * dx
        
    return (global_best.tolist(), global_best_value, n_points)
