import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Set up parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h.copy()
        h_sum = np.sum(h)
        if abs(h_sum * dx - 1.0) > 1e-10:
            h = h * (target_sum / h_sum)
    else:
        h = np.full(n_points, 0.5)
    
    # Helper to compute overlaps efficiently
    def compute_overlap(h_vals):
        # compute all overlaps for all lags
        n = len(h_vals)
        overlaps = np.zeros(2 * n - 1)
        for k in range(2 * n - 1):
            overlap = 0.0
            for i in range(n):
                j = (i + k) % n
                overlap += h_vals[i] * (1.0 - h_vals[j])
            overlaps[k] = overlap * dx
        return np.max(overlaps)
    
    # Helper to evaluate solution and return C5 bound
    def evaluate(h_vals):
        return compute_overlap(h_vals)
    
    # Helper to project h to feasible region
    def project_h(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        h_sum = np.sum(h_vals)
        if abs(h_sum) > 1e-10:
            h_vals = h_vals * (target_sum / h_sum)
        return h_vals
    
    # Main optimization loop
    start_time = time.time()
    best_h = project_h(h.copy())
    best_C5 = evaluate(best_h)
    patience = 0
    patience_limit = 50
    learning_rate = 0.1
    
    # Gradient-based update for overlaps
    def compute_gradient(h_vals):
        n = len(h_vals)
        grad = np.zeros(n)
        for i in range(n):
            for k in range(n):
                j = (i + k) % n
                grad[i] += (1.0 - h_vals[j]) * dx
        return grad
    
    iteration = 0
    while time.time() - start_time < budget_s - 1.0 and patience < patience_limit:
        iteration += 1
        
        # Estimate overlap for each lag using current h
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h[i] * (1.0 - h[j])
            overlaps[k] = overlap * dx
            
        # Use the lag that maximizes overlap to compute gradient
        max_lag = np.argmax(overlaps)
        
        # Compute gradient of max overlap with respect to h[i]
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_lag) % n_points
            grad[i] = (1.0 - h[j]) * dx
            
        # Update h in direction of negative gradient
        h_new = h - learning_rate * grad
        h_new = project_h(h_new)
        
        # Check improvement
        current_C5 = evaluate(h_new)
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h_new.copy()
            patience = 0
        else:
            patience += 1
            
        h = h_new
        if iteration % 10 == 0:
            learning_rate *= 0.9
            if learning_rate < 1e-6:
                learning_rate = 1e-6
    
    return (best_h.tolist(), best_C5, n_points)
