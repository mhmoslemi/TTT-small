import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Load initial solution if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Create a symmetric initial solution
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Make a plateau in the middle
        mid = n_points // 2
        left = n_points // 4
        right = 3 * n_points // 4
        h[left:right] = 1.0
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_C5 = compute_C5(h, dx)
    t0 = time.time()
    
    # Parameters
    lr = 0.1
    anneal_freq = 1000
    max_iter = 5000
    decay = 0.5
    
    for i in range(max_iter):
        if time.time() - t0 > budget_s:
            break
            
        # Compute cross-correlations (h * (1 - h)) for all lags
        h1 = 1 - h
        conv = correlate(h, h1, mode='full')
        # The overlap integrals are computed over [0,2] with dx spacing
        # corresponding to the full correlation result over [-1,1]
        # But we're actually interested in lags 0 to n_points-1
        # Which corresponds to indices [n_points-1, 2*n_points - 2] in fullcorr
        # That is because of the way correlate works
        # So we consider lags from 0 to n_points-1 (inclusive)
        # And we only want the first n_points lags
        C5_values = conv[n_points-1:] * dx
        max_idx = np.argmax(C5_values)
        max_C5 = C5_values[max_idx]
        
        # Compute gradient w.r.t. h, only for the max term 
        # The derivative of h[i] * (1 - h[i + k]) w.r.t. h[i] is (1 - h[i+k]) 
        # But we are taking this over the entire convolution
        # It's easier than the algebraic expression, directly modify the gradient
        
        # Let's do just the one contributing to the max overlap
        grad = np.zeros_like(h)
        if max_idx < n_points:
            # Cross correlate this with (1-h) as if we're taking derivative
            # It's like convolution with reverse of (1-h)
            rev_h1 = h1[::-1]  # This represents 1-h shifted backwards 
            # Construct kernel to represent derivative approximation  
            # We compute - gradient of max_C5_i = h@h1 with respect to h_i
            # For a fixed lag k, at position i, the contribution to the overlap is h[i]*h1[i+k]
            # So d/dh[i] is h1[i+k]
            for j in range(n_points):
                k = j + max_idx
                if k < n_points:
                    grad[j] += h1[k]
        
        # Update h
        h = h - lr * grad
        
        # Project to [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to satisfy the integral constraint
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
            
        # Evaluate new value
        new_C5 = compute_C5(h, dx)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            
        # Anneal learning rate
        if (i + 1) % anneal_freq == 0:
            lr *= decay
    
    return (best_h.tolist(), best_C5, n_points)

def compute_C5(h, dx):
    # Compute the full cross-correlation between h and (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Extract the valid overlaps (lag from 0 to n-1)
    C5_vals = corr[len(h)-1:] * dx
    return np.max(C5_vals)
