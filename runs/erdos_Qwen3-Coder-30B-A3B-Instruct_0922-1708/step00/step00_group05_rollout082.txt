import numpy as np
import cvxpy as cp
import time
from math import sqrt

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create a random one
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        n_points = 70
        h_init = np.ones(n_points) * 0.5
        # Ensure sum is exactly n_points / 2
        h_init[0] = h_init[0] * n_points / (2 * np.sum(h_init))
    
    # Discretization
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    max_attempts = 100
    attempts = 0
    
    # Iterative refinement with multiple random initializations
    for trial in range(5):
        if attempts >= max_attempts:
            break
        h_current = h_init.copy()
        # Add small random noise
        h_current += np.random.normal(0, 0.01, n_points)
        h_current = np.clip(h_current, 0, 1)
        # Normalize to sum = n_points / 2
        h_current = h_current * (n_points / 2) / np.sum(h_current)
        
        # Try to solve QCQP
        for _ in range(5):
            if attempts >= max_attempts or time.time() - start_time > budget_s - 10:
                break
            attempts += 1
            
            # Define variables
            h = cp.Variable(n_points)
            C = cp.Variable()
            
            # Objective
            objective = cp.Minimize(C)
            
            # Constraints
            constraints = [h >= 0, h <= 1, cp.sum(h) == n_points / 2]
            
            # Compute overlap integrals for each lag
            for k in range(-n_points + 1, n_points):
                # For lag k:
                # Compute indices that go into correlation
                if k >= 0:
                    left_idx = range(n_points - k)
                    right_idx = range(k, n_points)
                    corr_val = cp.sum(h[left_idx] * (1 - h[right_idx]))
                else:
                    left_idx = range(-k, n_points)
                    right_idx = range(n_points + k)
                    corr_val = cp.sum(h[left_idx] * (1 - h[right_idx]))
                constraints.append(corr_val <= C * dx)
            
            # Formulate problem
            prob = cp.Problem(objective, constraints)
            
            try:
                prob.solve(solver=cp.CVXOPT, verbose=False)
                if prob.status in ["optimal", "optimal_inaccurate"]:
                    # Get solution
                    h_sol = h.value
                    if h_sol is not None:
                        # Check constraints
                        if np.all(h_sol >= 0) and np.all(h_sol <= 1):
                            new_sum = np.sum(h_sol)
                            if abs(new_sum - n_points/2) < 1e-6:
                                # Re-evaluate C5
                                corr_full = np.correlate(h_sol, 1 - h_sol, mode='full')
                                max_overlap = np.max(corr_full) * dx
                                if max_overlap < best_c5:
                                    best_c5 = max_overlap
                                    best_h = h_sol.copy()
            except Exception:
                pass
            
            # Perturb slightly for next iteration
            h_current = h_current + np.random.normal(0, 0.005, n_points)
            h_current = np.clip(h_current, 0, 1)
            h_current = h_current * (n_points / 2) / np.sum(h_current)
            h_init = h_current.copy()
        
        # Early stopping if we're making progress
        if time.time() - start_time > budget_s - 10:
            break
    
    if best_h is None:
        best_h = h_init
        corr_full = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(corr_full) * dx
    
    return (best_h.tolist(), best_c5, n_points)
