import numpy as np
import time
from scipy import correlate
from scipy.signal import fftconvolve

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1 - h, mode="full")
    # Take the maximum (excluding the padding)
    max_corr = np.max(corr[len(h)-1:])
    return max_corr * dx

def normalize_h(h, target_sum):
    """Normalize h to have the target sum."""
    current_sum = np.sum(h)
    if abs(current_sum) < 1e-10:
        h = np.ones_like(h) * (target_sum / len(h))
    else:
        h = h * (target_sum / current_sum)
    return h

def project_to_bounds(h, lower=0, upper=1):
    """Clip h to bounds."""
    return np.clip(h, lower, upper)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_iter = int(budget_s / 0.05)  # approx 12000 iterations
    patience = 100
    
    # Initialize particles
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.ones(n_points) * 0.5
        
    # Normalize the initial solution to have sum = n_points / 2
    h0 = normalize_h(h0, n_points / 2)
    
    particles = []
    velocities = []
    
    # One particle from initial solution
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Others with small noise
    for _ in range(n_particles - 1):
        h_perturb = h0 + np.random.normal(0, 0.1, n_points)
        h_perturb = project_to_bounds(h_perturb)
        h_perturb = normalize_h(h_perturb, n_points / 2)
        particles.append(h_perturb)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate particles
    fitness = []
    for i in range(n_particles):
        f = evaluate_erdos_solution(particles[i])
        fitness.append(f)
    
    # Track best
    best_idx = np.argmin(fitness)
    global_best_h = particles[best_idx].copy()
    global_best_fitness = fitness[best_idx]
    
    iteration = 0
    last_improvement = 0
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        for p in range(n_particles):
            # Update velocity
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            v_new = (w * velocities[p] + 
                     c1 * r1 * (particles[p] - particles[p]) +
                     c2 * r2 * (global_best_h - particles[p]))
            
            # Limit velocity
            v_new = np.clip(v_new, -0.1, 0.1)
            
            # Update position
            h_new = particles[p] + v_new
            
            # Ensure bounds
            h_new = project_to_bounds(h_new)
            
            # Normalize to correct sum
            h_new = normalize_h(h_new, n_points / 2)
            
            # Evaluate
            f_new = evaluate_erdos_solution(h_new)
            
            if f_new < fitness[p]:
                fitness[p] = f_new
                particles[p] = h_new.copy()
                
                if f_new < global_best_fitness:
                    global_best_fitness = f_new
                    global_best_h = h_new.copy()
                    last_improvement = iteration
        
        # Tournament selection and diversity maintenance
        if iteration % 20 == 0:
            idxs = np.argsort(fitness)
            top_half = idxs[:n_particles//2]
            
            for p in range(n_particles):
                if p not in top_half:
                    # Replace with random perturbation of global best
                    h_new = global_best_h + np.random.normal(0, 0.01, n_points)
                    h_new = project_to_bounds(h_new)
                    h_new = normalize_h(h_new, n_points / 2)
                    particles[p] = h_new.copy()
                    fitness[p] = evaluate_erdos_solution(h_new)
        
        # Check for stagnation
        if iteration - last_improvement > patience and iteration > 500:
            # Introduce diversity
            for p in range(n_particles):
                h_new = particles[p] + np.random.normal(0, 0.01, n_points)
                h_new = project_to_bounds(h_new)
                h_new = normalize_h(h_new, n_points / 2)
                particles[p] = h_new.copy()
                fitness[p] = evaluate_erdos_solution(h_new)
            last_improvement = iteration
            
        iteration += 1
        
    return (global_best_h.tolist(), global_best_fitness, n_points)
