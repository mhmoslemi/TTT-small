import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Default construction - triangular pattern
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2 * x - 1)
        h = h * (n_points / 2) / np.sum(h)  # normalize to sum = n_points/2
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Objective function and gradient computation
    def compute_overlap(h_vals):
        # Compute overlap values for all shifts k
        h_vec = np.array(h_vals)
        h_flip = 1 - h_vec
        # Use convolution for overlap
        c = np.correlate(h_vec, h_flip, mode='full')
        return c
    
    def compute_c5(h_vals):
        overlaps = compute_overlap(h_vals)
        return np.max(overlaps) * dx
    
    def grad_overlap(h_vals, k):
        # Gradient of overlap with respect to h at shift k
        h_vec = np.array(h_vals)
        n = len(h_vec)
        grad = np.zeros_like(h_vec)
        # We need the derivative of sum_i h_i * (1 - h_{i+k})
        # Which is 1 - h_{i+k} if i-k >= 0 and i+k < n
        k_idx = k + n - 1  # adjust for zero-based indexing
        for i in range(n):
            if 0 <= i-k < n:
                if i-k >= 0:
                    grad[i] += 1 - h_vec[i-k]
                if i+k < n:
                    grad[i] += 1 - h_vec[i+k]
        return grad
    
    # Iteration parameters
    max_iter = 600  # approximate number of iterations
    learning_rate = 0.1
    tolerance = 1e-6
    last_improvement = 0
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    # History tracking for adaptive learning rate
    last_c5 = best_c5
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = compute_overlap(h)
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max] * dx
        
        # If we've found a good enough solution, stop early
        if max_overlap < 0.3807:
            break
            
        # Determine worst overlapping shift
        k_worst = k_max - (n_points - 1)
        
        # Construct local minimization problem around worst overlap shift
        # For consistency, we just use full gradient descent
        # We will update h towards reducing the maximum overlap
        
        # Compute gradients for the max overlap shift
        grad = np.zeros_like(h)
        if k_worst < 0:
            # h_i (1 - h_{i-k})  for i=0..n+k-1 inclusive
            for i in range(n_points + k_worst):
                if i >= 0 and i+k_worst < n_points:
                    grad[i] += 1 - h[i+k_worst]
        else:
            # h_i (1 - h_{i+k})  for i=k..n-1 inclusive  
            for i in range(k_worst, n_points):
                grad[i] -= (1 - h[i-k_worst])
        
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Enforce sum constraint exactly
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new configuration
        c5_new = compute_c5(h_new)
        
        # Adaptive learning rate
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            last_improvement = it
        else:
            learning_rate *= 0.99
            if learning_rate < 1e-6:
                learning_rate = 1e-6
                
        if abs(c5_new - last_c5) < tolerance:
            learning_rate *= 0.95
            if learning_rate < 1e-7:
                break
                
        last_c5 = c5_new
        h = h_new.copy()
    
    # The final result
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), float(final_c5), int(n_points))
