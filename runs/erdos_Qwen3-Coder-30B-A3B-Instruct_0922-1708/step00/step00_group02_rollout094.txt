import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem constants
    n_points = 100  # reasonable compromise size
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Basis functions: sigmoidal basis centered at uniform points
    m = int(2 * np.sqrt(n_points))  # number of basis functions
    if m < 2:
        m = 2
    centers = np.linspace(0, 2, m, endpoint=False)
    scales = 2.0 / m
    if scales < 1e-8:
        scales = 1e-8
    
    def basis(x, j):
        return expit((x - centers[j]) / scales)
    
    def eval_h(x, coeffs):
        return np.sum([coeffs[j] * basis(x, j) for j in range(m)], axis=0)
    
    def eval_h_single(x, coeffs):
        return np.sum([coeffs[j] * basis(x, j) for j in range(m)])
    
    def compute_overlap_from_coeffs(coeffs):
        h_vals = np.array([eval_h_single(x, coeffs) for x in np.linspace(0, 2, n_points)])
        # Clip to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        h_vals = np.array(h_vals)
        # Normalize to sum = target_sum
        h_vals = h_vals * (target_sum / np.sum(h_vals))
        # Compute overlap integral
        hval = np.concatenate([h_vals, np.zeros(n_points)])
        hval_rev = np.concatenate([np.zeros(n_points), 1 - h_vals])
        corrs = np.correlate(hval, hval_rev, mode='full')
        max_correlation = np.max(corrs[n_points:])
        return max_correlation * dx
    
    def update_coeffs(coeffs, grad, lr):
        new_coeffs = coeffs - lr * grad
        # Project to feasible region
        new_coeffs = np.clip(new_coeffs, -10, 10)
        return new_coeffs
    
    # Use initial_h_values if available as starting point
    best_coeffs = None
    best_c5 = float('inf')
    
    init_h = None
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
    else:
        init_h = np.ones(n_points) * 0.5  # default uniform start
    
    # Initialize coefficients using least squares projection
    def initialize_coefficients(h_start):
        A = np.zeros((n_points, m))
        x_vals = np.linspace(0, 2, n_points)
        for i in range(n_points):
            for j in range(m):
                A[i, j] = basis(x_vals[i], j)
        # Solve for coefficients using SVD
        try:
            coeffs = np.linalg.lstsq(A, h_start, rcond=None)[0]
        except:
            coeffs = np.random.rand(m) * 0.1
        return coeffs
    
    if init_h is not None:
        coeffs_start = initialize_coefficients(init_h)
    else:
        coeffs_start = np.random.rand(m) * 0.1
    
    coeffs = coeffs_start.copy()
    lr = 1e-2
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < 0.95 * budget_s:
        iteration += 1
        if iteration > 500:
            break
            
        try:
            c5 = compute_overlap_from_coeffs(coeffs)
            if c5 < best_c5:
                best_c5 = c5
                best_coeffs = coeffs.copy()
                last_improvement = iteration
                
            # Numerical gradient approximation
            eps = 1e-4
            grad = np.zeros_like(coeffs)
            for j in range(m):
                coeffs_plus = coeffs.copy()
                coeffs_minus = coeffs.copy()
                coeffs_plus[j] += eps
                coeffs_minus[j] -= eps
                c_plus = compute_overlap_from_coeffs(coeffs_plus)
                c_minus = compute_overlap_from_coeffs(coeffs_minus)
                grad[j] = (c_plus - c_minus) / (2 * eps)
                
            coeffs = update_coeffs(coeffs, grad, lr)
            
            if iteration % 100 == 0 and iteration > last_improvement + 50:
                lr *= 0.5
                if lr < 1e-6:
                    break
                    
        except Exception as e:
            break

    if best_coeffs is None:
        best_coeffs = coeffs_start.copy()
    
    # Final conversion to h vector
    h_vals = np.array([eval_h_single(x, best_coeffs) for x in np.linspace(0, 2, n_points)])
    h_vals = np.clip(h_vals, 0, 1)
    h_vals = h_vals * (target_sum / np.sum(h_vals))
    
    # Validate
    total = np.sum(h_vals)
    if abs(total - target_sum) > 1e-6:
        h_vals = h_vals * (target_sum / total)
        
    return (h_vals, best_c5, n_points)
