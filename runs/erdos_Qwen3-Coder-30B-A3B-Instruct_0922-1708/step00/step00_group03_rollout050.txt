from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap for given h"""
    conv = correlate(h, 1 - h, mode="full")
    max_lag = len(conv) // 2
    return np.max(conv[:max_lag]) * dx
