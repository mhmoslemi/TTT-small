import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Constants
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h, prefer initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5 * dx)
    
    # Ensure sum constraint
    h *= target_sum / np.sum(h)
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    # Optimization parameters
    initial_sigma = 0.1
    sigma_decay = 0.99
    max_iter = 1_000_000
    patience = 1000
    patience_counter = 0
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Anneal sigma
        sigma = initial_sigma * (sigma_decay ** iteration)
        
        # Generate candidate h by perturbing best solution
        h_new = best_h + np.random.normal(0, sigma, n_points)
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain sum constraint
        h_new *= target_sum / np.sum(h_new)
        
        # Evaluate
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    # Compute overlap integral for all shifts using correlation
    corr = correlate(h, 1 - h, mode='full') * dx
    # Extract the maximum value from valid ranges
    n = len(h)
    max_idx = n - 1
    c5_val = np.max(corr[max_idx - (n-1): max_idx + n])
    return c5_val
