import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    # Compute the overlap integral for all lags
    conv = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

    def objective(h_vec):
        h = h_vec.copy()
        # Ensure constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        c5 = evaluate_c5(h, dx)
        return c5

    def grad_objective(h_vec):
        eps = 1e-6
        grad = np.zeros_like(h_vec)

            h_pert = h_vec.copy()
            h_pert[i] += eps
            grad[i] = (objective(h_pert) - base_val) / eps
        return grad
