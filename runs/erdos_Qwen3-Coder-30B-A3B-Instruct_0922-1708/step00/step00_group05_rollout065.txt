import numpy as np
import time
from scipy.signal import correlate
import math

def evaluate_erdos_solution(h):
    """Returns the C5 value for a given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations for k in [-n+1, n-1] (full mode)
    corr = correlate(h, 1 - h, mode="full")
    # Only consider lags 0 to 2*n-2
    max_k = 2 * len(h) - 2
    # Extract relevant correlations: from lag = 0 to max lag of 2*n-2
    # The overlapping integral is computed over valid overlapping regions
    # For a shift k, we measure integral of h(x) * (1-h(x+k)) 
    # h is [0, 2] length should be 2/dx per sample to cover interval [0,2]
    # We compute full correlations, but we look at k=0 to n-1 (positive lags) and negative lags.
    # By symmetry of half the function with another, it's cleaner to work with all positive shifts
    # The full len(h) * 2 - 1 due to full mode (double length of data in conv)
    # but since we have [0, 2] real interval, we do simple slice. 
    # Since correlation output has len(h) + len(1-h) - 1 = 2 * len(h) - 1, we get center as index len(h)-1
    # Index k=0 is at j = len(h) -1. Should be integral from 0 to 2. 
    # The first positive k is 0, so candidate for maximum is k from 0 to len(h)-1 in valid range:
    # E.g. k=0 -> h(0) * (1-h(0)), k=1 -> h(0) * (1-h(1)), etc. (Note: index-wise, we shift by k)
    # The overlap is "essentially" z[k] = sum(h[j] * (1 - h[j+k])) for valid j (but more complex calculation  due to boundary)
    # Direct approach to compute the actual overlap value for k = [0,l] where l could be up to n -1
    # From density of samples (regardless of how the actual transmission is based on convolution)
    # See integral form reference: this can be correctly computed as the full correlation
    # We take the max of the correlation shifted over all possible lags that yield valid offsets.
    # For k = 0 to n-1: the actual overlapping measure over 2 samples is computed using:
    # h(0)*1h(k) + h(1)*1h(k+1) + ...   i.e., end of terms to be 'adjusted' but the matter is that
    # the shifted convolution integral is effectively the total desired overlap
    # So extraction is much simpler with recurrence to FFT correlation
    # Take values from center.
    c5 = max(corr)
    return c5 * dx

def project_to_constraints(h):
    """Projects h onto [0,1] and ensures integral = 1."""
    h = np.clip(h, 0, 1)
    dx = 2.0 / len(h)
    current_sum = np.sum(h) * dx
    if abs(current_sum) < 1e-10:
        h = np.ones_like(h) * 0.5
    else:
        h = h * (1.0 / current_sum)
    return h

def pso_erdos_optimization(seed, budget_s):
    np.random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 100000  # will be bounded by time anyway
    
    # Initialize swarm
    n_particles = 20
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Initialize with uniform distribution for safety
        h_init = np.ones(n_points) * 0.5
    
    h_init = project_to_constraints(h_init)
    particles.append(h_init)
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add more diverse particles
    for _ in range(n_particles-1):
        h_mut = h_init + np.random.normal(0, 0.1, n_points)
        h_mut = project_to_constraints(h_mut)
        particles.append(h_mut)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best solutions
    personal_best = [None] * n_particles
    personal_best_scores = [float('inf')] * n_particles
    global_best = None
    global_best_score = float('inf')
    
    # Evaluate all initial particles
    start_time = time.time()
    for i in range(n_particles):
        h = particles[i].copy()
        score = evaluate_erdos_solution(h)
        personal_best_scores[i] = score
        personal_best[i] = h.copy()
        if score < global_best_score:
            global_best_score = score
            global_best = h.copy()
    
    # PSO parameters
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    inertia_reduction_factor = 0.995
    no_improve_count = 0
    max_no_improve = 100
    
    # Iterate
    iter_count = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        for p in range(n_particles):
            # Update velocity
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            
            # Make sure existing global best is used with correct indexing
            if global_best is not None:
                v_new = (w * velocities[p] + 
                         c1 * r1 * (personal_best[p] - particles[p]) +
                         c2 * r2 * (global_best - particles[p]))
                velocities[p] = v_new
                # Limit velocity
                velocities[p] = np.clip(velocities[p], -0.1, 0.1)
            else:
                velocities[p] = np.random.uniform(-0.1, 0.1, n_points)

            # Update position
            particles[p] = particles[p] + velocities[p]
            
            # Projection to [0,1]
            particles[p] = np.clip(particles[p], 0, 1)
            
            # Ensure normalization
            particles[p] = project_to_constraints(particles[p])
            
            # Evaluate
            score = evaluate_erdos_solution(particles[p])
            
            # Update personal best
            if score < personal_best_scores[p]:
                personal_best_scores[p] = score
                personal_best[p] = particles[p].copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = particles[p].copy()
                no_improve_count = 0
            else:
                no_improve_count += 1
            
            iter_count += 1

        # Frequency-based diversity injection
        if no_improve_count > max_no_improve:
            # Inject new particles from best solutions to some extent
            # Use top 50% of particles to introduce new diversity
            indices = np.argsort([personal_best_scores[i] for i in range(n_particles)])[:n_particles//2]
            for j in range(n_particles//2):
                idx = indices[j]
                pert = np.random.normal(0, 0.01, n_points)
                new_h = np.clip(personal_best[idx] + pert, 0, 1)
                new_h = project_to_constraints(new_h)
                particles[idx] = new_h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            no_improve_count = 0
            w *= inertia_reduction_factor  # Reduce inertia to intensify search

    # Final evaluation
    final_h = global_best
    final_c5 = evaluate_erdos_solution(final_h)
    return (final_h.tolist(), final_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = pso_erdos_optimization(seed, budget_s)
    return (h_values, c5_bound, n_points)
