import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_h(h, dx):
    # Compute overlap integral for all lags
    conv = correlate(h, 1 - h, mode='full')
    max_overlap = np.max(conv) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    
    # start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Verify sum constraint
        if abs(np.sum(h) * dx - 1.0) > 1e-10:
            # Rescale to meet constraint
            h = h * (1.0 / (np.sum(h) * dx)) * dx
    else:
        # Fallback: uniform
        h = np.full(n_points, 0.5 * dx)
        
    # Reshape to be on [0,1] and meet sum constraint
    h = h / dx
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)
    
    # Precompute convolution indices
    half_len = n_points - 1
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    best_iter = 0
    
    # Gradient-based optimization with L-BFGS-B
    start_time = time.time()
    
    def objective(h_vec):
        h = h_vec.copy()
        h = h * (n_points / 2.0) / np.sum(h)  # Renormalize
        h = np.clip(h, 0, 1)  # Clamp to bounds
        c5 = evaluate_h(h, dx)
        return c5

    def gradient(h_vec):
        h = h_vec.copy()
        h = h * (n_points / 2.0) / np.sum(h)
        h = np.clip(h, 0, 1)
        
        # Compute convolution
        conv = correlate(h, 1 - h, mode='full')
        
        # Find the lag which maximizes the overlap
        max_lag_idx = np.argmax(conv)
        max_lag = max_lag_idx - half_len
        
        # Compute gradient of max overlap
        grad = np.zeros_like(h)
        # For lag=0 to n_points-1: 1-h[j] gets multiplied, -1*h[j] gets multiplied
        # For lag=-n_points+1 to -1: h[j] gets multiplied, -1*(1-h[j]) gets multiplied
        
        # Iterate through all lags
        # Rewrite to find the h[i] that contributes to the max overlap based on lag
        for i in range(n_points):
            if max_lag >= 0:
                # This corresponds to h[i] * (1 - h[i+max_lag])
                if i + max_lag < n_points:
                    grad[i] = -1.0 * (1 - h[i + max_lag]) + (1 - h[i]) * (max_lag > 0) * 1.0
            else:
                # max_lag < 0
                # This corresponds to (1 - h[i]) * h[i + max_lag]
                if i + max_lag >= 0:
                    grad[i] = (1 - h[i + max_lag]) - h[i + max_lag] * (max_lag < 0) * 1.0
                    
        return grad

    # Run optimization
    max_iter = 1000
    last_improvement = 0
    current_iter = 0
    
    h_t = h.copy()
    current_c5 = evaluate_h(h_t, dx)
    best_h = h_t.copy()
    best_c5 = current_c5
    best_iter = 0
    
    while time.time() - start_time < budget_s - 5 and current_iter < max_iter:
        current_iter += 1
        # Clamp h for safety
        h_t = np.clip(h_t, 0, 1)
        # Normalize to sum constraint
        h_t = h_t * (n_points / 2.0) / np.sum(h_t)
        
        # Call scipy minimize via L-BFGS-B
        res = minimize(
            objective,
            h_t,
            method='L-BFGS-B',
            bounds=[(0, 1)] * len(h_t),
            options={'maxiter': 10, 'ftol': 1e-10, 'gtol': 1e-10}
        )
        
        h_t = res.x.copy()
        h_t = np.clip(h_t, 0, 1)
        h_t = h_t * (n_points / 2.0) / np.sum(h_t)
        
        # Evaluate
        c5_new = evaluate_h(h_t, dx)
        
        if c5_new < best_c5 - 1e-9:
            best_h = h_t.copy()
            best_c5 = c5_new
            last_improvement = current_iter
            best_iter = current_iter
        
        if current_iter - last_improvement > 100:
            break
                
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    return (list(best_h), final_c5, n_points)
