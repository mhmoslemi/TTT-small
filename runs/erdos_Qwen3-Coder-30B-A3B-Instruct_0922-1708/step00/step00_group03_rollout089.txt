import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    lambda_reg = 1e-4
    max_iter = 1000
    tolerance = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    def compute_C5(h_vals):
        # Correlation computation
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # Only the overlapping parts
        n = len(h_vals)
        corr = corr[n-1:2*n-1]
        return np.max(corr) * dx

    def objective(h_vals):
        # Add regularization
        reg_term = lambda_reg * np.sum(h_vals ** 2)
        return compute_C5(h_vals) + reg_term

    def gradient(h_vals):
        # Approximate gradient
        n = len(h_vals)
        grad = np.zeros(n)
        # Compute centered finite difference for the derivative
        eps = 1e-6
        for i in range(n):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_minus = h_vals.copy()
            h_minus[i] -= eps
            
            grad[i] = (compute_C5(h_plus) - compute_C5(h_minus)) / (2 * eps)
        
        # Add regularization term
        grad += 2 * lambda_reg * h_vals
        return grad

    def project_and_rescale(h_vals):
        # Clamp values to [0,1]
        h_vals = np.clip(h_vals, 0, 1)
        # Rescale to satisfy the integral constraint
        current_sum = np.sum(h_vals)
        if current_sum > 0:
            h_vals = h_vals * (n_points / 2) / current_sum
        return h_vals

    # Optimization loop
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute objective and gradient
        obj_val = objective(h)
        grad = gradient(h)
        
        # Perform a line search
        alpha = 1.0
        max_alpha = 1.0
        for _ in range(10):
            trial_h = h - alpha * grad
            trial_h = project_and_rescale(trial_h)
            trial_obj = objective(trial_h)
            if trial_obj < obj_val:
                h = trial_h
                break
            alpha *= 0.5

        # Final projection
        h = project_and_rescale(h)
        
        # Simple check for convergence
        if np.linalg.norm(grad) < tolerance:
            break

    # Local shift search
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    for i in range(20):
        if time.time() - start_time > budget_s - 1:
            break
        # Try shifting
        shift = np.random.randint(0, n_points)
        shifted_h = np.roll(h, shift)
        shifted_C5 = compute_C5(shifted_h)
        if shifted_C5 < best_C5:
            best_C5 = shifted_C5
            best_h = shifted_h.copy()

    return (best_h, best_C5, n_points)
