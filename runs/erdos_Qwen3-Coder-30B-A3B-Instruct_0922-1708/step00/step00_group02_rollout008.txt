    h = h * target_sum / np.sum(h)
    
    # Precompute indices for fast overlap computation
    def compute_overlap(h, k):
        """Compute overlap for shift k"""
        n = len(h)
        if k >= n:
            return 0.0
        return np.sum(h[:n-k] * (1 - h[k:]))
    
    def compute_all_overlaps(h):
        """Compute all overlaps for all shifts"""
        n = len(h)
        overlaps = []
        for k in range(2*n - 1):
            overlap = 0.0
            # Compute the overlap for shift k
            if k < n:
                overlap = np.sum(h[:n-k] * (1 - h[k:]))
            else:
                overlap = np.sum(h[k-n+1:] * (1 - h[:n-k+1]))
            overlaps.append(overlap * dx)
        return overlaps
    
    def compute_C5(h):
        """Compute C5"""
        overlaps = compute_all_overlaps(h)
        return max(overlaps)
    
    # Gradient computation (approximate)
    def compute_gradient(h, k_max):
        """Compute gradient of overlap with respect to shift k_max"""
        n = len(h)
        grad = np.zeros_like(h)
        if k_max >= n:
            return grad
            
        # For overlap at shift k_max:
        #         sum_i h[i] * (1 - h[i + k_max])
        # Gradient w.r.t h[i]:
        #         (1 - h[i + k_max])     for i < n - k_max  
        #         -h[i - k_max]          for i >= k_max
        
        for i in range(n):
            if i < n - k_max:
                grad[i] += (1 - h[i + k_max])
            if i >= k_max:
                grad[i] -= h[i - k_max]
        return grad * dx
    
    # Optimization loop
    max_iter = 10000
    alpha = 0.01
    patience = 0
    best_C5 = float('inf')
    best_h = h.copy()

        # Compute all overlaps
        overlaps = compute_all_overlaps(h)
        k_max = np.argmax(overlaps)
        current_C5 = overlaps[k_max]
        
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()

        grad = compute_gradient(h, k_max)
        
        # Update h

    # Final evaluation
