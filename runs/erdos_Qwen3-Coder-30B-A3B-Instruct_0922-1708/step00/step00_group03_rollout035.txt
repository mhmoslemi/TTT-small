    def compute_overlap(h):
        h_padded = np.concatenate([h, h[:n_points-1]])
        corr = np.correlate(h_padded, 1 - h, mode='valid')
        return np.max(corr) * dx
    
    def objective(h):
        # Add regularization to make it differentiable
        overlap = compute_overlap(h)
        reg = lambda_reg * np.sum(h**2)
        return overlap + reg
    
    def grad(h):
        h_padded = np.concatenate([h, h[:n_points-1]])
        overlap = compute_overlap(h)
        # Compute gradient for overlap part
        corr = np.correlate(h_padded, 1 - h, mode='valid')
        k_max = np.argmax(corr)
        # Gradient of h[i] w.r.t. overlap with shift k_max
        grad_h = np.zeros_like(h)
        for i in range(n_points):
            j = (i - k_max) % n_points
            grad_h[i] += (1 - h[j]) * dx
            grad_h[j] -= h[i] * dx
        # Add gradient of regularization
        grad_h += 2 * lambda_reg * h
        return grad_h
    
    # Optimization loop
    max_iter = 800
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        # Get current overlap
        overlap = compute_overlap(h)
        # Compute gradient
        g = grad(h)
        # Line search
        alpha = 1.0
        while True:
            h_new = h - alpha * g
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            overlap_new = compute_overlap(h_new)
            # Armijo condition
            if overlap_new <= overlap - 0.5 * alpha * np.dot(g, h - h_new):
                h = h_new
                break
            alpha *= 0.5
    
    # Final evaluation
    final_c5 = compute_overlap(h)
    return (h.tolist(), final_c5, n_points)
