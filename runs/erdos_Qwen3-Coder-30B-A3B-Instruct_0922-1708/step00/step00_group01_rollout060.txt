import numpy as np
from scipy.optimize import differential_evolution, minimize
import time

def evaluate_C5(h, dx):
    # Compute the cross-correlation of h with (1 - h)
    n = len(h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Return the maximum value scaled by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points, dx):
    # Ensure h is in [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to satisfy sum constraint: sum(h) * dx = 1
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def fitness(h, n_points, dx):
    h_proj = project_to_feasible(h, n_points, dx)
    return evaluate_C5(h_proj, dx)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        h0 = project_to_feasible(h0, n_points, 2.0/n_points)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_C5 = fitness(best_h, n_points, dx)
    
    current_time = time.time()
    elapsed = current_time - start_time
    remaining_budget = budget_s - elapsed
    
    if remaining_budget < 1:
        return (best_h.tolist(), best_C5, n_points)
    
    # Local refinement with Nelder-Mead using the best solution found so far
    # Run local optimization on the current best
    try:
        res = minimize(
            lambda x: fitness(x, n_points, dx),
            best_h,
            method='Nelder-Mead',
            options={'maxiter': 500, 'adaptive': True}
        )
        if res.success:
            refined_h = project_to_feasible(res.x, n_points, dx)
            refined_C5 = evaluate_C5(refined_h, dx)
            if refined_C5 < best_C5:
                best_h = refined_h
                best_C5 = refined_C5
    except Exception:
        pass
    
    # Use differential evolution for global search
    bounds = [(0, 1) for _ in range(n_points)]
    try:
        de_res = differential_evolution(
            lambda x: fitness(x, n_points, dx),
            bounds,
            maxiter=int(remaining_budget * 10),
            popsize=15,
            seed=seed,
            polish=True,
            strategy='best1bin'
        )
        if de_res.success:
            final_h = project_to_feasible(de_res.x, n_points, dx)
            final_C5 = evaluate_C5(final_h, dx)
            if final_C5 < best_C5:
                best_h = final_h
                best_C5 = final_C5
    except Exception:
        pass
    
    # Final local refinement
    try:
        res = minimize(
            lambda x: fitness(x, n_points, dx),
            best_h,
            method='Nelder-Mead',
            options={'maxiter': 500, 'adaptive': True}
        )
        if res.success:
            refined_h = project_to_feasible(res.x, n_points, dx)
            refined_C5 = evaluate_C5(refined_h, dx)
            if refined_C5 < best_C5:
                best_h = refined_h
                best_C5 = refined_C5
    except Exception:
        pass
    
    # Final validation
    best_h = project_to_feasible(best_h, n_points, dx)
    best_C5 = evaluate_C5(best_h, dx)
    
    return (best_h.tolist(), best_C5, n_points)
