        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    def compute_c5(h_vals):
        # Convolution with 1-h
        conv = correlate(h_vals, 1 - h_vals, mode='full')
        # Find maximum
        max_val = np.max(conv)
        return max_val / n_points * 2  # dx = 2/n_points
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    last_improvement = 0
    
    # Optimization iterations
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 600:
        iter_count += 1
        
        # Compute all overlaps
        conv = correlate(h, 1 - h, mode='full')
        k_max = np.argmax(conv)
        
        # Worst shift index (centered at 0)
        shift = k_max - (len(h) - 1)
        
        # Create gradient w.r.t. h to minimize overlap at worst shift
        grad = np.zeros_like(h)
        
        # Compute derivatives of overlapping terms for h at indices affected by shift
        if shift >= 0:
            # h[i] affects term h[i] * (1 - h[i+shift])
            # Gradient: 1 - 2*h[i] - h[shift] (if index valid)
            for i in range(len(h) - shift):
                grad[i] += 1 - 2 * h[i] - h[i + shift]
        else:
            # shift < 0
            for i in range(len(h) + shift):
                grad[i] += 1 - 2 * h[i] - h[i - shift]
        
        # Update h using projected gradient descent
        alpha = 0.01
        h_new = h - alpha * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Ensure sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new.copy()
            best_c5 = new_c5
            last_improvement = iter_count
            
        # Update h
        h = h_new.copy()
        
        # Early termination condition

        # Check for stagnation
