import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Setup
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 500
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h.copy()
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.ones(n_points) * 0.5
        h += np.random.normal(0, 0.01, n_points)
    
    # Clamp to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_c5 = evaluate_erdos_solution(h)
    best_h = h.copy()
    
    # EM loop parameters
    eta = 0.1
    patience = 0
    patience_limit = 50
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # E-step
        overlaps = []
        for k in range(n_points):
            shifted_h = np.roll(h, -k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps.append(overlap)
        overlaps = np.array(overlaps)
        
        # Normalize to form probability
        if np.sum(overlaps) > 0:
            probs = overlaps / np.sum(overlaps)
        else:
            probs = np.ones(n_points) / n_points
            
        # M-step
        # Gradient computation
        grad = np.zeros_like(h)
        for k in range(n_points):
            shifted_h = np.roll(h, k)
            grad += probs[k] * (1 - shifted_h)
        
        # Update step
        h_new = h - eta * grad
        
        # Clamp and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Compute new C5
        c5_new = evaluate_erdos_solution(h)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        # Adaptive learning rate
        if patience % 10 == 0 and patience > 0:
            eta *= 0.5
            
        if patience > patience_limit:
            break
    
    # Final check
    if evaluate_erdos_solution(best_h) < best_c5:
        best_c5 = evaluate_erdos_solution(best_h)
        
    return (best_h.tolist(), best_c5, n_points)
    
def evaluate_erdos_solution(h):
    """Compute the C_5 bound for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps via correlate
    h_rev = np.flip(h)
    correlation = np.correlate(h, h_rev, mode='full')
    # Now compute 1-h convolved with h
    one_minus_h = 1 - np.array(h)
    corr_1mh_h = np.correlate(one_minus_h, h, mode='full')
    # Total overlaps are h * (1-h) overlaps
    # The peaks of the full convolution are at indices n_points-1 to 2*n_points-2
    # The maximum overlap is the max of these
    overlaps = corr_1mh_h[n_points-1:] * dx
    return np.max(overlaps)
