import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        # Ensure sum constraint
        current_sum = np.sum(h) * dx
        h = h * target_sum / current_sum
    else:
        h = np.full(n_points, 0.5 * dx)
        noise = np.random.uniform(-0.05, 0.05, n_points) * dx
        h += noise
        h = np.clip(h, 0, 1)
        # Rescale to meet sum constraint
        current_sum = np.sum(h) * dx
        h = h * target_sum / current_sum
    
    # Function to compute overlap
    def compute_overlap(h_vec):
        # h_vec: array of size n_points
        # 1-h_vec is the complement
        # Compute full convolution
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # The overlap integral is conv * dx
        return np.max(conv) * dx

    # Objective function (to minimize)
    def objective(h_vec):
        return compute_overlap(h_vec)

    # Constraint function for sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) * dx - target_sum

    # Bounds for h values
    bounds = [(0, 1) for _ in range(n_points)]

    # Constraints
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial value array
    x0 = h.copy()

    # Store best result
    best_h = x0.copy()
    best_c5 = objective(x0)
    
    start_time = time.time()
    
    # Use scipy.optimize.minimize with L-BFGS-B
    # Method BFGS is sensitive to constraint violations, so we use L-BFGS-B instead
    iteration = 0
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Create a new random perturbation to avoid local minima
        perturbation = np.random.normal(0, 0.01, n_points) * dx
        # Ensure the perturbed vector still satisfies sum constraint
        x_perturbed = x0 + perturbation
        current_sum = np.sum(x_perturbed) * dx
        x_perturbed = x_perturbed * target_sum / current_sum
        x_perturbed = np.clip(x_perturbed, 0, 1)
        
        # Optimize from this perturbed point
        result = minimize(
            objective,
            x_perturbed,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50, 'ftol': 1e-8, 'gtol': 1e-6}
        )
        
        if result.success:
            x_opt = result.x
            current_c5 = objective(x_opt)
            if current_c5 < best_c5:
                best_h = x_opt.copy()
                best_c5 = current_c5
                last_improvement = iteration
        else:
            # If optimization fails, try a slightly different approach
            pass

        # Check for termination conditions (early stopping if no improvement)
        if iteration - last_improvement > 100:
            break
            
        # Update for next iteration
        x0 = best_h.copy()  # Update starting point for next optimization
        iteration += 1
    
    # Final validation
    final_sum = np.sum(best_h) * dx
    best_h = best_h * target_sum / final_sum
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
