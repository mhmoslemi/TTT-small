import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h_initial = initial_h_values
        n_points = len(h_initial)
    except NameError:
        n_points = 70
        h_initial = np.random.rand(n_points)
        h_initial = h_initial / np.sum(h_initial) * (n_points / 2)
    
    # Parameters
    dx = 2.0 / n_points
    max_lag = n_points - 1
    
    # Initialize best solution
    best_h = h_initial.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Try multiple random initializations
    for trial in range(5):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Start with initial solution or random
        if trial == 0:
            h = h_initial.copy()
        else:
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
        
        # Local optimization using CVXPY
        for attempt in range(5):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            try:
                # Define CVXPY variables
                h_var = cp.Variable(n_points)
                C = cp.Variable()
                
                # Constraints
                constraints = [h_var >= 0, h_var <= 1]
                constraints.append(cp.sum(h_var) == n_points / 2)
                
                # Overlap constraints for all lags
                for k in range(-max_lag, max_lag + 1):
                    if time.time() - start_time > budget_s * 0.95:
                        break
                    # Compute indices for correlation
                    if k >= 0:
                        corr_indices1 = slice(0, n_points - k)
                        corr_indices2 = slice(k, n_points)
                    else:
                        corr_indices1 = slice(-k, n_points)
                        corr_indices2 = slice(0, n_points + k)
                    
                    # Constraint: sum(h[i] * (1 - h[i + k])) <= C * dx
                    if corr_indices1.stop > corr_indices1.start:
                        overlap = cp.sum(cp.multiply(h_var[corr_indices1], 
                                                   1 - h_var[corr_indices2]))
                        constraints.append(overlap <= C * dx)
                
                # Objective: minimize C
                objective = cp.Minimize(C)
                prob = cp.Problem(objective, constraints)
                
                # Try to solve
                prob.solve(solver=cp.CVXOPT, verbose=False)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_sol = h_var.value
                    if h_sol is not None and np.all(h_sol >= 0) and np.all(h_sol <= 1):
                        c5_new = evaluate_erdos_solution(h_sol)
                        if c5_new < best_c5:
                            best_c5 = c5_new
                            best_h = h_sol.copy()
                else:
                    # Perturb slightly and retry
                    h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                    h = h / np.sum(h) * (n_points / 2)
            except:
                # Fallback: perturb and continue
                h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
                h = h / np.sum(h) * (n_points / 2)
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute full correlation for all lags
    correlation = correlate(h, 1 - h, mode='full')
    # The overlap integral for lag k corresponds to correlation[n_points-1+k]
    # We want max over all lags
    max_corr = np.max(correlation)
    return max_corr * dx
