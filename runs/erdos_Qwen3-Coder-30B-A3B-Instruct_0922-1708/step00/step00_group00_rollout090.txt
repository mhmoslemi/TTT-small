import numpy as np
import time
import random
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given h"""
    # Compute correlation of h with (1-h) in full mode
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx
    corr *= dx
    # Return maximum value
    return np.max(corr)

def normalize_h(h, target_sum):
    """Normalize h to have the specified sum"""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    scaled = h * target_sum / current_sum
    return np.clip(scaled, 0, 1)

def ensure_constraints(h, n_points):
    """Ensure h respects constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    # Clip values to [0, 1]
    h = np.clip(h, 0, 1)
    # Normalize to desired sum
    target_sum = n_points / 2.0
    h = normalize_h(h, target_sum)
    return h

def initialize_population(n_points, pop_size, initial_h):
    """Initialize population with some individuals from initial solution"""
    population = []
    # If initial solution exists, add it
    if initial_h is not None:
        h = ensure_constraints(initial_h.copy(), n_points)
        population.append(h)
    
    # Fill remaining with random solutions
    remaining = pop_size - len(population)
    for _ in range(remaining):
        h = np.random.random(n_points)
        h = ensure_constraints(h, n_points)
        population.append(h)
        
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual via tournament selection"""
    tournament_indices = random.sample(range(len(population)), tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents"""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n_points = len(parent1)
    crossover_point = random.randint(1, n_points - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Apply small random mutations to h"""
    n_points = len(h)
    mutated = h.copy()
    for i in range(n_points):
        if random.random() < mutation_rate:
            delta = np.random.normal(0, 0.05)
            mutated[i] += delta
    return mutated

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    # Settings
    n_points = 100
    pop_size = 100
    max_generations = 120
    elite_size = max(1, pop_size // 5)  # Top 20%
    mutation_rate = 0.05
    dx = 2.0 / n_points
    
    start_time = time.time()
    
    best_h = None
    best_c5 = float('inf')
    
    # Initialize population
    if 'initial_h_values' in globals():
        initial_h = initial_h_values
    else:
        initial_h = None
    
    population = initialize_population(n_points, pop_size, initial_h)
    
    # Main optimization loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness
        fitnesses = []
        for h in population:
            f = evaluate_h(h, dx)
            fitnesses.append(f)
            
        # Track best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
        
        # Sort population by fitness
        indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in indices]
        sorted_fitnesses = [fitnesses[i] for i in indices]
        
        # Elitism - keep top individuals
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1, mutation_rate)
            child1 = ensure_constraints(child1, n_points)
            
            child2 = mutate(child2, mutation_rate)
            child2 = ensure_constraints(child2, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
    
    if best_h is None:
        # Fallback: look for a good candidate
        fitnesses = [evaluate_h(h, dx) for h in population]
        min_fitness_idx = np.argmin(fitnesses)
        best_c5 = fitnesses[min_fitness_idx]
        best_h = population[min_fitness_idx].copy()
    
    # Final evaluation
    final_c5 = evaluate_h(best_h, dx)
    
    return (best_h, final_c5, n_points)
