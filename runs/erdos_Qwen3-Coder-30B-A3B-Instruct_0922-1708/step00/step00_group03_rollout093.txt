import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h.copy()
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    # Precompute indices for cyclic operations
    indices = np.arange(n_points)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    # Optimization parameters
    learning_rate = 0.1
    patience = 50
    patience_counter = 0
    last_c5 = best_c5
    
    start_time = time.time()
    
    # EM-style optimization
    for iteration in range(500):
        if time.time() - start_time > budget_s - 1:
            break
            
        # E-step: compute all overlaps
        overlaps = np.zeros(n_points)
        for k in range(n_points):
            # Compute overlap for lag k
            shifted_h = np.roll(h, k)
            overlap = np.sum(h * (1 - shifted_h)) * dx
            overlaps[k] = overlap
        
        # M-step: compute gradient and update
        # For each point, accumulate gradient w.r.t. different lags
        new_h = np.zeros_like(h)
        for i in range(n_points):
            grad = 0.0
            for k in range(n_points):
                # Gradient of overlap_k w.r.t h[i]
                # overlap_k = sum_j h[j] * (1 - h[(j+k) % n])
                # d/dh[i] overlap_k = (1 - h[(i+k) % n]) if i = j else 0
                grad += (1 - h[(i + k) % n_points]) * overlaps[k]
            new_h[i] = h[i] - learning_rate * grad
        
        # Clip and rescale
        new_h = np.clip(new_h, 0, 1)
        h_sum = np.sum(new_h)
        if h_sum > 0:
            new_h = new_h * (n_points / 2) / h_sum
            
        h = new_h
        
        # Check for improvement
        current_c5 = evaluate_erdos_solution(h, dx)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Adaptive learning rate
        if current_c5 < last_c5:
            pass  # Improve
        elif patience_counter > 5:
            learning_rate *= 0.5
            patience_counter = 0
            
        last_c5 = current_c5
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h_vals, dx):
    h = np.array(h_vals)
    # Compute the full correlation integral for all lags
    full_corr = correlate(h, 1 - h, mode='full')
    # max_k integral over full_corr
    max_k = np.max(full_corr) * dx
    return max_k
