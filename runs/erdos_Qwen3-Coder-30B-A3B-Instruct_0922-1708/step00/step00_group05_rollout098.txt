import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        # Create a simple initial guess
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
    
    # Normalize to satisfy the integral constraint
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Fourier representation
    m = 20  # Number of cosine modes
    a = np.random.rand(m) * 0.5  # Coefficients for cosine modes
    a[0] = 0.5  # Set the DC component to ensure integral constraint
    
    # Optimization variables and parameters
    best_h = h_init.copy()
    best_C = float('inf')
    start_time = time.time()
    
    # Precomputed indices for cosine basis
    x = np.linspace(0, 2, n_points, endpoint=False)
    cos_basis = np.array([[np.cos(np.pi * k * xi / 2) for xi in x] for k in range(m)])
    
    # Gradient descent loop
    learning_rate = 0.01
    patience = 0
    max_patience = 100
    
    def compute_h_from_a(a):
        h = np.dot(a, cos_basis)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum = n_points/2
        return h
    
    def compute_C(h):
        # Compute the overlap integral using correlation
        corr = np.correlate(h, 1 - h, mode='full')
        # Take only the valid range where the shift is meaningful
        mid = len(corr) // 2
        corr_half = corr[mid - n_points + 1 : mid + n_points]
        C_val = np.max(np.abs(corr_half)) * dx
        return C_val
    
    # Run optimization
    while time.time() - start_time < budget_s:
        # Update h from a
        h = compute_h_from_a(a)
        
        # Evaluate current solution
        C = compute_C(h)
        
        if C < best_C:
            best_C = C
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience >= max_patience:
                break
        
        # Compute gradient approximation
        eps = 1e-5
        grad = np.zeros_like(a)
        for i in range(m):
            a_plus = a.copy()
            a_plus[i] += eps
            h_plus = compute_h_from_a(a_plus)
            C_plus = compute_C(h_plus)
            
            a_minus = a.copy()
            a_minus[i] -= eps
            h_minus = compute_h_from_a(a_minus)
            C_minus = compute_C(h_minus)
            
            grad[i] = (C_plus - C_minus) / (2 * eps)
        
        # Update coefficients
        a -= learning_rate * grad
        
        # Keep coefficients in bounds [0, 1]
        a = np.clip(a, 0, 1)
        
        # Reduce learning rate if no significant improvement
        if patience > 0 and patience % 100 == 0:
            learning_rate *= 0.5
    
    return (best_h.tolist(), best_C, n_points)
