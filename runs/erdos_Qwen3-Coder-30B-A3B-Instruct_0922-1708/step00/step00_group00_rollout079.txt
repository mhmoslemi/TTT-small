def evaluate_overlap(h, dx):
    """Evaluate C5 = max_k int h(x)(1 - h(x+k)) dx"""
    # Compute all overlaps for shifts k = -n+1 to n-1
    n = len(h)
    overlap = np.zeros(2*n - 1)
    # Use scipy correlate to get all shifts efficiently
    overlap_full = correlate(h, 1 - h, mode='full')
    # The full correlation has indices from -n+1 to n-1
    # We want the actual overlap values corresponding to shifts
    # indices = np.arange(-n+1, n)
    # k_vals = np.arange(-n+1, n)
    # Actually, we take the full cross-correlation result directly
    # The overlaps are at indices n-1 to 2*n-2  (i.e., from shift -n+1 to n-1)
    # But let's be careful:
    # When using correlate(h, 1-h, mode='full'), indices reflect:
    # 0 ... n-1  -> actual shifts 0 to n-1
    # n ... 2n-2 -> actual shifts -n+1  to -1
    # So shift k starts after n-1, and the valid index for shift k is n-1 + k  
    # To get shift k: use indices n-1 + k, not k directly
    
    # Step function h, 1-h, so overlap_k = sum h[i] (1-h[i+k])
    # correlate computes sum h[i] (1-h[i+k]) 
    # Looking at scipy documentation: the result is over all valid indices
    # The center result corresponds to shift k=0, index n-1
    
    # Extract all the overlaps (we use correlation for that)
    # A small fix to get shift k for overlap max
    # Assume h array with indices [0,...,n-1], 1-h with indices [0,...,n-1]
    # overlap[i] corresponds to shift k = i - (n-1) 
    # i=0 -> k = -n+1
    # i=n-1 -> k = 0
    # i=2*n-2 -> k = n-1
    
    # So shift k can go between -n+1 and n-1.
    # We want to get the maximum of all overlap_k values over valid k
    # out shape for full mode = 2*n - 1
    # So overlap[i] corresponds to shift k = i - (n-1) 
    
    # take maximum of the overlaps
    return np.max(overlap_full) * dx

def project_h(h, n_points):
    """Ensure 0 <= h[i] <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    s = np.sum(h)
    if abs(s) > 1e-10:
        h = h * (n_points / 2) / s
    return h

def evaluate_loss(h, n_points, dx, tau=0.1):
    """Evaluate a smooth approximation to max overlap"""
    # We compute overlaps for all shifts k = -n+1, ..., n-1
    # This is the full correlation of h with (1-h)
    # overlap contains all values for shifts -n+1 to n-1
    # So len(overlap) = 2*n - 1
    overlap = correlate(h, 1 - h, mode='full') 
    # The valid range of k is from -n+1 to n-1 which maps to indices n-1-n+1 to n-1+n-1 = 0 to 2*n-2
    # So overlap[i] corresponds to shift k = i - (n-1)
    # We only care about the extremes
    # for our smooth max    
    # Apply smooth max approximation
    
    # Convert to log-sum-exp form
    # stabilize for numerical reasons
    # Accounts for k = -n+1 to n-1
    # since mode full returns 2*n-1 elements
    # index 0 -> k = -n+1; index n-1 -> k = 0; index 2*n-2 -> k = n-1
    
    # Get the shifted values (k = 0 to n-1, der is -n+1 to -1 implicitly)
    # Just compute the exponents directly
    # A correction to words below: 
    # Let's take careful note that overlap_full is of length 2*n-1 where:
    # shift k = i - (n-1), from i = 0 to 2*n-2
    # For any i = 0..2*n-2, k = i - (n-1) 
    # so we get overlap at k = -n+1 to n-1
    # This assigns for i = n-1+k, k = -n+1..n-1
    
    # Apply the smooth maximum with temperature tau
    overlap_scaled = overlap * dx / tau
    loss = (1.0 / tau) * np.log(np.sum(np.exp(overlap_scaled)))
    return loss

def backtracking_line_search(h, grad, loss, n_points, dx, tau=0.1, alpha=0.1, beta=0.5):
    """Find step size alpha via backtracking line search"""
    # We maximize loss by minimizing -loss
    # But our function is not quite directly defined as derivative w.r.t. exit point  
    # Instead we should find alpha that reduces L(h - alpha * grad) < L(h)
    # But we do not control gradient direction, it's descent.
    # To avoid oversteps, make it tight.
    
    # We try alpha values until it improves
    # Start with step size alpha
    
    # loss = evaluate_loss(h, n_points, dx, tau)
    # Line search  
    # Stop when L(h - alpha * grad) < L(h) 
    # Try alpha, then alpha*beta until L(h - alpha * grad) < L(h) or alpha small
    
    # Cap steps to prevent overflow  
    while alpha > 1e-10:
        h_new = h - alpha * grad  
        h_new = project_h(h_new, n_points)  
        new_loss = evaluate_loss(h_new, n_points, dx, tau)
        if new_loss < loss:
            return alpha
        alpha *= beta
    return alpha

    tau = 0.1

        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current loss and gradient
        loss = evaluate_loss(h, n_points, dx, tau)
        grad = compute_grad(h, n_points, dx, tau)
        
        # Clip gradient  
        grad = np.clip(grad, -10, 10)
        
        # Use fixed step size
        alpha = step_size 
        # Try backtracking if too aggressive
        # Use finite-step for better empirics
        h_new = h - alpha * grad
        h_new = project_h(h_new, n_points)
        new_loss = evaluate_loss(h_new, n_points, dx, tau) 
        
        if new_loss < loss:
            h = h_new.copy()
        else:
            # Try a smaller step, or keep on same step if no change is a lot
            # Just set it back and move on if no enough decrease in objective 
            pass
        
        # Evaluate C5 and update best
        c5 = evaluate_overlap(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i
        if i - last_improvement > patience:
            break
    
    # Final evaluation  
    res = evaluate_overlap(best_h, dx)
    # Note: return h, res, n_points, but currently structure has 3 parts only
