import numpy as np
from scipy import optimize
import time
import random

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations for all shifts
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum overlap is the maximum value in the convolution
    # We're only interested in the shifts that are valid (non-zero)
    # The middle elements correspond to valid shifts from -1 to 1
    # But we get a full convolution of length 2*n-1
    # We shift conv to center the valid middle part
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to satisfy constraints: clip to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    """Initialize h using the provided initial_h_values if available."""
    try:
        # Assume initial_h_values is accessible in this scope
        # If defined, return a copy
        if 'initial_h_values' in globals():
            return np.array(initial_h_values).copy()
    except:
        pass
    # Default initialization: use a smoother function
    x = np.linspace(0, 2, n_points)
    h = 0.5 * (1 + np.sin(np.pi * x))  # Range [0,1]
    h = project_h(h, n_points)
    return h

def mutate_individual(h, mutation_rate):
    """Mutate an individual by slightly changing values."""
    h_copy = h.copy()
    for i in range(len(h_copy)):
        if random.random() < mutation_rate:
            # Add small random change to the value
            delta = (random.random() - 0.5) * 0.2
            h_copy[i] += delta
    return project_h(h_copy, len(h_copy))

def crossover(parent1, parent2):
    """Perform crossover of two parents to generate a child."""
    n = len(parent1)
    if n != len(parent2):
        raise ValueError("Parents must be of same length")
    # Uniform crossover
    child = np.zeros(n)
    for i in range(n):
        if random.random() < 0.5:
            child[i] = parent1[i]
        else:
            child[i] = parent2[i]
    return project_h(child, n)

def genetic_algorithm_phase(h, n_points, time_budget_s):
    """Run genetic algorithm to find a good starting point."""
    start_time = time.time()
    pop_size = 50
    mutation_rate = 0.1
    generations = 0
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    population = [h.copy()]
    for _ in range(pop_size - 1):
        individual = mutate_individual(h, mutation_rate)
        population.append(individual)
    
    while time.time() - start_time < time_budget_s * 0.7:
        generations += 1
        if generations % 10 == 0:
            pass  # Progress update
        
        # Evaluate all individuals
        fitness = []
        for individual in population:
            c5 = evaluate_c5(individual)
            fitness.append(c5)
        
        # Sort by fitness (lower C5 is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_h = population[0].copy()
            best_c5 = fitness[0]
        
        # Elitism: keep top 20%
        elite_size = max(1, pop_size // 5)
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_size = 5
            selected_indices = random.sample(range(pop_size), tournament_size)
            selected_fitness = [fitness[i] for i in selected_indices]
            winner_idx = selected_indices[np.argmin(selected_fitness)]
            parent1 = population[winner_idx]
            parent2 = population[random.choice(range(pop_size))]
            child = crossover(parent1, parent2)
            child = mutate_individual(child, mutation_rate)
            new_population.append(child)
        
        population = new_population
    
    return best_h, best_c5

def gradient_descent_phase(h, n_points, time_budget_s):
    """Run gradient descent with projected updates."""
    start_time = time.time()
    best_h = h.copy()
    best_c5 = evaluate_c5(best_h)
    learning_rate = 0.01
    iter_no_improvement = 0
    max_no_improvement = 10000
    
    def compute_gradient(h, shift_index):
        """Compute approximate gradient w.r.t. h for one specific shift."""
        # Let's compute gradient for the mode="full" convolution at a specific shift
        # Given the full convolution of two arrays of same length
        # We compute the derivative w.r.t. h
        # Still this is not trivial due to overlap calculation.
        # Instead, as an approximation, we compute finite differences
        N = len(h)
        dx = 2.0 / N
        # We want to find dC5/dh[i] for the particular shift where C5 is maximal
        # For simplicity, here we compute numeric gradients for a fixed shift.
        # but we actually want the gradient wrt the maximum correlation shift.
        # This is still handling approximation
    
        # Not computing exact gradient, use finite differences as fallback
        # Get the current contribution to max overlap at shift
        shift = shift_index - (N - 1)  # shift relative to center
        # Correlation at specific shift
        # This is clumsy but simple: using finite difference approach
        eps = 1e-5
        grad = np.zeros(N)
        for i in range(N):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, N)
            c5_plus = evaluate_c5(h_plus)
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, N)
            c5_minus = evaluate_c5(h_minus)
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        return grad
    
    def find_worst_shift(h):
        """Find shift that gives maximum overlap."""
        n = len(h)
        dx = 2.0 / n
        conv = np.correlate(h, 1 - h, mode="full")
        # Take max in absolute value correctly (shift between -n+1 and n-1)
        max_idx = np.argmax(conv)
        shift = max_idx - (n - 1)
        return shift
    
    iteration = 0
    while time.time() - start_time < time_budget_s * 0.3:
        # Structure of gradient descent:
        # Find shift that causes max overlap
        shift = find_worst_shift(best_h)
        # Get a suitable gradient estimate, or simple one
        # Set standard step
        shift = find_worst_shift(best_h)  # Simple approach: compute gradient 1 at a typical shift
        g = compute_gradient(best_h, shift)  # We approximate gradient
        
        # Assume gradient points in the right direction for now
        # GD update:
        h_new = best_h - learning_rate * g
        h_new = project_h(h_new, n_points)
        new_c5 = evaluate_c5(h_new)
        
        if new_c5 < best_c5:
            best_h = h_new
            best_c5 = new_c5
            iter_no_improvement = 0
            # Optionally reduce learning rate
        else:
            iter_no_improvement += 1
            
        if iter_no_improvement > max_no_improvement:
            break
        iteration += 1
        
        # Periodically adjust learning rate
        if iteration % 1000 == 0:
            learning_rate *= 0.5
            if learning_rate < 1e-6:
                break
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    """Main optimization function."""
    np.random.seed(seed)
    random.seed(seed)
    
    # Use smaller number for faster runtime
    n_points = 45  # or 60 if needed later
    h = initialize_h(n_points)
    start_time = time.time()
    
    # Phase 1: Genetic Algorithm
    h_ga, c5_ga = genetic_algorithm_phase(h, n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, c5_gd = gradient_descent_phase(h_ga, n_points, budget_s)
    
    # Return best result
    final_h = h_gd if c5_gd < c5_ga else h_ga
    final_c5 = min(c5_gd, c5_ga)
    
    # Ensure final result is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_c5(final_h)
    
    return (final_h, final_c5, n_points)
