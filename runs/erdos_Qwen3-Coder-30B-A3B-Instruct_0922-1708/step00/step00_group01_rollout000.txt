import numpy as np
from scipy import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Create symmetric initial construction
        n_points = 47
        h = np.zeros(n_points)
        center = n_points // 2
        width = 15
        h[center - width//2:center + width//2] = 1.0
        h = h * (n_points / 2.0) / np.sum(h)

    # Constants
    dx = 2.0 / n_points
    max_iter = 5000
    lr = 0.1
    anneal_every = 1000
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Compute cross-correlation
        corr = correlate(h, 1 - h, mode="full")
        # We only care about the valid overlaps (non-zero padding)
        full_corr = corr[len(h)-1:]
        max_idx = np.argmax(full_corr)
        max_val = full_corr[max_idx]
        
        # Gradient computation (simplified: approximate gradient via finite difference)
        # This computes gradient w.r.t. h[i] that most contributes to max overlap
        # For lag k, the contribution to C5 comes from h[i] * (1 - h[i+k])
        # So ∂(h[i] * (1 - h[i+k])) / ∂h[i] = (1 - h[i+k])
        # But now we operate on the cohesive overlap function
        
        # Compute gradient vector for the loss (max overlap) w.r.t h
        # Focusing on the dominant lag
        k = max_idx
        grad = np.zeros(n_points)
        for j in range(n_points):
            if j + k < n_points:
                grad[j] = 1 - h[j+k]
        
        # Update h
        h_new = h - lr * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint
        h_new = h_new * (n_points/2) / np.sum(h_new)
        
        # Evaluate new solution
        new_c5 = evaluate_erdos_solution(h_new, n_points)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Anneal learning rate
        if (i + 1) % anneal_every == 0:
            lr *= 0.5
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h_vals, n_points):
    """Evaluate the C5 constant for given h values"""
    h = np.array(h_vals)
    dx = 2.0 / n_points
    # Cross-correlation of h with 1-h
    corr = np.correlate(h, 1 - h, mode="full")
    # Take the overlapping part (non-zero padding)
    full_corr = corr[len(h)-1:]
    return np.max(full_corr) * dx
