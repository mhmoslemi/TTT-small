import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a reasonable starting point
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    max_iter = 6000
    eta = 0.01
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    for it in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        conv = signal.correlate(h, 1 - h, mode='full')
        overlaps = conv * dx
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        # Early termination if no improvement for a while
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > 100:
                break
                
        # Compute gradient for the worst-case lag
        grad = np.zeros_like(h)
        k = k_max - (n_points - 1)  # map full convolution index back to lag
        
        # Gradient of the overlap integral with respect to h
        if k >= 0:
            # h[i] * (1 - h[i+k]) 
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k]) * dx
            for i in range(k, n_points):
                grad[i] -= h[i - k] * dx
        else:
            # h[i] * (1 - h[i-k]) 
            k = -k
            for i in range(n_points - k):
                grad[i] += (1 - h[i + k]) * dx
            for i in range(k, n_points):
                grad[i] -= h[i - k] * dx
                
        # Update h using gradient descent
        h -= eta * grad
        
        # Project h to [0,1]
        h = np.clip(h, 0, 1)
        
        # Adjust sum to maintain constraint
        current_sum = np.sum(h)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-10:
            # Distribute difference proportionally
            diff = target_sum - current_sum
            h += diff / n_points
            
        # Adaptive learning rate
        if it % 100 == 0 or (it % 10 == 0 and patience > 50):
            eta *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
