    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Compute all overlaps
        h_vec = h_vec / dx  # Normalize for proper convolution
        overlaps = correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(overlaps) * dx
        return max_overlap
    
    # Constraints
    def constraint_integral(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    cons = [{'type': 'eq', 'fun': constraint_integral}]
    
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimize using L-BFGS-B for faster convergence
    result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, constraints=cons,
                      options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9})
    
    h_opt = result.x
    h_opt = np.clip(h_opt, 0, 1)
    h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
    
    # Final evaluation
    h_vec = h_opt / dx

    c5_bound = np.max(overlaps) * dx
