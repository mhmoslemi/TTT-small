import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, n_points):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / n_points
    # Compute full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # The maximum value is the worst case overlap
    max_overlap = np.max(conv)
    # Normalize by dx to get actual C5
    c5 = max_overlap * dx
    return c5

def project_h(h, n_points):
    """Project h to feasible region [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    # Rescale to make sum equal to n_points / 2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h
