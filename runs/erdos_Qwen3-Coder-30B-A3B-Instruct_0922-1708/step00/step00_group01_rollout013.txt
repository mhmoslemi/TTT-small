import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals() and initial_h_values is not None:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create initial symmetric construction
        n_points = 47
        h = np.ones(n_points)
        # Create smooth symmetric shape
        x = np.linspace(0, 2, n_points)
        # Create a plateau with symmetric tapering
        h = 0.5 + 0.5 * np.exp(-((x - 1.0)**2) / 0.1)
        # Scale to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)

    dx = 2.0 / n_points

    def compute_c5(h_vals):
        # Ensure valid range
        h_vals = np.clip(h_vals, 0, 1)
        # Compute cross-correlation
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # Max over all lags
        max_corr = np.max(corr)
        return max_corr * dx

    def feasibility_check(h_vals):
        return np.all((h_vals >= 0) & (h_vals <= 1)) and abs(np.sum(h_vals) * dx - 1.0) < 1e-10

    def project_to_feasible(h_vals):
        h_vals = np.clip(h_vals, 0, 1)
        # Scale to correct sum
        current_sum = np.sum(h_vals) * dx
        if abs(current_sum - 1.0) > 1e-10:
            h_vals = h_vals * (1.0 / current_sum)
        return h_vals

    # Genetic algorithm setup
    pop_size = 100
    n_generations = 1000
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = 20
    simplex_size = 10

    # Initialization
    population = [h.copy()]
    for _ in range(pop_size - 1):
        noise = np.random.normal(0, 0.05, n_points)
        indiv = h + noise
        indiv = project_to_feasible(indiv)
        population.append(indiv)
    
    best_h = h.copy()
    best_c5 = compute_c5(best_h)
    
    for gen in range(n_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness_vals = []
        for indiv in population:
            c5 = compute_c5(indiv)
            fitness_vals.append(c5)
        
        # Tournament selection
        selected = []
        for _ in range(pop_size):
            tournament_indices = np.random.choice(len(population), tournament_size)
            tournament_fitness = [fitness_vals[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            selected.append(population[winner_idx].copy())
        
        # Create next generation
        next_gen = []
        num_crossover = int(pop_size * crossover_rate)
        for i in range(0, num_crossover, 2):
            if i + 1 < len(selected):
                parent1 = selected[i]
                parent2 = selected[i + 1]
                # One-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                # Mutation
                if np.random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    offspring1 = offspring1 + noise
                if np.random.random() < mutation_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    offspring2 = offspring2 + noise
                # Projection
                offspring1 = project_to_feasible(offspring1)
                offspring2 = project_to_feasible(offspring2)
                next_gen.extend([offspring1, offspring2])
        
        # Fill the rest with mutated versions of current population
        while len(next_gen) < pop_size:
            parent_idx = np.random.randint(0, len(selected))
            parent = selected[parent_idx]
            mutated = parent + np.random.normal(0, 0.05, n_points)
            mutated = project_to_feasible(mutated)
            next_gen.append(mutated)
        
        population = next_gen[:pop_size]
        
        # Local optimization for top members
        top_indices = np.argsort(fitness_vals)[:simplex_size]
        for i in top_indices:
            indiv = population[i]
            try:
                # Minimize C5 with respect to this individual using Nelder-Mead
                res = minimize(lambda x: compute_c5(x), indiv, method='Nelder-Mead', 
                              options={'maxiter': 100, 'disp': False})
                if res.success:
                    refined = res.x
                    refined = project_to_feasible(refined)
                    c5_refined = compute_c5(refined)
                    if c5_refined < fitness_vals[i]:
                        # Replace old with refined
                        population[i] = refined
                        fitness_vals[i] = c5_refined
                        if c5_refined < best_c5:
                            best_c5 = c5_refined
                            best_h = refined.copy()
            except:
                pass
        
        if gen % 10 == 0:
            current_best = min(fitness_vals)
            if current_best < best_c5:
                best_c5 = current_best
                best_idx = np.argmin(fitness_vals)
                best_h = population[best_idx].copy()

    # Final check and return
    final_c5 = compute_c5(best_h)
    return (list(best_h), final_c5, n_points)
