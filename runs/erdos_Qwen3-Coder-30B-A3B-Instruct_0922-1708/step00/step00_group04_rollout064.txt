import numpy as np
from scipy.signal import correlate
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial construction
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Simple piecewise constant construction with 10 segments
        n_points = 20
        h_init = np.ones(n_points)
        h_init[:n_points//2] = 0
        h_init = h_init * (n_points/2) / np.sum(h_init)
        
    # Time remaining for optimization
    time_left = budget_s - (time.time() - start_time)
    if time_left < 5:
        return (h_init, evaluate_erdos_solution(h_init), n_points)
    
    # Refine with continuous optimization
    dx = 2.0 / n_points
    def objective(h):
        h = np.clip(h, 0, 1)
        integral_constraint = np.sum(h) * dx - 1.0
        if abs(integral_constraint) > 1e-6:
            # Penalize constraint violation
            penalty = 1000 * abs(integral_constraint)
        else:
            penalty = 0
        
        corr = correlate(h, 1 - h, mode='full')
        # Maximize over all lags
        max_overlap = np.max(corr) * dx
        return max_overlap + penalty
    
    # Constraints
    bounds = [(0, 1) for _ in range(n_points)]
    constraints = [{'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}]
    
    # Use scipy minimize with SLSQP
    opt_result = minimize(
        objective,
        h_init,
        method='SLSQP',
        bounds=bounds,
        constraints=constraints,
        options={'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    if opt_result.success:
        h_opt = np.clip(opt_result.x, 0, 1)
    else:
        h_opt = h_init
    
    # Possibly improve with a local search-like approach
    best_h = h_opt.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Shuffle iterations to escape local minima
    for _ in range(min(100, int(50 * time_left / 600))):
        if time.time() - start_time > budget_s - 1:
            break
        
        # Small perturbation
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        # Adjust to maintain integral constraint
        h_pert = h_pert * (n_points/2) / np.sum(h_pert)
        
        c5 = evaluate_erdos_solution(h_pert)
        if c5 < best_c5:
            best_h = h_pert
            best_c5 = c5
    
    return (best_h, best_c5, n_points)

def evaluate_erdos_solution(h):
    # h is a step function on [0, 2] with n points
    # Compute overlap integral for all lags and find max
    n = len(h)
    dx = 2.0 / n
    # Compute correlations
    corr = correlate(h, 1 - h, mode="full")
    # Find maximum value over valid lags
    max_k = len(corr) // 2
    c5 = np.max(corr[1:max_k]) * dx  # Exclude zero lag
    return c5
