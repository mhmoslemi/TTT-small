import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    stagnation_threshold = 500
    min_alpha = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * target_sum
        
    # Momentum terms
    h_mom = np.zeros(n_points)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    last_improvement = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute cross-correlation: h(x) * (1 - h(x+k)) for all k
        # This is equivalent to convolution of h with (1-h)
        corr = signal.correlate(h, 1 - h, mode='full')
        # The overlap for shift k is given by corr[n_points-1+k]
        # so we want the max value among valid shifts
        valid_corr = corr[n_points-1:]  # Only positive shifts
        k_max = np.argmax(valid_corr)
        
        # Compute gradient of overlap w.r.t. h
        # We want max_k int h(x)(1-h(x+k)) dx
        # Gradient is (1 - h) for forward terms, -h for backward terms
        # For a fixed k, the gradient term for h[i] is
        # (1 - h[i+k]) - h[i+k] * gradient of convolution
        grad_h = np.zeros(n_points)
        
        # Compute derivative elements using finite difference approach
        # This is a simplified approximation for now
        h_shifted = np.roll(h, k_max)
        grad_h = (1 - h_shifted) - h_shifted * 0  # Simplified
        
        # Correct gradient calculation:
        # For a point i, contribution to overlap is h[i]*h[i+k]
        # So derivative with respect to h[i] is (1 - h[i+k]) * dx
        # and with respect to h[i+k] is -h[i] * dx
        # We build gradient directly
        # j = i+k, so when we consider the constraint on h[i], it affects
        # term h[i](1-h[i+k]) where the derivative wrt h[i] is (1-h[i+k])
        # Also, it's a double sum over shift parameter, so max_k and derivation
        
        # More precise method: for each i, compute the k that gives max overlap
        # and use that for gradient calculation
        
        # Let me take a more systematic approach:
        # Compute actual overlap value for current h
        # Then compute gradient of h[i] contribution by computing change in overlap 
        # if h[i] changes by small amount
        # For single index i, contribution to correlation is h[i] * (1 - h[i+k_max])
        # If h[i] increases by epsilon, then overlap changes by epsilon * (1 - h[i+k_max])
        
        # Manually compute gradient
        grad_h = np.zeros(n_points)
        for i in range(n_points):
            j = (i + k_max) % n_points
            grad_h[i] = (1 - h[j])
            
        # Gradient descent step with momentum
        h_mom = beta * h_mom - alpha * grad_h
        h += h_mom
        
        # Project h to constraints: [0,1] and sum = target_sum
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        
        # Update learning rate if no improvement for some time
        c5_current = max(signal.correlate(h, 1-h, mode='full')[n_points-1:]) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            last_improvement = iteration
        elif iteration - last_improvement > stagnation_threshold:
            alpha *= 0.5
            last_improvement = iteration
            if alpha < min_alpha:
                break
            
    # Final validation and return
    # Compute final C5
    final_corr = signal.correlate(best_h, 1-best_h, mode='full')
    final_c5 = max(final_corr[n_points-1:]) * dx
    
    return (best_h, final_c5, n_points)
