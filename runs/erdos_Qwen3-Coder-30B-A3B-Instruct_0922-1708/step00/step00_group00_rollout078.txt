import numpy as np
from scipy import signal
import time

def evaluate_erdos_solution(h_values):
    """Return the C5 bound for given h_values."""
    h = np.array(h_values)
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using correlation
    corr = signal.correlate(h, 1 - h, mode='full')
    # Only take the relevant parts (shifts from -n+1 to n-1)
    # The valid overlap for shift k is from index n-1-k to n-1+k
    # but we want the largest one
    overlaps = []
    for k in range(-n+1, n):
        start = max(0, n-1-k)
        end = min(2*n-2, n-1+k)
        if start <= end:
            overlap = corr[n-1+k]
            overlaps.append(overlap * dx)
    return max(overlaps) if overlaps else 0

def project_h(h, n_points):
    """Project h to [0,1] and rescale to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def smooth_max(overlaps, tau=0.1):
    """Compute smooth approximation of max with temperature tau."""
    return np.log(np.sum(np.exp(tau * np.array(overlaps)))) / tau

def compute_gradients(h, n_points, dx, tau=0.1):
    """Compute gradients of smooth max."""
    # Compute all overlaps
    corr = signal.correlate(h, 1 - h, mode="full")
    overlaps = []
    for k in range(-n_points+1, n_points):
        start = max(0, n_points-1-k)
        end = min(2*n_points-2, n_points-1+k)
        if start <= end:
            overlap = corr[n_points-1+k]
            overlaps.append(overlap)
    
    # Get max overlap
    max_idx = np.argmax(overlaps)
    max_overlap = overlaps[max_idx]
    
    # approximate gradient with respect to h
    # Compute d_overlap_k/dh_i for each h_i using finite differences
    # For simplicity, we compute the gradient of log sum of exponentials
    weights = np.exp(tau * np.array(overlaps)) / np.sum(np.exp(tau * np.array(overlaps)))
    # now compute gradient numerically
    grad = np.zeros_like(h)
    for i in range(len(h)):
        # approximate derivative
        eps = 1e-6
        h_plus = h.copy()
        h_plus[i] += eps
        corr_plus = signal.correlate(h_plus, 1 - h_plus, mode="full")
        overlaps_plus = []
        for k in range(-n_points+1, n_points):
            start = max(0, n_points-1-k)
            end = min(2*n_points-2, n_points-1+k)
            if start <= end:
                overlap = corr_plus[n_points-1+k]
                overlaps_plus.append(overlap)
        overlap_prime = overlaps_plus[max_idx]
        grad[i] = (overlap_prime - overlaps[max_idx]) / eps

    grad = grad * tau * weights[max_idx]  # final gradient
    return grad
