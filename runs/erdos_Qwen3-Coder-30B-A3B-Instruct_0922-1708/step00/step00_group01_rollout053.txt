import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from provided initial_h_values or create a symmetric one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        n = 47
        h = np.ones(n) * 0.5
        mid = n // 2
        h[mid-5:mid+6] = 0.8  # plateau in the middle
    
    n_points = len(h)
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize h to meet constraint
    h = h * target_sum / np.sum(h)
    
    # Parameters
    learning_rate = 0.1
    anneal_every = 1000
    max_iter = 5000
    
    # Track best solution
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        correlation = signal.correlate(h, 1 - h, mode='full')
        # Only the non-negative lags are relevant (overlaps)
        overlap_values = correlation[len(h)-1:]
        # Convert to integral form (multiply by dx)
        overlap_integral = overlap_values * dx
        max_overlap_idx = np.argmax(overlap_integral)
        max_overlap_value = overlap_integral[max_overlap_idx]
        
        # Compute gradient for max overlap
        gradient = np.zeros_like(h)
        # Only consider the maximum overlap contributing lag
        lag = max_overlap_idx
        for j in range(len(h)):
            if lag >= len(h) - j:
                k = lag - (len(h) - j)
                if k < len(h):
                    gradient[j] = 1 - 2 * h[k]
            else:
                k = lag + j
                if k < len(h):
                    gradient[j] = 1 - 2 * h[k]
        
        # Update h
        h_new = h - learning_rate * gradient
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to meet sum constraint
        h_new = h_new * target_sum / np.sum(h_new)
        
        h = h_new
        
        # Evaluate and update best solution
        c5 = compute_c5(h, dx)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Anneal learning rate
        if (i + 1) % anneal_every == 0:
            learning_rate *= 0.5
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    # Compute the maximum overlap integral
    correlation = signal.correlate(h, 1 - h, mode='full')
    overlap_values = correlation[len(h)-1:]
    overlap_integral = overlap_values * dx
    return np.max(overlap_integral)
