from scipy.signal import correlate

def evaluate_c5(h):
    """Evaluate the C5 constant for given step function h."""
    n = len(h)
    dx = 2.0 / n

    corr = correlate(h, 1 - h, mode="full")
    # Compute the maximum overlap (excluding the trivial case where the overlap is zero)
    max_corr = np.max(corr)
    return max_corr * dx
