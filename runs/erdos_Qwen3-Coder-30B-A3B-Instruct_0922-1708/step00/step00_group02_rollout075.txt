import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Compute the maximum overlap integral."""
    n = len(h)
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode="full")
    # Extract the valid correlations (excluding padding)
    valid_corr = corr[n-1:2*n-1]
    return np.max(valid_corr) * dx

def project_h(h, n_points):
    """Project h onto the constraint set: h in [0,1] and sum(h) = n_points/2."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def gradient_descent_step(h, dx, lr=0.01, max_iter=100):
    """Perform one step of gradient descent on h to reduce overlap."""
    n = len(h)
    # Evaluate current overlap
    current_overlap = evaluate_overlap(h, dx)
    
    # Compute numerical gradient of overlap w.r.t. h
    grad = np.zeros(n)
    eps = 1e-6
    for i in range(n):
        h_plus = h.copy()
        h_plus[i] += eps
        h_plus = project_h(h_plus, n)
        overlap_plus = evaluate_overlap(h_plus, dx)
        
        h_minus = h.copy()
        h_minus[i] -= eps
        h_minus = project_h(h_minus, n)
        overlap_minus = evaluate_overlap(h_minus, dx)
        
        grad[i] = (overlap_plus - overlap_minus) / (2 * eps)
    
    # Update h using gradient descent
    h_new = h - lr * grad
    h_new = project_h(h_new, n)
    
    return h_new
