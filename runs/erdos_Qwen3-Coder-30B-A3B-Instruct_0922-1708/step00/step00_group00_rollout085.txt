import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for a given step function h."""
    # Compute correlation with 1-h
    correlation = signal.correlate(h, 1 - h, mode="full")
    # Convert to overlap integral values
    overlaps = correlation * dx
    # Maximum overlap
    return np.max(overlaps)

def normalize_h(h, target_sum):
    """Normalize h to achieve target sum."""
    current_sum = np.sum(h)
    if abs(current_sum - target_sum) < 1e-10:
        return h
    # Rescale
    h = h * target_sum / current_sum
    return h

def ensure_constraints(h, n_points, target_sum):
    """Ensure h is in [0,1] and sums to target_sum."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize
    h = normalize_h(h, target_sum)
    return h

def create_individual(n_points, initial_h=None):
    """Create a valid individual (step function h)."""
    if initial_h is not None:
        h = initial_h.copy()
    else:
        # Start with uniform vector
        h = np.ones(n_points) * 0.5
        h = np.clip(h, 0, 1)
    # Normalize to correct sum
    h = normalize_h(h, n_points / 2.0)
    return h

def crossover(parent1, parent2):
    """Perform crossover between two individuals."""
    n = len(parent1)
    # Single-point crossover
    crossover_point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate an individual."""
    n = len(h)
    num_mutations = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, size=num_mutations, replace=False)
    # Apply small random changes
    for i in indices:
        delta = np.random.normal(0, 0.05)
        h[i] += delta
    return h

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), size=tournament_size)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx]

def genetic_algorithm(n_points, initial_h, budget_s=600, pop_size=100):
    """Run genetic algorithm to optimize h."""
    start_time = time.time()
    target_sum = n_points / 2.0
    dx = 2.0 / n_points
    
    # Initialize population
    population = []
    if initial_h is not None:
        # Seed with initial solution
        population.append(initial_h.copy())
        pop_size -= 1
    # Fill rest with random solutions
    for i in range(pop_size):
        h = create_individual(n_points)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        # Evaluate fitness
        fitnesses = []
        for h in population:
            h = ensure_constraints(h, n_points, target_sum)
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Track improvement
        if no_improvement_count < max_no_improvement // 2:
            no_improvement_count += 1
        else:
            no_improvement_count = 0
        
        # Selection and reproduction
        new_population = []
        
        # Elitism: keep best 20%
        elite_count = int(0.2 * pop_size)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            if np.random.rand() < 0.8:  # Crossover probability
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            # Ensure constraints
            child1 = ensure_constraints(child1, n_points, target_sum)
            child2 = ensure_constraints(child2, n_points, target_sum)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Check for improvement (only once per generation)
        if len(new_population) > 0:
            min_fitness = min(fitnesses)
            if min_fitness < best_c5:
                no_improvement_count = 0
            else:
                no_improvement_count += 1
    
    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45  # from the initial construction
    
    initial_h = None

    # Run optimization
    best_h, best_c5 = genetic_algorithm(n_points, initial_h, budget_s)
    
    # Final evaluation
    dx = 2.0 / n_points
    final_c5 = evaluate_h(best_h, dx)
    
    if final_c5 > best_c5:
        # Use the evaluated one which might be slightly better
        best_c5 = final_c5
    
    # Ensure constraints once more
    best_h = ensure_constraints(best_h, n_points, n_points / 2.0)
    return (best_h.tolist(), float(best_c5), n_points)
