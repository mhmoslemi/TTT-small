import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral C_5
    corr = correlate(h, 1 - h, mode="full")
    return np.max(corr) * dx

def project_h(h, n_points):
    # Ensure h is in [0,1] and sum(h) = n_points / 2
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Variables
    n_points = 100  # Medium size for faster testing
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w, c1, c2 = 0.7, 2.0, 2.0
    stagnation_limit = 50
    diversity_reset = 20
    best_c5 = float('inf')
    best_h = None
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial construction as base
        base_h = np.array(initial_h_values)
        base_h = project_h(base_h, n_points)
        particles.append(base_h.copy())
        velocities.append(np.zeros_like(base_h))
        
        # Add random variations
        for _ in range(swarm_size - 1):
            h = base_h + np.random.normal(0, 0.05, n_points)
            h = project_h(h, n_points)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Random initialization
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Personal best tracking
    pbest = [particles[i].copy() for i in range(swarm_size)]
    pbest_c5 = [float('inf')] * swarm_size
    
    # Initialize global best
    gbest_idx = 0
    gbest_c5 = float('inf')
    
    iteration = 0
    last_improvement = 0
    while iteration < max_iter and time.time() - start_time < budget_s:
        # Evaluate all particles
        current_c5s = []
        for i in range(swarm_size):
            c5 = evaluate_h(particles[i], dx)
            current_c5s.append(c5)
            
            # Update personal best
            if c5 < pbest_c5[i]:
                pbest_c5[i] = c5
                pbest[i] = particles[i].copy()
                
            # Update global best
            if c5 < gbest_c5:
                gbest_c5 = c5
                gbest_idx = i
                last_improvement = iteration
                if gbest_c5 < best_c5:
                    best_c5 = gbest_c5
                    best_h = particles[gbest_idx].copy()
        
        # Check stagnation
        if iteration - last_improvement > stagnation_limit:
            # Reset some particles to inject diversity
            reset_count = max(1, int(swarm_size * diversity_reset / 100))
            indices = np.random.choice(swarm_size, reset_count, replace=False)
            for idx in indices:
                particles[idx] = np.random.rand(n_points)
                particles[idx] = project_h(particles[idx], n_points)
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            last_improvement = iteration
            
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) +
                             c2 * r2 * (particles[gbest_idx] - particles[i]))
            particles[i] += velocities[i]
            
            # Clip positions
            particles[i] = project_h(particles[i], n_points)
        
        iteration += 1
        
        # Early termination if no improvement in budget
        if time.time() - start_time > budget_s - 0.1:
            break
    
    if best_h is None:
        best_h = particles[gbest_idx]
        
    return (best_h.tolist(), best_c5, n_points)
