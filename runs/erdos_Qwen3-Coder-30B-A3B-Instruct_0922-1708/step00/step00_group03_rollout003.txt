from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_C5(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    conv = correlate(h, 1 - h, mode="full")
    max_lag = np.max(conv)
    return max_lag * dx
