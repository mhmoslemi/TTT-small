import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    except NameError:
        # Fallback: uniform distribution
        n_points = 45
        h = np.ones(n_points) * 0.5
        
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Target C5 value
    target = 0.3807
    
    # Parameters
    max_iter = 10000
    alpha = 0.01
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h_vec):
        # Compute overlap for all shifts
        h_flip = 1 - h_vec
        # Use full mode to get all overlapping shifts
        overlaps = correlate(h_vec, h_flip, mode='full') * (2.0 / n_points)
        # Return the maximum overlap
        return np.max(overlaps)
    
    def compute_gradient(h_vec, k_max):
        # Gradient w.r.t. C5 for shift k_max
        n = len(h_vec)
        grad = np.zeros(n)
        # For shift k_max
        for i in range(n):
            j = i + k_max
            if 0 <= j < n:
                grad[i] += -(1 - h_vec[j])
            # Check for shifting from the right end
            j2 = i - k_max
            if 0 <= j2 < n:
                grad[i] += h_vec[j2]
        return grad
    
    # Iterative optimization
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1

        if c5 <= target:
            break
            
        # Find the k that maximizes the overlap
        h_flip = 1 - h
        overlaps = correlate(h, h_flip, mode='full') * (2.0 / n_points)
        k_max = np.argmax(overlaps)
        
        # Compute gradient
        grad = compute_gradient(h, k_max)
        
        # Update h
        h = h - alpha * grad
        
        # Project h into [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
        # Adaptive step size
        if patience_counter > patience:
            alpha *= 0.5
            patience_counter = 0
            
    # Final verification
    final_c5 = compute_c5(best_h)
    
    return (list(best_h), float(final_c5), int(n_points))
