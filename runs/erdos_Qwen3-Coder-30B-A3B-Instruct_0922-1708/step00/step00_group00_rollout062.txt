import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for given h."""
    # Compute overlap integral for all shifts
    correlation = correlate(h, 1 - h, mode="full")
    overlaps = correlation * dx
    # The maximum overlap is the maximum value over all valid shifts
    # We need to extract the relevant part because of the mode="full"
    n = len(h)
    max_overlap = np.max(overlaps[n-1:2*n-1])
    return max_overlap

def enforce_constraints(h, n_points):
    """Enforce that all values are in [0,1] and sum equals n_points/2."""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust sum to be exactly n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-6:
        # Simple adjustment: scale to target sum
        h = h * target_sum / current_sum
    return h

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize a population of step functions."""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
        pop_size -= 1

    for _ in range(pop_size):
        # Create random valid solution
        h = np.random.random(n_points)
        h = enforce_constraints(h, n_points)
        population.append(h)
    return population

def mutate(h, mutation_rate=0.05):
    """Mutate h with given probability."""
    h = h.copy()
    if mutation_rate > 0:
        n_points = len(h)
        for i in range(n_points):
            if np.random.random() < mutation_rate:
                h[i] += np.random.normal(0, 0.1)
        h = enforce_constraints(h, n_points)
    return h

def crossover(h1, h2):
    """Perform crossover on two step functions."""
    n_points = len(h1)
    if n_points != len(h2):
        raise ValueError("h1 and h2 must be of same length")
    # Random crossover point
    point = np.random.randint(1, n_points)
    h1_new = np.concatenate([h1[:point], h2[point:]])
    h2_new = np.concatenate([h2[:point], h1[point:]])
    return h1_new, h2_new

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    # Select tournament_size individuals at random
    indices = np.random.choice(len(population), size=tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()
