import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    max_generations = 120
    population_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    tournament_size = 5
    elitism_size = population_size // 5
    
    def evaluate(h):
        """Evaluate the C5 value for a given h"""
        if abs(np.sum(h) * dx - 1.0) > 1e-6:
            # Re-normalize if needed
            h = h * (1.0 / (np.sum(h) * dx)) * (n_points / 2)
        overlaps = correlate(h, 1 - h, mode='full') * dx
        return np.max(overlaps)
    
    def initialize_population():
        pop = []
        if 'initial_h_values' in globals():
            pop.append(np.array(initial_h_values))
        else:
            # Generate a random starting solution
            h = np.ones(n_points) * 0.5
            # Perturb a bit to make it different
            for i in range(n_points):
                h[i] = max(0, min(1, h[i] + np.random.normal(0, 0.05)))
            h = h * (1.0 / (np.sum(h) * dx)) * (n_points / 2)
            pop.append(h)
        
        # Fill the rest with random valid individuals
        for _ in range(population_size - len(pop)):
            h = np.random.rand(n_points)
            # Ensure sum constraint
            h = h * (1.0 / (np.sum(h) * dx)) * (n_points / 2)
            # Clip to [0,1]
            h = np.clip(h, 0, 1)
            pop.append(h)
        return pop
    
    def tournament_selection(population, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(len(population), tournament_size)
            best_idx = tournament[np.argmin([fitnesses[i] for i in tournament])]
            selected.append(population[best_idx].copy())
        return selected
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_prob:
            return parent1.copy(), parent2.copy()
        
        point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Renormalize
        child1 = child1 * (1.0 / (np.sum(child1) * dx)) * (n_points / 2)
        child2 = child2 * (1.0 / (np.sum(child2) * dx)) * (n_points / 2)
        
        # Clip to [0,1]
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        return child1, child2
    
    def mutate(individual):
        mutated = individual.copy()
        for i in range(n_points):
            if np.random.rand() < mutation_rate:
                # Add small random change
                mutated[i] += np.random.normal(0, 0.05)
        # Renormalize
        mutated = mutated * (1.0 / (np.sum(mutated) * dx)) * (n_points / 2)
        # Clip
        mutated = np.clip(mutated, 0, 1)
        return mutated
    
    population = initialize_population()
    best_h = None
    best_c5 = float('inf')
    
    # Evaluate initial population
    fitnesses = []
    for h in population:
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        fitnesses.append(c5)
    
    # Iterative evolution
    generation = 0
    no_improvement_count = 0
    while generation < max_generations:
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
        
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Elitism - preserve best solutions
        elite_indices = np.argsort(fitnesses)[:elitism_size]
        elite = [population[i].copy() for i in elite_indices]
        
        # Generate new population
        new_population = elite.copy()
        
        # Crossover and mutation
        while len(new_population) < population_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
                
        # Evaluate new population
        population = new_population
        fitnesses = [evaluate(h) for h in population]
        
        # Update best
        current_best_idx = np.argmin(fitnesses)
        if fitnesses[current_best_idx] < best_c5:
            best_c5 = fitnesses[current_best_idx]
            best_h = population[current_best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count >= 100:
            # Early stopping if not improving
            break
            
        generation += 1
    
    # Ensure final solution meets constraints
    if best_h is not None:
        sum_h = np.sum(best_h)
        if abs(sum_h * dx - 1.0) > 1e-6:
            # Re-normalize
            best_h = best_h * (1.0 / (sum_h * dx)) * (n_points / 2)
            best_h = np.clip(best_h, 0, 1)
            best_c5 = evaluate(best_h)
    
    return (best_h, best_c5, n_points)
