import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize initial h to sum to target
    h = h * target_sum / np.sum(h)
    
    # Initialize learning rate
    eta = 0.1
    best_c5 = float('inf')
    best_h = h.copy()
    
    # Precompute indices for fast correlation
    indices = np.arange(n_points)
    
    def compute_c5(h_vec):
        # Compute overlap integral for all lags
        # Using correlation trick with flipped h
        h_flipped = 1 - h_vec
        overlaps = np.correlate(h_vec, h_flipped, mode='full') * dx
        # Only take the valid overlapping part
        # The maximum is over the overlapping range
        max_overlap = np.max(overlaps)
        return max_overlap
    
    def compute_gradients(h_vec):
        # For each position i, compute gradient of each overlap with respect to h[i]
        # This is the derivative of sum over k of h[i] * (1 - h[(i + k) % n])
        # = 1 - h[(i - k) % n] for each k
        # So gradient for element i is sum of (1 - h[(i - k) % n]) over k
        N = len(h_vec)
        grad = np.zeros(N)
        for i in range(N):
            for k in range(N):
                j = (i - k) % N
                grad[i] += (1 - h_vec[j])
        return grad
    
    # EM-like iterations
    last_improvement = 0
    iteration = 0
    max_iter = 500
    
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        # E-step: compute current overlaps (this matches the target)
        h_current = h.copy()
        c5_current = compute_c5(h_current)
        
        # M-step: update h using a modified gradient approach
        # Instead of using gradients directly, we'll use a simple projection approach
        # based on minimizing the maximum overlap
        
        # We'll do some gradient descent in the direction that improves the maximum
        grad = np.zeros_like(h)
        # For approximation, we'll use numerical gradient
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            # Rescale to maintain sum
            h_plus = h_plus * target_sum / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
            
        # Update with small step in negative gradient direction
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        # Rescale to preserve constraint
        h_new = h_new * target_sum / np.sum(h_new)
        
        h = h_new
        c5_new = compute_c5(h)
        
        # Update best if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            last_improvement = iteration
            
        # Adaptive learning rate
        if iteration > 0 and iteration % 10 == 0:
            if c5_new >= c5_current and iteration - last_improvement > 10:
                eta *= 0.5
        iteration += 1
        
        # Early stopping based on stability
        if iteration - last_improvement > 50:
            break
            
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
