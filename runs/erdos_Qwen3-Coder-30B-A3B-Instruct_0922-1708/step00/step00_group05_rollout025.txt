import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic initial guess
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * (n_points / (2 * np.sum(h)))
    
    # Normalize h to satisfy constraint
    h = h * (n_points / (2 * np.sum(h)))
    
    dx = 2.0 / n_points
    
    def compute_C5(h_vec):
        # Compute C5 using correlation
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def project_to_feasible(h_vec):
        # Project to [0,1] and adjust to satisfy integral constraint
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        target_sum = n_points / 2
        if abs(current_sum - target_sum) > 1e-10:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 0.1:
        iteration += 1
        if no_improvement_count > 100:
            break
            
        improved = False
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Create perturbed version
            h_new = best_h.copy()
            
            # Select random points to perturb
            n_perturb = max(1, n_points // 10)
            indices = np.random.choice(n_points, size=n_perturb, replace=False)
            
            # Perturb selected points
            delta = 0.05
            for i in indices:
                change = np.random.uniform(-delta, delta)
                h_new[i] += change
            
            # Project to feasible region
            h_new = project_to_feasible(h_new)
            
            # Apply smoothing with a small window
            if n_points > 2:
                window_size = min(5, n_points // 4)
                if window_size % 2 == 0:
                    window_size += 1
                if window_size > 1:
                    h_new = np.convolve(h_new, np.ones(window_size)/window_size, mode='same')
            
            # Project again after smoothing
            h_new = project_to_feasible(h_new)
            
            # Evaluate
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                best_h = h_new.copy()
                best_C5 = new_C5
                improved = True
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        if not improved:
            no_improvement_count += 1
            
    # Final validation check
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
