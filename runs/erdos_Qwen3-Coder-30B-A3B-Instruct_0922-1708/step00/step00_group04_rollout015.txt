import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    if 'initial_h_values' in globals():

        n_points = len(h)
    else:

        # Create a triangular pattern
        x = np.linspace(0, 2, n_points)
        h = 1 - np.abs(2 * x / 2 - 1)
        # Scale to make sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, n_points, dx)
    last_improvement = 0
    iteration = 0
    
    # Gradient-based optimization with alternating projections
    def gradient(h, k):
        # h is 1D array of length n_points
        # Compute gradient of the overlap term O_k w.r.t. h_i
        # O_k = sum over i of h_i * (1 - h_{i+k})
        grad = np.zeros_like(h)
        if k >= 0:
            # h_i * (1 - h_{i+k}) for i in [0, n-k-1]
            # gradient wrt h_i: 1 - h_{i+k}
            # gradient wrt h_{i+k}: -h_i
            grad[:-k] += 1 - h[k:]
            grad[k:] -= h[:-k]
        else:
            # h_i * (1 - h_{i+k}) for i in [0, n+k-1]
            # gradient wrt h_i: 1 - h_{i+k}
            # gradient wrt h_{i+k}: -h_i
            k = -k
            grad[k:] += 1 - h[:-k]
            grad[:-k] -= h[k:]
        return grad
    
    def project_h(h):
        # Project h to [0,1]^n and ensure sum = n_points/2
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        return h
    
    # Start main loop
    while time.time() - start_time < budget_s - 1 and iteration < 1000:
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        overlaps = np.array([np.sum(h * np.roll(1 - h, k)) for k in range(1 - n_points, n_points)])
        k_star = np.argmax(overlaps) - (n_points - 1)
        
        # If already below threshold, stop
        if overlaps[k_star] * dx < 0.3807:
            break
            
        # Optimization step: minimize O_{k_star}
        # Build a list of indices affected by k_star
        k = k_star
        affected_indices = set()
        if k >= 0:
            affected_indices.update(range(n_points - k))
            affected_indices.update(range(k, n_points))
        else:
            k = -k
            affected_indices.update(range(n_points - k))
            affected_indices.update(range(k, n_points))
        affected_indices = sorted(list(affected_indices))
        
        # Define objective function for minimizing overlap for k_star

            h_new = h_vec.copy()
            return np.sum(h_new * np.roll(1 - h_new, k_star)) / n_points * dx
            
        # Simple linear gradient descent method to update h
        # Gradient can be computed analytically
        alpha = 1e-3
        h_new = h.copy()
        step = gradient(h_new, k_star) * (-alpha * dx)
        h_new += step
        h_new = project_h(h_new)
        
        # Ensure this change is valid
        try:
            h_new = project_h(h_new)
            test_c5 = evaluate_erdos_solution(h_new, n_points, dx)
            if test_c5 < best_c5:
                # Accept the update
                best_h = h_new.copy()
                best_c5 = test_c5
                h = h_new.copy()
                last_improvement = iteration
        except:
            pass
        
        iteration += 1
        
        if iteration - last_improvement > 100:
            # Reinitialize if no improvement in 100 iterations 
            h = np.clip(best_h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, n_points, dx)
    
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h, n_points, dx):
    """Compute C5 for the given h."""
    # Correlate h with (1-h) in full mode
    correlation = correlate(h, 1 - h, mode='full')
    # Take the maximum value
    max_overlap = np.max(correlation)
    return max_overlap * dx
