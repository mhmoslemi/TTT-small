import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h):
    # Evaluate C5 using the full correlation
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h) for all lags
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def perturb_h(h, delta=0.05):
    # Return a perturbed version of h
    h_new = h.copy()
    n = len(h)
    # Randomly select some indices to modify
    n_perturb = max(1, n // 20)  # Perturb about 5% of points
    indices = np.random.choice(n, size=n_perturb, replace=False)
    for i in indices:
        # Randomly adjust and clip to [0,1]
        h_new[i] += np.random.uniform(-delta, delta)
        h_new[i] = np.clip(h_new[i], 0, 1)
    # Adjust to satisfy sum constraint
    old_sum = np.sum(h)
    new_sum = np.sum(h_new)
    if abs(new_sum - old_sum) > 1e-10:
        # Adjust all values by a scalar factor
        scale = old_sum / new_sum
        h_new *= scale
        # Enforce sum = n/2 by linear adjustment
        diff = old_sum - n / 2
        if abs(diff) > 1e-10:
            offset = diff / n
            h_new -= offset
            h_new = np.clip(h_new, 0, 1)
    return h_new

def smooth_h(h, window=3):
    # Simple moving average smoothing
    n = len(h)
    if window <= 1:
        return h
    smoothed = np.zeros(n)
    for i in range(n):
        left = max(0, i - window//2)
        right = min(n, i + window//2 + 1)
        smoothed[i] = np.mean(h[left:right])
    return smoothed

    start_time = time.time()

        h = np.clip(h, 0, 1)
    
    dx = 2.0 / n_points
    # Ensure sum constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points/2) > 1e-10:
        scale = (n_points / 2) / h_sum
        h *= scale
    h = np.clip(h, 0, 1)
    
    best_h = h.copy()
    best_c5 = evaluate_h(h)
    
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 100:
        iter_count += 1
        # Generate trial h's
        trial_count = 100
        improve = False
        for _ in range(trial_count):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = perturb_h(h, delta=0.05)
            h_new = smooth_h(h_new, window=3)
            
            # Ensure sum = n_points/2
            h_sum_new = np.sum(h_new)
            if abs(h_sum_new - n_points/2) > 1e-10:
                scale = (n_points / 2) / h_sum_new
                h_new *= scale
                h_new = np.clip(h_new, 0, 1)
            
            c5_new = evaluate_h(h_new)
            
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
