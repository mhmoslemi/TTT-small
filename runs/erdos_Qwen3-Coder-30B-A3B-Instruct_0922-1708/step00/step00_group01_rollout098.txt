import numpy as np
from scipy.signal import correlate
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 47  # Using the initial construction size
    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * target_sum
    else:
        # Create random feasible h
        h = np.random.random(n_points)
        h = h / np.sum(h) * target_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Normalize to correct sum
    h = h / np.sum(h) * target_sum
    
    # Frank-Wolfe optimization parameters
    max_iter = 500
    learning_rate = 0.1
    tolerance = 1e-5
    
    best_C5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1.0:
            break
            
        # Compute all overlaps for different lags
        # Using full correlation to get all lags
        h_padded = np.concatenate([h, np.zeros(n_points - 1)])
        one_minus_h = 1 - h
        one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
        
        cross_corr = correlate(h_padded, one_minus_h_padded, mode='full')
        # Extract the valid overlaps (correlation with negative lags)
        overlap_values = cross_corr[n_points - 1:] * dx
        
        # Find maximum overlap and its index
        max_overlap = np.max(overlap_values)
        max_k = np.argmax(overlap_values)
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
            
        # Gradient computation (based on the position that maximizes overlap)
        grad = np.zeros(n_points)
        k_idx = max_k - (n_points - 1)
        if k_idx >= 0:
            # h[i] * (1 - h[i+k_idx])
            for i in range(n_points - k_idx):
                grad[i] += 1.0 - h[i + k_idx]
        else:
            # h[i+k_idx] * (1 - h[i]) where i + k_idx >= 0
            k_idx = -k_idx
            for i in range(k_idx, n_points):
                grad[i - k_idx] += 1.0 - h[i]
        
        # Solve linear subproblem: minimize gradient^T * d s.t. 0 <= d <= 1, sum(d) = target_sum
        # This is a simple projection: we greedily set d to be the smallest components
        d = np.zeros(n_points)
        remaining_sum = target_sum
        grad_sorted = np.argsort(grad)
        
        # Set smallest gradient components to 1, others to 0 to minimize dot product
        # but we need to respect the sum constraint
        d[:] = 0
        sum_d = 0
        for i in range(n_points - 1, -1, -1):
            idx = grad_sorted[i]
            if remaining_sum >= 1:
                d[idx] = 1
                remaining_sum -= 1
                sum_d += 1
            else:
                d[idx] = remaining_sum
                sum_d += remaining_sum
                remaining_sum = 0
                break
        if sum_d < target_sum and remaining_sum > 0:
            # Re-adjust
            for i in range(n_points):
                if d[i] < 1:
                    if remaining_sum > 0:
                        added = min(1 - d[i], remaining_sum)
                        d[i] += added
                        remaining_sum -= added
                        if remaining_sum <= 0:
                            break
    
        # Update h
        h = (1 - learning_rate) * h + learning_rate * d
        
        # Ensure bounds
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum
        h = h / np.sum(h) * target_sum
        
        # Check convergence
        if iteration > 0 and abs(max_overlap - best_C5) < tolerance:
            break
    
    # Final evaluation
    h_padded = np.concatenate([best_h, np.zeros(n_points - 1)])
    one_minus_h = 1 - best_h
    one_minus_h_padded = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
    cross_corr = correlate(h_padded, one_minus_h_padded, mode='full')
    overlap_values = cross_corr[n_points - 1:] * dx
    final_C5 = np.max(overlap_values)

    return (best_h.tolist(), float(final_C5), n_points)
