import numpy as np
import time
from scipy import optimize

def evaluate_overlap(h):
    """Evaluate the maximum overlap for given h"""
    n = len(h)
    dx = 2.0 / n
    # Compute all overlaps using full convolution
    conv = np.correlate(h, 1 - h, mode="full")
    # Extract the relevant part (shifts from -n+1 to n-1)
    max_overlap = np.max(conv)
    return max_overlap * dx

def project_h(h, n_points):
    """Project h to feasible region and rescale to meet constraint"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def compute_gradient(h, k_max, n_points):
    """Compute gradient for specific shift k_max"""
    grad = np.zeros(n_points)
    # For shift k_max, compute gradient
    # Each h[i] contributes to overlap with (1-h[i+k_max]) and h[i]
    for i in range(n_points):
        if i + k_max < n_points:
            grad[i] += (1 - h[i + k_max])
        if i - k_max >= 0:
            grad[i] -= h[i - k_max]
    return grad

        # Compute gradient for k_max
        grad = compute_gradient(h, k_max, n_points)

        h = h - alpha * grad
