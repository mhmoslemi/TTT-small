import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def compute_C5(h):
        # Compute correlation: h * (1 - h_{shifted})
        corr = np.correlate(h, 1 - h, mode="full")
        # Only keep the relevant lags (0 to 2*length - 2)
        # But only look at shifts in [0, 2]
        # So indices from n_points-1 to 3*n_points-2
        corr = corr[n_points-1:3*n_points-1]
        # Normalize by dx
        return np.max(corr) * dx
    
    def project(h):
        # Ensure h is in [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint sum(h) * dx = 1
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    # Gradient-based optimization
    max_iter = 1000
    lambda_reg = 1e-4
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute current C5 value
        c5_current = compute_C5(h)
        
        # Compute gradient w.r.t. h using finite differences
        eps = 1e-6
        grad_h = np.zeros_like(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project(h_plus)
            c5_plus = compute_C5(h_plus)
            
            h_minus = h.copy()
            h_minus[j] -= eps
            h_minus = project(h_minus)
            c5_minus = compute_C5(h_minus)
            
            grad_h[j] = (c5_plus - c5_minus) / (2 * eps)
        
        # Add regularization gradient
        grad_h += 2 * lambda_reg * h
        
        # Line search
        alpha = 1.0
        max_alpha = 10.0
        while alpha > 1e-8:
            h_new = h - alpha * grad_h
            h_new = project(h_new)
            c5_new = compute_C5(h_new)
            
            if c5_new < c5_current:
                h = h_new
                break
            else:
                alpha *= 0.5
                
        # If no improvement found, reduce step size
        if alpha <= 1e-8:
            h = project(h)
    
    # Local shift search to further refine
    best_h = h.copy()
    best_c5 = compute_C5(best_h)
    
    # Try some shifts
    for shift_attempts in range(20):
        if time.time() - start_time > budget_s - 0.1:
            break
        # Try small shifts in continuous domain
        shift = np.random.uniform(-0.02, 0.02, n_points)
        h_shifted = np.roll(h + shift, shift_attempts % n_points)
        h_shifted = project(h_shifted)
        c5_shifted = compute_C5(h_shifted)
        if c5_shifted < best_c5:
            best_h = h_shifted
            best_c5 = c5_shifted
            
    # Final validation
    final_c5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
