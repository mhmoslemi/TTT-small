import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h):
    n = len(h)
    dx = 2.0 / n
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode="full")
    # Max over all shifts (k from -n+1 to n-1)
    max_corr = np.max(corr)
    return max_corr * dx

def project_h(h, n_points):
    # Clip h to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to have integral = 1
    integral = np.sum(h) * (2.0 / n_points)
    if integral > 0:
        h = h * (n_points / 2) / integral
    return h

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Uniform distribution as fallback
        h = np.ones(n_points) * 0.5

    h = project_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h)
    
    start_time = time.time()
    iter_count = 0
    
    # Simple gradient descent approach
    max_iter = 10000
    lr = 0.01
    
    while time.time() - start_time < budget_s and iter_count < max_iter:
        # Compute gradient numerically
        grad = np.zeros(n_points)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, n_points)
            c5_minus = evaluate_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Update h
        h = h - lr * grad
        h = project_h(h, n_points)
        
        # Evaluate current solution
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        iter_count += 1
        
        # Adaptive learning rate
        if iter_count % 100 == 0:
            lr = max(1e-5, lr * 0.95)
    
    return (list(best_h), best_c5, n_points)
