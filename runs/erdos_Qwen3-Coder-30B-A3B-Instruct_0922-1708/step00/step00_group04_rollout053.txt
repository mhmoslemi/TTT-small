import numpy as np
import cvxpy as cp
from time import time

def run(seed=42, budget_s=600, **kwargs):

    # Main optimization loop
    iter_count = 0
    no_improvement_count = 0
    max_no_improvement = 100

        iter_count += 1
        
        # Formulate and solve convex problem
        h = cp.Variable(n_points)
        obj = cp.Minimize(cp.max(cp.convolution(h, 1 - h, 'full')))
        
        constraints = [
            h >= 0,
            h <= 1,
            cp.sum(h) == n_points / 2
        ]
        
        prob = cp.Problem(obj, constraints)
        
        try:
            # Solve with CVXPY using CBC solver (not necessarily available but we try)
            prob.solve(solver=cp.CBC, verbose=False, max_seconds=budget_s - (time() - start_time) - 1)
        except:
            # Fallback to simpler method if CVXPY fails
            pass
        
        # If CVXPY failed or gave no solution, use a simple gradient descent-like approach
        if 'h.value' not in locals() or h.value is None:
            h_vals = best_h.copy()
            # Simple gradient descent on a linearized proxy
            for _ in range(10):
                # Evaluate gradient using finite differences
                grad = np.zeros_like(h_vals)
                eps = 1e-4
                for i in range(n_points):
                    h_plus = h_vals.copy()
                    h_plus[i] += eps
                    h_minus = h_vals.copy()
                    h_minus[i] -= eps
                    c5_plus = evaluate_erdos_solution(h_plus)
                    c5_minus = evaluate_erdos_solution(h_minus)
                    grad[i] = (c5_plus - c5_minus) / (2*eps)
                
                # Update with gradient step
                step_size = 0.01
                h_vals -= step_size * grad
                
                # Project to [0,1] bounds
                h_vals = np.clip(h_vals, 0, 1)
                
                # Rescale to sum = n_points / 2
                h_vals *= (n_points / 2) / np.sum(h_vals)
                
                if time() - start_time > budget_s - 1:
                    break
                    
            h_vals = h_vals.copy()

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    h = np.array(h)
    # Reverse h for correlation
    h_rev = h[::-1]
    # Compute correlation: (h * (1-h)) convolved with h from right
    corr = np.correlate(h, 1-h, mode="full")
    # Taking maximum over all lags
    return np.max(corr) * dx
