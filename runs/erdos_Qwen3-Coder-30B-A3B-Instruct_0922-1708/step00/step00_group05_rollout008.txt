import numpy as np
from scipy.optimize import minimize
import time
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Default construction
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize h0 to satisfy integral constraint
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    # Discretization
    dx = 2.0 / n_points
    
    # Try different configurations to avoid local optima
    n_initializations = 5
    n_solves_per_init = 5
    
    # Get the valid range for h
    lb = np.zeros(n_points)
    ub = np.ones(n_points)
    
    for init in range(n_initializations):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Perturb initial guess
        h = h0.copy()
        if init > 0:
            # Random perturbation
            perturb = np.random.normal(0, 0.01, n_points)
            h += perturb
            # Keep within bounds
            h = np.clip(h, 0, 1)
            # Renormalize
            h = h * (n_points / 2) / np.sum(h)
        
        for iter_solve in range(n_solves_per_init):
            if time.time() - start_time > budget_s * 0.95:
                break
                
            # Save current
            h_prev = h.copy()
            
            # Try non-linear optimization with constraints
            def obj(h_vec):
                # Compute C5 with current h
                corr = np.correlate(h_vec, 1 - h_vec, mode='full') * dx
                return np.max(corr)
            
            # Constraints
            def constraint_sum(h_vec):
                return np.sum(h_vec) - n_points / 2
            
            con = {'type': 'eq', 'fun': constraint_sum}
            
            # Bounds
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimization
            try:
                res = minimize(obj, h, method='SLSQP', bounds=bounds, constraints=con, 
                               options={'ftol': 1e-6, 'eps': 1e-4})
                if res.success:
                    h_candidate = res.x
                    c5_candidate = obj(h_candidate)
                    if c5_candidate < best_c5:
                        best_h = h_candidate.copy()
                        best_c5 = c5_candidate
            except:
                pass
                
            # Revert to previous if not improved significantly
            if np.allclose(h, h_prev, atol=1e-5):
                break
    
    return (best_h.tolist(), best_c5, n_points)
