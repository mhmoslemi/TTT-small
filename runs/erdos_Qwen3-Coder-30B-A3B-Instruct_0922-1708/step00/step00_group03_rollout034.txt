import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Use initial construction or create one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
        h0 = h0 * (n_points / 2) / np.sum(h0)  # Normalize to sum = n_points/2
    else:
        n_points = 70
        h0 = np.full(n_points, 0.5 * (2.0 / n_points))  # Initialize to indicator-like step function
    
    # Set up optimization variables
    x = cvx.Variable(n_points)
    t = cvx.Variable()
    
    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == n_points / 2
    ]
    
    # Correlation variables
    Z = cvx.Variable((n_points, n_points))
    
    # Relaxation constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                Z[i,j] <= x[i],
                Z[i,j] <= 1 - x[j],
                Z[i,j] >= x[i] + x[j] - 1
            ]
    
    # Compute all correlation values
    correlations = []
    for k in range(-n_points+1, n_points):
        if time() - start_time > budget_s * 0.95:
            break
        # Build correlation
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        corr_sum = cvx.sum(Z[np.arange(n_points)[valid], idx[valid]])
        correlations.append(corr_sum)
    
    # Maximize objective over all correlations
    max_corr = cvx.max(cvx.hstack(correlations))
    constraints.append(max_corr <= t)
    
    # Objective
    objective = cvx.Minimize(t)
    
    # Problem and solve
    prob = cvx.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status in ["optimal", "optimal_inaccurate"]:
            sol = np.array(x.value).flatten()
        else:
            sol = h0.copy()
    except:
        sol = h0.copy()
    
    # Post-process
    sol = np.clip(sol, 0, 1)
    sol = sol * (n_points / 2) / np.sum(sol)  # Ensure sum = n_points/2
    
    # Return the result
    c5 = evaluate_erdos_solution(sol)
    
    return (sol, c5, n_points)
