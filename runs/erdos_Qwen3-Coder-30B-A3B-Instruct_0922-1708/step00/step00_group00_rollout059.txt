from scipy.optimize import minimize
import time

    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():

        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5

    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_sum
    
    # Helper function to compute C5
    def compute_C5(h):
        h_ext = np.concatenate([h, np.zeros(n_points)])
        conv = np.correlate(h_ext, 1 - h, mode="full")
        C5 = np.max(conv[:2*n_points-1]) * dx
        return C5
    
    # Optimization loop
    best_C5 = compute_C5(h)
    best_h = h.copy()
    
    # ADMM inspired update
    rho = 1.0
    max_iter = 500
    patience = 50
    no_improvement = 0
    
    for iteration in range(max_iter):
        current_time = time.time()
        if current_time - start_time > budget_s - 1:
            break
            
        # Adaptive step size and noise-based perturbation
        step_size = 0.01 / (1 + iteration / 100)
        
        # Update h
        h_new = h.copy()
        
        # Simple gradient descent step on max correlation
        h_ext = np.concatenate([h, np.zeros(n_points)])
        h_ext_new = np.concatenate([h_new, np.zeros(n_points)])
        
        # Compute gradient approximation
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Compute gradient for h[i] by finite difference
            delta = 1e-4
            h_plus = h.copy()
            h_plus[i] += delta
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * target_sum
            
            h_ext_plus = np.concatenate([h_plus, np.zeros(n_points)])
            C5_plus = np.max(np.correlate(h_ext_plus, 1 - h_plus, mode="full")[:2*n_points-1]) * dx
            
            grad[i] = (C5_plus - best_C5) / delta
        
        # Apply gradient
        h_new = h - step_size * grad
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_sum
        
        # Update h
        h = h_new
        
        # Evaluate new solution
        new_C5 = compute_C5(h)
        
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h.copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        if no_improvement > patience:
            # Restart with a noisy version of best solution
            h = best_h + np.random.normal(0, 0.001, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * target_sum
            no_improvement = 0

    # Ensure final solution is valid
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h / np.sum(best_h) * target_sum
    
    return (best_h.tolist(), best_C5, n_points)
