import numpy as np
from scipy.optimize import minimize
import time

def evaluate_c5(h, dx):
    # Compute cross-correlation of h with (1-h)
    corr = np.correlate(h, 1-h, mode="full")
    # Extract the relevant parts (shifts from -1 to 1)
    mid = len(corr) // 2
    relevant = corr[mid - len(h) + 1:mid + len(h) - 1]
    return np.max(relevant) * dx

def project_h(h, dx):
    # Clip to bounds
    h = np.clip(h, 0, 1)
    # Rescale to meet constraint
    current_sum = np.sum(h) * dx
    if current_sum != 0.5:
        h = h * 0.5 / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize from parent if available

        n_points = len(h)
    else:
        n_points = 42
        h = np.zeros(n_points)
        # Use smooth pattern (triangular wave)
        for i in range(n_points):
            x = i / (n_points - 1) * 2
            h[i] = max(0, 1 - abs(x - 1))
    
    dx = 2.0 / n_points
    h = project_h(h, dx)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Reduced gradient descent approach
    max_iter = 10000
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        iter_count += 1
        current_c5 = evaluate_c5(h, dx)
        
        # Compute approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, dx)
            c5_plus = evaluate_c5(h_plus, dx)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus, dx)
            c5_minus = evaluate_c5(h_minus, dx)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        # Gradient step
        lr = 1e-3
        h = h - lr * grad
        h = project_h(h, dx)
        
        # Update best if improved
        new_c5 = evaluate_c5(h, dx)
        if new_c5 < best_c5:

    return (list(best_h), best_c5, n_points)
