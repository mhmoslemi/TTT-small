import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Small enough for fast optimization
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    inertia_w = 0.7
    c1, c2 = 2.0, 2.0
    stagnation_limit = 50
    diversity_reset = 20
    
    # Initialize swarm
    swarm = []
    velocities = []
    
    # Use initial construction if available, otherwise generate random
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    # Generate initial population
    swarm = [h0.copy()]
    velocities = [np.random.uniform(-0.1, 0.1, n_points)]
    
    for _ in range(swarm_size - 1):
        # Perturb from initial or random
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        swarm.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best
    global_best_h = None
    global_best_c5 = np.inf
    stagnation_count = 0
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all particles
        current_c5s = []
        for h in swarm:
            # Compute C5
            corr = correlate(h, 1 - h, mode="full")
            c5 = np.max(corr) * dx
            current_c5s.append(c5)
        
        # Update personal bests and global best
        for i, (h, c5) in enumerate(zip(swarm, current_c5s)):
            if c5 < global_best_c5:
                global_best_c5 = c5
                global_best_h = h.copy()
                
        # Update velocities and positions
        for i in range(swarm_size):
            h = swarm[i]
            v = velocities[i]
            pbest_h = h  # For now, we don't track individual pbest separately
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            v_new = (inertia_w * v + 
                    c1 * r1 * (pbest_h - h) + 
                    c2 * r2 * (global_best_h - h))
            
            # Update position
            h_new = h + v_new
            
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            
            # Rescale to ensure sum = n_points / 2
            h_new = h_new / np.sum(h_new) * (n_points / 2)
            
            # Update particle
            swarm[i] = h_new
            velocities[i] = v_new
        
        # Check for stagnation and inject diversity if needed
        if iteration > 0 and iteration % stagnation_limit == 0:
            if stagnation_count > 0:
                # Inject diversity
                reset_count = int(swarm_size * diversity_reset / 100)
                indices = np.random.choice(swarm_size, reset_count, replace=False)
                for i in indices:
                    h = np.random.rand(n_points)
                    h = h / np.sum(h) * (n_points / 2)
                    swarm[i] = h
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
            else:
                stagnation_count += 1
    
    # Final evaluation
    if global_best_h is None:
        # Fallback to initial if nothing found
        h_eval = h0 if 'h0' in locals() else np.random.rand(n_points)
        h_eval = h_eval / np.sum(h_eval) * (n_points / 2)
        corr = correlate(h_eval, 1 - h_eval, mode="full")
        global_best_c5 = np.max(corr) * dx
        global_best_h = h_eval
    
    return (global_best_h.tolist(), global_best_c5, n_points)
