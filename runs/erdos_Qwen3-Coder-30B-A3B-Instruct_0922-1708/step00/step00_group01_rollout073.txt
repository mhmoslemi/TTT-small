    h = np.clip(h, 0, 1)

    # Frank-Wolfe algorithm parameters
    max_iter = 500
    eta = 0.1
    tol = 1e-6
    best_c5 = float('inf')
    best_h = h.copy()

    k_range = np.arange(-n_points + 1, n_points)

        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Evaluate overlaps for all lags
        corr_result = correlate(h, 1 - h, mode='full')
        c5_values = corr_result * (2.0 / n_points)
        max_k_idx = np.argmax(c5_values)
        k_max = k_range[max_k_idx]
        c5_bound = c5_values[max_k_idx]
        
        # Track best
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
        
        # Compute gradient for linear subproblem: compute dervative of overlap w.r.t. h
        # We want derivative of h[i] in sum h[i] * (1 - h[i + k]) for each i
        grad = np.zeros(n_points)
        if k_max >= 0:
            # k_max >= 0 means h[i] is multiplied by (1 - h[i + k_max])
            # so gradient for h[i] is -h[i + k_max] if i + k_max < n_points
            for i in range(n_points):
                if i + k_max < n_points:
                    grad[i] -= h[i + k_max]
        else:
            # k_max < 0 means h[i] is multiplied by (1 - h[i + k_max]) 
            # where i + k_max < 0, so this means h[i] is multiplied by (1 - h[i + k_max if k_max<0])
            k_abs = -k_max
            for i in range(n_points):
                if i - k_abs >= 0:
                    grad[i] -= h[i - k_abs]
        
        # Linear subproblem: minimize grad^T * d subject to d in feasible set
        # The feasible set is: sum d = n_points/2 and 0 <= d[i] <= 1
        # Solution: sort by -grad and assign values greedily to satisfy the sum constraint
        # Items with highest -grad (i.e. smallest grad) get assigned 1 first, then as much as needed
        
        # We'll solve this stepwise; assign to d starting with the lowest -grad
        # Create indices sorted by decreasing -grad (i.e., increasing grad)
        sorted_indices = np.argsort(-grad)
        d = np.zeros(n_points)
        remaining_sum = n_points / 2
        
        # Assign 1 to all positions that can take it
        for idx in sorted_indices:
            if remaining_sum <= 0:
                break
            if remaining_sum >= 1:
                d[idx] = 1
                remaining_sum -= 1
            else:
                d[idx] = remaining_sum
                remaining_sum = 0
        
        # Update h
        h = (1 - eta) * h + eta * d
        # Clip and rescale
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final evaluation of best solution
    corr_result = correlate(best_h, 1 - best_h, mode='full')
    c5_values = corr_result * (2.0 / n_points)
    final_c5 = np.max(c5_values)
    
    return (best_h.tolist(), final_c5, n_points)
