import numpy as np
import scipy.signal
import time
from math import floor

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70
    n_particles = 20
    max_iter = int(budget_s / 0.05)  # Estimate: 0.05s per iteration
    w = 0.7
    c1 = c2 = 2.0
    v_max = 0.1
    dx = 2.0 / n_points
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        h_init = np.ones(n_points) * 0.5
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Generate additional particles
    for _ in range(n_particles - 1):
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate particles
    def compute_C(h):
        corr = scipy.signal.correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    particle_C = [compute_C(p) for p in particles]
    global_best_idx = np.argmin(particle_C)
    global_best_h = particles[global_best_idx].copy()
    global_best_C = particle_C[global_best_idx]
    
    # Main loop
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Update each particle
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            cognitive = c1 * r1 * (particles[i] - particles[i])
            social = c2 * r2 * (global_best_h - particles[i])
            velocities[i] = w * velocities[i] + cognitive + social
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            new_position = particles[i] + velocities[i]
            new_position = np.clip(new_position, 0, 1)
            
            # Normalize to sum to n_points / 2
            current_sum = np.sum(new_position)
            if abs(current_sum - n_points / 2) > 1e-10:
                # Rescale to maintain sum constraint
                factor = (n_points / 2) / current_sum
                new_position = new_position * factor
            
            # Store updated particle
            particles[i] = new_position.copy()
            
            # Evaluate new particle
            new_C = compute_C(particles[i])
            particle_C[i] = new_C
            
            # Update personal best
            if new_C < particle_C[i]:
                particle_C[i] = new_C
            
            # Update global best
            if new_C < global_best_C:
                global_best_C = new_C
                global_best_h = particles[i].copy()
                
        # Tournament selection
        if iteration % 10 == 0:
            # Keep top 50% and replace worst with new random solutions
            sorted_indices = np.argsort(particle_C)
            top_half = sorted_indices[:n_particles // 2]
            remaining = sorted_indices[n_particles // 2:]
            
            for i in remaining:
                # Create new solution based on best 
                parent = particles[top_half[np.random.randint(0, len(top_half))]
                h = parent + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                particles[i] = h.copy()
                particle_C[i] = compute_C(particles[i])
    
    # Return best solution
    return (global_best_h.tolist(), global_best_C, n_points)
