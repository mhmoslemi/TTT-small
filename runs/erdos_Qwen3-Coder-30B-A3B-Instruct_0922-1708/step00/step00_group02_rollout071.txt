import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    tolerance = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Keep track of best solution
    best_h = h.copy()
    best_C5 = np.inf
    
    # Optimization variables
    v_h = np.zeros_like(h)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlap for all shifts
        h1 = np.concatenate([h, np.zeros(n_points-1)])
        h2 = np.concatenate([np.zeros(n_points-1), 1-h])
        corr = correlate(h1, h2, mode='valid')[:n_points]
        
        # Find maximum overlap
        max_idx = np.argmax(corr)
        max_overlap = corr[max_idx]
        
        if max_overlap < best_C5:
            best_C5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient
        # ∂/∂h = (1-h) - (h * d/dh correlation)
        grad_h = (1 - h) - (h * (1 - 2 * h[max_idx]) if max_idx < n_points else 0)
        
        # Update with momentum
        v_h = beta * v_h + (1 - beta) * grad_h
        h = h - alpha * v_h
        
        # Project h to valid range [0, 1]
        h = np.clip(h, 0, 1)
        
        # Project h to correct sum
        h = h * (n_points / 2) / np.sum(h)
        
        # Adjust learning rate
        if i % 500 == 0 and i > 0:
            alpha *= 0.5
    
    # Final evaluation
    final_h = best_h
    h1 = np.concatenate([final_h, np.zeros(n_points-1)])
    h2 = np.concatenate([np.zeros(n_points-1), 1-final_h])
    corr = correlate(h1, h2, mode='valid')[:n_points]
    final_C5 = np.max(corr) * dx
    
    return (final_h, final_C5, n_points)
