import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    patience = 500
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h) 
    
    # Initialize momentum
    v_h = np.zeros_like(h)
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation
        h_rev = np.flip(h)
        corr = signal.convolve(h, 1 - h_rev, mode='full')
        
        # Find maximum overlap
        k_max = np.argmax(corr)
        
        # Compute gradient w.r.t. h
        # We need to compute d/dh [h(x)(1-h(x+k))]
        # This is (1-h(x+k)) - h(x) * d/dh h(x+k) = (1-h(x+k)) - h(x) * 0 = (1-h(x+k))
        # But we also need to account for how k affects the computation
        # This is a complex gradient calculation; we instead use a simpler approach
        # Gradient of overlap integral with respect to h at position 0:
        # d/dh_i [int h(x)(1-h(x+k)) dx] = (1-h[i+k]) - h[i] * partial_d_k h[i+k]
        # The derivative of correlation w.r.t. h is complex, so we focus on improving
        # the maximum overlap by optimizing h
        
        # Compute naive gradient for current h
        grad_h = np.zeros_like(h)
        for j in range(n_points):
            # Let's compute simple finite differences
            if j + k_max < n_points:
                grad_h[j] = 1 - h[j + k_max] 
                if j < n_points - 1:
                    grad_h[j] -= h[j]
                    
        # Gradient for simple view using convolution
        # d/dh (h(x)(1-h(x+k))) = (1-h(x+k)) - h(x) * d/dh h(x+k)  
        # For this case, we ignore the derivative of h w.r.t. k    
        # The simplest approach for now: want to reduce correlation at k_max
        
        # So we consider: how to reduce the correlation at k_max
        # Looking at gradient of the integral:
        # If we have integral h(x)(1 - h(x+k)), the derivative w.r.t. h[i] is
        # (1-h[i+k]) + (1-h[i+k+1]) + ... in case of convolution-like form
        # However, simple idea: look at full correlation.
        # The gradient wrt h[i] is: sum of terms of type (1-h[i+k]) (i.e., elements contributing)
        # But easier to just try to improve k_max by optimizing
        # Let's take another approach using projected gradient: 
        # Gradient from the overlap formula 

        # Reconstruct a better gradient using forward differences 
        grad_h = np.zeros_like(h)
        # Use the structure of cross-correlation around k_max
        for j in range(n_points):
            r = j + k_max
            if r < n_points:
                grad_h[j] += 1 - h[r]
            if j >= k_max and j < n_points:
                r2 = j - k_max
                if r2 < n_points:
                    grad_h[j] -= h[r2]
        
        # Apply momentum
        v_h = beta * v_h + (1 - beta) * grad_h
        h = h - alpha * v_h
        
        # Project onto feasible set
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points/2) / h_sum
        
        # Compute C5 bound
        corr_test = signal.convolve(h, 1 - h, mode='full')
        c5_current = np.max(corr_test) * dx
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            alpha *= 0.5
            patience_counter = 0
            
    # Final evaluation
    if best_h is not None:
        corr_final = signal.convolve(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr_final) * dx
    else:
        best_h = h
        corr_final = signal.convolve(h, 1 - h, mode='full')
        final_c5 = np.max(corr_final) * dx
    
    return (best_h, final_c5, n_points)
