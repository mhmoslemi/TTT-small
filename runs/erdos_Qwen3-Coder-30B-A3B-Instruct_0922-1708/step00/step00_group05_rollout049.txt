def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize from initial_h_values if available, otherwise create random
    if 'initial_h_values' in globals():

    else:
        h = np.random.rand(n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Fourier coefficients (using only even modes for symmetry)
    m = 10  # number of cosine modes
    a = np.random.rand(m) * 0.5  # coefficients, scaled for safe range
    a[0] = 0.5  # integral constraint
    best_h = h.copy()
    best_C = evaluate_C(h, dx)
    n_iter = 0
    iter_since_improvement = 0
    lr = 0.01
    
    while time.time() - start_time < budget_s and iter_since_improvement < 100:
        n_iter += 1
        # Convert to h(x) assuming cosine basis expansion
        x = np.linspace(0, 2, n_points, endpoint=False)
        h_new = a[0] * np.ones_like(x)
        for k in range(1, m):
            h_new += a[k] * np.cos(np.pi * k * x / 2)
        
        # Ensure h in [0,1] and integral constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate current C
        C = evaluate_C(h_new, dx)
        if C < best_C:
            best_C = C
            best_h = h_new.copy()
            iter_since_improvement = 0
        else:
            iter_since_improvement += 1
            
        # Compute gradient via finite difference
        grad = np.zeros(m)
        eps = 1e-4
        for k in range(m):
            a_plus = a.copy()
            a_plus[k] += eps
            h_plus = a_plus[0] * np.ones_like(x)
            for j in range(1, m):
                h_plus += a_plus[j] * np.cos(np.pi * j * x / 2)
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            C_plus = evaluate_C(h_plus, dx)
            
            a_minus = a.copy()
            a_minus[k] -= eps
            h_minus = a_minus[0] * np.ones_like(x)

                h_minus += a_minus[j] * np.cos(np.pi * j * x / 2)
            h_minus = np.clip(h_minus, 0, 1)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            C_minus = evaluate_C(h_minus, dx)
            
            grad[k] = (C_plus - C_minus) / (2 * eps)
        
        # Update a
        a = a - lr * grad
        a = np.clip(a, 0, 1)
        
        # Adaptive learning rate
        if n_iter % 100 == 0:
            lr *= 0.5
            
    return (list(best_h), best_C, n_points)

def evaluate_C(h, dx):
    # Evaluate C₅ = max_k \int h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1-h, mode="full")
    # Correct for the actually valid range of k
    # The full correlation has length 2*n - 1
    # We only look at non-negative lags, i.e., k from 0 to n-1
    # But correlation indices go from -(n-1) to n-1
    # corr[n-1] is for k=0, corr[n] for k=1, etc.
    # So max index for k=0 to n-1 is at n-1 to 2*n-2
    # But k starts at 0 so take i = n-1 to 2*n-2
    # We want index to be at i = 0 to n-1 (which corresponds to k = 0 to n-1)
    # Since corr index 0 means k = n-1, index n-1 means k=0 etc.
    # So valid k = 0 to n-1 corresponds to corr index (2*n-2) down to n-1
    corr_filtered = corr[n-1:]  # This should be equivalent to k=0, ..., n-1
    # We only want the lag k from 0 to n-1, so corr_filtered for valid indices is:
    # corr[n-1], corr[n], ..., corr[2*n-2], which is of length n; 
    # corresponding to k = 0, 1, ..., n-1.
    return np.max(corr_filtered) * dx
