import numpy as np
import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap for given h"""
    # Compute correlation of h with (1-h)
    corr = signal.correlate(h, 1-h, mode='full')
    # Convert to overlap values
    overlaps = corr * dx
    # Return maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with constraint enforcement"""
    population = []
    # Add initial solution if provided
    if initial_h is not None:
        h = np.array(initial_h)
        h = h / np.sum(h) * (n_points / 2)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    # Fill remaining population with random valid individuals
    while len(population) < pop_size:
        # Start with a scaled uniform distribution
        h = np.ones(n_points) * (n_points / 2) / n_points
        # Add small random perturbations
        h += np.random.normal(0, 0.1, n_points)
        # Enforce constraints
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        # Check constraint satisfaction
        if abs(np.sum(h) - n_points / 2) < 1e-5:
            population.append(h.copy())
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection"""
    tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_idx].copy()

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parents"""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate an individual"""
    n = len(h)
    for i in range(n):
        if np.random.rand() < mutation_rate:
            # Add small random change
            h[i] += np.random.normal(0, 0.05)
            # Clip to [0, 1]
            h[i] = np.clip(h[i], 0, 1)
    # Renormalize to sum to n/2
    h = h / np.sum(h) * (n / 2)
    return h

            break

        fitnesses = [evaluate_h(h, dx) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_score:
            best_score = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()

        new_population = []

        # Trim to exact population size

    return best_h, best_score
