import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
    
    # Initialize parameters
    max_iter = 1000
    eta = 0.1
    tol = 1e-6
    start_time = time.time()
    
    # Precompute dx
    dx = 2.0 / n_points
    
    def compute_C5(h_vals):
        # Compute all correlations for all lags
        corr = correlate(h_vals, 1 - h_vals, mode='full')
        # The overlap integral is approximated by the correlation convolution scaled by dx
        # Extract only valid lags
        mid = len(corr) // 2
        valid_corr = corr[mid - (n_points - 1):mid + n_points]
        return np.max(valid_corr) * dx

    def solve_linear_subproblem(grad):
        # Solve min <d, grad> subject to 0 <= d[i] <= 1 and sum(d) = n_points / 2
        # We use a simple projection heuristic and maintain the mean constraint
        indices = np.argsort(grad)[::-1]
        d = np.zeros_like(grad)
        total = 0
        # Start with those that help reduce gradient the most
        for i in indices:
            if total < n_points / 2:
                d[i] = min(1, n_points / 2 - total)
                total += d[i]
            else:
                break
        # Ensure the sum constraint
        if total > n_points / 2:
            adjustment = (total - n_points / 2) / np.sum(d > 0)
            d = np.maximum(0, d - adjustment)
        d = d + (n_points / 2 - np.sum(d)) / n_points
        d = np.clip(d, 0, 1)
        return d
    
    current_C5 = compute_C5(h)
    best_h = h.copy()
    best_C5 = current_C5
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlap for all shifts
        corr = correlate(h, 1 - h, mode='full')
        mid = len(corr) // 2
        valid_corr = corr[mid - (n_points - 1):mid + n_points]
        
        kmax = np.argmax(valid_corr)
        max_k = kmax - (n_points - 1)
        
        # Compute gradient at current point
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] = 1.0 - h[j]
        
        # Linear subproblem
        d = solve_linear_subproblem(grad)
        
        # Update step
        h_new = (1 - eta) * h + eta * d
        
        # Ensure constraints: clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * (n_points / 2)
        
        # Evaluate new solution
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new
            current_C5 = new_C5
            if new_C5 < best_C5:
                best_C5 = new_C5
                best_h = h.copy()
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    return (list(best_h), final_C5, n_points)
