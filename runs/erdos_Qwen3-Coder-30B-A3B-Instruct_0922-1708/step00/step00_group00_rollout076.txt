import numpy as np
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Compute the maximum overlap integral C5."""
    n = len(h)
    # Compute all overlaps for shifts k in [-n+1, n-1]
    # overlap[k] = sum h[i] * (1 - h[i+k]) for valid i
    overlaps = np.zeros(2*n - 1)
    for k in range(-n + 1, n):
        if k >= 0:
            # h[i] * (1 - h[i+k]) for i in [0, n-k-1]
            valid_range = slice(0, n - k)
            overlap = np.sum(h[valid_range] * (1 - h[k + valid_range]))
        else:
            # h[i] * (1 - h[i+k]) for i in [-k, n-1]
            valid_range = slice(-k, n)
            overlap = np.sum(h[valid_range] * (1 - h[valid_range - k]))
        overlaps[k + n - 1] = overlap
    return np.max(overlaps) * dx

def compute_gradient(h, dx, tau=0.1):
    """Compute gradient of smooth max of overlaps."""
    n = len(h)
    # Compute all overlap integrals with indices shifted by n-1
    overlaps = np.zeros(2 * n - 1)
    for k in range(-n + 1, n):
        if k >= 0:
            valid_range = slice(0, n - k)
            overlap = np.sum(h[valid_range] * (1 - h[k + valid_range]))
        else:
            valid_range = slice(-k, n)
            overlap = np.sum(h[valid_range] * (1 - h[valid_range - k]))
        overlaps[k + n - 1] = overlap

    # Smooth max using temperature tau
    max_val = np.max(overlaps)
    exp_term = np.exp(tau * (overlaps - max_val))
    grad = np.zeros(n)
    total_exp = np.sum(exp_term)
    # Compute gradient: d/dh[i] of log(sum(exp(tau*overlap)))
    for i in range(n):
        d_dh = np.zeros(n)
        # partial derivative for each overlap contribution
        for k in range(-n + 1, n):
            if k >= 0:
                if i < n - k:
                    idx = i
                    if idx + k < n:
                        d_dh[idx] += (1 - h[idx + k]) * exp_term[k + n - 1]
                elif i + k < n:
                    idx = i + k
                    if idx < n:
                        d_dh[idx] -= h[idx] * exp_term[k + n - 1]
            else:
                if i < n:
                    idx = i
                    if idx - k < n:
                        d_dh[idx] += (1 - h[idx - k]) * exp_term[k + n - 1]
                elif i - k < n:
                    idx = i - k
                    if idx < n:
                        d_dh[idx] -= h[idx] * exp_term[k + n - 1]

        grad[i] = (1.0 / (tau * total_exp)) * np.sum(d_dh * exp_term)

    return grad * dx

def project_h(h, n_points):
    """Project h into [0,1] range."""
    h = np.clip(h, 0., 1.)
    return h

def rescale_h(h, n_points):
    """Ensure that sum(h)*dx = 1.0 => sum(h) = n_points / 2."""
    s = np.sum(h)
    target = n_points / 2.0
    if abs(s - target) > 1e-10:
        h = h * target / s
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial construction if available, otherwise generate
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Small construction (30-60 points) is preferred for speed
        n_points = 45
        # Initialize with uniform distribution
        h = np.ones(n_points) * 0.5
        h = rescale_h(h, n_points)

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)

    last_improvement = 0
    iteration = 0
    converged = False
    while time.time() - start_time < budget_s - 0.1 and not converged:
        # Compute current gradient with temperature 0.1
        try:
            g = compute_gradient(h, dx, tau=0.1)
        except Exception:
            break

        # Gradient clipping
        norm = np.linalg.norm(g)
        if norm > 10:
            g = g / norm * 10

        # Backtracking line search
        alpha = 0.1
        beta = 0.5
        step = 0
        while step < 20:  # prevent too many steps
            h_new = h - alpha * g
            h_new = project_h(h_new, n_points)
            h_new = rescale_h(h_new, n_points)
            c5_new = evaluate_c5(h_new, dx)
            if c5_new < best_c5:
                h = h_new
                best_c5 = c5_new
                last_improvement = iteration
                break
            else:
                alpha *= beta
                step += 1

        if iteration - last_improvement > 50:
            # Restart with new initialization
            h = np.random.rand(n_points) * 0.1 + 0.45
            h = rescale_h(h, n_points)
            last_improvement = iteration
        iteration += 1

        # Early exit if small improvement
        if iteration > 500 and abs(best_c5 - evaluate_c5(best_h, dx)) < 1e-7:
            converged = True

    return (best_h.tolist(), best_c5, n_points)
