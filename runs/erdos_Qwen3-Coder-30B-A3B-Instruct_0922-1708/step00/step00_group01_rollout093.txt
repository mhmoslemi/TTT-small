import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Reasonable size for fast evaluation
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_inject = 20  # percentage
    
    # Initialize particles
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        # Use initial solution as seed
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0.copy())
        velocities.append(np.zeros_like(h0))
        
        # Add random variations
        for _ in range(swarm_size - 1):
            h = h0 + np.random.normal(0, 0.1, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            particles.append(h.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        # Generate random feasible solutions
        for _ in range(swarm_size):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            h = np.clip(h, 0, 1)
            particles.append(h)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Personal and global bests
    personal_best = particles.copy()
    personal_best_scores = np.full(swarm_size, np.inf)
    global_best_idx = 0
    global_best_score = np.inf
    
    start_time = time.time()
    iteration = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        # Evaluate current particles
        for i in range(swarm_size):
            h = particles[i]
            corr = correlate(h, 1 - h, mode='full')
            max_overlap = np.max(corr) * dx
            if max_overlap < personal_best_scores[i]:
                personal_best_scores[i] = max_overlap
                personal_best[i] = h.copy()
            if max_overlap < global_best_score:
                global_best_score = max_overlap
                global_best_idx = i
                stagnation_count = 0  # Reset stagnation on improvement
                
        # Update velocities and positions
        for i in range(swarm_size):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] +
                            c1 * r1 * (personal_best[i] - particles[i]) +
                            c2 * r2 * (personal_best[global_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project onto feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
            
        # Inject diversity if stagnating
        if stagnation_count > stagnation_limit:
            # Replace a portion of swarm with new random particles
            n_replace = int(swarm_size * diversity_inject / 100)
            indices_to_replace = np.random.choice(swarm_size, n_replace, replace=False)
            for idx in indices_to_replace:
                h = np.random.rand(n_points)
                h = h / np.sum(h) * (n_points / 2)
                h = np.clip(h, 0, 1)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0  # Reset after injecting diversity
            
        stagnation_count += 1
        iteration += 1
    
    # Return best solution found
    best_particle = personal_best[global_best_idx]
    return (best_particle.tolist(), global_best_score, n_points)
