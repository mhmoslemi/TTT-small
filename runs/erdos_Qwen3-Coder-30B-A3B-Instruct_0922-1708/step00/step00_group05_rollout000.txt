import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        # Create a basic symmetric construction
        n_points = 70
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        # Normalize to integral = 1
        h *= 2.0 / np.sum(h)
    
    # Ensure h is within bounds and integral constraint
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2.0) / np.sum(h)
    
    dx = 2.0 / n_points
    best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
    best_h = h.copy()
    
    # Parameters
    perturb_rate = 0.1  # Fraction of points to perturb
    delta = 0.05
    smoothing_window = 3
    patience = 100
    patience_counter = 0
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1 and patience_counter < patience:
        improved = False
        for _ in range(100):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Make a copy and perturb
            h_trial = h.copy()
            n_perturb = max(1, int(n_points * perturb_rate))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            for i in indices:
                # Random walk with bounds
                new_val = h_trial[i] + np.random.uniform(-delta, delta)
                h_trial[i] = np.clip(new_val, 0, 1)
            
            # Maintain integral constraint
            current_sum = np.sum(h_trial)
            target_sum = n_points / 2.0
            adjustment = (target_sum - current_sum) / n_points
            h_trial += adjustment
            
            # Clamp to valid range
            h_trial = np.clip(h_trial, 0, 1)
            
            # Apply smoothing
            if smoothing_window > 1:
                h_smooth = np.zeros_like(h_trial)
                for i in range(n_points):
                    start = max(0, i - smoothing_window//2)
                    end = min(n_points, i + smoothing_window//2 + 1)
                    h_smooth[i] = np.mean(h_trial[start:end])
                h_trial = h_smooth
            
            # Check constraints
            if abs(np.sum(h_trial) - target_sum) > 1e-6:
                h_trial = h_trial * target_sum / np.sum(h_trial)
            
            # Evaluate
            c5_trial = np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx
            
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
                h = h_trial.copy()
                improved = True
                patience_counter = 0
            else:
                patience_counter += 1
                
            iteration += 1
            if time.time() - start_time > budget_s - 0.1:
                break
                
        if not improved:
            patience_counter += 1
    
    return (best_h.tolist(), best_c5, n_points)
