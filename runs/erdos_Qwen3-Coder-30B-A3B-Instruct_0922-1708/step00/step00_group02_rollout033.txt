import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()

    # Use initial_h_values if it exists, else create a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5

    dx = 2.0 / n_points
    target_sum = n_points / 2.0

    # Ensure initial h sums to target_sum
    h = h * target_sum / np.sum(h)
    
    # Optimization parameters
    max_iter = 10000
    alpha = 0.01
    stagnation_limit = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed >= budget_s - 0.1:
            break
            
        # Compute all overlaps
        # Create 1-h array for correlation
        one_minus_h = 1 - h
        # Use mode='full' to get all shifts
        overlap_full = correlate(h, one_minus_h, mode='full') * dx
        # Extract valid overlaps (from shift -n+1 to n-1)
        overlap = overlap_full[n_points-1:2*n_points-1]
        
        # Get maximum overlap and its shift
        max_overlap = np.max(overlap)
        k_max = np.argmax(overlap) - (n_points - 1)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
        
        # Compute gradient for the worst shift k_max
        grad = np.zeros(n_points)
        # For shift k_max:
        # grad[i] = d/dh[i] (sum_j h[j] (1 - h[j + k_max]))
        # = (1 - h[i + k_max]) - h[i] for valid indices
        
        if k_max >= 0:
            # h[i] contributes to (1 - h[i + k_max]) term
            # for i in range(n_points - k_max), 
            # gradient[i] = 1 - h[i + k_max] - h[i]
            for i in range(n_points - k_max):
                grad[i] += 1 - h[i + k_max]
                grad[i] -= h[i]
        else:
            # k_max is negative, so we shift backward
            # h[i + k_max] contributes to (1 - h[i]) term
            # for i in range(-k_max, n_points)
            for i in range(-k_max, n_points):
                grad[i] += 1 - h[i + k_max]
                grad[i] -= h[i]
        
        # Update h with gradient descent
        h = h - alpha * grad
        
        # Project h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale h to maintain sum constraint
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
    
    # Final check for best solution
    one_minus_h = 1 - best_h
    overlap_full = correlate(best_h, one_minus_h, mode='full') * dx
    overlap = overlap_full[n_points-1:2*n_points-1]
    final_c5 = np.max(overlap)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
