import numpy as np

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h values"""
    # Compute correlation of h with (1-h) using full mode
    corr = np.correlate(h, 1 - h, mode='full')
    # The overlap integral is the maximum of the correlation values
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, dx):
    """Project h onto constraints: [0,1] and integral = 1"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to satisfy integral constraint
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * 1.0
    return h
