    scales = np.full(basis_size, 2.0 / basis_size)
    
    def basis(x, j):
        return sigmoid((x - centers[j]) / scales[j])
    
    # Compute basis matrix
    x_vals = np.linspace(0, 2, n_points, endpoint=False)
    Phi = np.array([[basis(x, j) for j in range(basis_size)] for x in x_vals])

    if len(h0) == n_points:

        try:
            a_init = np.linalg.lstsq(Phi, h0, rcond=None)[0]
        except:
            a_init = np.random.rand(basis_size) * 0.1
    else:
        a_init = np.random.rand(basis_size) * 0.1
    
    # Elastic net parameters
    gamma = 0.8
    alpha = 0.01
    
    # Initialize optimizer state
    a_best = a_init.copy()
    c5_best = float('inf')
    
    # Optimization loop
    max_iter = 500
    no_improvement_count = 0
    best_iter = 0

        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute current h and overlaps
        h = Phi @ a_best
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h) * dx
        h = h * (1.0 / h_sum)  # Normalize to correct integral
        
        # Compute overlaps for all lags
        overlaps = np.correlate(h, 1-h, mode='full') * dx
        c5_current = np.max(overlaps)
        
        if c5_current < c5_best:
            c5_best = c5_current
            a_best_current = a_best.copy()
            best_iter = iteration
            no_improvement_count = 0

        # Compute gradient

        eps = 1e-4
        grad = np.zeros(basis_size)
    
        for j in range(basis_size):
            a_plus = a_best.copy()

            h_plus = Phi @ a_plus

            h_plus = h_plus * (1.0 / h_sum_plus)
            
            overlap_plus = np.correlate(h_plus, 1-h_plus, mode='full') * dx

            grad[j] = (c5_plus - c5_current) / eps
            
        # Update coefficients with elastic net
        a_new = a_best - alpha * grad
        # Elastic net penalty
        a_new -= gamma * np.sign(a_new)  # L1 part
        a_new -= (1 - gamma) * a_new      # L2 part
        
        # Projection to feasibility
        a_new = np.clip(a_new, -10, 10)  # Prevent overflow

    h_sum_final = np.sum(h_final) * dx
    h_final = h_final * (1.0 / h_sum_final)
    
    final_overlaps = np.correlate(h_final, 1 - h_final, mode='full') * dx
    c5_final = np.max(final_overlaps)
    
    return (h_final, c5_final, n_points)
