import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h = np.array(initial_h_values)
        n_points = len(h)
    except NameError:
        # Default initialization
        n_points = 45
        h = np.full(n_points, 0.5)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    
    # Temperature for smooth max
    tau = 0.1
    
    # Optimization parameters
    max_iter = 10000
    step_size = 0.1
    backtracking_factor = 0.5
    patience = 50
    best_c5 = float('inf')
    best_h = h.copy()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps efficiently with correlation
        # h(x) (1 - h(x+k)) for all k
        # We compute all correlations at once
        # This is optimized and cached in the overlap computation
        
        # Compute overlaps for all shifts
        # Expand h: pad with zeros at end to compute full correlations
        h_extended = np.pad(h, (0, n_points-1), mode='constant', constant_values=0)
        one_minus_h = 1 - h
        one_minus_h_extended = np.pad(one_minus_h, (0, n_points-1), mode='constant', constant_values=0)
        
        # Cross-correlation: overlap_k = sum(h[i] * (1 - h[i+k]))
        overlaps = correlate(h_extended, one_minus_h_extended, mode='full')
        # Keep only valid overlaps
        overlaps = overlaps[n_points-1:2*n_points-1]  # indices correspond to k in [-n+1, n-1]
        
        # Smooth max approximation
        exp_overlaps = np.exp(tau * overlaps)
        smooth_max = np.log(np.sum(exp_overlaps)) / tau
        loss = smooth_max
          
        # Gradient computation
        # grad = d/dh[sum_k exp(tau * overlap_k) / exp(tau * max_k overlap_k)]
        # We compute the full gradient faster by:
        # Grad[i] = sum_k (d/dh[i] (exp(tau * overlap_k)) / exp(tau * max_k overlap_k))
        # d/dh[i] overlap_k = contribution from i and i-k (if valid)
        # We compute gradient per position
        grad = np.zeros_like(h)
        for k in range(n_points):
            # This will be inefficient for large n_points but, we are constrained
            # to smaller n_points or optimize inside constrained budget
            # We compute d/dh[j] for j = i to influence of shift k
            # This is the part of overlap_k that depends on h[j]
            # overlap_k = sum_i h[i] * (1-h[i+k]) for valid indices
            # Gradient wrt h[j] is (1 - h[j+k]), assuming valid shift
            if k >= n_points or k < 0:
                continue
            for i in range(n_points):
                if i + k < n_points:
                    grad[i] += tau * exp_overlaps[k] * (1 - h[i + k]) / exp_overlaps[n_points-1]  # limit
            # Fix index mapping for full correlation equivalent
            # extracting the right element from the full correlator
        # Since computation of doverlap_k/dh_i involves elements
        # of h shifted, we can use gradient of overlap by chain rule.
        # For numeric stability, compute gradient term-wise using pre-computed
        # full correlation structure.
        # Simplified accurate gradient using add layer:
        # Gradient w.r.t to h[i] 
        # is sum_k [ exp(tau * overlap_k) / exp(tau max_k overlap_k) * tau * d/ dh[i] overlap_k ]
        # For d/dh[i] overlap_k = sum_j [ h[j] * d/dh[i] (1 - h[j+k])] 
        # For valid i, positive j+k
        # = (1 - h[i+k]) if j=i and j+k valid, else 0 
        # = 1 - h[i+k] if i+k < n_points and i >= 0 
        # This analytical approach becomes cumbersome unless using optimized batch operations.
        # We campaign on a 'mean-field-like' approximation or leveraged cross-correlation for batch gradient.

        # Let's compute the gradient by multiplying by the exponential weights
        # Use discrete gradients
        h_grad = np.zeros_like(h)
        for i in range(n_points):
            # This contribution comes from the shift k's where index i-k is valid
            # and i is valid
            for k in range(n_points):
                p = i + k
                if p < n_points:
                    h_grad[i] += (1 - h[p]) * (exp_overlaps[k] / np.sum(exp_overlaps))
        
        # Algorithmic gradient of smooth max with projection and constraints
        # Element-wise gradient computation brought up by correlation expects full knowledge
        # Thus, grammar below is accurate in principle, but final actual implementation
        # requires more exact canonical approach via dual.
        # So we will compute simpler bracketed update by gradient step.
        
        # A simpler way: compute overlap gradient in the constrained way
        # where gradient of loss is defined as [d loss / dh]
        # d_loss_i = (sum_ k exp(tau overlap_k) * tau * d overlap_k / dh_i ) / (sum_k exp(tau overlap_k))
        # for k = n-1 + i - p to 2*(n-1) where p = 0 to n-1 - (i+k)
        # corresponds to 2*(n-1) = n + (n-1)
        # So we infer per component what the gradient contribution is.
        # i is position in h. k is shift against h.
        # For fixed k, overlap_k = sum_i h[i] * (1 - h[i+k]) => d/dh[i] overlap_k = (1 - h[i+k])
        # But sum of such gradient product terms over k with exp weighting is the gradient.
        dx = 2.0 / n_points
        
        # Reserve results
        grad = np.zeros_like(h)
        # Loop over shifts and compute gradient term
        for k in range(2*n_points - 1):
            #s = k - (n_points - 1)
            # We now make actual gradient per h[i]
            # If shift k is such that i+k is valid, than h[i] appears in that overlap
            # So, grad[i] += tau * exp(overlap_k) * (1 - h[i+k]) / sum(exp(overlap_*) )
            exp_sum = np.sum(exp_overlaps)
            for i in range(n_points):
                s = k - (n_points - 1)
                if 0 <= i + s < n_points:
                    grad[i] += (tau * exp_overlaps[k] / exp_sum) * (1 - h[i + s])
        
        # Backtracking line search
        alpha = step_size
        while alpha > 1e-10:
            test_h = h - alpha * grad
            test_h = np.clip(test_h, 0, 1)
            test_sum = np.sum(test_h)
            if test_sum > 0:
                test_h = test_h * (n_points / 2) / test_sum  # rescale
            test_overlaps = correlate(np.pad(test_h, (0, n_points-1), mode='constant', constant_values=0),
                                      np.pad(1 - test_h, (0, n_points-1), mode='constant', constant_values=0),
                                      mode='full')[n_points-1:2*n_points-1]
            test_loss = np.log(np.sum(np.exp(tau * test_overlaps))) / tau
            if test_loss < loss:  # better than current
                break
            alpha *= backtracking_factor

        # Update h
        h = h - alpha * grad
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum > 0:
            h = h * (n_points / 2) / h_sum
        
        # Evaluate current C5
        overlaps_eval = correlate(np.pad(h, (0, n_points-1), mode='constant', constant_values=0),
                                  np.pad(1 - h, (0, n_points-1), mode='constant', constant_values=0),
                                  mode='full')[n_points-1:2*n_points-1]
        # Compute max row  
        c5_bound = np.max(overlaps_eval) * dx
        if c5_bound < best_c5:
            best_c5 = c5_bound
            best_h = h.copy()
    
    # Final validation
    overlaps_val = correlate(np.pad(best_h, (0, n_points-1), mode='constant', constant_values=0),
                            np.pad(1 - best_h, (0, n_points-1), mode='constant', constant_values=0),
                            mode='full')[n_points-1:2*n_points-1]
    final_c5 = np.max(overlaps_val) * (2.0 / n_points)
    final_h = best_h.copy()
    
    return (final_h.tolist(), final_c5, n_points)
