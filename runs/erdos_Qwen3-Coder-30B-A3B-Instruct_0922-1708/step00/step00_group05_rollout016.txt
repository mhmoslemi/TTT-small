import numpy as np
import time
from scipy import signal
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = np.correlate(h, 1 - h, mode="full")
    # Take maximum over all lags
    max_corr = np.max(corr)
    return max_corr * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Default parameters
    n_points = 70
    max_iter = 100000
    n_particles = 20
    inertia_weight = 0.7
    c1 = c2 = 2.0
    max_vel = 0.1
    perturb_scale = 0.01
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available, otherwise create random one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        h0 = np.random.rand(n_points)
        
    # Normalize
    h0 = h0 * (n_points / 2) / np.sum(h0)
    
    # Create particles around initial solution
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for _ in range(n_particles - 1):
        # Perturb slight
        h_pert = h0 + np.random.normal(0, 0.1, n_points)
        # Clip
        h_pert = np.clip(h_pert, 0, 1)
        # Normalize
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Evaluate particles
    particle_costs = [evaluate_erdos_solution(p) for p in particles]
    
    # Track best
    global_best_idx = np.argmin(particle_costs)
    global_best_h = particles[global_best_idx].copy()
    global_best_cost = particle_costs[global_best_idx]
    
    # Optimization loop
    iter_count = 0
    no_improvement_count = 0
    last_best_cost = global_best_cost
    
    while time.time() - start_time < budget_s - 0.1 and iter_count < max_iter:
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.random(2)
            v_new = (inertia_weight * velocities[i] + 
                     c1 * r1 * (particles[i] - particles[i]) +
                     c2 * r2 * (global_best_h - particles[i]))
            # Clip velocity
            v_new = np.clip(v_new, -max_vel, max_vel)
            velocities[i] = v_new
            
            # Update position
            new_h = particles[i] + v_new
            # Clip
            new_h = np.clip(new_h, 0, 1)
            # Normalize
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            
            # Evaluate
            new_cost = evaluate_erdos_solution(new_h)
            
            # Update personal best
            if new_cost < particle_costs[i]:
                particles[i] = new_h
                particle_costs[i] = new_cost
            
            # Update global best
            if new_cost < global_best_cost:
                global_best_h = new_h.copy()
                global_best_cost = new_cost
                no_improvement_count = 0
                last_best_cost = global_best_cost
            else:
                no_improvement_count += 1
                
        iter_count += 1
        
        # Diversity maintenance
        if no_improvement_count > 100:
            no_improvement_count = 0
            for i in range(n_particles):
                # Add some noise to avoid getting stuck
                particles[i] = np.clip(particles[i] + np.random.normal(0, perturb_scale, n_points), 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                
            # Re-evaluate all
            particle_costs = [evaluate_erdos_solution(p) for p in particles]
            global_best_idx = np.argmin(particle_costs)
            if particle_costs[global_best_idx] < global_best_cost:
                global_best_h = particles[global_best_idx].copy()
                global_best_cost = particle_costs[global_best_idx]
    
    # Final evaluation
    final_cost = evaluate_erdos_solution(global_best_h)
    
    return (global_best_h.tolist(), final_cost, n_points)
