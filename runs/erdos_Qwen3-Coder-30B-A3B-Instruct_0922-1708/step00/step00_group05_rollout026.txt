import numpy as np
import time
from scipy import signal

def evaluate_erdos_solution(h, dx=2.0/len(h)):
    """Evalute the overlap integral for the given step function h."""
    # h is a numpy array of values in [0,1], of length n
    # The integral is computed as max_k correlation(h, 1-h, mode="full")
    h = np.asarray(h)
    inverted_h = 1 - h
    correlation = signal.correlate(h, inverted_h, mode="full")
    # Each correlation value is multiplied by dx to get true integral
    max_overlap = np.max(correlation) * dx
    return max_overlap

        h[35:70] = 0.0
    dx = 2.0 / n_points

    # Ensure the integral constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2.0) / h_sum
    
    # Evaluate initial solution
    current_c5 = evaluate_erdos_solution(h, dx)
    best_h = h.copy()
    best_c5 = current_c5

    # Search parameters
    max_iter = 10000
    perturb_delta = 0.05
    num_trials = 100
    smoothing_window = 3
    no_improve_limit = 100
    no_improve_count = 0

    # Local search loop
    for i_iter in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Perturb the solution multiple times
        for _ in range(num_trials):
            h_trial = h.copy()
            
            # Select random points to perturb
            n_perturb = max(1, int(0.1 * n_points))
            indices = np.random.choice(n_points, n_perturb, replace=False)
            
            # Apply small changes
            for idx in indices:
                delta = np.random.uniform(-perturb_delta, perturb_delta)
                h_trial[idx] += delta
                
            # Clamp to [0, 1]
            h_trial = np.clip(h_trial, 0, 1)
            
            # Rebalance the sum
            h_sum = np.sum(h_trial)
            h_trial = h_trial * (n_points / 2.0) / h_sum
            
            # Smooth to maintain shape
            if smoothing_window > 1:
                kernel = np.ones(smoothing_window) / smoothing_window
                h_trial = np.convolve(h_trial, kernel, mode='same')
                h_trial = np.clip(h_trial, 0, 1)
                
            # Evaluate
            new_c5 = evaluate_erdos_solution(h_trial, dx)
            
            # Accept better or equivalent solutions
            if new_c5 <= current_c5:
                h = h_trial.copy()
                current_c5 = new_c5
                if new_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = new_c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
            else:
                no_improve_count += 1
                
        # Early stopping
        if no_improve_count >= no_improve_limit:
            break

    # Final evaluation of best solution
    final_c5 = evaluate_erdos_solution(best_h, dx)
    
    return (best_h.tolist(), final_c5, len(best_h))
