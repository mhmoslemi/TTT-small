    np.random.seed(seed)
    start_time = time.time()
    
    # Prepare problem parameters
    n_points = 42  # As per initial construction
    dx = 2.0 / n_points

    h_start = np.clip(h_start, 0, 1)
    integral = np.sum(h_start) * dx
    if not np.isclose(integral, 1.0):
        h_start = h_start * (1.0 / integral)
    
    # MIP approach for optimization
    # Binary variables for transition points: indicating if a new segment starts
    # Continuous variable for intermediate level
    # Use a simplified piecewise approach to examine near-optimal structures
    
    best_h = h_start.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Helper function to evaluate overlap
    def compute_c5(h_vec):
        # compute C5 = max_k \int h(x)(1 - h(x+k)) dx
        # using scipy as in original problem
        h_vec = np.array(h_vec)
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        # need to compute max of this over full range
        max_val = np.max(corr) * dx
        return max_val
    
    # Discrete search for low complexity configurations
    # Sample 1000 random piecewise constant functions (few segments)
    np.random.seed(seed)
    max_attempts = 1000
    for _ in range(max_attempts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Create piecewise constant with random number of segments
        n_seg = np.random.randint(2, 10)  # 2 to 9 segments
        seg_points = np.sort(np.random.rand(n_seg - 1)) * 2
        seg_points = np.clip(seg_points, 0, 2)

def evaluate_erdos_solution(h_values):
    # Exact method of computing C5:
    h = np.array(h_values)
    dx = 2.0 / len(h)
    # Compute C5 = max_k \int h(x)(1 - h(x+k)) dx
    # Use scipy's full cross correlation and find the maximum
    corr = correlate(h, 1 - h, mode='full')
    # this gives a correlation of size 2*n - 1
    # corresponding to lags - (n-1) to n-1
    # We want maximum over positive lags, but 
    # the full cross-correlation contains both negative and positive lags
    # Convert to lags properly
    # max in middle is 0 lag, we want to ignore that (h*1-h) case
    # But we also want to look at all positive lags (as the problem says C₅ = max_k)
    # The correlations are from index 0 to 2*n - 2
    # Index n-1 corresponds to lag 0, index 0 to - (n-1) ... n-2 to -1, n to n-1...
    # We want all non-zero lags i.e., i != n-1
    # Positive lags: from index n to 2n - 2 (i.e., lag = 1 .. (n-1))
    # Negative lags: from index 0 to n-2 (i.e., lag = -(n-1) .. -1)
    # But to compute max_k integral ... for k=1,2,...,n-1 and also negative k
    # essentially all is fine
    
    # Let's more carefully implement:
    max_c5 = -1.0
    n = len(h)
    
    # Since it's a bit tricky with negative indices in Python correlation
    # and we are dealing with a periodic function, 
    # and h is over [0, 2], so domain from 0 to 2
    # Lag k ranges from 0 to n-1 (negative half and positive half)
    # But for indices with 0:2n-2, we extract all but the middle (lag=0, index=n-1)
    # Each index j in 0 to 2n-2 refers to a lag of (j - (n-1)) 
    # We need absolute maximum, so we consider both positive and negative lags
    # valid regimens:
    # All lags (index 0 to 2n−1) with index j is lag of (j - (n−1))
    # We exclude j = n−1 (lag=0) for (h(x) * (1 - h(x))) since they are not overlaps
    
    # Extract lags corresponding to k=1..n-1 (positive lags) + k=-n+1..-1 (negative lags)
    valid_indices = np.arange(2*n-1) 
    valid_indices = np.delete(valid_indices, n-1)  # remove lag=0
    
    corr_vals = corr[valid_indices]
    max_val = np.max(np.abs(corr_vals)) * dx
    return max_val
