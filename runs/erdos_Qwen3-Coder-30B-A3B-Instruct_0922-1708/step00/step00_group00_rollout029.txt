import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    max_iter = 10000
    tau = 0.1  # Temperature for smooth max
    alpha_init = 0.1
    beta = 0.5  # Backtracking factor
    patience = 50  # Early stopping
    best_c5 = float('inf')
    best_h = None

    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5 * dx

    def evaluate_c5(h_vec):
        # Compute overlaps for all shifts
        # h_vec is the current function values
        h_full = np.concatenate([h_vec, np.zeros(n_points)])
        overlaps = []
        for k in range(-n_points+1, n_points):
            if k >= 0:
                cross = h_vec * (1 - h_full[k:k+n_points])
            else:
                cross = h_full[-k:][:n_points] * (1 - h_vec)
            overlaps.append(np.sum(cross) * dx)
        
        return max(overlaps)

    def compute_loss_and_grad(h_vec, tau=0.1):
        h_full = np.concatenate([h_vec, np.zeros(n_points)])
        overlaps = []
        d_overlaps_dh = []
        n = len(h_vec)
        
        for k in range(-n+1, n):
            if k >= 0:
                cross = h_vec * (1 - h_full[k:k+n])
            else:
                cross = h_full[-k:][:n] * (1 - h_vec)
            overlaps.append(np.sum(cross) * dx)
            
            # Compute gradient of overlap w.r.t h_vec
            grad = np.zeros(n)
            if k >= 0:
                grad = 1 - h_full[k:k+n]  # h[i+k] is in slice
                grad = grad * dx
            else:
                grad = 1 - h_vec
                grad = grad * dx
            d_overlaps_dh.append(grad)
            
        # Smooth max approximation
        exps = np.exp(tau * np.array(overlaps))
        exp_sum = np.sum(exps)
        loss = np.log(exp_sum) / tau
        
        # Gradient of loss
        weights = exps / exp_sum
        grad = np.zeros(n)
        for i in range(len(weights)):
            grad += weights[i] * d_overlaps_dh[i]
        grad *= 1.0 / tau
        
        return loss, grad

    def project_and_rescale(h_vec):
        # Project to [0,1]
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to correct sum
        actual_sum = np.sum(h_vec)
        if actual_sum > 0:
            h_vec = h_vec * target_sum / actual_sum
        return h_vec

    # Main optimization loop
    last_improvement = 0
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute loss and gradient
        try:
            loss, grad = compute_loss_and_grad(h, tau)
        except Exception:
            # Fallback to direct evaluation in case of numerical issues
            c5 = evaluate_c5(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            break
            
        # Clip gradient
        grad = np.clip(grad, -10, 10)
        
        # Backtracking line search
        alpha = alpha_init
        h_trial = h - alpha * grad
        h_trial = project_and_rescale(h_trial)
        c5_trial = evaluate_c5(h_trial)
        
        while c5_trial >= evaluate_c5(h) and alpha > 1e-10:
            alpha *= beta
            h_trial = h - alpha * grad
            h_trial = project_and_rescale(h_trial)
            c5_trial = evaluate_c5(h_trial)
            
        # Update h
        h = h_trial
        
        # Check for improvement
        c5 = evaluate_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = i

        # Early stopping
        if i - last_improvement > patience:
            break

    if best_h is None:
        best_h = h.copy()
    c5_final = evaluate_c5(best_h)
    
    return (best_h.tolist(), c5_final, n_points)
