import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    except NameError:
        # Fallback to uniform distribution
        n_points = 70
        h0 = np.ones(n_points) * 0.5
    
    # Normalize h0 to satisfy sum constraint
    h0 = h0 * (n_points / 2.0) / np.sum(h0)
    
    # Define objective function for optimization
    def compute_C5(h_vec):
        dx = 2.0 / n_points
        # Compute all correlations for all shifts
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Max value is at the middle of the full correlation array
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Projection back to feasible set
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Rescale to meet integral constraint
        h_vec = h_vec * (n_points / 2.0) / np.sum(h_vec)
        return h_vec
    
    # Optimization loop using projected gradient descent
    h = h0.copy()
    
    # Gradient computation
    def grad_C5_at(h_vec):
        dx = 2.0 / n_points
        # Compute correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_k = np.argmax(corr)
        # Gradient of max_k correlation w.r.t. h at that shift
        grad = np.zeros_like(h_vec)
        # Compute gradient for the specific maximum shift
        k_max = max_k - (n_points - 1)  # Convert from full to shift index
        
        # Compute gradient components
        for i in range(n_points):
            if i - k_max >= 0:
                grad[i] += 1 - h_vec[i - k_max]
            if i + k_max < n_points:
                grad[i] += 1 - h_vec[i + k_max]
        return grad * dx
    
    # Line search parameters
    alpha = 0.05
    beta = 0.7
    
    # Optimization loop
    max_iter = 10000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        current_C5 = compute_C5(h)
        
        # Compute gradient
        g = grad_C5_at(h)
        
        # Update step
        h_new = h - alpha * g
        
        # Project to feasible set
        h_new = project(h_new)
        
        # Check if we made progress
        new_C5 = compute_C5(h_new)
        if new_C5 < current_C5:
            h = h_new
        else:
            alpha *= beta
    
    # Perform local search over discrete shifts
    def evaluate_shift(h_vec, shift):
        dx = 2.0 / n_points
        shifted = np.roll(h_vec, shift)
        corr = np.correlate(shifted, 1 - shifted, mode="full")
        return np.max(corr) * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    # Try small shifts to see if we can improve
    shifts = range(-10, 11)
    for s in shifts:
        if time.time() - start_time > budget_s - 1:
            break
        shifted_h = np.roll(h, s)
        shifted_h = project(shifted_h)
        shifted_C5 = compute_C5(shifted_h)
        if shifted_C5 < best_C5:
            best_C5 = shifted_C5
            best_h = shifted_h
    
    # Final evaluation
    final_C5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_C5, n_points)
