import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial_h_values if available

        n_points = 45

    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)

    tau = 0.1

    max_iter = 10000
    lr = 0.1
    beta = 0.5
    patience = 50

    best_c5 = evaluate_erdos_solution(h)
    
    def compute_overlap(h, dx):
        """Compute all overlaps efficiently using correlation."""
        h_extended = np.concatenate([h, np.zeros(n_points - 1)])
        h_flipped = h[::-1]
        correlations = signal.correlate(h_extended, h_flipped, mode='valid')
        overlaps = correlations[:-1] * dx
        # The actual overlap integral is h(x)(1-h(x+k))
        # which is -h(x)h(x+k) + h(x)
        # So we compute overlaps as: sum(h_i * (1 - h_{i+k})) = sum(h_i) - sum(h_i * h_{i+k})
        # But it's easier to directly compute: sum(h_i * (1 - h_{i+k}))
        h1 = np.concatenate([h, np.zeros(n_points-1)])
        h1_rev = h1[::-1]
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(1 - n_points, n_points):
            start_idx = max(0, k)
            end_idx = min(n_points, n_points + k)
            k_shift = max(0, -k)
            if end_idx > start_idx:
                overlaps[k + n_points - 1] = np.sum(h[start_idx:end_idx] * (1 - h1_rev[k_shift:k_shift + (end_idx - start_idx)])) * dx
        return overlaps
    
    def compute_loss_gradient(h, dx):
        """Approximate gradient of the loss using smooth max."""
        # Overlap for all shifts
        overlaps = compute_overlap(h, dx)
        # Smoothed maximum
        exp_overlaps = np.exp(tau * overlaps)
        sum_exp = np.sum(exp_overlaps)
        if sum_exp == 0:
            return np.zeros_like(h), 0
        # Gradient
        weights = exp_overlaps / sum_exp
        # Compute gradient
        grad = np.zeros_like(h)
        for k in range(len(overlaps)):
            # Manual derivative
            k_shift = k - (n_points - 1)
            start_idx = max(0, k_shift)
            end_idx = min(n_points, n_points + k_shift)
            k_rev = max(0, -k_shift)
            if end_idx > start_idx:
                h_part = h[start_idx:end_idx]
                h_rev_part = h1_rev[k_rev:k_rev + (end_idx - start_idx)]
                part_indices = np.arange(start_idx, end_idx)
                d_to_h = h_rev_part  # derivative of h[i] * (1 - h[i+k]) w.r.t. h[i] is (1 - h[i+k])
                d_to_h_r = (1 - h_part)  # derivative of h[i] * (1 - h[i+k]) w.r.t. h[i+k] is -h[i]
                part_grad = np.zeros_like(h_part)
                for i in range(len(part_indices)):
                    idx = part_indices[i]
                    idx_rev = k_rev + i
                    if idx_rev < n_points:
                        part_grad[i] += weights[k] * d_to_h[i]
                    if idx < n_points:
                        part_grad[i] -= weights[k] * h_part[i]  # -h[i] coefficient
                grad[start_idx:end_idx] += part_grad * dx
        return grad, np.log(sum_exp) / tau
    
    def project_and_rescale(h, n_points, dx):
        """Project h to [0,1] and rescale to sum to n_points/2."""
        h = np.clip(h, 0, 1)
        scale = (n_points / 2) / np.sum(h)
        h *= scale
        return h
    
    def backtrack_line_search(h, grad, lr, dx, tol=1e-6):
        """Backtracking line search for sufficient descent."""
        old_loss = None
        max_iter_backtrack = 10
        for _ in range(max_iter_backtrack):
            new_h = h - lr * grad
            new_h = project_and_rescale(new_h, n_points, dx)
            new_c5 = evaluate_erdos_solution(new_h)
            if old_loss is None or new_c5 <= old_loss - tol:
                return new_h, lr
            lr *= beta
        return h, lr
    
    iteration = 0
    last_improve = 0
    while iteration < max_iter:
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break
            
        grad, loss = compute_loss_gradient(h, dx)
        grad = np.nan_to_num(grad, nan=0.0, posinf=0.0, neginf=0.0)
        if np.allclose(grad, 0):
            break
            
        old_c5 = evaluate_erdos_solution(h)
        new_h, lr = backtrack_line_search(h, grad, lr, dx)
        
        # Update
        h = new_h
        
        # Check improvement
        new_c5 = evaluate_erdos_solution(h)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            last_improve = iteration
            
        # Early stopping
        if iteration - last_improve > patience:
            break
            
        iteration += 1
    
    return best_h.tolist(), best_c5, n_points

def evaluate_erdos_solution(h):
    """Evaluate final C5 (without changing h)."""
    import numpy as np
    from scipy import signal
    dx = 2.0 / len(h)
    # Build extended arrays for efficient convolution-style computation
    h1 = np.concatenate([h, np.zeros(len(h) - 1)])
    h1_rev = h1[::-1]
    # Compute all possible overlaps
    overlaps = np.zeros(2 * len(h) - 1)
    for k in range(1 - len(h), len(h)):
        start = max(0, k)
        end = min(len(h), len(h) + k)
        k_rev = max(0, -k)
        if end > start:
            h_part = h[start:end]
            h_rev_part = h1_rev[k_rev:k_rev + (end - start)]
            overlaps[k + len(h) - 1] = np.sum(h_part * (1 - h_rev_part)) * dx
    
    return np.max(overlaps)
