import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given step function h."""
    # Compute cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Scale by dx to get true integral values
    corr *= dx
    # Return the maximum overlap value
    return np.max(corr)

def enforce_constraints(h, n_points):
    """Enforce that h[i] in [0,1] and sum(h) == n_points/2"""
    # Clip values to [0,1]
    h = np.clip(h, 0, 1)
    # Normalize so sum equals n_points/2
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def mutate_individual(h, mutation_rate=0.05):
    """Apply small random mutations to h."""
    h = h.copy()
    n_points = len(h)
    # Apply random perturbations
    mutations = np.random.random(n_points) < mutation_rate
    delta = np.random.normal(0, 0.05, n_points)
    h[mutations] += delta[mutations]
    # Enforce constraints
    h = enforce_constraints(h, n_points)
    return h

def crossover(parent1, parent2):
    """Perform single-point crossover between two parents."""
    n_points = len(parent1)
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select one individual using tournament selection."""
    indices = np.random.choice(len(population), tournament_size)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 45  # Use the size of the current known good solution
    dx = 2.0 / n_points
    max_generations = 120
    population_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = int(0.2 * population_size)
    
    # Initialize population with a mix of initial solution and random solutions
    population = []
    if 'initial_h_values' in globals():
        # Add the initial solution
        initial_h = np.array(initial_h_values, dtype=float)
        initial_h = enforce_constraints(initial_h, n_points)
        population.append(initial_h)
        
        # Add more random solutions
        for _ in range(population_size - 1):
            # Create random solution
            h = np.random.random(n_points) * 0.2  # Small random values
            h = enforce_constraints(h, n_points)
            population.append(h)
    else:
        # All random solutions
        for _ in range(population_size):
            h = np.random.random(n_points) * 0.2  # Small random values
            h = enforce_constraints(h, n_points)
            population.append(h)
    
    # Evaluate initial population
    fitnesses = [evaluate_h(h, dx) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    # Track time
    start_time = time.time()
    
    # Evolution loop
    for generation in range(max_generations):
        elapsed = time.time() - start_time
        if elapsed >= budget_s * 0.95:
            break
            
        # Order population by fitness (ascending - lower is better)
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Save best solution so far
        current_best_fitness = sorted_fitnesses[0]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = sorted_population[0].copy()
        
        # Select and create new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(sorted_population[i].copy())
        
        # Create offspring
        while len(new_population) < population_size:
            # Selection
            parent1 = tournament_selection(sorted_population, sorted_fitnesses)
            parent2 = tournament_selection(sorted_population, sorted_fitnesses)
            
            # Crossover
            if np.random.random() < crossover_prob:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            child1 = mutate_individual(child1, mutation_rate)
            child2 = mutate_individual(child2, mutation_rate)
            
            new_population.append(child1)
            if len(new_population) < population_size:
                new_population.append(child2)
        
        # Update population
        population = new_population
        # Re-evaluate fitness
        fitnesses = [evaluate_h(h, dx) for h in population]
    
    # Final check and return best solution
    final_fitnesses = [evaluate_h(h, dx) for h in population]
    final_best_idx = np.argmin(final_fitnesses)
    if final_fitnesses[final_best_idx] < best_fitness:
        best_h = population[final_best_idx].copy()
        best_fitness = final_fitnesses[final_best_idx]
    
    return (best_h.tolist(), best_fitness, n_points)
