import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100  # Start with a reasonable number
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    h = np.clip(h, 0, 1)
    
    # Frank-Wolfe algorithm
    max_iter = 5000
    eta = 0.1
    tol = 1e-6
    
    best_h = h.copy()
    best_C5 = float('inf')
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps
        h_rev = np.flip(h)
        overlaps = correlate(h, h_rev, mode='full')
        # Adjust indices to get correct lag
        lag = np.arange(-n_points + 1, n_points)
        # Get the actual overlaps (multiply by dx)
        actual_overlaps = overlaps * dx
        
        # Find maximum overlap index and value
        max_idx = np.argmax(actual_overlaps)
        max_lag = lag[max_idx]
        max_overlap = actual_overlaps[max_idx] 
        
        # Compute gradient for the linear subproblem
        grad = np.zeros(n_points)
        if max_lag >= 0:
            grad[max_lag:] = -(h[:-max_lag] if max_lag < n_points else np.zeros(0))
        else:
            grad[:max_lag] = -(h[-max_lag:] if -max_lag < n_points else np.zeros(0))
            
        if max_lag <= 0:
            grad[:max_lag] = -(h[-max_lag:] if -max_lag < n_points else np.zeros(0))
        else:
            grad[max_lag:] = -(h[:-max_lag] if max_lag < n_points else np.zeros(0))
        
        # Modify to match correlation's style more accurately
        # The gradient corresponds to 1 - h[x + k] for each x
        # This gradient is essentially the cross-correlation with flipped h, 
        # but we need to ensure it's properly aligned for the actual operation.

        # Re-compute using correct method
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = (i + max_lag) % n_points
            if 0 <= j < n_points:
                grad[i] = -h[j]
            else:
                grad[i] = 0
        # Prefactor of 1 is implied
        # Monomial -h[j] for the correlation
        
        # Solve linear subproblem: minimize grad^T * d subject to 0 <= d <= 1, sum(d) = target_sum
        # This is a simple optimization over a polytope, solvable by taking
        # components of grad in descending order and filling with 1 or 0
        # while maintaining sum constraint
        
        # i-th coordinate of d (solution to linear subproblem) is 1 if grad[i] <= 0 and 0 if grad[i] >= 0
        # But we may need more nuanced scheme
        
        # We use a greedy approach:
        # Sort indices by gradient (descending)
        # Build an array d respecting sum constraint
        
        idx_sorted = np.argsort(-grad)  # Descending order of grad
        d = np.zeros(n_points)
        remaining = target_sum
        for i in idx_sorted:
            if remaining <= 0:
                break
            d[i] = min(1.0, remaining)
            remaining -= d[i]
        d = np.clip(d, 0, 1)
        d = d / np.sum(d) * target_sum  # rescale 
        
        # Update h
        h = (1 - eta) * h + eta * d
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        
        # Evaluate C5 
        # We already compute overlap on current iterate, save recompute if possible
        # We recompute now to be safe:
        h_rev = np.flip(h)
        overlaps = correlate(h, h_rev, mode='full') * dx
        C5 = np.max(overlaps)
        
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        if iteration % 100 == 0:
            pass  # Optional: log progress
            
    # Final check
    h_rev = np.flip(best_h)
    overlaps = correlate(best_h, h_rev, mode='full') * dx
    C5 = np.max(overlaps)
    
    return (best_h.tolist(), float(C5), n_points)
