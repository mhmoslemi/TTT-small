import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 70  # starting with the provided construction
    dx = 2.0 / n_points
    max_iter = 6000
    eta = 0.01
    patience = 100
    patience_counter = 0
    best_c5 = float('inf')
    best_h = None
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * (n_points / 2)  # Adjust to sum to n_points/2
    else:
        h = np.ones(n_points) * 0.5  # fallback initialization
        
    # Ensure sum constraint
    h_sum = np.sum(h)
    h = h * (n_points / 2) / h_sum
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all correlations (overlaps) for all lags
        corr = signal.correlate(h, 1 - h, mode='full')
        # Compute corresponding C5 values
        corr *= dx
        c5 = np.max(corr)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter > patience:
            break
            
        # Find the index of maximum overlap
        max_k = np.argmax(corr)
        # The actual lag is offset from full correlation array
        lag = max_k - (n_points - 1)
        
        # Compute gradient for this lag
        grad = np.zeros_like(h)
        
        if lag >= 0:
            # h[i] * (1 - h[i + lag]) is being maximized
            # Gradient is (1 - h[i + lag]) for many terms
            # Use slice-based assignment
            i_indices = np.arange(n_points - lag)
            grad[i_indices] = 1 - h[i_indices + lag]
            # Also handle the negative gradient part for lag < 0 (but here lag >= 0)
        else:
            # lag < 0: h[i] * (1 - h[i + lag]) 
            # still same idea but just shift in the other direction
            # i + lag >= 0 means i >= abs(lag)
            i_indices = np.arange(-lag, n_points)
            grad[i_indices] = -h[i_indices + lag]
            
        # Update h using gradient descent
        h -= eta * grad
        
        # Clip and project to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Maintain sum constraint exactly
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        
        # Adaptive learning rate
        if iteration % 100 == 0 or (iteration > 0 and patience_counter == 0):
            if eta > 1e-6:
                eta *= 0.95  # reduce learning rate

    # Final evaluation of the best solution
    if best_h is None:
        best_h = h
    corr = signal.correlate(best_h, 1 - best_h, mode='full')
    corr *= dx
    final_c5 = np.max(corr)
    
    return (best_h.tolist(), float(final_c5), n_points)
