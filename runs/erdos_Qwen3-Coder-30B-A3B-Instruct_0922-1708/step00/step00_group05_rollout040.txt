import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant for given step function h."""
    dx = 2.0 / len(h)
    # Compute all correlations
    corr = correlate(h, 1-h, mode="full")
    # The maximum correlation value scaled by dx
    return np.max(corr) * dx

    np.random.seed(seed)
    start_time = time.time()

        h0[10:20] = 1.0  # Some peaks
        h0[40:50] = 0.0  # Some valleys

    # Initialize PSO parameters
    n_particles = 20
    max_iter = int(budget_s / 0.05)
    w = 0.7
    c1 = c2 = 2.0
    v_max = 0.1
    dx = 2.0 / n_points
    
    # Normalize initial h
    h0_sum = np.sum(h0)
    if h0_sum != n_points / 2:
        h0 = h0 * (n_points / 2) / h0_sum

    # Particle initialization
    particles = []
    velocities = []
    personal_best = []
    personal_best_vals = []
    global_best = None
    global_best_val = float('inf')
    
    # Generate initial population
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-v_max, v_max, n_points))
    val = evaluate_erdos_solution(h0)
    personal_best.append(h0.copy())
    personal_best_vals.append(val)
    if val < global_best_val:
        global_best = h0.copy()
        global_best_val = val

    for i in range(1, n_particles):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h_sum = np.sum(h)
        if h_sum != n_points / 2:
            h = h * (n_points / 2) / h_sum
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        val = evaluate_erdos_solution(h)
        personal_best.append(h.copy())
        personal_best_vals.append(val)
        if val < global_best_val:
            global_best = h.copy()
            global_best_val = val

    # PSO optimization loop
    last_improvement = 0
    for iteration in range(max_iter):
        elapsed = time.time() - start_time
        if elapsed > budget_s:
            break

        # Update each particle
        for p in range(n_particles):
            # Update velocity
            r1 = random.random()
            r2 = random.random()
            vel = w * velocities[p] + c1 * r1 * (personal_best[p] - particles[p]) + c2 * r2 * (global_best - particles[p])
            velocities[p] = np.clip(vel, -v_max, v_max)
            
            # Update position
            particles[p] += velocities[p]
            # Clip to bounds
            particles[p] = np.clip(particles[p], 0, 1)
            # Normalize
            h_sum = np.sum(particles[p])
            if h_sum != n_points / 2:
                particles[p] = particles[p] * (n_points / 2) / h_sum
            # Evaluate
            val = evaluate_erdos_solution(particles[p])
            # Update personal best
            if val < personal_best_vals[p]:
                personal_best[p] = particles[p].copy()
                personal_best_vals[p] = val
                if val < global_best_val:
                    global_best = particles[p].copy()
                    global_best_val = val
                    last_improvement = iteration

        # Periodically introduce diversity
        if iteration - last_improvement > 100:
            for p in range(n_particles):
                particles[p] += np.random.normal(0, 0.01, n_points)
                particles[p] = np.clip(particles[p], 0, 1)
                h_sum = np.sum(particles[p])
                if h_sum != n_points / 2:
                    particles[p] = particles[p] * (n_points / 2) / h_sum
            last_improvement = iteration

    # Return best found solution
    return (global_best, global_best_val, n_points)
