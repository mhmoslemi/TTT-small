```python
import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize
import math

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for a given step function h."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral for all shifts
    conv = correlate(h, 1 - h, mode='full')
    # Extract the relevant part for k = 0 to n_points-1
    c5 = max(conv[n_points-1:]) * dx
    return c5

def initialize_particles(initial_h_values, n_particles, n_points):
    """Initialize swarm with diverse particles."""
    particles = []
    if initial_h_values is not None:
        particles.append(initial_h_values.copy())
    else:
        particles.append(np.full(n_points, 0.5))
    
    while len(particles) < n_particles:
        # Perturb initial vector to create diversity
        perturbed = particles[0] + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        # Normalize
        h_sum = np.sum(perturbed)
        if h_sum > 0:
            perturbed = perturbed * (n_points / 2) / h_sum
        particles.append(perturbed)
    
    return particles

def project_and_normalize(h, n_points):
    """Project h to [0,1] and normalize to sum to n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def update_velocity_and_position(particle, velocity, best_particle, global_best,
                                w, c1, c2, r1, r2):
    """Update particle velocity and position."""
    new_velocity = (w * velocity + 
                   c1 * r1 * (best_particle - particle) + 
                   c2 * r2 * (global_best - particle))
    
    # Ensure velocity doesn't exceed bounds
    new_velocity = np.clip(new_velocity, -0.1, 0.1)
    
    new_position = particle + new_velocity
    return new_position, new_velocity

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 70
    n_particles = 20
    max_iter = budget_s // 0.05  # ~12000 iterations with ~0.05s per
    w = 0.7
    c1, c2 = 2.0, 2.0
    max_no_improve_iter = 100
    
    # Initialize particles
    particles = initialize_particles(initial_h_values, n_particles, n_points)
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    particle_best = [p.copy() for p in particles]
    particle_best_values = [evaluate_erdos_solution(p) for p in particles]
    
    # Global best
    global_best_idx = np.argmin(particle_best_values)
    global_best = particle_best[global_best_idx].copy()
    global_best_value = particle_best_values[global_best_idx]
    
    start_time = time.time()
    
    iter_no_improve = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            
            new_pos, new_vel = update_velocity_and_position(
                particles[i], velocities[i], 
                particle_best[i], global_best,
                w, c1, c2, r1, r2
            )
            
            new_pos = project_and_normalize(new_pos, n_points)
            
            # Evaluate new position
            new_val = evaluate_erdos_solution(new_pos)
            
            # Update personal best
            if new_val < particle_best_values[i]:
                particle_best[i] = new_pos.copy()
                particle_best_values[i] = new_val
            
            # Update global best
            if new_val < global_best_value:
                global_best = new_pos.copy()
                global_best_value = new_val
                iter_no_improve = 0
            else:
                iter_no_improve += 1
                
            # Update particle
            particles[i] = new_pos
            velocities[i] = new_vel
            
        # Tournament selection: keep top 50%
        if iteration % 20 == 0:
            sorted_indices = np.argsort(particle_best_values)
            top_half = sorted_indices[:n_particles//2]
            
            # Replace worst with random particles from current pool
            worst_indices = sorted_indices[n_particles//2:]
            for idx in worst_indices:
                # Create new particle with slight perturbation
                new_h = particles[np.random.choice(top_half)] + np.random.normal(0, 0.01, n_points)
                new_h = project_and_normalize(new_h, n_points)
                particles[idx] = new_h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                particle_best[idx] = new_h.copy()
                particle_best_values[idx] = evaluate_erdos_solution(new_h)
                
        if iter_no_improve >= max_no_improve_iter:
            # Introduce more diversity
            for i in range(n_particles):
                noise = np.random.normal(0, 0.02, n_points)
                particles[i] = project_and_normalize(particles[i] + noise, n_points)
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
            iter_no_improve = 0
    
    return (global_best, global_best_value, n_points)
```
