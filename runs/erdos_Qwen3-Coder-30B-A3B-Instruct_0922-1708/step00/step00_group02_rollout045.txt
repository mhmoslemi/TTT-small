import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    min_alpha = 1e-6
    
    # Initial h from parent construction if available
    if 'initial_h_values' in globals():

    else:
        h = np.ones(n_points) * 0.5
    
    h = h / np.sum(h) * n_points / 2.0
    
    # Optimizer variables
    velocity_h = np.zeros(n_points)
    best_h = h.copy()
    best_C5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute cross-correlation h ⋆ (1 - h) 
        h1 = 1 - h
        corr = correlate(h, h1, mode='full')
        corr = corr * dx
        
        # Find maximum overlap index k
        k_max = np.argmax(corr)
        # Adjust k for the actual shift
        shift_idx = k_max - (n_points - 1)
        if shift_idx < 0:
            shift_idx = 0
        elif shift_idx > n_points - 1:
            shift_idx = n_points - 1
            
        # For computing gradient
        # We'll actually do a simplification by optimizing discrete overlaps
        # We compute C5 and corresponding derivative analytically
        C5 = np.max(corr)
        
        # Update according to gradients
        # grad_C5_wrt_h = (1 - h) + np.roll(h, shift_idx) - h * (we need more precise gradient)
        # For a simpler approach, we do a Gaussian smoothing / descent step
        # by computing the full overlap and gradient
        # Just compute the derivative Slightly different approach:
        grad_h = np.zeros(n_points)
        
        # For each position i, d/dh_i of overlap
        # This is complicated, lets approximate by finite difference or 
        # solve via minimization with constraints
        # For simplicity, let's adapt a better approximation:
        h_filled = np.concatenate([h, np.zeros(n_points - 1)])
        h1_filled = np.concatenate([1 - h, np.zeros(n_points - 1)])
        corr = correlate(h_filled, h1_filled, mode='full') * dx
        max_k_idx = np.argmax(corr)
        max_k = max_k_idx - (n_points - 1)
        
        # Gradient based on simplest approximation to gradient
        padded_h = np.concatenate([h, np.zeros(n_points)])  # Enlarged for shift
        shifted_h1 = np.roll(h1, -max_k)  # Stored for symmetric form as 1-h
        if max_k < 0:
            shifted_h1 = np.concatenate([shifted_h1[-max_k:], np.zeros(-max_k)])
        elif max_k > 0:
            shifted_h1 = np.concatenate([np.zeros(max_k), shifted_h1[:-max_k]])
        shifted_h1 = shifted_h1[:n_points]
        
        # Derivative term
        grad_h = h1 - shifted_h1
        # Apply gradients with momentum
        velocity_h = beta * velocity_h + (1 - beta) * grad_h
        h = h - alpha * velocity_h
        
        # Project h back to constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2.0) / np.sum(h)
        
        # Track best solution
        if C5 < best_C5:
            best_C5 = C5
            best_h = h.copy()
            
        # Adaptive learning rate
        if i % 500 == 0 and i > 0:
            if alpha > min_alpha:
                alpha *= 0.5
                
    # Final evaluation
    final_h = best_h
    corr = correlate(final_h, 1 - final_h, mode='full') * dx
    final_C5 = np.max(corr)
    
    return (final_h, final_C5, n_points)
