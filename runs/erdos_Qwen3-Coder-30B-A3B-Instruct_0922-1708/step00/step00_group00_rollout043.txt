import numpy as np
import time
from scipy import signal

def evaluate_h(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute correlation using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Convert to overlap integrals
    overlaps = corr * dx
    # The maximum overlap corresponds to the first and last elements of the correlation
    # since those are the ones for k in [-(n-1), n-1], the first element being k=-(n-1)
    # and second being k=-(n-2), ..., last being k=n-1. We cut the overlap array to
    # exclude the elements from k < 0, which gives us length 2*n - 1, but we look at
    # k from 0 to n-1 which is slice n-1:n-1+2*n-1 = slice n-1:3*n-2
    c5 = np.max(overlaps[n-1:2*n-1])
    return c5

def initialize_population(n_points, pop_size):
    """Initialize a population of step function candidates."""
    population = []
    
    # If initial_h_values is available, use it
    try:
        initial_h = initial_h_values
        if len(initial_h) == n_points:
            population.append(np.array(initial_h))
    except NameError:
        pass

    # Fill the rest with random valid configurations
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        # Adjust sum to exactly n_points / 2
        h_sum = np.sum(h)
        h = h * (n_points / 2) / h_sum
        # Ensure bounds
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def crossover(parent1, parent2, crossover_rate=0.8):
    """Perform crossover between two parent vectors."""
    if np.random.rand() > crossover_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a vector by adding Gaussian noise and re-normalizing."""
    n = len(h)
    noise = np.random.normal(0, 0.02, n)
    h_mut = h + noise
    
    # Clip to valid range
    h_mut = np.clip(h_mut, 0, 1)
    
    # Renormalize to sum to n/2
    h_sum = np.sum(h_mut)
    h_mut = h_mut * (n / 2) / h_sum
    
    return h_mut

def select_parents(population, fitnesses, tournament_size=5):
    """Select a parent pair via tournament selection."""
    def tournament():
        candidates = np.random.choice(len(population), tournament_size)
        best_idx = candidates[np.argmin([fitnesses[i] for i in candidates])]
        return population[best_idx]
    
    parent1 = tournament()
    parent2 = tournament()
    return parent1, parent2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45  # use the recommended size
    pop_size = 100
    generations = 0
    start_time = time.time()
    
    # Initialize population
    population = initialize_population(n_points, pop_size)
    
    best_c5 = float('inf')
    best_h = None
    
    while time.time() - start_time < budget_s - 0.1:  #留出一些时间做最后评估
        # Evaluate all individuals
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism: keep top 20%
        num_elite = pop_size // 5
        new_pop = population[:num_elite]
        
        # Generate new population
        while len(new_pop) < pop_size:
            parent1, parent2 = select_parents(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_pop.append(child1)
            if len(new_pop) < pop_size:
                new_pop.append(child2)
        
        population = new_pop
        generations += 1
    
    # Final evaluation of the best solution
    if best_h is not None:
        final_c5 = evaluate_h(best_h)
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to the initial solution
        initial_h = np.array(initial_h_values) if 'initial_h_values' in globals() else np.random.rand(n_points)
        initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
        initial_h = np.clip(initial_h, 0, 1)
        final_c5 = evaluate_h(initial_h)
        return (initial_h.tolist(), final_c5, n_points)
