import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate C5 for a given step function h"""
    # Compute overlap integral for all shifts
    overlaps = signal.correlate(h, 1 - h, mode="full") * dx
    # The overlap integral is defined for shifts from -(n-1) to (n-1)
    n = len(h)
    # Extract the relevant overlaps: maximum shift is n-1
    start_idx = n - 1
    end_idx = 2 * n - 2
    relevant_overlaps = overlaps[start_idx:end_idx]
    return np.max(relevant_overlaps)

def initialize_population(n_points, population_size, initial_h=None):
    """Initialize population with constraint satisfaction"""
    population = []
    
    # Add initial solution if available
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill remaining population with random samples
    while len(population) < population_size:
        # Generate random step function
        h = np.random.rand(n_points)
        # Ensure sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Tournament selection for parent selection"""
    selected = []
    for _ in range(len(population)):
        # Randomly select tournament participants
        participants = np.random.choice(len(population), tournament_size)
        # Choose best participant
        best_idx = participants[np.argmin([fitnesses[i] for i in participants])]
        selected.append(population[best_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_prob=0.8):
    """Single-point crossover"""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    crossover_point = np.random.randint(1, n - 1)
    
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return child1, child2

def mutate(h, mutation_rate=0.05):
    """Mutate a solution with soft constraints"""
    n = len(h)
    for i in range(n):
        if np.random.rand() < mutation_rate:
            # Add small random perturbation
            h[i] += np.random.normal(0, 0.02)
            # Ensure bounds
            h[i] = np.clip(h[i], 0, 1)
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    if abs(current_sum - n / 2) > 1e-6:
        # Rescale to meet exact sum constraint
        h = h * (n / 2) / current_sum
    
    return h

def genetic_algorithm(n_points, population_size=100, max_generations=120, 
                      initial_h=None, time_budget=600):
    """Main GA routine for minimizing C5"""
    # Initialize
    start_time = time.time()
    population = initialize_population(n_points, population_size, initial_h)
    dx = 2.0 / n_points
    
    best_fitness = float('inf')
    best_individual = None
    
    for generation in range(max_generations):
        # Early termination check
        if time.time() - start_time > time_budget * 0.95:
            break
        
        # Evaluate fitness
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best solution
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_fitness:
            best_fitness = fitnesses[min_fitness_idx]
            best_individual = population[min_fitness_idx].copy()
        
        # Elitism: preserve best solutions
        elite_count = population_size // 5
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elites = [population[i].copy() for i in elite_indices]
        
        # Selection, crossover, mutation
        selected = tournament_selection(population, fitnesses)
        new_population = elites.copy()
        
        # Create offspring
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(len(selected), 2, replace=False)
            child1, child2 = crossover(selected[parent1], selected[parent2])
            
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        # Trim to exact population size
        population = new_population[:population_size]
    
    return best_individual, best_fitness
