import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Get initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h / np.sum(h) * target_sum
    else:
        # Create a reasonable starting point
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = h * target_sum / np.sum(h)
    
    # Ensure bounds
    h = np.clip(h, 0, 1)
    
    # Normalize to satisfy constraint
    h = h * target_sum / np.sum(h)
    
    # Evaluate initial C5
    def compute_C5(h):
        h = np.array(h)
        conv = signal.correlate(h, 1 - h, mode='full')
        max_conv = np.max(conv)
        return max_conv * dx
    
    best_h = h.copy()
    best_C5 = compute_C5(h)
    
    start_time = time.time()
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1:
        improved = False
        for _ in range(100):  # 100 trials per iteration
            if time.time() - start_time >= budget_s - 1:
                break
                
            # Create a perturbation
            h_new = h.copy()
            
            # Select random points to perturb
            indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
            for i in indices:
                # Perturb by small amount
                delta = np.random.uniform(-0.05, 0.05)
                old_val = h_new[i]
                h_new[i] = old_val + delta
                
            # Keep within bounds
            h_new = np.clip(h_new, 0, 1)
            
            # Maintain sum constraint
            current_sum = np.sum(h_new)
            if current_sum > 0:
                scale_factor = target_sum / current_sum
                h_new = h_new * scale_factor
                
            # Apply smoothing to maintain structure
            window = min(5, n_points // 10)
            if window > 1:
                h_new = np.convolve(h_new, np.ones(window)/window, mode='same')
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
            
            # Evaluate
            new_C5 = compute_C5(h_new)
            
            if new_C5 < best_C5:
                best_h = h_new.copy()
                best_C5 = new_C5
                improved = True
                no_improvement_count = 0
            else:
                no_improvement_count += 1
                
        if not improved:
            no_improvement_count += 1
        
        if no_improvement_count >= max_no_improvement:
            break
            
        h = best_h.copy()
    
    # Final validation
    final_C5 = compute_C5(best_h)
    return (best_h.tolist(), final_C5, n_points)
