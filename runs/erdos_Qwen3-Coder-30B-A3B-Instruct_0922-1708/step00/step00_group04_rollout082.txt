import numpy as np
from scipy import optimize
from math import exp
import time

def evaluate_c5(h_values, n_points):
    dx = 2.0 / n_points
    # Compute cross-correlation of h with (1 - h)
    corr = np.correlate(h_values, 1 - h_values, mode="full")
    # Scale by dx and find maximum
    max_corr = np.max(corr) * dx
    return max_corr

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 42
        # Start with triangular wave
        h = np.linspace(0, 1, n_points)
        h = np.minimum(h, 1 - h)
        # Normalize
        h = h * (n_points / 2) / np.sum(h)
    
    # Ensure bounds and sum
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, n_points)
    
    # Optimization vars
    t0 = time.time()
    it = 0
    patience = 0
    max_patience = 100
    lr = 0.01
    
    # Store last few c5 values for convergence check
    recent_c5 = []
    
    while time.time() - t0 < budget_s - 1.0 and patience < max_patience:
        it += 1
        
        # Compute correlation at all lags (full mode)
        corr = np.correlate(h, 1 - h, mode="full")
        dx = 2.0 / n_points
        corr_scaled = corr * dx
        
        # Find maximum
        max_idx = np.argmax(corr_scaled)
        max_val = corr_scaled[max_idx]
        
        # Use log-sum-exp surrogate for max to enable gradient
        # We'll use a soft maximum
        beta = 10.0
        exp_corr = np.exp(beta * corr_scaled)
        soft_max = np.log(np.sum(exp_corr)) / beta
        
        # Compute gradient of the soft max w.r.t h
        # ∂softmax = (1/C) * exp(β * corr) * ∂corr/∂h
        # ∂corr/∂h at lag k is (1-h) for lags in [0,k], 
        # and is h shifted by k for k in [0,n-1]
        # But for full correlation, we compute directly from the
        # cross correlation formula
        # Use finite difference approximation
        eps = 1e-6
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_eps = h.copy()
            h_eps[i] += eps
            corr_eps = np.correlate(h_eps, 1 - h_eps, mode="full")
            corr_eps *= dx
            soft_max_eps = np.log(np.sum(np.exp(beta * corr_eps))) / beta
            grad[i] = (soft_max_eps - soft_max) / eps
        
        # Update
        h -= lr * grad
        
        # Project onto feasible set: [0,1] and sum = n_points / 2
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate
        c5 = evaluate_c5(h, n_points)
        recent_c5.append(c5)
        if len(recent_c5) > 10:
            recent_c5.pop(0)

        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # Adjust learning rate
        if len(recent_c5) >= 3:
            if recent_c5[-1] >= recent_c5[-2] and recent_c5[-2] >= recent_c5[-3]:
                lr *= 0.9
                if lr < 1e-6:
                    lr = 1e-6
        
        # Early termination
        if best_c5 < 0.3807:
            break
    
    # Final cleanup
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, n_points)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
