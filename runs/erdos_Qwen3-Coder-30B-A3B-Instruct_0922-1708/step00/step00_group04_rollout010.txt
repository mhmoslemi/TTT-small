import numpy as np
from scipy import signal
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use the initial construction if it exists
    try:
        h_init = initial_h_values
        n_points = len(h_init)
    except:
        # Fallback: create a simple piecewise constant function
        n_points = 42
        h_init = np.ones(n_points) * 0.5
        h_init[:n_points//2] = 1.0
        h_init[n_points//2:] = 0.0
        # Normalize to integral = 1
        h_init = h_init / np.sum(h_init) * 2.0 / n_points
        
    # Initialize problem
    dx = 2.0 / n_points
    h = Variable(n_points, name='h')
    a = Variable(name='a')  # Intermediate level
    
    # Constraints
    constraints = [h >= 0, h <= 1, 
                   sum(h) * dx == 1,  # integral constraint
                   a >= 0, a <= 1]  # level constraint
    
    # Create cross-correlation terms for maximum overlap
    # We'll model this with auxiliary variables
    c5 = Variable(name='c5')
    
    # Sample a few transitions to avoid overly complex formulation
    # We treat h as piecewise constant, but model this greedily
    
    # Solve directly with a heuristic approach using scipy optimization
    
    def compute_C5(h_vec):
        h_vec = np.array(h_vec)
        # Compute cross-correlations for all lags
        corr = signal.correlate(h_vec, 1 - h_vec, mode='full')
        # Scale to continuous interval
        corr *= dx
        # Return maximum value
        return np.max(corr)
    
    def objective(h_vec):
        return compute_C5(h_vec)
    
    # Using scipy's optimization with constraints
    # Simple heuristic: evolve the function in space to minimize overlap
    
    best_h = h_init.copy()
    best_c5 = compute_C5(best_h)
    
    # Parameters for optimization
    n_iter = 10000
    step_size = 0.01
    
    # Iterative hill-climbing
    for i in range(n_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Try small perturbations
        h_trial = best_h.copy()
        idx = np.random.randint(0, n_points)
        pert = np.random.uniform(-step_size, step_size)
        h_trial[idx] += pert
        # Project to [0,1]
        h_trial[idx] = np.clip(h_trial[idx], 0, 1)
        
        # Maintain integral constraint
        integral_diff = 1 - np.sum(h_trial) * dx
        # Distribute correction uniformly
        h_trial += integral_diff / n_points
        
        # Check if improved
        c5_trial = compute_C5(h_trial)
        if c5_trial < best_c5:
            best_h = h_trial
            best_c5 = c5_trial
    
    # Final validation
    final_c5 = compute_C5(best_h)
    return (list(best_h), final_c5, n_points)
