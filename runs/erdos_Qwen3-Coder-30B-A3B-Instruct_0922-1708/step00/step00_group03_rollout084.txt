import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_initial = np.array(initial_h_values)
        h_initial = h_initial * (target_sum / np.sum(h_initial))
    else:
        h_initial = np.full(n_points, 0.5 * dx)
        
    # Define optimization variables
    h = cp.Variable(n_points)
    
    # Add constraints to the optimization problem
    constraints = [
        h >= 0,
        h <= 1,
        cp.sum(h) == target_sum
    ]
    
    # Create a scalar variable for the maximum overlap
    t = cp.Variable()
    
    # Correlation computation part
    max_correlation = 0
    for k in range(-n_points + 1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >= 0) & (idx < n_points)
        corr = cp.sum(cp.multiply(h[valid], 1 - h[idx[valid]]))
        max_correlation = cp.maximum(max_correlation, corr)
        
    # Now we want to minimize max_correlation (or rather, make sure max_correlation <= t)
    constraints.append(max_correlation <= t)
    
    # Objective is to minimize t which is the upper bound on C5
    objective = cp.Minimize(t)
    
    # Problem definition
    prob = cp.Problem(objective, constraints)
    
    # Solver options
    solver = cp.MOSEK  # assuming MOSEK is available
    
    try:
        # Try to solve with timeout
        prob.solve(solver=solver, verbose=False, time_limit=budget_s)
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            h_opt = np.array(h.value).flatten()
        else:
            h_opt = h_initial
    except:
        h_opt = h_initial
    
    # Ensure constraints are met
    h_opt = np.clip(h_opt, 0, 1)
    current_sum = np.sum(h_opt)
    if abs(current_sum - target_sum) > 1e-6:
        h_opt = h_opt * target_sum / current_sum
        
    # Evaluate final result
    c5_bound = evaluate_erdos_solution(h_opt)
    
    return (h_opt, c5_bound, n_points)
