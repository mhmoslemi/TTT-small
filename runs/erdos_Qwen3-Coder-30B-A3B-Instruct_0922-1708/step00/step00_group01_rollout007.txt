import numpy as np
from scipy.signal import correlate
import time

def evaluate_C5(h):
    """Evaluate the exact C5 value using cross-correlation."""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute all overlaps for all lags
    overlaps = correlate(h, 1 - h, mode='full')
    # Take only the relevant part (positive lags up to n_points-1)
    # and convert to the actual values
    max_overlap = np.max(overlaps)
    return max_overlap * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a symmetric plateau with tapering
        n_points = 47
        h = np.ones(n_points)
        # Make it symmetric and taper it
        mid = n_points // 2
        for i in range(n_points):
            h[i] = max(0.1, 1.0 - abs(i - mid) / (mid * 0.8))
        # Normalize to sum = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
    
    # Check initial value
    assert np.all(h >= 0) and np.all(h <= 1)
    assert abs(np.sum(h) - n_points / 2) < 1e-10
    
    # Parameters
    tau = 1.0
    lr = 0.05
    anneal_factor = 0.999
    max_iter = 3000
    tol = 1e-5
    
    best_h = h.copy()
    best_C5 = evaluate_C5(h)
    last_C5 = best_C5
    last_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 0.1:
            break
            
        # Compute overlaps for all lags
        dx = 2.0 / n_points
        overlaps = correlate(h, 1 - h, mode='full')
        # We only need positive lags, so we extract the right half
        # The full length is 2*n_points - 1, we want from n_points-1 to 2*n_points-2
        overlaps = overlaps[n_points-1:]
        
        # Prepare for gradient computation
        exp_overlaps = np.exp(tau * overlaps)
        exp_sum = np.sum(exp_overlaps)
        
        # Gradient w.r.t. h
        grad = np.zeros(n_points)
        
        # For each point and each lag
        for k in range(n_points):
            lag = k
            # Contribution from overlap at lag k
            exp_overlap_k = np.exp(tau * overlaps[k])
            # derivative of overlap with respect to h[i]
            # overlap = sum(h[i] * (1 - h[(i+k)%n_points]))
            # d/dh[i] overlap = 1 - h[(i+k)%n_points]
            for i in range(n_points):
                j = (i + k) % n_points
                grad[i] += exp_overlap_k * (- (1 - h[j])) * dx
            
        grad /= exp_sum
        
        # Update with gradient descent
        h = h - lr * grad
        
        # Project back to [0,1]
        h = np.clip(h, 0, 1)
        
        # Project to sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate C5
        current_C5 = evaluate_C5(h)
        
        # Update best if improved
        if current_C5 < best_C5:
            best_C5 = current_C5
            best_h = h.copy()
        
        # Check convergence
        if abs(last_C5 - current_C5) < tol:
            # Early stopping if no improvement
            current_time = time.time()
            if current_time - last_time < 0.01:
                break
            last_time = current_time
        
        last_C5 = current_C5
        
        # Anneal temperature
        tau *= anneal_factor
    
    # Final check and return
    final_C5 = evaluate_C5(best_h)
    
    return (best_h.tolist(), float(final_C5), int(n_points))
