import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    """Compute the C5 constant for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    h_full = np.concatenate([h, np.zeros(n-1)])
    h_rev = np.concatenate([np.zeros(n-1), h])
    conv = correlate(h_full, 1-h_rev, mode="full")
    max_val = np.max(conv[:2*n-1]) * dx
    return max_val
