import numpy as np
from cvxpy import *
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        n_points = len(h_init)
    else:
        # Create a reasonable starting point
        n_points = 70
        h_init = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    h_true = h_init.copy()
    h_best = h_true.copy()
    c5_best = float('inf')
    start_time = time.time()
    
    # Precompute all lag indices
    max_lag = n_points - 1
    lags = list(range(-max_lag, max_lag + 1))
    
    # Function to compute overlap integral
    def compute_c5(h_vec):
        # Convolution approach
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        max_overlap = np.max(corr) * dx
        return max_overlap
    
    # Constraint: integral of h = 1
    def integral_constraint(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    # Function to check feasibility
    def is_feasible(h_vec):
        return np.all(h_vec >= 0) and np.all(h_vec <= 1) and abs(integral_constraint(h_vec)) < 1e-6
    
    # Try multiple random initializations
    n_init = 5
    for init_idx in range(n_init):
        if time.time() - start_time > budget_s - 5:
            break
        h_current = h_init.copy()
        if init_idx > 0:
            # Perturb slightly
            perturb = np.random.normal(0, 0.01, n_points)
            h_current += perturb
            h_current = np.clip(h_current, 0, 1)
            # Renormalize
            h_current *= (n_points / 2) / np.sum(h_current)
        h_current = h_current.copy()
        
        # Try multiple QCQP solves per initialization
        n_solves = 5
        for solve_idx in range(n_solves):
            if time.time() - start_time > budget_s - 5:
                break
            # Use CVXPY for QP
            h_var = Variable(n_points)
            # Variables
            C_var = Variable()
            
            # Constraints
            constraints = [0 <= h_var, h_var <= 1]
            # Integral constraint
            constraints.append(sum(h_var) == n_points / 2.0)
            
            # Overlap constraints
            for k in lags:
                # Correlation sum
                if k == 0:
                    val = np.sum(h_var * (1 - h_var))
                else:
                    if k > 0:
                        val = sum(h_var[i] * (1 - h_var[i + k]) for i in range(n_points - k))
                    else:
                        val = sum(h_var[i - k] * (1 - h_var[i]) for i in range(-k, n_points))
                constraints.append(val <= C_var * dx)
            
            # Objective
            objective = Minimize(C_var)
            prob = Problem(objective, constraints)
            
            try:
                prob.solve(solver=OSQP, verbose=False, max_iter=10000)
                if prob.status in ['optimal', 'optimal_inaccurate']:
                    c5_new = prob.value
                    # Check if solution is feasible
                    h_sol = h_var.value
                    if is_feasible(h_sol):
                        if c5_new < c5_best:
                            c5_best = c5_new
                            h_best = h_sol.copy()
                        # Update current to attempt further improvement
                        h_current = h_sol.copy()
                    else:
                        # Fix slightly
                        h_current = np.clip(h_sol, 0, 1)
                        h_current = h_current * (n_points / 2) / np.sum(h_current)
                        c5_new = compute_c5(h_current)
                        if c5_new < c5_best:
                            c5_best = c5_new
                            h_best = h_current.copy()
                else:
                    # Failed to solve optimally, continue with current
                    pass
            except:
                # Fall back to local improvement
                h_current = np.clip(h_current, 0, 1)
                h_current = h_current * (n_points / 2) / np.sum(h_current)
                c5_new = compute_c5(h_current)
                if c5_new < c5_best:
                    c5_best = c5_new
                    h_best = h_current.copy()
                
            # Refine with gradient-based method if time allows
            if time.time() - start_time > budget_s - 10:
                break
        # Update with better solution
        current_c5 = compute_c5(h_current)
        if current_c5 < c5_best:
            c5_best = current_c5
            h_best = h_current.copy()
    
    # Final validation
    if not is_feasible(h_best):
        h_best = np.clip(h_best, 0, 1)
        h_best = h_best * (n_points / 2) / np.sum(h_best)
    c5_best = compute_c5(h_best)
    
    return h_best.tolist(), c5_best, n_points
