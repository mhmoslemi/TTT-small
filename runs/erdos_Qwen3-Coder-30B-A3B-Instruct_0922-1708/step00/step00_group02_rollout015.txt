# Helper functions for overlap computation
def compute_overlap(h, dx):
    """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
    # We compute all overlaps using convolution
    # The overlap with lag k is given by convolution of h with (1-h) at that lag
    # We compute the full convolution and find max
    full_conv = np.correlate(h, 1 - h, mode='full')
    # Now we need to extract the right part of the convolution, to align with the lags we want
    # When convolving with flipped 1-h, lags will go from -n+1 to n-1
    # For lag k, h(x) and 1-h(x+k) overlap, for valid overlap lags, we need to shift properly
    # Let's actually use the direct approach:
    # Valid lags for overlapping support are from 0 to 2 (as h is supported on [0,2])
    # But actually we want all valid lags and compute maximum
    overlaps = []
    n = len(h)
    # For lag k, we want to compute int h(x) * (1 - h(x+k)) dx
    # This can be computed if x+k is within [0,2] - so k from -2 to 2 for valid overlap
    for k in range(-n+1, n):  # approximate all possible lags based on support
        lag_overlap = 0.0
        k_idx = k + n - 1 # adjustment to map lag k to full convolution index
        if 0 <= k_idx < len(full_conv):
            overlaps.append(full_conv[k_idx] * dx)
    if not overlaps:
        return 0.0
    return max(overlaps)

def evaluate_h(h_values, dx):
    """Evaluate the C5 bound of a given h with dx spacing"""
    return compute_overlap(h_values, dx)

def project_to_feasible(h, dx):
    """Ensure h is within [0, 1] and sum(h) * dx = 1"""
    h = np.clip(h, 0, 1)
    total = np.sum(h) * dx
    if total > 0:
        h = h / total * 1.0
    return h

def create_basis_functions(n_points, m_basis):
    """Create m_basis sigmoid basis functions on [0, 2]"""
    # centers
    centers = np.linspace(0, 2, m_basis)
    # scale
    sigma = 2.0 / m_basis
    x = np.linspace(0, 2, n_points)
    basis = np.zeros((n_points, m_basis))
    for i, c in enumerate(centers):
        basis[:, i] = 1.0 / (1.0 + np.exp(-(x - c) / sigma))
    return basis

def regularize_basis(a, gamma=0.8):
    """Elastic net regularization with L1 and L2 components"""
    l1 = gamma * np.sign(a)
    l2 = (1 - gamma) * a
    return l1 + l2

def setup_optimization(n_points, initial_h=None):
    """Initialize optimization variables"""
    m = int(2 * np.sqrt(n_points))  # number of basis functions
    basis = create_basis_functions(n_points, m)
    
    # Initialize coefficients
    if initial_h is not None:
        # Fit coefficients to initial_h
        a = np.linalg.lstsq(basis, initial_h, rcond=None)[0]
    else:
        a = np.random.uniform(-0.1, 0.1, m)
    
    # Scale such that sum(h)*dx = 1
    h_test = basis @ a
    h_test = np.clip(h_test, 0, 1)
    dx = 2.0 / n_points
    integral = np.sum(h_test) * dx
    if integral > 0:
        a = a * (1.0 / integral)
    
    return basis, a

        eps = 1e-5
