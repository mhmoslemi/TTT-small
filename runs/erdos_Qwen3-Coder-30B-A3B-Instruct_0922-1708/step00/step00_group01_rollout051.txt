import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)

        h0[mid:] = 0.5
        h0 = h0 * 2 / np.sum(h0)  # Normalize
    
    n_points = len(h0)
    dx = 2.0 / n_points
    max_iter = 5000
    
    # Optimize with gradient descent with adaptive learning rate
    h = h0.copy()
    best_h = h.copy()
    best_c5 = evaluate_overlap(h, dx)
    
    learning_rate = 0.1
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current overlap integrals
        overlaps = evaluate_overlap_all(h, dx)
        max_overlap_idx = np.argmax(overlaps)
        max_overlap_val = overlaps[max_overlap_idx]
        
        # Compute gradient for the maximum overlap component
        grad = compute_gradient(h, max_overlap_idx, dx)
        
        # Update with gradient
        h_new = h - learning_rate * grad
        
        # Project to feasible set [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy sum constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update h
        h = h_new
        
        # Evaluate new solution
        new_c5 = evaluate_overlap(h, dx)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
            
        # Anneal learning rate
        if i % 1000 == 0 and i > 0:
            learning_rate *= 0.5
            
    return (best_h, best_c5, n_points)

def evaluate_overlap(h, dx):
    """Compute max overlap integral"""
    # Compute cross-correlation of h with (1-h)
    h1 = 1 - h
    correlation = correlate(h, h1, mode='full')
    # Adjust for the size of the correlation and dx
    max_corr = np.max(correlation) * dx
    return max_corr

def evaluate_overlap_all(h, dx):
    """Get all overlap values as a vector"""
    h1 = 1 - h
    correlation = correlate(h, h1, mode='full')
    return correlation * dx

def compute_gradient(h, max_lag, dx):
    """Compute gradient w.r.t. h for the max overlap at max_lag"""
    n = len(h)
    
    # Prepare h1 = 1 - h
    h1 = 1 - h
    
    # Gradient for one lag is computed manually from the definition
    # For overlap at lag k, the integral involves h[i] * h1[i+k]
    # Which implies gradient w.r.t. h[i] includes h1[i+k]
    # And w.r.t. h1[i+k] includes h[i]
    # So at each h[i], we sum up the h1 terms with all correspondences
    
    grad = np.zeros_like(h)
    
    # Calculate contribution of h[i] to the overlap at max_lag
    for i in range(n):
        # For correlation at max_lag, look backward from position i
        j = i - max_lag + (n - 1)  # correspond to index j in h1 (w.r.t. forward indexing)
        if 0 <= j < n:
            grad[i] += h1[j]
            
        # For overlap contributions from h at other positions
        # The j in h1 should shift as we go through i and compute 
        # h[i] term in the integral for overlap max_lag
        # It's straigthforward:
        # overlap at k = sum_{i} h[i] * h1[i+k] for k-th lag
        # So gradient w.r.t. to h[i] = sum_{k} h1[i+k] (as long as i+k is valid)
        # But we're using the k-th overlap, so only contribute when k is the max lag
        
        # thus simpler: h[i] appears in the overlap sum if i+k = max_lag for some valid k
        # That's the lag that is max_lag in correlate default mode where most positive element is last element
        # So for the full correlation h * h1, if max_lag corresponds to k (where h1 = h1[-k] for positive k)
        # The lag in full correlation spans from -n+1 to n-1 
        # Which means h1 indexing involves offset = n - 1 - k
        # We are in terms of lag index
        # Let's map the full overlap lag to the original h, and gradient back
        # to find h contribution
        # Below code is cleaner if I compute it properly for a fixed lag in correlate
                
        # We use matrix approach:
        # But simple thought: h[i] term contributes h1[i+k] once in the overlap 
        # where k = max_lag (offset used in correlate)
        # Index in h1 for that element is `i + max_lag` 
        # But it's also abs(max_lag) offset in correlate, with sign depending

    # Reaching full matrix form conclusion:
    # If we let lag = max_lag be the best lag
    # for index i in h, its impact on overlap == sum of h1 indexed by (i - max_lag)  
    # because overlapping with h1 shifted by i
    # In full correlate, i-th element of h contributes to correlation at lag which is i + (n - 1) - idx_in_h1
    # Where the positive index is i + (n - 1) - ??  
    # Let k be lag, start from h induction:
    # h[i] * h1[j] in i+j = k in cross-correlation
    # So j = k - i, and thus derivative of overlap w.r.t. h[i] = h1[k - i]
    # provided both i and k-i are valid
    # That gives us gradient component:
    
    # Let's compute gradients properly using downward index
    for i in range(n):
        idx_h1 = i - max_lag
        if 0 <= idx_h1 < n:
            grad[i] = h1[idx_h1]
            
    # Unlike in typical gradient ascent, we optimize with minimize here.
    # So we can return the negative gradient.
    # But the multiple terms contribute (if current is max_lag). So even if we use
    # gradient for max overlap, use the full sum
    # Also try sparse approximation: we need gradient for the maximum k
    
    # Actually, iteration across all k is too expensive below:
    # So only one max_overlap_idx is active, and the derivative for that specific lag along h[i] is:
    # It's the h1 shifted back by the lag
    
    # Re-compute correctly by full batch calculation:
    # Gradient at h[i] is sum of terms over lag k: (if k is max_lag) then it's h1[i+k]
    # But this is complex to compute cleanly, so do baroque recalculations.
    # Fixing by explicit calculation:
    # We fix one overlap lag k = max_lag index
    # Its gradient from h[i] = h1[i+k] if 0 <= i+k < len(h)
    
    # Simpler take the cheap approximation: one dominant position
    # and compute grad correctly using cross-correlation indexing
    # Does not need symbolic derivation:
    # P = [0 1 2] and h = [0,1,2] then 
    # P starts from lag 0, saying (start from 0 + Lag k  involves making cross product of h[:k] with 1-h[k:])
    # Overlap at lag k = h[0] * (1-h[k]) + h[1] * (1-h[k+1]) + ...
    # Grad w.r.t. h[0] = (1-h[k]), Grad w.r.t. h[1] = (1-h[k+1]), etc.
    
    # So make more accurate gradient for lag k specified
    # For all it's valid entries:
    grad = np.zeros_like(h)
    for i in range(n):
        j = i - max_lag  # This will be to h1 (which is 1-h)
        if 0 <= j < n:
            grad[i] = h1[j]
    
    # Note: we want to *maximize* overlaps, so search *gradient* direction for descent 
    # But optimization is to minimize C5 which is max of overlaps
    # This means minimize the max overlap
    # For each k, overlap is fixed. But we took the max. 
    # Gradient of that max is gradient of that one k, and we're done.
    
    return grad - np.mean(grad) # subtract mean to ensure sum of gradient is zero
