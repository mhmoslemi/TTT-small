import numpy as np
from scipy.optimize import minimize
from scipy.special import expit
from time import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Discretization settings
    n_points = 100  # Fewer points to make it fast
    dx = 2.0 / n_points
    max_iter = 500
    alpha = 0.01
    gamma = 0.8  # Elastic net L1 penalty weight

    # Basis functions: sigmoidal shape functions centered on [0,2]
    m = int(2 * np.sqrt(n_points))  # Number of basis functions
    if m < 2:
        m = 2
    mu = np.linspace(0, 2, m, endpoint=False)
    sigma = 2.0 / m

    def basis(x, i):
        return expit((x - mu[i]) / sigma)

    def h_func(a, x):
        return np.dot(a, [basis(x, i) for i in range(m)])

    def compute_overlap(a):
        # Build h from coefficients a
        h_vals = np.array([h_func(a, x) for x in np.linspace(0, 2, n_points)])
        h_vals = np.clip(h_vals, 0, 1)
        # Overlaps for all lag values
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        corr = corr * dx
        # Find max lag (shifted because of correlation)
        max_k = np.argmax(corr)
        return corr[max_k]

    def compute_grad(a):
        # Compute gradient numerically with finite difference to avoid complex derivations
        eps = 1e-5
        grad = np.zeros_like(a)
        baseline = compute_overlap(a)
        for i in range(len(a)):
            a_plus = a.copy()
            a_plus[i] += eps
            grad[i] = (compute_overlap(a_plus) - baseline) / eps
        return grad

    # Initialize coefficients from initial_h_values or random
    if 'initial_h_values' in globals():
        h_start = initial_h_values
        # Fit basis functions to initial h
        A = np.array([basis(np.linspace(0, 2, n_points), i) for i in range(m)]).T
        a = np.linalg.lstsq(A, h_start, rcond=None)[0]
    else:
        a = np.random.uniform(-0.1, 0.1, m)
        
    # Normalize to integral constraint
    h_init = np.array([h_func(a, x) for x in np.linspace(0, 2, n_points)])
    integral = np.sum(h_init) * dx
    a *= (1.0 / integral)

    best_a = a.copy()
    best_c5 = compute_overlap(a)

    # Optimization
    for iter_num in range(max_iter):
        if time() - start_time > 0.95 * budget_s:
            break
            
        grad = compute_grad(a)
        
        # Elastic net update
        a -= alpha * grad
        a -= alpha * gamma * np.sign(a)  # L1
        a -= alpha * (1 - gamma) * a      # L2

        # Project a to feasible region
        h_vals = np.array([h_func(a, x) for x in np.linspace(0, 2, n_points)])
        h_vals = np.clip(h_vals, 0, 1)
        integral = np.sum(h_vals) * dx
        a *= (1.0 / integral)

        # Check if better
        c5 = compute_overlap(a)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a.copy()
            
        # Reduce learning rate occasionally
        if iter_num > 0 and iter_num % 100 == 0:
            alpha *= 0.5

    # Final discretization and validation
    h = np.array([h_func(best_a, x) for x in np.linspace(0, 2, n_points)])
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)

    # Evaluate final performance
    corr = np.correlate(h, 1 - h, mode='full')
    corr = corr * dx
    final_c5 = np.max(corr)

    return (h, float(final_c5), n_points)
