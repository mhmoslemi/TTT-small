import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 47
        h = np.ones(n_points) * 0.5
        # Simple symmetric construction
        mid = n_points // 2
        h[:mid] = np.linspace(0.1, 0.9, mid)
        h[mid:] = np.linspace(0.9, 0.1, n_points - mid)
        h = h * (n_points / 2) / np.sum(h)
        
    dx = 2.0 / n_points
    max_iter = 3000
    lr = 0.05
    tau = 1.0
    anneal = 0.999
    tol = 1e-5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode='full')
        overlaps = overlaps[len(overlaps)//2:] * dx  # positive lags
        
        # Smooth max using log-sum-exp
        c5_approx = np.log(np.sum(np.exp(tau * overlaps))) / tau
        c5_approx = c5_approx * 1.001  # small slack to avoid numerical issues
        
        # Gradient w.r.t. h
        exp_term = np.exp(tau * overlaps)
        sum_exp = np.sum(exp_term)
        
        # Compute ∂overlap(k)/∂h[i]
        grad = np.zeros_like(h)
        for k in range(len(overlaps)):
            if k == 0:
                grad += -exp_term[k] * (1 - h) / sum_exp
            else:
                # We're computing d/dh[i] (h[i] * (1 - h[i+k]))
                # So d/dh[i] = 1 - h[i+k] if k is valid
                # But we are computing correlation: conv(h, 1-h)
                # So d/dh[i] of overlap(k) is - (1 - h[i-k])
                grad += -exp_term[k] / sum_exp * np.roll(1 - h, k)
        
        # Update step
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        
        # Project onto sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        # New evaluation
        c5 = evaluate_erdos_solution(h, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Annealing
        tau *= anneal
        if tau < 1e-5:
            tau = 1e-5
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # BFS compatible function that computes actual C5 value
    h = np.array(h)
    overlap_vals = np.correlate(h, 1 - h, mode="full")
    overlap_vals = overlap_vals[len(overlap_vals)//2:]  # Just positive lags
    return np.max(overlap_vals) * dx
