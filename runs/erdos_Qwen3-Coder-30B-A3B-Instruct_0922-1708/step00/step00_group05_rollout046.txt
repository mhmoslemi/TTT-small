import time
from scipy.optimize import minimize

    target_sum = n_points / 2.0

    a[0] = 0.5  # Integral constraint: a_0 = 1/2
    
    best_c5 = float('inf')
    best_h = None
    current_h = None
    
    def h_from_coeffs(a_coeffs):
        x = np.linspace(0, 2, n_points)
        h_vals = np.zeros(n_points)
        for i, a_val in enumerate(a_coeffs):
            h_vals += a_val * np.cos(np.pi * i * x / 2)
        return np.clip(h_vals, 0, 1)
    
    def evaluate_h(h_vec):
        # Normalize to correct integral
        h_vec = h_vec * target_sum / np.sum(h_vec)
        c5 = np.max(np.correlate(h_vec, 1 - h_vec, mode='full')) * dx
        return c5
    
    def loss_and_grad(a_coeffs):
        h_vec = h_from_coeffs(a_coeffs)
        h_vec = h_vec * target_sum / np.sum(h_vec)
        
        # Compute current C5
        c5 = evaluate_h(h_vec)
        
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(a_coeffs)
        
        # For each coefficient, compute finite difference gradient
        for i in range(len(a_coeffs)):
            a_plus = a_coeffs.copy()
            a_plus[i] += eps
            a_minus = a_coeffs.copy()
            a_minus[i] -= eps
            
            h_plus = h_from_coeffs(a_plus)
            h_plus = h_plus * target_sum / np.sum(h_plus)
            
            h_minus = h_from_coeffs(a_minus)
            h_minus = h_minus * target_sum / np.sum(h_minus)
            
            c5_plus = evaluate_h(h_plus)
            c5_minus = evaluate_h(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        return c5, grad
    
    # Optimization loop
    learning_rate = 0.01
    iter_no_improve = 0
    max_iter_no_improve = 100
    
    for iteration in range(100000):
        if time.time() - start_time > budget_s - 0.1:
            break

            best_c5 = c5

            best_h = best_h * target_sum / np.sum(best_h)
            iter_no_improve = 0

            iter_no_improve += 1
            
        if iter_no_improve > max_iter_no_improve:
            break
            
        # Update coefficients
        a -= learning_rate * grad
        a = np.clip(a, 0, 1)
        
        # Occasionally reduce learning rate
        if iteration % 100 == 0 and iteration > 0:
            learning_rate *= 0.5

        return (h_final.tolist(), final_c5, n_points)
    
    # Final validation
    final_c5 = evaluate_h(best_h)
    return (best_h.tolist(), final_c5, n_points)
