import time
from scipy import signal

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for given h."""
    # Compute cross-correlation of h with (1-h)
    corr = signal.correlate(h, 1 - h, mode="full")
    # Scale by dx to get actual integral value for each shift
    corr *= dx
    # Return the maximum value (this is C5)
    return np.max(corr)

def check_constraints(h, n_points):
    """Check if h satisfies constraints: 0 <= h[i] <= 1 and sum(h) = n_points / 2."""
    return (np.all(h >= 0) and np.all(h <= 1) and 
            abs(np.sum(h) - n_points / 2) < 1e-6)

    """Create initial population of step functions."""
    population = []
    
    # Add initial solution if available
    if initial_h is not None and len(initial_h) == n_points:
        population.append(initial_h.copy())
    
    # Generate random individuals to fill population
    while len(population) < pop_size:
        # Start with uniform vector scaled to meet sum constraint
        h = np.full(n_points, 1.0 / n_points * n_points / 2)
        # Add small random perturbations
        for i in range(n_points):
            h[i] += np.random.normal(0, 0.05)
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        # Re-normalize to meet sum constraint
        h = h * (n_points / 2) / np.sum(h)
        
        if check_constraints(h, n_points):
            population.append(h)
    
    return population

def selection_tournament(population, fitnesses, tournament_size=5):
    """Tournament selection to select parents."""
    selected = []
    for _ in range(len(population)):
        # Choose random individuals for tournament
        indices = np.random.choice(len(population), tournament_size)
        # Select the one with best fitness (lowest C5)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        selected.append(population[best_idx])
    return selected

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform single-point crossover."""
    if np.random.rand() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Ensure constraints
    child1 = child1 * (len(child1) / 2) / np.sum(child1)
    child2 = child2 * (len(child2) / 2) / np.sum(child2)
    
    # Clamp to [0,1] after normalization
    child1 = np.clip(child1, 0, 1)
    child2 = np.clip(child2, 0, 1)
    
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Apply Gaussian noise to mutate an individual."""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.rand() < mutation_rate:
            mutated[i] += np.random.normal(0, 0.05)
    # Clamp to [0,1]
    mutated = np.clip(mutated, 0, 1)
    # Re-normalize to sum constraint
    mutated = mutated * (len(mutated) / 2) / np.sum(mutated)
    return mutated
