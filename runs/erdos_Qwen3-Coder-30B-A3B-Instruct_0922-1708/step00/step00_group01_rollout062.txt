import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Helper functions
    def evaluate_c5(h):
        """Evaluate C5 using cross-correlation"""
        h = np.array(h)
        n = len(h)
        dx = 2.0 / n
        # Compute correlation of h with (1-h) 
        corr = np.correlate(h, 1-h, mode="full")
        # Take maximum over all lags
        max_corr = np.max(corr)
        return max_corr * dx

    def project_h(h, n_points):
        """Project h to satisfy constraint sum(h) = n_points/2"""
        h = np.array(h)
        current_sum = np.sum(h)
        target_sum = n_points / 2.0
        if abs(current_sum - target_sum) < 1e-10:
            return h.copy()
        # Scale to meet sum constraint
        h = h * target_sum / current_sum
        # Clamp to [0,1]
        h = np.clip(h, 0, 1)
        return h

    def mutate_h(h, noise_scale=0.05):
        """Apply mutation with Gaussian noise"""
        h = np.array(h) + np.random.normal(0, noise_scale, len(h))
        return project_h(h, len(h))

    def crossover_h(h1, h2):
        """Perform segment-wise averaging crossover"""
        h1, h2 = np.array(h1), np.array(h2)
        n = len(h1)
        # Random crossover point
        cx = np.random.randint(1, n-1)
        child = np.concatenate([h1[:cx], h2[cx:]])
        return project_h(child, n)

    def get_initial_h(n_points):
        """Get initial solution"""
        if 'initial_h_values' in globals():
            initial = np.array(initial_h_values)
            if len(initial) == n_points:
                return project_h(initial, n_points)
        # Default initialization, symmetric plateau
        h = np.ones(n_points) * 0.5
        # Make it symmetric and taper to 0 at ends
        mid = n_points // 2
        steps = n_points // 4
        for i in range(steps):
            factor = float(i) / steps
            h[i] = 0.1 + 0.4 * factor
            h[n_points - 1 - i] = 0.1 + 0.4 * factor
        return project_h(h, n_points)

    # Begin optimization
    start_time = time.time()
    n_points = 47  # Use the parent's length if available
    pop_size = 100
    max_gen = 1000
    noise_scale = 0.05
    crossover_rate = 0.8
    mutation_rate = 0.1
    tournament_size = 20
    simplex_size = 10

    # Initialize population
    population = [get_initial_h(n_points)]
    while len(population) < pop_size:
        h = mutate_h(population[0], noise_scale)
        population.append(h)

    best_h = None
    best_c5 = float('inf')

    for gen in range(max_gen):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_c5(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]

        # Select parents for reproduction
        selected = []
        while len(selected) < pop_size and len(selected) < len(population):
            tournament = np.random.choice(len(population), tournament_size, replace=False)
            winner = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(winner)

        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - len(selected):
            if np.random.random() < crossover_rate and len(selected) >= 2:
                i1, i2 = np.random.choice(selected, 2, replace=False)
                child = crossover_h(population[i1], population[i2])
            else:
                i = np.random.choice(selected)
                child = mutate_h(population[i], noise_scale)
            offspring.append(child)

        # Nelder-Mead refinement
        refs = []
        for i in range(min(simplex_size, len(offspring))):
            h = offspring[i]
            # Wrap for scipy minimize
            def obj(x):
                return evaluate_c5(x)
            # Use Nelder-Mead
            result = minimize(obj, h, method='Nelder-Mead', options={'maxiter': 100})
            if result.success:
                offspring[i] = project_h(result.x, n_points)
                refs.append(result.x)

        # Reconstruct population
        new_pop = population[:len(selected)] + offspring
        population = new_pop

    if best_h is None:
        best_h = get_initial_h(n_points)

    return (list(best_h), best_c5, n_points)
