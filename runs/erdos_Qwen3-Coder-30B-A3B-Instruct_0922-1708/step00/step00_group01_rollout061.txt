import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time
import random

def evaluate_h(h, dx):
    # Compute C5 as max of cross-correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project_to_feasible(h, n_points, dx):
    # Ensure h is within [0,1] and sum(h)*dx = 1
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h) * dx
    if abs(current_sum - 1.0) > 1e-10:
        # Rescale to meet sum constraint
        h = h * (1.0 / current_sum)
    return h

def crossover(parent1, parent2, rate=0.8):
    if random.random() > rate:
        return parent1.copy(), parent2.copy()
    # Simple segment-wise averaging
    n = len(parent1)
    crossover_point = random.randint(1, n-1)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    return child1, child2

def mutate(h, noise_scale=0.05):
    # Gaussian noise mutation
    noise = np.random.normal(0, noise_scale, len(h))
    h_m = h + noise
    return project_to_feasible(h_m, len(h), 2.0/len(h))

def local_optimize(h, dx, maxiter=200):
    # Use Nelder-Mead on the objective
    n_points = len(h)
    def obj(h_flat):
        h_test = h_flat.reshape(-1)
        return evaluate_h(h_test, dx)
    
    # Flatten
    h_flat = h.flatten()
    # Use scipy minimize with Nelder-Mead
    res = minimize(obj, h_flat, method='Nelder-Mead', options={'maxiter': maxiter, 'disp': False})
    if res.success:
        h_opt = res.x.reshape(h.shape)
        return project_to_feasible(h_opt, n_points, dx)
    else:
        return h  # Return original if failed
