import numpy as np
import cvxpy as cvx
from time import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time()
    
    # Get n_points from initial_h_values if available, otherwise default
    try:
        n_points = len(initial_h_values)
    except:
        n_points = 70
    
    # Initialize
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.full(n_points, 0.5 * (2 / n_points))

    # Variables
    x = cvx.Variable(n_points)
    z = cvx.Variable((n_points, n_points))
    t = cvx.Variable()

    # Constraints
    constraints = [
        x >= 0,
        x <= 1,
        cvx.sum(x) == n_points / 2,
        t >= 0
    ]
    
    # Add z constraints
    for i in range(n_points):
        for j in range(n_points):
            constraints += [
                z[i,j] <= x[i],
                z[i,j] <= 1 - x[j],
                z[i,j] >= x[i] + x[j] - 1
            ]

    # Compute correlations
    C = []
    for k in range(-n_points+1, n_points):
        idx = np.arange(n_points) - k
        valid = (idx >=0) & (idx < n_points)
        sum_z = cvx.sum(z[np.arange(n_points)[valid], idx[valid]])
        C.append(sum_z)
    C = cvx.vstack(C)
    
    # Add max correlation constraint
    constraints += [cvx.max(C) <= t]

    # Objective
    objective = cvx.Minimize(t)

    # Setup problem
    prob = cvx.Problem(objective, constraints)
    
    # Solve
    try:
        prob.solve(solver=cvx.MOSEK, verbose=False, time_limit=budget_s)
        if prob.status in [cvx.OPTIMAL, cvx.OPTIMAL_INACCURATE]:
            h_sol = np.array(x.value).flatten()
            h_sol = np.clip(h_sol, 0, 1)
            h_sol = h_sol / np.sum(h_sol) * (n_points / 2)
        else:
            h_sol = h0
    except:
        h_sol = h0

    # Final evaluation
    c5 = evaluate_erdos_solution(h_sol)
    return (h_sol, c5, n_points)
