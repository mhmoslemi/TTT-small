import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize sum to exactly n_points / 2
    dx = 2.0 / n_points
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    
    # Time tracking
    start_time = time.time()
    best_h = h.copy()
    best_C = float('inf')
    
    # Parameters
    max_iter = 6000
    eta = 0.01
    patience = 0
    patience_limit = 100
    adaptive_interval = 100
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute all overlaps
        # We compute correlation of h with (1-h) to get overlaps for all shifts
        h_reflected = 1 - h
        correlations = signal.correlate(h_reflected, h, mode='full')
        # Corresponding indices:
        # k in [0, n_points - 1] corresponds to shift where first sequence is shifted
        # forward by k positions (i.e., h[i] -> h[i+k]). 
        # Negatives correspond to reversed indexing.
        # So we need to extract the relevant part for k in [0, n_points-1]
        # But we also need to consider shift k in [0, 2*n_points - 2] for full correlation.
        # For our domain [0,2), there are overlaps for all lag values where we align
        # h[x] with h[x+k], giving negative k, we do not consider these.
        # Actually, for h(x) in [0,2] we care about overlaps over [0,2], i.e. max lag=2,
        # so shift from 0 to 2 corresponds with shift from 0 to 2*n_points/2 = n_points.

        # Overlaps considered for lag k = 0 to n_points-1
        # k=0 when we compute h[i]*(1-h[i]) for all i
        # ... this is simple correlation of [h, 1-h] with itself.
        # But the code we're told evaluates with np.correlate(h, 1-h, mode="full") * dx
        # This returns a full correlation of length 2*n_points - 1
        # Let's use this exactly
        
        # The full correlation gives result of length 2*n_points - 1
        full_correlation = correlations * dx
        # Extract max overlap across all valid lags
        # The actual range of lags depends on our step function definition.
        # Let the first lags (from the left) be a negative shift
        # The middle ones are zero shift, plus positive shift.
        # But, for overlapping integration, we consider k = shift in first argument:
        # if we have h(x) and integrate h(x) * (1 - h(x+k))dx over domain [0,2],
        # then the overlapping integral can be evaluated at different lags k.
        # In the context of the evaluation function (from spec):
        # max_k integral h(x)(1 - h(x+k)) dx
        # This is valid only if for a given k, we have:
        # x in [0, 2] and x+k in [0, 2] => k in [-2, 2]
        # That means, for the discretized grid, with dx = 2/n,
        # evenly spaced samples at x[i] = i*dx = i * 2/n
        # To get x+k = j*dx  => k must map to index difference j-i
        # with valid indices in [0, n)
        # Maximum shift is 2 in (assuming finite support in [0,2])
        # This corresponds to k = 2 or k = -2, that's about 2/dx = n samples.
        # The motion should go from left shift (u_i = h_i * (1-h_{i+n-1}) = shift -2)
        # to k=0 for u_i = h_i * (1-h_i), up to k = 2 (u_i = h_i * (1-h_{i+1})
        # Actual usage: a full compute gives full array where index i in results
        # corresponds to shift of -n+1 to n-1.
        # So for shift k: index = k + (n - 1)
        # Thus we consider index range: 0 to 2*(n - 1)

        # So shift k=0 to 2 maps to index 0 to 2*(n-1) where index i+ (n-1) = k

        # index 0 to (0 + (n-1)) to (2 + (n-1)) → 0 to n + 1

        # Overlaps from k = 0 to n-1: include indices (n-1) to (n-1)+n = (2n-2)
        # This is for shift k = 0 to n-1 
        # But in integral context, use only for shift from 0 up to n_points
        # largest shift that makes sense is k=2 where shift = 2
        # => the last entry index (2n-2) is shift k=2 where k = 2*dx = 2*2/n = 4/n
        # This would mean for dx = 2/n, k =  2 dx = 4/n = 2 (i.e. shift 2 = 2 * dx)
        # So shift k = 0 to n-1, spans the largest valid shift k = 0 to 2,
        # which is logistically equal shift k=0 to n_points*2/2 = n_points.

        # Let's extract from center of full correlation:
        center = len(full_correlation) // 2
        n_off = n_points - 1
        # extract the indices that represent lags 0 to n_points (but it's still from -n_points to n_points)
        # Since we only care about overlaps coming from lag k=0 to 2 and that corresponds to shifts 
        # from indices center - n_points to center + n_points
        # Actually let's pay attention again
		
        # Use full correlation for shift 0 to 2 deduced from indices n_points-1 to n_points-1 + 2*n_points - 1
        # But that's not what we want. Let’s compute properly.

        # Valid lags are k in range [0,2], i.e. from -2 to 2.
        # For n equally spaced points over [0,2], dx = 2/n
        # Then the shift in samples is k/dx = k*n/2. 
        # So k in [-2,2] => shift in samples from -n_points to n_points.
        # So correct indices for lags 0 to 2: shift right, indices in [center-n_points, center+n_points]
        # Take all valid overlaps (i.e., within |k| <= 2)
        # We are computing max_k offsets, this gives indices [0, 2n - 2] indices.
        # We need to extract for lags 0 to 2, so for k from 0 to n_points samples,
        # to properly compute: k from -2 to 2 
        # i.e. indices from 0 to 2n-2 for which we extract for shift k from 0 to 2
        # Our full_correlation are indices 0 to 2n - 2 where index i corresponds to lag k = (i - (n - 1)) * dx
        
        # So corrected:
        k_bn = np.arange(len(full_correlation))
        k_vals = (k_bn - (n_points - 1)) * 2 / n_points  # Map indices to actual k values
        
        # Bounds:
        valid_mask = (k_vals >= 0.) & (k_vals <= 2.)  # Consider lags from 0 to 2
        # get those overlaps
        overlaps = full_correlation[valid_mask]
        
        # Now we want C = max_k overlaps
        max_k_index = np.argmax(overlaps)
        k_of_max_overlap = k_vals[valid_mask][max_k_index]
        C = overlaps[max_k_index]
        
        # Store the best solution
        if C < best_C:
            best_C = C
            best_h = h.copy()
            patience = 0
        else:
            patience += 1

        # If we get stuck with no improvement for 100 steps, break
        if patience >= patience_limit:
            break

        # If we update using h_i -> h_i - eta * gradient

        # So we compute vector of partial derivatives few elements
        # The gradient is not global, but based on the max k shift case.

        max_k_index_full = valid_mask.nonzero()[0][max_k_index]

        # based on how local computation with k affects it:
        # h_i * (1 - h_{i+k}) 
        # Gradient w.r.t h_i = (1 - h_{i+k})
        # We compute h like h -> h - eta * gradient
        gradient = np.zeros(n_points)
        
        # If we use [h, 1-h] correlation:
        # for the full correlation between (1-h) and h with inverted indexing,
        # We get overlaps, but isolate for that specific shift k = (k_bn - (n-1)) * 2/n

        # partial derivative of integral over x of h(x) * (1 - h(x+k)) dx
        # w.r.t. h(x) is just (1-h(x+k))
        # Since we are in discrete case:
        # For each point i, we compute: contribution is h_i * (1 - h_{i+k})
        # Look at d/dh_i of this = (1 - h_{i+k})
        # So we compose:
        # i -> gradient[i] = (1 - h_{i+k})
        # But out of bounds -> we fill with 0 or account for it carefully
        # Gradient from index `k` in shifted array
        # shift by +k, so:
        # gradient[i] = 1 - h[i+k] if i+k < n, else = 0

        # Equivalent equivalent approach:
        # Build a vector that represents gradient for the kth lag
        # that is (1 - h[i+k]) when *both* are in range
        # But what about x+k lands outside [0,2]? That’s either 1-k or k+2. 
        # But 
        # Let's be very careful with indices

        # Let’s think differently and see compute_brute_force_overlap
        # but we must follow same evaluation as in the function
        # Need index-shift from correlation with proper boundaries
        # The overlapping term becomes h[i] * (1 - h[i + k])
        # From full correlation, i_index corresponding to k breach have k_indices so:
        # center = n_points-1
        # k = (i - (n_points-1)) * (2/n)
        # => i = k*(n/2) + (n_points-1)
        # We are computing correlation between (1 − h) and h
        # The index in full correlations corresponds:
        # K at index i => shift k = (i - center) * dx = (i - (n_points-1)) * 2/n  
        # Memory:
        # Full correlation is of length 2n-1
        # So full_corr indices [0, 2n-2] 
        # correlation for shift k = 0 to n_points-1
        # indices in [n_points-1, n_points-1 + (n_points-1)] = [n_points-1, 2*(n_points-1)]
        # But we only care about lags 0 to 2 (i.e. k=0 to k=2)
        # => indices range accordingly. There are 2 * n_points - 1 points,
        # so indices from 0 to 2n - 2, mapping to shift from -2 to +2 
        # But what is the interpretation of such integral?
        # Deviations from the equation can be due to interpretation details
        # In general,
        # max_k integral h(x) (1 - h(x+k)) dx -> where 0 <= x <= 2, 0 <= x+k <= 2
        # So k in [-2,2]
        # Shift k maps into scaled integer index.
        
        # This is complexity ... use the correct strategy from initial implementation

        # Go back: using generate overlap integral ordering manually
        # Need: for a specific lag `k` choose:
        # This lag k corresponds to a shift in the full correlation
        # We simulate overlap computations directly now:
        # lag k means shift 1st array by k to match 2nd array
        # (h_i)_j = h[i+k], matching h_i * (1 - h_{i+k})
        
        k_int = int(round(k_of_max_overlap * n_points / 2)) 
        # But check k_int is within [0,n_points-1]
        if k_int < 0:
            k_int = 0
        elif k_int >= n_points:
            k_int = n_points - 1
            
        # Fill gradient w.r.t current max_k lag
        for i in range(n_points):
            # If i + k_int < n_points, then gradient[i] = 1 - h[i + k_int]
            # Else depends on only current document interpretation
            # updated: gradient[i] = 1 - h[i + k_int] unless k_int == 0
            if i + k_int < n_points:
                gradient[i] = 1 - h[i + k_int]
            else:
                gradient[i] = 0
                
        # Update h
        h = h - eta * gradient
        
        # Clip h to [0,1]
        np.clip(h, 0, 1, out=h)
        
        # Enforce sum constraint
        # Adjust so that sum equals n_points / 2 by redistribution
        current_sum = np.sum(h)
        desired_sum = n_points / 2
        # If not exact change, normally due to clipping, redistribute
        diff = desired_sum - current_sum
        
        if abs(diff) > 1e-10: # sum is off, redistribute
            # Scale down all values by small amount that will bring sum closer
            scale_factor = desired_sum / current_sum
            h[:] = h * scale_factor
        
        # After scaling, might go out of bounds again, keep bounds
        np.clip(h, 0, 1, out=h)
        
        # And again ensure final sum is exact
        current_sum = np.sum(h)
        if abs(current_sum - desired_sum) > 1e-10:
            # Just in case, reset final sum
            diff = desired_sum - current_sum
            h[0] += diff
        
        # Adjust learning rate
        if iter_count % adaptive_interval == 0:
            eta = max(1e-6, eta * 0.95)
            
    # Final validation and return best solution
    correlations = signal.correlate(1 - best_h, best_h, mode='full')
    full_correlation = correlations * dx
    center = len(full_correlation) // 2
    k_vals = (np.arange(len(full_correlation)) - (n_points - 1)) * 2 / n_points
    valid_mask = (k_vals >= 0.) & (k_vals <= 2.)
    overlaps = full_correlation[valid_mask]
    final_C = np.max(overlaps)
    return (best_h.tolist(), final_C, n_points)
