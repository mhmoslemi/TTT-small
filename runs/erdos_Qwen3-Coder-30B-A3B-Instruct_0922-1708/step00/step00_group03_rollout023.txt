import numpy as np
import time
from scipy.signal import correlate

def evaluate_erdos_solution(h):
    # This function evaluates the C5 constant for a given h
    # h should be a numpy array of values in [0,1]
    # Return the maximum overlap integral
    n = len(h)
    dx = 2.0 / n
    corr = correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr)
    return max_corr * dx

def rescale_h(h, target_sum):
    """Rescale h to have a specific sum."""
    current_sum = np.sum(h)
    if np.abs(current_sum - target_sum) < 1e-10:
        return h.copy()
    alpha = target_sum / current_sum
    return h * alpha

def initialize_population(n_points, num_individuals, initial_h):
    """Initialize a population of step functions."""
    population = []
    # Add the initial solution
    population.append(initial_h.copy())
    # Add random individuals
    target_sum = n_points / 2.0
    for _ in range(num_individuals - 1):
        h = np.random.random(n_points)
        h = rescale_h(h, target_sum)
        population.append(h)
    return population

def tournament_selection(population, fitnesses, k=5):
    """Select an individual using tournament selection."""
    tournament_indices = np.random.choice(len(population), k, replace=False)
    tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
    winner_index = tournament_indices[np.argmin(tournament_fitnesses)]
    return population[winner_index].copy()

def crossover(parent1, parent2):
    """Perform segment swap crossover."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, delta=0.1):
    """Mutate a step function."""
    n = len(h)
    if np.random.random() > mutation_rate:
        return h.copy()
    # Select points to mutate (about 10%)
    num_mutate = max(1, int(0.1 * n))
    indices = np.random.choice(n, num_mutate, replace=False)
    h_mut = h.copy()
    for i in indices:
        h_mut[i] += np.random.uniform(-delta, delta)
    h_mut = np.clip(h_mut, 0, 1)
    # Keep the sum constraint
    target_sum = len(h) / 2
    h_mut = rescale_h(h_mut, target_sum)
    return h_mut

    best_h = None
    best_c5 = float('inf')
    
    # Track time
    start_time = time.time()
    last_improvement_gen = 0
    generation = 0

    while generation < max_generations:
        elapsed = time.time() - start_time
        if elapsed > budget_s - 30:  # Leave 30s buffer
            break
            
        # Evaluate population
        fitnesses = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitnesses.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                last_improvement_gen = generation
        
        # Check for early exit
        if generation - last_improvement_gen > 50:
            break
            
        # Selection and evolution
        new_population = []
        
        # Elitism (keep best)
        sorted_indices = np.argsort(fitnesses)
        for i in range(pop_size // 4):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate rest via selection and crossover
        while len(new_population) < pop_size:
            p1 = tournament_selection(population, fitnesses)
            p2 = tournament_selection(population, fitnesses)
            c1, c2 = crossover(p1, p2)
            c1 = mutate(c1)
            c2 = mutate(c2)
            new_population.extend([c1, c2])
        
        # Trim to population size
        population = new_population[:pop_size]
        generation += 1
    
    return (best_h, best_c5, n_points)

def run(seed=42, budget_s=600, **kwargs):
    h_values, c5_bound, n_points = main_ga(seed, budget_s)
    return (h_values, c5_bound, n_points)
