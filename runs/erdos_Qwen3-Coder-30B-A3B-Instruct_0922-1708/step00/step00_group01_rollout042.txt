import numpy as np
import time
from scipy.signal import correlate

def evaluate(h, dx):
    """Evaluate the overlap integral for the given step function h."""
    # Compute correlation of h with (1-h) in full mode
    corr = correlate(h, 1 - h, mode="full")
    # Maximum of correlation values scaled by dx
    max_corr = np.max(corr) * dx
    return max_corr

def project(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points / 2
    scale = (n_points / 2) / np.sum(h)
    h = h * scale
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create random one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 100
        # Create random initial solution
        h0 = np.random.rand(n_points)
        h0 = project(h0, n_points)
    
    # Parameters for PSO
    n_particles = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    inertia_weight = w
    best_global = float('inf')
    best_global_h = None
    
    # Initialize particles randomly around the initial solution
    particles = []
    velocities = []
    personal_best = []
    personal_best_values = []
    for i in range(n_particles):
        # Start from random perturbation of h0
        h = h0.copy()
        if i > 0:  # Perturb only for particle > 0
            noise = np.random.normal(0, 0.05, n_points)
            h += noise
        h = project(h, n_points)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
        # Evaluate personal best
        val = evaluate(h, 2.0 / n_points)
        personal_best.append(h.copy())
        personal_best_values.append(val)
        if val < best_global:
            best_global = val
            best_global_h = h.copy()
    
    iter_no_improve = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            p_best = personal_best[i]
            g_best = best_global_h
            
            v = (inertia_weight * v + 
                 c1 * r1 * (p_best - h) + 
                 c2 * r2 * (g_best - h))
            
            # Update position
            h_new = h + v
            
            # Project new position
            h_new = project(h_new, n_points)
            
            # Evaluate
            val_new = evaluate(h_new, 2.0 / n_points)
            
            # Update personal best if improved
            if val_new < personal_best_values[i]:
                personal_best[i] = h_new.copy()
                personal_best_values[i] = val_new
                if val_new < best_global:
                    best_global = val_new
                    best_global_h = h_new.copy()
                    iter_no_improve = 0
                else:
                    iter_no_improve += 1
            
            # Update position and velocity
            particles[i] = h_new
            velocities[i] = v
        
        # Diversity injection if no improvement
        if iter_no_improve > 25:
            for i in range(int(0.2 * n_particles)):
                idx = np.random.randint(0, n_particles)
                # Reset particle with new random solution
                h_new = np.random.rand(n_points)
                h_new = project(h_new, n_points)
                particles[idx] = h_new
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
                val_new = evaluate(h_new, 2.0 / n_points)
                personal_best[idx] = h_new
                personal_best_values[idx] = val_new
                if val_new < best_global:
                    best_global = val_new
                    best_global_h = h_new
            iter_no_improve = 0
    
    return (best_global_h.tolist(), best_global, n_points)
