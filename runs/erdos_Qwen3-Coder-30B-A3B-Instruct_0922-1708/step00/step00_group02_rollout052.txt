import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use provided initial construction if available, otherwise create a uniform one

        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5  # uniform initial guess
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def project_h(h_vec):
        """Project h onto constraints: [0,1] and correct sum"""
        h_vec = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_vec)
        if current_sum > 0:
            h_vec = h_vec * target_sum / current_sum
        return h_vec
    
    def evaluate_objective(h_vec):
        """Compute C5 = max_k int h(x)(1-h(x+k)) dx"""
        # For efficiency, compute full correlation
        corr = np.correlate(h_vec, 1 - h_vec, mode="full")
        # Select the relevant portion for shifts in [0,2]
        # The valid range corresponds to lags in [-n_points+1, n_points-1]
        # But we only care about lag from 0 to 2, i.e., from 0 to n_points-1
        # Actually, we want the full autocorrelation in terms of shift k
        # Full correlation is of length 2*n_points - 1
        # Center element corresponds to lag=0
        # For our case, we want the maximum value across all valid lags
        # This is the maximum of the correlation array, scaled by dx
        # The correlation is already scaled properly for dx=1, so multiply by dx
        max_corr = np.max(corr)
        return max_corr * dx
    
    def gradient_step(h_vec, step_size=0.01):
        """Compute a simple gradient step using finite differences"""
        # Compute current objective value
        current_obj = evaluate_objective(h_vec)
        # Approximate gradient by finite differences
        grad = np.zeros_like(h_vec)
        eps = 1e-5
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            obj_plus = evaluate_objective(h_plus)
            
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            obj_minus = evaluate_objective(h_minus)
            
            grad[i] = (obj_plus - obj_minus) / (2 * eps)
        
        # Update h
        new_h = h_vec - step_size * grad
        new_h = project_h(new_h)
        return new_h
    
    best_h = h.copy()
    best_c5 = evaluate_objective(best_h)
    
    # Start optimization loop
    iter_count = 0
    step_size = 0.01
    no_improvement_count = 0
    max_no_improvement = 100
    last_best = best_c5
    
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Perform a few gradient steps
        for _ in range(10):
            new_h = gradient_step(best_h, step_size)
            new_c5 = evaluate_objective(new_h)
            if new_c5 < best_c5 - 1e-10:
                best_c5 = new_c5
                best_h = new_h.copy()
                no_improvement_count = 0
                last_best = best_c5
            else:
                no_improvement_count += 1
                if no_improvement_count > max_no_improvement:
                    step_size *= 0.5
                    no_improvement_count = 0
        
        # If no improvement in a while, reduce step size
        if no_improvement_count > max_no_improvement // 2:
            step_size *= 0.9
            
        # Break if quit criterion met
        if step_size < 1e-6:
            break
    
    # Final project to ensure constraints
    final_h = project_h(best_h)
    final_c5 = evaluate_objective(final_h)
    
    # Return best solution found
    return (final_h.tolist(), final_c5, n_points)
