import numpy as np
import time
from scipy.optimize import minimize
from scipy.spatial.distance import cdist

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    # Normalize initial h to have sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    # Optimization parameters
    alpha = 0.01
    max_iter = 10000
    patience = 100
    best_c5 = float('inf')
    best_h = h.copy()
    
    def compute_c5(h):
        # Compute all overlaps using correlation
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlaps = np.correlate(h_padded, h_reflected, mode='full')
        max_overlap = np.max(overlaps[n_points-1:])
        return max_overlap * (2.0 / n_points)  # dx = 2/n
    
    def compute_grad(h, k_max):
        # Gradient w.r.t. to shift k_max
        grad = np.zeros_like(h)
        for i in range(len(h)):
            if i + k_max < len(h):
                grad[i] += (1 - h[i + k_max])
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]
        return grad
    
    iter_count = 0
    last_improvement = 0
    
    while iter_count < max_iter and time.time() - start_time < budget_s - 1:
        # Compute all overlaps and find max
        h_padded = np.concatenate([h, np.zeros(n_points)])
        h_reflected = np.concatenate([np.zeros(n_points), 1 - h])
        overlaps = np.correlate(h_padded, h_reflected, mode='full')
        max_k = np.argmax(overlaps[n_points-1:]) + (n_points-1)
        
        # Compute gradient for this k
        grad = compute_grad(h, max_k - (n_points-1))
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new C5
        new_c5 = compute_c5(h_new)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new.copy()
            last_improvement = iter_count
        
        # Update
        h = h_new
        
        iter_count += 1
        
        # Early stopping
        if iter_count - last_improvement > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
