import numpy as np
import time
from scipy import signal

def evaluate_h(h_values, n_points):
    dx = 2.0 / n_points
    # Compute overlap integral for all shifts
    overlap = np.correlate(h_values, 1 - h_values, mode='full')
    # Multiply by dx to get actual overlap values
    overlap *= dx
    # Return the maximum overlap
    return np.max(overlap)

def initialize_population(n_points, pop_size, initial_h_values=None):
    population = []
    if initial_h_values is not None:
        population.append(initial_h_values.copy())
        pop_size -= 1
    
    for _ in range(pop_size):
        # Random initialization
        h = np.random.rand(n_points)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        population.append(h)
        
    return population

def mutate(h, mutation_rate):
    # Apply small random perturbations
    mask = np.random.rand(len(h)) < mutation_rate
    h = h + np.random.normal(0, 0.05, len(h)) * mask
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Re-normalize to sum to n_points / 2
    n_points = len(h)
    h = h * (n_points / 2) / np.sum(h)
    return h

def crossover(p1, p2):
    n_points = len(p1)
    # Single-point crossover
    crossover_point = np.random.randint(1, n_points)
    offspring1 = np.concatenate([p1[:crossover_point], p2[crossover_point:]])
    offspring2 = np.concatenate([p2[:crossover_point], p1[crossover_point:]])
    
    # Normalize to sum to n_points / 2
    offspring1 = offspring1 * (n_points / 2) / np.sum(offspring1)
    offspring2 = offspring2 * (n_points / 2) / np.sum(offspring2)
    
    # Clip to [0,1]
    offspring1 = np.clip(offspring1, 0, 1)
    offspring2 = np.clip(offspring2, 0, 1)
    
    return offspring1, offspring2

def select_parents(population, fitnesses, tournament_size=5):
    tournament_indices = np.random.choice(len(population), 
                                         size=(len(population), tournament_size), 
                                         replace=False)
    # Winner of each tournament is the one with lowest fitness
    winners = np.argmin([np.array(fitnesses)[indices].min() for indices in tournament_indices])
    return [population[i] for i in winners]

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 45
    pop_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    elite_size = pop_size // 5
    max_no_improve = 100
    no_improve = 0
    
    # Initialize population
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    # Evaluate initial population and track best
    fitnesses = [evaluate_h(h, n_points) for h in population]
    best_idx = np.argmin(fitnesses)
    best_h = population[best_idx].copy()
    best_fitness = fitnesses[best_idx]
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s:
        generation += 1
        # Sort population by fitness (lowest to highest)
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        elites = sorted_population[:elite_size]
        
        # Generate offspring
        offspring = []
        for i in range(0, pop_size - elite_size, 2):
            p1_idx = np.random.randint(0, pop_size - elite_size)
            p2_idx = np.random.randint(0, pop_size - elite_size)
            
            if np.random.rand() < crossover_prob:
                o1, o2 = crossover(sorted_population[p1_idx], sorted_population[p2_idx])
            else:
                o1, o2 = sorted_population[p1_idx].copy(), sorted_population[p2_idx].copy()
            
            # Mutate
            o1 = mutate(o1, mutation_rate)
            o2 = mutate(o2, mutation_rate)
            
            offspring.extend([o1, o2])
            
        # Ensure we have the right population size
        if len(offspring) > pop_size - elite_size:
            offspring = offspring[:pop_size - elite_size]
        elif len(offspring) < pop_size - elite_size:
            # If not enough offspring, add random individuals to fill
            extra = pop_size - elite_size - len(offspring)
            for _ in range(extra):
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
                offspring.append(h)
        
        # New population is elites + offspring
        population = elites + offspring
        
        # Evaluate new fitnesses
        fitnesses = [evaluate_h(h, n_points) for h in population]
        curr_best_idx = np.argmin(fitnesses)
        curr_best = population[curr_best_idx]
        curr_fitness = fitnesses[curr_best_idx]
        
        if curr_fitness < best_fitness:
            best_h = curr_best.copy()
            best_fitness = curr_fitness
            no_improve = 0
        else:
            no_improve += 1
            
        # Early stopping if no improvement for a while
        if no_improve > max_no_improve:
            break
            
    # Final validation
    final_fitness = evaluate_h(best_h, n_points)
    return (best_h, final_fitness, n_points)
