def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for given step function h."""
    n_points = len(h)

    # Compute all overlaps for lags 0 to 2*dx*n_points
    # Using full convolution
    h_rev = np.flip(h)
    overlaps = np.convolve(h, 1 - h_rev, mode='full')
    # Only consider the valid lags - smaller than n_points
    overlaps = overlaps[n_points-1:2*n_points-1]
    return np.max(overlaps) * dx
