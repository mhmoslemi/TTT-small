import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 70
    dx = 2.0 / n_points
    max_iter = 6000
    eta = 0.01
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        h = h * (n_points / 2) / np.sum(h)  # Normalize to sum to n_points/2
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure initial constraint
    h_sum = np.sum(h)
    if abs(h_sum - n_points / 2) > 1e-10:
        h = h * (n_points / 2) / h_sum
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find the worst lag k
        correlations = np.correlate(h, 1 - h, mode='full')
        k_indices = np.arange(-n_points + 1, n_points)
        max_k_idx = np.argmax(correlations)
        max_k = k_indices[max_k_idx]
        
        # Compute gradient for this k
        grad = np.zeros(n_points)
        if max_k >= 0:
            grad[max_k:] = (1 - h[:-max_k]) * dx
        else:
            grad[:max_k] = -h[-max_k:] * dx
            
        # Update h
        h_new = h - eta * grad
        
        # Clip to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Adjust to maintain sum constraint
        h_sum_new = np.sum(h_new)
        if abs(h_sum_new - n_points / 2) > 1e-10:
            h_new = h_new * (n_points / 2) / h_sum_new
        
        # Compute new C5
        c5_new = compute_c5(h_new, dx)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h = h_new
        
        # Adaptive learning rate
        if i % 100 == 0 or (i % 50 == 0 and i > 0):
            eta *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    corr = np.correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx
