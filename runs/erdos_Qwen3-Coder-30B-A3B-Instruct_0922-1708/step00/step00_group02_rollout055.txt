import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 10000
    alpha = 0.01
    tolerance = 1e-6
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    h *= (n_points / 2) / np.sum(h)
    
    best_c5 = float('inf')
    best_h = h.copy()
    
    start_time = time.time()
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap integrals for all shifts
        overlap = np.correlate(h, 1 - h, mode='full')
        shifted_overlap = overlap[len(h)-1:]
        c5 = np.max(shifted_overlap) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Find the shift that causes maximum overlap
        max_shift_idx = np.argmax(shifted_overlap)
        max_shift = max_shift_idx - (n_points - 1)
        
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            j = i + max_shift
            if 0 <= j < n_points:
                grad[i] += (1 - h[j]) - h[i]
        
        # Update h
        h = h - alpha * grad
        
        # Project h to [0, 1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain constraint sum(h) = n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        
    return (best_h.tolist(), float(best_c5), n_points)
