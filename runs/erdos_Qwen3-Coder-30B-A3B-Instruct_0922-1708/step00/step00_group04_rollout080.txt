import numpy as np
from scipy import optimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize h from initial_h_values if available, else create a smooth starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        # Create a smooth, symmetric function that sums to 1 over [0,2]
        n_points = 42
        x = np.linspace(0, 2, n_points)
        h = np.clip(0.5 + 0.3 * np.sin(np.pi * x), 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # rescale to sum to n_points/2
    
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Helper function to compute the C5 bound
    def compute_c5(h_vec):
        h_vec = np.array(h_vec)
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    # Surrogate for the max to make it differentiable
    def surrogate_max(arr, beta=100):
        # log-sum-exp approximation to the max
        return np.log(np.sum(np.exp(beta * arr))) / beta
    
    # Gradient computation function
    def compute_gradient(h_vec):
        # This is a simplified gradient approximation
        # In detail, we need to compute Jacobian of the correlation
        # We approximate with finite differences
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_c5 = compute_c5(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps if h_vec[i] < 1 else -eps
            if h_plus[i] < 0:
                h_plus[i] = 0
            if h_plus[i] > 1:
                h_plus[i] = 1
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        return grad
    
    # Optimization loop
    best_h = h.copy()
    best_c5 = compute_c5(h)
    last_improvement = 0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Alternate small random perturbations with gradient descent to avoid local optima
        if iteration % 10 == 0:
            # Random perturbation
            perturbation = np.random.normal(0, 0.01, n_points)
            h_new = h + perturbation
            h_new = np.clip(h_new, 0, 1)
            # Rescale to fix sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        else:
            # Gradient descent step
            grad = compute_gradient(h)
            step = 0.1 / (1 + iteration // 100)
            h_new = h - step * grad
            h_new = np.clip(h_new, 0, 1)
            # Rescale to fix sum
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        h = h_new
        
        # Early stopping
        if best_c5 < 0.3807 or last_improvement > 100:
            break
            
    return (best_h.tolist(), best_c5, n_points)
