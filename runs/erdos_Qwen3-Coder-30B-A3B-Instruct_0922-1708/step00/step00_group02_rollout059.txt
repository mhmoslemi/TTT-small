import numpy as np
import time
from scipy import optimize
from cvxpy import *

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 45
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.ones(n_points) * 0.5
    
    # Ensure sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    # Gradient-based optimization
    max_iter = 10000
    alpha = 0.01
    patience = 100
    no_improve_count = 0
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps
        overlaps = np.correlate(h, 1 - h, mode="full")
        k_max = np.argmax(overlaps)
        max_overlap = overlaps[k_max]
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count > patience:
            break
            
        # Compute gradient
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Contribution from h[i] to overlap at shift k_max
            if i + k_max < n_points:
                grad[i] += 1 - h[i + k_max]
            if i - k_max >= 0:
                grad[i] -= h[i - k_max]
        
        # Update h
        h = h - alpha * grad
        
        # Project onto [0,1]
        h = np.clip(h, 0, 1)
        
        # Rescale to maintain sum
        h = h * (n_points / 2) / np.sum(h)
    
    # Final validation
    final_c5 = compute_c5(best_h, dx)
    
    return (best_h.tolist(), final_c5 * dx, n_points)

def compute_c5(h, dx):
    # Compute overlaps
    overlaps = np.correlate(h, 1 - h, mode="full")
    return np.max(overlaps) * dx
