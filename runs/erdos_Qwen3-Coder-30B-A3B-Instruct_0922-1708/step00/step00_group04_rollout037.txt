import numpy as np
from scipy.signal import correlate
import time
from cvxpy import *
import cvxpy as cp

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C_5 for a given step function h"""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integrals for all lags
    corr = correlate(h, 1 - h, mode="full")
    # The overlap integral for lag k is corr[n-1+k] * dx
    # We want to maximize over all lags, which means minimizing the negative
    # But actually, we are maximizing the overlap integral over all k, so we want the maximum
    # of corr with the right indexing
    overlaps = np.array([corr[n-1+k] for k in range(2*n-1)]) * dx
    return np.max(overlaps)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start time tracking
    start_time = time.time()
    
    # Parameters
    n_points = 42  # Using the given initial construction size
    dx = 2.0 / n_points
    
    # If initial construction exists, use it; otherwise create a simple one
    try:
        h0 = initial_h_values
        h0 = np.array(h0)
        if len(h0) != n_points:
            h0 = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        h0 = h0 / np.sum(h0) * dx * n_points  # normalize to correct integral
    except:
        # Simple initialization
        h0 = np.ones(n_points) * 0.5
    
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # For this approach, we'll use a heuristic based on the initial solution
    # and refine it through simple local search

    def perturb_h(h, max_change=0.1):
        """Slightly perturb the step function h"""
        h_new = h.copy()
        # Randomly change values with small steps
        for i in range(n_points):
            if np.random.random() < 0.3:  # 30% chance to modify
                delta = np.random.uniform(-max_change, max_change)
                h_new[i] = np.clip(h_new[i] + delta, 0, 1)
        # Ensure integral constraint
        current_integral = np.sum(h_new) * dx
        h_new = h_new * (1 / current_integral) * dx * n_points
        return h_new
    
    def optimize_h_stochastic(h0, budget_s):
        """Stochastic local search"""
        current_h = h0.copy()
        current_c5 = evaluate_erdos_solution(current_h)
        best_h = current_h.copy()
        best_c5 = current_c5
        start_time = time.time()
        
        # Parameters for stochastic search
        max_change = 0.1
        step = 0
        
        while time.time() - start_time < budget_s - 1:  # Leave 1s for cleanup
            # Make a small perturbation
            h_new = perturb_h(current_h, max_change)
            c5_new = evaluate_erdos_solution(h_new)
            
            # Accept the new solution if it's better
            if c5_new < current_c5:
                current_h = h_new
                current_c5 = c5_new
                if c5_new < best_c5:
                    best_c5 = c5_new
                    best_h = h_new.copy()
                    # Print progress occasionally
                    if step % 100 == 0:
                        remaining = budget_s - (time.time() - start_time)
                        print(f"Step {step}: C5 = {c5_new:.6f}, Remaining time: {remaining:.1f}s")
            
            step += 1
            
            # Adjust search parameters over time
            if step % 1000 == 0:
                max_change = max(0.01, max_change * 0.95)
        
        return best_h, best_c5
    
    # Run the optimization
    final_h, final_c5 = optimize_h_stochastic(h0, budget_s)
    
    # Validate the result
    integral_check = np.sum(final_h) * dx
    if abs(integral_check - 1.0) > 1e-10:
        print(f"Warning: Integral constraint violated: {integral_check}")
    
    # Ensure h is in [0,1]
    final_h = np.clip(final_h, 0, 1)
    
    return (final_h.tolist(), final_c5, n_points)
