import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute overlap integral for h
    # Correlate h with (1-h) using full mode
    corr = correlate(h, 1 - h, mode='full')
    # The overlap values are at indices from -n+1 to n-1
    # We want to take the maximum of these (excluding the zero-lag case)
    # But the mode='full' gives us length 2*n-1 correlation
    # correlation indices: -n+1, ..., n-1
    # The first and last elements correspond to boundary cases
    # We want the maximum value excluding the center, which is at index n-1.
    # But due to how full mode works, the valid overlap indices range from 0 to 2*n-2
    # The lag k that corresponds to correl[0] is when the tail of h overlaps with the start of 1-h, giving lag n-1
    # For convenience: 0 corresponds to k = n-1    1 to k = n-2   ...  n-1 to k = 0 ...
    # ... 2*n-3 to k = -n+2   ...  2*n-2 to k = -n+1
    # Our definition maps back: 
    # if index i corresponds to shift k = i - (n-1) = i - n + 1
    # And we're excluding k = 0 (the zero lag)
    # So we exclude correlation[2*(n-1)] = correlation[n-1]
    # Actually, the zero-lag shift is at index n-1 (k=0)
    # We take max over all lags except k=0
    overlaps = corr * dx
    # We want the maximum value of the correlation computed with lag k != 0
    # The valid lags are from -n+1 to n-1, and we exclude k=0.
    # The corresponding indices in the correlation array are:
    # From 0 to n-2 (corresponding to k = -n+1 to -1) 
    # From n to 2*n-2 (corresponding to k = 1 to n-1)
    max_overlap = np.max(overlaps[:n-1])  # First part: k = -n+1 to -1
    max_overlap = np.maximum(max_overlap, np.max(overlaps[n+1:]))  # Second part: k = 1 to n-1
    return max_overlap

def initialize_population(n_points, pop_size, seed=42):
    """Initialize population with constraint adherence."""
    np.random.seed(seed)
    population = []
    
    # Add the initial solution (if available)
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Generate remaining individuals
    for _ in range(pop_size - len(population)):
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2) 
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        population.append(h.copy())
        
    return population

def mutate(h, mutation_rate=0.05):
    """Mutate h in-place."""
    n = len(h)
    num_mutations = int(n * mutation_rate)
    indices = np.random.choice(n, num_mutations, replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.05)
    # Clip to [0, 1] range
    h = np.clip(h, 0, 1)
    return h

def crossover(parent1, parent2):
    """Single-point crossover."""
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate((parent1[:point], parent2[point:]))
    child2 = np.concatenate((parent2[:point], parent1[point:]))
    return child1, child2

def fitness(h, dx):
    """Evaluate fitness: negative because we minimize"""
    return -evaluate_h(h, dx)

def run(seed=42, budget_s=600, **kwargs):
    start_time = time.time()
    
    # Parameters
    pop_size = 100
    n_points = 45  # Same as the parent construction
    mutation_rate = 0.05
    crossover_prob = 0.8
    elitism_rate = 0.2
    max_no_improvement = 100  # Stop if no improvement
    
    dx = 2.0 / n_points

    # Initialize
    population = initialize_population(n_points, pop_size, seed)
    
    best_fitness = float('inf')
    best_h = None
    no_improvement_count = 0
    
    while True:
        # Check if time is up
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate all individuals
        fitnesses = [fitness(h, dx) for h in population]  # Negative values to minimize
        
        # Sort by fitness (ascending order, negative is better)
        sorted_indices = np.argsort(fitnesses)
        
        # Update best solution if found
        current_best_fitness = fitnesses[sorted_indices[0]]
        if current_best_fitness < best_fitness:
            best_fitness = current_best_fitness
            best_h = population[sorted_indices[0]].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            break
            
        # Elitism
        num_elite = int(elitism_rate * pop_size)
        elite_indices = sorted_indices[:num_elite]
        new_population = [population[i].copy() for i in elite_indices]
        
        # Tournament selection + crossover + mutation
        while len(new_population) < pop_size:
            # Tournament selection
            tourn_size = 5
            tourn_indices = np.random.choice(len(population), tourn_size)
            tourn_fitnesses = [fitnesses[i] for i in tourn_indices]
            winner_idx = tourn_indices[np.argmin(tourn_fitnesses)]
            
            # Clone the winner for one parent
            parent1 = population[winner_idx].copy()
            
            # Choose second parent with some chance
            if np.random.rand() < crossover_prob:
                tourn_indices2 = np.random.choice(len(population), tourn_size)
                tourn_fitnesses2 = [fitnesses[i] for i in tourn_indices2]
                winner_idx2 = tourn_indices2[np.argmin(tourn_fitnesses2)]
                parent2 = population[winner_idx2].copy()
                child1, child2 = crossover(parent1, parent2)
            else:
                child1 = parent1
                
            # Apply mutation
            mutate(child1, mutation_rate)
            # Ensure sum is correct
            sum_old = np.sum(child1)
            child1 = child1 * (n_points / 2) / sum_old 
            # Clip
            child1 = np.clip(child1, 0, 1)
            new_population.append(child1)
            
        population = new_population
        
    # Final calculation of best solution
    if best_h is None:
        # Fallback if no good solution found, evaluate final population individuals
        best_fitness = float('inf')
        best_h = None
        for h in population:
            test_fitness = fitness(h, dx)
            if test_fitness < best_fitness:
                best_fitness = test_fitness
                best_h = h.copy()
            
    final_c5 = -best_fitness  # Convert back from negative to positive
    
    return (best_h, final_c5, n_points)
