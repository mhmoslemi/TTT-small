        corr = np.correlate(h, 1 - h, mode="full")

        return max_corr * dx
        
    def gradient(h):
        # Compute gradient
        n = len(h)
        grad = np.zeros(n)
        h_flipped = 1 - h
        
        # Find max correlation index
        corr = np.correlate(h, h_flipped, mode="full")
        max_idx = np.argmax(corr)
        
        # Compute gradient wrt to that index
        for i in range(n):
            k = i - (max_idx - n + 1)
            if 0 <= k < n:
                grad[i] += (1 - h_flipped[k]) * dx
            else:
                grad[i] += 0
        
        return grad

        # Apply step and project
        h_new = h - step_size * gradient(h)

        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
