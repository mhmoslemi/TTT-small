import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, otherwise create a uniform one
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 45
        h = np.ones(n_points) * 0.5
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Normalize h to meet constraint sum(h) * dx = 1
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Define the objective function
    def objective(h_vec):
        return compute_c5(h_vec, dx)
    
    # Define constraint function for the sum
    def constraint_sum(h_vec):
        return np.sum(h_vec) - target_sum
    
    # Define bounds for each variable (h[i] in [0, 1])
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraints for optimization
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization variables
    x0 = h.copy()
    
    # Gradient-free optimization using Nelder-Mead
    method = 'Nelder-Mead'
    
    while time.time() - start_time < budget_s - 1:
        # Optimize with current h
        try:
            res = minimize(objective, x0, method=method, bounds=bounds, constraints=cons, 
                           options={'maxiter': 100, 'disp': False})
            if res.success:
                h_candidate = res.x
                # Ensure it's within bounds
                h_candidate = np.clip(h_candidate, 0, 1)
                # Rescale to meet sum constraint
                h_candidate = h_candidate * target_sum / np.sum(h_candidate)
                c5_candidate = compute_c5(h_candidate, dx)
                
                if c5_candidate < best_c5:
                    best_h = h_candidate.copy()
                    best_c5 = c5_candidate
                x0 = h_candidate.copy()
            else:
                # If optimization fails, make small random change
                h_new = h.copy() + (np.random.rand(n_points) - 0.5) * 0.01
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * target_sum / np.sum(h_new)
                c5_new = compute_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                x0 = h_new.copy()
        except Exception:
            pass
        
        # Perturb slightly and try again
        h_perturb = h.copy() + (np.random.rand(n_points) - 0.5) * 0.01
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb * target_sum / np.sum(h_perturb)
        c5_perturb = compute_c5(h_perturb, dx)
        if c5_perturb < best_c5:
            best_h = h_perturb.copy()
            best_c5 = c5_perturb
        x0 = h_perturb.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def compute_c5(h, dx):
    """Compute the maximum overlap integral."""
    n = len(h)
    # Convolution of h with (1 - h)
    correlation = np.correlate(h, 1 - h, mode='full')
    # Only keep the valid range (for overlaps)
    # The overlap occurs when k ranges from 0 to 2, so we align with indices
    # Corresponds to offset in correlation: center is at index n-1
    # max for k in [0, 2] corresponds to indices from n-1 to 3*n-3
    # but let's compute across the full range to ensure we capture all k values
    # actually the full correlation should be fine if we're using it correctly
    # just take the maximum!
    max_overlap = np.max(correlation)
    return max_overlap * dx
