import numpy as np
import time
from scipy import correlate

def evaluate(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode="full")
    c5 = np.max(corr) * dx
    return c5

def project(h, n_points):
    """Project h to satisfy sum constraint and bounds"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2.0
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 1000000
    patience = 1000
    patience_counter = 0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5 * dx)
    
    # Normalize to sum = n_points / 2
    h = project(h, n_points)
    
    best_h = h.copy()
    best_c5 = evaluate(best_h, dx)
    
    start_time = time.time()
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate perturbation
        sigma = 0.1 * (0.99 ** i)
        perturbation = np.random.normal(0, sigma, n_points)
        h_new = h + perturbation
        
        # Project to feasible region
        h_new = project(h_new, n_points)
        
        # Evaluate
        c5_new = evaluate(h_new, dx)
        
        # Accept improvement
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Early termination if no improvement for a while
        if patience_counter > patience:
            break
    
    # Final evaluation
    final_c5 = evaluate(best_h, dx)
    
    return (best_h.tolist(), final_c5, n_points)
