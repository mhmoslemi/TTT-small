import numpy as np
from scipy.signal import correlate
import time
from copy import deepcopy

def evaluate_h(h):
    """Evaluate C5 for a given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation of h with (1-h) for all shifts
    corr = correlate(h, 1 - h, mode='full')
    # The max correlation is the max of all shifts
    # The max is over all possible shifts, so we take max
    C5 = np.max(corr) * dx
    return C5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    sum_h = np.sum(h)
    if sum_h != 0:
        h = h * (n_points / 2) / sum_h
    return h

def gradient_step(h, lr):
    """Compute gradient step to decrease C5"""
    n_points = len(h)
    dx = 2.0 / n_points
    
    # Compute the correlation between h and (1-h) - we only need the shift which maximizes it
    corr = correlate(h, 1 - h, mode='full')
    max_shift = np.argmax(corr)
    
    # Compute the gradient w.r.t. each component of h
    # For shift k=0: the gradient is -1 at h[i] for each i
    # For shift k: the gradient is 1 at h[i] for i such that h[i+k] exists
    # A more precise evaluation requires element-wise derivative
    gradient = np.zeros_like(h)
    
    # This gradient estimate uses finite differences:
    for i in range(n_points):
        # Compute how much changing h[i] by epsilon affects the maximum overlap
        h_plus = h.copy()
        h_plus[i] += 1e-6
        h_plus = project_h(h_plus, n_points)
        corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
        h_minus = h.copy()
        h_minus[i] -= 1e-6
        h_minus = project_h(h_minus, n_points)
        corr_minus = correlate(h_minus, 1 - h_minus, mode='full')
        gradient[i] = (np.max(corr_plus) - np.max(corr_minus)) / (2e-6)
        
    return h - lr * gradient

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 45
    
    # Initialize h
    if 'initial_h_values' in globals():

        h = project_h(h, n_points)
    else:

        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x))  
        h = project_h(h, n_points)
    
    best_h = h.copy()
    best_C5 = evaluate_h(h)
    
    # Phase 1: Genetic algorithm to escape local optima
    pop_size = 50
    generations = 1000
    mutation_rate = 0.1
    crossover_prob = 0.7
    elite_ratio = 0.2
    
    # Population initialization
    population = []
    population_C5 = []
    for _ in range(pop_size):
        hi = h + np.random.normal(0, 0.05, n_points)
        hi = project_h(hi, n_points)
        population.append(hi)
        population_C5.append(evaluate_h(hi))
        
    t_remaining = budget_s - (time.time() - start_time)
    
    ga_budget = int(0.7 * t_remaining)
    ga_start = time.time()
    
    while time.time() - ga_start < ga_budget:
        # Sort by fitness
        sorted_indices = np.argsort(population_C5)
        population = [population[i] for i in sorted_indices]
        population_C5 = [population_C5[i] for i in sorted_indices]
        
        # Check for best in generation
        if population_C5[0] < best_C5:
            best_C5 = population_C5[0]
            best_h = population[0].copy()
        
        # Elitism: Keep top 20%
        elite_size = int(elite_ratio * pop_size)
        next_gen = population[:elite_size]
        
        # Generate offspring via crossover and mutation
        while len(next_gen) < pop_size:
            # Tournament selection
            indices = np.random.choice(pop_size, 4, replace=False)
            selected = [population[i] for i in indices]
            selected_C5 = [population_C5[i] for i in indices]
            winner = selected[np.argmin(selected_C5)]
            
            if np.random.random() < crossover_prob:
                # Crossover with another random individual
                indices = np.random.choice(pop_size, 2, replace=False)
                parent1 = population[indices[0]]
                parent2 = population[indices[1]]
                
                # Simple uniform crossover
                mask = np.random.random(n_points) < 0.5
                child = np.where(mask, parent1, parent2)
            else:
                child = winner.copy()
                
            # Mutation
            if np.random.random() < mutation_rate:
                mut_indices = np.random.choice(n_points, size=10, replace=False)
                for i in mut_indices:
                    child[i] += np.random.normal(0, 0.05)
                    
            # Project to constraints
            child = project_h(child, n_points)
            
            next_gen.append(child)
            
        population = next_gen[:pop_size]
        population_C5 = [evaluate_h(hi) for hi in population]
    
    # Phase 2: Gradient descent optimization
    t_remaining = budget_s - (time.time() - start_time)
    gd_budget = t_remaining
    
    if gd_budget < 5:  # Not enough time for GD
        return (best_h, best_C5, n_points)
        
    gd_start = time.time()
    lr = 0.01
    last_improvement = 0
    iteration = 0
    
    while time.time() - gd_start < gd_budget:
        iteration += 1
        if iteration % 1000 == 0:
            # If no improvement, reduce learning rate
            if iteration - last_improvement > 1000:
                lr *= 0.5
                
        # Apply gradient descent
        h_new = gradient_step(best_h, lr)
        h_new = project_h(h_new, n_points)
        
        # Evaluate
        new_C5 = evaluate_h(h_new)
        if new_C5 < best_C5:
            best_C5 = new_C5
            best_h = h_new.copy()
            last_improvement = iteration
            
    return (best_h, best_C5, n_points)
