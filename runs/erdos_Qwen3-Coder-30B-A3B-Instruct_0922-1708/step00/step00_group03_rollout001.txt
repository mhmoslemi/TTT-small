import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    start_time = time.time()
    
    n_points = 70
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Initial sum constraint violated"
    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05, 0.05, n_points) * dx
        h = np.clip(h, 0, 1)
        # Scale to meet constraint
        h *= (n_points / 2) / np.sum(h)
    
    def compute_C5(h_vec):
        # Compute all overlaps
        h_vec = np.array(h_vec)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(conv) * dx
        return max_overlap

    def objective(h_vec):
        # Soft maximum approximation
        alpha = 0.1
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Add small value to prevent log(0)
        exp_vals = np.exp(alpha * conv)
        return np.log(np.sum(exp_vals)) * dx / alpha

    def gradient(h_vec):
        # Compute gradient of soft maximum
        alpha = 0.1
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        exp_vals = np.exp(alpha * conv)
        total = np.sum(exp_vals)
        if total <= 0:
            return np.zeros_like(h_vec)
        
        # Gradient with respect to h
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Compute contribution of h[i] to the convolution terms
            for j in range(len(h_vec)):
                k = i - j
                if 0 <= k < len(conv):
                    grad[i] += alpha * exp_vals[k] * (-1 if k == 0 else 1) * dx
        return grad / total

    # Optimization loop
    current_c5 = compute_C5(h)
    best_c5 = current_c5
    best_h = h.copy()
    
    # Use L-BFGS-B with early stopping
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define optimization callback for time tracking
    def callback(x):
        nonlocal current_c5, best_c5, best_h
        if time.time() - start_time > budget_s * 0.95:
            return True  # Stop optimization
        new_c5 = compute_C5(x)
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = x.copy()
        return False
    
    # Main optimization
    max_iter = 200
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        # Optimize using L-BFGS-B
        result = minimize(objective, h, bounds=bounds, method='L-BFGS-B', 
                          jac=gradient, callback=callback, options={'maxiter': 5})
        if not result.success:
            break
        h = np.clip(result.x, 0, 1)
        # Ensure sum constraint
        h *= (n_points / 2) / np.sum(h)
        if callback(h):
            break
        if i % 20 == 0:
            new_c5 = compute_C5(h)
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = compute_C5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
