import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial_h_values if available, otherwise create a default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        # Create a triangular wave-like initial guess
        n_points = 42
        h = np.zeros(n_points)
        x = np.linspace(0, 2, n_points)
        for i in range(n_points):
            h[i] = max(0, 1 - abs(x[i] - 1))
        # Normalize to sum to 1
        h = h / np.sum(h) * (n_points / 2)
    
    dx = 2.0 / n_points
    start_time = time.time()
    
    def compute_C5(h_vec):
        # Compute cross-correlation of h with (1-h) with full mode
        h_reflected = 1 - h_vec
        correlation = np.correlate(h_vec, h_reflected, mode="full")
        # The overlap integral in the paper is max of all shifts
        max_corr = np.max(correlation)
        return max_corr * dx
    
    def objective(h_vec):
        # Use a smooth approximation to the maximum
        return compute_C5(h_vec)
    
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    def project_h(h_vec):
        # Project onto feasible region: [0,1] and sum to n_points/2
        h_proj = np.clip(h_vec, 0, 1)
        current_sum = np.sum(h_proj)
        if current_sum > 0:
            h_proj = h_proj * (n_points / 2) / current_sum
        return h_proj
    
    # Create bounds: [0,1] for each component
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraints
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Initial solution
    best_h = h.copy()
    best_C5 = compute_C5(best_h)
    
    # Gradient-free optimization should be used
    # We use scipy's L-BFGS-B solver with bounds
    # First, initialize with unconstrained optimization
    
    x0 = h.copy()
    
    # Optimize using scipy minimize
    iter_count = 0
    last_improvement = 0
    max_iter_without_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter_without_improvement:
        try:
            # Reoptimize with current guess
            result = minimize(
                objective, 
                x0=x0, 
                method='L-BFGS-B', 
                bounds=bounds, 
                constraints=cons,
                options={'maxiter': 100}
            )
            
            if not result.success:
                break
                
            # Project the result to feasible region
            new_h = project_h(result.x)
            new_C5 = compute_C5(new_h)
            
            if new_C5 < best_C5:
                best_h = new_h.copy()
                best_C5 = new_C5
                last_improvement = iter_count
                x0 = new_h.copy()
                
            iter_count += 1
            
        except Exception as e:
            break
    
    return (best_h.tolist(), best_C5, n_points)
