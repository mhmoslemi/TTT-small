def run(seed=42, budget_s=600, **kwargs):

            no_improvement_count = 0

        conv = signal.correlate(h, 1 - h, mode='full')

        # Optimizing the max overlap term for the identified k

        corr = np.zeros(2 * n_points - 1)
        for i in range(2 * n_points - 1):
            idx = i - (n_points - 1)
            start_idx = max(0, -idx)
            end_idx = min(n_points, n_points - idx)
            corr[i] = np.sum(h[start_idx:end_idx] * (1 - h[max(0, idx):min(n_points, n_points + idx)]))
        
        # Find max correlated lag
        corr_idx = np.argmax(corr)
        k_best = corr_idx - (n_points - 1)

        h_new = cp.Variable(n_points)
        # Constraint
        constraints = [
            h_new >= 0,
            h_new <= 1,
            cp.sum(h_new) == n_points / 2
        ]

        # We approximate h(x + k) by shifting h by k
        # This is a bit tricky to get right.
        # A different more robust approach:
        # We already have h(t) and want to solve:
        # minimize sum(h_new(t) * (1 - h_new(t + k))) * dx
        # subject to h_new in [0,1], integral = 1
        
        # So for fixed k, we want to solve:
        # max over h_new: integral of h_new * (1 - h_new + k) dx = max over h_new: integral(h_new - h_new^2) dx
        
        # Mismatch! that's not the original.
        # Original is: integral h(x) (1 - h(x + k)) dx
        # Let's correct approach:
        
        # Minimize the max overlap, C_5 = max_k [int h(x) (1 - h(x + k)) dx]
        # Find best k from current h, fix that k, minimize the associated integral.
        # Optimize only for that k.
        
        # Get the best k from current solution
        max_loc = np.argmax(corr)
        best_k = max_loc - (n_points - 1)

            h_shift[:n_points-best_k] = h[best_k:]
            # Reverse stretch for wrap around
            h_shift[n_points-best_k:] = h[:best_k]
            
        else:
            # Forward shift
            h_shift[-best_k:] = h[:n_points+best_k]
            h_shift[:n_points+best_k] = h[-best_k:]

        # This term can be approximated for convex solver.
        # Our relaxation:
        # minimize sum(h_new * (1 - h_shift)) 
        # But that assumes constant shift, so we approximate differently.

        # So just compute h_{i + k} = one of the known values (can't be modeled directly due to non-linearity)
        
        # Simpler convex approximation approach:
        # Build a linear approximation of the non-convex term:
        # h(x)(1 - h(x+k)) linearized around current h(x) at 1st order:
        # = h(x)(1 - h(x + k)) \approx h(x) + (1 - h(x + k) - 1) = h(x) - h(x + k)
        # So minimize sum(h(x) - h(x + k)) = sum(h) - sum(h(x + k))
        # which simplifies to (sum(h) - sum(h)) = 0?
        # That's incorrect.
        
        # Let's use fixed current h value at non-overlapping points
        # For optimization: minimize \int h(x) (1 - h(x + k)) dx over h. 
        # When k is constant, it's an infinite-dimensional linear programming problem with one constraint.
        # We'll create a convex improved solution.
        
        # This is hard, reconstructing the optimization.
        # Let's go strait to iterative subgradient/gradient.
        
        # So the variable is h_new. For a fixed k, 
        # objective is sum_{i} h_new[i] * (1 - h[i-k]) * dx
        
        # Trace this carefully:
        # When computing the optimization, we want:
        # minimize sum_{i} h_new[i] * (1 - h_eval[i_k]) 
        # where h_eval[i_k] is h[t] at h_{i-k} taken carefully.
        
        # And our k: best_k
        # So h_new[i] * (1 - h[i-k]) = h_new[i] - h_new[i]*h[i-k] 
        # But that's still nonlinear in h_new.
        
        # A key optimization step:
        # Approx that we linearize term h_new * (1 - h[i-k]) around h_current[i-k]:
        # This is the first-order Taylor expansion, so we get:
        # h_current[i-k] * (1 - h_current[i-k]) + (h_new[i] - h_current[i]) * (1 - h_current[i-k]). 
        # But that's more complex.
        # Better: 
        # Let's linearize h_new[i] h[i-k] partially, apply standard dual smoothing:
        # min (h_new)(1-h[i-k]) 
        # This is not always convex. 
        # However,   min sum h_new[i] (1 - h_last[i + k])

        # Given a fixed (constant) h_last[x] 
        # Then h(x)(1 - h(x + k)) = h(x) - h(x) h(x + k)
        
        # So we minimize sum(h_new - h_new * h[x+k])
        # Which is same as sum(h_new - h_new * h[x+k])
        # We treat h_old[x+k] as a constant known values
        h_last_k = np.roll(h, best_k)
        try:
            # Solve the linear problem:
            # minimize sum(h_new - h_new * h_last_k) 
            # subject to h_new[i] \in [0,1], sum h_new = n_points / 2
            
            # The supporting linear expressions of h(x)(1-h(x+k))
            # is h(x) (1 - h(x + k)) 
            # which can be written in affine form by using auxiliary variable u for h(x+k) or by 
            # linearization around a known point.
            
            # Direct convex surrogate approximation
            # Let's define a system:
            # minimize sum(h_new[i] * (1 - h_old[i+k]))
            # subject to:
            # sum(h_new) = n_points / 2 
            # 0 <= h_new[i] <= 1
                    
            h_new = cp.Variable(n_points)
            # Linear approximation: h_new[i] (1 - h_old[i+k])
            # is h_new[i] - h_new[i] * h_old[i+k] = h_new[i] - b[i]*h_new[i], 
            # so we are minimizing sum h_new[i] - b[i] * h_new[i]
            # assuming h_old[i+k] is known as (fixed)
            # This is linear in h_new, so continuing
            h_new = cp.Variable(n_points)
            objective = cp.Minimize(
                cp.sum(cp.multiply(h_new, 1.0 - h_last_k))

            constraints = [
                h_new >= 0,
                h_new <= 1,
                cp.sum(h_new) == n_points / 2
            ]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=200)
            
            # Update h
            if prob.status == cp.OPTIMAL:
                h = h_new.value.copy()
        except:
            pass
            
        # Final projection: clamp to [0,1] & rescale to match integral constraint
        h = np.clip(h, 0, 1)
        rescale_factor = (n_points / 2) / np.sum(h)
        h = h * rescale_factor

    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h, dx)
    return (best_h.tolist(), final_c5, n_points)

def evaluate_erdos_solution(h, dx):
    # Corresponds to C_5 = max_k ∫ h(x)(1 - h(x+k)) dx
    corr = np.correlate(h, 1 - np.array(h), mode='full')
    max_overlap = np.max(corr) * dx
    return max_overlap
