import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        h = np.full(n_points, 0.5 * dx)
        h += np.random.uniform(-0.05, 0.05, n_points) * dx
        h = np.clip(h, 0, 1)
    
    # Ensure sum constraint
    current_sum = np.sum(h)
    h = h * target_sum / current_sum
    
    # Objective function that approximates max overlap
    def surrogate_objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Convolve h with (1 - h)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        # Only look at positive lags (the actual overlap values)
        lags = np.arange(len(conv) - n_points + 1, len(conv))
        overlaps = conv[lags]
        # Using log-sum-exp approximation for max
        alpha = 0.1
        return -np.log(np.sum(np.exp(alpha * overlaps))) / alpha
    
    # Gradient of surrogate (simplified)
    def surrogate_gradient(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        conv = correlate(h_vec, 1 - h_vec, mode='full')
        lags = np.arange(len(conv) - n_points + 1, len(conv))
        overlaps = conv[lags]
        max_idx = np.argmax(overlaps)
        max_overlap = overlaps[max_idx]
        
        # Approximate gradient via finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            conv_plus = correlate(h_plus, 1 - h_plus, mode='full')
            overlaps_plus = conv_plus[lags]
            grad[i] = (overlaps_plus[max_idx] - overlaps[max_idx]) / eps
        return -grad
    
    # Optimization loop
    best_c5 = float('inf')
    best_h = h.copy()
    last_improvement = 0
    iteration = 0
    max_iter = 1000
    
    while iteration < max_iter and (time.time() - start_time) < budget_s - 1:
        iteration += 1
        
        # Optimize surrogate
        res = minimize(
            surrogate_objective,
            h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            jac=surrogate_gradient,
            options={'maxiter': 50}
        )
        
        if not res.success:
            break
            
        h = res.x
        h = np.clip(h, 0, 1)
        h = h * target_sum / np.sum(h)
        
        # Compute exact C5
        conv = correlate(h, 1 - h, mode='full')
        lags = np.arange(len(conv) - n_points + 1, len(conv))
        overlaps = conv[lags]
        current_c5 = np.max(overlaps) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = 0
        else:
            last_improvement += 1
            
        if last_improvement > 50:
            break
    
    # Final refinement with exact objective if better solution found
    if best_c5 < 0.38085:
        # Small local refinement
        res = minimize(
            lambda x: np.max(correlate(np.clip(x, 0, 1), 1 - np.clip(x, 0, 1), mode='full')[n_points-1:2*n_points-1]) * dx,
            best_h,
            method='L-BFGS-B',
            bounds=[(0, 1)] * n_points,
            options={'maxiter': 20}
        )
        if res.success:
            final_h = np.clip(res.x, 0, 1)
            final_h = final_h * target_sum / np.sum(final_h)
            final_c5 = np.max(correlate(final_h, 1 - final_h, mode='full')[n_points-1:2*n_points-1]) * dx
            if final_c5 < best_c5:
                best_c5 = final_c5
                best_h = final_h
    
    # Collect final result
    return (best_h.tolist(), best_c5, n_points)
