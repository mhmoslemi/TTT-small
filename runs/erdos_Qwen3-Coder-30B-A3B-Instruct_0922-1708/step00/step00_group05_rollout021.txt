import numpy as np
import time
from scipy import optimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Starting point
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Fourier representation (only even modes for symmetry)
    m_modes = 10
    # We'll optimize coefficients a_k for k=1 to m_modes. a_0 is fixed.
    
    def h_from_coeffs(a):
        # a has shape (m_modes,)
        x = np.linspace(0, 2, n_points)
        h_vals = np.ones(n_points) * 0.5  # a_0 = 0.5
        for k in range(1, m_modes + 1):
            h_vals += a[k-1] * np.cos(np.pi * k * x / 2)
        return np.clip(h_vals, 0, 1)
    
    def rescale_h(h_vals):
        h_vals = np.copy(h_vals)
        h_sum = np.sum(h_vals) * dx
        h_vals *= target_sum / h_sum
        return h_vals

    def compute_c5(h_vals):
        # Correlation computation as in task
        corr = np.correlate(h_vals, 1 - h_vals, mode="full")
        # The full correlation will have length 2*n_points - 1
        # We skip symmetric entries (the middle one counts twice)
        c5 = np.max(np.abs(corr)) * dx
        return c5

    def loss_and_grad(a):
        h_vals = h_from_coeffs(a)
        h_vals = rescale_h(h_vals)
        c5 = compute_c5(h_vals)
        
        # Numerical gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(a)
        base_c5 = compute_c5(h_vals)
        for i in range(len(a)):
            a_plus = np.copy(a)
            a_plus[i] += eps
            h_plus = h_from_coeffs(a_plus)
            h_plus = rescale_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        
        return c5, grad

    # Optimization
    a_start = np.random.rand(m_modes) * 0.5  # Project initial point
    a_opt = a_start.copy()
    best_c5 = float('inf')
    best_a = a_opt.copy()
    
    alpha = 0.01
    iter_count = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        c5, grad = loss_and_grad(a_opt)
        if c5 < best_c5:
            best_c5 = c5
            best_a = a_opt.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        if no_improvement_count > 100:
            alpha *= 0.5
            no_improvement_count = 0
        
        a_opt = a_opt - alpha * grad
        a_opt = np.clip(a_opt, 0, 1)
        iter_count += 1
        
        # Periodic check
        if iter_count % 10 == 0:
            h_vals = h_from_coeffs(a_opt)
            h_vals = rescale_h(h_vals)
            c5_check = compute_c5(h_vals)
            if c5_check < best_c5:
                best_c5 = c5_check
                best_a = a_opt.copy()

    # Return best solution
    best_h_vals = h_from_coeffs(best_a)
    best_h_vals = rescale_h(best_h_vals)
    final_c5 = compute_c5(best_h_vals)
    return (best_h_vals.tolist(), final_c5, n_points)
