import numpy as np
from scipy import optimize
import time
import random

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for given h."""
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation for all shifts
    corr = np.correlate(h, 1 - h, mode="full")
    # The overlap C5 is the maximum of the correlation values scaled by dx
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h to meet constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum != 0:
        h = h * (n_points / 2) / current_sum
    return h

def genetic_algorithm_phase(initial_h, n_points, budget_s):
    """Perform genetic algorithm optimization"""
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_rate = 0.2
    max_generations = 100000  # Stop early if we're close to budget

    start_time = time.time()
    # Initialize population
    population = []
    for _ in range(pop_size):
        # Create random valid solution
        h = np.random.rand(n_points)
        h = project_h(h, n_points)
        population.append(h)

    # If initial_h exists, use it
    if initial_h is not None and len(initial_h) == n_points:
        population[0] = project_h(initial_h.copy(), n_points)

    best_h = None
    best_c5 = float('inf')

    for generation in range(max_generations):
        if time.time() - start_time > 0.7 * budget_s:
            break

        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_overlap(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]

        # Elitism
        num_elite = int(elitism_rate * pop_size)
        elite = population[:num_elite]

        # Generate new population
        new_pop = elite.copy()
        while len(new_pop) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(population, fitness, 3)
            parent2 = tournament_selection(population, fitness, 3)
            
            if random.random() < crossover_prob:
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate, n_points)
                child2 = mutate(child2, mutation_rate, n_points)
                new_pop.extend([child1, child2])
            else:
                child1 = mutate(parent1, mutation_rate, n_points)
                new_pop.append(child1)

        population = new_pop[:pop_size]

    return best_h, best_c5

def tournament_selection(population, fitness, tournament_size):
    """Select one individual using tournament selection."""
    indices = random.sample(range(len(population)), tournament_size)
    best_idx = indices[0]
    for i in indices[1:]:
        if fitness[i] < fitness[best_idx]:
            best_idx = i
    return population[best_idx]

def crossover(parent1, parent2):
    """Single-point crossover"""
    n = len(parent1)
    point = random.randint(1, n - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate, n_points):
    """Gaussian mutation"""
    h = h.copy()
    for i in range(n_points):
        if random.random() < mutation_rate:
            h[i] += np.random.normal(0, 0.05)  # Small Gaussian noise
    return project_h(h, n_points)

def gradient_descent_phase(initial_h, n_points, budget_s):
    """Perform gradient descent optimization on the best solution from GA"""
    max_iter = 10000
    lr = 0.01
    patience = 1000
    patience_counter = 0
    best_h = initial_h.copy()
    best_c5 = evaluate_overlap(best_h)
    start_time = time.time()
    
    # Use Lagrange multipliers or gradient ascent targeting the worst shift
    h = best_h.copy()
    
    for i in range(max_iter):
        if time.time() - start_time > 0.3 * budget_s:
            break
            
        # Compute the gradient of overlap with respect to h for maximum shift
        # since the overlap C5 is taken over maximum
        c5_before = evaluate_overlap(h)
        
        # Compute gradient approximated numerically for better convergence
        grad = np.zeros(n_points)
        eps = 1e-8
        c5_base = evaluate_overlap(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_overlap(h_plus)
            grad[j] = (c5_plus - c5_base) / eps
            
        # Update step
        h = h - lr * grad
        
        # Project to feasible region
        h = project_h(h, n_points)
        
        # Check for improvement
        c5_after = evaluate_overlap(h)
        if c5_after < best_c5:
            best_c5 = c5_after
            best_h = h.copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            # Reduce rate
            lr *= 0.5
            patience_counter = 0

    return best_h, best_c5

def run(seed=42, budget_s=600, **kwargs):
    random.seed(seed)
    np.random.seed(seed)
    
    start_time = time.time()
    n_points = 45  # Priority: faster convergence
    
    # Initialize h from the provided seed or alternate initialization
    initial_h = None
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        if len(initial_h) != n_points:
            initial_h = None
            
    # Phase 1: Genetic Algorithm
    ga_h, ga_c5 = genetic_algorithm_phase(initial_h, n_points, budget_s)
    
    # Phase 2: Gradient Descent on best solution from GA
    if time.time() - start_time < 0.8 * budget_s:
        gd_h, gd_c5 = gradient_descent_phase(ga_h, n_points, budget_s)
        final_h = gd_h
        final_c5 = gd_c5
    else:
        final_h = ga_h
        final_c5 = ga_c5
    
    # Ensure final solution is valid
    final_h = project_h(final_h, n_points)
    final_c5 = evaluate_overlap(final_h)
    
    return (final_h, final_c5, n_points)
