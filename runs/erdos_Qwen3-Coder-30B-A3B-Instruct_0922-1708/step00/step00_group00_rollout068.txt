import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_solution(h, dx):
    """Evaluate the maximum overlap for a given step function h."""
    # Compute the full correlation with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # The overlap integral values are scaled by dx
    overlaps = corr * dx
    # Return the maximum overlap
    return np.max(overlaps)

def initialize_population(n_points, pop_size, initial_h_values=None):
    """Initialize population with the provided initial solution or random solutions."""
    population = []
    if initial_h_values is not None and len(initial_h_values) == n_points:
        # Use the initial solution with some noise
        h0 = initial_h_values.copy()
        h0 = np.clip(h0, 0, 1)
        scale = (n_points / 2) / np.sum(h0)
        h0 = h0 * scale
        population.append(h0)
        # Add other random solutions
        for _ in range(pop_size - 1):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            population.append(h)
    else:
        # All random solutions
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            scale = (n_points / 2) / np.sum(h)
            h = h * scale
            population.append(h)
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    selected_idx = random.randint(0, len(population) - 1)
    for _ in range(tournament_size - 1):
        competitor_idx = random.randint(0, len(population) - 1)
        if fitnesses[competitor_idx] < fitnesses[selected_idx]:
            selected_idx = competitor_idx
    return population[selected_idx]

def crossover(parent1, parent2, crossover_prob=0.8):
    """Perform crossover between two parents."""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
    
    # Single-point crossover
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Ensure the sum constraint
    scale1 = (len(parent1) / 2) / np.sum(child1)
    child1 = child1 * scale1
    scale2 = (len(parent2) / 2) / np.sum(child2)
    child2 = child2 * scale2
    
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Mutate an individual with a given probability."""
    for i in range(len(individual)):
        if random.random() < mutation_rate:
            # Perturb the value slightly
            individual[i] += np.random.normal(0, 0.05)
            # Ensure it stays within [0,1]
            individual[i] = np.clip(individual[i], 0, 1)
    # Ensure the sum constraint
    scale = (len(individual) / 2) / np.sum(individual)
    individual = individual * scale
    return individual

def run(seed=42, budget_s=600, **kwargs):
    # Seed for reproducibility
    random.seed(seed)
    np.random.seed(seed)
    
    # Parameters
    n_points = 45
    pop_size = 100
    max_generations = 120
    dx = 2.0 / n_points
    
    # Initialize
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    best_h = None
    best_c5 = float('inf')
    
    # Genetic algorithm loop
    for gen in range(max_generations):
        # Calculate fitness for each individual
        fitnesses = [evaluate_solution(h, dx) for h in population]
        
        # Update best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitnesses.index(min_fitness)].copy()
        
        # Check time budget
        elapsed = time.time() - start_time
        if elapsed >= budget_s:
            break
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        elite_size = int(0.2 * pop_size)
        elite = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elite.copy()
        while len(new_population) < pop_size:
            # Selection
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            
            # Mutation
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
            
        # Truncate to population size
        population = new_population[:pop_size]
        
    # Final evaluation
    if best_h is not None:
        best_c5 = evaluate_solution(best_h, dx)
        
    return (best_h, best_c5, n_points)
