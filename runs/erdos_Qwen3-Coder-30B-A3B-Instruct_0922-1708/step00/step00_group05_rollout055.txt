import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available, otherwise create random one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
    else:
        n = 70
        h0 = np.random.rand(n)
        h0 = h0 / np.sum(h0) * (n / 2)
    
    n_points = len(h0)
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(h0)
    
    def evaluate_c5(h):
        # This is the evaluation function for C5
        h = np.array(h)
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr) * dx
        return max_corr
    
    # Multiple attempts with random restarts to escape local optima
    max_attempts = 50
    attempt = 0
    
    while time.time() - start_time < budget_s and attempt < max_attempts:
        attempt += 1
        
        # Try solving a QCQP formulation:
        try:
            h = cp.Variable(n_points)
            C = cp.Variable()
            
            # Constraints
            constraints = [
                h >= 0,              # h[i] >= 0
                h <= 1,              # h[i] <= 1
                cp.sum(h) == n_points / 2,  # sum(h) = n_points/2
            ]
            
            # Add overlap constraints
            # For each lag we add one constraint
            for lag in range(-n_points + 1, n_points):
                # Compute correlation manually
                if lag >= 0:
                    corr = cp.sum(cp.multiply(h[lag:], 1 - h[:n_points-lag]))
                else:
                    corr = cp.sum(cp.multiply(h[:n_points+lag], 1 - h[-lag:]))
                
                constraints.append(corr <= C * dx)
            
            # Minimize C
            prob = cp.Problem(cp.Minimize(C), constraints)
            
            # Try different solvers
            solvers = ['ECOS', 'SCS', 'OSQP', 'CVXOPT']
            solver_found = False
            
            for solver in solvers:
                try:
                    prob.solve(solver=solver, verbose=False)
                    if prob.status in ['optimal', 'optimal_inaccurate']:
                        solver_found = True
                        break
                except Exception:
                    continue
            
            if solver_found:
                h_val = h.value
                new_c5 = evaluate_c5(h_val)
                if new_c5 < best_c5:
                    best_h = h_val.copy()
                    best_c5 = new_c5
                    
        except Exception:
            pass
        
        # Fallback to slight perturbation of current best
        h_perturb = best_h + np.random.normal(0, 0.01, n_points)
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb = h_perturb / np.sum(h_perturb) * (n_points / 2)
        
        new_c5 = evaluate_c5(h_perturb)
        if new_c5 < best_c5:
            best_h = h_perturb.copy()
            best_c5 = new_c5
    
    return (best_h.tolist(), best_c5, n_points)
