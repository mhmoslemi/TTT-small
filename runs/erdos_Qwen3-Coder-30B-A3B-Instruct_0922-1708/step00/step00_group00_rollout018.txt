import numpy as np
import time
from scipy import signal

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral C5 for given step function h."""
    # Compute cross-correlation of h with (1-h) using full mode
    corr = signal.correlate(h, 1 - h, mode='full')
    # Multiply by dx to get actual integral values
    corr *= dx
    # Return maximum value across all shifts
    return np.max(corr)

def initialize_population(n_points, pop_size, seed=42):
    """Initialize a population of step functions."""
    np.random.seed(seed)
    population = []
    
    # If initial construction exists, use it as starting point
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = initial_h / np.sum(initial_h) * (n_points / 2)
        population.append(initial_h)
    
    # Fill the rest with random samples
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)  # Scale to meet sum constraint
        h = np.clip(h, 0, 1)  # Ensure bounds [0,1]
        population.append(h)
    
    return population

def tournament_selection(population, fitnesses, tournament_size=5):
    """Select an individual using tournament selection."""
    idxs = np.random.choice(len(population), tournament_size)
    best_idx = idxs[np.argmin([fitnesses[i] for i in idxs])]
    return population[best_idx]

def crossover(parent1, parent2):
    """Perform crossover between two parents."""
    n = len(parent1)
    cx_point = np.random.randint(1, n-1)
    child1 = np.concatenate([parent1[:cx_point], parent2[cx_point:]])
    child2 = np.concatenate([parent2[:cx_point], parent1[cx_point:]])
    return child1, child2

def mutate(individual, mutation_rate=0.05):
    """Mutate an individual with random perturbations."""
    n = len(individual)
    num_mutations = max(1, int(n * mutation_rate))
    indices = np.random.choice(n, num_mutations, replace=False)
    for i in indices:
        individual[i] += np.random.normal(0, 0.1)
    # Clip values to [0,1]
    individual = np.clip(individual, 0, 1)
    # Re-scale to maintain sum constraint
    individual = individual / np.sum(individual) * (n / 2)
    return individual

def genetic_algorithm(n_points, pop_size=100, max_generations=120, 
                      mutation_rate=0.05, crossover_prob=0.8, 
                      elite_ratio=0.2, seed=42, budget_s=600):
    """Run genetic algorithm to optimize the step function."""
    start_time = time.time()
    dx = 2.0 / n_points
    
    population = initialize_population(n_points, pop_size, seed)
    
    best_solution = None
    best_c5 = float('inf')
    
    # Elitism variables
    elite_size = int(elite_ratio * pop_size)
    
    for generation in range(max_generations):
        # Break early if time's up
        if time.time() - start_time > budget_s - 1:
            break
        
        # Evaluate fitness for all individuals
        fitnesses = [evaluate_c5(h, dx) for h in population]
        
        # Track best solution so far
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_solution = population[min_fitness_idx].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Create new population with elitism
        new_population = sorted_population[:elite_size]
        
        # Generate offspring through selection, crossover, mutation
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            if np.random.rand() < crossover_prob:
                parent2 = tournament_selection(population, fitnesses)
                child1, child2 = crossover(parent1, parent2)
                child1 = mutate(child1, mutation_rate)
                child2 = mutate(child2, mutation_rate)
                new_population.extend([child1, child2])
            else:
                child = mutate(parent1, mutation_rate)
                new_population.append(child)
        
        population = new_population[:pop_size]
    
    return best_solution, best_c5

    # Final check
    c5_final = evaluate_c5(h_values, dx)
