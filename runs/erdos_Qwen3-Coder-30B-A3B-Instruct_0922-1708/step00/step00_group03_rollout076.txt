import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5 * 2.0 / n_points)
        
    # Enforce the constraint sum(h) * dx = 1 (dx = 2/n_points)
    dx = 2.0 / n_points
    h *= (n_points / 2.0) / np.sum(h)
    
    # Objective function: smooth approximation of max overlap
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Compute overlaps for all lags
        overlap = correlate(h_vec, 1 - h_vec, mode='full')
        # Use log-sum-exp for smooth approximation to max
        alpha = 0.1  # Temperature parameter
        return -np.log(np.sum(np.exp(alpha * overlap))) / alpha
    
    # Gradient of objective (approximate)
    def gradient(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        # Quick approximation of gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_minus = h_vec.copy()
            h_minus[i] -= eps
            grad[i] = (objective(h_plus) - objective(h_minus)) / (2 * eps)
        return grad
    
    # Optimization with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    start_time = time.time()
    
    # Use L-BFGS-B with gradients
    res = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                   jac=gradient, options={'maxiter': 500, 'ftol': 1e-9})
    
    h_opt = np.clip(res.x, 0, 1)
    h_opt *= (n_points / 2.0) / np.sum(h_opt)
    
    # Final evaluation
    overlap = correlate(h_opt, 1 - h_opt, mode='full')
    c5_bound = np.max(overlap) * dx
    
    return (h_opt.tolist(), float(c5_bound), n_points)
