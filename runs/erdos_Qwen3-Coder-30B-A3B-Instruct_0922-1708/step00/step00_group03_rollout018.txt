import numpy as np
from scipy import optimize
import time

def evaluate_erdos_solution(h):
    """Compute C5 for given step function h."""
    n = len(h)
    dx = 2.0 / n
    # compute all overlaps for lags 0 to 2*n-2
    conv = np.correlate(h, 1-h, mode="full")
    # normalize
    conv *= dx
    # return max of all conv values
    return np.max(conv)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Set up discretization
    n_points = 70
    dx = 2.0 / n_points

    # Get or create initial h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        assert abs(np.sum(h) * dx - 1.0) < 1e-10, "Initial h does not satisfy constraint"
    else:
        h = np.ones(n_points) * 0.5

    # Ensure sum constraint
    h = h * (n_points / 2.0) / np.sum(h)
    h = np.clip(h, 0, 1)

    # EM optimization
    max_iter = 500
    eta = 0.1
    patience = 0
    best_c5 = float('inf')
    best_h = h.copy()
    prev_c5 = float('inf')
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # E-step: compute overlaps
        overlaps = np.zeros(2 * n_points - 1)
        for k in range(2 * n_points - 1):
            # compute overlap for lag k
            lag = k - (n_points - 1)
            if lag >= 0:
                # h[i] * (1 - h[i + lag]) for i = 0..n_points-1-lag
                overlap_val = sum(h[j] * (1 - h[j + lag]) for j in range(n_points - lag))
            else:
                # h[i] * (1 - h[i + lag]) for i = -lag..n_points-1
                overlap_val = sum(h[j] * (1 - h[j - lag]) for j in range(-lag, n_points))
            overlaps[k] = overlap_val

        # Normalize overlaps to form probability distribution
        norm_overlaps = overlaps / np.sum(overlaps)
        
        # M-step: compute gradients and update h
        grad = np.zeros(n_points)
        for k in range(2 * n_points - 1):
            lag = k - (n_points - 1)
            if lag >= 0:
                for j in range(n_points - lag):
                    grad[j] += norm_overlaps[k] * (1 - h[j + lag])
            else:
                for j in range(-lag, n_points):
                    grad[j] -= norm_overlaps[k] * h[j - lag]

        # Update h with gradient step
        h_new = h - eta * grad
        h_new = np.clip(h_new, 0, 1)
        
        # Scale to satisfy integral constraint
        h_new = h_new * (n_points / 2.0) / np.sum(h_new)
        
        # Update h and check improvement
        h = h_new
        current_c5 = evaluate_erdos_solution(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            
        if patience > 50:
            break
            
        # Adjust learning rate
        if current_c5 >= prev_c5:
            eta *= 0.5
            
        prev_c5 = current_c5
        
    return (best_h.tolist(), best_c5, n_points)
