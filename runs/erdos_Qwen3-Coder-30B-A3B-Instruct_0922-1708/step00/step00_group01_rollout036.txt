from scipy.optimize import differential_evolution, minimize
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial solution if available
    try:

    except NameError:

        h0[n_points//2:] = 0.0
        # Adjust to meet sum constraint
        h0 *= (n_points / 2) / np.sum(h0)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_c5 = evaluate_c5(best_h, dx)
    
    start_time = time.time()
    
    # Use scipy's differential evolution for global optimization
    def obj(h):
        return evaluate_c5(h, dx)
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Add constraint for sum
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    try:
        result = differential_evolution(
            obj,
            bounds,
            constraints=cons,
            maxiter=1000,
            popsize=15,
            seed=seed,
            callback=lambda x, convergence: None,
            updating='deferred'
        )
        
        if not result.failed:
            h_opt = result.x
            c5_opt = result.fun
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
        
    except Exception:
        pass  # Fallback to local method if global fails
    
    # Local refinement using Nelder-Mead over best so far
    try:
        result_local = minimize(
            obj,
            best_h,
            method='Nelder-Mead',
            options={'maxiter': 2000, 'disp': False}
        )
        
        if result_local.success:
            h_opt = result_local.x
            c5_opt = result_local.fun
            if c5_opt < best_c5:
                best_h = h_opt
                best_c5 = c5_opt
                
    except Exception:
        pass  # Skip local refinement if fails
    
    # Final evaluation
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h, final_c5, n_points)

def evaluate_c5(h, dx):
    """Evaluate C5 using cross-correlation."""
    # Compute cross-correlation of h and (1-h)
    # This computes the overlap integral for all lags
    corr = correlate(h, 1 - h, mode="full")
    # Maximum value over all lags, scaled by dx
    max_corr = np.max(corr)
    return max_corr * dx
