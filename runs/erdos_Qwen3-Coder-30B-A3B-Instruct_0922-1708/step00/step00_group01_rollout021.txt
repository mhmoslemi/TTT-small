import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100  # Reasonable size for fast evaluation
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * target_sum
    
    h = np.clip(h, 0, 1)
    
    # Frank-Wolfe optimization
    max_iter = 1000
    eta = 0.1
    tol = 1e-6
    start_time = time.time()
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute all overlaps to find maximum
        correlations = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlations)
        max_overlap = correlations[k_max]
        
        # Compute gradient for this k_max
        grad = np.zeros(n_points)
        for i in range(n_points):
            if k_max <= n_points - 1:
                j = i + k_max
                if j < n_points:
                    grad[i] = 1 - h[j]
                else:
                    grad[i] = 1 - h[j - n_points]
            else:
                # k_max is negative, so we're looking back
                j = i - (n_points - k_max)
                if j >= 0:
                    grad[i] = 1 - h[j]
                else:
                    grad[i] = 1 - h[j + n_points]
        
        # Solve linear subproblem: min h^T grad s.t. constraints
        # This is a simple optimization on simplex: min sum_{i=0}^{n-1} grad[i]*x[i]
        # subject to sum x[i] = 1/2, 0 <= x[i] <= 1
        
        # Sort gradient values to facilitate choosing min elements
        sorted_indices = np.argsort(grad)
        
        # Initialize proposed solution with zeros
        d = np.zeros(n_points)
        total = 0.0
        
        # Greedily assign values to minimize sum(grad * d)
        for i in sorted_indices:
            if total >= target_sum:
                break
            # Assign as much as we can to i
            assign = min(target_sum - total, 1.0)
            d[i] = assign
            total += assign
            
        # Ensure the sum is exactly target_sum
        if abs(total - target_sum) > 1e-10:
            d = d * target_sum / total
        
        # Update h
        h = (1 - eta) * h + eta * d
        h = np.clip(h, 0, 1)
        # Ensure sum constraint
        h = h / np.sum(h) * target_sum
        
        # Evaluate current solution
        current_c5 = evaluate_c5(h, dx)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)


def evaluate_c5(h, dx):
    # Compute the full correlation
    correlations = np.correlate(h, 1 - h, mode='full')
    # Get the maximum, but we want the maximum over all valid k values
    # The full correlation has length 2*n-1, corresponding to lags from -n+1 to n-1
    # The valid ones are from -n+1 to n-1
    # Take the maximum among all the overlaps
    return np.max(correlations) * dx
