import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the maximum overlap integral for given h"""
    # Compute the full convolution of h with (1-h)
    conv = correlate(h, 1 - h, mode='full')
    # Take the maximum value, scaled by dx
    return np.max(conv) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Get initial values
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.full(n_points, 0.5 * 2.0 / n_points)
    
    # Ensure sum constraint
    dx = 2.0 / n_points
    scale_factor = (n_points / 2.0) / np.sum(h)
    h *= scale_factor
    
    # Start timing
    start_time = time.time()
    
    # Soft approximation to max function for optimization
    def soft_max_overlaps(h, alpha=0.1):
        # Compute all overlaps
        conv = correlate(h, 1 - h, mode='full')
        # Apply soft maximum with temperature alpha
        return np.log(np.sum(np.exp(alpha * conv))) / alpha
    
    # Gradient of the soft maximum
    def grad_soft_max(h, alpha=0.1):
        conv = correlate(h, 1 - h, mode='full')
        exp_vals = np.exp(alpha * conv)
        sum_exp = np.sum(exp_vals)
        # Compute gradient
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Compute the gradient of overlap at position i
            grad_i = 0
            for k in range(len(h)):
                if i + k < len(h):
                    grad_i += exp_vals[k] * (-1) * (1 - h[i + k])  # Jacobian of h[i+k] contribution
                if i - k >= 0:
                    grad_i += exp_vals[len(h) - k - 1] * 1 * h[i - k]  # Jacobian of h[i-k] contribution
            grad[i] = grad_i / sum_exp
        return grad * alpha
    
    # Optimize using L-BFGS-B with bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Main optimization loop
    iter_count = 0
    last_improvement = 0
    max_iter = 1000
    
    while iter_count < max_iter and (time.time() - start_time) < budget_s - 1:
        # Try optimization with soft max objective
        try:
            result = minimize(
                lambda x: soft_max_overlaps(x, alpha=0.1),
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: grad_soft_max(x, alpha=0.1),
                options={'maxiter': 10, 'disp': False}
            )
            if result.success:
                h_new = np.clip(result.x, 0, 1)
                # Rescale to satisfy constraint ∫h = 1
                h_new *= (n_points / 2.0) / np.sum(h_new)
                c5_new = evaluate_c5(h_new, dx)
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
                    last_improvement = iter_count
        except:
            pass
        
        # Try local perturbation
        h_perturb = h.copy()
        noise = np.random.normal(0, 0.005, n_points)
        h_perturb += noise
        h_perturb = np.clip(h_perturb, 0, 1)
        h_perturb *= (n_points / 2.0) / np.sum(h_perturb)
        c5_perturb = evaluate_c5(h_perturb, dx)
        if c5_perturb < best_c5:
            best_h = h_perturb.copy()
            best_c5 = c5_perturb
            last_improvement = iter_count
        
        # Continue with simple gradient descent
        conv = correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(conv)
        max_lag_val = conv[max_lag_idx]
        
        # Compute finite differences for gradient
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] = min(1, h[i] + 1e-4)
            h_minus[i] = max(0, h[i] - 1e-4)
            conv_plus = correlate(h_plus, 1 - h_plus, mode='full')[max_lag_idx]
            conv_minus = correlate(h_minus, 1 - h_minus, mode='full')[max_lag_idx]
            grad[i] = (conv_plus - conv_minus) / (2e-4)
        
        # Update step
        step_size = 1e-2
        h = h - step_size * grad
        h = np.clip(h, 0, 1)
        h *= (n_points / 2.0) / np.sum(h)
        
        iter_count += 1
        
        # Stop early if no improvement for a while
        if iter_count - last_improvement > 100:
            break
    
    # Ensure final solution is valid and compute the exact C_5
    best_h = np.clip(best_h, 0, 1)
    best_h *= (n_points / 2.0) / np.sum(best_h)
    final_c5 = evaluate_c5(best_h, dx)
    
    return (best_h.tolist(), float(final_c5), int(n_points))
