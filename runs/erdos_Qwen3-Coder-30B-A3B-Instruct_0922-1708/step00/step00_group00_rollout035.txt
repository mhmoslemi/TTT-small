import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 45  # Using the initial construction size
    dx = 2.0 / n_points
    max_generations = 1000
    population_size = 100
    mutation_rate = 0.05
    crossover_prob = 0.8
    tournament_size = 5
    elite_size = population_size // 5
    
    # Generate initial population
    def init_population():
        population = []
        if 'initial_h_values' in globals():
            h0 = np.array(initial_h_values)
            h0 = h0 / np.sum(h0) * (n_points / 2)
            population.append(h0)
        else:
            # Create an initial random solution
            h0 = np.random.rand(n_points)
            # Normalize to meet constraint
            h0 = h0 / np.sum(h0) * (n_points / 2)
            population.append(h0)
            
        # Generate rest of population
        for _ in range(population_size - len(population)):
            h = np.random.rand(n_points)
            h = h / np.sum(h) * (n_points / 2)
            population.append(h)
        return population
    
    # Fitness evaluation
    def fitness(h):
        # Compute overlap integral for all shifts
        h1 = 1 - h
        corr = correlate(h, h1, mode='full')
        # Must shift corr back to correct alignment
        # correlate produces a length 2*n_points-1 array, centered at index n_points-1
        # negative indices k are reflected relative to n_points-1, so:
        # k in [-n_points+1, 0] corresponds to indices [0, n_points-1]
        # k in [1, n_points-1] corresponds to indices [n_points, 2*n_points-2]
        # The values of overlap are in indices that align with shifts k = -n_points+1 to n_points-1
        # So corr at index i corresponds to shift k = i - (n_points - 1)
        # Thus, peak overlap corresponds to max over range [0, 2*n_points-2] 
        # of corr[i] * dx
        return np.max(corr) * dx

    # Tournament selection
    def tournament_selection(pop, fitnesses):
        selected = []
        for _ in range(population_size):
            tournament_indices = np.random.choice(len(pop), size=tournament_size, replace=False)
            best_index = tournament_indices[np.argmin([fitnesses[i] for i in tournament_indices])]
            selected.append(pop[best_index].copy())
        return selected

    # Crossover
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_prob:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        return child1, child2

    # Mutation
    def mutate(h):
        h = h.copy()
        for i in range(len(h)):
            if np.random.rand() < mutation_rate:
                h[i] += np.random.normal(0, 0.05)
        # Clip values
        h = np.clip(h, 0, 1)
        # Re-normalize
        h = h / np.sum(h) * (n_points / 2)
        return h

    # Generate population
    population = init_population()
    
    start_time = time.time()
    best_fitness = float('inf')
    best_h = None

    # Main loop
    for gen in range(max_generations):
        if time.time() - start_time > budget_s:
            break
            
        # Evaluate fitness for current population
        fitnesses = [fitness(h) for h in population]
        
        # Update best solution
        min_fitness = min(fitnesses)
        if min_fitness < best_fitness:
            best_fitness = min_fitness
            best_h = population[np.argmin(fitnesses)].copy()
            
        # Elitism - keep best solutions
        elite_indices = np.argsort(fitnesses)[:elite_size]
        elite = [population[i].copy() for i in elite_indices]
        
        # Select parents
        selected = tournament_selection(population, fitnesses)
        
        # Generate children
        new_population = elite.copy()
        while len(new_population) < population_size:
            parent1, parent2 = np.random.choice(selected, size=2, replace=False)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Trim to population size
        population = new_population[:population_size]

    # Final check of best solution
    if best_h is not None:
        final_fitness = fitness(best_h)
        return (best_h, final_fitness, n_points)
    else:
        # Return worst-case from initial population
        fitnesses = [fitness(h) for h in population]
        best_index = np.argmin(fitnesses)
        return (population[best_index], fitnesses[best_index], n_points)
