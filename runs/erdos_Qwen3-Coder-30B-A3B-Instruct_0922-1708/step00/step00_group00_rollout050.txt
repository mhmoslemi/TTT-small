import numpy as np

def evaluate_overlap(h, dx):
    # Compute all overlaps for shifts k in [-n+1, n-1]
    n = len(h)
    # Use full correlation to get all shifts
    # h * (1 - h_shift)
    corr = correlate(h, 1 - h, mode='full')
    # But we only want the valid overlaps
    # For shift k, overlap is sum(h[i] * (1 - h[i+k]))
    # The correlation indices are k in [-n+1, n-1]
    # We take the range 0 to n-1 which corresponds to shifts k = 0 to n-1
    # But we should be more careful. The original is a convolution of h with (1-h)
    # See corr_indices = np.arange(-n+1, n)
    # overlap_k = sum(h[i] * (1 - h[i+k])) for all valid i
    # corr[k + (n-1)] = overlap_k 
    # where k = [-n+1, ..., n-1]
    overlap_vals = corr[n-1:]  # the positive shifts and zero shift
    # Normalize by dx
    overlap_vals *= dx
    # But actually, the corresponding shifts in corr are k in [-n+1, n-1],
    # So overlaps are corr[n-1+n-1] = corr[2n-2] for k = n-1, etc.
    # Actually no, for indices k corresponding to shift k:
    # corr[0] corresponds to shift -n+1, ..., corr[n-1] to shift 0, ..., corr[2*n-2] to shift n-1.
    # So overlap for shift k is corr[n-1+k]
    # We want the maximum overlap over all shifts k = -n+1, ..., n-1
    # This corresponds to indices k + (n-1) = k + n - 1 of corr, where k = -n+1 to n-1, so those indices are 0 to 2*n-2.
    # Which is the whole corr array.
    # But let's compute the overlaps explicitly.
    # Actually the discretization should not matter much if we compute correctly.
    # But to match the paper's formulation, use the scipy full correlation's output.
    # Let's say corr[i] = sum h[j] (1 - h[j + (i - (n-1))] ) for valid j and j + (i - (n-1))  
    overlaps = []
    for shift_idx in range(len(corr)):
        k = shift_idx - (n-1)
        # for shift k, compute sum h[i] * (1 - h[i+k]) 
        # i ranges over valid indices such that i + k is also valid.
        # i ranges from max(0, -k) to min(n-1, n-1-k) 
        # or 0 to n-1-k if k >= 0, or -k to n-1 if k < 0
        start = max(0, -k)
        end = min(n-1, n-1-k)
        if (k >= 0):
            # i goes from start to end = n-1-k, need h[i], h[i+k] 
            if end >= start:
                val = np.sum(h[start:end+1] * (1 - h[start+k:end+k+1]))
                overlaps.append(val * dx)
        else:
            # k < 0, so i+k ranges from i+k = start+k to end+k = n-1+k.
            # But i ranges from start to end where end = n-1-k, so i+k from start+k to n-1
            # such that start+k >= 0.
            if start+k >= 0 and end+k <= n-1:
                val = np.sum(h[start:end+1] * (1 - h[start+k:end+k+1]))
                overlaps.append(val * dx)
    # Normalize with dx to correspond to actual integral over [0,2]
    max_overlap = max(overlaps) if overlaps else 0.0
    return max_overlap

def evaluate_overlap_fast(h, dx):
    # Use correlation to compute overlaps
    n = len(h)
    # corr = correlate(h, 1-h, mode='full')
    # indices for shift k = -n+1,...,n-1 are 0,...,2*n-2
    # corr[0] = sum_i h[i] * (1-h[i-n+1]), corr[n-1] = sum h[i] * (1-h[i]), etc.  
    # We want all the overlaps.
    corr = correlate(h, 1-h, mode='full')
    # Shift the indices so we map index i to shift i - (n-1)
    # So index 0 of corr maps to shift -n+1, ..., index n-1 maps to 0, index 2*(n-1) maps to n-1
    overlaps = corr[n-1:]  # overlaps from shift 0 through n-1
    # The corresponding shift k = 0 to n-1.
    # The actual overlaps are 
    # overlap_k = sum_i h[i] * (1 - h[i+k]) for valid i, which means i in [0,..n-1-k] and i+k in [0, n-1]
    # i ranges from max(0, -k) to min(n-1, n-1-k) = n-1-k (because k >= 0)
    # So overlap_k = sum_{i=max(0, -k)}^{n-1-k} h[i] * (1-h[i+k])
    
    # But in correlation the k is done differently. Just compute max of all corr results and bound to [0, n-1] which is correct for positive k   
    # corr_indices from 0 to 2*n-2 map to shifts -n+1 to n-1
    # check those using simple 1D case or review an example 
    
    # For h = [0.5, 0.5] 
    # Corr h with 1-h = [0.5, 0.5] 
    # Result: [0.25, 0.25, 0.25]
    # This corresponds to shift -1, 0, 1  
    # Which means overlap -1: h[0]*(1-h[1]) = 0.5*0.5 = 0.25
    # overlap 0: h[0]*(1-h[0]) + h[1]*(1-h[1]) = 0.25 + 0.25 = 0.5
    # overlap 1: h[1]*(1-h[0]) = 0.25
    # But in correlation order:
    # 0th: shift -1 for i = 1: h[1]*(1-h[0]) = 0.25. Really call shift -1 overlap at i=1
    # but overlap_k should be over all i s.t. i+k valid.
    # Let's just compute
    overlaps = []
    for shift in range(-n+1, n):
        # For shift k, we compute sum_i h[i] * (1 - h[i+k])
        # This requires i in [0,n-1], and i+k in [0,n-1]
        # So, i in [0,n-1] and i in [-k, n-1-k]
        # i in [max(0, -k), min(n-1, n-1-k)]
        i0 = max(0, -shift)
        i1 = min(n-1, n-1-shift)
        if i1 >= i0:
            sub_h = h[i0:i1+1]
            sub_shift = h[i0+shift:i1+1+shift]
            # Now range of valid indices is [i0, i1]
            # and we compute over overlap k = i+k  
            # sub_h = h[i0:i1+1] 
            # sub_shift = h[i0+shift:i1+1+shift]
            # Then sum sub_h * (1 - sub_shift)
            # Note that shift can be negative.
            overlap = np.sum(sub_h * (1 - sub_shift))
            overlaps.append(overlap * dx)
    return max(overlaps) if overlaps else 0.0

def project_h(h, n_points):
    # Clip to [0,1] and rescale to sum h = n_points/2
    h = np.clip(h, 0, 1)
    total = np.sum(h)
    # Normalize / rescale
    if abs(total) > 1e-10:
        h = h * (n_points / 2.0) / total
    return h
