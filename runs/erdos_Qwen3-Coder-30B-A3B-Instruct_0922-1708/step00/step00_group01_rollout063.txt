import numpy as np
from scipy.optimize import minimize
from scipy.signal import correlate
import time

def evaluate_c5(h, dx):
    """Evaluate the C5 constant for given step function h."""
    # Cross-correlation of h with (1 - h) for all lags
    corr = correlate(h, 1 - h, mode='full')
    # Compute maximum overlap scaled by dx
    return np.max(corr) * dx

def project_to_feasible(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) * dx = 1."""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

    """Initialize h with a symmetric construction."""
    h = np.ones(n_points) * 0.5
    # Symmetric plateau with sine-like taper
    x = np.linspace(0, 2, n_points)
    taper = 0.5 * (1 - np.cos(np.pi * x))
    h = 0.5 + 0.4 * taper
    # Project to satisfy constraints
    return project_to_feasible(h, n_points, 2.0 / n_points)
