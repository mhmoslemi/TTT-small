import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        assert len(h) == n_points
        assert abs(np.sum(h) * dx - 1.0) < 1e-10
    else:
        # Create uniform initial guess
        h = np.full(n_points, 0.5)
        h = h * target_sum / np.sum(h)
    
    # Temperature for smooth max approximation
    tau = 0.1
    
    def evaluate(h_vec):
        """Computes C5 = max_k int h(x)(1-h(x+k)) dx"""
        # Compute all overlaps using correlation
        h_folded = np.concatenate([h_vec, np.zeros(n_points - 1)])
        one_minus_h = 1.0 - h_vec
        one_minus_h_folded = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
        # Compute cross-correlation
        corr = signal.correlate(h_folded, one_minus_h_folded, mode='full')
        # Extract valid correlations
        overlaps = corr[n_points - 1:]
        # Scale by dx
        overlaps *= dx
        # Return maximum overlap
        return np.max(overlaps)
    
    def smooth_max_overlaps(h_vec, temp=tau):
        """Returns differentiable approximation to max_k overlap"""
        h_folded = np.concatenate([h_vec, np.zeros(n_points - 1)])
        one_minus_h = 1.0 - h_vec
        one_minus_h_folded = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
        corr = signal.correlate(h_folded, one_minus_h_folded, mode='full')
        overlaps = corr[n_points - 1:]  # Shape: 2*n_points - 1
        overlaps *= dx  # Scale
        # Smooth max using temperature
        exp_term = np.exp(temp * overlaps)
        return np.log(np.sum(exp_term)) / temp
    
    def compute_grad(h_vec, temp=tau):
        """Computes gradient of smooth_max_overlaps"""
        h_folded = np.concatenate([h_vec, np.zeros(n_points - 1)])
        one_minus_h = 1.0 - h_vec
        one_minus_h_folded = np.concatenate([one_minus_h, np.zeros(n_points - 1)])
        # Cross correlate h with (1-h) to get overlaps
        corr_h_1mh = signal.correlate(h_folded, one_minus_h_folded, mode='full')
        overlaps = corr_h_1mh[n_points - 1:] * dx  # Scale with dx
        # Compute exp terms
        exp_term = np.exp(temp * overlaps)
        # Compute soft max weights
        weights = exp_term / np.sum(exp_term)
        # Gradient computation:
        # d/dh_i of max_log = sum_j weights_j * d/dh_i overlap_j
        # Use convolutions to get.grad
        # Derivative of correlation: additional derivative step
        # Use expression: overlap_k = sum_i h[i] * (1 - h[i+k])
        # grad = sum_k weights_k * (1 - h[i]) * (1 - h[i+k]) * delta[i,i+k]
        # Reformulate it as convolution-like:
        # grad[i] = sum_k weights_k * (1 - h[i+k])
        # But also consider h[i] so effectively,
        # But since overlaps are: h(i) * (1-h(i+k)), more accurate:
        # Partial of overlap_k wrt h[i] is: 1-h[i+k] if k = i-j,
        # so it is 1-h[i+k] for known k, x+1 = k -> i = k-x+1
        # Let's simplify using:
        # we compute d/dh_i of sum_k (exp(τ*overlap_k)) = sum_k (exp(τ*overlap_k) * τ * d/dh_i overlap_k)
        # d/dh_i overlap_k = (h(i) * (1-h(i+k)))'_i = 1-h(i+k) if i_k = k, else = 0
        # Implied: if i = k, then the derivative is 1-h[k] for derivative wrt h[i]
        # That is, it just returns the part of overlap_k that is with h[i]
        # More accurately:
        # overlap_k = sum_i h[i] * (1-h[i+k]) for all valid i with i+k in range
        # Then d/dh_i overlap_k = 1 - h[i+k]
        # If we need to build gradient of the sums:
        # Grad = sum_k weights_k * (1 - h[i+k]) for k = i-j
        # But we're differentiating wrt h[i] - so what matters is how j relates to k.
        # overlap_k = sum_i h[i] (1 - h[i + k])
        # d/dh[i] overlap_k = 1 - h[i + k]
        # So gradient of sum_k weights_k * overlap_k is sum_k weights_k * (1 - h[i + k])
        # Use a cross-correlation to build up this sum efficiently
        # Build res = signal.correlate one_minus_h with a set of weights
        # But weights are over k indices, but we have i and k such that overlap_k involves h[i]
        # For each i, the derivative wrt h[i] sums over k with k = i-index?
        # The correct approach:
        # overlap_k = sum_i h[i] * (1 - h[i + k])
        # For h[i], the contribution is only present in terms where i+k = some index i'
        # ...equiv to saying d/dh[i] overlap_{k} = (1 - h[i + k]) for valid i
        # So we build a vector whose i-th term is 1 - h[i]
        # Then we compute convolution with a unit vector shifted for each k...
        # Actually simpler:
        # We want: d/dh_i L = sum_k weights_k * [d/dh_i overlap_k]
        # d/dh_i overlap_k is (1-h[i+k]) if i+k is valid, else 0
        # So the best way: keep the reference vector h, compute lists of d_overlap_k/dh_i for some subset
        # and compute sum_k weights_k * d_overlap_k/dh_i
        # Let's just implement derivative calculation directly:
        # Compute gradient with respect to h[i]
        grad = np.zeros(n_points)
        # Precompute 1 - h for faster access
        one_minus_h = 1.0 - h_vec
        for i in range(n_points):
            # Sum over k of weights_k * d/dh_i overlap_k
            # This is 1 - h[i+k] if k is valid such that i+k is within bounds
            # i = 0..n-1
            # overlap_k = sum_j h[j] (1 - h[j + k])
            # d/dh_i overlap_k = 1 - h[i + k] if i + k >= 0 and i + k < n
            for k in range(-n_points+1, n_points):
                j = i + k
                if j >= 0 and j < n_points:
                    # d/dh[i] overlap_k = 1 - h[i + k]
                    # Weight is for the corresponding overlap_k index
                    # overlap_k corresponds to the index in the full overlap array which is k + n-1
                    jk = k + n_points - 1
                    grad[i] += weights[jk] * one_minus_h[j] * temp
        return grad
    
    # Optimization parameters
    learning_rate = 0.01
    max_iter = 10000
    patience = 50
    best_c5 = float('inf')
    best_h = None
    last_update = 0
    
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
        c5 = evaluate(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_update = iter_count
        if iter_count - last_update > patience:
            # Early stop if no improvement
            break
        
        # Compute gradient
        try:
            grad = compute_grad(h)
        except:
            # Fallback to finite difference if gradient fails
            eps = 1e-4
            grad = np.zeros_like(h)
            base_loss = smooth_max_overlaps(h)
            for i in range(n_points):
                h_eps = h.copy()
                h_eps[i] += eps
                loss_eps = smooth_max_overlaps(h_eps)
                grad[i] = (loss_eps - base_loss) / eps
        # Clip gradient
        grad = np.clip(grad, -100, 100)
        # Update
        h_new = h - learning_rate * grad
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint sum(h) = n_points / 2
        current_sum = np.sum(h_new)
        if abs(current_sum - target_sum) > 1e-6:
            h_new = h_new * target_sum / current_sum
        h = h_new
    
    if best_h is None:
        best_h = h
    
    # Final evaluation
    final_c5 = evaluate(best_h)
    return (best_h.tolist(), final_c5, n_points)
