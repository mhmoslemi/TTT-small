import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral C5 for given h."""
    # Compute correlation of h with (1-h)
    corr = correlate(h, 1 - h, mode='full')
    # Extract the relevant part and multiply by dx
    c5 = np.max(corr) * dx
    return c5

def project_h(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points / 2."""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if abs(current_sum - target_sum) > 1e-10:
        # Rescale to hit target sum
        h = h * target_sum / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    # Initialize parameters
    np.random.seed(seed)
    n_points = 45
    dx = 2.0 / n_points
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():

    else:

    # Project to satisfy constraints
    h = project_h(h, n_points)
    
    # Evaluate initial value
    c5_best = evaluate_h(h, dx)
    h_best = h.copy()
    
    # Optimization parameters
    sigma = 0.1
    decay_factor = 0.99
    max_iter = 1000000
    patience = 1000
    patience_counter = 0
    
    start_time = time.time()
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s:
            break
            
        # Generate perturbation
        delta = np.random.normal(0, sigma, n_points)
        h_new = h + delta
        
        # Project to constraints
        h_new = project_h(h_new, n_points)
        
        # Evaluate new candidate
        c5_new = evaluate_h(h_new, dx)
        
        # Accept improvement
        if c5_new < c5_best:
            h_best = h_new.copy()
            c5_best = c5_new
            patience_counter = 0
        else:
            patience_counter += 1
            
        # Update h for next iteration
        h = h_new.copy()
        
        # Decrease noise variance gradually
        sigma *= decay_factor
        
        # Early stopping if no improvement for a while
        if patience_counter > patience:
            break
    
    # Final check
    final_c5 = evaluate_h(h_best, dx)
    
    return (h_best.tolist(), final_c5, n_points)
