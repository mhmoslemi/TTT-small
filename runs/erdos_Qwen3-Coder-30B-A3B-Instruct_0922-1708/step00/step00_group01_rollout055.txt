import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the maximum overlap integral for a given step function h"""
    n = len(h)
    # Compute all overlaps for all lag values and find the maximum
    overlaps = correlate(h, 1 - h, mode='full') * dx
    # Take the max over all lags
    return np.max(overlaps)

def project_h(h, n_points):
    """Project h to satisfy the sum constraint and bounds [0,1]"""
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    # Adjust to meet sum constraint
    current_sum = np.sum(h)
    if current_sum > 0:
        target_sum = n_points / 2.0
        h = h * target_sum / current_sum
    return h
