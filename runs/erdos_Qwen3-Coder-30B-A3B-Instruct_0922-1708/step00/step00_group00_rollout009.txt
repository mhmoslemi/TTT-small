import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Problem parameters
    n_points = 45
    dx = 2.0 / n_points
    target_integral = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_integral
    else:
        h = np.ones(n_points) * 0.5  # Default initialization
    
    # Ensure h satisfies constraints
    h = np.clip(h, 0, 1)
    h = h / np.sum(h) * target_integral
    
    # Precompute 1-h for efficiency
    h_complement = 1 - h
    
    # ADMM parameters
    rho = 1.0
    max_iter = 1000
    tol = 1e-6
    backtracking_factor = 0.5
    max_backtrack = 10
    
    # Storage for tracking best solution
    best_c5 = float('inf')
    best_h = h.copy()
    
    # ADMM iterations
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1e-3:
            break
            
        # Compute overlap terms (full convolution)
        overlap = np.correlate(h, h_complement, mode='full')
        
        # Shift to align with the correct indices
        overlap_shift = overlap[len(h)-1:]  # Use valid part
        c5_value = np.max(overlap_shift) * dx
        
        if c5_value < best_c5:
            best_c5 = c5_value
            best_h = h.copy()
        
        # ADMM update step:
        # Primal update: minimize ||h - h_prev + rho * (h_prev - h) ||^2 + penalty for constraint
        # Noting that the overlap term is non-convex esp. due to max, use gradient descent first
        
        # Compute gradient with respect to h
        grad = np.zeros(n_points)
        for k in range(n_points):
            # Compute contribution of shift k to gradient
            if k == 0:
                # h[i] * (1 - h[i]) contribution
                grad += h_complement - h
            else:
                # For k>0: h[i] * (1 - h[i+k]) for i+k<n_points
                valid_i = slice(0, n_points-k)
                valid_k = slice(k, n_points)
                # Add to corresponding positions
                grad[valid_i] += h[valid_k] - h[valid_i]  # slight approximation of partial derivatives

        # Gradient descent step
        h_new = h - 0.01 * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new / np.sum(h_new) * target_integral
        
        h = h_new
        h_complement = 1 - h
        
    return (best_h.tolist(), best_c5, n_points)
