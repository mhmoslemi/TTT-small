import numpy as np
import time
from scipy.signal import correlate
from scipy.optimize import minimize_scalar
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given step function h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the overlap integral for all shifts
    corr = correlate(h, 1 - h, mode="full")
    # The maximum overlap value
    max_overlap = np.max(corr)
    return max_overlap * dx

def initialize_population(n_points, initial_h_values, pop_size):
    """Initialize a population of step functions."""
    population = []
    # Add the initial solution if available
    if initial_h_values is not None:
        h = np.array(initial_h_values)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    # Generate random initial solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    return population

def mutate(h, mutation_rate=0.2, dx=2.0/70):
    """Apply mutation to a step function."""
    new_h = h.copy()
    if np.random.rand() < mutation_rate:
        # Randomly select points to mutate
        n_mutate = int(0.1 * len(h))
        indices = np.random.choice(len(h), n_mutate, replace=False)
        for i in indices:
            new_h[i] += np.random.uniform(-0.1, 0.1) * dx
        # Clip values to [0,1]
        new_h = np.clip(new_h, 0, 1)
        # Rescale to keep sum constraint
        new_h = new_h * (len(h) / 2) / np.sum(new_h)
    return new_h

def crossover(h1, h2):
    """Perform crossover between two step functions."""
    if np.random.rand() < 0.5:
        # Swap segments
        split = np.random.randint(1, len(h1))
        new_h1 = np.concatenate([h1[:split], h2[split:]])
        new_h2 = np.concatenate([h2[:split], h1[split:]])
        # Rescale
        new_h1 = new_h1 * (len(h1) / 2) / np.sum(new_h1)
        new_h2 = new_h2 * (len(h2) / 2) / np.sum(new_h2)
        return new_h1, new_h2
    else:
        return h1.copy(), h2.copy()

def tournament_selection(population, fitnesses, tournament_size=20):
    """Select parents using tournament selection."""
    selected = []
    for _ in range(len(population)):
        tournament_indices = np.random.choice(len(population), tournament_size, replace=False)
        tournament_fitnesses = [fitnesses[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitnesses)]
        selected.append(population[winner_idx])
    return selected

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    n_points = 70
    dx = 2.0 / n_points
    pop_size = 100
    max_generations = 300
    patience = 50
    patience_counter = 0
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()

    # Initialize population
    population = initialize_population(n_points, initial_h_values, pop_size)
    fitnesses = [evaluate_erdos_solution(h) for h in population]
    
    # Store best solution
    best_idx = np.argmin(fitnesses)
    if fitnesses[best_idx] < best_c5:
        best_c5 = fitnesses[best_idx]
        best_h = population[best_idx].copy()
    
    # GA loop
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 30:
            break
            
        # Selection
        selected = tournament_selection(population, fitnesses)
        
        # Create offspring
        offspring = []
        while len(offspring) < pop_size:
            parent1 = selected[np.random.randint(0, len(selected))]
            parent2 = selected[np.random.randint(0, len(selected))]
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            offspring.extend([child1, child2])
        
        # Trim offspring to pop_size
        offspring = offspring[:pop_size]
        
        # Evaluate offspring
        offspring_fitnesses = [evaluate_erdos_solution(h) for h in offspring]
        
        # Combine population
        population = population + offspring
        fitnesses = fitnesses + offspring_fitnesses
        
        # Sort by fitness
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Trim to pop_size
        population = population[:pop_size]
        fitnesses = fitnesses[:pop_size]
        
        # Update best
        if fitnesses[0] < best_c5:
            best_c5 = fitnesses[0]
            best_h = population[0].copy()
            patience_counter = 0
        else:
            patience_counter += 1
            
        if patience_counter >= patience:
            break
        
    return (best_h.tolist(), best_c5, n_points)
