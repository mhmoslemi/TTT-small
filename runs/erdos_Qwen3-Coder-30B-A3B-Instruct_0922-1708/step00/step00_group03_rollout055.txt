import numpy as np

import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Problem parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * target_sum
    else:
        h = np.ones(n_points) * 0.5
    
    # Start timing
    start_time = time.time()
    
    # Do a simple gradient-based optimization as a baseline
    def evaluate(h):
        h = np.array(h)
        return np.max(np.correlate(h, 1 - h, mode="full")) * dx
    
    # Simple gradient descent
    learning_rate = 0.01
    max_iter = 10000
    tolerance = 1e-6
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        current_val = evaluate(h)
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus / np.sum(h_plus) * target_sum
            
            val_plus = evaluate(h_plus)
            grad[j] = (val_plus - current_val) / eps
        
        # Update h
        h = h - learning_rate * grad
        
        # Project to feasible set
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * target_sum
        
        # Check for convergence
        if np.linalg.norm(grad) < tolerance:
            break
            
    # Final evaluation
    final_c5 = evaluate(h)
    
    return (h.tolist(), final_c5, n_points)
