import numpy as np
import time
from scipy import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points / 2
    current_sum = np.sum(h)
    h = h * (n_points / 2) / current_sum
    
    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = np.inf
    
    # Parameters
    learning_rate = 0.01
    patience = 0
    max_patience = 100
    
    def compute_c5(h_vals):
        # Compute overlap integral for all shifts
        h_pad = np.concatenate([h_vals, np.zeros(n_points)])
        h_rev = np.flip(h_vals)
        result = correlate(h_pad, h_rev, mode='full')
        # Only take the relevant part of the correlation (positive lags)
        overlap = result[n_points-1:] * dx
        return np.max(overlap)
    
    def compute_gradient(h_vals, shift_k):
        grad = np.zeros_like(h_vals)
        for i in range(n_points):
            if i + shift_k < n_points:
                grad[i] = (1 - h_vals[i + shift_k]) * dx
            if i - shift_k >= 0:
                grad[i] -= h_vals[i - shift_k] * dx
        return grad
    
    def project_h(h_vals):
        # Clip to [0, 1]
        h_vals = np.clip(h_vals, 0, 1)
        # Adjust to sum to n_points / 2
        current_sum = np.sum(h_vals)
        if abs(current_sum - n_points/2) > 1e-10:
            scale = (n_points/2) / current_sum
            h_vals = h_vals * scale
        return h_vals
    
    while time.time() - start_time < budget_s - 0.1:
        # Compute current max overlap
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
        
        # Find the shift with maximum overlap
        h_pad = np.concatenate([h, np.zeros(n_points)])
        h_rev = np.flip(h)
        result = correlate(h_pad, h_rev, mode='full')
        overlap = result[n_points-1:] * dx
        max_k = np.argmax(overlap)
        
        # Compute gradient for the worst shift
        grad = compute_gradient(h, max_k)
        
        # Update h
        h = h - learning_rate * grad
        h = project_h(h)
        
        # Adaptive learning rate
        if patience % 100 == 0:
            learning_rate *= 0.95
        if patience > max_patience:
            break
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
