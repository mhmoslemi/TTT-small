import numpy as np
import time
from scipy.optimize import minimize
from scipy.special import expit

def sigmoid(x):
    return expit(x)

def run(seed=42, budget_s=600, n_points=45):
    np.random.seed(seed)
    start_time = time.time()
    
    # base function setup
    m = max(10, int(2 * np.sqrt(n_points)))
    x = np.linspace(0, 2, n_points)
    dx = 2.0 / n_points
    
    # basis functions
    centers = np.linspace(0, 2, m)
    scales = np.full(m, 2.0 / m)
    
    def basis_func(x, mu, sigma):
        return sigmoid((x - mu) / sigma)
    
    # create basis matrix
    Phi = np.zeros((n_points, m))
    for i in range(m):
        Phi[:, i] = basis_func(x, centers[i], scales[i])
    
    # Initialize from initial_h_values if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = h_start / np.sum(h_start) * n_points / 2
        # Compute coefficients
        coeffs = np.linalg.lstsq(Phi, h_start, rcond=None)[0]
        coeffs = np.clip(coeffs, -1.0, 1.0)
    else:
        # random initialization with scaling to ensure integral
        coeffs = np.random.uniform(-0.5, 0.5, m)
        # scale to meet integral constraint
        h_init = Phi @ coeffs
        h_init = np.clip(h_init, 0, 1)
        integral = np.sum(h_init) * dx
        coeffs = coeffs * (1.0 / integral) * (n_points / 2)
    
    # computation functions
    def compute_overlap(coeffs):
        h = Phi @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points / np.sum(h))  # Fix integral
        h = h * dx  # Scale to get actual integral
        # overlap computation
        h_full = np.concatenate([h, np.zeros(n_points)])
        overlap = np.correlate(h_full, 1 - h_full, mode='full')
        return np.max(overlap) * dx
    
    def gradient(coeffs):
        h = Phi @ coeffs
        h = np.clip(h, 0, 1)
        h = h * (n_points / np.sum(h))  # Fix integral
        h = h * dx
        h_full = np.concatenate([h, np.zeros(n_points)])
        k_range = range(n_points, 2*n_points)
        max_k = 0
        max_overlap = -np.inf
        for k in k_range:
            corr = np.sum(h_full[:n_points] * (1 - h_full[k:k+n_points]))
            if corr > max_overlap:
                max_overlap = corr
                max_k = k
        # Approximate gradient via finite differences
        eps = 1e-5
        grad = np.zeros_like(coeffs)
        for i in range(m):
            coeffs_pert = coeffs.copy()
            coeffs_pert[i] += eps
            h_pert = np.clip(Phi @ coeffs_pert, 0, 1)
            h_pert = h_pert * (n_points / np.sum(h_pert))
            h_pert = h_pert * dx
            h_pert_full = np.concatenate([h_pert, np.zeros(n_points)])
            corr_pert = np.sum(h_pert_full[:n_points] * (1 - h_pert_full[max_k:max_k+n_points]))
            grad[i] = (corr_pert - max_overlap) / eps
        return grad
    
    # Optimization loop
    best_coeffs = coeffs.copy()
    best_c5 = compute_overlap(coeffs)
    alpha = 0.01
    patience = 0
    max_patience = 20
    for i in range(500):
        if time.time() - start_time > 0.95 * budget_s:
            break
        g = gradient(coeffs)
        # Elastic net penalty
        l1_penalty = 0.8 * np.sign(coeffs)
        l2_penalty = 0.2 * coeffs
        coeffs = coeffs - alpha * (g + l1_penalty + l2_penalty)
        # Clip to feasible region
        coeffs = np.clip(coeffs, -1.0, 1.0)
        # Check new value
        current = compute_overlap(coeffs)
        if current < best_c5:
            best_c5 = current
            best_coeffs = coeffs.copy()
            patience = 0
        else:
            patience += 1
            if patience > max_patience:
                alpha *= 0.5
                patience = 0
        if patience > max_patience and alpha < 1e-6:
            break
    
    # Final conversion to discrete representation
    h_final = Phi @ best_coeffs
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_points / np.sum(h_final))
    
    # estimate final C5
    h_full_final = np.concatenate([h_final, np.zeros(n_points)])
    overlap_final = np.correlate(h_full_final, 1 - h_full_final, mode='full')
    C5_estimate = np.max(overlap_final) * dx
    
    # Cast to output format
    h_values = np.array(h_final).reshape(-1)
    
    return (h_values, C5_estimate, n_points)
