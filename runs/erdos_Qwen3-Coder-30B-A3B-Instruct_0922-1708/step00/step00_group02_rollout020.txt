import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initial setup
    n_points = 45
    dx = 2.0 / n_points
    max_iter = 3000
    alpha = 0.01
    beta = 0.9
    patience = 500
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Track previous values for learning rate adaptation
    prev_loss = float('inf')
    no_improvement_count = 0
    
    start_time = time.time()
    
    # Momentum terms
    v_h = np.zeros_like(h)
    
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlap for all shifts
        # Cross-correlation of h with (1-h)
        corr = signal.correlate(h, 1 - h, mode="full")
        # The full correlation has length 2*n_points - 1
        # We want the valid overlaps: from shift -1 to 1
        # But its index goes from 0 to 2*n_points-2
        # Shift 0 in correlation corresponds to shift 0 in our problem
        # The fold is in [0, 2] so shifts go from 0 to 2
        corr_shift = (np.arange(2*n_points - 1) - (n_points - 1)) * dx
        # We only care about shift in [0, 2]
        valid_indices = (corr_shift >= 0) & (corr_shift <= 2)
        corr_values = corr[valid_indices]
        corr_shifts = corr_shift[valid_indices]
        
        # Find the k that maximizes overlap
        max_corr_idx = np.argmax(corr_values)
        k_max = corr_shifts[max_corr_idx]
        c5 = corr_values[max_corr_idx] * dx
        
        # Track best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient w.r.t. h
        # From analytical gradient derivation
        # We compute overlap for current k_max
        # Then define a proper correlation-based gradient
        # Or use finite differences to approximate gradient
        # Let's compute one-step derivative:
        # ∂C₅/∂h_i = [1-h(i+k_max)] - h(i+k_max) * [1-h(i+k_max)] = (1-h(i+k_max)) - h(i+k_max)^2
        # But it's complex. Let's optimize h directly via descent in its representation.
        
        # Alternative method:
        # Define objective as minimization of max_k integral
        # ∂/∂h_i [∫ h(x)(1 - h(x+k_max)) dx] = (1 - h(i+k_max)) 
        # But we're not computing chain rule right. 
        # Let’s run a proper scheme with numeric gradient or
        # We rely on gradient-free first, then gradient-based if needed here
        
        # Simple gradient-free one-step update
        # Compute expected gradient via cross-correlation approximation
        # For each i, we compute ∂/∂h_i of overlap integral at k_max
        # If k_max induced the max, and no further 그러면 we can compute finitely
        
        # We compute expected gradient by differing from continuous formula
        grad = np.zeros_like(h)
        # Simple approach: compute finite difference in the most significant direction
        for i in range(n_points):
            # For current h, we have k_max for max overlap
            k_i = k_max
            # Compute gradient: d/df(h(x)) for f = h(x)(1-h(x+k_i))
            # It's just (1 - h(x+k_i)) in expression for x=i
            offset = int(k_i / dx) 
            if i + offset < n_points:
                grad[i] = (1 - h[i + offset])
        
        # Update h using momentum
        v_h = beta * v_h + (1 - beta) * grad
        h = h - alpha * v_h
        
        # Project h onto valid range and constraint sum
        h = np.clip(h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(h)
        h = h * scale_factor
        
        # Adaptive learning rate
        if c5 > prev_loss:
            no_improvement_count += 1
            if no_improvement_count > patience:
                alpha *= 0.5
                no_improvement_count = 0
        else:
            no_improvement_count = 0
        prev_loss = c5
        
        if no_improvement_count > 1000:
            # Reset if no progress
            if 'initial_h_values' in globals():
                h = np.array(initial_h_values)
                h = h / np.sum(h) * (n_points / 2)
            else:
                h = np.ones(n_points) * 0.5
            alpha = 0.01  # Reset learning rate

    # Final computation of C5 using the evaluator
    # To validate best_h (it's valid as it was constrained)
    # Compute full correlation and extract max:
    corr = signal.correlate(best_h, 1 - best_h, mode="full")
    corr_shift = (np.arange(2*n_points - 1) - (n_points - 1)) * dx
    valid_indices = (corr_shift >= 0) & (corr_shift <= 2)
    corr_values = corr[valid_indices]
    final_c5 = np.max(corr_values) * dx
    
    return (best_h, final_c5, n_points)
