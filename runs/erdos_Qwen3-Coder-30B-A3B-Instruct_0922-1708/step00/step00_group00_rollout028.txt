import numpy as np

import time

def evaluate_c5(h, dx):
    """Evaluate C5 = max_k int h(x)(1-h(x+k)) dx"""
    # Compute all overlaps using cross-correlation manually
    n = len(h)
    max_overlap = 0
    # For each shift k from -n+1 to n-1
    for k in range(1 - n, n):
        # Compute overlap for shift k
        if k >= 0:
            # h[i] * (1 - h[i + k]) for valid i
            overlap = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            # h[i] * (1 - h[i + k]) = h[i] * (1 - h[k+i]) for valid i
            overlap = np.sum(h[-k:] * (1 - h[:-k]))
        # Scale by dx and update max
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    return max_overlap

def project_h(h, n_points):
    """Project h to [0,1] and rescale to sum=1"""
    # Clamp h to [0,1]
    h = np.clip(h, 0, 1)
    # Rescale
    if np.sum(h) > 0:
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.full(n_points, 0.5)
    return h

def compute_gradient(h, dx):
    """Compute approximate gradient w.r.t. each h[i] for the smooth max of overlaps"""
    n = len(h)
    grad = np.zeros(n)
    # Prepare all overlaps for all shifts k
    overlaps = []
    for k in range(1 - n, n):
        if k >= 0:
            overlap = np.sum(h[:n-k] * (1 - h[k:]))
        else:
            overlap = np.sum(h[-k:] * (1 - h[:-k]))
        overlaps.append(overlap * dx)
    # Temporarily smooth max with temperature 0.1
    temp = 0.1
    exp_overlaps = np.exp(temp * np.array(overlaps))
    Z = np.sum(exp_overlaps)
    # Gradient for smooth max
    for i in range(n):
        # Partial derivative of smooth max w.r.t. h[i]
        d_max = 0
        for k_idx, k in enumerate(range(1 - n, n)):
            if k >= 0:
                if i < n - k and i + k < n:
                    # Original contribution to overlap_k from h[i] * (1 - h[i+k])
                    # The derivative w.r.t. h[i] is (1 - h[i+k]) and w.r.t. h[i+k] is -h[i]
                    # Chain rule with smooth max: smooth_max derivative = sum(exp(...) * ...)/sum(exp(...))
                    d_over_k = 0
                    if i < n - k:
                        d_over_k += 1 - h[i+k]
                    if i+k < n:
                        d_over_k += -h[i]
                    d_max += (temp * exp_overlaps[k_idx] * d_over_k) / Z
            else:
                if i+k >= 0 and i < n:
                    # Mirror case
                    d_over_k = 0
                    if i < n:
                        d_over_k += 1 - h[i+k]
                    if i+k < n:
                        d_over_k += -h[i]
                    d_max += (temp * exp_overlaps[k_idx] * d_over_k) / Z
        grad[i] = d_max
    return grad

def backtracking_line_search(h, grad, dx, alpha_init=0.1, beta=0.5):
    """Backtracking line search to find suitable step size"""
    alpha = alpha_init
    c5_prev = evaluate_c5(h, dx)
    while alpha > 1e-10:
        h_new = h - alpha * grad
        h_new = project_h(h_new, len(h))
        c5_new = evaluate_c5(h_new, dx)
        if c5_new < c5_prev:
            return alpha, h_new
        else:
            alpha *= beta
    return alpha, h
