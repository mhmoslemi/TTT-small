import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h."""
    corr = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(corr) * dx
    return max_overlap

def project_to_feasible(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points / 2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Define parameters
    n_points = 100
    dx = 2.0 / n_points
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    stagnation_limit = 50
    diversity_reset_ratio = 0.2
    
    # Initialize particles
    if 'initial_h_values' in globals():
        initial_h = np.array(initial_h_values)
        initial_h = project_to_feasible(initial_h, n_points)
    else:
        # Create initial random solution
        initial_h = np.random.rand(n_points)
        initial_h = project_to_feasible(initial_h, n_points)
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Set up particles around the initial solution
    for i in range(swarm_size):
        if i == 0:
            h = initial_h.copy()
        else:
            # Perturb initial solution
            h = initial_h + np.random.normal(0, 0.05, n_points)
            h = project_to_feasible(h, n_points)
        particles.append(h.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h.copy())
        pbest_vals.append(evaluate_h(h, dx))
    
    # Initialize global best
    global_best_idx = np.argmin(pbest_vals)
    global_best = pbest[global_best_idx].copy()
    global_best_val = pbest_vals[global_best_idx]
    
    # PSO loop
    last_improvement = 0
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (global_best - particles[i]))
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_to_feasible(particles[i], n_points)
            
            # Evaluate new solution
            new_val = evaluate_h(particles[i], dx)
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = new_val
                    last_improvement = iteration
                    
        # Check for stagnation
        if iteration - last_improvement > stagnation_limit:
            # Inject diversity
            num_to_reset = int(swarm_size * diversity_reset_ratio)
            indices_to_reset = np.random.choice(swarm_size, num_to_reset, replace=False)
            for idx in indices_to_reset:
                particles[idx] = np.random.rand(n_points)
                particles[idx] = project_to_feasible(particles[idx], n_points)
                velocities[idx] = np.zeros(n_points)
                pbest[idx] = particles[idx].copy()
                pbest_vals[idx] = evaluate_h(particles[idx], dx)
            last_improvement = iteration
            
    # Final evaluation of global best
    final_c5 = global_best_val
    
    return (global_best.tolist(), final_c5, n_points)
