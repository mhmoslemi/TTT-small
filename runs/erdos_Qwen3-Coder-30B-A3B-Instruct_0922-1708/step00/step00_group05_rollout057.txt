import numpy as np
import cvxpy as cp
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use initial construction if available, else create random one
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
    else:
        n_points = 70
        h0 = np.random.rand(n_points)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    
    dx = 2.0 / n_points
    best_h = h0.copy()
    best_C = evaluate_erdos_solution(h0, dx)
    
    start_time = time.time()
    max_iterations = 1000
    iterations = 0
    
    # Try multiple random initializations
    for trial in range(5):
        if time.time() - start_time > budget_s * 0.9:
            break
        # Generate random perturbed starting point
        h = h0 + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        
        # Try multiple iterations
        for _ in range(5):
            if time.time() - start_time > budget_s * 0.95:
                break
            
            try:
                # Solve QCQP
                h_var = cp.Variable(n_points)
                C = cp.Variable()
                
                # Constraints
                constraints = [h_var >= 0, h_var <= 1]
                constraints.append(cp.sum(h_var) == n_points / 2)
                
                # Overlap constraints
                for k in range(-n_points+1, n_points):
                    # Correlation for each lag
                    if k >= 0:
                        corr = cp.sum(cp.multiply(h_var[:n_points-k], 1 - h_var[k:]))
                    else:
                        corr = cp.sum(cp.multiply(h_var[-k:], 1 - h_var[:n_points+k]))
                    constraints.append(corr <= C * dx)
                
                # Objective
                objective = cp.Minimize(C)
                prob = cp.Problem(objective, constraints)
                
                # Solve
                prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=1000)
                
                if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                    h_new = h_var.value
                    if h_new is not None:
                        # Validate
                        if np.all(h_new >= 0) and np.all(h_new <= 1):
                            C_val = prob.value
                            if C_val < best_C:
                                best_C = C_val
                                best_h = h_new.copy()
            except:
                pass
            
            # Perturb slightly for next iteration
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            iterations += 1
            if iterations >= max_iterations:
                break
    
    return (best_h, best_C, n_points)

def evaluate_erdos_solution(h, dx):
    # Compute C5 using full convolution
    corr = correlate(h, 1-h, mode="full")
    return np.max(corr) * dx
