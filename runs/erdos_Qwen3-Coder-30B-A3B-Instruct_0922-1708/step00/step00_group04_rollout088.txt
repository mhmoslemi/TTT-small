import numpy as np
from scipy.signal import correlate
from cvxpy import *
import time

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute all correlations
    corr = correlate(h, 1 - h, mode='full')
    # The maximum overlap is the maximum of the correlation values
    max_overlap = np.max(corr) * dx
    return max_overlap

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    try:
        h0 = initial_h_values
        n_points = len(h0)
    except:
        # Create a simple initial construction
        n_points = 42
        h0 = np.ones(n_points) * 0.5
        # Ensure integral = 1
        h0 = h0 * 2.0 / n_points
    
    # Start with the initial solution
    best_h = h0.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Define the variables and MIP problem
    # Set a small number of segments to work with
    n_segments = 10
    max_transitions = n_segments - 1
    
    # Try a simpler, more direct approach with gradient descent
    # since MIP is too slow for high resolution
    
    # We'll optimize a step function with up to n_segments
    # Try to keep it as simple as possible
    
    # Direct optimization using scipy.optimize
    from scipy.optimize import differential_evolution
    
    # Objective function to minimize
    def objective(h_vals):
        # Convert h_vals to a proper step function
        h = np.array(h_vals)
        # Make sure h is in [0,1] and sum is correct
        h = np.clip(h, 0.0, 1.0)
        # Adjust so that sum*h_width = 1
        h_width = 2.0 / len(h)
        h = h * (1.0 / np.sum(h) * h_width) / h_width
        c5 = evaluate_erdos_solution(h)
        return c5
    
    def constraint_sum(h_vals):
        h = np.array(h_vals)
        h_width = 2.0 / len(h)
        return np.sum(h) * h_width - 1.0
    
    bounds = [(0., 1.) for _ in range(n_points)]
    
    # Use differential evolution to search the space of step functions
    result = differential_evolution(objective, bounds, seed=seed, maxiter=100, popsize=15)
    
    # Check if we found a better solution and validate it
    if result.success:
        h_opt = np.array(result.x)
        h_opt = np.clip(h_opt, 0.0, 1.0)
        # Adjust to get correct integral
        h_width = 2.0 / n_points
        h_opt = h_opt * (1.0 / np.sum(h_opt) * h_width) / h_width
        c5_new = evaluate_erdos_solution(h_opt)
        if c5_new < best_c5:
            best_h = h_opt
            best_c5 = c5_new
    else:
        # fallback to original
        h_opt = best_h
    
    # Validate final result
    h_final = np.array(h_opt)
    h_final = np.clip(h_final, 0.0, 1.0)
    h_width = 2.0 / n_points
    h_final = h_final * (1.0 / np.sum(h_final) * h_width) / h_width
    
    # Final check
    final_c5 = evaluate_erdos_solution(h_final)
    
    return (h_final.tolist(), final_c5, n_points)
