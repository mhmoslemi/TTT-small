import numpy as np
from scipy import signal
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initial setup
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values, dtype=float)
        n_points = len(h)
    else:
        n_points = 70
        # Create a symmetric initial guess
        h = np.ones(n_points) * 0.5
        # Make it piecewise constant with some variation
        segment_len = n_points // 4
        h[:segment_len] = 0.2
        h[segment_len:2*segment_len] = 0.8
        h[2*segment_len:3*segment_len] = 0.3
        h[3*segment_len:] = 0.7
        # Normalize to sum to n_points/2
        h *= (n_points / 2) / np.sum(h)
    
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    def evaluate_c5(h_func):
        # Use full correlation to get all lags
        corr = signal.correlate(h_func, 1 - h_func, mode='full')
        # Convert to proper integral scale
        c5 = np.max(corr) * dx
        return c5
    
    def project_h(h_vec):
        # Clamp to [0,1]
        h_proj = np.clip(h_vec, 0, 1)
        # Adjust to meet sum constraint
        current_sum = np.sum(h_proj)
        if abs(current_sum - target_sum) > 1e-10:
            # Simple proportional adjustment
            ratio = target_sum / current_sum
            h_proj *= ratio
        return h_proj
    
    def smooth_h(h_vec, window=3):
        # Apply a simple moving average
        if window > len(h_vec):
            window = len(h_vec)
        if window <= 1:
            return h_vec
        weights = np.ones(window) / window
        # Pad for boundary conditions
        padded = np.pad(h_vec, window//2, mode='edge')
        smoothed = np.convolve(padded, weights, mode='valid')
        return smoothed[:len(h_vec)]
    
    # Initialize best
    h_best = h.copy()
    c5_best = evaluate_c5(h_best)
    
    patience = 0
    max_patience = 100
    min_improvement = 1e-6
    
    # Perturbation parameters
    delta = 0.05
    num_trials = 100
    
    iteration = 0
    while time.time() - start_time < budget_s - 0.1:
        improved = False
        for _ in range(num_trials):
            if time.time() - start_time > budget_s - 0.1:
                break
                
            # Create perturbed version
            h_new = h_best.copy()
            # Select random indices to modify
            num_modify = max(1, n_points // 10)
            indices = np.random.choice(n_points, size=num_modify, replace=False)
            for idx in indices:
                # Perturb with small random step
                step = np.random.uniform(-delta, delta)
                h_new[idx] += step
            
            # Apply smoothing to keep structure
            h_new = smooth_h(h_new, window=3)
            
            # Project back to feasible region
            h_new = project_h(h_new)
            
            # Evaluate
            c5_new = evaluate_c5(h_new)
            
            # Accept if improvement
            if c5_new < c5_best - min_improvement:
                h_best = h_new
                c5_best = c5_new
                improved = True
                patience = 0
            elif c5_new < c5_best + min_improvement:
                # Accept with small probability even if not better
                if np.random.rand() < 0.05:
                    h_best = h_new
                    c5_best = c5_new
                    patience = 0
                    improved = True
        
        patience += 1
        iteration += 1
        if not improved:
            patience += 1
        else:
            patience = 0
            
        if patience > max_patience:
            break
    
    return (h_best.tolist(), c5_best, n_points)
