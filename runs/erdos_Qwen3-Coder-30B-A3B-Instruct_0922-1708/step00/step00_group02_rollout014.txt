import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 45
    dx = 2.0 / n_points
    target_sum = n_points / 2.0
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        h = np.clip(h, 0, 1)
    
    # Normalize
    h = h * target_sum / np.sum(h)
    
    # Evaluation function
    def evaluate_C5(h):
        # Compute all cross-correlations for all shifts
        cross_corr = np.correlate(h, 1 - h, mode='full')
        # The maximum is over all lags
        max_corr = np.max(cross_corr)
        return max_corr * dx
    
    # Constraints and projection
    def project(h):
        h = np.clip(h, 0, 1)
        current_sum = np.sum(h)
        if current_sum > 0:
            h = h * target_sum / current_sum
        return h
    
    # Gradient computation - approximate gradient of the max element
    def compute_grad(h, max_shift_idx):
        # For a fixed shift, compute the gradient
        n = len(h)
        grad = np.zeros(n)
        
        # For shift k, we have integral h(x) * (1 - h(x+k))
        # So partial derivative w.r.t. h(x) is (1 - h(x+k)) where x+k is within bounds
        # For simplicity, we only consider the case when grad is non-zero
        for i in range(n):
            j = i + max_shift_idx - n + 1
            if j >= 0 and j < n:
                grad[i] = (1 - h[j]) * dx
        
        return grad
    
    best_h = h.copy()
    best_c5 = evaluate_C5(h)
    
    # Phase 1: Genetic Algorithm
    # Early termination condition: 70% of time
    ga_budget = 0.7 * budget_s
    ga_end_time = start_time + ga_budget
    
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    
    population = []
    fitness = []
    
    # Initialize population
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            new_h = np.array(initial_h_values)
        else:
            new_h = 0.5 * (1 + np.sin(2 * np.pi * np.random.rand(n_points) / 1.0))
            new_h = np.clip(new_h, 0, 1)
        new_h = new_h * target_sum / np.sum(new_h)
        population.append(new_h)
        fitness.append(evaluate_C5(new_h))
    
    # Sort by fitness
    sorted_indices = np.argsort(fitness)
    population = [population[i] for i in sorted_indices]
    fitness = [fitness[i] for i in sorted_indices]
    
    # Evolution loop
    generation = 0
    while time.time() < ga_end_time and generation < 1000:
        # Elitism: keep best 20%
        elite_count = max(1, pop_size // 5)
        elite = population[:elite_count]
        elite_fitness = fitness[:elite_count]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_count:
            # Selection (tournament)
            idx1 = np.random.randint(0, pop_size//2)
            idx2 = np.random.randint(0, pop_size//2)
            parent1 = population[idx1]
            parent2 = population[idx2]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
            
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
            
            # Clip and rescale
            child1 = project(child1)
            child2 = project(child2)
                
            offspring.extend([child1, child2])
            if len(offspring) >= pop_size - elite_count:
                break

        # Evaluate new population
        all_population = elite + offspring[:pop_size - elite_count]
        all_fitness = elite_fitness + [evaluate_C5(h) for h in offspring[:pop_size - elite_count]]
        
        # Sort and keep best
        sorted_indices = np.argsort(all_fitness)
        population = [all_population[i] for i in sorted_indices]
        fitness = [all_fitness[i] for i in sorted_indices]
        
        # Update best solution
        current_best_c5 = fitness[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[0].copy()
            
        generation += 1
        
    # Phase 2: Gradient Descent on best solution
    # Remaining time for GD
    gd_budget = budget_s - (time.time() - start_time)
    gd_end_time = start_time + gd_budget
    
    # Start from best solution of GA phase
    current_h = best_h.copy()
    prev_c5 = best_c5
    
    # Optimizer counter
    no_improvement_count = 0
    max_no_improvement = 100
    
    # Apply gradient descent to optimize the maximum
    # This gradient decent simply looks at the maximum shift and minimizes that
    iterations = 0
    while time.time() < gd_end_time and no_improvement_count < max_no_improvement:
        # Calculate current cross correlation
        cross_corr = np.correlate(current_h, 1 - current_h, mode='full')
        max_shift_idx = np.argmax(cross_corr)
        max_lag_value = cross_corr[max_shift_idx]
        # Minimize the max cross-correlation by descent on its derivative
        
        # Use the gradient of max cross-correlation to update
        grad = compute_grad(current_h, max_shift_idx)
        
        # Standard gradient descent step
        step_size = 0.01
        current_h -= step_size * grad
        
        # Project to constraints
        current_h = project(current_h)
        
        # Re-evaluate C5
        new_c5 = evaluate_C5(current_h)
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = current_h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        iterations += 1
    
    # Return final result
    return (best_h, best_c5, n_points)
