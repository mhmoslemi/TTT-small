import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 70
    dx = 2.0 / n_points
    target_sum = n_points / 2
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
    else:
        h = np.full(n_points, 0.5 * (2.0 / n_points))
    
    # Normalize to sum = target_sum
    h = h * target_sum / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # SDP approach to minimize C5
    try:
        # Define variables
        x = cp.Variable(n_points)
        Z = cp.Variable((n_points, n_points))
        t = cp.Variable()
        
        # Constraints
        constraints = [
            x >= 0,
            x <= 1,
            cp.sum(x) == target_sum
        ]
        
        # Z constraints
        for i in range(n_points):
            for j in range(n_points):
                constraints += [
                    Z[i,j] <= x[i],
                    Z[i,j] <= 1 - x[j],
                    Z[i,j] >= x[i] + x[j] - 1
                ]
        
        # Compute correlations
        correlations = []
        for k in range(-n_points+1, n_points):
            idx = np.arange(n_points) - k
            valid = (idx >= 0) & (idx < n_points)
            sum_z = cp.sum(Z[np.arange(n_points)[valid], idx[valid]])
            correlations.append(sum_z)
        
        # Maximize correlation but minimize t
        for corr in correlations:
            constraints.append(corr <= t)
        
        # Objective
        objective = cp.Minimize(t)
        
        # Problem
        prob = cp.Problem(objective, constraints)
        
        # Solve
        prob.solve(solver=cp.MOSEK, verbose=False, time_limit=budget_s)
        
        if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
            x_result = x.value
            # Clip to [0,1]
            x_result = np.clip(x_result, 0, 1)
            # Rescale to sum = target_sum
            x_result = x_result * target_sum / np.sum(x_result)
            
            # Evaluate
            c5 = evaluate_erdos_solution(x_result)
            if c5 < best_c5:
                best_h = x_result
                best_c5 = c5
                
    except Exception as e:
        # fallback to initial + slight optimization
        pass

    # Local optimization 
    if time.time() - start_time < budget_s - 1:
        # Simple gradient descent with bound constraints
        h = best_h.copy()
        step_size = 0.01
        max_iter = 50000
        for i in range(max_iter):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Compute gradient (Approximate via finite differences)
            h_copy = h.copy()
            h_plus = h_copy.copy()
            h_minus = h_copy.copy()
            grad = np.zeros(n_points)
            
            for j in range(n_points):
                h_plus[j] += 1e-4
                h_minus[j] -= 1e-4
                
                if h_plus[j] <= 1 and h_minus[j] >= 0:
                    c5_plus = evaluate_erdos_solution(h_plus)
                    c5_minus = evaluate_erdos_solution(h_minus)
                    grad[j] = (c5_plus - c5_minus) / (2e-4)
                
                h_plus[j] = h_copy[j]
                h_minus[j] = h_copy[j]
                
            # Gradient descent step
            h = h - step_size * grad
            
            # Project to bounds
            h = np.clip(h, 0, 1)
            # Project to sum constraint
            h = h * target_sum / np.sum(h)
            
            # Evaluate
            c5 = evaluate_erdos_solution(h)
            if c5 < best_c5:
                best_h = h.copy()
                best_c5 = c5
                
    return (best_h.tolist(), best_c5, n_points)
