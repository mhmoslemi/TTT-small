import numpy as np
from scipy.optimize import minimize
import time

def evaluate_overlap(h, dx):
    """Evaluate the maximum overlap integral for given h values."""
    # Compute correlation of h with (1-h)
    corr = np.correlate(h, 1 - h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * dx

def project_h(h, dx):
    """Project h onto feasible set: [0,1] and sum(h)*dx = 1."""
    h = np.clip(h, 0, 1)
    # Rescale to meet sum constraint
    h_sum = np.sum(h) * dx
    if h_sum > 0:
        h = h / h_sum * (2.0 / dx)  # We want sum(h)*dx = 1
    return h

def gradient_descent_step(h, dx, learning_rate):
    """Compute one gradient descent step for minimizing maximum overlap."""
    # Compute current value of overlap
    corr_full = np.correlate(h, 1 - h, mode='full')
    max_idx = np.argmax(corr_full)
    max_val = corr_full[max_idx]
    
    # Gradient approximation: compute gradient of max overlap wrt h
    # For each h[i], finite difference approximation of derivative
    grad = np.zeros_like(h)
    epsilon = 1e-6
    for i in range(len(h)):
        h_plus = h.copy()
        h_plus[i] += epsilon
        h_plus = project_h(h_plus, dx)
        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
        max_val_plus = np.max(corr_plus)
        grad[i] = (max_val_plus - max_val) / epsilon
    
    # Update h
    h_new = h - learning_rate * grad
    h_new = project_h(h_new, dx)
    return h_new
