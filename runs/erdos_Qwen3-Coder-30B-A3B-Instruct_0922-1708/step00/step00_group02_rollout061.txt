import numpy as np
from scipy import optimize
import time
import random

def evaluate_erdos_solution(h):
    n = len(h)
    dx = 2.0 / n
    # Compute cross-correlation
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum value of correlation corresponds to the maximum overlap
    max_overlap = np.max(correlation)
    return max_overlap * dx

def project_h(h, n_points):
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to sum to n_points/2
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def initialize_h(n_points):
    if 'initial_h_values' in globals():
        return np.array(initial_h_values)
    else:
        # Initialize with a sinusoidal function
        x = np.linspace(0, 2, n_points)
        h = 0.5 * (1 + np.sin(2 * np.pi * x / 1.0))
        # Ensure sum = n_points/2
        h = project_h(h, n_points)
        return h

def genetic_algorithm_phase(h_init, n_points, budget_s):
    # Genetic algorithm parameters
    pop_size = 50
    mutation_rate = 0.1
    crossover_prob = 0.7
    elitism_ratio = 0.2
    max_generations = 1000
    
    # Initialize population
    population = []
    for _ in range(pop_size):
        individual = h_init + np.random.normal(0, 0.05, n_points)
        individual = project_h(individual, n_points)
        population.append(individual)
        
    # Current best
    best_h = h_init.copy()
    best_C5 = evaluate_erdos_solution(best_h)
    
    time_start = time.time()
    generation = 0
    
    while time.time() - time_start < 0.7 * budget_s and generation < max_generations:
        # Evaluate all individuals
        fitness = []
        for individual in population:
            c5 = evaluate_erdos_solution(individual)
            fitness.append(c5)
            
        # Sort by fitness (lower C5 better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_C5:
            best_h = population[0].copy()
            best_C5 = fitness[0]
            
        # Elitism: keep top 20%
        elite_size = int(elitism_ratio * pop_size)
        elite = population[:elite_size]
        
        # Create new population
        new_population = elite[:]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Select parents
            parent1_idx = np.random.randint(0, int(0.5 * pop_size))
            parent2_idx = np.random.randint(0, int(0.5 * pop_size))
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_prob:
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1, child2 = parent1.copy(), parent2.copy()
                
            # Mutation
            for i in range(n_points):
                if np.random.rand() < mutation_rate:
                    child1[i] += np.random.normal(0, 0.05)
                if np.random.rand() < mutation_rate:
                    child2[i] += np.random.normal(0, 0.05)
                    
            # Project to feasible region
            child1 = project_h(child1, n_points)
            child2 = project_h(child2, n_points)
            
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        generation += 1
    
    return best_h, best_C5

def gradient_descent_phase(h_init, n_points, budget_s):
    # Gradient descent parameters
    max_iter = 5000
    learning_rate = 0.01
    tolerance = 1e-6
    decay_every = 1000
    
    h = h_init.copy()
    last_improvement = 0
    current_iter = 0
    
    time_start = time.time()
    
    while time.time() - time_start < 0.3 * budget_s and current_iter < max_iter:
        # Compute gradient using finite differences
        # We compute gradient of the objective function (C5) for the current worst shift
        dx = 2.0 / n_points
        c5_current = evaluate_erdos_solution(h)
        # Find which shift gives maximum overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        max_shift_idx = np.argmax(correlation)
        # We now need to compute partial derivatives numerically
        eps = 1e-6
        h_grad = np.zeros(n_points)
        
        # Approximate gradient via finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus, n_points)
            c5_plus = evaluate_erdos_solution(h_plus)
            
            h_grad[i] = (c5_plus - c5_current) / eps
        
        # Update h
        h_new = h - learning_rate * h_grad
        h_new = project_h(h_new, n_points)
        
        # Project and check halting condition
        h = h_new
        if current_iter % decay_every == 0 and current_iter > 0:
            learning_rate *= 0.5
            
        current_iter += 1
        
    return h, evaluate_erdos_solution(h)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 45
    h = initialize_h(n_points)
    C5 = evaluate_erdos_solution(h)
    
    # Phase 1: Genetic Algorithm
    h_gen, C5_gen = genetic_algorithm_phase(h.copy(), n_points, budget_s)
    
    # Phase 2: Gradient Descent
    h_gd, C5_gd = gradient_descent_phase(h_gen.copy(), n_points, budget_s)
    
    # Final evaluation
    final_h = h_gd
    final_C5 = C5_gd
    
    # Ensure final solution is feasible
    final_h = project_h(final_h, n_points)
    final_C5 = evaluate_erdos_solution(final_h)
    
    return (final_h, final_C5, n_points)
