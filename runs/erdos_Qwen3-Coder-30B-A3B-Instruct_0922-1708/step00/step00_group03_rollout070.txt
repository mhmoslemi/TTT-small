import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for the given step function h"""
    dx = 2.0 / len(h)
    # Compute the overlap integral for all shifts
    conv = correlate(h, 1 - h, mode="full")
    max_overlap = np.max(conv) * dx
    return max_overlap

def initialize_population(n_points, pop_size, initial_h=None):
    """Initialize population with one or more step functions"""
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
    else:
        # Create a random initial solution
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    # Fill up the rest of population with random solutions
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
        
    return population

def tournament_selection(population, fitness, tournament_size=5):
    """Select a parent using tournament selection"""
    selected = []
    pop_size = len(population)
    for _ in range(pop_size):
        tournament_indices = np.random.choice(pop_size, tournament_size, replace=False)
        tournament_fitness = [fitness[i] for i in tournament_indices]
        winner_idx = tournament_indices[np.argmin(tournament_fitness)]
        selected.append(population[winner_idx].copy())
    return selected

def crossover(parent1, parent2, crossover_prob=0.5):
    """Perform crossover between two parents"""
    if random.random() > crossover_prob:
        return parent1.copy(), parent2.copy()
        
    # Single point crossover
    point = random.randint(1, len(parent1) - 1)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    return child1, child2

def mutate(h, mutation_rate=0.2, step_size=0.1):
    """Mutate a solution with Gaussian noise"""
    if random.random() > mutation_rate:
        return h.copy()
        
    # Select random portion to mutate
    n_points = len(h)
    mutation_count = int(n_points * 0.1)
    indices = np.random.choice(n_points, mutation_count, replace=False)
    
    # Mutate
    h = h.copy()
    for i in indices:
        h[i] += random.uniform(-step_size, step_size)
    
    # Clip to [0,1]
    h = np.clip(h, 0, 1)
    
    # Rescale to satisfy constraint
    total = np.sum(h)
    if abs(total - n_points / 2) > 1e-6:
        alpha = (n_points / 2) / total
        h = h * alpha
        
    return h

def genetic_algorithm(initial_h_values, n_points, pop_size, max_generations, budget_s):
    """Genetic Algorithm for minimizing C5 bound"""
    start_time = time.time()
    population = initialize_population(n_points, pop_size, initial_h_values)
    
    best_h = None
    best_c5 = float('inf')
    no_improvement_count = 0
    
    for generation in range(max_generations):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_erdos_solution(h)
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement_count = 0
                
        no_improvement_count += 1
        if no_improvement_count > 50:
            break
            
        # Selection
        selected = tournament_selection(population, fitness)
        
        # Crossover & Mutation to create new population
        new_population = [best_h.copy()]
        while len(new_population) < pop_size:
            parent1, parent2 = random.sample(selected, 2)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_population.extend([child1, child2])
            
        # Keep top pop_size individuals
        fitness_pairs = list(zip(new_population, fitness))
        fitness_pairs.sort(key=lambda x: x[1])
        population = [h for h, _ in fitness_pairs[:pop_size]]
        
    return best_h, best_c5

def run(seed=42, budget_s=600):
    np.random.seed(seed)
    n_points = 70
    pop_size = 100
    max_generations = 300
    
    # Use the initial solution if available
    try:
        initial_h_values
    except NameError:
        initial_h_values = None
        
    best_h, best_c5 = genetic_algorithm(
        initial_h_values, n_points, pop_size, max_generations, budget_s
    )
    
    return (best_h, best_c5, n_points)
