import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def project_h(h, n_points):
    """Project h to feasible region: [0,1] and sum = n_points/2"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 100  # Start with a reasonable size
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_start = np.array(initial_h_values)
        h_start = project_h(h_start, n_points)
    else:
        # Create a basic construction
        h_start = np.ones(n_points) * 0.5
        h_start = project_h(h_start, n_points)
    
    # PSO parameters
    swarm_size = 50
    max_iter = 500
    w = 0.7
    c1 = c2 = 2.0
    
    # Initialize swarm
    swarm = []
    velocities = []
    personal_best = []
    personal_best_values = []
    
    # Generate initial particles
    for _ in range(swarm_size):
        if 'initial_h_values' in globals() and np.random.rand() < 0.7:
            # Start from initial solution with noise
            h = h_start + np.random.normal(0, 0.05, n_points)
            h = project_h(h, n_points)
        else:
            # Random feasible solutions
            h = np.random.rand(n_points)
            h = project_h(h, n_points)
        swarm.append(h.copy())
        velocities.append(np.zeros(n_points))
        personal_best.append(h.copy())
        personal_best_values.append(evaluate_h(h, dx))
    
    # Global best
    best_idx = np.argmin(personal_best_values)
    global_best = personal_best[best_idx].copy()
    global_best_value = personal_best_values[best_idx]
    
    # Keep track of stagnation
    stagnation_count = 0
    prev_best = global_best_value
    
    start_time = time.time()
    
    # PSO iteration
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        for i in range(swarm_size):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_best[i] - swarm[i]) + 
                            c2 * r2 * (global_best - swarm[i]))
            
            # Update position
            swarm[i] += velocities[i]
            
            # Project to feasible region
            swarm[i] = project_h(swarm[i], n_points)
            
            # Evaluate
            current_value = evaluate_h(swarm[i], dx)
            
            # Update personal best
            if current_value < personal_best_values[i]:
                personal_best[i] = swarm[i].copy()
                personal_best_values[i] = current_value
                
                # Update global best
                if current_value < global_best_value:
                    global_best = swarm[i].copy()
                    global_best_value = current_value
                    stagnation_count = 0
                else:
                    stagnation_count += 1
                    
        # Diversity injection if stagnating
        if stagnation_count > 25:
            for i in range(int(0.2 * swarm_size)):
                idx = np.random.randint(0, swarm_size)
                swarm[idx] = np.random.rand(n_points)
                swarm[idx] = project_h(swarm[idx], n_points)
                personal_best[idx] = swarm[idx].copy()
                personal_best_values[idx] = evaluate_h(swarm[idx], dx)
            stagnation_count = 0
            
        # Early stopping
        if np.abs(prev_best - global_best_value) < 1e-6:
            stagnation_count += 1
        else:
            stagnation_count = 0
        prev_best = global_best_value
        
        if stagnation_count > 50:
            break
    
    # Final evaluation
    final_c5 = evaluate_h(global_best, dx)
    
    return (global_best.tolist(), final_c5, n_points)
