import numpy as np
from scipy.signal import correlate
import time
import random

def evaluate_erdos_solution(h):
    # Evaluate the C5 constant for given step function h
    dx = 2.0 / len(h)
    h_array = np.array(h)
    corr = correlate(h_array, 1 - h_array, mode="full")
    max_corr = np.max(corr) * dx
    return max_corr

def initialize_population(n_points, pop_size, initial_h=None):
    # Initialize population with one or more step functions
    population = []
    if initial_h is not None:
        population.append(initial_h.copy())
    
    # Fill the rest with random step functions
    for i in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)  # rescale to sum to n_points/2
        population.append(h)
    
    return population

def mutate(h, rate=0.2, delta=0.1):
    # mutating step function h in place
    h_new = h.copy()
    n_points = len(h_new)
    if np.random.rand() < rate:
        # select indices to mutate
        n_mutate = int(0.1 * n_points)
        indices = np.random.choice(n_points, n_mutate, replace=False)
        for i in indices:
            h_new[i] += np.random.uniform(-delta, delta) * (2.0 / n_points)
            h_new[i] = np.clip(h_new[i], 0, 1)
        # rescale to ensure sum constraint
        scale_factor = (n_points / 2) / np.sum(h_new)
        h_new *= scale_factor
    return h_new

def crossover(h1, h2):
    # uniform crossover between h1 and h2
    n = len(h1)
    split_point = np.random.randint(1, n)
    h_child = np.concatenate([h1[:split_point], h2[split_point:]])
    # rescale
    scale_factor = (n / 2) / np.sum(h_child)
    h_child *= scale_factor
    return h_child

def selection(population, fitness, k=20):
    # tournament selection
    selected = []
    n_pop = len(population)
    for _ in range(k):
        idx = np.random.randint(0, n_pop)
        selected.append(population[idx])
    return selected

    best_h = population[best_idx].copy()
    best_c5 = fitness[best_idx]
    
    best_gen = 0
    
    # Main GA loop
    for gen in range(max_generations):
        # Check time
        if time.time() - start_time > budget_s - 30:

        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            best_gen = gen
            
        # Early stopping
        if gen - best_gen > patience:
            break

        new_pop = selected.copy()

            p1, p2 = random.sample(selected, 2)
            child = crossover(p1, p2)
            child = mutate(child)

        # Update population
