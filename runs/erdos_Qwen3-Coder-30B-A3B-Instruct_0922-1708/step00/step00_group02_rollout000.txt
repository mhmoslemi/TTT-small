import numpy as np

import time

def evaluate_overlap(h):
    """Evaluate the maximum overlap integral for h."""
    n = len(h)
    dx = 2.0 / n
    # Compute the full convolution of h with (1-h)
    conv = np.correlate(h, 1 - h, mode="full")
    # The maximum value is the peak of the convolution
    max_overlap = np.max(conv) * dx
    return max_overlap

def project_h(h, n_points):
    """Project h onto the constraint set: 0 <= h[i] <= 1, sum(h) = n_points/2."""
    # Clip to [0, 1]
    h = np.clip(h, 0, 1)
    # Rescale to have the right sum
    current_sum = np.sum(h)
    target_sum = n_points / 2
    if current_sum > 0:
        h = h * target_sum / current_sum
    return h

    h_copy = h.copy()
    
    # Perturb each point slightly to estimate the gradient direction

    eps = 1e-5

    h_new = h_copy - learning_rate * grad
