import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a modified Barker-like pattern
        h = np.zeros(n_points)
        # Use a pattern that has good autocorrelation properties
        # For a 100-point sequence, start with a 4-pulse pattern as a base
        pulse_width = n_points // 4  # about 25 points per pulse
        for i in range(4):
            start_idx = i * pulse_width
            end_idx = start_idx + pulse_width
            if end_idx <= n_points:
                h[start_idx:end_idx] = 1.0
                
        # Normalize to satisfy integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Objective function to minimize (negative because we're minimizing max overlap)
    def objective(h_vec):
        # Rescale h to satisfy integral constraint
        h_scaled = h_vec * (n_points / 2) / np.sum(h_vec)
        h_scaled = np.clip(h_scaled, 0, 1)
        
        correlation = np.correlate(h_scaled, 1 - h_scaled, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Gradient function (simplified for efficiency)
    def gradient(h_vec):
        h_scaled = h_vec * (n_points / 2) / np.sum(h_vec)
        h_scaled = np.clip(h_scaled, 0, 1)
        
        # Compute numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - objective(h_vec)) / eps
            
        return grad
    
    # Constraint function for the integral
    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    # Bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Apply optimization using L-BFGS-B
    try:
        # Optimization with multiple restarts
        num_restarts = 3
        for restart in range(num_restarts):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Try random restarts
            if restart > 0:
                h = np.random.rand(n_points) * 0.5 + 0.25  # Random distribution
                h = h * (n_points / 2) / np.sum(h)
            
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=gradient,
                options={'maxiter': 200, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
                corr = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(corr) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
                    
    except Exception as e:
        # If optimization fails, keep current best
        pass

    # Final evaluation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
