import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the overlap integral for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def rescale_h(h, n_points):
    """Rescale h to satisfy sum constraint"""
    h = np.clip(h, 0, 1)
    current_sum = np.sum(h)
    if current_sum > 0:
        h = h * (n_points / 2) / current_sum
    return h

def crossover(parent1, parent2, rate=0.5):
    """Blend two parents to create offspring"""
    mask = np.random.random(len(parent1)) < rate
    offspring = np.where(mask, parent1, parent2)
    return offspring

def mutate(h, mutation_rate=0.05, noise_level=0.1):
    """Mutate h with Gaussian noise"""
    h_mut = h.copy()
    n_points = len(h)
    n_mutate = int(n_points * mutation_rate)
    indices = np.random.choice(n_points, n_mutate, replace=False)
    noise = np.random.normal(0, noise_level, n_mutate)
    h_mut[indices] += noise
    return rescale_h(h_mut, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = rescale_h(h, n_points)
        population.append(h)
    
    # Fill rest of population
    for i in range(pop_size - 1):
        h = np.random.random(n_points)
        h = rescale_h(h, n_points)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = evaluate_h(best_h, dx)
    
    start_time = time.time()
    generation = 0
    stagnation_count = 0
    stagnation_limit = 100
    
    while time.time() - start_time < budget_s - 1 and stagnation_count < stagnation_limit:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Update best solution
        min_fitness = min(fitness)
        if min_fitness < best_c5:
            best_c5 = min_fitness
            best_h = population[fitness.index(min_fitness)].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Selection: keep top 40%
        sorted_indices = np.argsort(fitness)
        keep_count = int(pop_size * 0.4)
        selected = [population[i] for i in sorted_indices[:keep_count]]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover: create offspring
        while len(new_population) < pop_size:
            p1, p2 = np.random.choice(selected, 2, replace=False)
            offspring = crossover(p1, p2)
            offspring = mutate(offspring)
            new_population.append(offspring)
            
        population = new_population[:pop_size]
        
    return (best_h.tolist(), best_c5, n_points)
