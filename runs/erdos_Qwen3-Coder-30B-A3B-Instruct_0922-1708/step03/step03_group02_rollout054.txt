import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured starting point - piecewise constant
        h = np.zeros(n_points)
        # High values in middle regions
        start1, end1 = int(0.2/dx), int(0.4/dx)
        start2, end2 = int(1.2/dx), int(1.4/dx)
        if end1 > start1:
            h[start1:end1] = 1.0
        if end2 > start2:
            h[start2:end2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    T = 0.1  # initial temperature
    alpha = 0.995  # cooling factor
    lr = 0.01  # learning rate
    perturbation_rate = 0.1  # fraction of points to perturb
    eps = 1e-5
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing step
        h_trial = h.copy()
        n_perturb = int(n_points * perturbation_rate)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            # Add small random perturbation
            perturb = np.random.uniform(-0.1, 0.1)
            h_trial[i] += perturb
        
        # Clip and rescale
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5 for trial
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(corr_trial) * dx
        
        # Accept with simulated annealing probability
        if c5_trial < best_c5 or random.random() < np.exp(-(c5_trial - best_c5) / T):
            h = h_trial
            c5 = c5_trial
        else:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Gradient-based step
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Apply gradient descent step
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
        
        h = h_new
        c5 = c5_new
        
        # Adaptive learning rate
        if c5_new < c5:
            lr *= 1.01
        else:
            lr *= 0.95
        lr = np.clip(lr, 0.001, 0.02)
        
        # Update temperature
        T *= alpha
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
