import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with Nelder-Mead refinement
    T = 0.1
    cooling = 0.995
    perturb_scale = 0.1
    
    # Annealing phase
    anneal_steps = 2000
    for step in range(anneal_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb randomly
        h_new = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, perturb_scale)
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        # Simulated annealing acceptance
        if c5_new < best_c5 or np.random.random() < math.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= cooling
        
    # Nelder-Mead refinement phase
    def obj_func(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Create initial simplex around best solution
    simplex = [best_h]
    for _ in range(4):
        perturbed = best_h + np.random.normal(0, 0.01, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        simplex.append(perturbed)
        
    # Nelder-Mead iteration
    while time.time() - start_time < budget_s - 1:
        # Evaluate simplex
        scores = [obj_func(s) for s in simplex]
        sort_idx = np.argsort(scores)
        simplex = [simplex[i] for i in sort_idx]
        scores = [scores[i] for i in sort_idx]
        
        if scores[0] < best_c5:
            best_c5 = scores[0]
            best_h = simplex[0].copy()
        
        # Nelder-Mead reflection, expansion, contraction
        # Simplex vertices are simplex[0] (best) to simplex[4] (worst)
        worst = simplex[4]
        best = simplex[0]
        
        # Compute centroid of best 4
        centroid = np.mean(simplex[:4], axis=0)
        
        # Reflection
        reflected = centroid + centroid - worst
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (n_points / 2) / np.sum(reflected)
        
        r_score = obj_func(reflected)
        
        if scores[0] <= r_score < scores[2]:
            # Accept reflection
            simplex[4] = reflected
        elif r_score < scores[0]:
            # Expansion
            expanded = centroid + 2 * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (n_points / 2) / np.sum(expanded)
            e_score = obj_func(expanded)
            if e_score < r_score:
                simplex[4] = expanded
            else:
                simplex[4] = reflected
        else:
            # Contraction
            if r_score < scores[4]:
                # Outside contraction
                contracted = centroid + 0.5 * (worst - centroid)
            else:
                # Inside contraction
                contracted = centroid - 0.5 * (worst - centroid)
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (n_points / 2) / np.sum(contracted)
            c_score = obj_func(contracted)
            if c_score < scores[4]:
                simplex[4] = contracted
            else:
                # Shrink
                for i in range(1, 5):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])

    return (best_h.tolist(), best_c5, n_points)
