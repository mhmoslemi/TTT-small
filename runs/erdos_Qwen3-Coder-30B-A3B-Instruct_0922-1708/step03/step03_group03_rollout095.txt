import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Precompute all shifts for overlap calculation
    def compute_c5(h_vals):
        h_array = np.array(h_vals)
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        # Only consider non-circular overlaps
        c5_vals = []
        for k in range(1, n_points):
            overlap = np.sum(h_array[:-k] * (1 - h_array[k:]))
            c5_vals.append(overlap * dx)
        return max(c5_vals)
    
    best_h = h.copy()
    best_c5 = compute_c5(h)
    
    start_time = time.time()
    
    # Resolver schedule with decreasing step sizes
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    temperature = 0.1
    num_iterations = 1000
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
        for _ in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
            # Generate candidate
            h_candidate = h + np.random.normal(0, resolver, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            # Re-normalize
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            c5_candidate = compute_c5(h_candidate)
            
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.rand() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
            
            temperature *= 0.99
            if temperature < 1e-6:
                temperature = 1e-6
    
    # Final computation of C5 using the full convolution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
