import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Genetic Algorithm parameters
    population_size = 50
    generations = 300
    elite_size = int(0.2 * population_size)
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = int(0.1 * population_size)
    
    # Initialize population
    population = []
    population.append(h.copy())
    for _ in range(population_size - 1):
        if 'initial_h_values' in globals():
            perturbed = h + np.random.normal(0, 0.05, n_points)
            perturbed = np.clip(perturbed, 0, 1)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        else:
            perturbed = np.random.uniform(0, 1, n_points)
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        population.append(perturbed)
    
    best_overall = h.copy()
    best_overall_c5 = float('inf')
    
    stagnation_counter = 0
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < generations:
        generation += 1
        
        # Evaluate fitness for all population members
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
            if c5 < best_overall_c5:
                best_overall_c5 = c5
                best_overall = individual.copy()
        
        if best_overall_c5 < best_c5:
            best_c5 = best_overall_c5
            best_h = best_overall.copy()
            stagnation_counter = 0
        else:
            stagnation_counter += 1
            
        if stagnation_counter > 100:
            break
            
        # Rank population by fitness
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population through selection, crossover and mutation
        new_population = elites[:]
        
        while len(new_population) < population_size:
            # Tournament selection
            selected = []
            for _ in range(2):
                tournament = random.sample(population, min(tournament_size, len(population)))
                tournament_fitness = [fitness[population.index(ind)] for ind in tournament]
                winner = tournament[np.argmin(tournament_fitness)]
                selected.append(winner)
            
            # Crossover
            if random.random() < crossover_rate and len(selected) == 2:
                parent1, parent2 = selected
                crossover_point = random.randint(1, n_points - 1)
                child = np.zeros(n_points)
                child[:crossover_point] = parent1[:crossover_point]
                child[crossover_point:] = parent2[crossover_point:]
                # Blend crossover
                blend_factor = random.random() * 0.5 + 0.25  # Random blending 0.25 to 0.75
                child = child * (1 - blend_factor) + parent1 * blend_factor
                
                # Mutate
                if random.random() < mutation_rate:
                    for i in range(n_points):
                        if random.random() < 0.1:  # 10% chance to mutate each gene
                            child[i] += np.random.normal(0, 0.02)
                
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
            else:
                # Just mutate an elite
                parent = elites[random.randint(0, len(elites) - 1)]
                child = parent.copy()
                if random.random() < mutation_rate:
                    for i in range(n_points):
                        if random.random() < 0.1:
                            child[i] += np.random.normal(0, 0.02)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:]
        
        # Occasionally add random individuals
        if generation % 10 == 0:
            for _ in range(5):
                new_ind = np.random.uniform(0, 1, n_points)
                new_ind = new_ind * (n_points / 2) / np.sum(new_ind)
                population.append(new_ind)
                if len(population) > population_size:
                    population.pop()
    
    # Final refinement step
    h = best_h.copy()
    for _ in range(100):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        lr = 0.001
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        correlation = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(correlation) * dx
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
