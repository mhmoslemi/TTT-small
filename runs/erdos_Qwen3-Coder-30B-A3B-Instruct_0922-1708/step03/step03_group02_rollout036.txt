import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        particles.append(h0.copy())
    else:
        h0 = np.ones(n_points) * 0.5
        h0 = h0 * (n_points / 2) / np.sum(h0)
        particles.append(h0)
    
    # Generate other particles with small perturbations
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Track global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
                
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            
            # Update velocity
            v = velocities[i]
            pbest = personal_best[i]
            gbest = global_best
            
            v = w * v + c1 * r1 * (pbest - particles[i]) + c2 * r2 * (gbest - particles[i])
            
            # Clamp velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            new_h = particles[i] + v
            
            # Keep within bounds and rescale
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            particles[i] = new_h
            
        # Adaptation: slightly decrease inertia
        if step % 100 == 0:
            w = max(0.7, w * 0.995)
            
        # Occasionally reset some particles to maintain diversity
        if step % 100 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.1:
                    h = np.random.uniform(0.3, 0.7, n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h
                    
    # Final check
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
