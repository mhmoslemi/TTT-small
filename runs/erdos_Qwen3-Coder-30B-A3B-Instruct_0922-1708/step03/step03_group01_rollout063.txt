import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute correlation for all shifts
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the parent construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Tabu search parameters
    tabu_list_size = 50
    tabu_list = set()
    max_perturb_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    aspirate_prob = 0.1
    start_time = time.time()
    iteration = 0
    
    # Simple hash function for tabu list
    def hash_h(h_vec):
        return hashlib.md5(h_vec.tobytes()).hexdigest()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate multiple candidates
        candidates = []
        for _ in range(50):  # Generate 50 candidates
            # Select random subset for perturbation
            indices = np.random.choice(n_points, size=max_perturb_count, replace=False)
            
            # Create new candidate
            candidate_h = h.copy()
            
            # Perturb selected indices
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
                
            # Clip and rescale
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h = candidate_h * scale_factor
            
            # Check if in tabu list
            h_hash = hash_h(candidate_h)
            if h_hash not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5))
                
        if candidates:
            # Find best candidate not in tabu list
            best_candidate = min(candidates, key=lambda x: x[1])
            candidate_h, candidate_c5 = best_candidate
            
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                stagnation_count = 0
            else:
                # Allow aspiration with probability
                if np.random.random() < aspirate_prob:
                    best_h = candidate_h.copy()
                    best_c5 = candidate_c5
                    stagnation_count = 0
                    
            # Update tabu list
            current_hash = hash_h(h)
            if current_hash in tabu_list:
                tabu_list.remove(current_hash)
                
            tabu_list.add(hash_h(best_h))
            if len(tabu_list) > tabu_list_size:
                # Remove oldest (simple approach)
                tabu_list.pop()
                
            h = best_h.copy()
            
            # Update step size if stagnation
            stagnation_count += 1
            if stagnation_count >= max_stagnation:
                step_size *= 1.2  # Increase step size
                stagnation_count = 0
                # Clear tabu list to allow more exploration
                tabu_list.clear()
                
        else:
            # Fallback to random perturbation
            h_new = h.copy()
            idx = np.random.randint(0, n_points)
            h_new[idx] += np.random.uniform(-step_size, step_size)
            h_new = np.clip(h_new, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h_new)
            h_new = h_new * scale_factor
            h = h_new
            
    return (best_h.tolist(), best_c5, n_points)
