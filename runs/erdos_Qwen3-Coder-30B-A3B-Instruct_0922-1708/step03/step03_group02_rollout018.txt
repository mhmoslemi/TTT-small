import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    # Track the best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.995
    max_anneal_steps = 2000
    start_time = time.time()
    
    # Simulated annealing phase
    for step in range(max_anneal_steps):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Create perturbed solution
        h_new = h.copy()
        indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices:
            perturb = np.random.normal(0, 0.1)
            h_new[i] += perturb
            
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            c5 = c5_new
            
        T *= alpha
        
    # Refinement with Nelder-Mead
    def objective(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Nelder-Mead with simplex around current best solution
    simplex_size = min(10, n_points + 1)  # Ensure not larger than feasible
    simplex_points = []
    
    # Initial simplex
    base_point = best_h.copy()
    simplex_points.append(base_point)
    
    # Generate random perturbations
    for i in range(simplex_size - 1):
        perturbed = base_point + np.random.normal(0, 0.05, n_points)
        perturbed = np.clip(perturbed, 0, 1)
        perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
        simplex_points.append(perturbed)
        
    simplex_points = np.array(simplex_points)
    
    # Minimize with Nelder-Mead
    max_iter = 1000
    for iter_count in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate all points
        values = [objective(p) for p in simplex_points]
        
        # Sort by values
        indices = np.argsort(values)
        simplex_points = simplex_points[indices]
        values = np.array(values)[indices]
        
        # Stop if solution is good enough
        if values[0] < best_c5:
            best_c5 = values[0]
            best_h = simplex_points[0].copy()
            
        # Simplex update

        x_bar = np.mean(simplex_points[:-1], axis=0)
        x_r = x_bar + (x_bar - simplex_points[-1])

        if values[0] <= value_r < values[-2]:
            simplex_points[-1] = x_r

            # Expansion
            x_e = x_bar + 2 * (x_bar - simplex_points[-1])
            x_e = np.clip(x_e, 0, 1)
            x_e = x_e * (n_points / 2) / np.sum(x_e)
            value_e = objective(x_e)
            if value_e < value_r:
                simplex_points[-1] = x_e
            else:
                simplex_points[-1] = x_r
        else:
            # Contraction
            x_c = x_bar + 0.5 * (simplex_points[-1] - x_bar)
            x_c = np.clip(x_c, 0, 1)
            x_c = x_c * (n_points / 2) / np.sum(x_c)
            value_c = objective(x_c)
            if value_c < values[-1]:
                simplex_points[-1] = x_c
            else:
                # Shrink
                for i in range(1, len(simplex_points)):
                    simplex_points[i] = simplex_points[0] + 0.5 * (simplex_points[i] - simplex_points[0])
    
    # Final check and return
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure final normalization
    best_h = best_h * (n_points / 2) / np.sum(best_h)
    if np.sum(best_h) != n_points / 2:
        best_h = best_h * (n_points / 2) / np.sum(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
