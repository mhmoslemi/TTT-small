import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using parent construction or a Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a modified Barker-like pattern
        h = np.zeros(n_points)
        # Use a pattern with alternating high/low regions
        for i in range(n_points):
            if i % 4 < 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Objective function for optimization
    def objective(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Gradient computation (approximate)
    def gradient(h_vec):
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (objective(h_plus) - base_obj) / eps
        return grad
    
    # Optimization with restarts
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        iteration += 1
        
        # Current evaluation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Random perturbation to escape local minima
        if iteration % 10 == 0:
            # Add small noise to escape local minima
            noise = np.random.normal(0, 0.01, n_points)
            h = h + noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # L-BFGS-B optimization step
        try:
            # Optimize within bounds [0,1], sum = 1
            bounds = [(0, 1) for _ in range(n_points)]
            
            # This reduces to a simpler step using the known sum constraint
            # Gradient-free approach using scipy's L-BFGS-B
            result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                            options={'maxiter': 50, 'ftol': 1e-9, 'gtol': 1e-9})
            if result.success:
                h = np.clip(result.x, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # If optimization fails, fall back to a small gradient step
                grad = gradient(h)
                h = h - 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # Fallback
            h = h + np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        if iteration % 20 == 0:
            restart_count += 1
    
    # Final evaluation with best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
