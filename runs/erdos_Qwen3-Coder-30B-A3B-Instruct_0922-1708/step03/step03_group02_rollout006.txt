import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a more sophisticated approach: simulated annealing with local refinement
    # Initialize with some randomness to avoid local minima
    h = np.random.rand(n_points) * 0.5 + 0.25
    h = h / np.sum(h) * (n_points / 2)
    
    temp = 1.0
    cooling_rate = 0.999
    min_temp = 1e-6
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def local_refinement(h):
        """Improve the solution using gradient descent with bounds"""
        h = h.copy()
        for _ in range(10):
            # Compute gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            c5 = compute_c5(h)
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / eps
            
            # Update using gradient
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Check if new solution is better
            new_c5 = compute_c5(h)
            if new_c5 < c5:
                c5 = new_c5
            else:
                break
        return h
    
    while time.time() - start_time < budget_s - 1:
        # Annealing step
        h_trial = h.copy()
        
        # Random perturbation
        idx = np.random.randint(n_points)
        perturbation = np.random.normal(0, 0.02)
        h_trial[idx] += perturbation
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Local refinement of the trial solution
        h_trial = local_refinement(h_trial)
        
        c5_trial = compute_c5(h_trial)
        
        # Accept or reject based on temperature
        if c5_trial < compute_c5(h) or np.random.rand() < np.exp(-(c5_trial - compute_c5(h)) / temp):
            h = h_trial
        
        # Update temperature
        temp = max(min_temp, temp * cooling_rate)
        
        # Track best solution
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Periodic refinement to improve quality
        if (time.time() - start_time) % 2 < 0.1:
            h_refined = local_refinement(h)
            c5_refined = compute_c5(h_refined)
            if c5_refined < c5_current:
                h = h_refined
                if c5_refined < best_c5:
                    best_c5 = c5_refined
                    best_h = h.copy()
    
    # Final local refinement
    best_h = local_refinement(best_h)
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
