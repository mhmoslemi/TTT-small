import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Compute C5 for given h function."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Generate multiple candidates via perturbation
        candidates = []
        for k in range(50):
            # Create new candidate
            candidate_h = h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Perturb selected indices
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Ensure feasibility
            candidate_h = np.clip(candidate_h, 0, 1)
            # Rescale
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            
            # Compute hash to check TABU list
            h_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            
            if h_hash not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5, h_hash))
        
        if not candidates:
            # If all candidates are in TABU list, allow aspiration
            if np.random.random() < 0.1:
                # Create a new random perturbed solution that's not TABU
                candidate_h = h.copy()
                indices = np.random.choice(n_points, size=perturbation_count, replace=False)
                for i in indices:
                    delta = np.random.uniform(-step_size, step_size)
                    candidate_h[i] += delta
                candidate_h = np.clip(candidate_h, 0, 1)
                candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
                h_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5, h_hash))
            else:
                continue
        
        # Select best candidate not in TABU list
        best_candidate = min(candidates, key=lambda x: x[1])
        candidate_h, c5, h_hash = best_candidate
        
        # Update best solution if improved
        if c5 < best_c5:
            best_c5 = c5
            best_h = candidate_h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Update TABU list
        if len(tabu_list) >= tabu_length:
            # Remove oldest entry
            tabu_list.pop()
        tabu_list.add(h_hash)
        
        # Update current solution
        h = candidate_h.copy()
        
        # Adaptive step size
        if stagnation_count >= max_stagnation:
            step_size *= 1.2  # Increase step size
            stagnation_count = 0
            tabu_list.clear()  # Clear tabu list to avoid getting stuck

    return (best_h.tolist(), best_c5, n_points)
