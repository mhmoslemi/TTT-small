import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize h using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    T = T0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate neighbor
        h_new = h.copy()
        
        # Select random points to perturb (5% of points)
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        # Perturb selected points
        for i in indices:
            delta = (np.random.random() - 0.5) * 0.2
            h_new[i] += delta
            h_new[i] = np.clip(h_new[i], 0, 1)
            
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability based on temperature
            if np.random.random() < np.exp((current_c5 - new_c5) / T):
                h = h_new
                
        iteration += 1
        T *= alpha
        
    # Refinement phase
    # Increase resolution to 200 points
    n_points_refined = 200
    dx_refined = 2.0 / n_points_refined
    h_refined = np.interp(np.linspace(0, 2, n_points_refined), 
                         np.linspace(0, 2, len(best_h)), best_h)
    h_refined = h_refined * (n_points_refined / 2) / np.sum(h_refined)
    
    # Additional optimization at higher resolution
    for i in range(500):
        if time.time() - start_time > budget_s - 1:
            break
        h_new = h_refined.copy()
        n_perturb = max(1, n_points_refined // 20)
        indices = np.random.choice(n_points_refined, n_perturb, replace=False)
        for j in indices:
            delta = (np.random.random() - 0.5) * 0.1
            h_new[j] += delta
            h_new[j] = np.clip(h_new[j], 0, 1)
        h_new = h_new * (n_points_refined / 2) / np.sum(h_new)
        
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx_refined
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h_new
            
        # Accept based on temperature
        current_c5 = np.max(np.correlate(h_refined, 1 - h_refined, mode='full')) * dx_refined
        if new_c5 < current_c5 or np.random.random() < np.exp((current_c5 - new_c5) / T):
            h_refined = h_new
            
        T *= alpha
        
    # Final check
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * (2.0 / len(best_h))
    
    return (best_h.tolist(), final_c5, len(best_h))
