import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Generate initial swarm
    particles = [h_init.copy()]
    velocities = [np.random.uniform(-0.1, 0.1, n_points)]
    
    for i in range(1, n_particles):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    v_max = 0.2
    
    # Track best particles
    particle_best = [p.copy() for p in particles]
    particle_best_scores = [float('inf')] * n_particles
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        # Update each particle
        for i in range(n_particles):
            # Clip and normalize particle
            h = np.clip(particles[i], 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute score
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (particle_best[i] - particles[i]) + c2 * r2 * (global_best - particles[i])
            
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            particles[i] = particles[i] + v
            
        # Periodically reset divergent particles
        if step % 20 == 0:
            for i in range(1, n_particles):  # Skip first particle (initial)
                if np.random.rand() < 0.1:
                    h = global_best + np.random.normal(0, 0.1, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Update parameters
        if step % 100 == 0:
            w = max(0.8, w - 0.0005)  # Linearly decrease inertia
            
        step += 1
    
    # Final verification
    final_corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
