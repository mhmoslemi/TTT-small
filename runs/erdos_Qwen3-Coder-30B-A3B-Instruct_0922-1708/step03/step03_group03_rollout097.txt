import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Resolvers for different scales of perturbation
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    temperature = 0.1
    perturbation_scale = 0.1
    success_count = 0
    total_count = 0
    
    def overlap_computation(h_vec):
        # Compute cross-correlation manually
        corr = np.zeros(2 * n_points - 1)
        h_vec = np.array(h_vec)
        for k in range(1, n_points):
            # Compute sum h[i] * (1 - h[i + k]) for valid i
            valid_len = n_points - k
            if valid_len > 0:
                corr_full = np.convolve(h_vec, 1 - h_vec, mode='full')
                corr_k = corr_full[n_points - 1 : n_points - 1 + valid_len]
                corr[k] = np.sum(corr_k) * dx
            else:
                corr[k] = 0
        return np.max(corr)
    
    for resolver in resolvers:
        perturbation_scale = resolver
        num_iterations = 1000
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate by adding noise
            noise = np.random.normal(0, perturbation_scale, n_points)
            h_candidate = h + noise
            # Clip to feasible region
            h_candidate = np.clip(h_candidate, 0, 1)
            # Rescale to sum constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap
            c5_candidate = overlap_computation(h_candidate)
            
            # Accept or reject based on improvement or simulated annealing
            if c5_candidate < best_c5:
                h = h_candidate
                best_c5 = c5_candidate
                success_count += 1
                # Reduce perturbation scale slightly to refine
                perturbation_scale *= 0.99
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if temperature > 1e-6:
                    accept_prob = np.exp(-delta / temperature)
                    if np.random.random() < accept_prob:
                        h = h_candidate
                        success_count += 1
                else:
                    # Eventually stop accepting worse solutions
                    pass
                    
            total_count += 1
            
            # Adjust perturbation scale based on success
            if i % 50 == 0 and i > 0:
                if success_count > 10:
                    perturbation_scale *= 1.5
                elif success_count < 5:
                    perturbation_scale *= 0.5
                success_count = 0
                perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
                
            # Decrease temperature
            if i % 100 == 0:
                temperature *= 0.99

    # Final correction
    best_c5 = overlap_computation(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
