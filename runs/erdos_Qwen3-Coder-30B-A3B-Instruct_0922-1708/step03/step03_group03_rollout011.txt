import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Helper to compute C5 from h
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    # Gradient approximation using finite differences
    def compute_gradient(h_vals, eps=1e-5):
        grad = np.zeros_like(h_vals)
        c5_base = compute_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
    
    # Main optimization loop
    iteration = 0
    last_improvement = 0
    while time.time() - start_time < budget_s - 1 and iteration - last_improvement < 200:
        iteration += 1
        
        # Compute current value and check for improvement
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update h using gradient descent
        lr = 0.01 / (1 + iteration / 100)  # Adaptive learning rate
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally try a second approach with CVXPY
        if iteration % 50 == 0:
            try:
                # Try to solve with CVXPY for a potential better solution
                h_vars = cp.Variable(n_points)
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
                # Approximate objective (Maximize the largest overlap)
                # We'll use a heuristic proxy for max correlation
                corr = cp.reshape(h_vars, (n_points, 1))
                corr_reflect = cp.reshape(1 - h_vars, (n_points, 1))
                conv = cp.conv(corr, corr_reflect)
                objective = cp.Minimize(cp.max(conv))  # Using Minimize since we negated terms
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h_cvx = h_vars.value
                    if h_cvx is not None:
                        h_cvx = np.clip(h_cvx, 0, 1)
                        h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                        c5_cvx = compute_c5(h_cvx)
                        if c5_cvx < best_c5:
                            best_c5 = c5_cvx
                            best_h = h_cvx.copy()
                            last_improvement = iteration
            except Exception:
                pass
                
    return (best_h.tolist(), best_c5, n_points)
