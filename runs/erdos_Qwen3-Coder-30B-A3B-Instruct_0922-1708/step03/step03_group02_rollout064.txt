import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init)
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add random particles
    for i in range(n_particles - len(particles)):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    
    # PSO parameters
    omega = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Track personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [float('inf')] * n_particles
    
    # Track global best
    global_best_idx = 0
    global_best_val = float('inf')
    
    start_time = time.time()
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_vals[i]:
                personal_best_vals[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_val:
                global_best_val = c5
                global_best_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            p_best = personal_best[i]
            g_best = particles[global_best_idx]
            
            # Update velocity
            new_v = (omega * v + 
                    c1 * r1 * (p_best - h) + 
                    c2 * r2 * (g_best - h))
            
            # Clamp velocity
            new_v = np.clip(new_v, -max_vel, max_vel)
            velocities[i] = new_v
            
            # Update position
            new_h = h + new_v
            particles[i] = new_h
            
        # Periodic reset to maintain diversity
        if step % 100 == 0:
            for i in range(1, n_particles):  # Skip first particle (initial)
                if np.random.rand() < 0.1:
                    particles[i] = np.random.uniform(0.4, 0.6, n_points)
                    particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Adjust inertia weight if needed
        if step % 500 == 0:
            omega = max(0.8, omega - 0.01)
    
    # Return the best solution found
    best_h = particles[global_best_idx]
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
