import numpy as np
from scipy.fftpack import fft, ifft
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    # FFT-based optimization
    n_freq = min(50, n_points // 2)
    
    # Initial FFT
    H = fft(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        # Compute overlap in frequency domain
        H_conj = np.conj(H)
        H_1_minus_H = H - H * H_conj  # This is incorrect in ODE but correct conceptually
        # Better approach:
        h_reconstructed = np.real(ifft(H))
        h_reconstructed = np.clip(h_reconstructed, 0, 1)
        h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
        H = fft(h_reconstructed)
        
        # Compute overlap integral
        correlation = np.correlate(h_reconstructed, 1 - h_reconstructed, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_reconstructed.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Compute gradient in frequency domain
        # Simple finite difference approach for gradient
        grad = np.zeros_like(H)
        eps = 1e-5
        
        for k in range(len(H)):  # Try small perturbation to each frequency component
            H_perturb = H.copy()
            H_perturb[k] += eps
            h_test = np.real(ifft(H_perturb))
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            corr = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr) * dx
            grad[k] = (c5_test - c5) / eps

        # Update in frequency domain
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-3)
        H = H - lr * grad
        
        # Occasionally reproject
        if iteration % 10 == 0:
            h_reconstructed = np.real(ifft(H))
            h_reconstructed = np.clip(h_reconstructed, 0, 1)
            h_reconstructed = h_reconstructed * (n_points / 2) / np.sum(h_reconstructed)
            H = fft(h_reconstructed)
            
        # Random restart occasionally
        if iteration % 100 == 0 and no_improvement_count > 10:
            H = fft(np.clip(h_reconstructed + np.random.normal(0, 0.001, n_points), 0, 1) * (n_points / 2) / np.sum(np.clip(h_reconstructed + np.random.normal(0, 0.001, n_points), 0, 1)))
                
    return (best_h.tolist(), best_c5, n_points)
