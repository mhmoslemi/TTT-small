import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Create initial population
    population = []
    
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # Fill with random initial solutions
    for _ in range(pop_size - 1):
        h = np.random.uniform(0.2, 0.8, n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h.copy())
    
    # For tracking best solution
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness (ascending)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite individuals
        elite_count = int(elite_rate * pop_size)
        new_population = population[:elite_count]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            def tournament_select():
                indices = np.random.choice(pop_size, tournament_size, replace=False)
                best_idx = indices[np.argmin([fitness[i] for i in indices])]
                return population[best_idx].copy()
            
            # Crossover
            if np.random.random() < cross_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutate children
                if np.random.random() < mut_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child1 += noise
                    child1 = np.clip(child1, 0, 1)
                    child1 = child1 / np.sum(child1) * (n_points / 2)
                    
                if np.random.random() < mut_rate:
                    noise = np.random.normal(0, 0.05, n_points)
                    child2 += noise
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 / np.sum(child2) * (n_points / 2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Just mutate a selected parent
                parent = tournament_select()
                noise = np.random.normal(0, 0.05, n_points)
                child = parent + noise
                child = np.clip(child, 0, 1)
                child = child / np.sum(child) * (n_points / 2)
                new_population.append(child)
        
        population = new_population
    
    # Final evaluation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    
    # Fallback to the last population member
    h = population[0]
    corr = correlate(h, 1 - h, mode='full')
    c5 = np.max(corr) * dx
    return (h.tolist(), c5, n_points)
