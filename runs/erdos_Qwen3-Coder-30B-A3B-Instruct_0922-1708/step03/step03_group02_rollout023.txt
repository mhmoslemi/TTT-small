import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    else:
        h_init = np.ones(n_points) * 0.5
        population.append(h_init.copy())
    
    # Fill population with diverse individuals
    for _ in range(pop_size - 1):
        h = np.random.uniform(0.2, 0.8, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    def compute_c5(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def check_feasibility(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def tournament_selection(pop, fitnesses):
        idxs = np.random.choice(len(pop), tournament_size)
        best = np.argmin([fitnesses[i] for i in idxs])
        return pop[idxs[best]]
    
    def crossover(parent1, parent2):
        if np.random.random() > cross_rate:
            return parent1.copy(), parent2.copy()
        crossover_point = np.random.randint(1, n_points)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        return child1, child2
    
    def mutate(h):
        h = h.copy()
        for i in range(n_points):
            if np.random.random() < mut_rate:
                h[i] += np.random.normal(0, 0.05)
        return check_feasibility(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitnesses = [compute_c5(h) for h in population]
        
        # Track best
        current_best_idx = np.argmin(fitnesses)
        current_best_c5 = fitnesses[current_best_idx]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = population[current_best_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
            
        # Selection
        elite_count = int(elite_rate * pop_size)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        elite = [population[i] for i in elite_indices]
        
        # Create new population
        new_pop = elite.copy()
        
        while len(new_pop) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            new_pop.extend([child1, child2])
        
        # Trim to exact population size
        new_pop = new_pop[:pop_size]
        population = new_pop
        
        generation += 1
    
    if best_h is None:
        # Fallback to best from last population if none found
        fitnesses = [compute_c5(h) for h in population]
        best_idx = np.argmin(fitnesses)
        best_h = population[best_idx]
        best_c5 = fitnesses[best_idx]
    
    return (best_h.tolist(), best_c5, n_points)
