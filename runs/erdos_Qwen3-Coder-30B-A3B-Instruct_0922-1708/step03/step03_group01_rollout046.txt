import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a Frank-Zadoff-like sequence
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = np.abs(np.sin(np.pi * i * (np.sqrt(2) - 1)) * 0.5 + 0.5)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    alpha = 0.01
    tau = 1.0
    max_iter = 1000
    
    # Phase 1: Binary search to narrow down C5
    low = 0.38
    high = 0.4
    while high - low > 1e-4 and time.time() - start_time < budget_s * 0.6:
        c_mid = (low + high) / 2
        # Check feasibility using QCQP
        try:
            # This is a simplified check - actual implementation would solve QCQP
            corr = np.correlate(h, 1 - h, mode='full')
            max_corr = np.max(corr)
            if max_corr * dx <= c_mid:
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
            
    # Phase 2: Gradient-based refinement
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        
        # Compute current overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the shift that produces current max overlap
        max_shift = np.argmax(correlation)
        
        # Compute gradient analytically
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Gradient w.r.t. h[i]
            # The contribution to overlap when shifting by max_shift
            j = (i + max_shift) % n_points
            grad[i] = dx * (1 - h[j]) - dx * h[(i - max_shift) % n_points]
        
        # Gradient descent step
        h_new = h - alpha * grad
        
        # Project to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint ∫ h dx = 1
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step size
        if iteration % 100 == 0:
            alpha *= 0.9
        
        if iteration % 1000 == 0:
            # Occasionally check the C5 to avoid stagnation
            corr = np.correlate(h, 1 - h, mode='full')
            c5_check = np.max(corr) * dx
            if c5_check > best_c5 * 1.1 and alpha > 1e-6:
                alpha *= 0.1
                
    # Downsample to 100 points as final result
    final_h = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, n_points), best_h)
    final_h = np.clip(final_h, 0, 1)
    final_h = final_h * 50.0 / np.sum(final_h)  # Normalize for sum = 50
    
    # Final evaluation
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * 0.02  # dx = 2/100 = 0.02
    
    return (final_h.tolist(), final_c5, 100)
