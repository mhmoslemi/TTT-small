import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured approach
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization
        h = np.zeros(n_points)
        # Place larger values in regions that improve bounds
        for i in range(20, 40):
            h[i] = 0.8
        for i in range(60, 80):
            h[i] = 0.6
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling rate
    lr = 0.01  # Initial learning rate
    eps = 1e-5  # Finite difference step
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_new = h.copy()
        # Perturb ~10% of elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.05)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for new candidate
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5 or np.random.random() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            c5 = c5_new
        else:
            # Recompute for later use
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient Descent Step
        grad = np.zeros_like(h)
        
        # Compute gradient with finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new.copy()
        
        # Recompute C5 after GD step
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update learning rate
        if c5 < best_c5:
            lr *= 1.01
        else:
            lr *= 0.95
        
        # Clip learning rate
        lr = np.clip(lr, 0.001, 0.02)
        
        # Update temperature
        T *= alpha
    
    # Final computation of C5
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
