import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    # Compute the overlap integral C_5
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def project_h(h, n_points, dx):
    # Clip to [0,1] and rescale to sum to n_points/2
    h = np.clip(h, 0, 1)
    scale_factor = (n_points / 2) / np.sum(h)
    h = h * scale_factor
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values for 10% of particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        for _ in range(num_particles // 10):
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        for _ in range(num_particles // 10):
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Initialize remaining particles randomly
    for _ in range(num_particles - len(particles)):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Convert to arrays for efficiency
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # Track best positions
    p_best = particles.copy()
    p_best_vals = np.array([evaluate_h(p, dx) for p in particles])
    
    # Global best
    g_best_idx = np.argmin(p_best_vals)
    g_best = particles[g_best_idx].copy()
    g_best_val = p_best_vals[g_best_idx]
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (p_best[i] - particles[i]) + 
                             c2 * r2 * (g_best - particles[i]))
            
            # Limit velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate
            c5 = evaluate_h(particles[i], dx)
            
            # Update personal best
            if c5 < p_best_vals[i]:
                p_best_vals[i] = c5
                p_best[i] = particles[i].copy()
                
                # Update global best
                if c5 < g_best_val:
                    g_best_val = c5
                    g_best = particles[i].copy()
    
    return (g_best.tolist(), g_best_val, n_points)
