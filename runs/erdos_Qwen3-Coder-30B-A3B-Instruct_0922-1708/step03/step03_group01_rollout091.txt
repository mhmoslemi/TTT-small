import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low, high = 0.38, 0.4
    tolerance = 1e-4
    
    # Precompute shift indices
    shifts = [i for i in range(n_points)]
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        # Formulate QCQP problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # This is a quadratic constraint
            # sum h[i] * (1 - h[i+k]) <= c_mid / dx
            idx1 = np.arange(n_points)
            idx2 = (idx1 + k) % n_points
            expr = cp.sum(cp.multiply(h_var[idx1], 1 - h_var[idx2]))
            constraints.append(expr <= c_mid / dx)
            
        # Objective is to minimize c_mid (not a variable here)
        # We just validate feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                # Verify that h_sol is valid and compute actual C5
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                # Compute actual C5
                corr = np.correlate(h_sol, 1 - h_sol, mode='full')
                actual_c5 = np.max(corr) * dx
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_sol.copy()
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
            
    if best_c5 == float('inf'):
        # fallback to initial solution
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
