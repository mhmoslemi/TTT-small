import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Higher resolution for global search
    dx = 2.0 / n_points
    
    # Initialize h using Frank-Zadoff-like sequence or parent
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Frank-Zadoff-like initialization
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 + 0.5 * np.sin(2 * np.pi * i * 0.37 + np.pi/4)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Binary search with QCQP feasibility checks
    low, high = 0.38, 0.4
    while high - low > 1e-4 and time.time() - start_time < budget_s * 0.6:
        C_mid = (low + high) / 2
        # Formulate QCQP feasibility problem
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0, 
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        for k in range(n_points):
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= C_mid / dx)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status == cp.OPTIMAL:
                low = C_mid
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                high = C_mid
        except:
            high = C_mid
    
    # Phase 2: Local refinement with gradient-based optimization
    h = best_h if best_c5 < float('inf') else h
    alpha = 0.01
    max_iter = 2000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute maximum overlapping shift
        correlation = np.correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(correlation)
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Analytical gradient computation for current max shift
        grad = np.zeros_like(h)
        k = max_shift - (n_points - 1)  # offset for shift
        for i in range(n_points):
            # Gradient w.r.t. h[i]
            j = (i + k) % n_points
            grad[i] = dx * ((1 - h[j]) - h[(i - k) % n_points] if (i - k) >= 0 else dx * (1 - h[j]))
        
        # Update step
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate and trust region updates
        if iteration % 100 == 0:
            alpha *= 0.95
        if iteration % 50 == 0:
            h = h * (n_points / 2) / np.sum(h)  # Rescale
    
    # Downsample to 100 points for final output
    n_output = 100
    dx_output = 2.0 / n_output
    h_output = np.interp(np.linspace(0, 2, n_output), np.linspace(0, 2, n_points), best_h)
    h_output = np.clip(h_output, 0, 1)
    h_output = h_output * (n_output / 2) / np.sum(h_output)
    
    # Final re-evaluation
    correlation_final = np.correlate(h_output, 1 - h_output, mode='full')
    c5_final = np.max(correlation_final) * dx_output
    
    return (h_output.tolist(), c5_final, n_output)
