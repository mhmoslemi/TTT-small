import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.99
    max_iter = 1000
    
    t = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h.copy()
        
        # Perturb a random subset of points
        num_perturb = max(1, int(n_points * 0.05))  # 5% of points
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            # Add small random change
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            # Clamp to [0, 1]
            h_new[i] = max(0, min(1, h_new[i]))
        
        # Rescale to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability based on temperature
            prob = np.exp((current_c5 - new_c5) / t)
            if random.random() < prob:
                h = h_new
                
        # Decrease temperature
        t *= alpha
        
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
