import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Try local optimization around best solution
    try:
        # Try using scipy minimize with bounds and constraint
        def objective(h_vals):
            h = np.array(h_vals)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx
        
        def constraint_fn(h_vals):
            h = np.array(h_vals)
            return np.sum(h) - n_points / 2
            
        bounds = [(0, 1) for _ in range(n_points)]
        cons = {'type': 'eq', 'fun': constraint_fn}
        
        h_init = best_h.copy()
        result = minimize(objective, h_init, method='SLSQP', bounds=bounds, constraints=cons, 
                         options={'maxiter': 100, 'ftol': 1e-6})
        if result.success:
            h_final = np.clip(result.x, 0, 1)
            h_final = h_final * (n_points / 2) / np.sum(h_final)
            correlation = np.correlate(h_final, 1 - h_final, mode='full')
            c5_final = np.max(correlation) * dx
            if c5_final < best_c5:
                best_h = h_final
                best_c5 = c5_final
    except:
        pass
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
