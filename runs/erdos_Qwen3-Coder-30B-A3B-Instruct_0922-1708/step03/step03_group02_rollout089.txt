import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = np.zeros((n_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize particles with diversity
    for i in range(n_particles):
        if i == 0:
            particles[i] = h_init.copy()
        else:
            # Add small random perturbations
            perturbation = np.random.normal(0, 0.05, n_points)
            particles[i] = h_init + perturbation
            particles[i] = np.clip(particles[i], 0, 1)
            # Rescale to meet constraint
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (n_particles, n_points))
    
    # Initialize personal bests
    personal_bests = particles.copy()
    personal_best_scores = np.full(n_particles, float('inf'))
    
    # Initialize global best
    global_best_idx = 0
    global_best_score = float('inf')
    
    # PSO parameters
    w = 1.0  # inertia weight
    c1 = c2 = 2.0  # acceleration coefficients
    v_max = 0.2  # maximum velocity
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Enforce constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_bests[i] = h.copy()
            
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best_idx = i
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (personal_bests[i] - particles[i]) + 
                            c2 * r2 * (particles[global_best_idx] - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Position update
            particles[i] = particles[i] + velocities[i]
        
        # Occasionally reset some particles to maintain diversity
        if step % 50 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.1:
                    particles[i] = np.random.uniform(0.4, 0.6, n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        # Gradually decrease inertia
        w = max(0.8, 1.0 - (step / 10000.0))
    
    # Return best solution found
    best_h = particles[global_best_idx]
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
