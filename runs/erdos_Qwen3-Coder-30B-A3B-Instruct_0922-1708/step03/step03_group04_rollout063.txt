import numpy as np
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx

    def compute_worst_shift_and_grad(h_array):
        # Compute all correlations for all shifts
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        max_k = np.argmax(correlation)
        max_value = correlation[max_k]
        
        # Convert index back to shift value
        # In scipy correlation, the shift corresponds to (max_k - (len(h_array)-1))
        shift = max_k - (len(h_array) - 1)
        
        # Compute gradient w.r.t. h for the worst case shift
        grad = np.zeros_like(h_array)
        for i in range(len(h_array)):
            # For shift k, gradient w.r.t. h[i] is:
            # dx * (1 - h[(i - shift) % len(h)] - h[(i + shift) % len(h)])
            left_idx = (i - shift) % len(h_array)
            right_idx = (i + shift) % len(h_array)
            grad[i] = dx * (1 - h_array[left_idx] - h_array[right_idx])
        
        return shift, grad

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute worst shift and analytic gradient
        worst_shift, grad = compute_worst_shift_and_grad(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))

                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
