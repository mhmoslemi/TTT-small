import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple initial configuration
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 1000
    iteration = 0
    
    # Precompute all lags
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Simple gradient estimation with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Gradient descent update
        lr = 0.002
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally apply more sophisticated optimization
        if iteration % 10 == 0:
            try:
                h_vars = cp.Variable(n_points)
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
                
                # Create a better objective by approximating the max over lags
                objective_vals = []
                for k in range(n_points):
                    # This is an approximation: compute correlation term for specific lags
                    corr_term = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, k)))
                    objective_vals.append(corr_term)
                
                objective = cp.Minimize(cp.max(objective_vals))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False, max_seconds=10)
                
                if prob.status == cp.OPTIMAL:
                    h_new = h_vars.value
                    if h_new is not None:
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        
                        correlation = np.correlate(h_new, 1 - h_new, mode='full')
                        c5_opt = np.max(correlation) * dx
                        
                        if c5_opt < best_c5:
                            best_h = h_new
                            best_c5 = c5_opt
                            h = h_new.copy()
            except:
                pass
                
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
