import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with local refinement
    T = 0.1
    alpha = 0.995
    steps = 0
    max_anneal_steps = 2000
    
    while steps < max_anneal_steps and time.time() - start_time < budget_s - 1:
        # Propose new solution with simulated annealing
        h_new = h.copy()
        # Perturb about 10% of the elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.05)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to enforce constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        # Compute overlap integral for both solutions
        correlation_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation_current) * dx
        
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject based on Metropolis criteria
        if c5_new < c5_current or np.random.rand() < np.exp(-(c5_new - c5_current) / T):
            h = h_new.copy()
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        steps += 1
        T *= alpha
    
    # Local Nelder-Mead refinement
    while time.time() - start_time < budget_s:
        # Start with best solution  
        init_h = best_h.copy()
        
        # Create initial simplex for Nelder-Mead
        simplex = [init_h]
        for _ in range(4):
            perturbed = init_h.copy()
            indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
            for i in indices:
                perturbed[i] += np.random.normal(0, 0.02)
            perturbed = np.clip(perturbed, 0, 1)
            # Normalize to sum = n_points/2
            perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
            simplex.append(perturbed)
        
        # Evaluate function at each simplex vertex
        def objective(x):
            corr = np.correlate(x, 1 - x, mode='full')
            return np.max(corr) * dx
        
        # Run Nelder-Mead step
        # Use scipy minimize with Nelder-Mead
        result = minimize(objective, simplex[0], method='Nelder-Mead', 
                          options={'maxiter': 100, 'disp': False})
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            correlation = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
