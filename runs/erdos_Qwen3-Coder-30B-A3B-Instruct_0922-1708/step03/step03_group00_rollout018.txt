import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with provided construction or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        # Interpolate to target resolution
        x_orig = np.linspace(0, 2, n_points)
        x_new = np.linspace(0, 2, 100)
        h = np.interp(x_new, x_orig, h)
        n_points = 100
    else:
        h = np.ones(100) * 0.5
        n_points = 100

    dx = 2.0 / n_points
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    max_iter = 500
    patience = 100
    
    # Gradient descent refinement parameters
    lr = 0.001
    gd_iter = 200
    
    # Annealing loop
    last_improvement = 0
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Compute overlap
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_num
            
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        corr_new = signal.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            
        T *= alpha
        
        if iter_num - last_improvement > patience:
            break
            
    # Gradient descent refinement
    h = best_h.copy()
    for iter_num in range(gd_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = signal.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update with adaptive learning rate
        h = h - lr * grad
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Recompute loss
        corr = signal.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Decrease learning rate
        if iter_num % 50 == 0 and iter_num > 0:
            lr *= 0.95
    
    return (best_h.tolist(), best_c5, n_points)
