import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    # Constrain sum to exactly n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Multi-resolution search params
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature-based acceptance for simulated annealing
    temperature = 0.1
    min_temperature = 1e-6
    cooling_rate = 0.99
    
    # Track recent acceptance
    recent_accepts = []
    
    start_time = time.time()
    iteration = 0
    success_counter = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Use coarsest resolution first
        resolver_index = min(iteration // 1000, len(resolvers) - 1)
        pert_scale = resolvers[resolver_index]
        
        # Generate perturbed candidate
        h_candidate = h + np.random.normal(0, pert_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute overlap
        c5_candidate = 0
        for k in range(1, n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h_candidate[i] * (1 - h_candidate[i + k])
            overlap *= dx
            c5_candidate = max(c5_candidate, overlap)
        
        # Check if better or accept based on temperature
        if c5_candidate < best_c5:
            h = h_candidate.copy()
            best_c5 = c5_candidate
            success_counter += 1
            recent_accepts.append(1)
        else:
            recent_accepts.append(0)
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if temperature > 0 and np.random.random() < np.exp(-delta / temperature):
                h = h_candidate.copy()
                success_counter += 1
        
        # Update temperature
        if iteration % 100 == 0:
            temperature *= cooling_rate
            temperature = max(temperature, min_temperature)
        
        # Adaptive perturbation scaling
        if len(recent_accepts) >= 50:
            success_ratio = sum(recent_accepts[-50:]) / 50.0
            if success_ratio > 0.2:
                pert_scale *= 1.5
            elif success_ratio < 0.1:
                pert_scale *= 0.5
            pert_scale = max(0.001, min(pert_scale, 1.0))
            recent_accepts = recent_accepts[-50:]
        
        # Ensure we return within time
        if iteration % 500 == 0 and time.time() - start_time > budget_s - 1:
            break
    
    # Final verification of solution
    final_c5 = 0
    for k in range(1, n_points):
        overlap = 0
        for i in range(n_points - k):
            overlap += best_h[i] * (1 - best_h[i + k])
        overlap *= dx
        final_c5 = max(final_c5, overlap)
    
    return (best_h.tolist(), final_c5, n_points)
