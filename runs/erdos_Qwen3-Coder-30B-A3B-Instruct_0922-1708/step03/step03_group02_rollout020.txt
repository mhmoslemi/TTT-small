import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a hybrid approach: start with gradient descent, then switch to
    # genetic algorithm after initial convergence
    ga_start_time = start_time + 100  # Switch to GA after 100s
    
    # Perform gradient descent
    while time.time() - start_time < budget_s - 1 and time.time() < ga_start_time:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    # Now use genetic algorithm to refine
    if time.time() < ga_start_time + 500:
        pop_size = 50
        elite_rate = 0.2
        cross_rate = 0.7
        mut_rate = 0.1
        tournament_size = 5
        
        # Initialize population
        def normalize(h):
            return h * (n_points / 2) / np.sum(h)
            
        population = []
        for i in range(pop_size):
            if i == 0 and 'initial_h_values' in globals():
                h_init = h.copy()
            else:
                h_init = np.random.uniform(0.2, 0.8, n_points)
                h_init = normalize(h_init)
            population.append(h_init)
            
        def compute_c5(h):
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx
        
        # GA main loop
        generation = 0
        patience = 0
        last_best = best_c5
        
        while time.time() - start_time < budget_s - 1 and patience < 100:
            # Compute fitness
            fitness = [compute_c5(h) for h in population]
            
            # Track best
            min_fitness = min(fitness)
            if min_fitness < best_c5:
                best_c5 = min_fitness
                best_h = population[fitness.index(min_fitness)].copy()
                patience = 0
            else:
                patience += 1
                
            # Selection (tournament)
            selected = []
            for _ in range(pop_size):
                tournament = np.random.choice(pop_size, tournament_size)
                winner = tournament[np.argmin([fitness[i] for i in tournament])]
                selected.append(population[winner].copy())
                
            # Elitism
            elite_count = int(elite_rate * pop_size)
            elite_indices = np.argsort(fitness)[:elite_count]
            elite = [population[i].copy() for i in elite_indices]
            
            # Crossover and mutation
            new_population = elite[:]
            
            while len(new_population) < pop_size:
                if np.random.rand() < cross_rate and len(new_population) + 1 < pop_size:
                    # Crossover
                    p1, p2 = np.random.choice(selected, 2, replace=False)
                    point = np.random.randint(1, n_points - 1)
                    child1 = np.concatenate([p1[:point], p2[point:]])
                    child2 = np.concatenate([p2[:point], p1[point:]])
                    
                    # Add mutations
                    for i in range(n_points):
                        if np.random.rand() < mut_rate:
                            child1[i] += np.random.normal(0, 0.05)
                        if np.random.rand() < mut_rate:
                            child2[i] += np.random.normal(0, 0.05)
                    
                    # Clip and rescale
                    child1 = np.clip(child1, 0, 1)
                    child1 = normalize(child1)
                    child2 = np.clip(child2, 0, 1)
                    child2 = normalize(child2)
                    
                    new_population.extend([child1, child2])
                else:
                    # Mutation only
                    parent = np.random.choice(selected)
                    child = parent.copy()
                    for i in range(n_points):
                        if np.random.rand() < mut_rate:
                            child[i] += np.random.normal(0, 0.1)
                    
                    # Clip and rescale
                    child = np.clip(child, 0, 1)
                    child = normalize(child)
                    new_population.append(child)
            
            # Trim to exact size
            population = new_population[:pop_size]
            generation += 1
            
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
