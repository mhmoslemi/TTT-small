import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_generations = 10000
    generation = 0
    
    # GA parameters
    pop_size = 50
    elite_size = int(0.2 * pop_size)
    mutation_rate = 0.1
    crossover_rate = 0.7
    
    # Create initial population
    population = [h.copy()]
    for _ in range(pop_size - 1):
        pert = np.random.normal(0, 0.1, n_points)
        new_h = h + pert
        new_h = np.clip(new_h, 0, 1)
        new_h = new_h * (n_points / 2) / np.sum(new_h)
        population.append(new_h)
    
    stagnation_count = 0
    last_best = float('inf')
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitness = []
        for ind in population:
            correlation = np.correlate(ind, 1 - ind, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            # Random restart
            population = [h.copy()]
            for _ in range(pop_size - 1):
                pert = np.random.normal(0, 0.1, n_points)
                new_h = h + pert
                new_h = np.clip(new_h, 0, 1)
                new_h = new_h * (n_points / 2) / np.sum(new_h)
                population.append(new_h)
            stagnation_count = 0
        
        # Create new population
        new_population = population[:elite_size]
        
        while len(new_population) < pop_size:
            # Tournament selection
            if len(population) >= 2:
                parent1_idx = np.argmin([fitness[random.randint(0, len(population)-1)] for _ in range(3)])
                parent2_idx = np.argmin([fitness[random.randint(0, len(population)-1)] for _ in range(3)])
                
                parent1 = population[parent1_idx]
                parent2 = population[parent2_idx]
                
                if random.random() < crossover_rate:
                    # Uniform crossover
                    mask = np.random.random(n_points) < 0.5
                    child = np.where(mask, parent1, parent2)
                else:
                    child = parent1.copy()
                
                # Mutation
                if random.random() < mutation_rate:
                    mut_indices = np.random.choice(n_points, size=int(mutation_rate * n_points), replace=False)
                    for i in mut_indices:
                        child[i] += np.random.normal(0, 0.05)
                
                # Ensure feasibility
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                
                new_population.append(child)
            
        population = new_population
        generation += 1
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
