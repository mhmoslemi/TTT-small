import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function plus some noise
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0 
        h += np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature for simulated annealing
    temperature = 0.1
    min_temp = 1e-6
    temp_decay = 0.99
    success_count = 0
    total_count = 0
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 2:
            break
            
        num_iterations = 1000
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Perturb h
            h_candidate = h + np.random.normal(0, resolver, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                # Reduce scale slightly for next step
                resolver = max(resolver * 0.9, 0.001)
            else:
                # Simulated annealing acceptance
                delta = c5_candidate - best_c5
                if delta > 0 and np.random.rand() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
                    success_count += 1
            
            total_count += 1
            
            # Adjust temperature
            if (i + 1) % 100 == 0:
                temperature = max(temperature * temp_decay, min_temp)
                
            # Adaptive scale adjustment
            if total_count % 50 == 0 and total_count > 0:
                success_ratio = success_count / total_count
                if success_ratio > 0.2:
                    resolver = min(resolver * 1.5, 1.0)
                elif success_ratio < 0.1:
                    resolver = max(resolver * 0.5, 0.001)
                success_count = 0
                total_count = 0
    
    # Final computation  
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
