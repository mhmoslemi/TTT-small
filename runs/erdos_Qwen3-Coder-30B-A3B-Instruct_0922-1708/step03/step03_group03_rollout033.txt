import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():

        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new

    # Apply additional improvements with binary search approach
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Perform binary search on C5
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    # Store best feasible solution
    best_feasible_h = best_h.copy()
    best_feasible_c5 = best_c5
    
    # Calculate initial feasible solution
    current_c5 = compute_c5(best_h)
    if current_c5 < best_feasible_c5:
        best_feasible_c5 = current_c5
        best_feasible_h = best_h.copy()

    while C_high - C_low > tolerance and time.time() - start_time < budget_s - 1:
        C_test = (C_low + C_high) / 2.0

        # This is a simplified approach: try optimization with a convex relaxation

        h_vars = cp.Variable(n_points)

        # Use a different approach: box constraints with summary constraints
        try:
            # Use CVXPY for constrained optimization  
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))

            prob.solve(solver=cp.MOSEK, verbose=False)

                h_new = h_vars.value

                    # Feasible, try to improve

                else:
                    C_low = C_test
            else:
                C_low = C_test

    final_h = best_feasible_h.copy()
    final_c5 = best_feasible_c5

    if time.time() - start_time < budget_s - 1:

        h = final_h.copy()
        max_iter = 100

            if time.time() - start_time > budget_s - 1:
                break
        
            # Compute gradient
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for j in range(n_points):
                h_plus = h.copy()

                c5_plus = compute_c5(h_plus)
                c5_curr = compute_c5(h)
                grad[j] = (c5_plus - c5_curr) / eps
            
            # Gradient step
            lr = 0.001
            h_new = h - lr * grad
            
            # Projection and rescaling
            h_new = np.clip(h_new, 0, 1)

            h = h_new
            
        # Check final result
        final_c5 = compute_c5(h)
        if final_c5 < best_feasible_c5:
            final_h = h
            final_c5 = final_c5

    return (final_h.tolist(), final_c5, n_points)
