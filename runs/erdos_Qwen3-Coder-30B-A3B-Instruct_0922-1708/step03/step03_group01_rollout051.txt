import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define constraint for sum(h) = n_points / 2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Define bounds for each h[i] in [0, 1]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Constraint dictionary
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Objective function to minimize
    def objective(h):
        # Compute all overlaps
        correlation = np.correlate(h, 1 - h, mode='full')
        # Return negative because we want to minimize
        return -np.max(correlation) * dx
    
    # Gradient of objective function
    def gradient(h):
        # Find which shift gives the maximum overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(correlation)
        
        # Compute gradient for this specific shift
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # j = i + max_k
            j = (i + max_k) % len(h)
            if j < len(h):
                grad[i] = -dx * (1 - h[j])
            # j = i - max_k 
            j = (i - max_k) % len(h)
            if j < len(h):
                grad[i] -= dx * h[j]
        return grad
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-8
    
    # Run optimization with SLSQP
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Solve optimization problem
        result = minimize(objective, h, method='SLSQP', bounds=bounds, 
                          constraints=constraints, jac=gradient, 
                          options={'maxiter': 10, 'ftol': tol})
        
        if result.success:
            h = result.x
            # Ensure constraints are satisfied
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate current solution
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
