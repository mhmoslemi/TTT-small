```python
import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def frank_zadoff_sequence(n):
    """Generate a Frank-Zadoff sequence of length n."""
    # For simplicity, returning a scaled sine wave approximation
    # This is not a true Frank-Zadoff sequence but should provide a good initialization
    return np.sin(np.pi * np.arange(n) * (np.arange(n) + 1) / n)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    n_points = 200  # Higher resolution for global search
    dx = 2.0 / n_points
    low = 0.38
    high = 0.4
    
    # Initialize using the provided parent or Frank-Zadoff sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = frank_zadoff_sequence(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Phase 1: Binary search with QCQP checks
    best_h = h.copy()
    best_c5 = float('inf')
    
    while time.time() - start_time < (budget_s / 2) and high - low > 1e-4:
        C_mid = (low + high) / 2
        # QCQP feasibility check
        h_vars = cp.Variable(n_points)
        constraints = [0 <= h_vars[i] <= 1 for i in range(n_points)]
        constraints.append(cp.sum(h_vars) == n_points / 2)
        
        # Constraint for each shift
        shifted = [np.roll(h_vars, k) for k in range(n_points)]
        shift_constraints = []
        for k in range(n_points):
            right_side = C_mid / dx
            # Sum over all positions for shift k
            overlap = cp.sum(cp.multiply(h_vars, 1 - shifted[k]))
            shift_constraints.append(overlap <= right_side)
        constraints.extend(shift_constraints)
        
        # Dummy objective
        objective = cp.Minimize(0)
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            # Feasible
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                if 'h_vars.value' in locals():
                    temp_h = np.array(h_vars.value)
                    temp_h = np.clip(temp_h, 0, 1)
                    temp_h = temp_h * (n_points / 2) / np.sum(temp_h)
                    c5_temp = 0.0
                    for k in range(n_points):
                        shift = np.roll(temp_h, k)
                        overlap = np.dot(temp_h, shift)
                        c5_temp = max(c5_temp, overlap)
                    c5_temp *= dx
                    if c5_temp < best_c5:
                        best_c5 = c5_temp
                        best_h = temp_h.copy()
                high = C_mid
            else:
                low = C_mid
        except:
            low = C_mid
            
    # Phase 2: Local refinement
    # Downsample to actual final resolution
    final_n_points = 100
    final_dx = 2.0 / final_n_points
    if n_points > final_n_points:
        # Interpolate to final resolution
        h = h[:final_n_points]
        h = np.clip(h, 0, 1)
        h = h * (final_n_points / 2) / np.sum(h)
    else:
        final_n_points = n_points
    
    n_points = final_n_points
    dx = final_dx
    
    # Update best_h if necessary
    if best_c5 == float('inf'):
        best_c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        best_h = h.copy()
        
    # Local refinement with analytical gradients
    alpha = 0.001
    max_iter = 5000
    iteration = 0
    while iteration < max_iter and time.time() - start_time < budget_s - 5:
        correlation = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(correlation)
        max_overlap = correlation[max_idx]
        c5 = max_overlap * dx    # This is max C5
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Analytical gradients
        grad = np.zeros_like(h)
        k = max_idx - (len(h) - 1)  # shift corresponding to max overlap
        
        # Compute gradient for current penalty
        # d/dh[j] of sum_{i} h[i](1 - h[i+k]) = 1 - h[i + k] - h[i - k] if i is valid
        for i in range(len(h)):
            left = (i - k) % len(h)
            right = (i + k) % len(h)
            grad[i] = 1 - h[left] - h[right]
        
        # Gradient descent step
        h_new = h - alpha * grad
        
        # Projection
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            alpha *= 0.99
        
        iteration += 1
    
    # Final evaluation with exact correlation
    corr_shifted = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_shifted) * dx
        
    return (best_h.tolist(), final_c5, n_points)
```
