import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Problem parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute norms for efficiency
    def compute_c5(h_vec):
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Simulated Annealing move
        h_new = h.copy()
        # Select random points to modify (5%)
        indices_to_modify = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        
        # Modify selected points
        for i in indices_to_modify:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        new_c5 = compute_c5(h_new)
        T = T0 * (alpha ** iteration)
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
        
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
