import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a uniform distribution
        h = np.ones(n_points) * (n_points / 2) / n_points
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Compute initial overlap
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    best_c5 = c5
    best_h = h.copy()
    
    max_iterations = 1000
    max_no_improvement = 100
    
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute gradients using finite differences
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Restart with a better distribution
            h = np.ones(n_points) * (n_points / 2) / n_points
            h[:n_points//4] = 2.0
            h[3*n_points//4:] = 2.0
            h = h * (n_points / 2) / np.sum(h)
            no_improvement_count = 0
            
        h = h_new
        c5 = c5_new
        iteration += 1
        
        # Occasionally, try a cvxpy-based solution on a subset for a small improvement
        if iteration % 10 == 0:
            try:
                # Only use cvxpy on a certain resolution to save time
                size = min(30, n_points)
                indices = np.linspace(0, n_points-1, size, dtype=int)
                h_sub = h[indices]
                h_vars = cp.Variable(size)
                constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == size / 2]
                objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (size, 1)))))
                prob = cp.Problem(objective, constraints)
                prob.solve(solver=cp.CBC, verbose=False)
                if prob.status == cp.OPTIMAL:
                    h_cvx = h_vars.value
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (size / 2) / np.sum(h_cvx)
                    # Interpolate back to full resolution
                    h_full = np.interp(np.linspace(0, n_points-1, n_points), indices, h_cvx)
                    h_full = h_full * (n_points / 2) / np.sum(h_full)
                    corr_cvx = np.correlate(h_full, 1 - h_full, mode='full')
                    c5_cvx = np.max(corr_cvx) * dx
                    if c5_cvx < best_c5:
                        best_c5 = c5_cvx
                        best_h = h_full
            except:
                pass
                
    return (best_h.tolist(), best_c5, n_points)
