import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Compute overlaps for all lags
        overlaps = []
        for i in range(n_points):
            # Overlap at lag i
            if i == 0:
                overlap = cp.sum(h_relaxed * (1 - h_relaxed))
            else:
                overlap = cp.sum(h_relaxed[i:] * (1 - h_relaxed[:n_points-i]))
            overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(cp.stack(overlaps)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=1000, abstol=1e-6)
            if prob.status == cp.OPTIMAL:
                h_relaxed_vals = h_relaxed.value
            else:
                # Fallback if not optimal
                h_relaxed_vals = h
        except:
            h_relaxed_vals = h
            
        # Rounding
        h_rounded = (h_relaxed_vals >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        target_sum = n_points / 2
        
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            diff = int(current_sum - target_sum)
            indices = np.argsort(h_relaxed_vals)
            for i in range(diff):
                h_rounded[indices[i]] = 0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            diff = int(target_sum - current_sum)
            indices = np.argsort(-h_relaxed_vals)
            for i in range(diff):
                h_rounded[indices[i]] = 1
                
        # Rescale to exact sum
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Evaluate C5
        correlation = np.correlate(h_rounded, 1 - h_rounded, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
        
        # Perturb to escape local minima
        h = h_rounded + np.random.normal(0, 0.001, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check with refined evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
