import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.extend([h_init.copy() for _ in range(num_particles//10)])
        velocities.extend([np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles//10)])
    else:
        h_init = np.ones(n_points) * 0.5
        particles.extend([h_init.copy() for _ in range(num_particles//10)])
        velocities.extend([np.random.uniform(-0.1, 0.1, n_points) for _ in range(num_particles//10)])
        
    # Fill remaining particles with random feasible values
    remaining = num_particles - len(particles)
    for _ in range(remaining):
        h = np.random.uniform(0, 1, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Track bests
    p_best = particles.copy()
    p_best_values = np.full(num_particles, np.inf)
    g_best_idx = 0
    g_best_value = np.inf
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            corr = np.correlate(h, 1 - h, mode='full') * dx
            c5 = np.max(corr)
            
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
                
            if c5 < g_best_value:
                g_best_value = c5
                g_best_idx = i
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (p_best[g_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project onto feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Return best solution
    best_h = p_best[g_best_idx]
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
