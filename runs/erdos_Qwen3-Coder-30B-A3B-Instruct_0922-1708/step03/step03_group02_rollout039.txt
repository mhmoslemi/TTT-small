import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add random particles
    for i in range(n_particles - len(particles)):
        h = np.random.rand(n_points) * 0.2 + 0.4
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    # Initialize velocities
    for i in range(n_particles):
        v = np.random.rand(n_points) * 0.2 - 0.1
        velocities.append(v)
    
    # PSO parameters
    w = 1.0          # inertia weight
    c1 = c2 = 2.0    # acceleration coefficients
    
    # Best positions
    personal_best = [p.copy() for p in particles]
    personal_best_vals = [float('inf')] * n_particles
    
    global_best = particles[0].copy()
    global_best_val = float('inf')
    
    start_time = time.time()
    step = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            
            # Evaluate objective
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_vals[i]:
                personal_best[i] = h.copy()
                personal_best_vals[i] = c5
                
                # Update global best if needed
                if c5 < global_best_val:
                    global_best = h.copy()
                    global_best_val = c5
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -0.2, 0.2)
            velocities[i] = v
            
            # Update position
            h = h + v
            particles[i] = h
            
        # Handle stagnation
        if stagnation_count > max_stagnation:
            # Reset some particles
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                h = np.random.rand(n_points) * 0.2 + 0.4
                h = h / np.sum(h) * (n_points / 2)
                particles[idx] = h
                velocities[idx] = np.random.rand(n_points) * 0.2 - 0.1
            stagnation_count = 0
            
        # Adaptive inertia weight
        if step % 100 == 0:
            w = max(0.8, w * 0.99)
    
    # Final evaluation
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
