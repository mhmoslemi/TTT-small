import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Annealing parameters
    T = 0.1
    alpha = 0.995
    perturbation_std = 0.1
    steps = 0
    max_annealing_steps = 2000
    
    # Nelder-Mead parameters
    simplex_size = 5
    max_nm_steps = 1000
    
    # Simulated Annealing phase
    while steps < max_annealing_steps and time.time() - start_time < budget_s - 1:
        # Perturb a subset of elements
        idxs = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_new = h.copy()
        h_new[idxs] += np.random.normal(0, perturbation_std, len(idxs))
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
                
        T *= alpha
        steps += 1
    
    # Nelder-Mead optimization phase
    def obj_func(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Create initial simplex
    simplex = [h.copy()]
    for i in range(simplex_size - 1):
        pert = np.random.normal(0, 0.01, n_points)
        h_pert = h + pert
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)
    
    # Nelder-Mead loop
    nm_steps = 0
    while time.time() - start_time < budget_s - 1 and nm_steps < max_nm_steps:
        # Evaluate all vertices
        vals = [obj_func(v) for v in simplex]
        
        # Order simplex by function values
        indices = np.argsort(vals)
        simplex = [simplex[i] for i in indices]
        vals = [vals[i] for i in indices]
        
        # Compute centroid
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        xr = centroid + (centroid - simplex[-1])
        xr = np.clip(xr, 0, 1)
        xr = xr * (n_points / 2) / np.sum(xr)
        val_r = obj_func(xr)
        
        if vals[0] <= val_r < vals[-2]:
            simplex[-1] = xr
        elif val_r < vals[0]:
            # Expansion
            xe = centroid + 2 * (xr - centroid)
            xe = np.clip(xe, 0, 1)
            xe = xe * (n_points / 2) / np.sum(xe)
            val_e = obj_func(xe)
            if val_e < val_r:
                simplex[-1] = xe
            else:
                simplex[-1] = xr
        else:
            # Contraction
            xc = centroid + 0.5 * (simplex[-1] - centroid)
            xc = np.clip(xc, 0, 1)
            xc = xc * (n_points / 2) / np.sum(xc)
            val_c = obj_func(xc)
            if val_c < vals[-1]:
                simplex[-1] = xc
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        
        # Update best if improved
        if vals[0] < best_c5:
            best_c5 = vals[0]
            best_h = simplex[0].copy()
            
        nm_steps += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
