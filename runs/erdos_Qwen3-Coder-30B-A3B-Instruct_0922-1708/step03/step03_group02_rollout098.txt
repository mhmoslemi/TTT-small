import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize population
    population = []
    population.append(h_init)
    
    # Fill remaining population with random perturbations
    for _ in range(pop_size - 1):
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Track best solution
    best_h = h_init.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improve_count = 0
    max_no_improve = 100
    
    while time.time() - start_time < budget_s - 1 and no_improve_count < max_no_improve:
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
        
        # Create new population
        new_population = []
        
        # Elite preservation
        elite_count = int(elite_rate * pop_size)
        for i in range(elite_count):
            new_population.append(population[i].copy())
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = tournament_selection(population, fitness, tournament_size)
            parent2_idx = tournament_selection(population, fitness, tournament_size)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.random() < cross_rate:
                child1, child2 = crossover(parent1, parent2)
            else:
                child1, child2 = parent1.copy(), parent2.copy()
            
            # Mutation
            if np.random.random() < mut_rate:
                child1 = mutate(child1)
            if np.random.random() < mut_rate:
                child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        generation += 1
    
    # Final evaluation
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)

def tournament_selection(population, fitness, k):
    """Select an individual using tournament selection"""
    indices = np.random.choice(len(population), size=k, replace=False)
    best_idx = indices[np.argmin([fitness[i] for i in indices])]
    return best_idx

def crossover(parent1, parent2):
    """Single point crossover"""
    point = np.random.randint(1, len(parent1))
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    # Ensure feasibility
    child1 = np.clip(child1, 0, 1)
    child1 = child1 * (len(child1) / 2) / np.sum(child1)
    child2 = np.clip(child2, 0, 1)
    child2 = child2 * (len(child2) / 2) / np.sum(child2)
    
    return child1, child2

def mutate(h):
    """Gaussian mutation with normalization"""
    # Apply mutation to selected indices
    indices = np.random.choice(len(h), size=int(0.1 * len(h)), replace=False)
    for i in indices:
        h[i] += np.random.normal(0, 0.1)
    
    # Ensure constraints
    h = np.clip(h, 0, 1)
    h = h * (len(h) / 2) / np.sum(h)
    return h
