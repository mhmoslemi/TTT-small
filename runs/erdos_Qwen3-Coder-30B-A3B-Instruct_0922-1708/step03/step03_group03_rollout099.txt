import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        # Add slight random noise to break symmetry
        h += np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Resolver levels for perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature schedule
    temperature = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    
    # For perturbation adaptation
    last_improvements = []
    adapt_threshold_low = 0.1
    adapt_threshold_high = 0.2
    adapt_factor_up = 1.5
    adapt_factor_down = 0.5
    
    for resolver in resolvers:
        perturbation_scale = resolver
        num_iterations = 1000
        
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate
            h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute the overlap integral
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if delta < 0 or np.random.rand() < np.exp(-delta / max(temperature, min_temp)):
                h = h_candidate.copy()
                if c5_candidate < best_c5:
                    best_c5 = c5_candidate
                    best_h = h.copy()
                    perturbation_scale *= 0.9  # Adaptive refinement
                    if perturbation_scale < 1e-6:
                        perturbation_scale = 1e-6
            else:
                # Keep current h
                pass
                
            # Update temperature
            if i % 100 == 0:
                temperature *= temp_decay
                if temperature < min_temp:
                    temperature = min_temp
                    
            # Track improvements for adaptation
            if i % 10 == 0 and len(last_improvements) > 0:
                success = sum(1 for x in last_improvements[-50:] if x < 0) / min(50, len(last_improvements))
                if success > adapt_threshold_high:
                    perturbation_scale *= adapt_factor_up
                elif success < adapt_threshold_low:
                    perturbation_scale *= adapt_factor_down
                perturbation_scale = np.clip(perturbation_scale, 1e-3, 1.0)
    
    # Final check using larger number of iterations if we have time
    remaining_time = budget_s - (time.time() - start_time)
    if remaining_time > 5 and best_c5 > 0.3808:
        # Perform a few more iterations at lower perturbation scale
        perturbation_scale = 0.0005
        for _ in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(corr) * dx
            
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()

    # Verify final solution
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
