import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    while (high - low > tolerance) and (time.time() - start_time < budget_s - 1):
        C_mid = (low + high) / 2.0
        
        # Create optimization variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute the overlap term for shift k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= C_mid / dx)
        
        # Objective: minimize C5 (implicitly handled via feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h_sol = np.clip(h_sol, 0, 1)
                    h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                    # Evaluate C5 for this solution
                    correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                    c5_eval = np.max(correlation) * dx
                    if c5_eval < best_c5:
                        best_c5 = c5_eval
                        best_h = h_sol.copy()
                    high = C_mid
                else:
                    low = C_mid
            else:
                low = C_mid
        except Exception:
            low = C_mid
        
        if time.time() - start_time > budget_s - 1:
            break
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
