import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a modified Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a modified Barker-like pattern to start
        h = np.zeros(n_points)
        # Use a pattern that roughly mimics a Barker sequence for better overlap properties
        # We will place two pulses, one at the start and one at the end, with proper normalization
        h[:n_points//4] = 1.0
        h[3*n_points//4:] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def objective(h_vec):
        # Ensure h_vec is normalized
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        
        # Compute the overlap integral
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    def constraint_func(h_vec):
        # Normalization constraint
        return np.sum(h_vec) * dx - 1.0
    
    # Set up bounds for the variables
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Set up constraints
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Use L-BFGS-B with bounds and constraints for optimization
    # Try several restarts to avoid local minima
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Current objective value
        current_c5 = objective(best_h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = best_h.copy()
            
        # Attempt a local optimization step
        result = minimize(
            objective, 
            best_h,
            method='L-BFGS-B', 
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 50}
        )
        
        if result.success:
            h_new = result.x
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Update best if better
            c5_new = objective(h_new)
            if c5_new < best_c5:
                best_h = h_new
                best_c5 = c5_new
                
            # Perturb a bit to avoid getting stuck in local optima
            h_pert = h_new + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            c5_pert = objective(h_pert)
            if c5_pert < best_c5:
                best_h = h_pert
                best_c5 = c5_pert
        
        # Occasionally, explore with a random step
        if iteration % 10 == 0:
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand) 
            h_rand = np.clip(h_rand, 0, 1)
            c5_rand = objective(h_rand)
            if c5_rand < best_c5:
                best_h = h_rand
                best_c5 = c5_rand
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
