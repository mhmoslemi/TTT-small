import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Convex relaxation approach
        h_relaxed = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
        
        # Compute overlap for each lag to get max
        overlaps = []
        for k in range(n_points):
            overlap = 0
            for i in range(n_points - k):
                overlap += h_relaxed[i] * (1 - h_relaxed[i + k])
            overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_new = h_relaxed.value
            else:
                # Fallback to gradient descent if convex solves fail
                h_new = h.copy()
                grad = np.zeros_like(h)
                eps = 1e-5
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                lr = 0.005
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
        except:
            # Fallback to gradient descent if convex fails
            h_new = h.copy()
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            lr = 0.005
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Rounding and adjustment
        h_rounded = (h_new >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        if current_sum > n_points / 2:
            # Need to reduce sum
            diff = int(current_sum - n_points / 2)
            to_zero = np.argsort(h_new)[:diff]
            h_rounded[to_zero] = 0
        elif current_sum < n_points / 2:
            # Need to increase sum
            diff = int(n_points / 2 - current_sum)
            to_one = np.argsort(h_new)[-diff:]
            h_rounded[to_one] = 1
        
        h = h_rounded.copy()
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
