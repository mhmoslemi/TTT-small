import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    max_generations = 120
    crossover_rate = 0.5
    mutation_rate = 0.2
    elite_size = 20
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            h = np.random.uniform(0, 1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def evaluate_individual(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def crossover(parent1, parent2):
        if random.random() > crossover_rate:
            return parent1.copy(), parent2.copy()
        
        # One-point crossover with blending
        point = random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        
        # Rescale to maintain constraint
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        return child1, child2
    
    def mutate(individual):
        if random.random() > mutation_rate:
            return individual.copy()
        
        # Mutate about 5% of points
        num_mutations = max(1, int(0.05 * n_points))
        indices = random.sample(range(n_points), num_mutations)
        child = individual.copy()
        
        for i in indices:
            noise = random.uniform(-0.1, 0.1)
            child[i] += noise
            child[i] = max(0, min(1, child[i]))
        
        # Rescale to maintain constraint
        child = child * (n_points / 2) / np.sum(child)
        return child
    
    # Create initial population
    population = [create_individual() for _ in range(pop_size)]
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1 and generation < max_generations:
        # Evaluate fitness
        fitness_scores = []
        for ind in population:
            c5 = evaluate_individual(ind)
            fitness_scores.append((c5, ind))
        
        # Sort by fitness (lower is better)
        fitness_scores.sort(key=lambda x: x[0])
        
        # Update best solution
        current_best_c5, current_best_h = fitness_scores[0]
        if current_best_c5 < best_c5:
            best_c5 = current_best_c5
            best_h = current_best_h.copy()
        
        # Selection: keep elite
        elites = [ind for _, ind in fitness_scores[:elite_size]]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            parent1 = tournament_selection(fitness_scores)
            parent2 = tournament_selection(fitness_scores)
            
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.extend([child1, child2])
        
        population = new_population[:pop_size]
        generation += 1
    
    return (best_h.tolist(), best_c5, n_points)

def tournament_selection(fitness_scores, tournament_size=3):
    """Select an individual using tournament selection."""
    selected = random.sample(fitness_scores, min(tournament_size, len(fitness_scores)))
    selected.sort(key=lambda x: x[0])  # Sort by fitness
    return selected[0][1]  # Return the individual with best fitness
