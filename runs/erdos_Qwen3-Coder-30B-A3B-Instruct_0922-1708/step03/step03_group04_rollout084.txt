import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient for a specific shift
    def compute_grad_for_shift(h_array, shift_idx):
        grad = np.zeros_like(h_array)
        n = len(h_array)
        for i in range(n):
            # For shift k, the contribution to correlation at k is h[i] * (1 - h[i-k])
            # gradient w.r.t h[i] for the max term
            # Actually we compute the gradient of a specific cross-correlation term
            grad[i] = 1 - h_array[(i - shift_idx) % n]
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute full correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        
        # Find the shift that maximizes the correlation
        max_shift_idx = np.argmax(correlation)
        
        # Compute gradient for the maximum shift case
        grad = compute_grad_for_shift(h, max_shift_idx)
        
        # Update step using exact gradient for that shift
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 50 == 0:
            lr *= 0.95
            
        # Occasionally random restart to escape local optima
        if iteration % 300 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5

    return (best_h.tolist(), best_c5, n_points)
