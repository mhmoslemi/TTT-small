import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        # Compute full correlation 
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        # Return maximum value scaled by dx
        return np.max(correlation) * dx
    
    # Helper to compute exact gradient for a given shift
    def compute_gradient(h_array, shift_idx):
        n = len(h_array)
        grad = np.zeros(n)
        # For shift k, compute partial derivatives for correlation at that shift
        # The k-th correlation value is: sum_{i=0}^{n-1} h[i] * (1 - h[(i-k)%n])
        # So partial derivative d/dh[i] of correlation at shift k is:
        # - if i == j+k: coeff = -1, where j is such that (i-k)%n = j
        # which in practice means j = (i-k) % n, same as: j = (i-k+n)%n
        # So grad[i] = -h[(i-shift)%n] for valid shift
        
        # Actually using the definition directly:
        # Since we care about maximum from all shifts:
        # Caution: correct shift to be in proper indexing
        for i in range(n):
            # At shift = shift_idx
            # Correlation at shift_idx involves h[j] * (1 - h[j-shift_idx])
            # So d/dh[i] of this term is:
            # -1 if i = j-shift_idx, 0 otherwise
            # But we're summing over all j
            # j runs from 0 to n-1 and the sum is over term h[j] * (1 - h[j-shift_idx])
            # We want to compute d/dh[i] of this sum = sum over j of d/dh[i] of h[j]*(1-h[j-shift_idx])
            # If i != j: d/dh[i] of h[j]*(1-h[j-shift_idx]) = 0
            # If i == j: d/dh[i] of h[j]*(1-h[j-shift_idx]) = 1*(1 - h[j-shift_idx]) = 1 - h[j-shift_idx] 
            # But the way it's written, h[j-shift_idx] is for the first in the product h[j] * (1 - h[j-shift_idx]).
            # For shift idx, cross-correlation = sum_{j=0}^{n-1} h[j] * (1 - h[j - idx])
            # So gradient of that for h[i] = - (1 - h[i - idx]) = h[i - idx] - 1
            
            # Therefore: grad[i] = h[(i-shift_idx) % n] - 1
            
            old_i = i
            shifted_idx = (i - shift_idx) % n
            grad[old_i] = h[shifted_idx] - 1
            
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1 
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute full cross-correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        # Find lag k with max overlap
        max_k = np.argmax(correlation)
        # The maximum correlation is at max_k
        # The corresponding shift (in lags) is:
        # If we have n points and do full convolution then the valid indices are 0 to 2*n-2
        # The central part corresponds to shifts 0 to n-1 (0 is same as shift-n)
        # The actual shift in [0, n-1] range would be (max_k - n + 1) % n but let's get
        # the correct logic: 
        # The overlap for lag k (from 0 to 2n-2) is the k-th element of convolution
        # We want shift: 
        true_shift = max_k - (n_points - 1) 
        if true_shift < 0:
            true_shift = n_points + true_shift
        
        # Compute gradient for this shift
        grad = compute_gradient(h, true_shift)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
