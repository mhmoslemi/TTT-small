import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convex relaxation with McCormick envelopes
    solver = cp.MOSEK  # or cp.ECOS for open source
    
    iter_count = 0
    while time.time() - start_time < budget_s - 1 and iter_count < 100:
        # Create convex optimization problem
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # McCormick envelopes
        z_vars = {}
        constraints = []
        
        # Bound constraints
        constraints += [0 <= h_var[i] for i in range(n_points)]
        constraints += [h_var[i] <= 1 for i in range(n_points)]
        constraints += [cp.sum(h_var) == n_points / 2]
        constraints += [t >= 0]
        
        # McCormick constraints for z_ij ~ h[i] * (1 - h[j])
        for i in range(n_points):
            for j in range(n_points):
                z = cp.Variable()
                z_vars[(i,j)] = z
                constraints += [z >= h_var[i] + h_var[j] - 1]
                constraints += [z <= h_var[i]]
                constraints += [z <= 1 - h_var[j]]
                constraints += [z >= 0]
        
        # Objective: minimize t
        constraints += [t >= sum(z_vars[(i, (i+k)%n_points)] * dx 
                                for i in range(n_points - k) 
                                for k in range(1, n_points)) 
                        for k in range(1, n_points)]
        
        prob = cp.Problem(cp.Minimize(t), constraints)
        
        try:
            prob.solve(solver=solver, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_sol = h_var.value
                
                # Check while taking a more direct approach using known approx
                # Interleave current solution with better-known formulas to avoid
                # local minima
                h_approx = h_sol.copy()
                h_approx = np.clip(h_approx, 0, 1)
                h_approx = h_approx * (n_points / 2) / np.sum(h_approx)
                
                # Use standard correlation for verification
                corr = np.correlate(h_approx, 1 - h_approx, mode='full')
                c5 = np.max(corr) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_approx.copy()
                    
                h = h_approx.copy()
            else:
                # Fallback to gradient-based on failure
                break
        except:
            # If anything fails, fallback to gradient method
            break
        
        iter_count += 1
    
    # Final refinement using simple gradient descent on misbehaving case
    if iter_count >= 100 or best_c5 > 0.381:
        # Reinitialize with best solution so far
        h = best_h.copy()
        while time.time() - start_time < budget_s - 1:
            # Compute gradient using finite differences
            correlation = np.correlate(h, 1-h, mode='full')
            c5 = np.max(correlation) * dx
            
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Gradient step
            lr = 0.005
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
            # Check for improvement
            corr = np.correlate(h, 1-h, mode='full')
            c5 = np.max(corr) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
    
    # Final computation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
