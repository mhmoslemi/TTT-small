import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Start with the parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
        
    dx = 2.0 / n_points
    
    # Normalize to the correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Optimization parameters
    temp_start = 0.1
    temp_decay = 0.99
    max_iter = 1000
    perturb_fraction = 0.05
    perturb_step = 0.1
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        temp = temp_start * (temp_decay ** iteration)
        
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighborhood: perturb random points
        n_perturb = max(1, int(n_points * perturb_fraction))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        h_new = h.copy()
        
        for i in indices:
            # Random perturbation
            delta = np.random.uniform(-perturb_step, perturb_step)
            h_new[i] += delta
            
        # Clamp values
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability
            if np.random.rand() < np.exp((current_c5 - new_c5) / temp):
                h = h_new
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
