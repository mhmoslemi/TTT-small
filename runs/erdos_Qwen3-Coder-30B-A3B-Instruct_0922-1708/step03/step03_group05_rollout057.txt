import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Create initial particles
    particles = []
    velocities = []
    p_best = []
    p_best_values = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
        for _ in range(num_particles // 5):
            particles.append(h0.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        h0 = np.ones(n_points) * 0.5
        for _ in range(num_particles // 5):
            particles.append(h0.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Fill remaining particles
    for _ in range(num_particles - len(particles)):
        h = np.random.rand(n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize personal best
    for i in range(num_particles):
        p_best.append(particles[i].copy())
        p_best_values.append(float('inf'))
    
    # Initialize global best
    g_best = None
    g_best_value = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_value:
                g_best_value = c5
                g_best = h.copy()
        
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + c1 * r1 * (p_best[i] - particles[i]) + c2 * r2 * (g_best - particles[i])
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            new_h = particles[i] + v
            
            # Project new position to feasible region
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            particles[i] = new_h
    
    # Final evaluation of best solution
    correlation = np.correlate(g_best, 1 - g_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (g_best.tolist(), final_c5, n_points)
