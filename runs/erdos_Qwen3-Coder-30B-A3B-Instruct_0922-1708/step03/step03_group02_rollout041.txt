import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturbation_magnitude = 0.1
    max_annealing_steps = 2000
    
    # Nelder-Mead parameters
    simplex_size = 5
    
    # Annealing phase
    steps = 0
    while steps < max_annealing_steps and time.time() - start_time < budget_s - 1:
        # Perturb a random subset of elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new = h.copy()
        for i in indices:
            perturbation = np.random.normal(0, perturbation_magnitude)
            h_new[i] += perturbation
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        elif np.random.rand() < math.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            
        T *= alpha
        steps += 1

    # Nelder-Mead refinement
    def objective_simplex(h_simplex):
        # h_simplex: 1D array with shape (n_points,)
        # We have a 1D simplex array representing points.
        # This function should return a scalar cost for the simplex.
        # We'll compute the objective value for each point in the simplex
        # and return the minimum.
        costs = []
        for h_point in h_simplex:
            correlation = np.correlate(h_point, 1 - h_point, mode='full')
            c5 = np.max(correlation) * dx
            costs.append(c5)
        return np.min(costs)
    
    # Initialize Simplex with multiple points
    simplex_points = [best_h.copy()]
    for i in range(simplex_size - 1):
        h_pert = best_h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex_points.append(h_pert)
    
    simplex = np.array(simplex_points)
    
    # Refinement with Nelder-Mead in a simplified loop
    nelder_mead_steps = 0
    max_nelder_mead_steps = 1000
    
    while nelder_mead_steps < max_nelder_mead_steps and time.time() - start_time < budget_s - 1:
        # Compute objective for each vertex
        costs = []
        for i in range(len(simplex)):
            h_point = simplex[i]
            correlation = np.correlate(h_point, 1 - h_point, mode='full')
            c5 = np.max(correlation) * dx
            costs.append(c5)
        costs = np.array(costs)

        # Sort vertices
        indices = np.argsort(costs)
        simplex = simplex[indices]
        costs = costs[indices]

        # Reflection
        x_r = 2 * np.mean(simplex[:-1], axis=0) - simplex[-1]
        x_r = np.clip(x_r, 0, 1)
        x_r = x_r * (n_points / 2) / np.sum(x_r)
        correlation = np.correlate(x_r, 1 - x_r, mode='full')
        c5_r = np.max(correlation) * dx

        if costs[0] <= c5_r < costs[-2]:
            simplex[-1] = x_r
        elif c5_r < costs[0]:
            # Expansion
            x_e = 2 * x_r - np.mean(simplex[:-1], axis=0)
            x_e = np.clip(x_e, 0, 1)
            x_e = x_e * (n_points / 2) / np.sum(x_e)
            correlation = np.correlate(x_e, 1 - x_e, mode='full')
            c5_e = np.max(correlation) * dx
            if c5_e < c5_r:
                simplex[-1] = x_e
            else:
                simplex[-1] = x_r
        else:
            # Contraction
            x_c = 0.5 * (simplex[-1] + np.mean(simplex[:-1], axis=0))
            x_c = np.clip(x_c, 0, 1)
            x_c = x_c * (n_points / 2) / np.sum(x_c)
            correlation = np.correlate(x_c, 1 - x_c, mode='full')
            c5_c = np.max(correlation) * dx
            if c5_c < costs[-1]:
                simplex[-1] = x_c
            else:
                # Shrink
                simplex = simplex[0] + 0.5 * (simplex - simplex[0])
                for i in range(len(simplex)):
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])

        # Check for improvement
        new_cost = np.min([np.max(np.correlate(s, 1 - s, mode='full')) * dx for s in simplex])
        if new_cost < best_c5:
            best_c5 = new_cost
            best_h = simplex[0].copy()

        nelder_mead_steps += 1

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
