import numpy as np
import cvxpy as cp
import time
from scipy.interpolate import interp1d

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search bounds
    low = 0.38
    high = 0.4
    
    while high - low > 1e-4 and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2
        
        # Create CVXPY variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute the overlap constraint for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= c_mid / dx)
        
        # Objective - minimize C5 (but constraints handle this)
        objective = cp.Minimize(0)  # dummy objective, constraints define the problem
        
        # Formulate and solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate this solution
                correlation = np.correlate(h_new, 1 - h_new, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < best_c5:
                    best_c5 = c5
                    best_h = h_new.copy()
                    
                high = c_mid
            else:
                low = c_mid
        except Exception:
            low = c_mid
    
    # Final evaluation of best solution
    if best_c5 == float('inf'):
        # Fallback computation
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        best_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), best_c5, n_points)
