import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        if len(h0) != n_points:
            x_orig = np.linspace(0, 2, len(h0))
            x_target = np.linspace(0, 2, n_points)
            h0 = np.interp(x_target, x_orig, h0)
        h0 = h0 / np.sum(h0) * (n_points / 2)
    else:
        h0 = np.ones(n_points) * 0.5
        
    # Create initial population
    particles.append(h0.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    for _ in range(n_particles - 1):
        h = h0 + np.random.normal(0, 0.02, n_points)
        h = np.clip(h, 0, 1)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    omega = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    step = 0
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i, h in enumerate(particles):
            # Ensure constraint sum(h) = n_points/2
            h = h / np.sum(h) * (n_points / 2)
            particles[i] = h
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
                stagnation_count = 0

        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = (omega * v + 
                 c1 * r1 * (personal_best[i] - h) + 
                 c2 * r2 * (global_best - h))
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            h = h + v
            
            # Clip position to bounds
            h = np.clip(h, 0, 1)
            
            # Rescale to satisfy constraint
            h = h / np.sum(h) * (n_points / 2)
            particles[i] = h
            
        step += 1
        stagnation_count += 1
        
        # Adjust parameters periodically
        if step % 500 == 0:
            omega = max(0.8, omega - 0.001)
            
        # Reset some particles periodically to maintain diversity
        if step % 1000 == 0:
            for i in range(n_particles // 10):
                idx = np.random.randint(0, n_particles)
                h = global_best + np.random.normal(0, 0.05, n_points)
                h = np.clip(h, 0, 1)
                h = h / np.sum(h) * (n_points / 2)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final evaluation of global best
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_score = np.max(correlation) * dx
    
    return (global_best.tolist(), final_score, n_points)
