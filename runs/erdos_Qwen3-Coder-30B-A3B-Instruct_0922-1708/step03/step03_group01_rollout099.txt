import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use higher resolution for global search
    dx = 2.0 / n_points

    # Initialize using the provided parent or Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Frank-Zadoff-like sequence
        h = np.sin(np.pi * np.arange(n_points) * (np.arange(n_points) + 1) / n_points)
        h = np.clip((h + 1) / 2, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    start_time = time.time()
    
    # Phase 1: Binary search via QCQP feasibility
    low, high = 0.38, 0.4
    best_feasible = high
    
    while time.time() - start_time < budget_s - 5 and high - low > 1e-4:
        c_mid = (low + high) / 2
        # Check QCQP feasibility
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]

        for k in range(n_points):
            left = h_var
            right = cp.hstack([1 - h_var[k:], 1 - h_var[:k]])
            constraint_expr = cp.sum(cp.multiply(left, right)) <= c_mid / dx
            constraints.append(constraint_expr)

        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                best_feasible = c_mid
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid

    # Phase 2: Local refinement
    # Start from best feasible h found in phase 1 or initial h
    h = h / np.sum(h) * (n_points / 2)  # normalize to sum=100
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    iteration = 0
    max_iter = 1000 if 'initial_h_values' in globals() else 500
    tau = 1.0
    alpha = 0.01
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the index that maximizes overlap
        max_shift_idx = np.argmax(correlation)
        if max_shift_idx >= n_points:
            max_shift = max_shift_idx - n_points
        else:
            max_shift = max_shift_idx
            
        # Analytical gradient for the max shift
        grad = np.zeros(n_points)
        # Gradient w.r.t. terms at shift k
        for i in range(n_points):
            j = (i + max_shift) % n_points
            grad[i] = (1 - h[j]) - h[i]  # derivative of h[i]*(1 - h[i+k]) w.r.t. h[i]
        
        # Gradient descent update
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step-size
        if iteration % 50 == 0:
            alpha *= 0.95
            
    # Downsample to 100 points
    h_final = best_h[::2]  # Simple downsampling
    h_final = h_final * (100 / 2) / np.sum(h_final)
    
    # Final evaluation
    correlation = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(correlation) * (2.0 / 100)
    
    return (h_final.tolist(), c5_final, 100)
