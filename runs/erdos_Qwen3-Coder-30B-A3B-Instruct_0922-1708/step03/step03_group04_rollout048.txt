import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient efficiently
    def compute_gradient(h_array):
        c5_current = compute_c5(h_array)
        grad = np.zeros_like(h_array)
        for i in range(n_points):
            h_plus = h_array.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_current) / eps
        return grad
        
    # Convex initialization step to try to get a better starting point
    if True:  # Simple convex relaxation: only minimize sum of h[i] - 2*sum(h[i]^2)  
        try:
            h_cvx = cp.Variable(n_points)
            objective = cp.Minimize(cp.sum(h_cvx) - 2 * cp.sum_squares(h_cvx))
            constraints = [h_cvx >= 0, h_cvx <= 1, cp.sum(h_cvx) == n_points / 2]
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_cvx_sol = np.array(h_cvx.value)
                h_cvx_sol = np.clip(h_cvx_sol, 0, 1)
                h_cvx_sol = h_cvx_sol * (n_points / 2) / np.sum(h_cvx_sol)
                h = h_cvx_sol
        except:
            pass  # Fall back to standard initialization if convex optimization fails
    
    # Adaptive learning rate and tracking
    last_improvement = 0
    patience = 200
    max_patience = 500
    while time.time() - start_time < budget_s - 1 and iteration < max_patience:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Compute gradient via finite differences
        grad = compute_gradient(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Periodic restart to avoid local minima
        if iteration % 200 == 0 and iteration - last_improvement > 100:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
        # Early termination if no progress
        if iteration - last_improvement > patience:
            break
            
    return (best_h.tolist(), best_c5, n_points)
