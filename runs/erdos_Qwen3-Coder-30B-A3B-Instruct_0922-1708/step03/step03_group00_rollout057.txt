import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Discretize shifts
    shift_points = 200
    shifts = np.linspace(0, 2, shift_points, endpoint=False)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Convert to cvxpy variables
    h_var = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [
        h_var >= 0,
        h_var <= 1,
        cp.sum(h_var) * dx == 1,
        C >= 0
    ]
    
    # Track index of most violated constraint
    most_violated_index = 0
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 200:
        iteration += 1
        
        # Compute the overlap integral with current h
        corr = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the most violated constraint (max overlap)
        max_overlap = -1
        best_shift_idx = 0
        for i, shift in enumerate(shifts):
            if shift == 0:
                continue
            # Compute correlation for this shift
            k_start = int(shift / dx)
            if k_start >= n_points:
                continue
            corr_shift = np.correlate(h, 1 - h, mode='full')
            corr_shift = corr_shift[k_start + n_points - 1 : k_start + 2 * n_points - 1]
            overlap = np.sum(corr_shift) * dx
            if overlap > max_overlap:
                max_overlap = overlap
                best_shift_idx = i
                
        # Add constraint based on most violated shift
        shift_val = shifts[best_shift_idx]
        k_start = int(shift_val / dx)
        if k_start >= n_points:
            continue
        
        # Create constraint manually
        constraint_terms = []
        for i in range(n_points):
            if i - k_start < 0:
                continue
            constraint_terms.append(h_var[i] * (1 - h_var[i - k_start]))
        
        if constraint_terms:
            constraints.append(cp.sum(constraint_terms) * dx <= C)
        
        # Solve the optimization problem
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        # Try different solvers
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
        except:
            try:
                prob.solve(solver=cp.CVXOPT, max_iters=100)
            except:
                try:
                    prob.solve(solver=cp.SCS, max_iters=100)
                except:
                    try:
                        prob.solve(solver=cp.GLPK, max_iters=100)
                    except:
                        pass
        
        if prob.status == cp.OPTIMAL:
            new_h = h_var.value
            if new_h is not None:
                h = np.clip(new_h, 0, 1)
                # Rescale
                h = h * (n_points / 2) / np.sum(h)
                C_val = C.value
                if C_val is not None:
                    if C_val < best_c5 * 0.99999:
                        # Only update C if significantly improved
                        pass
        
        # Simple fallback update to prevent stagnation
        if iteration % 10 == 0:
            # Small perturbation
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
