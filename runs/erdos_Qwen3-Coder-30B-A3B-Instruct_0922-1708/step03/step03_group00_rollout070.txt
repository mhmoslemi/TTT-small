    n_generations = 300

        h = np.array(initial_h_values)

        h = np.clip(h, 0, 1)

        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        population.append(h)

    for _ in range(population_size - 1):
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)

    for generation in range(n_generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate fitness (C5) of each candidate
        fitness = []
        for h in population:
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)

        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
        
        # Create new generation via selection, crossover, mutation
        new_population = [population[best_idx]]
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        elite = [population[i] for i in sorted_indices[:population_size//5]]
        
        # Add elite to new population
        new_population.extend(elite)
        
        while len(new_population) < population_size:

            tournament_size = 5
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if len(new_population) < population_size - 1:
                # Crossover with another selected candidate
                parent2_idx = np.random.choice(population_size)
                parent1 = population[winner_idx]

                # Blend crossover (simple average at points)
                rate = np.random.rand()

                child = np.clip(child, 0, 1)

                # Mutation
                if np.random.rand() < 0.1:

                # Ensure population size
                new_population.append(population[winner_idx])
        
        population = new_population
    
    # Final gradient-based refinement
    h_refine = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx

        grad = np.zeros_like(h_refine)

            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update
        lr = 0.001
        h_new = h_refine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_refine = h_new
        
    # Re-evaluate final result for consistency
    correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
    best_c5 = np.max(correlation) * dx
    
    return (h_refine.tolist(), best_c5, n_points)
