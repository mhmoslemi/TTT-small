import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured construction
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Place peaks in the middle regions
        start1, end1 = int(0.2 * n_points), int(0.4 * n_points)
        start2, end2 = int(1.2 * n_points), int(1.4 * n_points)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters for the hybrid approach
    temp = 0.1  # Initial temperature for simulated annealing
    cooling_rate = 0.995
    lr = 0.01
    min_lr = 0.001
    max_lr = 0.02
    perturb_rate = 0.1
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Simulated Annealing Step
        h_candidate = h.copy()
        # Perturb ~10% of elements
        indices = np.random.choice(n_points, size=int(n_points * perturb_rate), replace=False)
        for i in indices:
            # Add Gaussian noise
            h_candidate[i] += np.random.normal(0, 0.05)
        # Clip and rescale
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5 for the candidate
        corr_candidate = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr_candidate) * dx
        
        # Compute current C5
        corr_current = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(corr_current) * dx

        # Acceptance criterion for SA
        if c5_candidate < c5_current or np.random.rand() < np.exp(-(c5_candidate - c5_current) / temp):
            h = h_candidate
            c5_current = c5_candidate
        
        # Gradient Descent Step
        eps = 1e-5
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5 after gradient step
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Update if improvement
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
            lr = min(max_lr, lr * 1.01)
        else:
            lr = max(min_lr, lr * 0.95)
            
        # Temperatures update
        temp *= cooling_rate

        # Update best solution
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()

    # Final validation
    corr_best = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_best) * dx
    
    return (best_h.tolist(), final_c5, n_points)
