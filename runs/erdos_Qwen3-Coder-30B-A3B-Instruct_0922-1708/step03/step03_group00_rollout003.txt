import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia_w = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Generate initial swarm
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_p = np.array(initial_h_values)
            if len(h_p) != n_points:
                x_orig = np.linspace(0, 2, len(h_p))
                x_new = np.linspace(0, 2, n_points)
                h_p = np.interp(x_new, x_orig, h_p)
            h_p = np.clip(h_p, 0, 1)
            h_p = h_p * (n_points / 2) / np.sum(h_p)
        else:
            h_p = np.random.rand(n_points) * 0.5
            h_p = h_p * (n_points / 2) / np.sum(h_p)
        particles.append(h_p)
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(h_p.copy())
        # Compute initial pbest_val
        corr = np.correlate(h_p, 1 - h_p, mode='full')
        pbest_vals.append(np.max(corr) * dx)

    # Global best
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    if gbest_val < best_c5:
        best_c5 = gbest_val
        best_h = gbest.copy()

    # PSO iterations
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(n_points), np.random.rand(n_points)
            velocities[i] = (inertia_w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Ensure feasibility: clip and rescale
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate new fitness
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            new_val = np.max(corr) * dx
            
            # Update personal best
            if new_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < gbest_val:
                    gbest = particles[i].copy()
                    gbest_val = new_val
                    no_improvement_count = 0
                    
                    if gbest_val < best_c5:
                        best_c5 = gbest_val
                        best_h = gbest.copy()
            else:
                no_improvement_count += 1

    # Local refinement with gradient descent on best solution
    h_refine = gbest.copy()
    refine_iters = 500
    
    for _ in range(refine_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute correlation and C5
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            # Ensure constraint
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refine - lr * grad
        
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_refine = h_new

    # Final evaluation
    correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refine.copy()
        
    return (best_h.tolist(), best_c5, n_points)
