import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute C5 using numpy correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try convex relaxation approach
        try:
            # Create relaxed problem
            h_relaxed = cp.Variable(n_points)
            constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
            
            # Formulate the max overlap constraint
            overlaps = []
            for k in range(n_points):
                # Overlap at lag k: sum_{i=0}^{n_points-1-k} h[i]*(1-h[i+k])
                overlap_sum = 0
                for i in range(n_points - k):
                    overlap_sum += h_relaxed[i] * (1 - h_relaxed[i + k])
                overlaps.append(overlap_sum)
            
            objective = cp.Minimize(cp.max(overlaps))
            prob = cp.Problem(objective, constraints)
            
            prob.solve(solver=cp.ECOS, verbose=False, eps=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_sol = h_relaxed.value
                # Rounding with perturbation
                h_new = h_relaxed_sol.copy()
                # Add small noise to improve rounding
                noise = np.random.normal(0, 0.01, n_points)
                h_new = h_new + noise
                h_new = np.clip(h_new, 0, 1)
                
                # Convert to binary via thresholding
                h_rounded = (h_new >= 0.5).astype(float)
                
                # Adjust so sum equals n_points/2
                current_sum = np.sum(h_rounded)
                diff = current_sum - n_points / 2
                
                if diff > 0:
                    # Need to flip some 1s to 0s
                    indices = np.argsort(h_new)
                    for i in range(int(diff)):
                        h_rounded[indices[i]] = 0
                elif diff < 0:
                    # Need to flip some 0s to 1s
                    indices = np.argsort(-h_new)
                    for i in range(int(-diff)):
                        h_rounded[indices[i]] = 1
                
                h_new = h_rounded.copy()
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
            else:
                # Fallback to gradient step
                eps = 1e-4
                grad = np.zeros_like(h)
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') 
                    c5_plus = np.max(corr_plus) * dx
                    grad[i] = (c5_plus - c5) / eps
                
                h = h - 0.01 * grad
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        except Exception:
            # Fallback to gradient descent
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full') 
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            h = h - 0.01 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
