import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.99
    T = T0
    
    start_time = time.time()
    
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate neighbor
        h_new = h.copy()
        
        # Perturb a few points
        n_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Small random perturbation
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            
        # Clamp to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability
            if np.random.random() < np.exp((current_c5 - new_c5) / T):
                h = h_new
                
        # Update temperature
        T = T * alpha
        
    # Final evaluation
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
