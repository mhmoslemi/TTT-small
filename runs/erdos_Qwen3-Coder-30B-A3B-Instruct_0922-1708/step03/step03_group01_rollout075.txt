import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective_and_gradient(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute correlation
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient
        max_k = np.argmax(correlation)
        grad = np.zeros_like(h)
        
        # The gradient is complex to compute exactly since we're dealing
        # with the maximum over shifts. For now, use finite differences.
        # We'll approximate using a small perturbation
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        return c5, grad
    
    # Simple gradient descent with more sophisticated gradient computation
    iteration = 0
    alpha = 0.01
    max_iter = 10000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Evaluate function and gradient
        c5, grad = objective_and_gradient(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update step
        h_new = h - alpha * grad
        
        # Project back to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            alpha *= 0.995
    
    # Final re-evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    return (best_h.tolist(), final_c5, n_points)
