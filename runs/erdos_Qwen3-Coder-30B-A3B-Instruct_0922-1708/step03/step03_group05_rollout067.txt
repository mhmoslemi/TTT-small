import numpy as np
import cvxpy as cp
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Convex relaxation approach
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        
        # Compute overlap for all lags
        overlaps = []
        for k in range(n_points):
            # For lag k, we compute sum h[i] * (1 - h[i+k])
            # But the lag is from 0 to n_points-1 index-wise
            lag_expr = 0
            for i in range(n_points - k):
                if i + k < n_points:
                    lag_expr += h_relaxed[i] * (1 - h_relaxed[i + k])
            overlaps.append(lag_expr)
        
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                h_relaxed_sol = h_relaxed.value
            else:
                # fallback
                h_relaxed_sol = h
        except:
            # fallback
            h_relaxed_sol = h
            
        # Rounding heuristic from relaxed solution
        h_rounded = (h_relaxed_sol >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust to meet sum constraint
        if current_sum > n_points / 2:
            # Need to flip some 1s to 0s
            diff = int(current_sum - n_points / 2)
            # Flip indices with smallest h_relaxed_sol values
            flip_indices = np.argsort(h_relaxed_sol)[:diff]
            h_rounded[flip_indices] = 0
        elif current_sum < n_points / 2:
            # Need to flip some 0s to 1s
            diff = int(n_points / 2 - current_sum)
            # Flip indices with largest h_relaxed_sol values
            flip_indices = np.argsort(-h_relaxed_sol)[:diff]
            h_rounded[flip_indices] = 1
            
        # Ensure the final sum is correct
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Perturb slightly to escape local minima
        h = h_rounded + np.random.normal(0, 0.001, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
