    no_improve_count = 0

    while iteration < max_iter and time.time() - start_time < budget_s - 1:

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1

        h_new = h + np.random.normal(0, epsilon, n_points)
