import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    def objective(h):
        # Compute correlation for all shifts
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def gradient(h):
        # Compute the gradient for the current maximum overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        max_idx = np.argmax(correlation)
        k_opt = max_idx - len(h) + 1
        
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j = (i - k_opt) % len(h)
            grad[i] = dx * (1 - 2 * h[i])
        return grad
    
    # Use scipy.optimize.minimize with SLSQP
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization loop
    try:
        # Perform the optimization with early stopping
        for iteration in range(10000):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Run optimization step
            res = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, 
                          constraints=constraints, options={'maxiter': 10, 'ftol': 1e-8})
            
            if not res.success:
                break
            
            h = res.x
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            # Evaluate current solution
            c5 = objective(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
    except Exception as e:
        # In case of any error, just return the best solution so far
        pass
    
    # Final evaluation to make sure we have the correct C5
    best_c5 = objective(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
