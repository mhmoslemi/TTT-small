import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    p_best = np.zeros((num_particles, n_points))
    p_best_values = np.full(num_particles, np.inf)
    
    # Initialize particles
    if 'initial_h_values' in globals():
        # Use initial construction for 10% of particles
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        
        for i in range(num_particles // 10):
            particles[i] = h_init.copy()
            # Add small random noise
            particles[i] += np.random.normal(0, 0.01, n_points)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    else:
        # Initialize with uniform distribution
        for i in range(num_particles):
            particles[i] = np.ones(n_points) * 0.5
    
    # Initialize velocities
    for i in range(num_particles):
        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Initialize p_best
    for i in range(num_particles):
        p_best[i] = particles[i].copy()
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Global best
    g_best = particles[0].copy()
    g_best_value = np.inf
    
    start_time = time.time()
    iteration = 0
    
    def project(h):
        # Clip to [0, 1]
        h = np.clip(h, 0, 1)
        # Rescale to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        # Compute correlation
        corr = correlate(h, 1 - h, mode='full')
        # Max correlation value
        c5 = np.max(corr) * dx
        return c5
    
    # Main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            c5 = compute_c5(h)
            
            # Update p_best
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
                
            # Update g_best
            if c5 < g_best_value:
                g_best_value = c5
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(num_particles):
            r1 = np.random.rand(n_points)
            r2 = np.random.rand(n_points)
            
            # Update velocity
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (g_best - particles[i]))
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project(particles[i])
    
    # Final check
    final_c5 = compute_c5(g_best)
    
    return (g_best.tolist(), final_c5, n_points)
