import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    T = T0
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current C₅
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate neighbor
        h_new = h.copy()
        # Perturb 5% of points
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in indices:
            # Add noise
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to correct sum
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C₅
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp((current_c5 - new_c5) / T):
                h = h_new
                
        # Lower temperature
        T = T * alpha
        
        # Occasionally do a larger step to escape local minima
        if iteration % 100 == 0 and iteration > 0:
            h = best_h.copy()
            h += np.random.normal(0, 0.05, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check for best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
