from scipy.optimize import minimize
import time
import math

    best_h = h.copy()
    best_c5 = float('inf')

    alpha = 0.995

    max_anneal_steps = 2000

    for step in range(max_anneal_steps):

            break

        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_new = h.copy()
        for i in perturb_indices:
            perturbation = np.random.normal(0, perturb_scale)
            h_new[i] += perturbation
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx

            best_c5 = c5_new

            h = h_new
            c5 = c5_new
            
        # Cooling
        T *= alpha

    def objective_simplex(h_simplex):
        # Compute C5 for each point in simplex
        c5_vals = []
        for h_point in h_simplex:
            # Normalize to sum = n_points/2
            h_norm = h_point * (n_points / 2) / np.sum(h_point)
            # Clamp to [0,1]
            h_norm = np.clip(h_norm, 0, 1)
            correlation = np.correlate(h_norm, 1 - h_norm, mode='full')
            c5_val = np.max(correlation) * dx
            c5_vals.append(c5_val)
        return c5_vals

    simplex_size = min(10, n_points + 1)
    simplex = [best_h]

        h_perturbed = best_h + np.random.normal(0, 0.01, n_points)
        h_perturbed = np.clip(h_perturbed, 0, 1)
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        simplex.append(h_perturbed)
        
    # Nelder-Mead iterations
    while time.time() - start_time < budget_s - 1:
        # Get objective values
        c5_vals = objective_simplex(simplex)
        
        # Sort simplex by objective value
        sorted_indices = np.argsort(c5_vals)
        simplex = [simplex[i] for i in sorted_indices]

        xbest = simplex[0]

                simplex[-1] = xexp

                simplex[-1] = xrefl

                continue
                
        # Shrink
        for i in range(1, len(simplex)):
            simplex[i] = xbest + 0.5 * (simplex[i] - xbest)
            simplex[i] = np.clip(simplex[i], 0, 1)
            simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
    
    # Final_check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
