def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute exact gradient
    def compute_exact_gradient(h_array):
        # Find the worst-case shift k
        n = len(h_array)
        correlations = np.correlate(h_array, 1 - h_array, mode='full')
        max_idx = np.argmax(correlations)
        # max_idx corresponds to index in full correlation array starting from -n+1 to n-1
        k_max = max_idx - (n - 1)
        
        # Compute gradient for the worst-case k
        grad = np.zeros(n)
        h_shifted = np.roll(h_array, k_max)
        for i in range(n):
            # Partial derivative wrt h[i]
            # C = sum over i of h[i] * (1 - h[i-k])
            # dC/dh[i] = 1 - h[i-k] - h[i+k] (but if k is negative, we adjust)
            # What we have is k_max; compute shift inverse
            # Let's compute: for each i, we want to know how changing h[i] affects C_k_max
            grad[i] = dx * (1 - h_array[(i - k_max) % n] - h_array[(i + k_max) % n])
        return grad

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient using exact computation
        grad = compute_exact_gradient(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():

    return (best_h.tolist(), best_c5, n_points)
