from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)

    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_values):
        # Compute overlap integral for all shifts
        correlation = np.correlate(h_values, 1 - h_values, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def objective(h_values):
        return compute_c5(h_values)
    
    def gradient(h_values):
        # Compute gradient using finite differences
        eps = 1e-6
        grad = np.zeros_like(h_values)
        c5_base = compute_c5(h_values)
        for i in range(len(h_values)):
            h_plus = h_values.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5_base) / eps
        return grad
        
    # Constraints
    def constraint_sum(h_values):
        return np.sum(h_values) - n_points / 2
        
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    iteration = 0
    step_size = 0.001
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute current value
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Compute gradient
        grad = gradient(h)
        
        # Update using gradient descent
        h_new = h - step_size * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive step size
        if iteration % 50 == 0:
            step_size *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
