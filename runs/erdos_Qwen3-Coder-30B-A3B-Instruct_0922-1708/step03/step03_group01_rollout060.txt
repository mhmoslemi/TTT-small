import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for a given h"""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or default
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to the required sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_counter = 0
    stagnation_threshold = 100
    aspiration_prob = 0.1
    
    start_time = time.time()
    iteration = 0
    
    def hash_h(h_vec):
        h_bytes = h_vec.tobytes()
        return hashlib.md5(h_bytes).hexdigest()
    
    def perturb_h(current_h, step):
        candidate = current_h.copy()
        # Select indices to perturb
        indices = np.random.choice(n_points, size=perturbation_count, replace=False)
        # Apply random perturbations
        for i in indices:
            delta = np.random.uniform(-step, step)
            candidate[i] += delta
        # Clip to [0,1]
        candidate = np.clip(candidate, 0, 1)
        # Normalize to sum = n_points/2
        candidate = candidate * (n_points / 2) / np.sum(candidate)
        return candidate
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate multiple candidates
        candidates = []
        for _ in range(50):
            candidate_h = perturb_h(h, step_size)
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            # Add to candidates list (h, c5)
            candidates.append((candidate_h.copy(), candidate_c5))
        
        # Find best candidate not in tabu list
        best_candidate = None
        best_candidate_c5 = float('inf')
        for candidate_h, candidate_c5 in candidates:
            h_hash = hash_h(candidate_h)
            if h_hash not in tabu_list and candidate_c5 < best_candidate_c5:
                best_candidate = candidate_h
                best_candidate_c5 = candidate_c5
        
        # If no good candidate found, try aspiration
        if best_candidate is None:
            for candidate_h, candidate_c5 in candidates:
                h_hash = hash_h(candidate_h)
                if candidate_c5 < best_candidate_c5:
                    if np.random.random() < aspiration_prob:
                        best_candidate = candidate_h
                        best_candidate_c5 = candidate_c5
        
        # Update if we found a better solution
        if best_candidate is not None and best_candidate_c5 < best_c5:
            best_h = best_candidate.copy()
            best_c5 = best_candidate_c5
            stagnation_counter = 0  # reset stagnation counter
            
            # Add previous solution to tabu list
            current_h_hash = hash_h(h)
            tabu_list.add(current_h_hash)
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
                
            h = best_candidate.copy()
        else:
            # No improvement found, add current to tabu list and perturb
            stagnation_counter += 1
            current_h_hash = hash_h(h)
            tabu_list.add(current_h_hash)
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
            
            # Perturb current solution to escape local minimum
            h = perturb_h(h, step_size)
        
        # Increase step size if stagnant
        if stagnation_counter >= stagnation_threshold:
            step_size *= 1.2
            stagnation_counter = 0
            # For memory management, clear tabu list
            tabu_list.clear()
        
        # Periodic check
        if iteration % 100 == 0:
            if np.isnan(best_c5) or np.isinf(best_c5):
                # Recover from bad state
                h = np.ones(n_points) * 0.5
                h = h * (n_points / 2) / np.sum(h)
                best_c5 = evaluate_erdos_solution(h)
                best_h = h.copy()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
