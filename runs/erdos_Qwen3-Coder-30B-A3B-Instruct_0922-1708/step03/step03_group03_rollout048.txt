import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or as uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5

    h = h * (n_points / 2) / np.sum(h)
    
    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = float('inf')
    temp = 0.1
    perturbation_scale = 0.1
    success_count = 0
    total_count = 0
    
    start_time = time.time()
    
    # Define resolver levels (perturbation scales)
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for resolver in resolvers:
        perturbation_scale = resolver
        iterations = 1000
        
        for _ in range(iterations):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Generate perturbation
            noise = np.random.randn(n_points) * perturbation_scale
            h_candidate = h + noise
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlaps
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                perturbation_scale *= 1.1  # Adaptive refinement
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.random() < np.exp(-delta / temp):
                    h = h_candidate.copy()
                    success_count += 1
            
            total_count += 1
            
            # Adjust temperature
            if _ % 100 == 0 and _ > 0:
                temp *= 0.99
                temp = max(temp, 1e-6)
                
                # Adjust perturbation based on success ratio
                if total_count > 0:
                    success_ratio = success_count / total_count
                    if success_ratio > 0.2:
                        perturbation_scale *= 1.5
                    elif success_ratio < 0.1:
                        perturbation_scale *= 0.5
                    perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
                    success_count = 0
                    total_count = 0
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Ensure feasibility
    best_h = np.clip(best_h, 0, 1)
    best_h = best_h * (n_points / 2) / np.sum(best_h)

    return (best_h.tolist(), final_c5, n_points)
