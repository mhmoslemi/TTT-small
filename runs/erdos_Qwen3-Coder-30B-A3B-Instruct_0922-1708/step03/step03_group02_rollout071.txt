import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_init))
        x_target = np.linspace(0, 2, n_points)
        h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Create initial population
    population = []
    population.append(h_init)
    for _ in range(pop_size - 1):
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        # Evaluate current population
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        # Check for new best
        for i, c5 in enumerate(fitness):
            if c5 < best_c5:
                best_c5 = c5
                best_h = population[i].copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Elitism
        elite_size = int(elite_rate * pop_size)
        new_population = sorted_population[:elite_size]
        
        # Create offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            def tournament_select():
                indices = np.random.choice(pop_size, tournament_size)
                best_idx = indices[np.argmin([fitness[i] for i in indices])]
                return population[best_idx].copy()
            
            # Crossover
            if np.random.random() < cross_rate:
                parent1 = tournament_select()
                parent2 = tournament_select()
                
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Mutation
                if np.random.random() < mut_rate:
                    for i in range(n_points):
                        if np.random.random() < 0.1:
                            child1[i] += np.random.normal(0, 0.1)
                if np.random.random() < mut_rate:
                    for i in range(n_points):
                        if np.random.random() < 0.1:
                            child2[i] += np.random.normal(0, 0.1)
                
                # Ensure constraints
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child2 = np.clip(child2, 0, 1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Just mutation
                parent = tournament_select()
                child = parent.copy()
                for i in range(n_points):
                    if np.random.random() < 0.1:
                        child[i] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population
    
    # Final evaluation
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        return (best_h.tolist(), final_c5, n_points)
    else:
        # Fallback to the last evaluated solution
        corr = correlate(population[0], 1 - population[0], mode='full')
        final_c5 = np.max(corr) * dx
        return (population[0].tolist(), final_c5, n_points)
