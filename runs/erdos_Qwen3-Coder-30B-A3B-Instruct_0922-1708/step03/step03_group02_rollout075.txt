import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured pattern: two peaks
        h = np.zeros(n_points)
        # First peak: [0.2, 0.4]
        i_start1 = int(0.2 / 2.0 * n_points)
        i_end1 = int(0.4 / 2.0 * n_points)
        h[i_start1:i_end1] = 1.0
        # Second peak: [1.2, 1.4]
        i_start2 = int(1.2 / 2.0 * n_points)
        i_end2 = int(1.4 / 2.0 * n_points)
        h[i_start2:i_end2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    temp = 0.1
    cooling_rate = 0.995
    lr = 0.01
    perturb_rate = 0.1
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_trial = h.copy()
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            # Perturb by a small amount
            h_trial[i] += np.random.normal(0, 0.05)
        # Clip and rescale
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5 for trial
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(corr_trial) * dx
        
        # Compute original C5
        corr_orig = np.correlate(h, 1 - h, mode='full')
        c5_orig = np.max(corr_orig) * dx
        
        # Accept or reject
        if c5_trial < c5_orig or np.random.random() < np.exp(-(c5_trial - c5_orig) / temp):
            h = h_trial
            c5 = c5_trial
        else:
            c5 = c5_orig
            
        # Gradient Descent Step
        if np.random.random() < 0.7:  # 70% chance to do gradient update
            # Compute gradient using central differences
            grad = np.zeros_like(h)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update with gradient descent
            h_new = h - lr * grad
            
            # Clip and project
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Recompute C5 after gradient update
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update learning rate
            if c5 < best_c5:
                lr = min(lr * 1.01, 0.02)
            else:
                lr = max(lr * 0.95, 0.001)
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Update temperature
        temp *= cooling_rate
    
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
