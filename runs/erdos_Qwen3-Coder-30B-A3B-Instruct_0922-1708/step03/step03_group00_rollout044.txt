import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Simulated Annealing
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < (budget_s - 100) and stagnation_count < max_stagnation:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
            
        T *= alpha
        
        # Occasionally try a small random perturbation to escape local optima
        if iteration % 10 == 0:
            h = np.clip(h + np.random.normal(0, 0.005, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Gradient Descent Refinement on best solution
    h = best_h.copy()
    learning_rate = 0.001
    for i in range(200):
        if time.time() - start_time > budget_s - 10:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Update h
        h_new = h - learning_rate * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate
        if i % 50 == 0 and i > 0:
            learning_rate *= 0.95
            
        # Recompute C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final check with higher resolution if needed
    if n_points < 100:
        n_points = 100
        dx = 2.0 / n_points
        x_orig = np.linspace(0, 2, len(best_h))
        x_new = np.linspace(0, 2, n_points)
        h_interp = np.interp(x_new, x_orig, best_h)
        h_interp = np.clip(h_interp, 0, 1)
        h_interp = h_interp * (n_points / 2) / np.sum(h_interp)
        
        correlation = np.correlate(h_interp, 1 - h_interp, mode='full')
        c5 = np.max(correlation) * dx
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_interp
    
    return (best_h.tolist(), best_c5, n_points)
