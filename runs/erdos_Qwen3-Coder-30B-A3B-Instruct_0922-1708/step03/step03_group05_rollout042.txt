import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    while time.time() - start_time < budget_s - 1:
        # Solve convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Build overlap constraints
        overlaps = []
        for i in range(n_points):
            overlap = 0
            for j in range(n_points - i):
                overlap += h_relaxed[j] * (1 - h_relaxed[i + j])
            overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(cp.hstack(overlaps)) * dx)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, abstol=1e-6)
            if prob.status == 'optimal':
                h_solution = h_relaxed.value
            else:
                # Fallback to gradient descent
                h_solution = h.copy()
        except:
            h_solution = h.copy()
        
        # Round and adjust
        h_rounded = (h_solution > 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        target_sum = n_points / 2
        
        if current_sum > target_sum:
            # Need to reduce
            diff = int(current_sum - target_sum)
            indices = np.argsort(h_solution)
            for i in range(diff):
                h_rounded[indices[i]] = 0
        elif current_sum < target_sum:
            # Need to increase
            diff = int(target_sum - current_sum)
            indices = np.argsort(-h_solution)
            for i in range(diff):
                h_rounded[indices[i]] = 1
        
        h = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Compute final C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
