import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Resolvers for perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    num_iterations = 1000
    
    # For adaptive perturbation
    success_count = 0
    success_window = 50
    last_successes = [0] * success_window
    
    # Temperature for simulated annealing
    temperature = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    
    # Function to compute C5
    def compute_c5(h_vec):
        # Compute overlap for all shifts 1 to n_points-1
        corr = np.correlate(h_vec, 1 - h_vec, mode='full')
        # Only consider valid overlaps (from shift 1 to n_points-1)
        overlap_vals = corr[n_points-1:2*n_points-2] * dx
        return np.max(overlap_vals)
    
    # Adaptive perturbation scale
    perturbation_scale = 0.1
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
        perturbation_scale = resolver
        
        for it in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate by perturbing with noise scaled by current scale
            h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
            # Clip to feasible range
            h_candidate = np.clip(h_candidate, 0, 1)
            # Rescale to sum constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute C5 for candidate
            c5_candidate = compute_c5(h_candidate)
            
            # Update best if improved
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                # Adjust scale: reduce scale if we're finding good solutions
                perturbation_scale = max(0.001, perturbation_scale * 0.9)
                success_count += 1
            else:
                # Use simulated annealing acceptance probability
                delta = c5_candidate - best_c5
                if delta > 0 and temperature > min_temp:
                    prob_accept = np.exp(-delta / temperature)
                    if np.random.rand() < prob_accept:
                        h = h_candidate.copy()
                        success_count += 1
                # Reduce temperature
                if it % 100 == 0:
                    temperature = max(min_temp, temperature * temp_decay)
            
            # Update sliding window of successes
            last_successes[it % success_window] = 1 if c5_candidate < best_c5 else 0
            
            # Adjust perturbation scale based on recent success rate
            if it % 10 == 0:
                success_rate = sum(last_successes) / float(len(last_successes))
                if success_rate > 0.2:
                    perturbation_scale = min(1.0, perturbation_scale * 1.5)
                elif success_rate < 0.1:
                    perturbation_scale = max(0.001, perturbation_scale * 0.5)
            
            # Update current solution
            h = h_candidate.copy()
    
    # Final check with largest possible overlap value
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
