import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    # Compute C5 = max_k int h(x)(1 - h(x+k)) dx
    # Using correlation
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * (2.0 / len(h))
    return c5

def hash_array(arr, digits=6):
    # Hash an array to detect visited solutions
    return hashlib.md5(np.round(arr, digits).tobytes()).hexdigest()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        # Normalize to sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    perturbation_count = int(0.2 * n_points)
    step_size = 0.1
    stagnation_count = 0
    stagnation_threshold = 100
    
    start_time = time.time()
    
    current_h = h.copy()
    current_c5 = best_c5
    
    # Main Tabu search loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        candidates = []
        
        # Generate multiple candidates by perturbing current solution
        for _ in range(50):
            # Copy current solution and perturb
            candidate_h = current_h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Add perturbation
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
                
            # Clip to [0, 1]
            candidate_h = np.clip(candidate_h, 0, 1)
            
            # Rescale to correct integral
            candidate_h = candidate_h * (n_points / 2) / np.sum(candidate_h)
            
            # Check if this solution is in TABU list
            h_hash = hash_array(candidate_h)
            if h_hash not in tabu_list:
                # Evaluate candidate
                c5_candidate = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5_candidate))
                
        # If we found good candidates, select the best one
        if candidates:
            # Sort by C5 value
            candidates.sort(key=lambda x: x[1])
            best_candidate_h, best_candidate_c5 = candidates[0]
            
            # Accept improvement or with aspiration criteria
            if best_candidate_c5 < current_c5 or (np.random.random() < 0.1):
                current_h = best_candidate_h.copy()
                current_c5 = best_candidate_c5
                
                if best_candidate_c5 < best_c5:
                    best_h = current_h.copy()
                    best_c5 = current_c5
                    
                # Add old solution to tabu list (hash-based)
                old_hash = hash_array(current_h)
                tabu_list.add(old_hash)
                if len(tabu_list) > tabu_length:
                    # Remove oldest entry by simple remove
                    tabu_list.pop()
                    
                # Reset stagnation
                stagnation_count = 0
            else:
                # No improvement, still update tabu and increase step if stagnated
                stagnation_count += 1
                
        else:
            # No valid candidate found, add current to TABU and mutate
            old_hash = hash_array(current_h)
            tabu_list.add(old_hash)
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
                
            # Try other random mutation
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                current_h[i] += delta
            current_h = np.clip(current_h, 0, 1)
            current_h = current_h * (n_points / 2) / np.sum(current_h)
            
        # Adaptive perturbation step size
        if stagnation_count >= stagnation_threshold:
            step_size *= 1.2  # Increase perturbation
            # Reset tabu list to avoid overcrowding
            tabu_list.clear()
            stagnation_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
