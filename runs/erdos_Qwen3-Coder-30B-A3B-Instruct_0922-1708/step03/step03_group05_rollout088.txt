import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Optimization parameters
    lambda_smooth = 0.1
    lr = 0.01
    smoothing_decay = 0.99
    max_iterations = 1000
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Smoothly decay lambda
        if iteration % 100 == 0:
            lambda_smooth *= smoothing_decay
        
        # Solve relaxed problem
        # Minimize E(h) = C5 + lambda * sum((h[i] - h[j])^2)  s.t. sum(h) = n/2
        
        # For the energy function, we will make a simpler direct approach
        # First compute the gradient of C5 w.r.t h
        
        # Compute gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Add smoothing regularizer
        if n_points > 1:
            smooth_grad = np.zeros_like(h)
            for i in range(n_points):
                smooth_grad[i] = -2 * lambda_smooth * (h[i] - np.mean(h))
            grad += smooth_grad
            
        # Update h using gradient descent
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Periodically apply perturbation to avoid local minima
        if iteration % 50 == 0:
            perturb_factor = 0.002
            perturb = np.random.normal(0, perturb_factor, n_points)
            h = h + perturb
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation on best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
