import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    best_h = None
    best_c5 = float('inf')
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Structured initialization
        h = np.zeros(n_points)
        # Place step functions in key regions
        idx1 = int(0.2 / 2 * n_points)
        idx2 = int(0.4 / 2 * n_points)
        idx3 = int(1.2 / 2 * n_points)
        idx4 = int(1.4 / 2 * n_points)
        h[idx1:idx2] = 1.0
        h[idx3:idx4] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    # Helper functions
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project_h(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def perturb_h(h_vec, rate=0.1, scale=0.1):
        h_new = h_vec.copy()
        n_perturb = int(rate * n_points)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, scale)
        return project_h(h_new)
    
    # Cooling schedule and parameters
    T = 0.1
    alpha = 0.995
    lr = 0.01
    eps = 1e-5
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing step
        h_candidate = perturb_h(h, rate=0.1, scale=0.1)
        c5_candidate = compute_c5(h_candidate)
        
        if c5_candidate < best_c5:
            best_c5 = c5_candidate
            best_h = h_candidate.copy()
        
        # Accept with probability
        if c5_candidate < compute_c5(h) or np.random.rand() < np.exp(-(c5_candidate - compute_c5(h)) / T):
            h = h_candidate.copy()
        
        # Gradient Descent step
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            
            h_minus = h.copy()
            h_minus[i] -= eps
            h_minus = project_h(h_minus)
            c5_minus = compute_c5(h_minus)
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
        
        h = h - lr * grad
        h = project_h(h)
        
        # Update learning rate
        if compute_c5(h) < best_c5:
            lr *= 1.01
        else:
            lr *= 0.95
        lr = np.clip(lr, 0.001, 0.02)
        
        # Update temperature
        T *= alpha
    
    # Final check
    if best_h is not None:
        c5_final = compute_c5(best_h)
        return (best_h.tolist(), c5_final, n_points)
    else:
        c5_final = compute_c5(h)
        return (h.tolist(), c5_final, n_points)
