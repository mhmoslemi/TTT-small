import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            particle = np.array(initial_h_values)
            if len(particle) != n_points:
                x_orig = np.linspace(0, 2, len(particle))
                x_new = np.linspace(0, 2, n_points)
                particle = np.interp(x_new, x_orig, particle)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        else:
            particle = np.random.rand(n_points) * 0.5
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle.copy())
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(particle.copy())
        corr = np.correlate(particle, 1 - particle, mode='full')
        pbest_vals.append(np.max(corr) * dx)
    
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO iterations
    no_improvement = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        iteration += 1
        
        for i in range(n_particles):
            particle = particles[i]
            velocity = velocities[i]
            current_pbest = pbest[i]
            current_pbest_val = pbest_vals[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocity = (inertia * velocity + 
                       c1 * r1 * (current_pbest - particle) + 
                       c2 * r2 * (gbest - particle))
            
            # Clip velocity
            velocity = np.clip(velocity, -max_velocity, max_velocity)
            
            # Update particle
            particle = particle + velocity
            
            # Apply constraints
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Evaluate
            corr = np.correlate(particle, 1 - particle, mode='full')
            new_val = np.max(corr) * dx
            
            # Update personal best
            if new_val < current_pbest_val:
                pbest[i] = particle.copy()
                pbest_vals[i] = new_val
                
                # Update global best
                if new_val < gbest_val:
                    gbest = particle.copy()
                    gbest_val = new_val
                    no_improvement = 0
                    
            # Update particle and velocity
            particles[i] = particle
            velocities[i] = velocity
            
        no_improvement += 1
        
    # Local refinement with gradient descent on best solution
    h = gbest
    
    # Gradient descent loop with backtracking line search
    max_iter = 200
    for _ in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Project to feasible set
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update best solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
