import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5  # Uniform distribution
        
    # Add small noise to break symmetry
    h += np.random.normal(0, 0.01, n_points)
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    temperature = 0.1
    success_count = 0
    total_accepted = 0
    
    # Resolver levels with decreasing scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        num_iterations = 1000
        for iteration in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate by perturbing h
            perturbation = np.random.normal(0, resolver, n_points)
            h_candidate = h + perturbation
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap integral
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject based on simulated annealing criterion
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                total_accepted += 1
                success_count += 1
            else:
                delta = c5_candidate - best_c5
                if random.random() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
                    total_accepted += 1
                    
            # Adaptively adjust temperature
            if iteration > 0 and iteration % 100 == 0:
                temperature *= 0.99
                temperature = max(temperature, 1e-6)
                
            # Adaptively adjust perturbation scale based on success
            if iteration > 0 and iteration % 50 == 0:
                success_rate = success_count / 50.0
                if success_rate > 0.2:
                    resolver *= 1.1
                elif success_rate < 0.1:
                    resolver *= 0.9
                resolver = max(0.001, min(1.0, resolver))
                success_count = 0
                
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
