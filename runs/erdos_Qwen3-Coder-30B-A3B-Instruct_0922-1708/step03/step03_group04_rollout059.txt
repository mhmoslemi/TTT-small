import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient for max correlation
    def compute_gradient(h_array, max_shift):
        n = len(h_array)
        grad = np.zeros_like(h_array)
        # For shift k, correlation[i] = sum(h[j] * (1 - h[(j+k) % n])) for j in range(max(0,i-k), min(n,i-k+n))
        # So for fixed i: 
        #   if i-k < 0, then j ranges from 0 to i-k+n-1
        #   if i-k >= 0, then j ranges from i-k to i-k+n-1
        # For our case, compute partial derivative
        for i in range(n):
            grad[i] = 0
            # When h[i] contributes to the max shift, it multiplies with (1-h[(i-k)%n])
            # So partial derivative wrt h[i] is -1 * (1 - h[(i-k)%n]) 
            # This derivation is more straightforward when you consider the contribution to the overlap at shift k
            # For each j, grad of h[i] wrt to the overlap at shift k is 0 if i != j-k (mod n), 
            # and -1 * (1-h[j]) if i == j-k (mod n),  
            # So total is sum over j of (-1 * 1_{i == (j-k)%n}* (1-h[j]) ) =  -1*(1-h[(i+k)%n])
            # Since the contribution of h[j] to overlap at k = h[j]*(1-h[j+k%n])
            # But because of circular nature: grad[i] = sum over j of -1 * 1_{i=(j-k)%n} * (1-h[j])
            # Simplified: grad[i] = - (1-h[(i+k)%n]) 
            # But not sure. Simpler approach: compute 1st derivative with respect to each h[i] for maximum shift
            # let's reframe this as we're essentially computing the gradient of h^T * (1 - h.shifted_by_k)
            # for the k which gives the maximum overlap
            
            # Let's just compute the gradient via maximum shift directly
            # The contribution of h[i] to overlap at shift k is h[i] * (1-h[(i-k)%n])
            # So the partial derivative wrt h[i] is (1- h[(i-k)%n]) when i corresponds to that term.
            # Clarification:
            # overlap_k = sum_over_i h[i] * (1 - h[(i-k)%n])
            # d_overlap_k / d h[i] = (1 - h[(i-k)%n]) for that value of i
            # for the k that maximizes overlap, 
            # grad[i] = 1 - h[(i-max_shift)%n]
            
            # Very careful way:
            k = max_shift
            for j in range(n):
                if j == (i + k) % n:
                    # contribution to overlap at shift k: h[j]*(1-h[i])
                    # d(h[j] * (1-h[i])) / d h[i] = -h[j]
                    grad[i] -= h[j]
                if j == (i - k) % n:
                    # contribution to overlap at shift k: h[i]*(1-h[j])
                    # d(h[i] * (1-h[j])) / d h[i] = (1-h[j])
                    grad[i] += (1 - h[j])
        
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Get all correlations and max shift
        full_corr = np.correlate(h, 1-h, mode='full')
        # Full correlation has 2*n_points - 1 values indexed from -(n_points-1) to (n_points-1)
        # We want to find the maximum shift k in [0,2] which is the real k that corresponds to peak
        max_shift_idx = np.argmax(full_corr)
        max_corr_val = full_corr[max_shift_idx]
        
        max_shift = max_shift_idx - (n_points-1)  # convert index to shift k = -n_points+1 to n_points-1
        # That's the shift k that maximizes the overlap. So we want to reduce that overlap
        
        # Compute exact gradient for this shift k
        n = len(h)
        grad = np.zeros_like(h)
        for i in range(n):
            # For shift max_shift, the term in overlap_sum at shift max_shift is h[i] * (1 - h[(i - max_shift) % n])
            # So d/dh[i] of overlap at shift max_shift = (1 - h[(i - max_shift) % n])
            # Loop over all positions that contribute 
            # Actually, because the overlaps are computed from h * (1 - h.shifted_by_k)
            # The gradient of h[i] in the overlap contribution from h[j] is:
            # for shift k = max_shift, we have term = h[i] * (1 - h[(i - k) % n])  
            # But we also have (i+k) % n factor for the contribution from each h[j] affected
            
            # Think carefully:
            # For shift k (where k can be 0 to 2*n-3 for full correlation), 
            # the overlap is sum of h[i] * (1 - h[(i-k)%n])
            # d/dh[i] = (1 - h[(i-k)%n]) since derivative on h[i] in that term
            
            # But we're targeting the maximum over k
            # So derivative of overlap_k w.r.t. h[i] = ( if i-k >= 0, then (1 - h[(i-k)%n]) else 0)  
            # No wait, that's wrong. I should directly compute the correlation gradients for this specific k.
            k = max_shift
            # given that i determines term h[i]* (1-h[(i-k)%n]) in overlap corresponding to shift k  
            grad[i] = 1 - h[(i - k) % n]
            
        # Apply gradient update
        h_new = h - lr * grad
        
        # Project h_new to [0,1] and rescale to satisfy sum constraint
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5

    return (best_h.tolist(), best_c5, n_points)
