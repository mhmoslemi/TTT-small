import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization: step functions
        h = np.zeros(n_points)
        # Place two step functions
        start1, end1 = int(0.2/dx), int(0.4/dx)
        start2, end2 = int(1.2/dx), int(1.4/dx)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.995
    lr = 0.01
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing step
        h_trial = h.copy()
        # Perturb 10% of the points
        perturb_indices = np.random.choice(n_points, size=int(0.1*n_points), replace=False)
        for i in perturb_indices:
            h_trial[i] += np.random.normal(0, 0.05)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute correlation for both
        c5_trial = np.max(np.correlate(h_trial, 1 - h_trial, mode='full')) * dx
        c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Accept or reject based on SA
        if c5_trial < c5_current or np.random.rand() < math.exp(-(c5_trial - c5_current) / T):
            h = h_trial
            c5_current = c5_trial
        
        # Gradient Descent step
        grad = np.zeros_like(h)
        eps = 1e-5

        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
            
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5_new = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Update learning rate
        if c5_new < c5_current:
            lr = min(lr * 1.01, 0.02)
        else:
            lr = max(lr * 0.95, 0.001)
        
        # Update temperature
        T *= alpha
        
        # Update best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
