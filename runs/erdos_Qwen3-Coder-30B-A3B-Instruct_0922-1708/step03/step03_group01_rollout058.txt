import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    crossover_rate = 0.5
    mutation_rate = 0.2
    elite_size = 10
    
    # Initialize population
    def initialize_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            h = np.random.rand(n_points)
        # Normalize to sum to n_points / 2
        h = h * (n_points / 2) / np.sum(h)
        return np.clip(h, 0, 1)
    
    def evaluate_individual(h):
        corr = correlate(h, 1 - h, mode='full')
        return np.max(corr) * dx
    
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
        point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([parent1[:point], parent2[point:]])
        child2 = np.concatenate([parent2[:point], parent1[point:]])
        # Renormalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return np.clip(child1, 0, 1), np.clip(child2, 0, 1)
    
    def mutate(h):
        h_mut = h.copy()
        n_mutate = int(mutation_rate * n_points)
        indices = np.random.choice(n_points, n_mutate, replace=False)
        for i in indices:
            noise = np.random.normal(0, 0.1)
            h_mut[i] += noise
        # Renormalize
        h_mut = h_mut * (n_points / 2) / np.sum(h_mut)
        return np.clip(h_mut, 0, 1)
    
    # Initialize population
    population = [initialize_individual() for _ in range(pop_size)]
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    patience = 0
    
    while time.time() - start_time < budget_s - 1:
        generation += 1
        
        # Evaluate population
        fitness = [evaluate_individual(ind) for ind in population]
        
        # Update best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            patience = 0
        else:
            patience += 1
        
        # Selection
        sorted_indices = np.argsort(fitness)
        elites = [population[i] for i in sorted_indices[:elite_size]]
        
        # Create new population
        new_population = elites.copy()
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.choice(sorted_indices[:pop_size//2])
            parent2_idx = np.random.choice(sorted_indices[:pop_size//2])
            
            child1, child2 = crossover(population[parent1_idx], population[parent2_idx])
            
            if np.random.rand() < mutation_rate:
                child1 = mutate(child1)
            if np.random.rand() < mutation_rate:
                child2 = mutate(child2)
                
            new_population.extend([child1, child2])
            
        population = new_population[:pop_size]
        
        # Early stopping
        if patience > 100:
            break
            
    return (best_h.tolist(), best_c5, n_points)
