import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 200
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define shift points for constraint generation
    shift_points = np.linspace(0, 2, 200)
    
    # Cutting plane method loop
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute overlaps for all shifts
        overlaps = []
        for k in shift_points:
            # Compute the overlap integral for shift k
            # Use numpy's correlate with mode='full' and properly handle truncation
            h_shifted = np.roll(h, -int(k / dx))
            # Ensure proper alignment for convolution-like operation
            correlation = np.correlate(h, 1 - h_shifted, mode='full')
            # Select valid indices
            valid_indices = range(len(correlation) // 2, len(correlation) // 2 + n_points)
            result = np.sum(correlation[valid_indices]) * dx
            overlaps.append(result)
        
        max_overlap = max(overlaps)
        
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Add constraint to the optimization problem
        # Formulate as convex optimization:
        # Minimize C
        # subject to: sum(h[i] * (1 - h[i+k])) * dx <= C for all k
        
        # Create variable for h
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Constraints based on sampled shifts
        constraints = []
        
        # Normalize the integral constraint
        constraints.append(cp.sum(h_var) * dx == 1.0)
        for i in range(n_points):
            constraints.append(0 <= h_var[i])
            constraints.append(h_var[i] <= 1.0)
            
        # Re-sample on a finer grid
        fine_shifts = np.linspace(0, 2, 200)
        for k in fine_shifts[:50]:  # Use fewer to save time
            # Approximate correlation with sliding window approach
            indices_k = np.arange(int(k / dx), min(int(k / dx) + n_points, n_points)) 
            if len(indices_k) == 0:
                continue
            h_shifted = np.zeros(n_points)
            h_shifted[indices_k] = h[ : len(indices_k)]
            corr_val = np.sum(h_var * (1 - h_shifted)) * dx
            constraints.append(corr_val <= C)
            
        # Solve the problem
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.CVXOPT, verbose=False, max_iters=50)
            if prob.status != cp.INFEASIBLE:
                h = np.array(h_var.value).flatten()
                h = np.clip(h, 0, 1)
                # Rescale h to meet sum(h)*dx = 1 constraint
                h = h * (n_points / 2) / np.sum(h)
        except:
            pass
    
    # Final computation and return
    correlation_vals = np.correlate(best_h, 1 - best_h, mode='full')
    max_corr = np.max(correlation_vals) * dx
    return (best_h.tolist(), max_corr, n_points)
