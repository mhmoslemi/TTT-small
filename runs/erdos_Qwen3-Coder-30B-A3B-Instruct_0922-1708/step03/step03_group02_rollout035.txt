import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    
    # Generate other particles with random perturbations
    for _ in range(n_particles - 1):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Initialize global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    ω = 1.0  # inertia weight
    c1 = c2 = 2.0  # acceleration constants
    v_max = 0.2  # maximum velocity
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure the constraint sum(h) = n_points / 2
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = ω * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h = h + v
            particles[i] = h
        
        # Periodically reset some particles to maintain diversity
        if step % 100 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:  # 10% chance
                    h = global_best + np.random.normal(0, 0.05, n_points)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h
        
        # Adjust inertia weight if needed
        if step > 1000:
            ω = max(0.8, 1.2 - (step / 10000) * 0.4)
    
    return (global_best.tolist(), global_best_score, n_points)
