import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing with Nelder-Mead refinement
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling rate
    perturbation_sigma = 0.1
    
    # Set up simplex for Nelder-Mead in the first iteration
    simplex_size = 5
    simplex = [h.copy()]
    for _ in range(simplex_size - 1):
        h_pert = h + np.random.normal(0, perturbation_sigma, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Annealing phase - global search
    for _ in range(2000):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Make a random perturbation to one element
        idx = np.random.randint(0, n_points)
        h_new = h.copy()
        h_new[idx] += np.random.normal(0, perturbation_sigma)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        c5_new = compute_c5(h_new)
        c5_old = compute_c5(h)
        
        if c5_new < c5_old:
            h = h_new
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(c5_new - c5_old) / T):
                h = h_new
                
        # Cool down
        T *= alpha
        
        # Track best solution
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
    
    # Refinement with Nelder-Mead
    def obj_func_simplex(x):
        # Normalize
        x = x * (n_points / 2) / np.sum(x)
        x = np.clip(x, 0, 1)
        return compute_c5(x)
    
    while time.time() - start_time < budget_s - 1:
        # Update simplex values
        simplex_vals = [obj_func_simplex(p) for p in simplex]
        sort_idx = np.argsort(simplex_vals)
        simplex = [simplex[i] for i in sort_idx]
        simplex_vals = [simplex_vals[i] for i in sort_idx]
        
        # Nelder-Mead step
        # Reflect
        centroid = np.mean(simplex[:-1], axis=0)
        x_r = centroid + (centroid - simplex[-1])
        x_r = np.clip(x_r, 0, 1)
        x_r = x_r * (n_points / 2) / np.sum(x_r)
        f_r = obj_func_simplex(x_r)
        
        if simplex_vals[0] <= f_r < simplex_vals[-2]:
            simplex[-1] = x_r
        elif f_r < simplex_vals[0]:
            # Expand
            x_e = centroid + 2 * (x_r - centroid)
            x_e = np.clip(x_e, 0, 1)
            x_e = x_e * (n_points / 2) / np.sum(x_e)
            f_e = obj_func_simplex(x_e)
            if f_e < f_r:
                simplex[-1] = x_e
            else:
                simplex[-1] = x_r
        else:
            # Contract
            x_c = simplex[-1] + 0.5 * (centroid - simplex[-1])
            x_c = np.clip(x_c, 0, 1)
            x_c = x_c * (n_points / 2) / np.sum(x_c)
            f_c = obj_func_simplex(x_c)
            if f_c < simplex_vals[-1]:
                simplex[-1] = x_c
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
        
        # Update best solution
        best_val_in_simplex = min([obj_func_simplex(p) for p in simplex])
        if best_val_in_simplex < best_c5:
            best_c5 = best_val_in_simplex
            # Find best point in simplex
            best_simplex_idx = np.argmin([obj_func_simplex(p) for p in simplex])
            best_h = simplex[best_simplex_idx].copy()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
