import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using provided parent or uniform distribution
    if 'initial_h_values' in globals():

        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # GA parameters
    pop_size = 50
    elite_size = 20
    crossover_rate = 0.5
    mutation_rate = 0.2
    mutation_intensity = 0.1
    
    # Create initial population
    population = []
    for _ in range(pop_size):
        if 'initial_h_values' in globals():
            individual = h.copy()
        else:
            individual = np.random.rand(n_points)
        # Normalize
        individual = individual * (n_points / 2) / np.sum(individual)
        # Clip to [0,1]
        individual = np.clip(individual, 0, 1)
        population.append(individual)
    
    # Ensure at least one valid starting individual
    if 'initial_h_values' not in globals():
        population[0] = h.copy()
    
    no_improvement_count = 0
    max_no_improvement = 100
    
    # GA main loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate fitness of all individuals
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Update best solution
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        if no_improvement_count > max_no_improvement:
            # Reset population if no improvement
            population = []
            for _ in range(pop_size):
                if 'initial_h_values' in globals():
                    individual = h.copy()
                else:
                    individual = np.random.rand(n_points)
                individual = individual * (n_points / 2) / np.sum(individual)
                individual = np.clip(individual, 0, 1)
                population.append(individual)
            no_improvement_count = 0
        
        # Select elite
        new_population = population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            else:
                child1 = parent1.copy()
                child2 = parent2.copy()
                
            # Mutation
            def mutate(individual):
                mutated = individual.copy()
                num_mutations = int(mutation_rate * n_points)
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for i in mutation_indices:
                    delta = np.random.normal(0, mutation_intensity)
                    mutated[i] += delta
                # Clip to [0,1]
                mutated = np.clip(mutated, 0, 1)
                # Rescale
                mutated = mutated * (n_points / 2) / np.sum(mutated)
                return mutated
                
            child1 = mutate(child1)
            child2 = mutate(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
                
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
