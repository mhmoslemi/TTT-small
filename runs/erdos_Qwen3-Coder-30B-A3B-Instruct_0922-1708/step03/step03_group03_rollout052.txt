import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a modified Barker initialization
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a modified Barker-like initialization
        h = np.zeros(n_points)
        # Use a pattern similar to Barker codes but adapted for our constraints
        # Place two pulses with a specific spacing to reduce correlations
        pulse_width = n_points // 4
        h[:pulse_width] = 1.0
        h[pulse_width:2*pulse_width] = 0.0
        h[2*pulse_width:3*pulse_width] = 1.0
        h[3*pulse_width:] = 0.0
        
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)  # Keep integral constraint
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def gradient(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        eps = 1e-6
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = objective(h_plus)
            c5 = objective(h)
            
            grad[i] = (c5_plus - c5) / eps
        
        return grad
    
    # Optimization using L-BFGS-B
    # Restart optimization several times to avoid local minima
    restarts = 5
    for restart in range(restarts):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Generate a randomized starting point for each restart
        if restart > 0:
            h = np.random.rand(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
        
        # Optimize including bounds
        res = minimize(objective, h, method='L-BFGS-B', 
                       bounds=[(0, 1) for _ in range(n_points)],
                       options={'maxiter': 500, 'ftol': 1e-8},
                       jac=gradient if restart == 0 else None)
        
        h_opt = res.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        c5_opt = objective(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
    
    # Final refinement using simple gradient descent
    h_refine = best_h.copy()
    lr = 0.001
    max_iter = 2000
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 2:
            break
            
        grad = gradient(h_refine)
        h_new = h_refine - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check if we're improving
        c5_new = objective(h_new)
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            
        h_refine = h_new
        lr *= 0.999  # Decay learning rate
    
    # Final check to make sure it meets constraints
    final_h = np.clip(best_h, 0, 1)
    final_h = final_h * (n_points / 2) / np.sum(final_h)
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    # Double check our final result
    return (final_h.tolist(), final_c5, n_points)
