import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure initial normalization
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing
    T = 0.1
    alpha = 0.995
    perturb_sigma = 0.1
    sa_steps = 0
    max_sa_steps = 2000
    
    while sa_steps < max_sa_steps and time.time() - start_time < budget_s - 1:
        # Make a random perturbation to current solution
        h_new = h.copy()
        # Perturb about 10% of elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, perturb_sigma)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing probability
        if c5_new < best_c5 or np.random.rand() < math.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= alpha
        sa_steps += 1
    
    # Nelder-Mead refinement
    def objective(h_flat):
        h = h_flat.reshape(-1)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Create initial simplex
    h_base = best_h.copy()
    simplex = [h_base]
    for i in range(4):
        h_pert = h_base + np.random.normal(0, 0.01, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)
    
    # Simplex optimization loop
    nm_steps = 0
    max_nm_steps = 1000
    
    while nm_steps < max_nm_steps and time.time() - start_time < budget_s - 1:
        # Evaluate all vertices
        values = [objective(h) for h in simplex]
        sorted_indices = np.argsort(values)
        
        # Update best if necessary
        if values[sorted_indices[0]] < best_c5:
            best_c5 = values[sorted_indices[0]]
            best_h = simplex[sorted_indices[0]].copy()
        
        # Simplex operations (minimal implementation)
        # Use the best and worst points for reflection
        x_best = simplex[sorted_indices[0]]
        x_worst = simplex[sorted_indices[-1]]
        x_second_worst = simplex[sorted_indices[-2]]
        
        # Compute centroid excluding the worst point
        x_centroid = np.mean([simplex[i] for i in sorted_indices[:-1]], axis=0)
        
        # Reflection
        x_reflect = x_centroid + x_centroid - x_worst
        x_reflect = np.clip(x_reflect, 0, 1)
        x_reflect = x_reflect * (n_points / 2) / np.sum(x_reflect)
        
        c_reflect = objective(x_reflect)
        
        # Replace worst point
        if c_reflect < values[sorted_indices[-2]] and c_reflect > values[sorted_indices[0]]:
            simplex[sorted_indices[-1]] = x_reflect
        elif c_reflect < values[sorted_indices[0]]:
            # Expansion
            x_expand = x_centroid + 2 * (x_reflect - x_centroid)
            x_expand = np.clip(x_expand, 0, 1)
            x_expand = x_expand * (n_points / 2) / np.sum(x_expand)
            
            c_expand = objective(x_expand)
            if c_expand < c_reflect:
                simplex[sorted_indices[-1]] = x_expand
            else:
                simplex[sorted_indices[-1]] = x_reflect
        else:
            # Contraction
            x_contract = x_centroid + 0.5 * (x_worst - x_centroid)
            x_contract = np.clip(x_contract, 0, 1)
            x_contract = x_contract * (n_points / 2) / np.sum(x_contract)
            
            c_contract = objective(x_contract)
            if c_contract < values[sorted_indices[-1]]:
                simplex[sorted_indices[-1]] = x_contract
            else:
                # Shrink
                x_best_orig = x_best.copy()
                for i in range(1, len(simplex)):
                    simplex[i] = x_best_orig + 0.5 * (simplex[i] - x_best_orig)
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        
        nm_steps += 1
    
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
