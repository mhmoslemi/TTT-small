import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Ensure constraint sum(h) = n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturb_scale = 0.1
    
    # Nelder-Mead parameters
    simplex_size = 5
    max_nelder_mead_iter = 1000
    
    # Annealing phase
    steps = 0
    max_anneal_steps = 2000
    
    while steps < max_anneal_steps and time.time() - start_time < budget_s - 1:
        # Create a perturbed version
        h_new = h.copy()
        # Perturb ~10% of elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, perturb_scale)
        
        # Clip to feasible range [0,1]
        h_new = np.clip(h_new, 0, 1)
        
        # Normalize to satisfy sum(h) = n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing criteria
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        else:
            # Accept with some probability
            delta = c5_new - best_c5
            if np.random.rand() < np.exp(-delta / T):
                h = h_new.copy()
        
        # Cool down temperature
        T *= alpha
        steps += 1
    
    # Nelder-Mead refinement
    def objective_func(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Create a simplex around the current best solution
    simplex = [best_h.copy()]
    for i in range(simplex_size - 1):
        h_pert = best_h.copy()
        for j in range(int(0.05 * n_points)):
            idx = np.random.randint(0, n_points)
            h_pert[idx] += np.random.normal(0, 0.01)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)
    
    # Perform Nelder-Mead optimization
    stepcount = 0
    while time.time() - start_time < budget_s - 1 and stepcount < max_nelder_mead_iter:
        # Evaluate all points in simplex
        f_vals = []
        for h_tmp in simplex:
            f_vals.append(objective_func(h_tmp))
        
        # Sort indices
        idx_sorted = np.argsort(f_vals)
        
        # Best and worst
        best_idx = idx_sorted[0]
        worst_idx = idx_sorted[-1]
        
        # Compute centroid of all but worst
        centroid = np.mean([simplex[i] for i in range(simplex_size) if i != worst_idx], axis=0)
        
        # Reflection point
        r = 1 + 1
        reflected = centroid + r * (centroid - simplex[worst_idx])
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (n_points / 2) / np.sum(reflected)
        f_reflected = objective_func(reflected)
        
        if f_vals[best_idx] <= f_reflected < f_vals[-2]:
            # Replace worst
            simplex[worst_idx] = reflected
        elif f_reflected < f_vals[best_idx]:
            # Expansion
            e = 2
            expanded = centroid + e * (reflected - centroid)
            expanded = np.clip(expanded, 0, 1)
            expanded = expanded * (n_points / 2) / np.sum(expanded)
            f_expanded = objective_func(expanded)
            
            if f_expanded < f_reflected:
                simplex[worst_idx] = expanded
            else:
                simplex[worst_idx] = reflected
        else:
            # Contraction
            c = 0.5
            if f_reflected < f_vals[worst_idx]:
                # Outside contraction
                contracted = centroid + c * (reflected - centroid)
            else:
                # Inside contraction
                contracted = centroid + c * (simplex[worst_idx] - centroid)
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (n_points / 2) / np.sum(contracted)
            f_contracted = objective_func(contracted)
            
            if f_contracted < f_vals[worst_idx]:
                simplex[worst_idx] = contracted
            else:
                # Shrink
                best_h_simplex = simplex[best_idx]
                for i in range(1, simplex_size):
                    simplex[i] = best_h_simplex + 0.5 * (simplex[i] - best_h_simplex)
        
        # Update best
        f_vals_new = []
        for h_tmp in simplex:
            f_vals_new.append(objective_func(h_tmp))
        best_idx_new = np.argmin(f_vals_new)
        if f_vals_new[best_idx_new] < best_c5:
            best_c5 = f_vals_new[best_idx_new]
            # Update best_h using the best point in simplex
            best_h = simplex[best_idx_new].copy()
        
        stepcount += 1
    
    # Final computation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
