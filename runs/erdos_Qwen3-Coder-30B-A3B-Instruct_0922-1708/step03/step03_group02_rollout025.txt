import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with two intervals
        h = np.zeros(n_points)
        start1, end1 = int(0.2/dx), int(0.4/dx)
        start2, end2 = int(1.2/dx), int(1.4/dx)
        h[start1:end1] = 1
        h[start2:end2] = 1
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    T = 0.1  # initial temperature
    alpha = 0.995  # cooling rate
    lr = 0.01  # learning rate
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step: Perturb 10% of elements
        h_trial = h.copy()
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            delta = np.random.normal(0, 0.05)
            h_trial[i] += delta
        
        h_trial = project(h_trial)
        c5_trial = compute_c5(h_trial)
        
        # Accept or reject based on SA criterion
        if c5_trial < c5 or np.random.rand() < np.exp(-(c5_trial - c5) / T):
            h = h_trial
            c5 = c5_trial
        
        # Ensure constraint satisfaction
        h = project(h)
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Gradient Descent Step
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            
            grad[i] = (c5_plus - c5) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = project(h_new)
        
        c5_new = compute_c5(h_new)
        if c5_new < c5:
            h = h_new
            c5 = c5_new
            lr = min(0.02, lr * 1.01)
        else:
            lr = max(0.001, lr * 0.95)
            
        # Update temperature
        T *= alpha
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
