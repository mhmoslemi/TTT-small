import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Add refinement tracking
    last_improvement = 0
    improvement_count = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iteration
            improvement_count += 1
            
        # Periodic restart after no improvement
        if iteration - last_improvement > 50:
            # Random perturbation
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            last_improvement = iteration
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute gradient using two-sided finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_minus = h.copy()
            h_plus[i] += eps
            h_minus[i] -= eps
            
            h_plus = np.clip(h_plus, 0, 1)
            h_minus = np.clip(h_minus, 0, 1)
            
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            h_minus = h_minus * (n_points / 2) / np.sum(h_minus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            corr_minus = np.correlate(h_minus, 1 - h_minus, mode='full')
            
            c5_plus = np.max(corr_plus) * dx
            c5_minus = np.max(corr_minus) * dx
            
            grad[i] = (c5_plus - c5_minus) / (2 * eps)
            
        # Update h using gradient descent with adaptive learning rate
        lr = 0.005 / (1.0 + 0.0001 * iteration)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
    # Final refinement with CVX optimization if beneficial
    # Create a refined version for potential improvement
    if time.time() - start_time < budget_s - 1:
        try:
            # Use a second-order approximation with CVXPY
            h_vars = cp.Variable(n_points)
            h_vars.value = best_h
            
            # Constraint
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Use a simple direct convex approximation
            # Although not perfect, with about 100-point discretization
            # we can try to clarify the overlap computation
            # For the optimal step function, the maximum should occur at lags
            # when significant overlap is possible, so we sample a few lags
            # to visualize potential for minimizing.
            
            # Solve weaker relaxation to improve final solution
            objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
            prob = cp.Problem(objective, constraints)
            
            # Use faster solver if available
            try:
                prob.solve(solver=cp.CBC, verbose=False)
            except:
                try:
                    prob.solve(solver=cp.SCIP, verbose=False)
                except:
                    try:
                        prob.solve(solver=cp.GLPK, verbose=False)
                    except:
                        prob.solve(solver=cp.ECOS, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_cvx = h_vars.value
                if h_cvx is not None and not np.any(np.isnan(h_cvx)):
                    h_cvx = np.clip(h_cvx, 0, 1)
                    h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                    
                    correlation = np.correlate(h_cvx, 1 - h_cvx, mode='full')
                    c5_cvx = np.max(correlation) * dx
                    
                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
        except:
            pass
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
