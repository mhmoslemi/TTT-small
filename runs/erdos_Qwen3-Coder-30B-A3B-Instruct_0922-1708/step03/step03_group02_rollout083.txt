import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # The objective is to minimize max_k int h(x)(1-h(x+k)) dx
    
    # Constraint: sum(h)*dx = 1 => sum(h) = n_points/2
    
    # We explore multiple strategies to avoid local minima
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Strategy 1: Gradient-based optimization with adaptive step size
    def optimize_with_gradient(h):
        for _ in range(100):  # Limited iterations per call
            if time.time() - start_time > budget_s - 1:
                break
                
            c5 = compute_c5(h)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Compute gradient with finite differences
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / eps
            
            # Adaptive learning rate
            lr = 0.005 / (1.0 + 0.01 * np.linalg.norm(grad))
            h_new = h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            
        return h
    
    # Strategy 2: Local search with random perturbations
    def local_search(h):
        for _ in range(50):
            if time.time() - start_time > budget_s - 1:
                break
            # Make small random perturbations
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            c5 = compute_c5(h_new)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_new.copy()
            h = h_new
        return h
    
    # Strategy 3: Simulated annealing inspired approach
    def simulated_annealing(h):
        temp = 1.0
        cooling_rate = 0.99
        for _ in range(200):
            if time.time() - start_time > budget_s - 1:
                break
            # Random neighbor
            h_new = h + np.random.normal(0, 0.05, n_points) * temp
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            c5_old = compute_c5(h)
            c5_new = compute_c5(h_new)
            
            if c5_new < c5_old or np.random.rand() < np.exp((c5_old - c5_new) / temp):
                h = h_new
                
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h_new.copy()
                
            temp *= cooling_rate
        return h
    
    # Run alternating strategies
    strategy_counter = 0
    while time.time() - start_time < budget_s - 1:
        strategy_counter += 1
        if strategy_counter % 4 == 1:
            h = optimize_with_gradient(h)
        elif strategy_counter % 4 == 2:
            h = local_search(h)
        elif strategy_counter % 4 == 3:
            h = simulated_annealing(h)
        else:  # reset with better h if found
            h = best_h.copy()
            h += np.random.normal(0, 0.005, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final check with exact evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
