import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing loop
    start_time = time.time()
    iterations = 0
    
    # Annealing
    while time.time() - start_time < budget_s * 0.7:
        iterations += 1
        # Compute current overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # New correlation
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(correlation_new) * dx
        
        # Accept or reject
        if new_c5 < current_c5 or np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
        
        T = alpha * T
        
        # Early exit if we're stuck
        if iterations % 100 == 0 and best_c5 < 0.381:
            break
    
    # Gradient descent refinement
    h = best_h.copy()
    lr = 0.001
    decay_rate = 0.95
    max_iter = 200
    
    for i in range(max_iter):
        if time.time() - start_time > budget_s * 0.95:
            break
            
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - best_c5) / eps
        
        # Update
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Recompute c5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_new.copy()
            
        h = h_new.copy()
        
        # Learning rate decay
        if i % 50 == 0 and i > 0:
            lr *= decay_rate
    
    return (best_h.tolist(), best_c5, n_points)
