import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200  # Use higher resolution in phase 1
    dx = 2.0 / n_points
    start_time = time.time()
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Frank-Zadoff-like sequence
        h = np.ones(n_points) * 0.5
        h[:n_points//2] = 1.0
        h[n_points//2:] = 0.0
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Phase 1: Binary search on C5 with QCQP
    low, high = 0.38, 0.4
    while high - low > 1e-4 and time.time() - start_time < budget_s * 0.7:
        C_mid = (low + high) / 2
        c = C_mid / dx
        
        # QCQP formulation
        h_var = cp.Variable(n_points)
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2]
        
        # Add overlap constraints
        for k in range(n_points):
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= c)
        
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_feasible = h_var.value
                # Update best feasible solution
                overlap = np.correlate(h_feasible, 1 - h_feasible, mode='full')
                c5_feasible = np.max(overlap) * dx
                if c5_feasible < best_c5:
                    best_c5 = c5_feasible
                    best_h = h_feasible.copy()
                high = C_mid
            else:
                low = C_mid
        except:
            low = C_mid
            
    # Phase 2: Refinement with gradient descent
    h = best_h.copy()
    alpha = 0.01
    max_iter = 1000
    for i in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Find shift with maximum overlap
        overlap = np.correlate(h, 1 - h, mode='full')
        max_shift = np.argmax(overlap)
        
        # Gradient of overlap at max shift
        grad = np.zeros_like(h)
        for j in range(n_points):
            # derivative w.r.t. h[j] for shift max_shift
            k = max_shift
            j_shift = (j + k) % n_points
            grad[j] = dx * (1 - h[j_shift]) - dx * h[(j - k) % n_points]
        
        # Update h
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        overlap = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(overlap) * dx
        
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Adaptive learning rate
        if i % 10 == 0:
            alpha *= 0.995
    
    # Downsample to 100 points
    final_h = best_h[::2]
    final_h = final_h * (100 / 2) / np.sum(final_h)
    
    return (final_h.tolist(), best_c5, 100)
