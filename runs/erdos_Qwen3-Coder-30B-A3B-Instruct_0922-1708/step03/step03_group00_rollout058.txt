import numpy as np
import cvxpy as cp
import time

def solve_convex_subproblem(h_init, max_shifts, dx):
    n_points = len(h_init)
    h = cp.Variable(n_points)
    C = cp.Variable()
    
    # Constraints
    constraints = [h >= 0, h <= 1, cp.sum(h) * dx == 1, C >= 0]
    
    # Add constraints for each shift
    for k in range(max_shifts):
        # Compute the correlation for shift k
        corr = 0
        for i in range(n_points):
            j = i + k
            if j < n_points:
                corr += h[i] * (1 - h[j])
        constraints.append(corr <= C)
    
    # Objective
    objective = cp.Minimize(C)
    prob = cp.Problem(objective, constraints)
    
    try:
        prob.solve(solver=cp.ECOS, max_iters=1000)
        return h.value, C.value
    except:
        return h_init, float('inf')

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_shifts = 200
    
    # Start with initial solution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Convex subproblem
        new_h, new_c5 = solve_convex_subproblem(h, max_shifts, dx)
        
        if new_c5 < best_c5:
            h = new_h
        else:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Rescale to sum constraint
        h = h * (n_points / 2) / np.sum(h)
    
    return (best_h.tolist(), best_c5, n_points)
