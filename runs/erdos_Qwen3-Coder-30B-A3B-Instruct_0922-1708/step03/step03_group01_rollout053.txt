import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to correct sum
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute the overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def objective(h_vals):
        return compute_c5(h_vals)
        
    def gradient(h_vals):
        # Compute gradient analytically
        # For each index i, the gradient is influenced by overlaps with nearby indices
        grad = np.zeros_like(h_vals)
        c5 = compute_c5(h_vals)
        
        # Find which k maximizes the correlation
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        k_max = np.argmax(correlation)
        
        # The maximum correlation is at k_max
        # derivative contribution from h[i] is (1-h[i+k_max]) - h[i-k_max]
        n = len(h_vals)
        for i in range(n):
            # Position i contributes to correlation with positions (i+k_max) and (i-k_max)
            j1 = (i + k_max) % n
            j2 = (i - k_max) % n
            if j1 < n and j2 < n:
                grad[i] = dx * ((1 - h_vals[j1]) - h_vals[j2])
        
        return grad
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization
    options = {'maxiter': 1000, 'ftol': 1e-8, 'gtol': 1e-8}
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimize with SLSQP
        try:
            result = minimize(objective, h, method='SLSQP', jac=gradient, bounds=bounds, 
                              constraints=constraints, options=options)
            if result.success:
                h = result.x
                # Re-normalize to exact constraint
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to random restart
                h = np.random.rand(n_points)
                h = h * (n_points / 2) / np.sum(h)
        except Exception:
            # Fallback to random restart
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
        
        # Evaluate C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
    return (best_h.tolist(), best_c5, n_points)
