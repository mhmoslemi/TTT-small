import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Convex relaxation approach
        h_relaxed = cp.Variable(n_points)
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points / 2
        ]
        
        # Evaluate the maximum overlap over all lags
        max_overlap = 0
        overlap_exprs = []
        for k in range(n_points):
            # Overlap at lag k
            if k == 0:
                overlap_expr = cp.sum(h_relaxed * (1 - h_relaxed))
            else:
                overlap_expr = cp.sum(h_relaxed[k:] * (1 - h_relaxed[:n_points-k]))
            overlap_exprs.append(overlap_expr)
            
        # Minimize the maximum overlap
        objective = cp.Minimize(cp.max(cp.stack(overlap_exprs)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
            if prob.status == cp.OPTIMAL:
                h_sol = h_relaxed.value
            else:
                h_sol = h
        except:
            h_sol = h
        
        # Project to a feasible binary-like solution
        h_rounded = np.round(h_sol).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust to exactly n_points / 2
        if current_sum > n_points / 2:
            # Need to remove some 1s
            diff = int(current_sum - n_points / 2)
            # Find indices with smallest relaxed values and flip them
            flip_indices = np.argsort(h_sol)[:diff]
            for idx in flip_indices:
                h_rounded[idx] = 0
        elif current_sum < n_points / 2:
            # Need to add some 1s
            diff = int(n_points / 2 - current_sum)
            # Find indices with largest relaxed values and flip them
            flip_indices = np.argsort(h_sol)[-diff:]
            for idx in flip_indices:
                h_rounded[idx] = 1
                
        # Further local optimization with gradient steps
        h_new = h_rounded.copy()
        # Ensure it sums to target
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h_new = np.clip(h_new, 0, 1)
        
        h = h_new
    
    # Final verification
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
