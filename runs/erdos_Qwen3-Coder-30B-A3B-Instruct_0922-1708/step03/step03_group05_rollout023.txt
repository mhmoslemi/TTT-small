import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    T = T0
    iteration = 0
    
    # Precompute all lags for efficiency
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate a neighbor solution
        h_new = h.copy()
        
        # Select 5% of points to perturb
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        
        for i in indices:
            # Add small random perturbation
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
            # Clip to [0,1]
            h_new[i] = np.clip(h_new[i], 0, 1)
        
        # Rescale to meet constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept with simulated annealing probability
        if c5_new < c5 or np.random.rand() < np.exp((c5 - c5_new) / T):
            h = h_new
        
        # Decrease temperature
        T *= alpha
        
    # Ensure final solution is feasible and recompute
    h_final = best_h.copy()
    h_final = h_final * (n_points / 2) / np.sum(h_final)
    correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
    final_c5 = np.max(correlation_final) * dx
    
    return (h_final.tolist(), final_c5, n_points)
