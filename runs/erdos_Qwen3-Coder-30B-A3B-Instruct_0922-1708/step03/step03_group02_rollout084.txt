import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():

        x_orig = np.linspace(0, 2, len(h_orig))

        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Try different initializations to avoid local minima
    for i in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb the current best solution as a starting point
        if i > 0:
            h = best_h.copy()
            # Add small random perturbations
            perturb = np.random.normal(0, 0.02, n_points)
            h = np.clip(h + perturb, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Iteratively refine using constrained optimization
        for iteration in range(500):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Compute current C5 value
            correlation = np.correlate(h, 1 - h, mode='full')
            c5_val = np.max(correlation) * dx
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h.copy()
                
            # Use scipy minimize for better convergence
            def objective(h_vec):
                correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
                return -np.max(correlation) * dx  # Negative for maximization
            
            # Define simple bounds and constraint for scipy
            bounds = [(0, 1) for _ in range(n_points)]
            constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points/2}
            
            try:
                result = minimize(objective, h, bounds=bounds, constraints=constraint,
                                method='SLSQP', options={'maxiter': 10, 'ftol': 1e-6})
                if result.success:
                    h = np.clip(result.x, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)

                    # Fallback: gradient descent step
                    eps = 1e-4
                    grad = np.zeros_like(h)

                        h_plus = np.clip(h_plus, 0, 1)
                        h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                        
                        corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                        c5_plus = np.max(corr_plus) * dx
                        
                        grad[j] = (c5_plus - c5_val) / eps
                    
                    lr = 0.01
                    h_new = h - lr * grad
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    h = h_new
            except:
                # Fallback to basic gradient descent if optimization fails
                eps = 1e-4
                grad = np.zeros_like(h)

                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)

    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
