import numpy as np
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h - start with the provided parent or a block function
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    # Normalize to satisfy the constraint sum(h) * dx = 1
    h = h * (n_points / 2) / np.sum(h)
    
    # Track the best solution
    best_h = h.copy()
    correlation = np.correlate(h, 1 - h, mode='full')
    best_c5 = np.max(correlation) * dx
    
    start_time = time.time()
    time_left = lambda: budget_s - (time.time() - start_time)
    
    # Adaptive temperature and perturbation parameters
    temperature = 0.1
    perturbation_scale = 0.1
    success_count = 0
    last_success_count = 0
    iteration = 0
    
    # Resolution schedule
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Main optimization loop
    iteration = 0
    while time_left() > 1:
        iteration += 1
        
        # Every 100 iterations, adjust perturbation scale based on recent success
        if iteration % 100 == 0:
            success_ratio = success_count / max(1, iteration - last_success_count)
            if success_ratio > 0.2:
                perturbation_scale *= 1.5
            elif success_ratio < 0.1:
                perturbation_scale *= 0.5
            perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
            last_success_count = iteration
            success_count = 0
        
        if iteration % 1000 == 0:
            # Periodically update the best solution
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Generate candidate h
        h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute overlap for candidate
        correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        # Accept or reject based on Metropolis criterion
        if c5_candidate < best_c5:
            h = h_candidate
            best_c5 = c5_candidate
            success_count += 1
        else:
            # Accept with probability
            delta = c5_candidate - best_c5
            if random.random() < np.exp(-delta / temperature):
                h = h_candidate
                success_count += 1
                
        # Reduce temperature
        temperature *= 0.99

    # Final check of the best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
