import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use 200 points for better resolution while staying under budget
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 500
    
    # Precompute lags
    lags = list(range(1, n_points))
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def project_feasible(h_vec):
        # Clip to [0,1] and rescale to sum = n_points/2
        h_proj = np.clip(h_vec, 0, 1)
        h_proj = h_proj * (n_points / 2) / np.sum(h_proj)
        return h_proj
    
    # Physics-inspired relaxation with smoothing
    smoothing_coeff = 0.1
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Check current solution
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Create variables for convex relaxation
        h_var = cp.Variable(n_points)
        
        # Define the objective: minimize C5 + smoothing term
        # The overlap term is approximated using the correlation
        # For convex relaxation, we use a proxy for the maximum
        # We'll use a quadratic approximation of the smoothing term
        
        # Just use the current points in a convex solve
        # Constraint: sum(h) = n_points/2
        constraints = [cp.sum(h_var) == n_points / 2]
        
        # Define quadratic smoothing terms
        smoothing = 0
        for i in range(n_points):
            for j in range(i+1, n_points):
                smoothing += (h_var[i] - h_var[j])**2
                
        objective_expr = cp.Minimize(
            cp.max(cp.conv(cp.reshape(h_var, (1, n_points)), 
                           cp.reshape(1 - h_var, (n_points, 1)), 
                           method='direct')) * dx + 
            smoothing_coeff * smoothing
        )
        
        prob = cp.Problem(objective_expr, constraints)
        
        # Solve the convex problem
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
            else:
                # Fall back to simpler approach if convex solver fails
                h_new = h.copy()
        except:
            # Fallback to gradient descent when convex fails
            h_new = h.copy()
            
        # Project back to feasible region
        h_new = project_feasible(h_new)
        
        # Dilute with a small amount of random perturbation
        if iteration % 20 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h_new += noise
            h_new = project_feasible(h_new)
        
        # Apply some damped oscillation to explore more
        if iteration % 10 == 0:
            oscillation = 0.0005 * np.sin(iteration * 0.1 + np.arange(n_points))
            h_new += oscillation
            h_new = project_feasible(h_new)
            
        h = h_new
        
        # Gradually decrease smoothing 
        if iteration % 100 == 0 and smoothing_coeff > 0.001:
            smoothing_coeff *= 0.99

    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
