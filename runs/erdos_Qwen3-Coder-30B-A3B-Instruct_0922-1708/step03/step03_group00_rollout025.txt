import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_velocity = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for i in range(n_particles):
        # Start from different locations
        if i == 0:
            particle = h.copy()
        else:
            # Random perturbation
            particle = np.clip(h + np.random.normal(0, 0.05, n_points), 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        
        particles.append(particle)
        velocities.append(np.random.uniform(-max_velocity, max_velocity, n_points))
        pbest.append(particle.copy())
        pbest_vals.append(float('inf'))
    
    # Initialize global best
    gbest = h.copy()
    gbest_val = float('inf')
    
    # PSO iterations
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        iteration += 1
        converged = True
        
        for i in range(n_particles):
            particle = particles[i]
            
            # Compute the overlap integral
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particle.copy()
                
            if c5 < gbest_val:
                gbest_val = c5
                gbest = particle.copy()
                no_improvement_count = 0
                converged = False
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                             c1 * r1 * (pbest[i] - particle) + 
                             c2 * r2 * (gbest - particle))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -max_velocity, max_velocity)
            
            # Update position
            particle_new = particle + velocities[i]
            
            # Project to feasible region
            particle_new = np.clip(particle_new, 0, 1)
            particle_new = particle_new * (n_points / 2) / np.sum(particle_new)
            
            particles[i] = particle_new
            
        if not converged:
            no_improvement_count = 0
        else:
            no_improvement_count += 1

    # Refine with gradient descent on best particle
    h = gbest.copy()
    for _ in range(100):  # limited iterations
        if time.time() - start_time > budget_s - 0.1:
            break
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Update with adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad

        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        new_c5 = 0
        new_correlation = np.correlate(h, 1 - h, mode='full')
        new_c5 = np.max(new_correlation) * dx
        
        if new_c5 < best_c5:
            best_c5 = new_c5
            best_h = h.copy()
        else:
            # Early stopping if no improvement
            break
            
    return (best_h.tolist(), best_c5, n_points)
