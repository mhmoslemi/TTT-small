import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h0 = np.array(initial_h_values)
        n_points = len(h0)
        h = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(h0)), h0)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    max_anneal = 500
    no_improve_limit = 100
    
    # Initial C5 computation
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * (2.0 / n_points)
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    # Annealing loop
    no_improve_count = 0
    for i in range(max_anneal):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        # Rescale to sum to n_points/2
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * (2.0 / n_points)
        
        delta = c5_new - c5
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            h = h_new
            c5 = c5_new
            
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        T = alpha * T
        if no_improve_count > no_improve_limit:
            break

    # Refinement with gradient descent
    max_refine = 200
    lr = 0.001
    
    # Gradient descent with finite differences
    for i in range(max_refine):
        if time.time() - start_time > budget_s - 2:
            break
            
        # Estimate gradient
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * (2.0 / n_points)
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * (2.0 / n_points)
        
        # Update if better or occasionally accept worse
        if c5_new < c5 or np.random.rand() < 0.01:
            h = h_new
            c5 = c5_new
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
        # Adjust learning rate
        if i % 50 == 0 and i > 0:
            lr *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
