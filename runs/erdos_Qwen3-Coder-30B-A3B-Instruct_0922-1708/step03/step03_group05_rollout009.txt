import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    velocities = np.zeros((num_particles, n_points))
    
    # Initialize particles with some from initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        
        for i in range(min(5, num_particles)):  # Seed 10% with initial
            particles[i] = h_init.copy()
            # Add small random perturbation
            perturbation = np.random.normal(0, 0.05, n_points)
            particles[i] += perturbation
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    else:
        # Initialize completely random
        for i in range(num_particles):
            particles[i] = np.random.uniform(0.1, 0.9, n_points)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Personal best positions and values
    p_best = particles.copy()
    p_best_vals = np.full(num_particles, float('inf'))
    
    # Global best
    g_best = None
    g_best_val = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        for i in range(num_particles):
            # Evaluate current particle
            h = particles[i]
            
            # Compute correlation (overlap integral)
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < p_best_vals[i]:
                p_best_vals[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_val:
                g_best_val = c5
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final evaluation of best solution
    correlation = np.correlate(g_best, 1 - g_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (g_best.tolist(), final_c5, n_points)
