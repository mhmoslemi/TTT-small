import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    # Compute C5 as max of correlation
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    start_time = time.time()
    tabu_list = set()
    tabu_size = 50
    step_size = 0.1
    stagnation_count = 0
    last_improvement = 0
    
    # Ensure initial feasible
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Main TABU search loop
    while time.time() - start_time < budget_s - 1:
        # Generate candidates by perturbing current solution
        candidates = []
        perturbation_count = max(1, int(0.2 * n_points))
        
        for _ in range(50):
            # Make perturbation
            candidate_h = h.copy()
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Project constraints
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h = candidate_h * scale_factor
            
            # Hash for TABU check
            h_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            
            if h_hash not in tabu_list:
                candidate_c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h, candidate_c5, h_hash))
        
        if len(candidates) == 0:
            # Fall back to random perturbation
            candidate_h = h.copy()
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h = candidate_h * scale_factor
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            candidates.append((candidate_h, candidate_c5, hashlib.md5(candidate_h.tobytes()).hexdigest()))
        
        # Find best candidate
        best_candidate = min(candidates, key=lambda x: x[1])
        candidate_h, candidate_c5, candidate_hash = best_candidate
        
        # Check for improvement
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            last_improvement = 0
        else:
            # Aspiration criterion for occasional worse moves
            if np.random.random() < 0.1:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                last_improvement = 0
            else:
                last_improvement += 1
        
        # Update TABU list
        tabu_list.add(candidate_hash)
        if len(tabu_list) > tabu_size:
            # Remove oldest entry (simple FIFO replacement)
            tabu_list.pop()
        
        # Update current solution
        h = best_h.copy()
        
        # Adaptive step size and reset
        stagnation_count += 1
        if last_improvement > 100:
            step_size *= 1.2
            tabu_list.clear()
            stagnation_count = 0
            last_improvement = 0
        elif stagnation_count > 500:
            # Periodic restart
            h = np.random.rand(n_points) * 0.2 + 0.4  # Random slightly biased
            h = h * (n_points / 2) / np.sum(h)
            stagnation_count = 0
        
        # Re-evaluate after tuning
        if stagnation_count % 10 == 0:
            h = np.clip(h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h = h * scale_factor
            
    return (best_h.tolist(), best_c5, n_points)
