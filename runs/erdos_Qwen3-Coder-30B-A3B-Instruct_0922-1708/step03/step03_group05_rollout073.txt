import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    perturb_fraction = 0.05
    perturb_magnitude = 0.1
    
    # Initialize
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    T = T0
    iteration = 0
    
    # Precompute full correlation for efficiency
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Generate neighborhood
        h_new = h.copy()
        indices_to_modify = np.random.choice(n_points, size=int(n_points * perturb_fraction), replace=False)
        
        for i in indices_to_modify:
            delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
            h_new[i] += delta
            
        # Ensure feasibility
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject
        if new_c5 < current_c5:
            h = h_new
        else:
            # Accept with probability exp(-(new_c5 - current_c5) / T)
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                h = h_new
                
        # Update temperature
        T *= alpha
        
    # Final refinement at higher resolution if possible
    if n_points < 200 and time.time() - start_time < budget_s - 5:

        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, np.linspace(0, 2, len(best_h)), best_h)
        h = h * (n_points / 2) / np.sum(h)
        
        # Second optimization phase
        max_iterations = 500
        T = T0
        for i in range(max_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            current_c5 = compute_c5(h)
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = h.copy()

            h_new = h.copy()
            indices_to_modify = np.random.choice(n_points, size=int(n_points * perturb_fraction), replace=False)
            
            for j in indices_to_modify:
                delta = np.random.uniform(-perturb_magnitude, perturb_magnitude)
                h_new[j] += delta

            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Compute new C5
            new_c5 = compute_c5(h_new)
            
            # Accept or reject
            if new_c5 < current_c5:
                h = h_new
            else:
                if np.random.rand() < np.exp(-(new_c5 - current_c5) / T):
                    h = h_new
                    
            T *= alpha
            
        # Final check
        final_c5 = compute_c5(best_h)
        
        return (best_h.tolist(), final_c5, n_points)
        
    # Final check
