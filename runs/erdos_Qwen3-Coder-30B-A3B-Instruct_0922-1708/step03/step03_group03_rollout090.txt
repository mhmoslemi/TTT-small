import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Binary search over C5
    low = 0.0
    high = 0.38085
    tolerance = 1e-5
    
    while time.time() - start_time < budget_s - 1 and high - low > tolerance:
        C_test = (low + high) / 2.0
        
        # Try to find h that satisfies overlap constraints
        feasible_h = None
        try:
            # Compute current correlations to identify problematic lags
            current_correlation = np.correlate(h, 1 - h, mode='full')
            max_correlation = np.max(current_correlation)
            # Focus on lags that contribute most to the maximum
            # This is a greedy heuristic to reduce optimization complexity
            problematic_lags = np.argsort(current_correlation)[-10:]  # Top 10
            
            # Use smaller set of constraints to improve efficiency
            lags_to_use = problematic_lags[problematic_lags > 0]  # Exclude lag 0
            
            # Convex feasibility check
            h_vars = cp.Variable(n_points)
            
            # Basic constraints
            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
            
            # Add overlap constraints for main lags
            for k in lags_to_use:
                # Approximate linearization of the correlation term
                # We approximate: h[i] * (1 - h[(i + k) % n]) ≈ 
                # 1/2 * (h[i] + h[(i + k) % n]) - 1/2 * h[i] * h[(i + k) % n]
                # For now, keep the tight constraint directly for simplicity
                # Using piecewise linearization is not straightforward in cvxpy
                # Thus, we will directly put a constraint
                corr_k = cp.sum(cp.multiply(h_vars, cp.roll(1 - h_vars, k)))
                # Approximation: linear constraints are often better for feasibility
                # But we stay closer to original formulation
                constraints.append(corr_k <= C_test * n_points / dx)
            
            # Objective - minimize a perturbation to stay near current solution
            objective = cp.Minimize(cp.sum_squares(h_vars - h))
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                if h_feasible is not None:
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                    # Check if this new feasible solution is actually better
                    corr_feasible = np.correlate(h_feasible, 1 - h_feasible, mode='full')
                    c5_feasible = np.max(corr_feasible) * dx
                    
                    if c5_feasible < best_c5:
                        best_c5 = c5_feasible
                        best_h = h_feasible
                    feasible_h = h_feasible
                    
        except Exception as e:
            # If MoSeK fails, fall back to a simpler restart strategy
            pass
        
        # If we found a feasible solution, update our search
        if feasible_h is not None:
            low = C_test
            h = feasible_h
        else:
            high = C_test
    
    # Final optimization with gradient descent on a feasible solution
    h = best_h.copy()
    
    # Run gradient updates to refine further
    while time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Try to perturb with gradient descent
        grad = np.zeros_like(h)
        eps = 1e-8
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Update h using gradient descent
        lr = 0.005
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
