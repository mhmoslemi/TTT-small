import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Create initial population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        population.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        population.append(h_init)
    
    # Fill the rest of the population with random variations
    for _ in range(pop_size - 1):
        h = np.random.uniform(0.2, 0.8, n_points)
        h = h / np.sum(h) * (n_points / 2)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep elite
        elite_count = int(elite_rate * pop_size)
        new_population = population[:elite_count]
        
        # Generate offspring through crossover and mutation
        while len(new_population) < pop_size:
            # Tournament selection
            if len(population) < tournament_size:
                selected_indices = list(range(len(population)))
            else:
                selected_indices = np.random.choice(len(population), tournament_size, replace=False)
            selected_fitness = [fitness[i] for i in selected_indices]
            parent1_idx = selected_indices[np.argmin(selected_fitness)]
            
            if np.random.random() < cross_rate and len(population) > 1:
                parent2_idx = np.random.choice([i for i in range(len(population)) if i != parent1_idx])
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([population[parent1_idx][:crossover_point], 
                                        population[parent2_idx][crossover_point:]])
                child2 = np.concatenate([population[parent2_idx][:crossover_point], 
                                        population[parent1_idx][crossover_point:]])
            else:
                child1 = population[parent1_idx].copy()
                child2 = population[parent1_idx].copy()
            
            # Mutation
            if np.random.random() < mut_rate:
                mut_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                for idx in mut_indices:
                    child1[idx] += np.random.normal(0, 0.1)
            if np.random.random() < mut_rate:
                mut_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
                for idx in mut_indices:
                    child2[idx] += np.random.normal(0, 0.1)
            
            # Ensure feasibility
            child1 = np.clip(child1, 0, 1)
            child1 = child1 * (n_points / 2) / np.sum(child1)
            
            child2 = np.clip(child2, 0, 1)
            child2 = child2 * (n_points / 2) / np.sum(child2)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population
        generation += 1
        
        # Check for improvement
        if generation % 10 == 0:
            curr_best = min(fitness)
            if curr_best < best_c5:
                no_improvement_count = 0
            else:
                no_improvement_count += 1
    
    # Final re-evaluation of best solution
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
