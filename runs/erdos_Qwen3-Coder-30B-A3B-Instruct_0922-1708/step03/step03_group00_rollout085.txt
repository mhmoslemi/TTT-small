import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Optimization loop
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use a better gradient estimation with finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Use adaptive learning rate based on gradient magnitude
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally try a small random perturbation to escape local optima
        if iteration % 100 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Refine solution with a local optimizer
    if time.time() - start_time < budget_s - 5:
        # Define the objective function to minimize
        def obj(h_flat):
            h = h_flat.reshape(-1)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            correlation = np.correlate(h, 1 - h, mode='full')
            return np.max(correlation) * dx
        
        # Define bounds and constraints
        bounds = [(0, 1) for _ in range(n_points)]
        constraints = [{'type': 'eq', 'fun': lambda h: np.sum(h) * dx - 1.0}]
        
        # Use scipy minimize with L-BFGS-B
        try:
            result = minimize(obj, h, method='L-BFGS-B', bounds=bounds, constraints=constraints, options={'maxiter': 100})
            if result.success:
                h_final = result.x.reshape(-1)
                h_final = np.clip(h_final, 0, 1)
                h_final = h_final * (n_points / 2) / np.sum(h_final)
                correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
                c5_final = np.max(correlation_final) * dx
                if c5_final < best_c5:
                    best_c5 = c5_final
                    best_h = h_final
        except:
            pass
    
    return (best_h.tolist(), best_c5, n_points)
