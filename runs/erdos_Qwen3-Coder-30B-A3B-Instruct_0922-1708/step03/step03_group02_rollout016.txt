import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturb_std = 0.1
    sa_steps = 2000
    
    # Nelder-Mead parameters
    simplex_size = 5
    
    start_time = time.time()
    
    # Simulated Annealing phase
    sa_elapsed = 0
    while sa_elapsed < sa_steps and (time.time() - start_time) < budget_s - 10:
        # Perturb a random subset of elements
        h_new = h.copy()
        indices_to_perturb = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        for i in indices_to_perturb:
            h_new[i] += np.random.normal(0, perturb_std)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Acceptance probability
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= alpha
        sa_elapsed += 1
    
    # Nelder-Mead refinement phase
    def objective(h_vec):
        h = h_vec.reshape(-1)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Initialize simplex
    simplex = [best_h]
    for _ in range(simplex_size - 1):
        h_pert = best_h + np.random.normal(0, 0.01, n_points)
        simplex.append(project_to_feasible(h_pert))
    
    # Nelder-Mead iterations
    nm_elapsed = 0
    while (time.time() - start_time) < budget_s - 1 and nm_elapsed < 1000:
        # Evaluate simplex
        values = [objective(h) for h in simplex]
        sorted_indices = np.argsort(values)
        
        # Sort simplex by values
        sorted_simplex = [simplex[i] for i in sorted_indices]
        sorted_values = [values[i] for i in sorted_indices]
        
        # Reflect
        x_bar = np.mean(sorted_simplex[:-1], axis=0)
        x_r = x_bar + (x_bar - sorted_simplex[-1])
        x_r = project_to_feasible(x_r)
        
        f_r = objective(x_r)
        if sorted_values[0] <= f_r < sorted_values[-2]:
            # Accept reflection
            simplex[-1] = x_r
        elif f_r < sorted_values[0]:
            # Expand
            x_e = x_bar + 2 * (x_r - x_bar)
            x_e = project_to_feasible(x_e)
            f_e = objective(x_e)
            if f_e < f_r:
                simplex[-1] = x_e
            else:
                simplex[-1] = x_r
        else:
            # Contract
            x_c = sorted_simplex[-1] + 0.5 * (x_bar - sorted_simplex[-1])
            x_c = project_to_feasible(x_c)
            f_c = objective(x_c)
            if f_c < sorted_values[-1]:
                simplex[-1] = x_c
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = project_to_feasible(simplex[i])
        
        # Update best if improved
        current_best_value = min(values)
        if current_best_value < best_c5:
            best_c5 = current_best_value
            best_h = simplex[np.argmin(values)].copy()
        
        nm_elapsed += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
