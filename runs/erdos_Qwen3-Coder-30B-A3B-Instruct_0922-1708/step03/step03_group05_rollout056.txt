import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = np.zeros((num_particles, n_points))
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        
        # Seed 10% of particles with initial solution
        for i in range(num_particles // 10):
            particles[i] = h_init
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize remaining particles randomly
    for i in range(num_particles // 10, num_particles):
        particles[i] = np.random.rand(n_points) * 0.5
        particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
    
    # Initialize velocities
    velocities = np.random.uniform(-0.1, 0.1, (num_particles, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Track personal bests
    p_best = particles.copy()
    p_best_C5 = np.full(num_particles, float('inf'))
    
    # Global best
    g_best_idx = 0
    g_best_C5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            correlation = np.correlate(h, 1 - h, mode='full')
            C5 = np.max(correlation) * dx
            
            # Update personal best
            if C5 < p_best_C5[i]:
                p_best_C5[i] = C5
                p_best[i] = h.copy()
                
            # Update global best
            if C5 < g_best_C5:
                g_best_C5 = C5
                g_best_idx = i
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (p_best[i] - particles[i]) + 
                            c2 * r2 * (p_best[g_best_idx] - particles[i]))
            
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
    
    # Return best solution found
    final_h = p_best[g_best_idx]
    correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_C5 = np.max(correlation) * dx
    
    return (final_h.tolist(), final_C5, n_points)
