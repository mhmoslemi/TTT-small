import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Sampling for shifts in [0, 2] with 200 points
    shift_samples = np.linspace(0, 2, 200)
    
    # Set an initial upper bound
    current_C = 0.38085
    max_iterations = 200  # Safe number of iterations
    
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current correlation values
        correlation_full = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation_full) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find the shift that produces the maximal overlap
        max_idx = np.argmax(correlation_full)
        # Get corresponding shift (handle the full convolution properly)
        shift_idx = max_idx - (len(h) - 1)
        shift = shift_idx * dx
        
        # Build the constraint matrix for the new constraint:
        # sum_{x} h[x] * (1 - h[x + shift]) <= C * dx
        # We sample the integral as sum_{j=0}^{n-1} h[j] * (1 - h[(j+shift_idx) modulo n])
        
        # Discretize shift to nearest index
        shift_idx = int(round(shift / dx))
        if shift_idx < 0:
            shift_idx = len(h) + shift_idx  # wrap around
        
        # Build constraint efficiently
        constraint_sum = 0
        for i in range(len(h)):
            j = (i + shift_idx) % len(h)
            constraint_sum += h[i] * (1 - h[j])
            
        # Add to global upper bound because we want to minimize C
        current_C = min(current_C, c5)
        
        # Use CVXPY to solve with new constraint
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(current_C)
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add the constraint corresponding to max overlap
        constraint_expr = 0
        for i in range(len(h)):
            j = (i + shift_idx) % len(h)
            constraint_expr += h_var[i] * (1 - h_var[j])
        constraints.append(constraint_expr <= current_C * dx)
        
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=50)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_var.value
                if h_sol is not None:
                    h = np.clip(h_sol, 0, 1)
                    # Rescale to sum to n_points/2
                    h = h * (n_points / 2) / np.sum(h)
        except:
            pass  # Fall back to old h if optimization fails
            
        # Occasionally inject small random perturbation
        if iteration % 100 == 0:
            pert = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + pert, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final validation
    correlation_final = np.correlate(best_h, 1 - best_h, mode='full')
    best_c5 = np.max(correlation_final) * dx
    
    return (best_h.tolist(), best_c5, n_points)
