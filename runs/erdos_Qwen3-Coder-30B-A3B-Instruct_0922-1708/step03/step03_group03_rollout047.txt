import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Resolvers for different perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature schedule parameters
    initial_temp = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    
    iterations_per_resolver = 1000
    success_count = 0
    recent_successes = 0
    
    # Track best solution
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    current_temp = initial_temp
    
    for resolver in resolvers:
        current_perturbation = resolver
        for i in range(iterations_per_resolver):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate h with perturbation
            h_candidate = h + np.random.normal(0, current_perturbation, n_points)
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap for candidate
            c5_candidate = evaluate_c5(h_candidate, dx)
            
            # Accept or reject
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                success_count += 1
                recent_successes += 1
                current_perturbation *= 0.95  # Adaptive refinement
                current_perturbation = max(current_perturbation, 0.001)
            else:
                # Metropolis acceptance with temperature
                delta = c5_candidate - best_c5
                if current_temp > min_temp and np.random.rand() < np.exp(-delta / current_temp):
                    h = h_candidate.copy()
                    recent_successes += 1
                else:
                    recent_successes = max(0, recent_successes - 1)
            
            # Adjust temperature
            if i % 100 == 0:
                current_temp *= temp_decay
                current_temp = max(current_temp, min_temp)
            
            # Adjust perturbation based on success ratio
            if i % 50 == 0 and i > 0:
                success_ratio = min(1.0, float(recent_successes) / 50.0)
                if success_ratio > 0.2:
                    current_perturbation *= 1.5
                elif success_ratio < 0.1:
                    current_perturbation *= 0.5
                current_perturbation = np.clip(current_perturbation, 0.001, 1.0)
                recent_successes = 0
            
            # Update best solution
            c5_current = evaluate_c5(h, dx)
            if c5_current < best_c5:
                best_c5 = c5_current
                best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def evaluate_c5(h, dx):
    """Compute maximum overlap integral."""
    n = len(h)
    max_overlap = 0
    
    # For each shift k from 1 to n-1
    for k in range(1, n):
        # Compute overlap for specific shift k
        overlap = 0.0
        for a in range(n - k):
            overlap += h[a] * (1 - h[a + k])
        overlap *= dx
        max_overlap = max(max_overlap, overlap)
    
    return max_overlap
