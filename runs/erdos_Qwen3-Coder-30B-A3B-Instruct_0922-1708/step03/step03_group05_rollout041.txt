import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Try convex relaxation approach
        try:
            # Define CVXPY variables
            h_relaxed = cp.Variable(n_points)
            
            # Constraints
            constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
            
            # Compute overlap for each lag, but avoid stack for memory
            overlaps = []
            for i in range(n_points):
                # We need to compute sum over valid indices
                # For lag i, h[j] * (1 - h[j+i]) for j from 0 to n_points - i - 1
                over = cp.sum(cp.multiply(h_relaxed[:n_points-i], 1-h_relaxed[i:]))
                overlaps.append(over)
            
            # Use maximum of these overlaps (equivalent to max_k)
            objective = cp.Minimize(cp.max(overlaps))
            prob = cp.Problem(objective, constraints)
            
            # Solve the relaxed problem
            prob.solve(solver=cp.ECOS, verbose=False, feastol=1e-6, reltol=1e-6)
            
            if prob.status == cp.OPTIMAL:
                h_relaxed_sol = h_relaxed.value
                
                # Rounding step
                h_rounded = (h_relaxed_sol >= 0.5).astype(float)
                
                # Adjust to meet the sum constraint
                current_sum = np.sum(h_rounded)
                needed_sum = n_points / 2
                
                if current_sum > needed_sum:
                    # Need to flip some 1s to 0s
                    # Find indices where h_relaxed_sol is closest to 0.5 (largest deviation)
                    diffs = np.abs(h_relaxed_sol - 0.5)
                    flip_indices = np.argsort(diffs)[:int(current_sum - needed_sum)]
                    h_rounded[flip_indices] = 0
                elif current_sum < needed_sum:
                    # Need to flip some 0s to 1s
                    diffs = np.abs(h_relaxed_sol - 0.5)
                    flip_indices = np.argsort(diffs)[-int(needed_sum - current_sum):]
                    h_rounded[flip_indices] = 1
                
                # Rescale
                h = h_rounded * (n_points / 2) / np.sum(h_rounded)
            else:
                # Fallback to gradient descent if convex solve fails
                grad = np.zeros_like(h)
                eps = 1e-5
                
                for i in range(n_points):
                    h_plus = h.copy()
                    h_plus[i] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[i] = (c5_plus - c5) / eps
                
                # Update h using gradient descent
                lr = 0.003
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except Exception:
            # Fallback to gradient descent if convex solve fails
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps
            
            # Update h using gradient descent
            lr = 0.003
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
