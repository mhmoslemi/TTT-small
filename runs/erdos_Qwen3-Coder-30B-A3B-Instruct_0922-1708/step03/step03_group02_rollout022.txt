import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        population.append(h_init.copy())
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        population.append(h_init.copy())
    
    # Fill population with diverse individuals
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points) * 0.5 + 0.25  # Random values in [0.25, 0.75]
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    population = np.array(population)
    
    def compute_c5(h):
        # Compute convolution
        corr = correlate(h, 1 - h, mode='full')
        max_corr = np.max(corr)
        return max_corr * dx
    
    def fitness(h):
        return compute_c5(h)
    
    def feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def tournament_select(pop, fitnesses, k):
        indices = np.random.choice(len(pop), size=k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return pop[best_idx].copy()
    
    def crossover(p1, p2):
        if np.random.rand() > cross_rate:
            return p1.copy()
        cut = np.random.randint(1, n_points)
        child = np.concatenate([p1[:cut], p2[cut:]])
        return child
    
    def mutate(h):
        mask = np.random.rand(n_points) < mut_rate
        noise = np.random.normal(0, 0.1, n_points)
        h = h + mask * noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        # Evaluate fitness
        fitnesses = [fitness(h) for h in population]
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Selection
        elite_count = int(pop_size * elite_rate)
        elite_indices = np.argsort(fitnesses)[:elite_count]
        new_pop = [population[i].copy() for i in elite_indices]
        
        # Generate offspring
        while len(new_pop) < pop_size:
            parent1 = tournament_select(population, fitnesses, tournament_size)
            parent2 = tournament_select(population, fitnesses, tournament_size)
            child = crossover(parent1, parent2)
            child = mutate(child)
            new_pop.append(child)
        
        # Keep population size
        population = np.array(new_pop[:pop_size])
        generation += 1
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
