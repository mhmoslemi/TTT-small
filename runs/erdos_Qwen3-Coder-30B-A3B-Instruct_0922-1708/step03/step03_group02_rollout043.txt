import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with the provided parent construction or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to exact sum constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    temp = 0.1
    cooling_rate = 0.995
    perturbation_magnitude = 0.1
    
    # Nelder-Mead optimizer
    def nelder_mead_optimize(current_h):
        # Create initial simplex around current point
        simplex_size = min(10, n_points)
        simplex = [current_h.copy()]
        for _ in range(simplex_size-1):
            perturb = np.random.normal(0, perturbation_magnitude, n_points)
            new_h = current_h + perturb
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            simplex.append(new_h)
        
        # Objective function for simplex vertices
        def objective(h_vec):
            h_vec = np.clip(h_vec, 0, 1)
            h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
            corr = np.correlate(h_vec, 1 - h_vec, mode='full')
            return np.max(corr) * dx
        
        # Simplex optimization loop
        iters = 0
        while iters < 50 and time.time() - start_time < budget_s - 1:
            # Evaluate all points
            values = [objective(point) for point in simplex]
            
            # Sort by value
            sorted_indices = np.argsort(values)
            simplex = [simplex[i] for i in sorted_indices]
            values = [values[i] for i in sorted_indices]
            
            # Calculate centroid of best points (excluding worst)
            centroid = np.mean(simplex[:-1], axis=0)
            
            # Reflect worst point
            worst = simplex[-1]
            reflect = centroid + (centroid - worst)
            reflect = np.clip(reflect, 0, 1)
            reflect = reflect * (n_points / 2) / np.sum(reflect)
            reflect_val = objective(reflect)
            
            if values[0] <= reflect_val < values[-2]:
                # Accept reflection
                simplex[-1] = reflect
            elif reflect_val < values[0]:
                # Expand
                expand = centroid + 2 * (reflect - centroid)
                expand = np.clip(expand, 0, 1)
                expand = expand * (n_points / 2) / np.sum(expand)
                expand_val = objective(expand)
                if expand_val < reflect_val:
                    simplex[-1] = expand
                else:
                    simplex[-1] = reflect
            else:
                # Contract
                if reflect_val < values[-1]:
                    # Outside contraction
                    contract = centroid + 0.5 * (worst - centroid)
                else:
                    # Inside contraction
                    contract = centroid - 0.5 * (worst - centroid)
                contract = np.clip(contract, 0, 1)
                contract = contract * (n_points / 2) / np.sum(contract)
                contract_val = objective(contract)
                if contract_val < values[-1]:
                    simplex[-1] = contract
                else:
                    # Shrink towards best point
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
            
            iters += 1
        
        # Return best point from refined simplex
        best_simplex = [objective(point) for point in simplex]
        best_index = np.argmin(best_simplex)
        return simplex[best_index]
    
    # Simulated Annealing loop
    steps = 0
    max_annealing_steps = 2000
    while steps < max_annealing_steps and time.time() - start_time < budget_s - 1:
        # Perturb h (5% of points at a time)
        num_perturb = max(1, n_points // 20)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        h_new = h.copy()
        for i in indices:
            h_new[i] += np.random.normal(0, perturbation_magnitude)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        # Accept or reject based on temperature
        if c5_new < best_c5:
            h = h_new.copy()
            best_h = h.copy()
            best_c5 = c5_new
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / temp):
            h = h_new.copy()
        
        # Update temperature
        temp *= cooling_rate
        steps += 1
    
    # Final optimizations using Nelder-Mead
    while time.time() - start_time < budget_s - 1:
        h_improved = nelder_mead_optimize(best_h)
        corr = np.correlate(h_improved, 1 - h_improved, mode='full')
        c5_improved = np.max(corr) * dx
        
        if c5_improved < best_c5:
            best_h = h_improved
            best_c5 = c5_improved

    # Final verification
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
