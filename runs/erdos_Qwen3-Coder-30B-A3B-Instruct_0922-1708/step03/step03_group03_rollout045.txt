import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Resolvers for different scales of perturbation
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature for simulated annealing
    temperature = 0.1
    start_time = time.time()
    
    # Success tracking for perturbation adaptation
    recent_successes = []
    last_improvement_iter = 0
    
    iteration = 0
    num_iterations_per_resolver = 1000
    
    for resolver in resolvers:
        for _ in range(num_iterations_per_resolver):
            iteration += 1
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate with perturbation
            perturbation = np.random.normal(0, resolver, n_points)
            h_candidate = h + perturbation
            
            # Clip to feasible range
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Rescale to meet sum constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute maximum overlap
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept or reject based on improvement or temperature
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                last_improvement_iter = iteration
                recent_successes.append(1)
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.rand() < np.exp(-delta / temperature):
                    h = h_candidate.copy()
                    recent_successes.append(1)
                else:
                    recent_successes.append(0)
            
            # Update temperature
            if iteration % 100 == 0:
                temperature *= 0.99
                temperature = max(temperature, 1e-6)
            
            # Adjust perturbation scale based on success ratio
            if len(recent_successes) > 50:
                recent_successes = recent_successes[-50:]
                success_ratio = sum(recent_successes) / len(recent_successes)
                if success_ratio > 0.2:
                    resolver *= 1.5
                elif success_ratio < 0.1:
                    resolver *= 0.5
                resolver = max(0.001, min(1.0, resolver))
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
