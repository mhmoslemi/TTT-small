import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use higher resolution for global search, then downsample
    n_points = 200
    dx = 2.0 / n_points

    # Initialize using the provided parent or Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create Frank-Zadoff-like step function (approximate)
        h = np.ones(n_points)
        # Sample a few intervals with different values
        intervals = [(0, 0.5), (0.5, 1.0), (1.0, 1.5), (1.5, 2.0)]
        for i, (start, end) in enumerate(intervals):
            idx_start = int(start / 2.0 * n_points)
            idx_end = int(end / 2.0 * n_points)
            if i % 2 == 0:
                h[idx_start:idx_end] = 0.2
            else:
                h[idx_start:idx_end] = 0.8
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    # Phase 1: Binary search for lower bound on C5
    low = 0.38
    high = 0.4
    best_h = h.copy()
    best_c5 = float('inf')

    while high - low > 1e-4 and time.time() - start_time < budget_s - 5:
        C_mid = (low + high) / 2.0
        
        # Build constraints for QCQP checking feasibility
        # We do this roughly by checking a few key overlaps
        num_shifts = min(100, n_points)
        feasible = True
        
        # For small enough shifts, we enforce:
        # sum(h[i] * (1 - h[i+k])) <= C_mid / dx for small k
        # This is a relaxation, but sufficient for early cut-off
        if num_shifts > 50:
            shifts_to_check = list(range(0, num_shifts, 10))
        else:
            shifts_to_check = list(range(num_shifts))

        # Try small number of shifts to see feasibility
        tmp_h = h.copy()
        corr_max_all = []
        for k in shifts_to_check[:30]:  # try first few
            if k == 0:
                continue
            shift = np.roll(h, k)
            overlap = np.sum(h * (1 - shift))
            corr_max_all.append(overlap)
        if corr_max_all and np.max(corr_max_all) * dx > C_mid:
            low = C_mid
            continue
        
        # If we got here, try some refinement in phase 2
        break
    
    # Phase 2: Refinement with gradient-based optimization
    # Use a portion of the remaining budget for local optimization
    max_iter = 3000
    alpha = 0.01
    step_size_decay = 0.995
    max_iter_local = min(2500, max_iter)
    
    last_improvement = 0
    for iter_num in range(max_iter_local):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            last_improvement = iter_num
            
        # Get the shift that gives the maximum overlap
        max_shift_idx = np.argmax(correlation)
        max_shift = max_shift_idx - len(h) + 1  # because numpy returns full convolution

        # Compute the gradient with respect to h for this shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Analytical gradient for shift k
            j_plus = (i + max_shift) % n_points
            j_minus = (i - max_shift) % n_points
            grad[i] = dx * ((1 - h[j_plus]) - h[j_minus])

        # Update h using gradient descent
        h_new = h - alpha * grad

        # Project and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adjust step size
        alpha *= step_size_decay
        
        # Stop early if no improvement in recent iterations
        if iter_num - last_improvement > 100:
            break
          
    # Final processing - downsample to 100 points
    final_h = best_h.copy()
    final_n_points = 100
    final_dx = 2.0 / final_n_points
    if len(final_h) != final_n_points:
        x_orig = np.linspace(0, 2, len(final_h))
        x_new = np.linspace(0, 2, final_n_points)
        final_h = np.interp(x_new, x_orig, final_h)
        final_h = np.clip(final_h, 0, 1)
        final_h = final_h * (final_n_points / 2) / np.sum(final_h)
    
    # Compute final C5 score
    corr_final = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr_final) * final_dx
    
    return (final_h.tolist(), final_c5, final_n_points)
