import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Timer
    start_time = time.time()
    
    # Simulated Annealing Loop
    T = T0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Perturb h
        h_new = h.copy()
        # Select random points to perturb (5%)
        indices = np.random.choice(n_points, size=max(1, n_points // 20), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        # Clamp to [0,1]
        h_new = np.clip(h_new, 0, 1)
        # Rescale to maintain constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Evaluate current solution
        correlation_old = np.correlate(h, 1 - h, mode='full')
        c5_old = np.max(correlation_old) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            h = h_new.copy()
        elif np.random.rand() < np.exp((c5_old - c5_new) / T):
            h = h_new.copy()
            
        # Update temperature
        T *= alpha
        
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
