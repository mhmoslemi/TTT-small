import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5  # Default uniform initialization
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Temperature schedule
    temp = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    
    # Resolver levels - perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Track recent improvements
    recent_improvements = []
    max_recent = 50
    
    # Optimization loop
    for resolver in resolvers:
        if time.time() - start_time > budget_s - 1:
            break
            
        num_iterations = 1000
        perturbation_scale = resolver
        
        for i in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            # Generate candidate
            noise = np.random.normal(0, perturbation_scale, n_points)
            h_candidate = h + noise
            
            # Clip and rescale
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap integral using numpy's correlate
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5_candidate = np.max(correlation) * dx
            
            # Accept if better
            if c5_candidate < best_c5:
                h = h_candidate.copy()
                best_c5 = c5_candidate
                recent_improvements.append(1)
                # Reduce perturbation scale by 10% after improvement
                perturbation_scale *= 0.9
            else:
                # Accept with probability based on temperature
                delta = c5_candidate - best_c5
                if np.random.rand() < np.exp(-delta / temp):
                    h = h_candidate.copy()
                    recent_improvements.append(1)
                else:
                    recent_improvements.append(0)
                    
            # Decrement temperature
            if i % 100 == 0:
                temp = max(temp * temp_decay, min_temp)
                
            # Adapt perturbation scale
            if len(recent_improvements) > max_recent:
                recent_improvements.pop(0)
            success_ratio = sum(recent_improvements) / len(recent_improvements)
            if success_ratio > 0.2:
                perturbation_scale *= 1.5
            elif success_ratio < 0.1:
                perturbation_scale *= 0.5
                
            # Clip perturbation scale
            perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
    
    # Final check with best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
