import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a Barker-like initialization
        h = np.zeros(n_points)
        # Use a pattern that balances high and low values to reduce overlap
        # Divide into 4 segments, alternating high/low
        segment_size = n_points // 4
        for i in range(4):
            start_idx = i * segment_size
            end_idx = (i + 1) * segment_size if i < 3 else n_points
            if i % 2 == 0:
                h[start_idx:end_idx] = 1.0  # High values
            else:
                h[start_idx:end_idx] = 0.0  # Low values
        # Normalize to meet constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Gradient-based optimization
    def objective(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def grad_objective(h):
        # Numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h)
        base_c5 = objective(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
            
        return grad
    
    # Define bounds for optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with restarts
    max_iterations = 5000
    iteration = 0
    inner_iter = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Call optimizer
        try:
            result = minimize(
                objective,
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=grad_objective,
                options={'maxiter': 100, 'ftol': 1e-9, 'gtol': 1e-9}
            )
            
            if result.success:
                h_new = result.x
                # Rescale to meet constraint
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
                # Check if this is a better solution
                current_c5 = objective(h)
                if current_c5 < best_c5:
                    best_h = h.copy()
                    best_c5 = current_c5
                    
        except:
            pass
            
        # Periodic restarts with different initialization
        if iteration % 50 == 0 and iteration > 0:
            # Apply a different initialization pattern
            h = np.zeros(n_points)
            segment_size = n_points // 4
            for i in range(4):
                start_idx = i * segment_size
                end_idx = (i + 1) * segment_size if i < 3 else n_points
                if i % 2 == 0:
                    h[start_idx:end_idx] = 0.5  # Medium values
                else:
                    h[start_idx:end_idx] = 1.5
            h = h * (n_points / 2) / np.sum(h)
            
        iteration += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
