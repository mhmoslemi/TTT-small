import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate C5 for given h"""
    correlation = correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def normalize_h(h, n_points):
    """Ensure sum(h) = n_points / 2 and h in [0,1]"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def mutate(individual, mut_rate, n_points):
    """Mutation: Gaussian noise with clipping and rescaling"""
    mutation_mask = np.random.random(n_points) < mut_rate
    individual = individual.copy()
    individual[mutation_mask] += np.random.normal(0, 0.1, np.sum(mutation_mask))
    return normalize_h(individual, n_points)

def crossover(parent1, parent2, cross_rate, n_points):
    """Single-point crossover"""
    if np.random.random() > cross_rate:
        return parent1.copy(), parent2.copy()
    
    crossover_point = np.random.randint(1, n_points)
    child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
    child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
    
    return normalize_h(child1, n_points), normalize_h(child2, n_points)

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = normalize_h(h, n_points)
        population.append(h)
    else:
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Fill the rest of population
    for i in range(pop_size - 1):
        h = np.random.uniform(0.25, 0.75, n_points)
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Best solution tracking
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    
    gen = 0
    no_improvement = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement < max_no_improvement:
        gen += 1
        
        # Evaluate population
        fitness = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitness.append(c5)
        
        # Sort by fitness
        sorted_indices = np.argsort(fitness)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitness = [fitness[i] for i in sorted_indices]
        
        # Track best
        if sorted_fitness[0] < best_c5:
            best_c5 = sorted_fitness[0]
            best_h = sorted_population[0].copy()
            no_improvement = 0
        else:
            no_improvement += 1
        
        # Selection and reproduction
        new_population = []
        
        # Elitism
        elite_count = int(pop_size * elite_rate)
        new_population.extend(sorted_population[:elite_count])
        
        # Tournament selection + crossover + mutation
        while len(new_population) < pop_size:
            # Tournament selection
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            
            parent1 = sorted_population[winner_index]
            
            # Select second parent
            tournament_indices = np.random.choice(pop_size, tournament_size)
            tournament_fitness = [sorted_fitness[i] for i in tournament_indices]
            winner_index = tournament_indices[np.argmin(tournament_fitness)]
            parent2 = sorted_population[winner_index]
            
            # Crossover
            child1, child2 = crossover(parent1, parent2, cross_rate, n_points)
            
            # Mutation
            child1 = mutate(child1, mut_rate, n_points)
            child2 = mutate(child2, mut_rate, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
    
    # Final check
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
        return (best_h.tolist(), final_c5, n_points)
    else:
        return ([0.5] * n_points, 0.5, n_points)
