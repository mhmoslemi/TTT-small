import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-6
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient for the maximum correlation case
        correlation = np.correlate(h, 1 - h, mode='full')
        max_lag_idx = np.argmax(correlation)
        # The full correlation is of length 2*n_points - 1
        # So the actual lag k is from 0 to 2*n_points - 2
        # For shift k, correlation[k] corresponds to shift k if k < n_points 
        # or shift k-2*n_points if k >= n_points  
        # We care about which k gives the maximum value
        k_max = max_lag_idx - (n_points - 1)  # k is in range -n_points+1 to n_points-1
  
        # Compute gradient analytically for k_max
        grad = np.zeros_like(h)
        # For each i, compute gradient component for shift k_max
        # This is the derivative of h[i] * (1 - h[i-k_max]) w.r.t h[i] and h[i+k_max], if not out of bounds
        for i in range(n_points):
            if (i - k_max) >= 0 and (i - k_max) < n_points:
                grad[i] -= dx  # derivative of h[i] * (1 - h[i-k_max])
            if (i + k_max) >= 0 and (i + k_max) < n_points:
                grad[i] -= dx  # derivative of h[i] * (1 - h[i+k_max])
                
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            # restart with a new piecewise linear function
            slope = 1.0 / n_points  # Note: y=mx + b where b = 0
            t = np.linspace(0, 2, n_points)
            h = np.clip(0.5 + 0.5 * np.sin(np.pi * t), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
                
    return (best_h.tolist(), best_c5, n_points)
