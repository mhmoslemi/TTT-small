import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Identify the shift that maximizes overlap
        correlation = np.correlate(h, 1 - h, mode='full')
        max_lag = np.argmax(correlation)
        # Convert lag index to actual shift k
        k_max = max_lag - (n_points - 1)
        
        # Compute exact gradient w.r.t. this maximum shift
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Gradient contribution from h[i] to overlap with shifted h
            # For shift k_max, i contributes to overlap with (i-k_max)%n_points
            j1 = (i - k_max) % n_points  # h[i] * (1 - h[j1])
            j2 = (i + k_max) % n_points  # h[i] * (1 - h[j2]) - only if k_max > 0
            
            if k_max > 0:
                # h[i] contributes to the overlap at positions (i-k_max) and (i+k_max)
                grad[i] = dx * (1 - h[j1] - h[j2])
            elif k_max < 0:
                # h[i] contributes to the overlap at positions (i+k_max) and (i-k_max)
                grad[i] = dx * (1 - h[j1] - h[j2])
            else:
                # k_max = 0, which should not happen due to max_lag calculation
                grad[i] = dx * (1 - h[(i - k_max) % n_points] - h[(i + k_max) % n_points])
                
        # Special handling for k=0 case
        if k_max == 0:
            # This should not happen in practice, but handle it just in case
            for i in range(n_points):
                grad[i] = dx * (1 - 2 * h[i])  # Optimize origin contribution differently
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decrease learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart with a different mixture
        if iteration % 500 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                # Perturb the current solution
                h_perturbed = h + 0.01 * (np.random.rand(n_points) - 0.5)
                h_perturbed = np.clip(h_perturbed, 0, 1)
                h = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
                
    return (best_h.tolist(), best_c5, n_points)
