import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vel_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest_positions = []
    pbest_values = []
    
    # Start with the best solution or initial
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest_positions.append(h.copy())
    pbest_values.append(float('inf'))
    
    # Generate additional random particles
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand.copy())
        velocities.append(np.random.uniform(-vel_max, vel_max, n_points))
        pbest_positions.append(h_rand.copy())
        pbest_values.append(float('inf'))
    
    # Global best
    gbest_position = None
    gbest_value = float('inf')
    
    # PSO main loop
    pso_iterations = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        pso_iterations += 1
        improved = False
        
        # Evaluate all particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Evaluate this particle's C5
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update pbest
            if c5 < pbest_values[i]:
                pbest_values[i] = c5
                pbest_positions[i] = particle.copy()
                improved = True
                if c5 < gbest_value:
                    gbest_value = c5
                    gbest_position = particle.copy()
                    
        # Update best overall
        if improved:
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Update particles
        for i in range(n_particles):
            if i == 0:  # First particle is the current best found
                continue
                
            # Update velocity
            r1, r2 = np.random.rand(2)
            vel = (inertia * velocities[i] + 
                   c1 * r1 * (pbest_positions[i] - particles[i]) + 
                   c2 * r2 * (gbest_position - particles[i]))
            
            # Clip velocity
            vel = np.clip(vel, -vel_max, vel_max)
            
            # Update position
            new_pos = particles[i] + vel
            
            # Apply constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
            velocities[i] = vel
            
        # Occasionally remove particles that are too far from the best
        if pso_iterations % 20 == 0:
            current_best_idx = np.argmin(pbest_values)
            current_best = pbest_positions[current_best_idx]
            
            for i in range(n_particles):
                dist = np.linalg.norm(particles[i] - current_best)
                if dist > 1.0 and i != 0:
                    h_new = np.random.rand(n_points)
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    particles[i] = h_new.copy()
                    velocities[i] = np.random.uniform(-vel_max, vel_max, n_points)
                    
    # Refine with gradient descent on the very best
    if gbest_position is not None:
        h_refined = gbest_position.copy()
        
        # Define callback to store the best value
        best_c5_refined = gbest_value

        # Perform gradient descent
        for _ in range(1000):
            if time.time() - start_time > budget_s - 1:
                break
            
            # Gradient estimation with finite differences
            grad = np.zeros_like(h_refined)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - best_c5_refined) / eps
            
            # Adaptive learning rate
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h_refined - lr * grad
            
            # Apply constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Evaluate new value
            correlation = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(correlation) * dx
            
            if c5_new < best_c5_refined:
                best_c5_refined = c5_new
                h_refined = h_new.copy()
                
            h_refined = h_new.copy()
            
        # Final update if better than evaluated
        if best_c5_refined < best_c5:
            best_c5 = best_c5_refined
            best_h = h_refined.copy()
            
    return (best_h.tolist(), best_c5, n_points)
