import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Barker-like initialization pattern
        h = np.zeros(n_points)
        # Create a pattern with alternating high/low regions
        for i in range(n_points):
            if i < n_points // 4 or (i >= n_points // 2 and i < 3*n_points//4):
                h[i] = 1.0
            else:
                h[i] = 0.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute correlation indices
    def compute_overlap(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def objective(h_vec):
        return compute_overlap(h_vec)
    
    def gradient(h_vec):
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_overlap(h_plus)
            c5_orig = compute_overlap(h_vec)
            grad[i] = (c5_plus - c5_orig) / eps
        return grad
    
    # Main optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        current_c5 = compute_overlap(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Use L-BFGS-B for optimization
        try:
            # Rescale to satisfy integral constraint in optimization
            h_scaled = h.copy()
            h_scaled = h_scaled * (n_points / 2) / np.sum(h_scaled)
            
            # Define bounds for each variable
            bounds = [(0, 1) for _ in range(n_points)]
            
            # Optimize with bounds
            result = minimize(objective, h_scaled, method='L-BFGS-B', 
                             bounds=bounds, jac=gradient, 
                             options={'maxiter': 10, 'gtol': 1e-6})
            
            if result.success:
                h = np.clip(result.x, 0, 1)
                # Renormalize
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to gradient descent if optimization fails
                lr = 0.001
                grad = gradient(h)
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except:
            # Fallback to simple gradient descent
            lr = 0.001
            grad = gradient(h)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check with full evaluation
    final_c5 = compute_overlap(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
