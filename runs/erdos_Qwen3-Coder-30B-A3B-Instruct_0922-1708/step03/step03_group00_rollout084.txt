import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_shifts = 200
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Use a cutting plane approach
    # Start with a large C
    C = 0.38085

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Build constraints based on current h
        constraints = []
        
        # Compute overlaps for a number of sample shifts
        shifts = np.linspace(0, 2, max_shifts)
        for shift in shifts:
            # Discretize the shift
            shift_idx = int(shift / dx)
            if shift_idx >= n_points or shift_idx < 0:
                continue
            # For shift k, we have integral h(x)(1-h(x+k))dx
            # Evaluate at all positions, accumulate
            # Correlation with 1-h shifted
            h_shifted = np.roll(1 - h, shift_idx)
            h_shifted[:shift_idx] = 0  # zero padding
            
            overlap = np.sum(h * h_shifted) * dx
            
            # Add constraint: integral h(x)(1 - h(x+k)) dx <= C
            # This becomes: sum(h[i] * (1 - h[i+shift_idx])) * dx <= C
            # sum(h[i] * (1 - h[i+shift_idx])) * dx <= C
            if overlap > C:
                C = overlap  # Tighten bound

        # Now we're trying to minimize C subject to constraints
        # We want to minimize the maximum overlap, so we'll solve:
        # minimize C
        # subject to integral h(x)(1 - h(x+k)) dx <= C for all k
        # subject to 0 <= h[i] <= 1 
        # subject to sum(h) * dx = 1
        
        # Simplified, Euclidean-like, gradient-based solve attempt
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite difference
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        # Adaptive learning rate
        lr = 0.05 / (1 + np.linalg.norm(grad) * 1e-2)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally try a small random perturbation
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally tighten C if we're making progress
        if iteration % 20 == 0:
            correlation = np.correlate(h, 1 - h, mode='full')
            new_c5 = np.max(correlation) * dx
            if new_c5 < C:
                C = new_c5
                
    return (best_h.tolist(), best_c5, n_points)
