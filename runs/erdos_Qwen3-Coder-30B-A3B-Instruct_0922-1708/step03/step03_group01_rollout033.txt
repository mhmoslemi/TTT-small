import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_generations = 120
    pop_size = 50
    crossover_rate = 0.5
    mutation_rate = 0.2
    elite_size = pop_size // 5
    no_improve_limit = 100
    
    # Initialize population
    population = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        # Default initial random population
        for _ in range(pop_size):
            h = np.random.rand(n_points)
            h = h * (n_points / 2) / np.sum(h)
            population.append(h.copy())
    
    # Add some diversity with different starting points
    for _ in range(pop_size - len(population)):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
        
    best_h = None
    best_c5 = float('inf')
    
    generation = 0
    no_improve_count = 0
    start_time = time.time()
    
    while generation < max_generations and (time.time() - start_time) < budget_s - 1:
        generation += 1
        
        # Evaluate fitness
        fitness_scores = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness_scores.append(c5)
            
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness_scores)
        population = [population[i] for i in sorted_indices]
        fitness_scores = [fitness_scores[i] for i in sorted_indices]
        
        # Update best solution
        if fitness_scores[0] < best_c5:
            best_c5 = fitness_scores[0]
            best_h = population[0].copy()
            no_improve_count = 0
        else:
            no_improve_count += 1
            
        if no_improve_count >= no_improve_limit:
            break
            
        # Generate new population
        new_population = []
        
        # Elitism: keep best individuals
        for i in range(elite_size):
            new_population.append(population[i].copy())
            
        # Create offspring through crossover
        while len(new_population) < pop_size:
            # Select parents (tournament selection)
            parent1_idx = np.random.randint(0, pop_size // 2)
            parent2_idx = np.random.randint(0, pop_size // 2)
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.rand() < crossover_rate:
                # Single-point crossover
                crossover_point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
                child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
                
                # Rescale to sum constraint
                child1 = child1 * (n_points / 2) / np.sum(child1)
                child2 = child2 * (n_points / 2) / np.sum(child2)
                
                new_population.append(child1)
                if len(new_population) < pop_size:
                    new_population.append(child2)
            else:
                # Clone
                new_population.append(parent1.copy())
                if len(new_population) < pop_size:
                    new_population.append(parent2.copy())
                    
        # Mutation
        for i in range(len(new_population)):
            if np.random.rand() < mutation_rate:
                h = new_population[i]
                # Mutate 5% of the points
                num_mutations = max(1, n_points // 20)
                mutation_indices = np.random.choice(n_points, num_mutations, replace=False)
                for idx in mutation_indices:
                    # Add small random change
                    h[idx] += np.random.normal(0, 0.05)
                    # Clip to [0,1]
                    h[idx] = np.clip(h[idx], 0, 1)
                    
                # Rescale to maintain constraint
                h = h * (n_points / 2) / np.sum(h)
                new_population[i] = h
                
        population = new_population
        
    # Final validation of best solution
    if best_h is not None:
        corr = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(corr) * dx
        # Ensure we are reporting the best value found
        if final_c5 < best_c5:
            best_c5 = final_c5
            
    # If no solution found, return initial or uniform
    if best_h is None:
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            h = np.ones(n_points) * 0.5
            h = h * (n_points / 2) / np.sum(h)
        corr = correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h
        
    return (best_h.tolist(), best_c5, n_points)
