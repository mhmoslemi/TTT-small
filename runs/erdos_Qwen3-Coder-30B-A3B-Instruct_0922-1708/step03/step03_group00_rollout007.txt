import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iterations = 200
    iteration = 0
    
    # Discretize shifts
    shift_points = 200
    shifts = np.linspace(0, 2, shift_points)
    
    # Cutting-plane approach
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        # Compute current max overlap for all shifts
        max_overlap = -1
        max_shift_idx = 0
        
        for shift_idx, k in enumerate(shifts):
            # Compute overlap integral for specific shift k
            if k == 0:
                overlap = np.sum(h * (1 - h)) * dx
            else:
                # Use convolution approach
                h_shifted = np.concatenate([h[-int(k/dx):], h[:-int(k/dx)]])
                overlap = np.sum(h * (1 - h_shifted)) * dx
                
            if overlap > max_overlap:
                max_overlap = overlap
                max_shift_idx = shift_idx
                
        if max_overlap < best_c5:
            best_c5 = max_overlap
            best_h = h.copy()
            
        # Add constraint for the most violating shift
        k = shifts[max_shift_idx]
        
        # Create variables for new linearized constraint
        h_var = cp.Variable(n_points)
        C_var = cp.Variable()
        
        # Objective: minimize C
        objective = cp.Minimize(C_var)
        
        # Constraints
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add constraint for current shift
        if k == 0:
            # Direct constraint for shift 0
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_var)) <= C_var
        else:
            # For non-zero shift
            # Create shifted version
            shift_int = int(k / dx)
            if shift_int >= n_points:
                h_shifted = np.zeros(n_points)
            else:
                h_shifted = np.concatenate([h_var[-shift_int:], h_var[:-shift_int]])
            
            constraint_expr = cp.sum(cp.multiply(h_var, 1 - h_shifted)) <= C_var
            
        constraints.append(constraint_expr)
        
        # Solve the optimization problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, max_iters=100)
            if prob.status == cp.OPTIMAL:
                h_new = h_var.value
                # Clip and rescale
                h = np.clip(h_new, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fall back to gradient approach if CVXPY fails
                break
        except:
            # Fall back to gradient approach if CVXPY fails
            break
            
        iteration += 1
        
    # Final verification and polishing
    for _ in range(10):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Gradient step
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
            
    # Final computation of C5
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
