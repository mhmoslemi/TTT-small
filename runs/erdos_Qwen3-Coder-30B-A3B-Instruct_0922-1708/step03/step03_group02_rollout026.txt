import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured construction
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization: two peaks
        h = np.zeros(n_points)
        i1 = int(0.2 * n_points)
        i2 = int(0.4 * n_points)
        i3 = int(1.2 * n_points)
        i4 = int(1.4 * n_points)
        h[i1:i2] = 1.0
        h[i3:i4] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    temp = 0.1
    cooling_rate = 0.995
    lr = 0.01
    perturb_rate = 0.1
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_trial = h.copy()
        n_perturb = int(n_points * perturb_rate)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_trial[i] += np.random.normal(0, 0.05)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute c5 for trial
        correlation_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(correlation_trial) * dx
        
        # Accept or reject based on temperature
        if c5_trial < best_c5 or np.random.rand() < np.exp(-(c5_trial - best_c5) / temp):
            h = h_trial.copy()
            c5_current = c5_trial
        else:
            c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
            
        # Gradient Descent Step
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update with gradient
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Update best if improved
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Adaptive learning rate
        lr = max(0.001, min(0.02, lr * 1.01 if c5_current < best_c5 else lr * 0.95))
        
        # Update temperature
        temp *= cooling_rate
        
        if temp < 1e-6:
            temp = 1e-6
    
    # Final recomputation of best_c5
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
