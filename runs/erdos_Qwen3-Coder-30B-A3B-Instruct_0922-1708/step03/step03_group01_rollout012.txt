import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    # Compute the overlap integral C_5
    n_points = len(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, otherwise use uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_list_size = 50
    stagnation_count = 0
    max_stagnation = 100
    step_size = 0.1
    perturbation_count = max(1, int(0.2 * n_points))
    aspiration_prob = 0.1
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < 10000:
        iteration += 1
        
        # Generate multiple candidate solutions
        candidates = []
        for k in range(50):
            # Make a copy of current solution
            candidate_h = h.copy()
            
            # Randomly select indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Add random perturbations
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
                
            # Clip and normalize
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h = candidate_h * scale_factor
            
            # Hash the candidate to check_TABU list
            candidate_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            
            if candidate_hash not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5, candidate_hash))
        
        # Find the best candidate not in TABU list
        if candidates:
            # Sort by C5 values
            candidates.sort(key=lambda x: x[1])
            best_candidate_h, best_candidate_c5, candidate_hash = candidates[0]
            
            # Update the best solution if we found a better one
            if best_candidate_c5 < best_c5:
                best_h = best_candidate_h.copy()
                best_c5 = best_candidate_c5
                stagnation_count = 0
            else:
                # Update stagnation count
                stagnation_count += 1
                
                # Occasionally allow worse solutions for escape
                if np.random.rand() < aspiration_prob:
                    best_h = best_candidate_h.copy()
                    best_c5 = best_candidate_c5
                    stagnation_count = 0
            
            # Add the previous h to the tabu list
            h_hash = hashlib.md5(h.tobytes()).hexdigest()
            tabu_list.add(h_hash)
            
            # Remove oldest entry if list exceeds size
            if len(tabu_list) > tabu_list_size:
                # Remove a random element to keep size track
                if tabu_list:
                    tabu_list.pop()
            
            # Update h to the selected candidate
            h = best_candidate_h.copy()
            
            # Adjust step size if stagnating
            if stagnation_count > max_stagnation:
                step_size *= 1.2
                stagnation_count = 0
                tabu_list.clear()
        else:
            # If no valid candidates, perturb the current solution
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                h[i] += delta
                
            h = np.clip(h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(h)
            h = h * scale_factor
            
            # Periodically reset tabu list if stuck
            if iteration % 1000 == 0:
                tabu_list.clear()
    
    # Final evaluation of best solution
    best_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), best_c5, n_points)
