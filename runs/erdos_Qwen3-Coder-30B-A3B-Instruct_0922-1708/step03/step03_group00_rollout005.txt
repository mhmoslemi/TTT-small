import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_shifts = 200
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Sampling shift values
    shift_values = np.linspace(0, 2, max_shifts)
    
    # Constraint list for the optimization
    constraints = []
    
    # Add initial constraints (from the initial h)
    corr_vals = np.correlate(h, 1 - h, mode='full')
    max_corr = np.max(corr_vals)
    if max_corr < best_c5:
        best_c5 = max_corr
        best_h = h.copy()
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < 200:
        iteration += 1
        
        h_old = h.copy()
        
        # Find the current maximum overlap
        corr_vals = np.correlate(h, 1 - h, mode='full')
        shift_idx = np.argmax(corr_vals)
        max_corr = corr_vals[shift_idx] * dx
        
        if max_corr < best_c5:
            best_c5 = max_corr
            best_h = h.copy()
            
        # Add this as a constraint to our convex problem
        # We'll linearize by computing the gradient (finite differences)
        # Parameters for approximation
        eps = 1e-4
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_corr_plus = np.max(corr_plus) * dx
            grad[i] = (max_corr_plus - max_corr) / eps
        
        # New constraint: 
        # \sum_i grad_i * h_i >= max_corr
        # or equivalently, -\sum_i grad_i * h_i <= -max_corr
        constraints.append(-cp.sum(cp.multiply(grad, cp.Variable(n_points))) <= -max_corr)
        
        # Solve the CVXPY problem
        h_var = cp.Variable(n_points)
        objective = cp.Minimize(cp.sum(cp.multiply(np.ones(n_points), h_var)))
        constraints_cvx = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ] + constraints
        
        # Set up problem
        prob = cp.Problem(objective, constraints_cvx)
        try:
            prob.solve(solver=cp.SCS, max_iters=1000, eps=1e-4)
            if prob.status == cp.OPTIMAL:
                h = h_var.value
            else:
                # Fallback: use gradient info to do a small step
                lr = 0.01
                h_new = h - lr * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
        except:
            # Fallback: gradient descent step
            lr = 0.01
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Ensuring constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Small chance of a random perturbation
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
