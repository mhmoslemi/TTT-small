import numpy as np
import cvxpy as cp
import time

def compute_c5(h, dx):
    """Compute C5 from h values."""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = compute_c5(h, dx)
    
    start_time = time.time()
    
    # Convex relaxation approach
    while time.time() - start_time < budget_s - 1:
        # Define convex problem
        h_relaxed = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Objective: minimize max overlap
        # Compute overlaps for each lag
        overlaps = []
        for k in range(n_points):
            # Compute sum for lag k
            if k == 0:
                overlap_k = cp.sum(cp.multiply(h_relaxed, 1 - h_relaxed))
            else:
                # This is a bit tricky; use proper slicing
                h1 = h_relaxed[k:]
                h2 = 1 - h_relaxed[:-k] if k < n_points else cp.Parameter(n_points-k, value=0)
                overlap_k = cp.sum(cp.multiply(h1, h2))
            overlaps.append(overlap_k)
        
        # For the max, we can reformulate this in a standard way
        # If using max_k overlap_k, we can add additional constraint:
        # t >= overlap_k for all k
        # Then minimize t. 
        # Let's construct explicitly and make convex problem:
        t = cp.Variable()
        for i, overlap in enumerate(overlaps):
            constraints.append(t >= overlap)
            
        objective = cp.Minimize(t)
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed_sol = h_relaxed.value
            else:
                h_relaxed_sol = h
        except:
            h_relaxed_sol = h
            
        # Rounding step
        h_rounded = (h_relaxed_sol >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust to meet sum constraint
        if abs(current_sum - n_points/2) < 1e-9:
            pass  # Already correct
        elif current_sum > n_points/2:
            # Need to flip some 1s to 0s
            diff = int(current_sum - n_points/2)
            # Get indices where h_relaxed_sol is smallest among 1s
            indices_1 = np.where(h_rounded == 1)[0]
            # Sort these by h_relaxed_sol
            sorted_indices = sorted(indices_1, key=lambda i: h_relaxed_sol[i])
            # Flip the smallest of these
            for i in range(min(diff, len(sorted_indices))):
                h_rounded[sorted_indices[i]] = 0
        else:
            # Need to flip some 0s to 1s  
            diff = int(n_points/2 - current_sum)
            # Get indices where h_relaxed_sol is largest among 0s
            indices_0 = np.where(h_rounded == 0)[0]
            # Sort these by h_relaxed_sol descending
            sorted_indices = sorted(indices_0, key=lambda i: h_relaxed_sol[i], reverse=True)
            # Flip the largest of these
            for i in range(min(diff, len(sorted_indices))):
                h_rounded[sorted_indices[i]] = 1
                
        # Final check
        c5 = compute_c5(h_rounded, dx)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
            
        # Update h for next iteration
        h = h_rounded
        
    return (best_h.tolist(), best_c5, n_points)
