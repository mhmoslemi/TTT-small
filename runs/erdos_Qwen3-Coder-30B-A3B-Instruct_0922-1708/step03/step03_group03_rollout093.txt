import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        # Rescale to meet constraint
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for tight bound
    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    while C_high - C_low > tolerance and time.time() - start_time < budget_s - 10:
        C_test = (C_low + C_high) / 2.0
        
        # Try to prove feasibility with convex optimization
        feasibility = check_feasibility(h, C_test, n_points, dx)
        
        if feasibility:
            C_high = C_test
            # Keep the better h found
            if feasibility[1] is not None:
                h = feasibility[1]
                if feasibility[0] < best_c5:
                    best_c5 = feasibility[0]
                    best_h = h.copy()
        else:
            C_low = C_test
            
    # If binary search failed to tighten further, try refinement via gradient descent
    if C_high - C_low <= tolerance:
        # Try local optimization
        h_refined = h.copy()
        for _ in range(100):
            if time.time() - start_time > budget_s - 10:
                break
            # Compute gradient
            grad = np.zeros_like(h_refined)
            eps = 1e-5
            for i in range(n_points):
                h_plus = h_refined.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                corr = np.correlate(h_refined, 1 - h_refined, mode='full')
                c5 = np.max(corr) * dx
                grad[i] = (c5_plus - c5) / eps
            
            # Update
            lr = 0.005
            h_new = h_refined - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h_refined = h_new
            
        # Final check
        corr = np.correlate(h_refined, 1 - h_refined, mode='full')
        c5_fin = np.max(corr) * dx
        if c5_fin < best_c5:
            best_c5 = c5_fin
            best_h = h_refined
            
    # Final validation
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)

def check_feasibility(h_init, C_test, n_points, dx):
    """Check if there exists h with overlap constraint <= C_test"""
    try:
        # Create variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Create overlap constraints for top overlapping lags
        corr = np.correlate(h_init, 1 - h_init, mode='full')
        max_lags = np.argsort(corr)[-n_points//4:]  # Top 25% overlaps
        
        # Add overlap constraints as linear approximations
        for k in max_lags:
            # Linear approximation around current h
            h_current = h_init
            # Select indices for convolution
            h1 = h_current
            h2 = 1 - h_current
            # Compute approximate overlap
            overlap = 0.0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h1[i] * h2[j]
            overlap *= dx
            
            # Convex relaxation constraint
            constraints.append(overlap <= C_test)
            
        # Low degree polynomial relaxation could make it better, but we'll try a simpler method
        objective = cp.Minimize(cp.sum_squares(h_vars - np.array(h_init)))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False, eps_abs=1e-6, eps_rel=1e-6)
        
        if prob.status == cp.OPTIMAL:
            h_result = h_vars.value
            if h_result is not None:
                h_result = np.clip(h_result, 0, 1)
                h_result = h_result * (n_points / 2) / np.sum(h_result)
                
                corr_final = np.correlate(h_result, 1 - h_result, mode='full')
                c5_final = np.max(corr_final) * dx
                
                return (c5_final, h_result)
        return False
    except:
        return False
