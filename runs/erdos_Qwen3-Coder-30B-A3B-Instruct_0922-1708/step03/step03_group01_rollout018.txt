import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to sum to n_points/2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        # Define variables
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points/2]
        
        # Add overlap constraints for each shift
        for k in range(n_points):
            # Compute the overlap constraint for shift k
            overlap_expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_expr += h_var[i] * (1 - h_var[j])
            constraints.append(overlap_expr <= c_mid / dx)
        
        # Objective: minimize C5 (this is actually a feasibility problem)
        # So we just need to test feasibility
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            # If successful, this is feasible
            if h_var.value is not None:
                h_sol = np.array(h_var.value).flatten()
                # Project to [0,1] and re-normalize
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                # Evaluate C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                c5_test = np.max(correlation) * dx
                
                if c5_test < best_c5:
                    best_c5 = c5_test
                    best_h = h_sol.copy()
                
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
    
    # Final check of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
