import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    # Define the objective function
    def objective(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    # Define gradient estimation
    def compute_gradient(h, eps=1e-4):
        grad = np.zeros_like(h)
        current_obj = objective(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            grad[i] = (objective(h_plus) - current_obj) / eps
            
        return grad
    
    # Optimization loop with hybrid approach
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate current solution
        c5 = objective(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Adaptive learning rate based on gradient magnitude
        lr = 0.005 / (1 + np.linalg.norm(grad) * 1e-3)
        
        # Update with adaptive step size
        h_new = h - lr * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perturb to escape local optima
        if iteration % 50 == 0:
            noise = np.random.normal(0, 0.0005, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally use a random good point if improving
        if iteration % 200 == 0 and iteration > 100:
            # Random search
            h_rand = np.random.rand(n_points)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            c5_rand = objective(h_rand)
            if c5_rand < c5:
                h = h_rand.copy()
                
    # Final refinement with gradient descent
    for _ in range(50):
        grad = compute_gradient(best_h)
        lr = 0.001
        best_h = best_h - lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        c5 = objective(best_h)
        if c5 < best_c5:
            best_c5 = c5

    return (best_h.tolist(), best_c5, n_points)
