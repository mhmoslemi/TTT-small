import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    # Binary search on C5
    low_bound = 0.0
    high_bound = 0.38085
    tolerance = 1e-5
    
    while time.time() - start_time < budget_s - 1 and high_bound - low_bound > tolerance:
        iteration += 1
        test_c5 = (low_bound + high_bound) / 2.0
        
        # Formulate feasibility problem with the test C5 value
        h_vars = cp.Variable(n_points)
        
        # Constraints: h in [0,1] and sum = n_points/2
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        
        # Add overlap constraints
        # We'll approximate this by considering a small number of lag values
        for i in range(1, n_points):
            # Approximate the sum of correlations for lag i
            lag_sum = 0
            for j in range(n_points - i):
                lag_sum += h_vars[j] * (1 - h_vars[j + i])
            # Add constraint for this lag
            constraints.append(lag_sum <= test_c5 * dx * (n_points - i))
        
        # Objective: minimize something (not needed, just solve feasibility)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.MOSEK, verbose=False, max_iters=1000)
            
            if prob.status == cp.OPTIMAL:
                h_feasible = h_vars.value
                if h_feasible is not None:
                    h_feasible = np.clip(h_feasible, 0, 1)
                    h_feasible = h_feasible * (n_points / 2) / np.sum(h_feasible)
                    c5_feasible = compute_c5(h_feasible)
                    if c5_feasible < best_c5:
                        best_c5 = c5_feasible
                        best_h = h_feasible.copy()
                    high_bound = test_c5
                else:
                    low_bound = test_c5
            else:
                low_bound = test_c5
        except Exception:
            low_bound = test_c5
    
    # Final optimize with gradient descent if we still have time
    if time.time() - start_time < budget_s - 2:
        # Use more sophisticated optimization routine
        # Just make a few passes of gradient descent to refine the current solution
        
        # Compute current best value
        corr = np.correlate(best_h, 1 - best_h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Try several local optimizations
        for _ in range(20):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Compute gradient numerically
            grad = np.zeros_like(best_h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = best_h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - current_c5) / eps
            
            # Update
            lr = 0.001
            h_new = best_h - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            corr_new = np.correlate(h_new, 1 - h_new, mode='full')
            c5_new = np.max(corr_new) * dx
            
            if c5_new < current_c5:
                best_h = h_new
                current_c5 = c5_new
                
    # Final check
    corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
