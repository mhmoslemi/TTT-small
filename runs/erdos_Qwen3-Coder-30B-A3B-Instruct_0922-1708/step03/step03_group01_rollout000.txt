import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Objective function
    def obj(h_vec):
        # Compute correlations for all shifts
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Gradient function
    def grad(h_vec):
        # Find shift that maximizes overlap
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_shift = np.argmax(correlation)
        # Adjust for the fact correlation is full length
        shift = max_shift - (len(h_vec) - 1)
        
        # Compute gradient at the points contributing to the max overlap
        gradient = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            j = (i - shift) % len(h_vec)
            if 0 <= j < len(h_vec):
                gradient[i] = dx * (1 - 2 * h_vec[i])
        return gradient
    
    start_time = time.time()
    
    # Define bounds and constraint
    bounds = [(0, 1) for _ in range(n_points)]
    constraint = {'type': 'eq', 'fun': lambda x: np.sum(x) - n_points / 2}
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Optimize with SLSQP
        result = minimize(obj, h, method='SLSQP', bounds=bounds, 
                         constraints=constraint, jac=grad, 
                         options={'maxiter': 100, 'ftol': 1e-8})
        
        if result.success:
            h = result.x
            c5 = obj(h)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        else:
            # If optimization fails, slightly perturb the solution
            h = h + np.random.normal(0, 1e-5, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        # Occasionally rescale to ensure constraint satisfaction
        h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
