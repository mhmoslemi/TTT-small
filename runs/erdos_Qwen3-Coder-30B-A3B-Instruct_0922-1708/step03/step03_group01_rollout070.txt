def frank_zadoff_sequence(n):
    """Generate a Frank-Zadoff sequence for initialization."""
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = np.sin(2 * np.pi * i * (i + 1) / (2 * n))
    return (seq + 1) / 2  # Scale to [0,1]
