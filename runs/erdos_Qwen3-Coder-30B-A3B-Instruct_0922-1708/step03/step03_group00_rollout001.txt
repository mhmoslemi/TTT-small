import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_c5 = float('inf')

    # PSO settings
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_new = np.linspace(0, 2, n_points)
                h_init = np.interp(x_new, x_orig, h_init)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        else:
            h_init = np.random.rand(n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.zeros(n_points))
        pbest.append(h_init.copy())
        pbest_vals.append(float('inf'))
    
    gbest = h.copy()
    gbest_val = float('inf')
    
    # PSO optimization loop
    pso_iterations = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    while time.time() - start_time < budget_s - 2 and no_improvement_count < max_no_improvement:
        pso_iterations += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (inertia * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate fitness
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            fit = np.max(correlation) * dx
            
            if fit < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = fit
                
            if fit < gbest_val:
                gbest = particles[i].copy()
                gbest_val = fit
                no_improvement_count = 0

        if pso_iterations % 5 == 0:

            for _ in range(10):
                # Compute the overlap integral
                correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
                c5 = np.max(correlation) * dx
                
                # Compute gradient using finite differences
                grad = np.zeros_like(h_refine)
                eps = 1e-4
                
                for j in range(n_points):

                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - c5) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h_refine - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                h_refine = h_new
                
            # Update global best if better
            correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
            c5_refine = np.max(correlation) * dx
            
            if c5_refine < gbest_val:
                gbest = h_refine.copy()
                gbest_val = c5_refine
                
    if gbest_val < best_c5:
        best_h = gbest.copy()
        best_c5 = gbest_val
        
    return (best_h.tolist(), best_c5, n_points)
