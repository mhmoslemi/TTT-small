import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a modified Barker-like initialization
        h = np.zeros(n_points)
        # Use a pattern that places high values in a way to reduce overlap
        # We'll use a pattern similar to a Barker code but in continuous domain
        half_points = n_points // 2
        h[:half_points] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Optimization function
    def objective(h_vec):
        # Rescale to meet constraint
        h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
        h_rescaled = np.clip(h_rescaled, 0, 1)
        
        # Compute overlap
        correlation = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Gradient function (numerical approximation)
    def gradient(h_vec):
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_obj = objective(h_vec)
        
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_obj) / eps
            
        return grad
    
    # Run optimization with multiple restarts
    max_restarts = 10
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Start with current best
        x0 = best_h.copy()
        
        # Bounds for optimization
        bounds = [(0, 1) for _ in range(n_points)]
        
        # Optimize using L-BFGS-B with gradient
        result = minimize(objective, x0, method='L-BFGS-B', bounds=bounds, 
                         jac=gradient, options={'maxiter': 100, 'ftol': 1e-9})
        
        if result.success:
            h_opt = result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Compute overlap
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_opt.copy()
        else:
            # If optimization fails, just try a small perturbation
            h_pert = best_h + np.random.normal(0, 0.01, n_points)
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            
            correlation = np.correlate(h_pert, 1 - h_pert, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_pert.copy()
                
        restart_count += 1
        
        # Occasionally try some random perturbations
        if restart_count % 3 == 0:
            # Random perturbation
            h_rand = np.random.rand(n_points)
            h_rand = np.clip(h_rand, 0, 1)
            h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
            
            correlation = np.correlate(h_rand, 1 - h_rand, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h_rand.copy()
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
