import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Normalize to sum to n_points/2
    h_init = h_init * (n_points / 2) / np.sum(h_init)
    
    # Create initial particles
    particles.append(h_init.copy())
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    inertia = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate all particles
        for i, h in enumerate(particles):
            # Ensure feasibility
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute correlation
            corr = np.correlate(h, 1 - h, mode='full')
            score = np.max(corr) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_bests[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            cognitive = c1 * r1 * (personal_bests[i] - h)
            social = c2 * r2 * (global_best - h)
            v = inertia * v + cognitive + social
            
            # Clamp velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h = h + v
            
            # Store updated values
            particles[i] = h
            velocities[i] = v
            
        step += 1
        
        # Occasionally reset some particles for diversity
        if step % 100 == 0 and step > 0:
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h * (n_points / 2) / np.sum(h)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final evaluation
    corr = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (global_best.tolist(), final_c5, n_points)
