import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Initialize swarm
    n_particles = 40
    particles = np.zeros((n_particles, n_points))
    velocities = np.zeros((n_particles, n_points))
    
    # Initialize particles around the initial solution
    for i in range(n_particles):
        if i == 0:
            particles[i] = h_init.copy()
        else:
            # Add small random perturbations
            pert = np.random.normal(0, 0.05, n_points)
            particles[i] = h_init + pert
        # Clip and rescale each particle to meet constraints
        particles[i] = np.clip(particles[i], 0, 1)
        particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        # Initialize velocities
        velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Initialize personal bests
    personal_best = particles.copy()
    personal_best_scores = np.full(n_particles, np.inf)
    
    # Initialize global best
    global_best_idx = 0
    global_best_score = np.inf
    
    start_time = time.time()
    step = 0
    
    # PSO parameters
    omega = 1.0
    c1 = 2.0
    c2 = 2.0
    velocity_max = 0.2
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        # Update inertia weight if needed
        if step % 500 == 0:
            omega = max(0.8, omega - 0.01)
            
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Compute C5 using correlation
            corr = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best_idx = i
        
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = omega * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (personal_best[global_best_idx] - h)
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            
            # Update position
            h = h + v
            
            # Ensure bounds and scaling
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            particles[i] = h
            velocities[i] = v
            
        # Occasionally reset some particles to maintain diversity
        if step % 200 == 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:  # 10% chance
                    # Resample with small random perturbation
                    particles[i] = np.random.uniform(0.2, 0.8, n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
    
    # Final best solution check
    final_h = personal_best[global_best_idx].copy()
    corr = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (final_h.tolist(), final_c5, n_points)
