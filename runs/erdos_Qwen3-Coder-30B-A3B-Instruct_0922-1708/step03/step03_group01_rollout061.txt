import numpy as np
import time
import hashlib

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Configuration
    max_tabu_size = 50
    perturbation_count = max(1, int(0.2 * n_points))
    initial_step = 0.1
    step_increase_factor = 1.2
    stagnation_threshold = 100
    aspiration_prob = 0.1
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # Tabu list
    tabu_list = set()
    
    # Core variables
    current_h = h.copy()
    current_c5 = best_c5
    step_size = initial_step
    last_improvement = 0
    iteration = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Perturb current solution
        candidate_h = current_h.copy()
        indices = np.random.choice(n_points, size=perturbation_count, replace=False)
        for i in indices:
            delta = np.random.uniform(-step_size, step_size)
            candidate_h[i] += delta
        
        # Clip and normalize
        candidate_h = np.clip(candidate_h, 0, 1)
        scale_factor = (n_points / 2) / np.sum(candidate_h)
        candidate_h = candidate_h * scale_factor
        
        # Compute candidate C5
        candidate_c5 = evaluate_erdos_solution(candidate_h)
        
        # Hash the candidate to check tabu status
        hashable_candidate = tuple(np.round(candidate_h, decimals=6))
        tabu_hash = hashlib.md5(str(hashable_candidate).encode()).hexdigest()
        
        # Check if candidate is tabu
        is_tabu = tabu_hash in tabu_list
        
        # Check improvement
        if candidate_c5 < best_c5:
            # Accept better solution
            best_c5 = candidate_c5
            best_h = candidate_h.copy()
            last_improvement = iteration
            tabu_list.add(hashlib.md5(str(tuple(np.round(current_h, decimals=6))).encode()).hexdigest())
            current_h = candidate_h.copy()
            current_c5 = candidate_c5
        else:
            # Allow aspiration with some probability
            if not is_tabu and np.random.random() < aspiration_prob:
                tabu_list.add(hashlib.md5(str(tuple(np.round(current_h, decimals=6))).encode()).hexdigest())
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
            elif not is_tabu:
                # Regular move to the tabu solution
                tabu_list.add(hashlib.md5(str(tuple(np.round(current_h, decimals=6))).encode()).hexdigest())
                current_h = candidate_h.copy()
                current_c5 = candidate_c5
            else:
                # Choose the best non-tabu neighbor
                pass
            
        # Manage tabu list size and step size
        if len(tabu_list) > max_tabu_size:
            tabu_list.pop()
            
        if iteration - last_improvement > stagnation_threshold:
            step_size *= step_increase_factor
            last_improvement = iteration
            
    return (best_h.tolist(), best_c5, n_points)

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap integral."""
    n_points = len(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5
