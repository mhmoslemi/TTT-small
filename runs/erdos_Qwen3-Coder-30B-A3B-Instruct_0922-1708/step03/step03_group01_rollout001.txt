import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    def objective_and_gradient(h):
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient analytically
        k_max = np.argmax(correlation)
        grad = np.zeros_like(h)
        
        # For each position, compute derivative of max overlap term
        # The maximum overlap occurs at shift k_max
        # Gradients for h[i] depend on terms h[i] * (1 - h[i+k_max]) and h[i-k_max] * (1 - h[i])
        for i in range(n_points):
            # Contribution from h[i] * (1 - h[i+k_max])
            if i + k_max < n_points:
                grad[i] += dx * (-h[i + k_max])
            # Contribution from h[i-k_max] * (1 - h[i])
            if i - k_max >= 0:
                grad[i] += dx * (-h[i - k_max])
                
        return c5, grad
    
    # Define bounds and constraints
    bounds = [(0, 1) for _ in range(n_points)]
    
    def constraint_func(h):
        return np.sum(h) - n_points / 2
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Optimization loop
    start_time = time.time()
    
    # Use SLSQP with analytical gradients
    while time.time() - start_time < budget_s - 1:
        # Compute current value and gradient
        c5, grad = objective_and_gradient(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Perform one optimization step using L-BFGS-B or trust-constr
        try:
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h,
                method='L-BFGS-B',
                jac=lambda x: objective_and_gradient(x)[1],
                bounds=bounds,
                constraints=constraints,
                options={'maxiter': 10, 'ftol': 1e-8, 'gtol': 1e-8}
            )
            
            if result.success:
                h = result.x
                # Ensure the constraint is still satisfied
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback: use simple gradient step
                step_size = 1e-3
                h_new = h - step_size * grad
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new
                
        except:
            # Fallback: simple gradient descent step
            step_size = 1e-3
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
