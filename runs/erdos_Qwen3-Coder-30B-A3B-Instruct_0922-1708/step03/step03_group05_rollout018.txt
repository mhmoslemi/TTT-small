import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points/2]
        
        # Compute overlaps for all lags
        overlaps = []
        for k in range(n_points):
            overlap_expr = 0
            for i in range(n_points - k):
                overlap_expr += h_relaxed[i] * (1 - h_relaxed[i + k])
            overlaps.append(overlap_expr)
        
        objective = cp.Minimize(cp.max(cp.stack(overlaps)))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_relaxed_val = h_relaxed.value
            else:
                # Fallback to gradient descent
                h_relaxed_val = h
        except:
            h_relaxed_val = h
        
        # Rounding
        h_rounded = (h_relaxed_val >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust sum to n_points/2
        if current_sum > n_points / 2:
            # Need to flip some 1's to 0's
            diff = int(current_sum - n_points / 2)
            # Flip indices with smallest relaxed values
            flip_indices = np.argsort(h_relaxed_val)[:diff]
            h_rounded[flip_indices] = 0
        elif current_sum < n_points / 2:
            # Need to flip some 0's to 1's
            diff = int(n_points / 2 - current_sum)
            # Flip indices with largest relaxed values
            flip_indices = np.argsort(h_relaxed_val)[-diff:]
            h_rounded[flip_indices] = 1
        
        # Final validation check
        h_rounded = np.clip(h_rounded, 0, 1)
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        
        # Compute final C5
        correlation = np.correlate(h_rounded, 1 - h_rounded, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
        
        # Slight perturbation to continue exploration
        h = h_rounded + np.random.normal(0, 0.01, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
