import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def objective_and_gradient(h):
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        max_overlap = np.max(correlation)
        c5 = max_overlap * dx
        
        # Gradient computation
        # Find k that gives max overlap
        k_max = np.argmax(correlation)
        # k ranges from 0 to 2*n_points-2, we want k in [0, n_points-1]
        if k_max >= n_points:
            k_max_shift = k_max - n_points + 1
        else:
            k_max_shift = k_max
            
        # Gradient w.r.t. each h[i]
        grad = np.zeros_like(h)
        for i in range(n_points):
            # Compute derivative of overlap with respect to h[i]
            if i + k_max_shift < n_points:
                grad[i] += dx * (- (1 - h[i + k_max_shift]))
            if i - k_max_shift >= 0:
                grad[i] += dx * (- h[i - k_max_shift])
                
        return c5, grad
    
    # Constraint: sum(h) = n_points/2
    def constraint_sum(h):
        return np.sum(h) - n_points / 2
    
    # Bounds for each h[i]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint dict
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Optimization parameters
    max_iter = 1000
    tol = 1e-8
    
    # Optimization loop
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute objective and gradient
        c5, grad = objective_and_gradient(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Make a scipy minimize step
        # Use L-BFGS-B solver which can handle bounds well
        result = minimize(
            lambda x: objective_and_gradient(x)[0],
            h,
            method='L-BFGS-B',
            bounds=bounds,
            constraints=cons,
            jac=lambda x: objective_and_gradient(x)[1],
            options={'maxiter': 1, 'gtol': tol}
        )
        
        # Update h with new values
        if result.success or result.message.startswith('CONVERGENCE'):
            h = result.x
        else:
            # If optimization failed, fallback to simple gradient descent
            step_size = 0.001
            grad = objective_and_gradient(h)[1]
            h_new = h - step_size * grad
            
            # Project to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet sum constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Occasionally re-evaluate global best
        if iter_num % 10 == 0:
            c5_test, _ = objective_and_gradient(h)
            if c5_test < best_c5:
                best_c5 = c5_test
                best_h = h.copy()
    
    # Final evaluation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
