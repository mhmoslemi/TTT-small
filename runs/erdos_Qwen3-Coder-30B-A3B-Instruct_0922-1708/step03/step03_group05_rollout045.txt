import numpy as np
import time
import cvxpy as cp

def evaluate_c5(h, dx):
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_h(h, n_points):
    # Ensure h is in [0,1] and sum = n_points/2
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    T0 = 0.1
    alpha = 0.99
    
    # Initialize
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    h = project_h(h, n_points)
    best_h = h.copy()
    best_c5 = evaluate_c5(h, dx)
    
    # Optimization loop
    T = T0
    for iteration in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Generate neighbor
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
        for i in indices:
            perturbation = np.random.uniform(-0.1, 0.1)
            h_new[i] += perturbation
        
        h_new = project_h(h_new, n_points)
        c5_new = evaluate_c5(h_new, dx)
        
        # Accept or reject
        if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
            h = h_new
            if c5_new < best_c5:
                best_h = h_new.copy()
                best_c5 = c5_new
        
        T *= alpha
    
    # Refinement: 2x resolution
    if time.time() - start_time < budget_s - 1:
        n_points_refined = 200
        dx_refined = 2.0 / n_points_refined
        
        # Interpolate
        x_orig = np.linspace(0, 2, n_points)
        x_refined = np.linspace(0, 2, n_points_refined)
        h_refined = np.interp(x_refined, x_orig, best_h)
        h_refined = h_refined * (n_points_refined / 2) / np.sum(h_refined)
        
        # Optimize refined version
        h = h_refined
        best_h = h.copy()
        best_c5 = evaluate_c5(h, dx_refined)
        
        T = T0
        for iteration in range(max_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            h_new = h.copy()
            indices = np.random.choice(n_points_refined, size=int(0.05 * n_points_refined), replace=False)
            for i in indices:
                perturbation = np.random.uniform(-0.1, 0.1)
                h_new[i] += perturbation
            
            h_new = project_h(h_new, n_points_refined)
            c5_new = evaluate_c5(h_new, dx_refined)
            
            if c5_new < best_c5 or np.random.rand() < np.exp((best_c5 - c5_new) / T):
                h = h_new
                if c5_new < best_c5:
                    best_h = h_new.copy()
                    best_c5 = c5_new
            
            T *= alpha
    
    final_c5 = evaluate_c5(best_h, dx_refined if 'dx_refined' in locals() else dx)
    return (best_h.tolist(), final_c5, len(best_h))
