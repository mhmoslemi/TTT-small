import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Binary search for C5
    c_low = 0.0
    c_high = 0.38085
    tolerance = 1e-5
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
        
    def check_feasibility(h, c):
        # For constraint: sum(h[i] * (1 - h[i+k])) <= c * dx for all k
        # This is a bilinear constraint, we'll do a linearization approach
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx <= c
        
    while time.time() - start_time < budget_s - 1 and c_high - c_low > tolerance:
        c_test = (c_low + c_high) / 2.0
        iteration += 1
        
        # Try to construct a valid h with C5 <= c_test
        # Perform gradient-based optimization for improving the current h
        h_current = h.copy()
        
        # Simple optimization approach: gradient descent step
        n_iter = 10
        for _ in range(n_iter):
            # Compute gradient estimation
            grad = np.zeros_like(h_current)
            eps = 1e-4
            
            for i in range(n_points):
                h_plus = h_current.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                c_plus = compute_c5(h_plus)
                grad[i] = (c_plus - compute_c5(h_current)) / eps
            
            # Update with gradient
            lr = 0.001
            h_new = h_current - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_current = h_new
            
        # Check feasibility of new solution
        c_val = compute_c5(h_current)
        if c_val <= c_test:
            c_high = c_test
            if c_val < best_c5:
                best_h = h_current.copy()
                best_c5 = c_val
            h = h_current.copy()
        else:
            c_low = c_test
            
    # Final refinement via CVX optimization
    try:
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]
        objective = cp.Minimize(cp.max(cp.conv(h_vars, cp.reshape(1 - h_vars, (len(h_vars), 1)))))
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.MOSEK, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_cvx = h_vars.value
            if h_cvx is not None:
                h_cvx = np.clip(h_cvx, 0, 1)
                h_cvx = h_cvx * (n_points / 2) / np.sum(h_cvx)
                c_cvx = compute_c5(h_cvx)
                
                if c_cvx < best_c5:
                    best_h = h_cvx
                    best_c5 = c_cvx
    except:
        pass
    
    # Final validation of best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
