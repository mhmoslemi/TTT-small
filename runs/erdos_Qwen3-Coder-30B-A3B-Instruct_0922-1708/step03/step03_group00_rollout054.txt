import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO Parameters
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Start with initial solution and random particles
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_vals.append(float('inf'))
    
    for _ in range(n_particles - 1):
        # Generate a random feasible solution
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand.copy())
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(h_rand.copy())
        pbest_vals.append(float('inf'))
    
    gbest_idx = 0
    gbest_val = float('inf')
    
    # PSO Iteration Loop
    pso_iterations = 0
    no_improvement_count = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 50:
        pso_iterations += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            # Ensure the particle is feasible
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Compute correlation
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5_val = np.max(correlation) * dx
            
            # Update personal best
            if c5_val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5_val
                
            # Update global best
            if c5_val < gbest_val:
                gbest_val = c5_val
                gbest_idx = i
                no_improvement_count = 0
                
        # Update best if we've found a better one through PSO
        if gbest_val < best_c5:
            best_c5 = gbest_val
            best_h = particles[gbest_idx].copy()
            
        # Update velocities and positions
        for i in range(n_particles):
            r1 = np.random.rand()
            r2 = np.random.rand()
            
            # Velocity update
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Clip velocities
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            
            # Position update
            particles[i] += velocities[i]
            
        # Ensure all particles are feasible
        for i in range(n_particles):
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        no_improvement_count += 1
        
        # Occasionally add some diversity
        if pso_iterations % 20 == 0:
            for i in range(1, n_particles):
                if np.random.rand() < 0.3:
                    h_rand = np.random.rand(n_points)
                    h_rand = np.clip(h_rand, 0, 1)
                    h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
                    particles[i] = h_rand.copy()
                    velocities[i] = np.random.rand(n_points) * 0.01
    
    # Local refinement with gradient descent
    h_refine = best_h.copy()
    refine_start = time.time()
    
    # Gradient Descent
    while time.time() - refine_start < 50 and time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimation
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h_refine.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refine - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h_refine = h_new
        
        # Update best if improved
        correlation_refine = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5_refine = np.max(correlation_refine) * dx
        
        if c5_refine < best_c5:
            best_c5 = c5_refine
            best_h = h_refine.copy()
    
    return (best_h.tolist(), best_c5, n_points)
