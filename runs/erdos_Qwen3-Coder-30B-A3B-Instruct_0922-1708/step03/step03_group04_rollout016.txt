import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    def compute_gradient_for_max_shift(h_array):
        # Compute full correlation
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        # Find the lag k that maximizes correlation
        k_max = np.argmax(corr)
        # For a full correlation of length 2*n_points - 1, 
        # the shift k is: k = k_max - (n_points - 1)
        k = k_max - (n_points - 1)
        
        # Compute exact gradient for that shift
        grad = np.zeros_like(h_array)
        for i in range(n_points):
            # In the overlap integral: h(i) * (1 - h(i+k))
            # The derivative w.r.t h(i) is (1 - h(i+k))
            # The derivative w.r.t h(i+k) is -h(i)
            # But we're computing the gradient with respect to the variables h(i)
            # so we dol not directly differentiate h(i+k) w.r.t. h(i) 
            # we should consider how modifying h(i) affects the correlation sum
            # For any given shift k, the contribution of h(i) to the overlap
            # is h(i) * (1 - h(i + k)) 
            # w.r.t. h(i), it's (1 - h(i + k))
            # but we're computing the effect on correlation & eventually gradient for h[i]
            # The derivative of h[i] w.r.t h[i] is 1 and it multiplies (1 - h[i+k]) if i belongs to 
            # the nonzero range of the convolution contribution
            # update in correct loop manner
            # Let's think carefully:
            # We have a convolution h * (1 - h) where we're varying h
            # For a specific value k, the correlation is:
            # sum over i of h[i] * (1 - h[(i+k) mod n])
            # so the gradient of this with respect to h[j] is:
            # if (j + k) mod n == i then that term contributes -h[i],  
            # if i == j then it contributes (1 - h[(j+k) mod n])
            # which is a sum over i of delta[i, j] * (1 - h[(i+k) mod n]) - delta[i, j-k] * h[j]
            # iff i = j, we get (1 - h[(j+k) mod n])  
            # iff i = j-k (mod n), we get -h[j]
            # so for fixed j:
            # grad[j] = (1 - h[(j+k) mod n]) - h[(j-k) mod n]
            # This gives the true gradient for THAT specific correlation term
            
            i_pos = (j + k) % n_points
            i_neg = (j - k) % n_points
            # gradient of [sum h[i] * (1 - h[(i+k) mod n])] 
            # w.r.t h[j] is:
            # 1 - h[(j+k) mod n] if j is part of h[(i+k) mod n] = h[j], which is true only if i = j-k
            # -h[j] if j is part of h[(i-k) mod n] = h[j], which is true only if i+k = j, i = j-k
            # Biased onto the term h[(j-k) mod n] and h[(j+k) mod n]
            # Actually better to think via what happens to the whole convolution in position j:
            # In corr = np.correlate(h, 1-h, mode='full'), 
            # at position k_max, they maximize correlation.
            # So we compute the derivative for that mode.
            
            # Let us instead derive manually for the specific given shift:
            # f = sum_{i=0}^{n-1} h_i * (1 - h_{(i+k) mode n})
            # df/dh_j = 1_{j-k mod n = i} * (1- h_{i+k}) - 1_{i+k = j} * h_j  
            # = h_{j+k} - h_j where the placements are with modular arithmetic
            # for k = k_max, this is:
            # grad_j = (1 - h_{(j+k) mod n}) - h_{(j-k) mod n} 
            
            grad[j] = (1 - h[(j+k) % n_points]) - h[(j-k) % n_points]
        
        # Slightly recompute with optimized steps
        grad = np.zeros_like(h_array)
        k_pos = k if k >= 0 else n_points + k
        k_neg = (-k) if k <= 0 else n_points - k
        # Index of correlation maximum is k_max = k + (n-1) 
        # So k = k_max - (n-1)
        for i in range(n_points):
            grad[i] = (1 - h[(i + k) % n_points]) - h[(i - k) % n_points]
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient via exact formula for the worst-case shift
        grad = np.zeros_like(h)
        # Compute all correlations
        corr = np.correlate(h, 1 - h, mode='full')
        # Find max shift
        k_max = np.argmax(corr)
        k = k_max - (n_points - 1)
        
        # Compute exact gradient for that shift
        for i in range(n_points):
            grad[i] = (1 - h[(i + k) % n_points]) - h[(i - k) % n_points]
            
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
