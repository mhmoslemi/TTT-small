import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    # Start with base solution and perturbations
    for i in range(n_particles):
        if i == 0:
            h = h_base.copy()
        else:
            # Add small random perturbations
            h = h_base + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    v_max = 0.2
    
    # Track personal bests
    personal_bests = [p.copy() for p in particles]
    personal_best_values = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    # PSO loop
    start_time = time.time()
    step = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s - 1:
        # Update each particle
        for i in range(n_particles):
            # Update particle position
            velocities[i] = w * velocities[i] + \
                           c1 * np.random.rand() * (personal_bests[i] - particles[i]) + \
                           c2 * np.random.rand() * (global_best - particles[i])
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            particles[i] = particles[i] + velocities[i]
            
            # Ensure bounds
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to correct sum
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate current solution
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            current_c5 = np.max(correlation) * dx
            
            # Update personal best
            if current_c5 < personal_best_values[i]:
                personal_best_values[i] = current_c5
                personal_bests[i] = particles[i].copy()
                
                # Update global best
                if current_c5 < global_best_value:
                    global_best_value = current_c5
                    global_best = particles[i].copy()
                    stagnation_count = 0
                else:
                    stagnation_count += 1
        
        # Occasionally reset some particles to maintain diversity
        if step % 50 == 0 and step > 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:  # Reset 10% of particles
                    if 'initial_h_values' in globals():
                        h_init = np.array(initial_h_values)
                        if len(h_init) != n_points:
                            x_orig = np.linspace(0, 2, len(h_init))
                            x_target = np.linspace(0, 2, n_points)
                            h_init = np.interp(x_target, x_orig, h_init)
                        particles[i] = h_init + np.random.normal(0, 0.05, n_points)
                    else:
                        particles[i] = np.random.uniform(0.3, 0.7, n_points)
                    particles[i] = np.clip(particles[i], 0, 1)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Adaptive inertia weight
        if step > 0 and step % 100 == 0:
            w = max(0.8, w - 0.0001)
        
        step += 1
        
        # Break early if stagnation occurs
        if stagnation_count > max_stagnation:
            break
    
    # Final check
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
