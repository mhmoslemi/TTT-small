import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize to satisfy constraint sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturbation_magnitude = 0.1
    
    # Nelder-Mead parameters
    simplex_size = 5
    
    # Simulated Annealing phase
    sa_steps = 2000
    for step in range(sa_steps):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb a random subset of points
        perturbed_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new = h.copy()
        for i in perturbed_indices:
            h_new[i] += np.random.normal(0, perturbation_magnitude)
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing
        if c5_new < best_c5 or np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        # Cool down
        T *= alpha
    
    # Nelder-Mead refinement phase
    def compute_c5(h_vals):
        h = np.array(h_vals)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return -np.max(correlation) * dx  # Negative because we minimize
    
    # Create initial simplex
    def create_simplex(center, size=0.1, n_points=5):
        simplex = [center.copy()]
        for _ in range(n_points - 1):
            pert = np.random.normal(0, size, len(center))
            new_point = center + pert
            # Ensure it is valid in [0,1]^n_points
            new_point = np.clip(new_point, 0, 1)
            new_point = new_point * (n_points / 2) / np.sum(new_point)
            simplex.append(new_point)
        return simplex
    
    # Refinement with Nelder-Mead-like approach
    current_simplex = create_simplex(h, size=0.05, n_points=simplex_size)
    current_values = [compute_c5(point) for point in current_simplex]
    
    # Simplex Nelder-Mead step
    iters = 0
    max_iters = 1000
    
    while iters < max_iters and time.time() - start_time < budget_s - 1:
        # Sort simplex points by function value
        sorted_indices = np.argsort(current_values)
        best_idx = sorted_indices[0]
        worst_idx = sorted_indices[-1]
        second_worst_idx = sorted_indices[-2]
        
        # Reflect the worst point
        centroid = np.mean([current_simplex[i] for i in sorted_indices[:-1]], axis=0)
        reflected = 2 * centroid - current_simplex[worst_idx]
        reflected = np.clip(reflected, 0, 1)
        reflected = reflected * (n_points / 2) / np.sum(reflected)
        reflected_value = compute_c5(reflected)
        
        # Replace worst point if it's better than the worst
        if reflected_value < current_values[worst_idx]:
            current_simplex[worst_idx] = reflected
            current_values[worst_idx] = reflected_value
        else:
            # Otherwise contract towards best point
            contracted = 0.5 * (current_simplex[best_idx] + current_simplex[worst_idx])
            contracted = np.clip(contracted, 0, 1)
            contracted = contracted * (n_points / 2) / np.sum(contracted)
            contracted_value = compute_c5(contracted)
            
            if contracted_value < current_values[worst_idx]:
                current_simplex[worst_idx] = contracted
                current_values[worst_idx] = contracted_value
            else:
                # Shrink all points toward best point
                for idx in range(len(current_simplex)):
                    if idx != best_idx:
                        current_simplex[idx] = 0.5 * (current_simplex[best_idx] + current_simplex[idx])
                        current_simplex[idx] = np.clip(current_simplex[idx], 0, 1)
                        current_simplex[idx] = current_simplex[idx] * (n_points / 2) / np.sum(current_simplex[idx])
                        current_values[idx] = compute_c5(current_simplex[idx])
        
        if current_values[best_idx] < -best_c5:  # Note: we use negative values
            best_c5 = -current_values[best_idx]
            best_h = current_simplex[best_idx].copy()

        iters += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
