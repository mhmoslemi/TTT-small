import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Genetic Algorithm parameters
    pop_size = 50
    elite_size = int(0.4 * pop_size)
    crossover_rate = 0.5
    mutation_rate = 0.2
    mutation_strength = 0.1
    
    # Initialize population with diverse individuals
    population = []
    for i in range(pop_size):
        if i == 0 and 'initial_h_values' in globals():
            # Use initial solution first
            individual = h.copy()
        else:
            # Perturb initial solution slightly to create diversity
            individual = h + np.random.normal(0, 0.05, n_points)
            individual = np.clip(individual, 0, 1)
        # rescale
        individual = individual * (n_points / 2) / np.sum(individual)
        population.append(individual)
    
    # Track best individual
    best_individual = h.copy()
    best_fitness = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Evaluate fitness of entire population
        fitnesses = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitnesses.append(c5)
            if c5 < best_fitness:
                best_fitness = c5
                best_individual = individual.copy()
        
        # Update global best
        if best_fitness < best_c5:
            best_c5 = best_fitness
            best_h = best_individual.copy()
        
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        sorted_fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Elitism
        new_population = sorted_population[:elite_size]
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parent1_idx = sorted_indices[np.random.randint(0, elite_size)]
            parent2_idx = sorted_indices[np.random.randint(0, elite_size)]
            
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover
            if np.random.random() < crossover_rate:
                # Uniform crossover
                mask = np.random.random(n_points) > 0.5
                offspring1 = np.where(mask, parent1, parent2)
                offspring2 = np.where(mask, parent2, parent1)
                
                # Mutate
                if np.random.random() < mutation_rate:
                    indices_to_mutate = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                    for idx in indices_to_mutate:
                        offspring1[idx] += np.random.normal(0, mutation_strength)
                    offspring1 = np.clip(offspring1, 0, 1)
                    offspring1 = offspring1 * (n_points / 2) / np.sum(offspring1)
                
                if np.random.random() < mutation_rate:
                    indices_to_mutate = np.random.choice(n_points, size=int(0.05 * n_points), replace=False)
                    for idx in indices_to_mutate:
                        offspring2[idx] += np.random.normal(0, mutation_strength)
                    offspring2 = np.clip(offspring2, 0, 1)
                    offspring2 = offspring2 * (n_points / 2) / np.sum(offspring2)
                
                new_population.append(offspring1)
                if len(new_population) < pop_size:
                    new_population.append(offspring2)
            else:
                # Clone best parent
                new_population.append(parent1)
                if len(new_population) < pop_size:
                    new_population.append(parent2)
        
        population = new_population
    
    return (best_h.tolist(), best_c5, n_points)
