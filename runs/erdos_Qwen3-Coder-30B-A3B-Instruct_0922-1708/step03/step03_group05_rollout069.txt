import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute C5 using the existing method
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Convex relaxation approach
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points/2]
        
        # Compute all overlaps for the relaxed optimization
        overlaps = []
        for i in range(n_points):
            # Compute the overlap at lag i
            lagged = h_relaxed[i:] if i < n_points else np.array([])
            if len(lagged) > 0:
                overlap = cp.sum(cp.multiply(h_relaxed[:len(lagged)], 1 - lagged))
                overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(overlaps))
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
            if prob.status == cp.OPTIMAL:
                h_new = h_relaxed.value
                # Round to create a feasible integer solution
                h_rounded = (h_new >= 0.5).astype(float)
                current_sum = np.sum(h_rounded)
                
                if current_sum > n_points / 2:
                    # We have too many 1s, flip some to 0s
                    mask = h_new < 0.5
                    flip_indices = np.where(mask)[0]
                    to_flip = int(current_sum - n_points / 2)
                    if len(flip_indices) >= to_flip:
                        flip_indices = np.argpartition(h_new[flip_indices], to_flip)[:to_flip]
                        flip_indices = flip_indices[flip_indices >= 0]
                        h_rounded[flip_indices] = 0
                elif current_sum < n_points / 2:
                    # We have too many 0s, flip some to 1s
                    mask = h_new >= 0.5
                    flip_indices = np.where(~mask)[0]
                    to_flip = int(n_points / 2 - current_sum)
                    if len(flip_indices) >= to_flip:
                        flip_indices = np.argpartition(h_new[flip_indices], -to_flip)[-to_flip:]
                        flip_indices = flip_indices[flip_indices >= 0]
                        h_rounded[flip_indices] = 1
                
                # Make sure the sum is exactly right
                current_sum = np.sum(h_rounded)
                diff = current_sum - n_points / 2
                if diff > 0:
                    # Remove some 1s
                    ones_indices = np.where(h_rounded > 0.5)[0]
                    to_remove = int(np.round(diff))
                    if to_remove > 0:
                        final_indices = np.argsort(h_new[ones_indices])[:to_remove]
                        h_rounded[ones_indices[final_indices]] = 0
                elif diff < 0:
                    # Add some 1s
                    zeros_indices = np.where(h_rounded < 0.5)[0]
                    to_add = int(np.round(-diff))
                    if to_add > 0:
                        final_indices = np.argsort(h_new[zeros_indices])[-to_add:]
                        h_rounded[zeros_indices[final_indices]] = 1
                
                h = np.clip(h_rounded, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback approach if optimization fails: small perturbation
                lr = 0.0001
                noise = np.random.normal(0, 0.005, n_points)
                h_new = h + noise
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                h = h_new 
        except:
            # Fallback if convex solve fails
            lr = 0.0001
            noise = np.random.normal(0, 0.005, n_points)
            h_new = h + noise
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
