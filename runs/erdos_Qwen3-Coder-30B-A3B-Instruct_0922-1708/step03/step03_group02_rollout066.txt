import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    # Set hyperparameters for the optimization
    T = 0.1    # Initial temperature
    alpha = 0.995  # Cooling rate
    sigma = 0.1    # Perturbation scale
    max_anneal_steps = 2000
    max_neldermead_steps = 1000
    
    # Track best solution
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing loop
    start_time = time.time()
    steps = 0
    
    # Initialization for Nelder-Mead
    simplex_size = min(5, n_points)
    
    # Annealing phase
    while steps < max_anneal_steps and time.time() - start_time < budget_s - 10:
        # Make a random perturbation to h
        h_new = h.copy()
        indices = np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False)
        for i in indices:
            perturbation = np.random.normal(0, sigma)
            h_new[i] += perturbation
            
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            h = h_new.copy()
        elif np.random.random() < math.exp(-(c5_new - np.max(np.correlate(h, 1 - h, mode='full')) * dx) / T):
            h = h_new.copy()
            
        # Cooling
        T *= alpha
        steps += 1
    
    # Nelder-Mead refinement
    # Create initial simplex
    simplex = [best_h.copy()]
    for _ in range(simplex_size - 1):
        h_pert = best_h.copy()
        for i in np.random.choice(n_points, size=max(1, int(0.1 * n_points)), replace=False):
            h_pert[i] += np.random.normal(0, 0.01)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)
    
    # Simplex optimization loop
    nm_steps = 0
    step_limit = max_neldermead_steps
    
    while nm_steps < step_limit and time.time() - start_time < budget_s - 2:
        # Evaluate all points in simplex
        values = []
        for point in simplex:
            corr = np.correlate(point, 1 - point, mode='full')
            c5_val = np.max(corr) * dx
            values.append(c5_val)
        
        # Sort simplex by function value
        sorted_indices = np.argsort(values)
        simplex = [simplex[i] for i in sorted_indices]
        values = [values[i] for i in sorted_indices]
        
        # Calculate centroid
        centroid = np.mean(simplex[:-1], axis=0)
        
        # Reflection
        r_point = centroid + 2 * (centroid - simplex[-1])
        r_point = np.clip(r_point, 0, 1)
        r_point = r_point * (n_points / 2) / np.sum(r_point)
        
        r_corr = np.correlate(r_point, 1 - r_point, mode='full')
        r_val = np.max(r_corr) * dx
        
        # Check if it's better than the best point
        if values[0] <= r_val < values[-2]:
            simplex[-1] = r_point
        elif r_val < values[0]:
            # Expansion
            e_point = centroid + 2 * (r_point - centroid)
            e_point = np.clip(e_point, 0, 1)
            e_point = e_point * (n_points / 2) / np.sum(e_point)
            
            e_corr = np.correlate(e_point, 1 - e_point, mode='full')
            e_val = np.max(e_corr) * dx
            
            if e_val < r_val:
                simplex[-1] = e_point
            else:
                simplex[-1] = r_point
        else:
            # Contraction
            if r_val < values[-1]:
                c_point = centroid + 0.5 * (r_point - centroid)
            else:
                c_point = centroid + 0.5 * (simplex[-1] - centroid)
                
            c_point = np.clip(c_point, 0, 1)
            c_point = c_point * (n_points / 2) / np.sum(c_point)
            
            c_corr = np.correlate(c_point, 1 - c_point, mode='full')
            c_val = np.max(c_corr) * dx
            
            if c_val < values[-1]:
                simplex[-1] = c_point
            else:
                # Shrink simplex
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        
        nm_steps += 1
    
    # Final computation of best solution
    best_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(best_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
