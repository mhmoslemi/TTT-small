import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a Barker-like initialization
        h = np.zeros(n_points)
        # Use a simple pattern: alternating high/low regions
        for i in range(n_points):
            if (i // 10) % 2 == 0:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Scale to meet the integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Objective function for optimization
    def objective(h_vec):
        # Rescale h to satisfy the constraint
        h_rescaled = h_vec * (n_points / 2) / np.sum(h_vec)
        # Compute overlap with all lags
        correlation = np.correlate(h_rescaled, 1 - h_rescaled, mode='full')
        max_overlap = np.max(correlation) * dx
        return max_overlap
    
    # Gradient computation (approximate)
    def gradient(h_vec):
        eps = 1e-5
        grad = np.zeros_like(h_vec)
        base_val = objective(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            grad[i] = (objective(h_plus) - base_val) / eps
        return grad
    
    # Optimization with restarts
    max_restarts = 5
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Optimize with L-BFGS-B
        bounds = [(0, 1) for _ in range(n_points)]
        result = minimize(objective, h, method='L-BFGS-B', bounds=bounds, 
                         jac=gradient, options={'maxiter': 500})
        
        if result.success:
            h_opt = result.x
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            # Evaluate final result
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5_val = np.max(correlation) * dx
            
            if c5_val < best_c5:
                best_c5 = c5_val
                best_h = h_opt.copy()
                
            # Perturb the solution for next restart
            h = h_opt.copy() + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        else:
            # If optimization fails, try a simple perturbation
            h = h + np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        restart_count += 1
    
    # Final optimization with full gradient
    bounds = [(0, 1) for _ in range(n_points)]
    result = minimize(objective, best_h, method='L-BFGS-B', bounds=bounds, 
                     jac=gradient, options={'maxiter': 1000})
    
    if result.success:
        final_h = result.x
        final_h = final_h * (n_points / 2) / np.sum(final_h)
        correlation = np.correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(correlation) * dx
        best_h = final_h
        best_c5 = final_c5
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
