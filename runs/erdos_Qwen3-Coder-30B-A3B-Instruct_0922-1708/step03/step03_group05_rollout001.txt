import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        h = h / np.sum(h) * (n_points / 2)
    
    # Adam optimizer parameters
    beta1, beta2 = 0.9, 0.999
    eps = 1e-7
    alpha = 0.001
    
    # Initialize Adam moments
    m = np.zeros_like(h)
    v = np.zeros_like(h)
    
    # Best solution tracking
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Precompute indices for correlation
    max_lag = n_points - 1
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < 500:
        iteration += 1
        
        # Compute the overlap integral using correlation
        correlation = correlate(h, 1 - h, mode='full')
        # The full correlation includes lags from -max_lag to +max_lag
        # We only care about positive lags, so we're looking at positions [max_lag:]
        c5 = np.max(correlation[max_lag:]) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            correlation_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(correlation_plus[max_lag:]) * dx
            
            grad[i] = (c5_plus - c5) / eps_fd
        
        # Update using Adam optimizer
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** iteration)
        v_hat = v / (1 - beta2 ** iteration)
        
        h_new = h - alpha * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project to feasible region
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally perturb solution to escape local minima
        if np.random.rand() < max(0.1 * np.exp(-iteration/100), 0.01):
            perturbation_magnitude = 0.05
            perturbation = np.random.normal(0, perturbation_magnitude, n_points)
            h = h + perturbation
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final evaluation
    correlation_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation_final[max_lag:]) * dx
    
    return (best_h.tolist(), final_c5, n_points)
