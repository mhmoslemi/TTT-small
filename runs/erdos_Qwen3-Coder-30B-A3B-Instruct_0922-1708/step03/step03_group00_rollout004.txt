import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Add the initial solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points) * 0.5 + 0.25
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # PSO loop
    pbest = [p.copy() for p in particles]
    pbest_vals = [float('inf')] * n_particles
    
    # Compute initial global best
    gbest_idx = 0
    gbest_val = float('inf')
    
    # Evaluate initial particles
    for i in range(n_particles):
        correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
        c5 = np.max(correlation) * dx
        pbest_vals[i] = c5
        if c5 < gbest_val:
            gbest_val = c5
            gbest_idx = i
            best_h = particles[i].copy()
            best_c5 = c5
        
    # Optimization loop
    max_no_improvement = 50
    no_improvement_count = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(), np.random.rand()
            velocities[i] = (w * velocities[i] + 
                             c1 * r1 * (pbest[i] - particles[i]) + 
                             c2 * r2 * (particles[gbest_idx] - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -max_vel, max_vel)
            
            # Update position
            particles[i] += velocities[i]
            
            # Clip and rescale to satisfy constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate the new particle
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest_vals[i] = c5
                pbest[i] = particles[i].copy()
                
                # Check if global best
                if c5 < gbest_val:
                    gbest_val = c5
                    gbest_idx = i
                    best_h = particles[i].copy()
                    best_c5 = c5
                    no_improvement_count = 0
                    
        no_improvement_count += 1
        if no_improvement_count > max_no_improvement:
            break
            
    # Local refinement on the global best particle
    h_refine = best_h.copy()
    
    # Gradient descent refinement using finite differences
    patience = 0
    prev_c5 = best_c5
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
            
        correlation = np.correlate(h_refine, 1 - h_refine, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_refine.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h_refine)
        eps = 1e-4
        for j in range(n_points):
            h_plus = h_refine.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
            
        # Adaptive learning rate based on gradient
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h_refine - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check for no improvement
        if np.abs(c5 - prev_c5) < 1e-8:
            patience += 1
            if patience > 50:
                break
        else:
            patience = 0
            
        h_refine = h_new
        prev_c5 = c5
        
    return (best_h.tolist(), best_c5, n_points)
