import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 constant for given h"""
    corr = correlate(h, 1 - h, mode='full')
    return np.max(corr) * dx

def normalize_h(h, n_points):
    """Ensure sum(h) = n_points / 2 and h in [0,1]"""
    h = np.clip(h, 0, 1)
    scale = (n_points / 2) / np.sum(h)
    return h * scale

def crossover(parent1, parent2, cross_rate):
    """Single-point crossover with probability cross_rate"""
    if np.random.random() > cross_rate:
        return parent1.copy(), parent2.copy()
    
    n = len(parent1)
    point = np.random.randint(1, n)
    child1 = np.concatenate([parent1[:point], parent2[point:]])
    child2 = np.concatenate([parent2[:point], parent1[point:]])
    
    return child1, child2

def mutate(individual, mut_rate, n_points):
    """Gaussian mutation with probability mut_rate"""
    mutated = individual.copy()
    for i in range(len(mutated)):
        if np.random.random() < mut_rate:
            mutated[i] += np.random.normal(0, 0.1)
    return normalize_h(mutated, n_points)

def tournament_selection(population, fitnesses, k=5):
    """Select one individual using tournament selection"""
    indices = np.random.choice(len(population), k)
    best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
    return population[best_idx].copy()

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_size = int(0.2 * pop_size)
    cross_rate = 0.7
    mut_rate = 0.1
    
    population = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            # Interpolate to target resolution
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = normalize_h(h_init, n_points)
        population.append(h_init)
    else:
        # Start with a simple configuration
        h = np.ones(n_points) * 0.5
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Fill population
    while len(population) < pop_size:
        h = np.random.rand(n_points) * 0.5 + 0.25  # [0.25, 0.75]
        h = normalize_h(h, n_points)
        population.append(h)
    
    # Main optimization loop
    start_time = time.time()
    best_h = None
    best_c5 = float('inf')
    stagnation_count = 0
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_h(h, dx)
            fitnesses.append(c5)
        
        # Track best
        min_idx = np.argmin(fitnesses)
        if fitnesses[min_idx] < best_c5:
            best_c5 = fitnesses[min_idx]
            best_h = population[min_idx].copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Check for stagnation
        if stagnation_count > 100:
            # Add some random diversity
            for i in range(10):
                idx = np.random.randint(0, pop_size)
                population[idx] = np.random.rand(n_points) * 0.5 + 0.25
                population[idx] = normalize_h(population[idx], n_points)
            stagnation_count = 0
        
        # Sort by fitness (ascending)
        sorted_indices = np.argsort(fitnesses)
        population = [population[i] for i in sorted_indices]
        fitnesses = [fitnesses[i] for i in sorted_indices]
        
        # Keep elite
        new_population = [population[i].copy() for i in range(elite_size)]
        
        # Generate offspring
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            
            child1, child2 = crossover(parent1, parent2, cross_rate)
            
            child1 = mutate(child1, mut_rate, n_points)
            child2 = mutate(child2, mut_rate, n_points)
            
            new_population.append(child1)
            if len(new_population) < pop_size:
                new_population.append(child2)
        
        population = new_population[:pop_size]
    
    # Final evaluation
    if best_h is not None:
        final_c5 = evaluate_h(best_h, dx)
    else:
        # If no solution found, evaluate the best in the final population
        final_fitnesses = [evaluate_h(h, dx) for h in population]
        min_idx = np.argmin(final_fitnesses)
        best_h = population[min_idx].copy()
        final_c5 = final_fitnesses[min_idx]
    
    return (best_h.tolist(), final_c5, n_points)
