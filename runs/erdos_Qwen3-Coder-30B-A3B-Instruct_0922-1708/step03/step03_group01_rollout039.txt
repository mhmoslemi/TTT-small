import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h"""
    # Compute correlation with 1-h
    corr = np.correlate(h, 1-h, mode='full')
    # Return maximum value scaled by dx
    return np.max(corr) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    start_time = time.time()
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    step_size = 0.1
    stagnation_count = 0
    stagnation_threshold = 100
    
    current_h = h.copy()
    current_c5 = best_c5
    
    # Hash function for storing in tabu list
    def hash_h(h_vector):
        return hashlib.md5(h_vector.astype(np.float32).tobytes()).hexdigest()
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidates by perturbing current solution
        candidates = []
        perturbation_count = max(1, int(0.2 * n_points))
        
        for _ in range(50):
            h_candidate = current_h.copy()
            
            # Randomly select indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Perturb selected indices
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                h_candidate[i] += delta
            
            # Clip values to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Rescale to maintain constraint
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Create hash of candidate
            candidate_hash = hash_h(h_candidate)
            
            # Skip if in tabu list
            if candidate_hash in tabu_list:
                continue
                
            # Evaluate candidate
            c5 = evaluate_erdos_solution(h_candidate)
            candidates.append((h_candidate, c5, candidate_hash))
        
        if not candidates:
            # If no valid candidates, perturb current slightly
            h_candidate = current_h.copy()
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                h_candidate[i] += delta
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            candidate_hash = hash_h(h_candidate)
            c5 = evaluate_erdos_solution(h_candidate)
            candidates.append((h_candidate, c5, candidate_hash))
            
        # Sort by C₅ value
        candidates.sort(key=lambda x: x[1])
        
        # Select best valid candidate
        h_new, c5_new, candidate_hash = candidates[0]
        
        # Update best if improved
        if c5_new < best_c5:
            best_h = h_new.copy()
            best_c5 = c5_new
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Add current to tabu list
        current_hash = hash_h(current_h)
        tabu_list.add(current_hash)
        
        # Maintain tabu list size
        if len(tabu_list) > tabu_length:
            # Remove oldest entries
            tabu_list.clear()
            
        # Update current solution
        current_h = h_new
        current_c5 = c5_new
        
        # Occasionally accept worse solution for escape
        if stagnation_count > stagnation_threshold and np.random.random() < 0.1:
            # Accept worse candidate for escape
            pass
            
        # Increase step size if stagnating
        if stagnation_count > stagnation_threshold:
            step_size *= 1.2
            stagnation_count = 0
            
    return (best_h.tolist(), best_c5, n_points)
