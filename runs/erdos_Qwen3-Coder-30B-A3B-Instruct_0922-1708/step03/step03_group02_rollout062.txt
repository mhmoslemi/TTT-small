import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Create initial population
    particles.append(h_init)
    velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Add random particles
    for _ in range(n_particles - 1):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    omega = 1.0
    c1 = c2 = 2.0
    max_velocity = 0.2
    
    # Initialize personal bests
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    
    # Initialize global best
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            
            # Clip to bounds
            h = np.clip(h, 0, 1)
            
            # Rescale to satisfy constraint
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < personal_best_scores[i]:
                personal_best_scores[i] = score
                personal_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = omega * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Clamp velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            h = h + v
            
            # Store updated particle
            particles[i] = h
        
        # Periodic reset to prevent stagnation
        if step % 100 == 0 and step > 0:
            for i in range(1, n_particles):  # Keep first particle as reference
                particles[i] = np.random.uniform(0.4, 0.6, n_points)
                particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
                velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Decrease inertia over time
        omega = 1.0 - (step / 2000.0) * 0.2  # Linear decay
        omega = max(0.8, omega)
        
        step += 1
    
    # Final evaluation
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
