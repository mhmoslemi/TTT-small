        # 1. loop over + deterministic,  
        #    compute i+lag in range,  
        #    compute i-lag in range, 
        #    then compute  gradient[i] += coeff .* (1 - h[np.clip(...)]) 
        # or, even more precise:
        # values due to single-term effect?
        # So we need to build to concrete version:
        # Last correct approach:  
        # Let C_k = \sum_j h[j] (1 - h[j+k])
        # dC_k / dh[i] = 1 - h[i+k] if i+k in indices  
        #           = - h[i-k] if i-k in indices  
        # Correction for periodicity:
        # But perimeter part cannot be considered for漁 conclusions in nu 
        # Sum: gradient[i] = 1 - h[(i + lag) % N] - h[(i - lag) % N]
        
        for i in range(len(h_array)):  
            gradient[i] = 1.0 - h_array[(i + lag) % len(h_array)] - h_array[(i - lag) % len(h_array)]
        
        return gradient * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient via exact formula for max lag
        grad = compute_grad_correct(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
