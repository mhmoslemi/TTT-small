import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    max_iter = 100
    iter_count = 0
    
    while time.time() - start_time < budget_s - 1 and iter_count < max_iter:
        # Compute correlation to get current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_current = np.max(correlation) * dx
        
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
        
        # Build cvxpy optimization problem to refine h
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # Constraints
        constraints = [h_var >= 0, h_var <= 1, cp.sum(h_var) == n_points / 2, t >= 0]
        
        # Add McCormick envelopes for bilinear terms
        # For each lag k, we want to bound h[i] * (1 - h[i+k]) = h[i] - h[i]*h[i+k]
        # We introduce auxiliary variables for products
        lag_length = 2 * n_points - 1  # Full correlation length
        full_correlation = np.zeros(lag_length)
        
        # Iterate through lags
        for k in range(lag_length):
            # Get slices for the current lag k
            if k < n_points:
                # h[i] * (1 - h[i+k]) where i = 0,...,n_points-1-k
                idx_i = slice(0, n_points - k)
                idx_j = slice(k, n_points)
            else:
                idx_i = slice(k - n_points + 1, n_points)
                idx_j = slice(0, n_points - (k - n_points + 1))
                
            if idx_i.stop > idx_i.start and idx_j.stop > idx_j.start:
                # Use McCormick envelopes for each pair
                zi = cp.Variable((idx_i.stop - idx_i.start, idx_j.stop - idx_j.start))
                
                # Add McCormick constraints
                for i in range(idx_i.start, idx_i.stop):
                    for j in range(idx_j.start, idx_j.stop):
                        # z[i][j] = h[i] * (1 - h[j])
                        idx_i_loc = i - idx_i.start
                        idx_j_loc = j - idx_j.start
                        z_var = zi[idx_i_loc, idx_j_loc]
                        
                        # McCormick envelopes for z = h[i] * (1 - h[j])
                        constraints.extend([
                            z_var >= h_var[i] + (1 - h_var[j]) - 1,
                            z_var <= h_var[i],
                            z_var <= 1 - h_var[j],
                            z_var >= 0
                        ])
                
                # Add contribution to t
                # ∫ h(x)(1-h(x+k)) dx ≈ sum((h[i] - h[i]*h[i+k]) * dx)
                # So we want to upper bound ∑ (h_i - h_i * h_j)
                # Sum of z: sum(zi) * dx <= t
                constraints.append(cp.sum(zi) * dx <= t)
            else:
                # no valid pairs
                pass
                
        # Objective
        objective = cp.Minimize(t)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False, max_iters=100)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_new = h_var.value
                if h_new is not None:
                    h = np.clip(h_new, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
            else:
                # Fall back on a small optimization step if solver fails
                h = h * 0.99 + 0.01 * np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            # fall back to simple rescaling
            h = h * 0.99 + 0.01 * np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        
        iter_count += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
