import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm with particles
    num_particles = 50
    particles = []
    velocities = []
    
    # Use initial solution if available
    if 'initial_h_values' in globals():
        h_base = np.array(initial_h_values)
        if len(h_base) != n_points:
            x_orig = np.linspace(0, 2, len(h_base))
            x_target = np.linspace(0, 2, n_points)
            h_base = np.interp(x_target, x_orig, h_base)
    else:
        h_base = np.ones(n_points) * 0.5
    
    # Initialize particles
    for i in range(num_particles):
        if i < num_particles // 10 and 'initial_h_values' in globals():
            h = h_base.copy()
        else:
            h = np.random.uniform(0, 1, n_points)
        # Ensure constraint sum = n_points/2
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 0.7
    c1 = 1.5
    c2 = 1.5
    
    # Track best positions
    p_best = [p.copy() for p in particles]
    p_best_values = [float('inf')] * num_particles
    g_best = None
    g_best_value = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            # Compute overlap integral
            correlation = correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < p_best_values[i]:
                p_best_values[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_value:
                g_best_value = c5
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = (w * v + 
                 c1 * r1 * (p_best[i] - h) + 
                 c2 * r2 * (g_best - h))
            
            # Update position
            h_new = h + v
            
            # Clip and rescale to satisfy constraints
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
            velocities[i] = v
    
    # Final evaluation of global best
    if g_best is not None:
        correlation = correlate(g_best, 1 - g_best, mode='full')
        final_c5 = np.max(correlation) * dx
        return (g_best.tolist(), final_c5, n_points)
    else:
        # Fallback to the best particle found
        best_idx = np.argmin(p_best_values)
        final_h = p_best[best_idx]
        correlation = correlate(final_h, 1 - final_h, mode='full')
        final_c5 = np.max(correlation) * dx
        return (final_h.tolist(), final_c5, n_points)
