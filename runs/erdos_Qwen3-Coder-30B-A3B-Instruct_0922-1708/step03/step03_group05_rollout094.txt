import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute C5 from current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy() 
        
        # Solve convex relaxation using CVXPY
        h_relaxed = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_relaxed >= 0,
            h_relaxed <= 1,
            cp.sum(h_relaxed) == n_points/2
        ]
        
        # Compute all overlap terms for objective
        overlaps = []
        for i in range(n_points):
            # h[i:]*(1-h[i:]) - exploit overlap computation for full correlation
            overlap_terms = []
            for k in range(n_points - i):
                overlap_terms.append(h_relaxed[i+k] * (1 - h_relaxed[i]))
            if overlap_terms:
                overlaps.append(cp.sum(overlap_terms))
        
        # Minimize maximum overlap (requires reformulation)
        max_overlap = cp.Variable()
        for i, overlap in enumerate(overlaps):
            constraints.append(overlap <= max_overlap)
            
        obj = cp.Minimize(max_overlap)
        prob = cp.Problem(obj, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False, abstol=1e-6, reltol=1e-6)
            if prob.status in ["optimal", "optimal_inaccurate"]:
                h_relaxed_sol = h_relaxed.value
                if h_relaxed_sol is not None:
                    # Rounding approach
                    h_rounded = (h_relaxed_sol >= 0.5).astype(float)
                    current_sum = np.sum(h_rounded)
                    
                    # Adjust to sum constraint
                    if current_sum > n_points / 2:
                        # Remove 1's starting from smallest relaxed values
                        indices = np.argsort(h_relaxed_sol)
                        indices_to_zero = indices[:int(current_sum - n_points/2)]
                        h_rounded[indices_to_zero] = 0
                    elif current_sum < n_points / 2:
                        # Add 1's starting from largest relaxed values
                        indices = np.argsort(-h_relaxed_sol)
                        indices_to_one = indices[:int(n_points/2 - current_sum)]
                        h_rounded[indices_to_one] = 1
                    
                    # Normalize by rescaling to sum == n_points/2
                    h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
                    
                    # Final validation and replacement
                    corr = np.correlate(h_rounded, 1 - h_rounded, mode='full')
                    c5_new = np.max(corr) * dx
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_rounded.copy()
                        
                    # Use rounded value as new starting point
                    h = h_rounded.copy()
                    continue
        except:
            pass
        
        # Fallback optimization with gradient ascent
        expanded_h = np.pad(h, (0, n_points - 1), 'constant', constant_values=0)
        expanded_h_rev = np.pad(1 - h, (0, n_points - 1), 'constant', constant_values=0)
        
        # Compute gradient step
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            corr = np.correlate(h, 1 - h, mode='full')
            c5_here = np.max(corr) * dx
            
            grad[i] = (c5_plus - c5_here) / eps
            
        # Gradient descent update
        lr = 0.01
        h_new = h - lr * grad
        
        # Clip and normalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
    
    # Final validation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
