import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define a simple, adaptive gradient descent with momentum
    def adaptive_gradient_descent(h, iterations=100, lr_base=0.01):
        h_curr = h.copy()
        velocity = np.zeros_like(h)
        for i in range(iterations):
            # Compute gradient with finite differences
            grad = np.zeros_like(h_curr)
            eps = 1e-5
            c5_curr = compute_c5(h_curr)
            
            for j in range(n_points):
                h_plus = h_curr.copy()
                h_plus[j] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[j] = (c5_plus - c5_curr) / eps
                
            # Adaptive learning rate
            lr = lr_base / (1 + np.linalg.norm(grad) * 1e-3)
            
            # Momentum
            velocity = 0.9 * velocity + lr * grad
            h_new = h_curr - velocity
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h_curr = h_new
            
        return h_curr
    
    # Helper to compute C5
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Fast multi-start local search with restarting
    restart_count = 0
    max_restarts = 50
    stagnation_count = 0
    max_stagnation = 50
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Run gradient descent
        h_opt = adaptive_gradient_descent(h, iterations=50)
        c5_opt = compute_c5(h_opt)
        
        if c5_opt < best_c5:
            best_c5 = c5_opt
            best_h = h_opt.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count >= max_stagnation:
            # Random restart
            h = np.random.rand(n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            restart_count += 1
            stagnation_count = 0
        else:
            # Local improvement around current best
            h = best_h.copy()
            h += np.random.normal(0, 0.01, n_points)
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
