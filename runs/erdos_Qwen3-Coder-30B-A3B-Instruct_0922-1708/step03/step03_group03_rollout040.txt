            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)

        h = np.zeros(n_points)

    C_low = 0.0
    C_high = 0.38085
    tolerance = 1e-5
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def is_feasible(h_vec, C_test):
        # Check if maximum overlap integral is <= C_test
        c5 = compute_c5(h_vec)
        return c5 <= C_test
    
    # Main binary search loop
    while time.time() - start_time < budget_s - 1 and C_high - C_low > tolerance:
        C_test = (C_low + C_high) / 2.0
        
        # Try to solve a modified optimization to find a feasible solution
        h_vars = cp.Variable(n_points)
        constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]

        test_corr = np.correlate(h, 1 - h, mode='full')
        # Focus on top lags with highest overlap; test with local optimization approach
        max_lags = np.argsort(test_corr)[-min(20, len(test_corr)):][::-1]

            pass
            
        # Use gradient descent with constrained optimization to improve solution
        h_method = h.copy()
        for iter in range(50):  # limited iterations
            if time.time() - start_time > budget_s - 2:
                break
            # Compute gradient numerically
            grad = np.zeros_like(h_method)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h_method.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_orig = np.correlate(h_method, 1 - h_method, mode='full')
                c5_orig = np.max(corr_orig) * dx
                
                grad[i] = (c5_plus - c5_orig) / eps
            
            # Update step
            lr = 0.005
            h_new = h_method - lr * grad
            
            # Clip and rescale
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            # Keep the best feasible solution
            if is_feasible(h_new, C_test):

        if is_feasible(h_method, C_test):
            C_high = C_test
            if compute_c5(h_method) < best_c5:

            constraints = [h_vars >= 0, h_vars <= 1, cp.sum(h_vars) == n_points / 2]

                    if c5_cvx < best_c5:
                        best_h = h_cvx
                        best_c5 = c5_cvx
        except:
            pass
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
