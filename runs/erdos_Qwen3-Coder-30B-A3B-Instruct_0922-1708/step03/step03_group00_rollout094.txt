import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    cooling_rate = 0.995
    perturbation_scale = 0.05
    
    # Optimization parameters
    max_iterations = 1000
    patience = 100
    improvement_count = 0
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def adjust_h(h):
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        # Rescale to meet constraint
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    for i in range(max_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        current_c5 = compute_c5(h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            improvement_count = 0
        else:
            improvement_count += 1
            
        # Generate neighbor
        h_new = h + np.random.normal(0, perturbation_scale, n_points)
        h_new = adjust_h(h_new)
        
        new_c5 = compute_c5(h_new)
        
        # Simulated Annealing acceptance
        if new_c5 < current_c5 or np.random.random() < np.exp(-(new_c5 - current_c5) / T):
            h = h_new
        
        # Cooling
        T *= cooling_rate
        
        # Early stopping
        if improvement_count > patience:
            break
    
    # Gradient descent refinement
    def objective(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def gradient(h_vec):
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        eps = 1e-4
        grad = np.zeros_like(h)
        base_c5 = objective(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = objective(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Use scipy to optimize the refined solution
    h_refined = best_h.copy()
    lr = 0.01
    
    # Refine using gradient-based method
    for _ in range(1000):
        if time.time() - start_time > budget_s - 1:
            break
        grad = gradient(h_refined)
        h_new = h_refined - lr * grad
        h_refined = adjust_h(h_new)
        
        # Adaptive learning rate
        if np.max(np.abs(grad)) < 1e-6:
            lr *= 0.95
        if lr < 1e-6:
            lr = 1e-6
    
    # Final evaluation
    final_c5 = compute_c5(h_refined)
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h_refined.copy()
    
    return (best_h.tolist(), best_c5, n_points)
