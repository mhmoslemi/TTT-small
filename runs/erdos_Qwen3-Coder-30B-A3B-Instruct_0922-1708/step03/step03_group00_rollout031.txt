import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_shifts = 200
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Precompute shifts
    shifts = np.linspace(0, 2, max_shifts)
    
    start_time = time.time()
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Cutting plane approach
    C = cp.Variable()
    h_var = cp.Variable(n_points)
    
    # Constraints
    constraints = [
        0 <= h_var,
        h_var <= 1,
        cp.sum(h_var) == n_points / 2
    ]
    
    # Add initial constraint
    correlation = np.correlate(h, 1 - h, mode='full')
    max_overlap = np.max(correlation) * dx
    constraints.append(C >= max_overlap)
    
    # Objective
    prob = cp.Problem(cp.Minimize(C), constraints)
    prob.solve(solver=cp.ECOS, verbose=False)
    
    best_c5 = C.value
    best_h = h_var.value
    
    # Iterative refinement
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlaps for all shifts
        overlaps = []
        for k in shifts:
            # Use convolution for shifts
            h_shifted = np.roll(h, int(k / dx))
            overlap = np.sum(h * (1 - h_shifted)) * dx
            overlaps.append(overlap)
            
        max_overlap_idx = np.argmax(overlaps)
        max_k = shifts[max_overlap_idx]
        
        # Add new constraint
        h_shifted = np.roll(h, int(max_k / dx))
        new_constraint = cp.sum(cp.multiply(h_var, 1 - h_shifted)) <= C
        constraints.append(new_constraint)
        
        # Re-solve
        prob = cp.Problem(cp.Minimize(C), constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if C.value is not None and C.value < best_c5:
            best_c5 = C.value
            best_h = h_var.value
            
    return (best_h.tolist(), best_c5, n_points)
