import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    # Start with initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        h = h / np.sum(h) * (n_points / 2)
    else:
        n_points = 100
        dx = 2.0 / n_points
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T0 = 0.1
    alpha = 0.995
    max_iter = 1000
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    def project_to_feasible(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    # Temporarily store values
    T = T0
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Current C5
        current_c5 = compute_c5(h)
        
        # Save current configuration
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h.copy()
        
        # Perturb some points
        n_perturb = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_new[i] += delta
        
        # Project to feasible set
        h_new = project_to_feasible(h_new)
        
        # Compute new C5
        new_c5 = compute_c5(h_new)
        
        # Accept or reject based on SA criteria
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
        
        # Cool down
        T *= alpha
    
    # Final evaluation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
