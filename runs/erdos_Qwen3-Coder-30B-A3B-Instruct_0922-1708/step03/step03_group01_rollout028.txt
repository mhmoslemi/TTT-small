import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define the objective and gradient
    def objective(h_vec):
        # Compute the overlap integral
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def gradient(h_vec):
        # Find the shift k that maximizes overlap
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_k = np.argmax(correlation)
        
        # Compute analytical gradient
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            # Gradient contribution from shift k
            # ∂/∂h[i] ∑ⱼ h[j](1 - h[j+k]) = (1 - h[i+k]) - h[i-k]
            k = max_k
            # avoid boundary issues by using appropriate indices
            if (i + k) < len(h_vec):
                grad[i] += -h_vec[i + k] * dx
            if (i - k) >= 0:
                grad[i] += -h_vec[i - k] * dx
                
            # also add the 1 term for diff with 1-h
            if (i + k) < len(h_vec):
                grad[i] += dx
        return grad
    
    # Define constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_sum}
    
    # Define bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization with SLSQP
    while time.time() - start_time < budget_s - 1:
        # Run optimization step
        result = minimize(objective, h, method='SLSQP', jac=gradient, 
                          bounds=bounds, constraints=cons, 
                          options={'maxiter': 1000, 'ftol': 1e-8})
        
        if not result.success:
            break
            
        h = result.x
        current_c5 = objective(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Early stopping if we've found a good enough solution
        if best_c5 < 0.3807:
            break

    return (best_h.tolist(), best_c5, n_points)
