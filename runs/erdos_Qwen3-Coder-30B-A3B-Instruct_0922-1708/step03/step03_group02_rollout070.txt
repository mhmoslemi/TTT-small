import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Load initial solution if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
    
    # Create initial population
    population = []
    population.append(h_init)
    for _ in range(pop_size - 1):
        h = np.random.random(n_points) * 0.5 + 0.25
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    # Convert to numpy array for faster computation
    population = np.array(population)
    
    # Track best solution
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    gen = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        # Evaluate fitness for all individuals
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
        
        fitness = np.array(fitness)
        
        # Update best solution
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Rank individuals
        sorted_indices = np.argsort(fitness)
        
        # Create new population
        new_population = []
        
        # Preserve elite
        elite_size = int(pop_size * elite_rate)
        for i in range(elite_size):
            new_population.append(population[sorted_indices[i]].copy())
        
        # Generate offspring
        while len(new_population) < pop_size:
            # Tournament selection
            parents = []
            for _ in range(2):
                tourn = np.random.choice(pop_size, tournament_size)
                winner = tourn[np.argmin(fitness[tourn])]
                parents.append(population[winner])
            
            # Crossover
            if np.random.random() < cross_rate:
                crossover_point = np.random.randint(1, n_points-1)
                child1 = np.concatenate([parents[0][:crossover_point], parents[1][crossover_point:]])
                child2 = np.concatenate([parents[1][:crossover_point], parents[0][crossover_point:]])
            else:
                child1 = parents[0].copy()
                child2 = parents[1].copy()
            
            # Mutation for both children
            for child in [child1, child2]:
                for i in range(n_points):
                    if np.random.random() < mut_rate:
                        child[i] += np.random.normal(0, 0.05)
                
                # Clip and normalize
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        # Trim to exact population size
        population = new_population[:pop_size]
        
        gen += 1
    
    # Final recomputation of best solution  
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
