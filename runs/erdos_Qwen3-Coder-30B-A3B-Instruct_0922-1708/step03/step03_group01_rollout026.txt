import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vec):
        # Compute overlap integral
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
        
    def objective(h_vec):
        return compute_c5(h_vec)
        
    def gradient(h_vec):
        # Compute the gradient of the max overlap
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_idx = np.argmax(correlation)
        # This is an approximation based on the idea that
        # derivative of max(f_i) at i = argmax(f_i) is approximately
        # derivative of f_i at i = argmax(f_i)
        # For simplicity, we will calculate a numerical gradient
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        for i in range(len(h_vec)):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            grad[i] = (compute_c5(h_plus) - compute_c5(h_vec)) / eps
        return grad
        
    # Constraints
    def constraint_sum(h_vec):
        return np.sum(h_vec) - n_points / 2
        
    # Bounds
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Trying a more robust optimization approach
    constraints = {'type': 'eq', 'fun': constraint_sum}
    
    iteration = 0
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current value and update best
        c5_current = compute_c5(h)
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
            
        # Use SLSQP with analytical gradient
        try:
            result = minimize(
                objective,
                h,
                method='SLSQP',
                bounds=bounds,
                constraints=constraints,
                jac=gradient,
                options={'maxiter': 5},
                callback=lambda x: None
            )
            if result.success:
                h = result.x
                # Ensure it's valid
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
            else:
                # Fallback to basic optimization if needed
                h = np.clip(h + np.random.normal(0, 0.01, len(h)), 0, 1)
                h = h * (n_points / 2) / np.sum(h)
        except:
            h = np.clip(h + np.random.normal(0, 0.01, len(h)), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Final refinement
    c5_final = compute_c5(best_h)
    return (best_h.tolist(), c5_final, n_points)
