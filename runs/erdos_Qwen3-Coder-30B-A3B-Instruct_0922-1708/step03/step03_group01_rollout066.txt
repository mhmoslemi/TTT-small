import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low = 0.38
    high = 0.4
    tol = 1e-4
    
    while high - low > tol and time.time() - start_time < budget_s - 1:
        c_mid = (low + high) / 2.0
        
        # Define variables
        h_var = cp.Variable(n_points)
        
        # Define constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute the overlap for shift k
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            constraints.append(overlap <= c_mid / dx)
        
        # Define objective (minimize C5)
        objective = cp.Minimize(c_mid)
        
        # Create and solve the problem
        problem = cp.Problem(objective, constraints)
        
        try:
            problem.solve(solver=cp.ECOS, verbose=False)
            if problem.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_solution = h_var.value
                # Clip and normalize
                h_solution = np.clip(h_solution, 0, 1)
                h_solution = h_solution * (n_points / 2) / np.sum(h_solution)
                
                # Evaluate the actual C5
                c5_eval = np.max(np.correlate(h_solution, 1 - h_solution, mode='full')) * dx
                
                if c5_eval < best_c5:
                    best_c5 = c5_eval
                    best_h = h_solution.copy()
                
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
            
        # Avoid spending too much time in binary search
        if time.time() - start_time > budget_s - 1:
            break
    
    # Final evaluation of best solution
    if best_c5 == float('inf'):
        # Fallback to standard evaluation
        corr = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(corr) * dx
        best_h = h.copy()
        
    return (best_h.tolist(), best_c5, n_points)
