import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Multi-population GA approach
    population_size = 50
    generations = 300
    elite_size = population_size // 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    tournament_size = population_size // 10
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        pop_h = h.copy()
        population.append(pop_h)
        for i in range(population_size - 1):
            # Add slight perturbations of initial
            pert = np.random.normal(0, 0.05, n_points)
            new_h = pop_h + pert
            new_h = np.clip(new_h, 0, 1)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            population.append(new_h)
    else:
        for i in range(population_size):
            new_h = np.random.uniform(0, 1, n_points)
            new_h = new_h * (n_points / 2) / np.sum(new_h)
            population.append(new_h)
    
    # Keep track of best globally
    for gen in range(generations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate population
        fitness = []
        for h_candidate in population:
            correlation = np.correlate(h_candidate, 1 - h_candidate, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Track best solution in this generation
        best_idx = np.argmin(fitness)
        if fitness[best_idx] < best_c5:
            best_c5 = fitness[best_idx]
            best_h = population[best_idx].copy()
            
        # Selection: tournament selection
        selected = []
        for _ in range(population_size):
            tournament = np.random.choice(population_size, tournament_size, replace=False)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx])
            
        # Elitism
        elite_indices = np.argsort(fitness)[:elite_size]
        new_population = [population[i] for i in elite_indices]
        
        # Crossover and mutation
        while len(new_population) < population_size:
            # Crossover
            if np.random.random() < crossover_rate:
                parent1_idx, parent2_idx = np.random.choice(population_size, 2, replace=False)
                parent1, parent2 = selected[parent1_idx], selected[parent2_idx]
                
                # Blend crossover
                crossover_point = np.random.randint(1, n_points-1)
                child = np.zeros(n_points)
                child[:crossover_point] = parent1[:crossover_point]
                child[crossover_point:] = parent2[crossover_point:]
                # Blend the transition region
                blend_ratio = 0.5
                child[crossover_point-1] = blend_ratio * parent1[crossover_point-1] + (1-blend_ratio) * parent2[crossover_point-1]
                child[crossover_point] = blend_ratio * parent1[crossover_point] + (1-blend_ratio) * parent2[crossover_point]
                
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            else:
                # Just copy one parent
                parent_idx = np.random.choice(population_size)
                child = selected[parent_idx].copy()
                
            # Mutation
            if np.random.random() < mutation_rate:
                for i in range(n_points):
                    if np.random.random() < 0.1:  # 10% chance per gene
                        child[i] += np.random.normal(0, 0.01)
                # Clip and rescale
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:population_size]  # Trim to exact size if necessary
        
    # Final refinement with gradient descent
    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break

        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Use finite differences for gradient estimation
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
    
        # Update with adaptive learning rate
        lr = 0.001 / (1 + np.linalg.norm(grad) * 1e-4)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Update if better
        correlation = np.correlate(h, 1 - h, mode='full')
        c5_new = np.max(correlation) * dx
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
            
        # Very simple convergence check
        if np.abs(c5 - c5_new) < 1e-10:
            break
    
    return (best_h.tolist(), best_c5, n_points)
