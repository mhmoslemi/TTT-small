import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # 10% seeded from initial construction
    for i in range(int(0.1 * num_particles)):
        if 'initial_h_values' in globals():
            h_init = np.array(initial_h_values)
            if len(h_init) != n_points:
                x_orig = np.linspace(0, 2, len(h_init))
                x_target = np.linspace(0, 2, n_points)
                h_init = np.interp(x_target, x_orig, h_init)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        else:
            h_init = np.ones(n_points) * 0.5
        particles.append(h_init)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
    
    # Remaining particles randomly initialized
    for i in range(int(0.1 * num_particles), num_particles):
        h_init = np.random.rand(n_points) * 0.5 + 0.25
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init)
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        
    # Best positions for each particle
    p_best = [p.copy() for p in particles]
    p_best_c5 = [float('inf')] * num_particles
    
    # Global best
    g_best = best_h.copy()
    g_best_c5 = best_c5
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update global best
        for i in range(num_particles):
            particle = particles[i]
            # Compute C5 for this particle
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = particle.copy()
                
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = particle.copy()
                
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            v = w * velocities[i] + \
                c1 * r1 * (p_best[i] - particles[i]) + \
                c2 * r2 * (g_best - particles[i])
            v = np.clip(v, -max_vel, max_vel)
            
            new_pos = particles[i] + v
            
            # Project to feasible region
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos / np.sum(new_pos) * (n_points / 2)
            
            velocities[i] = v
            particles[i] = new_pos
            
        # Update global best
        if g_best_c5 < best_c5:
            best_c5 = g_best_c5
            best_h = g_best.copy()
    
    return (best_h.tolist(), best_c5, n_points)
