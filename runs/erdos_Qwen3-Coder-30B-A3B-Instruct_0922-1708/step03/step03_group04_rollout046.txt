import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient numerically 
    def compute_gradient(h_array):
        grad = np.zeros_like(h_array)
        current_c5 = compute_c5(h_array)
        for i in range(n_points):
            h_plus = h_array.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart with better initialization
        if iteration % 500 == 0:
            # Try convex relaxation as a better initialization
            try:
                h_relaxed = np.ones(n_points) * 0.5
                # Convex relaxation: min sum h_i - 2*sum h_i^2
                # Using a simple iterative method
                for _ in range(50):
                    h_prev = h_relaxed.copy()
                    # Simple gradient ascent step on the relaxed objective
                    grad_relaxed = 1 - 2 * h_relaxed
                    h_relaxed = h_relaxed + 0.001 * grad_relaxed
                    h_relaxed = np.clip(h_relaxed, 0, 1)
                    h_relaxed = h_relaxed * (n_points / 2) / np.sum(h_relaxed)
                    if np.max(np.abs(h_relaxed - h_prev)) < 1e-6:
                        break
                h = h_relaxed
            except:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
