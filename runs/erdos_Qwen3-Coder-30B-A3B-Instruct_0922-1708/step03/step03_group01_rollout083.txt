import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill remaining population with random valid individuals
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = None
    best_c5 = float('inf')
    
    start_time = time.time()
    gen = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        gen += 1
        fitness = []
        
        # Evaluate fitness for all individuals
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                no_improvement = 0
        
        # Select top 40%
        selected_indices = np.argsort(fitness)[:int(0.4 * pop_size)]
        selected = [population[i] for i in selected_indices]
        
        # Generate new population
        new_population = selected.copy()
        
        # Crossover: create offspring
        while len(new_population) < pop_size:
            parent1, parent2 = np.random.choice(selected, 2, replace=False)
            
            # Uniform crossover
            mask = np.random.rand(n_points) > 0.5
            child = np.where(mask, parent1, parent2)
            
            # Mutation: perturb 5% of points
            mutate_indices = np.random.choice(n_points, int(0.05 * n_points), replace=False)
            for i in mutate_indices:
                child[i] += np.random.normal(0, 0.1)
            
            # Clip and rescale
            child = np.clip(child, 0, 1)
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population[:pop_size]
        no_improvement += 1
    
    return (best_h.tolist(), best_c5, n_points)
