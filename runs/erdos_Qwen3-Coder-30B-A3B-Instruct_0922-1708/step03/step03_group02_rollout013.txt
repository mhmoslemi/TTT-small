import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add random particles
    for i in range(n_particles - len(particles)):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    # Initialize velocities
    for i in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    v_max = 0.2
    
    # Personal best positions and fitness
    personal_best = [particles[i].copy() for i in range(n_particles)]
    personal_best_fitness = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_fitness = float('inf')
    
    start_time = time.time()
    step = 0
    stagnation_count = 0
    max_stagnation = 200
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Update global best
        for i in range(n_particles):
            # Evaluate fitness for particle i
            correlation = np.correlate(particles[i], 1 - particles[i], mode='full')
            fitness = np.max(correlation) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = particles[i].copy()
            
            # Update global best
            if fitness < global_best_fitness:
                global_best_fitness = fitness
                global_best = particles[i].copy()
                stagnation_count = 0
        
        # Update stagnation
        stagnation_count += 1
        if stagnation_count > max_stagnation:
            # Reset some particles
            for i in range(n_particles // 5):
                idx = np.random.randint(0, n_particles)
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h / np.sum(h) * (n_points / 2)
                particles[idx] = h
                velocities[idx] = np.random.uniform(-0.1, 0.1, n_points)
            stagnation_count = 0
        
        # Update particles
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (personal_best[i] - particles[i]) + 
                           c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Clamp to [0,1]
            particles[i] = np.clip(particles[i], 0, 1)
            
            # Rescale to meet constraint sum(h) = n_points / 2
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
        
        # Adaptively adjust parameters
        if step % 50 == 0:
            w = max(0.8, w * 0.99)
            if step % 200 == 0:
                v_max *= 0.95
    
    # Final check
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
