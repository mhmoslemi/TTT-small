import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Use a more sophisticated approach: alternating optimization
    # and local refinement
    iter_count = 0
    while time.time() - start_time < budget_s - 1:
        iter_count += 1
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Refine the solution with gradient-based method but with smarter steps
        # Use a projected gradient descent with adaptive learning rate
        grad = np.zeros_like(h)
        eps = 1e-5
        
        # Compute gradient by finite differences
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps

        # Update with adaptive step size
        # Start with a fixed step but reduce if needed
        lr = 0.01
        
        # Try a few different learning rates to see improvement
        best_lr = lr
        best_improvement = 0
        
        for test_lr in [0.001, 0.005, 0.01, 0.02]:
            h_test = h - test_lr * grad
            h_test = np.clip(h_test, 0, 1)
            h_test = h_test * (n_points / 2) / np.sum(h_test)
            
            corr_test = np.correlate(h_test, 1 - h_test, mode='full')
            c5_test = np.max(corr_test) * dx
            
            improvement = c5 - c5_test
            if improvement > best_improvement:
                best_improvement = improvement
                best_lr = test_lr
                
        # Apply best learning rate
        h_new = h - best_lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Occasionally do a local search refinement
        if iter_count % 10 == 0:
            # Try local search around current point
            h_local = h.copy()
            for _ in range(5):
                # Small perturbations and evaluate
                perturbed = h_local + np.random.normal(0, 0.001, n_points)
                perturbed = np.clip(perturbed, 0, 1)
                perturbed = perturbed * (n_points / 2) / np.sum(perturbed)
                
                corr = np.correlate(perturbed, 1 - perturbed, mode='full')
                c5_local = np.max(corr) * dx
                
                if c5_local < c5:
                    h_local = perturbed
                    c5 = c5_local
            
            h = h_local

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
