import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    temp = 0.1
    cooling_rate = 0.99
    perturbation_scale = 0.05
    
    # Annealing phase
    max_annealing_iterations = 500
    no_improvement_count = 0
    max_no_improvement = 100
    
    while iteration < max_annealing_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Generate neighbor
        h_new = h + np.random.normal(0, perturbation_scale, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new overlap
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject based on simulated annealing
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / temp):
            h = h_new
            
        # Update temperature
        temp *= cooling_rate
        
        # Early termination if no improvement
        if no_improvement_count > max_no_improvement:
            break
    
    # Gradient descent refinement
    max_grad_iterations = 200
    lr = 0.001
    decay_rate = 0.95
    
    for i in range(max_grad_iterations):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current c5
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(best_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update h
        best_h = best_h - lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        # Decay learning rate
        if i % 50 == 0 and i > 0:
            lr *= decay_rate
    
    return (best_h.tolist(), best_c5, n_points)
