import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Place high values in specific intervals
        idx1 = np.where((np.linspace(0, 2, n_points) >= 0.2) & 
                        (np.linspace(0, 2, n_points) <= 0.4))[0]
        idx2 = np.where((np.linspace(0, 2, n_points) >= 1.2) & 
                        (np.linspace(0, 2, n_points) <= 1.4))[0]
        h[idx1] = 1.0
        h[idx2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    # Set up annealing parameters
    T = 0.1
    alpha = 0.995
    lr = 0.01
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def project_h(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_new = h.copy()
        # Perturb about 10% of elements
        n_perturb = max(1, n_points // 10)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, 0.05)
        h_new = project_h(h_new)
        
        new_c5 = compute_c5(h_new)
        old_c5 = compute_c5(h)
        
        if new_c5 < old_c5 or np.random.rand() < np.exp(-(new_c5 - old_c5) / T):
            h = h_new
            if new_c5 < best_c5:
                best_c5 = new_c5
                best_h = h.copy()
        
        # Gradient Descent Step
        eps = 1e-5
        grad = np.zeros(n_points)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_h(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - old_c5) / eps
        
        # Adaptive learning rate
        if new_c5 < old_c5:
            lr *= 1.01
        else:
            lr *= 0.95
        lr = np.clip(lr, 0.001, 0.02)
        
        h_new = h - lr * grad
        h_new = project_h(h_new)
        h = h_new
        
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
        
        # Update temperature
        T *= alpha
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
