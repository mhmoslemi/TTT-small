import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    w = 0.7
    c1 = 2.0
    c2 = 2.0
    vmax = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            h_init = h.copy()
            # Perturb slightly
            h_init += np.random.normal(0, 0.01, n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        else:
            h_init = np.random.rand(n_points)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-vmax, vmax, n_points))
        corr = np.correlate(h_init, 1 - h_init, mode='full')
        val = np.max(corr) * dx
        pbest.append(h_init.copy())
        pbest_vals.append(val)
        
    gbest_idx = np.argmin(pbest_vals)
    gbest = pbest[gbest_idx].copy()
    gbest_val = pbest_vals[gbest_idx]
    
    # PSO loop
    no_improvement = 0
    while time.time() - start_time < budget_s - 1 and no_improvement < 50:
        iteration += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                            c1 * r1 * (pbest[i] - particles[i]) + 
                            c2 * r2 * (gbest - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vmax, vmax)
            
            # Update position
            particles[i] += velocities[i]
            
            # Enforce constraints
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
            
            # Evaluate
            corr = np.correlate(particles[i], 1 - particles[i], mode='full')
            val = np.max(corr) * dx
            
            if val < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = val
            
            if val < gbest_val:
                gbest = particles[i].copy()
                gbest_val = val
                no_improvement = 0
                
        no_improvement += 1
        
        # Local refinement every few iterations
        if iteration % 10 == 0:
            # Refine gbest with gradient descent
            h = gbest.copy()
            for _ in range(20):
                if time.time() - start_time >= budget_s - 1:
                    break
                    
                # Compute gradient with finite differences
                grad = np.zeros_like(h)
                eps = 1e-4
                
                for j in range(n_points):
                    h_plus = h.copy()
                    h_plus[j] += eps
                    h_plus = np.clip(h_plus, 0, 1)
                    h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                    
                    corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                    c5_plus = np.max(corr_plus) * dx
                    
                    grad[j] = (c5_plus - gbest_val) / eps
                
                # Adaptive learning rate
                lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
                h_new = h - lr * grad
                
                # Clip and rescale
                h_new = np.clip(h_new, 0, 1)
                h_new = h_new * (n_points / 2) / np.sum(h_new)
                
                # Evaluate
                corr = np.correlate(h_new, 1 - h_new, mode='full')
                c5_new = np.max(corr) * dx
                
                if c5_new < gbest_val:
                    h = h_new
                    gbest_val = c5_new
                    no_improvement = 0
            
            gbest = h.copy()
            
    # Final evaluation
    correlation = np.correlate(gbest, 1 - gbest, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (gbest.tolist(), final_c5, n_points)
