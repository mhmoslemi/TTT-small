import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population with the provided parent or uniform distribution
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill up population
    for _ in range(pop_size - 1):
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    gen_no_improvement = 0
    max_no_improvement = 100
    
    def evaluate_individual(h):
        # Ensure h is within bounds
        h = np.clip(h, 0, 1)
        # Rescale to satisfy sum constraint
        h = h * (n_points / 2) / np.sum(h)
        # Compute the correlation
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    # Function to mutate individual
    def mutate(h):
        h = h.copy()
        # Select points to modify (5% of total)
        n_mutate = max(1, int(n_points * 0.05))
        indices = np.random.choice(n_points, n_mutate, replace=False)
        for i in indices:
            noise = np.random.normal(0, 0.1)
            h[i] += noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    # Function to crossover
    def crossover(h1, h2):
        h1 = h1.copy()
        h2 = h2.copy()
        # Point crossover
        crossover_point = np.random.randint(1, n_points - 1)
        child1 = np.concatenate([h1[:crossover_point], h2[crossover_point:]])
        child2 = np.concatenate([h2[:crossover_point], h1[crossover_point:]])
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        return child1, child2
    
    generation = 0
    while time.time() - start_time < budget_s - 1 and gen_no_improvement < max_no_improvement:
        generation += 1
        # Evaluate fitness
        fitnesses = []
        for h in population:
            c5 = evaluate_individual(h)
            fitnesses.append(c5)
        
        # Update best
        min_fitness_idx = np.argmin(fitnesses)
        if fitnesses[min_fitness_idx] < best_c5:
            best_c5 = fitnesses[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            gen_no_improvement = 0
        else:
            gen_no_improvement += 1
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitnesses)
        sorted_population = [population[i] for i in sorted_indices]
        
        # Keep top 40%
        elite_size = int(0.4 * pop_size)
        elite = sorted_population[:elite_size]
        
        # Generate offspring
        offspring = []
        while len(offspring) < pop_size - elite_size:
            # Select two parents
            parent_indices = np.random.choice(elite_size, 2, replace=False)
            parent1 = elite[parent_indices[0]].copy()
            parent2 = elite[parent_indices[1]].copy()
            # Crossover
            child1, child2 = crossover(parent1, parent2)
            # Mutate
            child1 = mutate(child1)
            child2 = mutate(child2)
            offspring.extend([child1, child2])
            
        # Combine elite and offspring
        population = elite + offspring[:pop_size - elite_size]
        
    # Final check
    final_fitnesses = [evaluate_individual(h) for h in population]
    min_fitness_idx = np.argmin(final_fitnesses)
    if final_fitnesses[min_fitness_idx] < best_c5:
        best_c5 = final_fitnesses[min_fitness_idx]
        best_h = population[min_fitness_idx].copy()
    
    return (best_h.tolist(), best_c5, n_points)
