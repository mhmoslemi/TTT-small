import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Population-based approach using genetic algorithm
    pop_size = 50
    generations = 120
    crossover_rate = 0.5
    mutation_rate = 0.2
    mutation_intensity = 0.1
    elite_size = 5
    
    # Initialize population
    def create_individual():
        if 'initial_h_values' in globals():
            h = np.array(initial_h_values)
            if len(h) != n_points:
                x_orig = np.linspace(0, 2, len(h))
                x_new = np.linspace(0, 2, n_points)
                h = np.interp(x_new, x_orig, h)
        else:
            # Create random valid individual
            h = np.random.rand(n_points)
        # Normalize to sum to n_points/2
        h = h * (n_points / 2) / np.sum(h)
        # Clip to [0,1]
        h = np.clip(h, 0, 1)
        return h
    
    # Initialize population
    population = [create_individual() for _ in range(pop_size)]
    
    def evaluate_individual(h):
        correlation = correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    def tournament_selection(population, fitnesses, k=3):
        indices = np.random.choice(len(population), size=k, replace=False)
        best_idx = indices[np.argmin([fitnesses[i] for i in indices])]
        return population[best_idx]
        
    def crossover(parent1, parent2):
        if np.random.rand() > crossover_rate:
            return parent1.copy(), parent2.copy()
            
        # Single point crossover
        crossover_point = np.random.randint(1, n_points-1)
        child1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
        child2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
        
        # Normalize
        child1 = child1 * (n_points / 2) / np.sum(child1)
        child2 = child2 * (n_points / 2) / np.sum(child2)
        
        # Clip
        child1 = np.clip(child1, 0, 1)
        child2 = np.clip(child2, 0, 1)
        
        return child1, child2
        
    def mutate(h, intensity=0.1):
        h = h.copy()
        # Mutate 20% of points
        n_mutate = int(n_points * mutation_rate)
        indices = np.random.choice(n_points, size=n_mutate, replace=False)
        for i in indices:
            noise = np.random.normal(0, intensity)
            h[i] += noise
        # Rescale
        h = h * (n_points / 2) / np.sum(h)
        # Clip
        h = np.clip(h, 0, 1)
        return h
    
    start_time = time.time()
    best_overall_c5 = float('inf')
    best_overall_h = None
    
    for generation in range(generations):
        # Evaluate fitness
        fitnesses = [evaluate_individual(h) for h in population]
        
        # Update best overall
        min_fitness = min(fitnesses)
        if min_fitness < best_overall_c5:
            best_overall_c5 = min_fitness
            best_idx = np.argmin(fitnesses)
            best_overall_h = population[best_idx].copy()
            
        # Selection and reproduction
        new_population = []
        
        # Elitism
        elite_indices = np.argsort(fitnesses)[:elite_size]
        for i in elite_indices:
            new_population.append(population[i].copy())
        
        # Create rest of population
        while len(new_population) < pop_size:
            parent1 = tournament_selection(population, fitnesses)
            parent2 = tournament_selection(population, fitnesses)
            child1, child2 = crossover(parent1, parent2)
            child1 = mutate(child1, mutation_intensity)
            child2 = mutate(child2, mutation_intensity)
            new_population.extend([child1, child2])
            
        # Trim to exact size
        population = new_population[:pop_size]
        
        # Break if timeout approaching
        if time.time() - start_time > budget_s - 1:
            break
    
    return (best_overall_h.tolist(), best_overall_c5, n_points)
