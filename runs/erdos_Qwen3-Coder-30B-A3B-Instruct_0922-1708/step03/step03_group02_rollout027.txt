import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Binary pattern: 1 in intervals [0.2, 0.4] and [1.2, 1.4]
        i1_start = int(0.2 / 2 * n_points)
        i1_end = int(0.4 / 2 * n_points)
        i2_start = int(1.2 / 2 * n_points)
        i2_end = int(1.4 / 2 * n_points)
        h[i1_start:i1_end] = 1.0
        h[i2_start:i2_end] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    temp = 0.1
    cooling_rate = 0.995
    lr = 0.01
    min_lr = 0.001
    max_lr = 0.02
    perturb_rate = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_sa = h.copy()
        num_perturb = max(1, int(n_points * perturb_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            h_sa[i] += np.random.normal(0, 0.05)
        h_sa = np.clip(h_sa, 0, 1)
        h_sa = h_sa * (n_points / 2) / np.sum(h_sa)
        
        # Evaluate SA candidate
        corr_sa = correlate(h_sa, 1 - h_sa, mode='full')
        c5_sa = np.max(corr_sa) * dx
        
        # Accept or reject based on temperature
        if c5_sa < best_c5 or np.random.rand() < np.exp(-(c5_sa - best_c5) / temp):
            h = h_sa
            c5_current = c5_sa
        else:
            corr_current = correlate(h, 1 - h, mode='full')
            c5_current = np.max(corr_current) * dx
        
        # Gradient Descent Step (with smoothing)
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_current) / eps
        
        # Apply gradient with learning rate and smoothing
        descent = -grad
        h_new = h + lr * descent
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new candidate
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Update if improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            # Increase learning rate
            lr = min(max_lr, lr * 1.01)
        else:
            # Decrease learning rate
            lr = max(min_lr, lr * 0.95)
        
        h = h_new
        
        # Update temperature
        temp *= cooling_rate
    
    # Final evaluation
    final_corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
