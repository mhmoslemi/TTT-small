import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper to get the index of maximum cross-correlation
    def get_max_shift(h_array):
        correlation = correlate(h_array, 1 - h_array, mode='full')
        max_idx = np.argmax(correlation)
        # Adjust index to get the actual shift k
        k = max_idx - (len(h_array) - 1)
        return k
    
    # Helper to compute exact gradient for the max shift
    def compute_exact_gradient(h_array, max_k):
        n = len(h_array)
        grad = np.zeros(n)
        for j in range(n):
            # Gradient for j-th component w.r.t. max_k shift
            # Simple formula from convolution derivative
            # We take forward differences
            prev_idx = (j - max_k) % n
            next_idx = (j + max_k) % n
            grad[j] = dx * (1 - h_array[prev_idx] - h_array[next_idx])
        return grad
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Get the shift that maximizes overlap
        max_k = get_max_shift(h)
        
        # Compute exact gradient for that shift
        grad = compute_exact_gradient(h, max_k)
        
        # Update step with gradient
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
