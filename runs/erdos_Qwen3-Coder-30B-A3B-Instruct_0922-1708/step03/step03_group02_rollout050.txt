import numpy as np
from scipy.signal import correlate
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization: step function with two intervals
        h = np.zeros(n_points)
        left = int(0.2 / 2 * n_points)
        right = int(0.4 / 2 * n_points)
        h[left:right] = 1.0
        left2 = int(1.2 / 2 * n_points)
        right2 = int(1.4 / 2 * n_points)
        h[left2:right2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling factor
    lr = 0.01  # Learning rate
    perturb_rate = 0.1  # Fraction of elements to perturb
    eps = 1e-5  # Finite difference step
    
    start_time = time.time()
    
    def compute_c5(h_vec):
        corr = correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(corr) * dx
    
    def project(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_candidate = h.copy()
        n_perturb = int(perturb_rate * n_points)
        indices = np.random.choice(n_points, n_perturb, replace=False)
        for i in indices:
            h_candidate[i] += np.random.normal(0, 0.05)
        h_candidate = project(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        
        # Accept with probability based on temperature
        if c5_candidate < c5 or np.random.rand() < np.exp(-(c5_candidate - c5) / T):
            h = h_candidate
            c5 = c5_candidate
        
        # Gradient Descent Step
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
        
        # Update h using gradient and adaptive learning rate
        h_new = h - lr * grad
        h_new = project(h_new)
        c5_new = compute_c5(h_new)
        
        # Update learning rate based on improvement
        if c5_new < c5:
            lr = min(lr * 1.01, 0.02)
        else:
            lr = max(lr * 0.95, 0.001)
            
        h = h_new
        c5 = c5_new
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Update temperature
        T *= alpha
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
