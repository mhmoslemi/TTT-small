import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Initialize h using convex relaxation
    h_init = np.ones(n_points) * 0.5
    h = h_init.copy()
    
    # Use CVXPY for convex relaxation (simpler relaxation)
    # Minimize sum(h) - sum(h^2)
    h_cvx = cp.Variable(n_points)
    objective = cp.Minimize(cp.sum(h_cvx) - cp.sum(cp.square(h_cvx)))
    constraints = [
        h_cvx >= 0,
        h_cvx <= 1,
        cp.sum(h_cvx) == n_points / 2
    ]
    
    try:
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        if prob.status == cp.OPTIMAL:
            h = np.array(h_cvx.value).flatten()
            h = np.clip(h, 0, 1)
            # Ensure sum constraint
            h = h * (n_points / 2) / np.sum(h)
    except:
        pass

    iteration = 0
    lr = 0.02
    eps = 1e-6
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        # We need the max correlation index
        correlation = np.correlate(h, 1 - h, mode='full')
        max_k = np.argmax(correlation)
        
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 50 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 100 == 0:
            h = h_init.copy()  # Use the convex relaxation solution as restart
            
    return (best_h.tolist(), best_c5, n_points)
