import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    best_c5 = float('inf')
    best_h = None
    
    # Initialize h from initial_h_values or create a starting point
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a simple construction: block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
    
    start_time = time.time()
    
    # Resolver levels for perturbation scales
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature for simulated annealing
    temperature = 0.1
    temp_decay = 0.99
    min_temp = 1e-6
    
    # Perturbation adaptivity
    last_improvements = []
    success_threshold_high = 0.2
    success_threshold_low = 0.1
    adapt_interval = 50
    
    # Compute initial C5
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    if c5 < best_c5:
        best_c5 = c5
        best_h = h.copy()
    
    iteration = 0
    perturbation_scale = resolvers[0]
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidate by perturbing h
        h_candidate = h + np.random.normal(0, perturbation_scale, n_points)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute overlap for candidate
        correlation = correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(correlation) * dx
        
        # Accept or reject with simulated annealing
        if c5_candidate < best_c5:
            h = h_candidate
            c5 = c5_candidate
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
            
            # Reduce perturbation scale on improvement
            perturbation_scale = max(0.001, perturbation_scale * 0.9)
            last_improvements.append(1)
        else:
            # Accept with probability based on temperature
            delta = c5_candidate - best_c5
            if np.random.rand() < np.exp(-delta / max(temperature, min_temp)):
                h = h_candidate
                c5 = c5_candidate
            last_improvements.append(0)
        
        # Decay temperature
        if iteration % 100 == 0:
            temperature *= temp_decay
            temperature = max(temperature, min_temp)
        
        # Adaptive perturbation scale based on recent success rate
        if iteration % adapt_interval == 0 and len(last_improvements) > 0:
            success_rate = sum(last_improvements) / len(last_improvements)
            if success_rate > success_threshold_high:
                perturbation_scale *= 0.7
            elif success_rate < success_threshold_low:
                perturbation_scale *= 1.5
            perturbation_scale = np.clip(perturbation_scale, 0.001, 1.0)
            last_improvements = []
        
        # Try escalating resolver levels
        if iteration % 2000 == 0:
            # Advance to next resolver if one exists
            resolver_idx = 0
            for i, r in enumerate(resolvers):
                if r == perturbation_scale:
                    resolver_idx = i
                    break
            if resolver_idx + 1 < len(resolvers):
                perturbation_scale = resolvers[resolver_idx + 1]
    
    # Final check with full computation
    if best_h is not None:
        correlation = correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
        return (best_h.tolist(), final_c5, n_points)
    
    # Fallback to initial solution
    correlation = correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    return (h.tolist(), final_c5, n_points)
