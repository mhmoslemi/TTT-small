import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent or a smart construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a smart initial step function
        h = np.zeros(n_points)
        # Try to distribute the mass efficiently
        h[:n_points//2] = 2.0
        
    # Normalize to satisfy integral constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vec):
        # Compute overlap using correlation
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx

    def objective(h_vec):
        return compute_c5(h_vec)

    def callback(h_vec):
        nonlocal best_h, best_c5
        c5 = compute_c5(h_vec)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_vec.copy()

    # Define bounds for each variable
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Define constraint for integral
    def constraint_func(h_vec):
        return np.sum(h_vec) * dx - 1.0
    
    constraints = {'type': 'eq', 'fun': constraint_func}
    
    # Try multiple restarts with different random initializations
    max_restarts = 5
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # 1. Try with current h
        result = minimize(
            objective,
            h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            callback=callback,
            options={'maxiter': 1000, 'ftol': 1e-9, 'gtol': 1e-9}
        )
        
        if result.success:
            c5_result = compute_c5(result.x)
            if c5_result < best_c5:
                best_c5 = c5_result
                best_h = result.x.copy()
        
        # Generate a new random initialization
        h = np.random.uniform(0, 1, n_points)
        h = h * (n_points / 2) / np.sum(h)
        restart_count += 1
        
        # Early exit if we have good solution
        if best_c5 < 0.38085:
            break
    
    # Final refinement with smaller tolerance
    try:
        result = minimize(
            objective,
            best_h.copy(),
            method='L-BFGS-B',
            bounds=bounds,
            constraints=constraints,
            callback=callback,
            options={'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
        )
        
        if result.success:
            c5_final = compute_c5(result.x)
            if c5_final < best_c5:
                best_c5 = c5_final
                best_h = result.x.copy()
    except:
        pass
    
    # Final validation
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
