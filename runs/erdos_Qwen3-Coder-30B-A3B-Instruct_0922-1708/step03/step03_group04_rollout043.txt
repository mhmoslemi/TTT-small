        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx

    def compute_max_grad(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        k_max = np.argmax(correlation)

        grad = np.zeros_like(h_array)
