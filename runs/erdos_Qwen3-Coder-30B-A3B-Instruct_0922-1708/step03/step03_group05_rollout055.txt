import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    max_vel = 0.1
    
    # Initialize PSO
    particles = []
    velocities = []
    p_best = []
    p_best_c5 = []
    
    # Initialize particles
    for i in range(num_particles):
        if i < num_particles // 10 and 'initial_h_values' in globals():
            # Use initial values for some particles
            h_init = h.copy()
        else:
            # Random initialization
            h_init = np.random.uniform(0, 1, n_points)
            h_init = h_init / np.sum(h_init) * (n_points / 2)
        
        particles.append(h_init.copy())
        velocities.append(np.random.uniform(-max_vel, max_vel, n_points))
        p_best.append(h_init.copy())
        p_best_c5.append(float('inf'))
    
    g_best = particles[0].copy()
    g_best_c5 = float('inf')
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate each particle
        for i in range(num_particles):
            h = particles[i]
            
            # Compute the overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = h.copy()
            
            # Update global best
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = h.copy()
        
        # Update particles
        for i in range(num_particles):
            r1, r2 = np.random.random(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = w * v + c1 * r1 * (p_best[i] - h) + c2 * r2 * (g_best - h)
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            
            # Update position
            h_new = h + v
            
            # Project to feasible region
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new.copy()
            velocities[i] = v.copy()
        
        # Update best solution
        if g_best_c5 < best_c5:
            best_c5 = g_best_c5
            best_h = g_best.copy()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
