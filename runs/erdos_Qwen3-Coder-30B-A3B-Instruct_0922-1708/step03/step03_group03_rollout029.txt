import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like pattern
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a Barker-like initialization pattern
        h = np.zeros(n_points)
        # Use a pattern with alternating high/low values
        for i in range(n_points):
            if i % 4 < 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
        # Normalize to meet integral constraint
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute useful values for overlap computation
    def compute_c5(h_vals):
        # Compute overlap integral
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_overlap = np.max(correlation)
        return max_overlap * dx
    
    # Gradient computation
    def compute_gradient(h_vals, eps=1e-6):
        grad = np.zeros_like(h_vals)
        base_c5 = compute_c5(h_vals)
        for i in range(len(h_vals)):
            h_plus = h_vals.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    max_iter = 5000
    iteration = 0
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute current overlap
        current_c5 = compute_c5(h)
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Compute gradient
        grad = compute_gradient(h)
        
        # Update using gradient descent
        lr = 0.01
        h_new = h - lr * grad
        
        # Apply bounds and normalization
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check for significant change
        if np.allclose(h, h_new, rtol=1e-8):
            break
            
        h = h_new
        
    # Final computation for best solution
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
