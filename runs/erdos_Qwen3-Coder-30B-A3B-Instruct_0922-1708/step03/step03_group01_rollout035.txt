import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h"""
    # Compute overlap integral
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * (2.0 / len(h))
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize h to satisfy constraint
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    start_time = time.time()
    
    # Hash function for solutions
    def hash_solution(vec):
        return hashlib.md5(vec.tobytes()).hexdigest()
    
    # Perturb function
    def perturb(h, step_size):
        h_new = h.copy()
        # Select indices to perturb (20% of points)
        indices = np.random.choice(n_points, size=int(0.2 * n_points), replace=False)
        
        # Apply perturbations
        for i in indices:
            delta = np.random.uniform(-step_size, step_size)
            h_new[i] += delta
        
        # Clip and renormalize
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        return h_new
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidates
        candidates = []
        num_candidates = 50
        
        for _ in range(num_candidates):
            candidate_h = perturb(h, step_size)
            key = hash_solution(candidate_h)
            
            # Only consider candidates not in TABU list
            if key not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h, c5))
        
        if not candidates:
            # No new candidates, just perturb current solution and add to TABU
            h_new = perturb(h, step_size)
            h = h_new
            tabu_list.add(hash_solution(h))
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
            
            stagnation_count += 1
            if stagnation_count > max_stagnation:
                step_size *= 1.2
                tabu_list.clear()
                stagnation_count = 0
            continue
        
        # Find best candidate
        best_candidate = min(candidates, key=lambda x: x[1])
        candidate_h, candidate_c5 = best_candidate
        
        # Update best
        if candidate_c5 < best_c5:
            best_h = candidate_h.copy()
            best_c5 = candidate_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update TABU list
        tabu_list.add(hash_solution(h))
        if len(tabu_list) > tabu_length:
            tabu_list.pop()
            
        # Update current solution
        h = candidate_h.copy()
        
        # Adaptive step size
        if stagnation_count > max_stagnation:
            step_size *= 1.2
            tabu_list.clear()
            stagnation_count = 0
        elif stagnation_count % 20 == 0 and step_size > 0.01:
            step_size *= 0.95
            stagnation_count = 0
    
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
