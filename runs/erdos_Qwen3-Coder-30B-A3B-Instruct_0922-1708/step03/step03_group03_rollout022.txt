import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize to sum = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Resolver schedule
    resolvers = [0.1, 0.05, 0.02, 0.01, 0.005, 0.002, 0.001]
    
    # Temperature schedule
    temperature = 0.1
    temp_decay = 0.99
    temp_min = 1e-6
    
    # Track recent acceptance
    recent_accepts = []
    success_threshold_high = 0.2
    success_threshold_low = 0.1
    
    start_time = time.time()
    total_iterations = 0
    
    for resolver in resolvers:
        num_iterations = 1000
        perturbation_scale = resolver
        
        for _ in range(num_iterations):
            if time.time() - start_time > budget_s - 1:
                break
                
            total_iterations += 1
            
            # Generate perturbation
            delta = np.random.normal(0, perturbation_scale, n_points)
            h_candidate = h + delta
            
            # Clip to [0,1]
            h_candidate = np.clip(h_candidate, 0, 1)
            
            # Rescale to sum = n_points / 2
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Compute overlap
            # Direct computation avoids circular shifts
            overlap_scores = []
            for k in range(1, n_points):
                # Compute sum_{a=0}^{n_points - k - 1} h[a] * (1 - h[a + k])
                score = 0
                for a in range(n_points - k):
                    score += h_candidate[a] * (1 - h_candidate[a + k])
                overlap_scores.append(score * dx)
            
            c5_candidate = max(overlap_scores)
            
            # Accept or reject
            if c5_candidate < best_c5:
                best_c5 = c5_candidate
                best_h = h_candidate.copy()
                h = h_candidate.copy()
                recent_accepts.append(1)
            else:
                # Metropolis acceptance
                if temperature > temp_min:
                    prob = np.exp(-(c5_candidate - best_c5) / temperature)
                    if np.random.random() < prob:
                        h = h_candidate.copy()
                        recent_accepts.append(1)
                    else:
                        recent_accepts.append(0)
                else:
                    recent_accepts.append(0)
            
            # Update temperature
            if total_iterations % 100 == 0:
                temperature *= temp_decay
                temperature = max(temperature, temp_min)
            
            # Adapt perturbation scale based on recent success
            if len(recent_accepts) > 50:
                recent_accepts = recent_accepts[-50:]
                success_ratio = sum(recent_accepts) / len(recent_accepts)
                if success_ratio > success_threshold_high:
                    perturbation_scale *= 1.5
                elif success_ratio < success_threshold_low:
                    perturbation_scale *= 0.5
                perturbation_scale = max(0.001, min(1.0, perturbation_scale))
    
    # Final check
    overlap_scores = []
    for k in range(1, n_points):
        score = 0
        for a in range(n_points - k):
            score += best_h[a] * (1 - best_h[a + k])
        overlap_scores.append(score * dx)
    
    final_c5 = max(overlap_scores)
    
    return (best_h.tolist(), final_c5, n_points)
