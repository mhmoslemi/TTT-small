import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute correlation using numpy's correlate function
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        # Extract the valid range of correlations (excluding padding)
        mid = len(h_vals) - 1
        corr_valid = corr[mid:]
        c5 = np.max(corr_valid) * dx
        return c5, corr
    
    def objective_and_gradient(h_vals):
        c5, corr = compute_c5(h_vals)
        # Find which shift gives the maximum overlap
        mid = len(h_vals) - 1
        corr_valid = corr[mid:]
        max_idx = np.argmax(corr_valid)
        
        # Compute analytical gradient
        grad = np.zeros_like(h_vals)
        # Gradient for the maximal correlation term
        for i in range(len(h_vals)):
            # For shift k = max_idx, we look at term h[i] * (1 - h[i+k])
            k = max_idx
            if i + k < len(h_vals):
                grad[i] += dx * (-h_vals[i + k])
            if i - k >= 0 and i != 0:
                grad[i] += dx * (-h_vals[i - k])
        
        # Add penalty for sum constraint
        penalty_grad = np.ones_like(h_vals) * (2.0 / n_points)
        grad += penalty_grad
        
        return c5, grad
    
    def constraint_sum(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    def constraint_bounds(i):
        return lambda h_vals: h_vals[i]
    
    # Setup constraints
    cons = [{'type': 'eq', 'fun': constraint_sum}]
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Optimization loop
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute C5 and gradient
        c5, grad = objective_and_gradient(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use scipy minimize with SLSQP
        try:
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h,
                method='SLSQP',
                jac=lambda x: objective_and_gradient(x)[1],
                bounds=bounds,
                constraints=cons,
                options={'maxiter': 10, 'ftol': 1e-8}
            )
            if result.success:
                h = result.x
                # Ensure the sum constraint is met
                h = h * (n_points / 2) / np.sum(h)
                h = np.clip(h, 0, 1)
        except Exception:
            # If optimization fails, fall back to simple gradient step
            step_size = 0.001
            h_new = h - step_size * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    _, final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
