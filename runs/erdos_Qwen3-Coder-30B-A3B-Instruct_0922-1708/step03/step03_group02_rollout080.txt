import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # The objective is to minimize max_k int h(x)(1-h(x+k)) dx
    # This is equivalent to maximizing min_k int h(x)h(x+k) dx
    
    # Constraint: sum(h)*dx = 1 => sum(h) = n_points/2
    # We try to optimize using both local optimization and gradient-based approach
    def get_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    # Run local search optimization with different strategies
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        # Local search on current solution
        c5 = get_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use scipy.optimize with constraints to do better local optimization
        # Simple gradient step to improve the solution
        eps = 1e-5
        
        # Try to compute a gradient with finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)  # Keep in [0,1]
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)  # Rescale to meet constraint
            
            c5_plus = get_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient descent step
        lr = 0.01
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        iteration += 1

    # Final evaluation
    final_c5 = get_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
