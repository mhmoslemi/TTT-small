import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization
        h = np.zeros(n_points)
        # Place peaks in regions [0.2, 0.4] and [1.2, 1.4]
        i1_start = int(0.2 / 2.0 * n_points)
        i1_end = int(0.4 / 2.0 * n_points)
        i2_start = int(1.2 / 2.0 * n_points)
        i2_end = int(1.4 / 2.0 * n_points)
        h[i1_start:i1_end] = 1.0
        h[i2_start:i2_end] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Parameters
    T = 0.1  # Initial temperature
    alpha = 0.995  # Cooling rate
    lr = 0.01  # Initial learning rate
    eps = 1e-5   # Gradient epsilon
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Step 1: Simulated Annealing Perturbation
        h_trial = h.copy()
        num_perturb = max(1, int(0.1 * n_points))
        indices = np.random.choice(n_points, size=num_perturb, replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_trial[i] += delta
        
        # Project to constraints
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5
        correlation_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(correlation_trial) * dx
        
        # Accept or reject with SA
        if c5_trial < best_c5 or np.random.rand() < math.exp(-(c5_trial - best_c5) / T):
            h = h_trial.copy()
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h.copy()
        
        # Step 2: Gradient descent for local refinement
        grad = np.zeros_like(h)
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5_trial) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        
        # Project back to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5 for refined step
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Adaptive learning rate
        if c5_new < best_c5:
            lr *= 1.01
        else:
            lr *= 0.95
        
        lr = np.clip(lr, 0.001, 0.02)
        
        h = h_new.copy()
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h.copy()
        
        # Update temperature
        T *= alpha
        
        # Ensure we're still in a reasonable time
        if time.time() - start_time > budget_s - 0.1:
            break
    
    # Final evaluation
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
