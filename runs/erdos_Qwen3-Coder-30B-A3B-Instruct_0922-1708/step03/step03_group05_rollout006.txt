import numpy as np
import time
from scipy.optimize import minimize
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    num_particles = 50
    w = 0.7
    c1 = c2 = 1.5
    v_max = 0.1
    
    # Initialize swarm
    particles = []
    velocities = []
    
    # Use initial solution for some particles
    num_seeded = max(1, num_particles // 10)
    for i in range(num_seeded):
        particles.append(h.copy())
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Initialize other particles randomly
    for i in range(num_seeded, num_particles):
        h_init = np.random.uniform(0, 1, n_points)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        
    # Keep track of personal bests
    p_best = [p.copy() for p in particles]
    p_best_c5 = [float('inf')] * num_particles
    g_best = best_h.copy()
    g_best_c5 = best_c5
    
    # Precompute all lags for efficiency
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = h.copy()
                
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = h.copy()
                
        # Update velocities and positions
        for i in range(num_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = w * v + c1 * r1 * (p_best[i] - h) + c2 * r2 * (g_best - h)
            # Clip velocity
            v = np.clip(v, -v_max, v_max)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            # Clip to [0,1]
            h_new = np.clip(h_new, 0, 1)
            # Rescale to meet constraint
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            particles[i] = h_new
            
        # Update best if needed
        if g_best_c5 < best_c5:
            best_c5 = g_best_c5
            best_h = g_best.copy()
    
    # Make sure we return the best result within the time limit
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
