import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add random particles
    for _ in range(n_particles - len(particles)):
        h = np.random.uniform(0.2, 0.8, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Personal best positions and their fitness
    personal_best = [p.copy() for p in particles]
    personal_best_fitness = [float('inf')] * n_particles
    
    # Global best
    global_best = particles[0].copy()
    global_best_fitness = float('inf')
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            fitness = np.max(correlation) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best_fitness[i] = fitness
                personal_best[i] = h.copy()
                
                # Update global best
                if fitness < global_best_fitness:
                    global_best_fitness = fitness
                    global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            h = particles[i]
            v = velocities[i]
            
            # Update velocity
            v = w * v + c1 * r1 * (personal_best[i] - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            h = h + v
            particles[i] = h
            
            # Rescale to meet constraint
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
        
        # Adapt inertia weight
        if step % 100 == 0:
            w = max(0.8, w * 0.99)
            
        # Periodic reset to maintain diversity
        if step % 100 == 0:
            for i in range(n_particles):
                # Randomly reset some particles
                if np.random.rand() < 0.1:
                    h = np.random.uniform(0.2, 0.8, n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h
    
    # Final evaluation
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
