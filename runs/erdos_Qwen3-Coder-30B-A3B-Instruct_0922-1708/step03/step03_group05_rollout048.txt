import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 1000
    
    # Initialize h
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Constraint: sum(h) * dx = 1, which means sum(h) = n_points / 2
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Temperature schedule
    T0 = 0.1
    alpha = 0.99
    T = T0
    
    # Iteration counter
    iteration = 0
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations * 2:
        iteration += 1
        
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Update best solution
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h.copy()
        
        # Select random subset of points to modify
        n_modify = max(1, int(0.05 * n_points))
        indices = np.random.choice(n_points, size=n_modify, replace=False)
        
        # Adjust selected points
        for i in indices:
            # Add noise with adaptive magnitude based on temperature
            delta = np.random.uniform(-0.1, 0.1) * T
            h_new[i] += delta
        
        # Clamp values to [0, 1]
        h_new = np.clip(h_new, 0, 1)
        
        # Rescale to satisfy constraint
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        new_c5 = np.max(corr_new) * dx
        
        # Acceptance probability
        if new_c5 < current_c5 or np.random.rand() < np.exp((current_c5 - new_c5) / T):
            h = h_new
        
        # Update temperature
        T = max(1e-6, T * alpha)
    
    # Final check and refine
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
