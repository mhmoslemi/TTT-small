import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing approach combined with Nelder-Mead
    T = 0.1
    cooling_rate = 0.995
    perturbation_magnitude = 0.1
    
    # Nelder-Mead parameters
    simplex_size = 5
    max_simplex_iterations = 1000
    
    # Track best solution found
    current_h = h.copy()
    
    # First phase: Simulated Annealing
    sa_steps = 0
    while sa_steps < 2000 and time.time() - start_time < budget_s - 5:
        # Perturb some elements
        perturb_indices = np.random.choice(n_points, size=max(1, n_points // 10), replace=False)
        h_trial = current_h.copy()
        for i in perturb_indices:
            delta = np.random.normal(0, perturbation_magnitude)
            h_trial[i] += delta
        h_trial = np.clip(h_trial, 0, 1)
        
        # Normalize
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Evaluate
        correlation = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(correlation) * dx
        
        # Accept probabilistically
        if c5_trial < best_c5 or np.random.rand() < np.exp(-(c5_trial - best_c5) / T):
            current_h = h_trial.copy()
            if c5_trial < best_c5:
                best_c5 = c5_trial
                best_h = h_trial.copy()
        
        T *= cooling_rate
        sa_steps += 1
        
        # Occasionally check if we're stuck and restart if needed
        if sa_steps % 500 == 0 and best_c5 < 0.4:
            T = max(0.01, T * 0.8)
    
    # Second phase: Local optimization with Nelder-Mead
    # Creating an initial simplex
    def nelder_mead_optimization(h_start, max_iter=1000):
        # Generate initial simplex
        simplex = [h_start.copy()]
        for _ in range(simplex_size - 1):
            h_pert = h_start.copy()
            for i in range(n_points):
                if np.random.rand() < 0.1:
                    h_pert[i] = np.clip(h_pert[i] + np.random.normal(0, 0.05), 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
            simplex.append(h_pert)
        
        for _ in range(max_iter):
            if time.time() - start_time > budget_s - 2:
                break
                
            # Evaluate function at each point
            values = []
            for h_point in simplex:
                correlation = np.correlate(h_point, 1-h_point, mode='full')
                c5 = np.max(correlation) * dx
                values.append(c5)
            
            # Sort indices by function values
            sorted_indices = np.argsort(values)
            simplex = [simplex[i] for i in sorted_indices]
            values = [values[i] for i in sorted_indices]
            
            # Compute centroid (excluding worst point)
            centroid = np.mean(simplex[:-1], axis=0)
            centroid = np.clip(centroid, 0, 1)
            centroid = centroid * (n_points / 2) / np.sum(centroid)
            
            # Reflection
            reflection = centroid + (centroid - simplex[-1])
            reflection = np.clip(reflection, 0, 1)
            reflection = reflection * (n_points / 2) / np.sum(reflection)
            
            ref_value = np.max(np.correlate(reflection, 1-reflection, mode='full')) * dx
            
            if values[0] <= ref_value < values[-2]:
                simplex[-1] = reflection
            elif ref_value < values[0]:
                # Expansion
                expansion = centroid + 2 * (reflection - centroid)
                expansion = np.clip(expansion, 0, 1)
                expansion = expansion * (n_points / 2) / np.sum(expansion)
                
                exp_value = np.max(np.correlate(expansion, 1-expansion, mode='full')) * dx
                
                if exp_value < ref_value:
                    simplex[-1] = expansion
                else:
                    simplex[-1] = reflection
            else:
                # Contraction
                contraction = centroid + 0.5 * (simplex[-1] - centroid)
                contraction = np.clip(contraction, 0, 1)
                contraction = contraction * (n_points / 2) / np.sum(contraction)
                
                con_value = np.max(np.correlate(contraction, 1-contraction, mode='full')) * dx
                
                if con_value < values[-1]:
                    simplex[-1] = contraction
                else:
                    # Shrink simplex
                    for i in range(1, len(simplex)):
                        simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                        simplex[i] = np.clip(simplex[i], 0, 1)
                        simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
            
            # Update best if we found a better solution
            best_simplex_value = np.min([np.max(np.correlate(h, 1-h, mode='full')) * dx 
                                         for h in simplex])
            if best_simplex_value < best_c5:
                best_c5 = best_simplex_value
                # Find the corresponding h
                for h in simplex:
                    potential_value = np.max(np.correlate(h, 1-h, mode='full')) * dx
                    if abs(potential_value - best_c5) < 1e-8:
                        best_h = h.copy()
                        break
        return best_h, best_c5
    
    # Run Nelder-Mead from current best
    try:
        current_h, current_c5 = nelder_mead_optimization(current_h)
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = current_h.copy()
    except Exception:
        # If simplex fails, just continue with current best
        pass
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
