import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for the minimum C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    max_iterations = 100
    
    while (high - low > tolerance) and (time.time() - start_time < budget_s - 1):
        C5_mid = (low + high) / 2.0
        if time.time() - start_time > budget_s - 2:
            break
            
        # Define the QCQP problem
        h_var = cp.Variable(n_points)
        
        # Objective: minimize C5
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1
        ]
        
        # Add overlap constraints for all shifts k
        for k in range(n_points):
            # Compute the correlation for shift k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= C5_mid / dx)
        
        # Minimize C5 (this is just to satisfy the formulation,
        # actually we're checking feasibility of the constraints)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL or prob.status == cp.OPTIMAL_INACCURATE:
                h_new = h_var.value
                if h_new is not None:
                    h_new = np.clip(h_new, 0, 1)
                    h_new = h_new * (n_points / 2) / np.sum(h_new)
                    
                    # Compute C5 for this h
                    correlation = np.correlate(h_new, 1 - h_new, mode='full')
                    c5_new = np.max(correlation) * dx
                    
                    if c5_new < best_c5:
                        best_c5 = c5_new
                        best_h = h_new.copy()
                        
                    high = C5_mid
                else:
                    low = C5_mid
            else:
                low = C5_mid
        except:
            low = C5_mid

    # Final check with official evaluator if needed
    try:
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        final_c5 = np.max(correlation) * dx
        if final_c5 < best_c5:
            best_c5 = final_c5
    except:
        pass

    return (best_h.tolist(), best_c5, n_points)
