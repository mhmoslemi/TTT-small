import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    num_particles = 50
    particles = []
    velocities = []
    
    # If initial construction exists, use it to seed some particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        for i in range(num_particles // 5):
            particles.append(h_init.copy())
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    else:
        for i in range(num_particles // 5):
            particles.append(np.ones(n_points) * 0.5)
            velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Fill remaining particles with random valid constructions
    for i in range(num_particles - len(particles)):
        h = np.random.random(n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Convert to arrays for vectorized operations
    particles = np.array(particles)
    velocities = np.array(velocities)
    
    # PSO parameters
    w = 0.7
    c1 = c2 = 1.5
    
    # Track best solutions
    p_best = particles.copy()
    p_best_c5 = np.full(num_particles, float('inf'))
    g_best = None
    g_best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Evaluate all particles
        for i in range(num_particles):
            h = particles[i]
            # Compute C5 using the given correlation approach
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            if c5 < p_best_c5[i]:
                p_best_c5[i] = c5
                p_best[i] = h.copy()
            if c5 < g_best_c5:
                g_best_c5 = c5
                g_best = h.copy()
        
        # Update velocities and positions
        for i in range(num_particles):
            r1 = np.random.random(n_points)
            r2 = np.random.random(n_points)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Return best solution
    return (g_best.tolist(), g_best_c5, n_points)
