import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 200
    shift_samples = 200
    shift_discretization = np.linspace(0, 2, shift_samples)
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Convex optimization approach
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        # Compute all correlations
        correlations = []
        for k in shift_discretization:
            # Convolution with shifted version of (1-h)
            shifted_h = np.roll(1 - h, int(k / dx))
            corr = np.sum(h * shifted_h) * dx
            correlations.append(corr)
        
        c5 = np.max(correlations)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Add a constraint and solve a new optimization problem
        h_var = cp.Variable(n_points)
        C = cp.Variable()
        
        # Convex constraints based on the maximum overlap
        constraints = [0 <= h_var, h_var <= 1]
        constraints.append(cp.sum(h_var) == n_points / 2)
        
        # Add the constraint from the current max overlap
        max_corr_idx = np.argmax(correlations)
        max_k = shift_discretization[max_corr_idx]
        k_index = int(max_k / dx)
        
        # Shifted version of 1-h
        shifted_ones_minus_h = np.roll(1 - h, k_index)
        # Constraint for this particular shift
        corr_constraint = cp.sum(cp.multiply(h_var, shifted_ones_minus_h)) <= C
        constraints.append(corr_constraint)
        
        # Value of C is the max overlap from previous iteration (upper bound)
        objective = cp.Minimize(C)
        prob = cp.Problem(objective, constraints)
        prob.solve(solver=cp.ECOS, verbose=False)
        
        if prob.status == cp.OPTIMAL:
            h_new = np.array(h_var.value).flatten()
            h = np.clip(h_new, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
        else:
            # Fall back to gradient descent if convex problem fails
            eps = 1e-4
            grad = np.zeros_like(h)
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                grad[i] = (c5_plus - c5) / eps
            
            lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
        # Occasionally try a small random perturbation to escape local optima
        if iteration % 50 == 0:
            h = np.clip(h + np.random.normal(0, 0.001, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        iteration += 1

    return (best_h.tolist(), best_c5, n_points)
