import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured construction
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured starting point: piecewise constant with sharp transitions
        h = np.zeros(n_points)
        # Place high values in regions where overlap is minimized
        start1, end1 = int(0.2 * n_points), int(0.4 * n_points)
        start2, end2 = int(1.2 * n_points), int(1.4 * n_points)
        h[start1:end1] = 1.0
        h[start2:end2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    lr = 0.01
    eps = 1e-5
    
    start_time = time.time()
    
    # Initialize state tracking
    last_improvement = 0
    
    while time.time() - start_time < budget_s - 1:
        # Step 1: Simulated Annealing Perturbation
        h_temp = h.copy()
        # Randomly perturb 10% of elements
        perturb_indices = np.random.choice(n_points, size=int(n_points * 0.1), replace=False)
        for i in perturb_indices:
            h_temp[i] += np.random.normal(0, 0.05)  # Perturb by small amount
        
        # Clip and rescale
        h_temp = np.clip(h_temp, 0, 1)
        h_temp = h_temp * (n_points / 2) / np.sum(h_temp)
        
        # Compute C5 for perturbed solution
        correlation = np.correlate(h_temp, 1 - h_temp, mode='full')
        c5_temp = np.max(correlation) * dx
        
        # Metropolis acceptance probability
        if c5_temp < best_c5 or np.random.rand() < math.exp((best_c5 - c5_temp) / T):
            h = h_temp
            c5 = c5_temp
        else:
            c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        
        # Step 2: Gradient-based Refinement
        # Compute gradient via finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
        
        # Gradient descent update
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Check new C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Update if improvement
        if c5_new < c5:
            h = h_new
            c5 = c5_new
        
        # Step 3: Adaptive Learning Rate
        if c5 < best_c5:
            lr *= 1.01
            last_improvement = 0
        else:
            lr *= 0.95
            last_improvement += 1
        
        lr = max(0.001, min(0.02, lr))
        
        # Step 4: Update Temperature
        T *= alpha
        
        # Update best solution
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Early stopping for stagnation
        if last_improvement > 200:
            break
    
    # Final confirmation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
