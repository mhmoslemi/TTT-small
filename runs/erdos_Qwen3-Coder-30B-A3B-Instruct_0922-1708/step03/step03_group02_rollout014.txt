import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        particles.append(h_init / np.sum(h_init) * (n_points / 2))
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Add random particles
    for _ in range(n_particles - len(particles)):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h / np.sum(h) * (n_points / 2)
        particles.append(h)
    
    velocities = [np.random.uniform(-0.1, 0.1, n_points) for _ in range(n_particles)]
    
    # PSO parameters
    w = 1.0
    c1 = c2 = 2.0
    max_vel = 0.2
    
    # Track best positions
    personal_best = [particles[i].copy() for i in range(n_particles)]
    personal_best_scores = [float('inf')] * n_particles
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # Optimization loop
    start_time = time.time()
    step = 0
    stagnation_count = 0
    max_stagnation = 200
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h / np.sum(h) * (n_points / 2)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
                
                # Update global best
                if c5 < global_best_score:
                    global_best_score = c5
                    global_best = h.copy()
                    stagnation_count = 0
        
        # Update velocities and positions
        for i in range(n_particles):
            if i == 0:  # Skip first particle (it's the global best or from parent)
                continue
                
            r1, r2 = np.random.rand(2)
            
            # Velocity update
            vel_update = (w * velocities[i] + 
                         c1 * r1 * (personal_best[i] - particles[i]) + 
                         c2 * r2 * (global_best - particles[i]))
            
            # Clip velocity
            vel_update = np.clip(vel_update, -max_vel, max_vel)
            velocities[i] = vel_update
            
            # Position update
            particles[i] = particles[i] + velocities[i]
            
            # Ensure that position is valid (clipping and rescaling)
            particles[i] = np.clip(particles[i], 0, 1)
            particles[i] = particles[i] / np.sum(particles[i]) * (n_points / 2)
        
        # Adaptation mechanisms
        if step % 50 == 0:
            w = max(0.8, w - 0.001)  # Decrease inertia over time
        
        if step % 50 == 0 and step > 0:
            # Check for stagnation
            if abs(global_best_score - global_best_score) < 1e-8:
                stagnation_count += 1
            else:
                stagnation_count = 0
                
        if stagnation_count > max_stagnation or step % 200 == 0:
            # Reinitialize some particles to maintain diversity
            for i in range(n_particles // 5):
                r = np.random.randint(0, n_particles)
                h = np.random.uniform(0.4, 0.6, n_points)
                h = h / np.sum(h) * (n_points / 2)
                particles[r] = h
                velocities[r] = np.random.uniform(-0.1, 0.1, n_points)
                
        if step % 100 == 0:
            # Periodic check for new best
            for i in range(n_particles):
                h = particles[i]
                correlation = np.correlate(h, 1 - h, mode='full')
                c5 = np.max(correlation) * dx
                
                if c5 < global_best_score:
                    global_best_score = c5
                    global_best = h.copy()

    # Final validation and return
    final_h = global_best.copy()
    final_correlation = np.correlate(final_h, 1 - final_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    return (final_h.tolist(), final_c5, n_points)
