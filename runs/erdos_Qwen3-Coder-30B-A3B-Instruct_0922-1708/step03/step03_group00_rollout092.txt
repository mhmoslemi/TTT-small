import numpy as np
import time
from scipy.optimize import minimize
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Initialize with parent construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
    else:
        n_points = 100
        h = np.ones(n_points) * 0.5
    
    # Ensure correct normalization
    dx = 2.0 / n_points
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    
    # Annealing iterations
    sa_iters = 500
    for i in range(sa_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        corr = correlate(h, 1 - h, mode='full')
        c5 = np.max(corr) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept with probability
        if c5_new < c5 or np.random.rand() < np.exp(-(c5_new - c5) / T):
            h = h_new
        
        # Cool down
        T *= alpha
        
        # Occasionally perturb globally
        if i % 30 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
    
    # Gradient-based refinement
    max_refine_iters = 300
    lr = 0.001
    
    for i in range(max_refine_iters):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current C5
        corr = correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(corr) * dx
        
        # Estimate gradient
        grad = np.zeros_like(best_h)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Update
        best_h = best_h - lr * grad
        best_h = np.clip(best_h, 0, 1)
        best_h = best_h * (n_points / 2) / np.sum(best_h)
        
        # Adaptive learning rate
        if i % 50 == 0:
            lr *= 0.95
            
        if i % 100 == 0:
            # Early stopping if improved very little
            corr = correlate(best_h, 1 - best_h, mode='full')
            new_c5 = np.max(corr) * dx
            if abs(new_c5 - c5) < 1e-8:
                break
    
    # Final evaluation
    corr_final = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr_final) * dx
    
    return (best_h.tolist(), final_c5, n_points)
