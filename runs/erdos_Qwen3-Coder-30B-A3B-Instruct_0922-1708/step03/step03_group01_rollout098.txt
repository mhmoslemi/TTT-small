import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Frank-Zadoff-like sequence - minimal cross-correlation
        h = np.sin(np.pi * np.arange(n_points) * (np.arange(n_points) + 1) / n_points)
        h = (h + 1) / 2  # Scale to [0,1]
        h = h * (n_points / 2) / np.sum(h)
        
    best_h = h.copy()
    best_c5 = float('inf')
    start_time = time.time()
    iteration = 0
    alpha = 0.01
    max_iter = 5000
    
    # Phase 1: Binary search for feasibility
    low = 0.38
    high = 0.4
    tol = 1e-4
    max_feasible = high
    
    # Binary search parameters
    max_iterations = 20
    binary_iter = 0
    
    # Phase 1: Feasibility check using QCQP
    while binary_iter < max_iterations and (high - low) > tol and time.time() - start_time < budget_s - 10:
        mid = (low + high) / 2
        # Build QCQP constraints to check if such h exists
        h_vars = cp.Variable(n_points)
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # For finite support, we compute shift correctly
            shifted = cp.roll(h_vars, k)
            overlap = cp.sum(cp.multiply(h_vars, 1 - shifted))
            constraints.append(overlap <= mid / dx)
            
        objective = cp.Minimize(0)  # Feasibility problem
        prob = cp.Problem(objective, constraints)
        
        try:
            # Try to solve with solver
            prob.solve(solver=cp.MOSEK, verbose=False)
            # If successful, feasible
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                max_feasible = mid
                high = mid
            else:
                low = mid
        except:
            low = mid
        binary_iter += 1
        
    # Use the closest feasible found so far
    h = best_h.copy() if best_c5 < float('inf') else h
    # Refinement phase: Gradient descent
    while iteration < max_iter and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap integral
        c5_vals = []
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            val = np.dot(h, shifted)
            c5_vals.append(val)
        c5 = max(c5_vals) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Find current maximum overlap shift
        max_k_index = np.argmax(c5_vals)
        
        # Compute analytical gradient for max overlap shift
        grad = np.zeros_like(h)
        k = max_k_index
        for i in range(n_points):
            # Gradient of h[i] in the overlap at shift k
            shift_val = h[(i + k) % n_points]
            grad[i] = dx * (1 - 2 * h[(i + k) % n_points]) if i < n_points - k else dx * (1 - 2 * h[i - n_points + k])
            
        # Update with gradient descent
        h_new = h - alpha * grad
        
        # Clip to valid range
        h_new = np.clip(h_new, 0, 1)
        
        # Re-normalize
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            alpha *= 0.95
            
    # Final downsampling to 100 points
    if n_points > 100:
        h_ds = np.zeros(100)
        for i in range(100):
            start = i * 2
            end = start + 2
            h_ds[i] = np.mean(h[start:end])
        h_ds = h_ds * (100 / 2) / np.sum(h_ds)
        h = h_ds
        n_points = 100
        
        # Recompute best_c5
        c5_vals = []
        for k in range(n_points):
            shifted = np.roll(1 - h, k)
            val = np.dot(h, shifted)
            c5_vals.append(val)
        best_c5 = max(c5_vals) * (2.0 / n_points)
    
    return (best_h.tolist(), best_c5, n_points)
