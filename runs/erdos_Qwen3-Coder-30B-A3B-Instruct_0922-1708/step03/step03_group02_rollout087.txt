import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    # Use initial_h_values if available, otherwise create random particles
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        particles.append(h_init / np.sum(h_init) * (n_points / 2))
    else:
        particles.append(np.ones(n_points) * 0.5)
    
    # Initialize remaining particles with varied starting points
    for i in range(1, n_particles):
        if 'initial_h_values' in globals():
            # Add perturbation to initial construction
            pert = np.random.normal(0, 0.05, n_points)
            h_pert = particles[0] + pert
            h_pert = np.clip(h_pert, 0, 1)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        else:
            # Random initialization
            h_pert = np.random.uniform(0.3, 0.7, n_points)
            h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        particles.append(h_pert)
    
    # Initialize velocities
    for i in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Track best particles
    particle_best = [particles[i].copy() for i in range(n_particles)]
    particle_best_scores = [float('inf')] * n_particles
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    omega = 1.0
    c1 = c2 = 2.0
    velocity_max = 0.2
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        # Update particles
        for i in range(n_particles):
            h = particles[i]
            
            # Clip and rescale
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Evaluate
            correlation = np.correlate(h, 1 - h, mode='full')
            score = np.max(correlation) * dx
            
            # Update personal best
            if score < particle_best_scores[i]:
                particle_best_scores[i] = score
                particle_best[i] = h.copy()
            
            # Update global best
            if score < global_best_score:
                global_best_score = score
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(), np.random.rand()
            v = velocities[i]
            h = particles[i]
            p_best = particle_best[i]
            
            # Update velocity
            v = omega * v + c1 * r1 * (p_best - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -velocity_max, velocity_max)
            velocities[i] = v
            
            # Update position
            h = h + v
            particles[i] = h
            
            # Rescale for sum constraint
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
        
        step += 1
        
        # Periodic reset to maintain diversity
        if step % 100 == 0 and step > 0:
            for i in range(n_particles):
                if np.random.rand() < 0.1:  # 10% chance
                    particles[i] = np.random.uniform(0.3, 0.7, n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Adjust hyperparameters gradually
        if step % 1000 == 0 and step > 0:
            omega = max(0.8, omega - 0.001)
    
    # Final evaluation
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
