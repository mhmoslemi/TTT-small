import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():

            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # The objective is to minimize max_k int h(x)(1-h(x+k)) dx
    # This is equivalent to maximizing min_k int h(x)h(x+k) dx
    
    # Constraint: sum(h)*dx = 1 => sum(h) = n_points/2
    
    # Use a more robust approach: try different initialization strategies and local search variations
    def compute_c5(h_vals):
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def objective(h_vals):
        return compute_c5(h_vals)
    
    # Try different strategies
    strategies = [
        # Strategy 1: Gradient descent with adaptive learning rate
        lambda h_vals: _adaptive_gradient_descent(h_vals, dx, n_points, start_time, budget_s),
        # Strategy 2: Local optimization with bounds
        lambda h_vals: _local_optimization(h_vals, dx, n_points, start_time, budget_s)
    ]
    
    # Start with ensuring feasible initial point
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    # Normalize the starting point
    h = h / np.sum(h) * (n_points / 2)
    
    for i, strategy in enumerate(strategies):
        if time.time() - start_time > budget_s - 1:

            best_c5 = final_c5
            best_h = final_h.copy()
    
    return (best_h.tolist(), best_c5, n_points)

def _adaptive_gradient_descent(h, dx, n_points, start_time, budget_s):
    # Adaptive gradient descent with different learning rates
    max_iter = 200
    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        c5 = np.max(np.correlate(h, 1 - h, mode='full')) * dx
        grad = np.zeros_like(h)
        eps = 1e-6
        lr = 0.003
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            c5_plus = np.max(np.correlate(h_plus, 1 - h_plus, mode='full')) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Adaptive LR
        lr = max(1e-6, lr / (1 + 0.01 * iteration))
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1) 
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

    return h, np.max(np.correlate(h, 1 - h, mode='full')) * dx

def _local_optimization(h, dx, n_points, start_time, budget_s):
    # More robust local search using scipy minimize with bounds
    def obj_func(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Use constrained optimization
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Normalize constraint
    def constraint_func(h_vals):
        return np.sum(h_vals) - n_points / 2
    
    cons = {'type': 'eq', 'fun': constraint_func}
    
    # Add a small perturbation to avoid plateaus
    perturb = np.random.normal(0, 0.01, n_points)
    initial_h = np.clip(h + perturb, 0, 1)
    initial_h = initial_h * (n_points / 2) / np.sum(initial_h)
    
    try:
        result = minimize(obj_func, initial_h, method='SLSQP', bounds=bounds, constraints=cons, 
                          options={'maxiter': 50, 'ftol': 1e-6})
        if result.success:
            optimized_h = np.clip(result.x, 0, 1)
            optimized_h = optimized_h * (n_points / 2) / np.sum(optimized_h)
            c5 = obj_func(optimized_h)
            return optimized_h, c5
    except:
        pass
    
    return h, np.max(np.correlate(h, 1 - h, mode='full')) * dx
