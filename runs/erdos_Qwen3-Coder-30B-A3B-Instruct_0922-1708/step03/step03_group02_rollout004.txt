import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Structured initialization
        h = np.zeros(n_points)
        # Place high values in intervals [0.2, 0.4] and [1.2, 1.4]
        idx1 = np.where((x_target >= 0.2) & (x_target <= 0.4))[0]
        idx2 = np.where((x_target >= 1.2) & (x_target <= 1.4))[0]
        h[idx1] = 1.0
        h[idx2] = 1.0
        h = h / np.sum(h) * (n_points / 2)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    T = 0.1  # initial temperature
    alpha = 0.995  # cooling factor
    lr = 0.01  # initial learning rate
    eps = 1e-5  # finite difference step
    perturb_rate = 0.1  # percentage of elements to perturb
    
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Simulated Annealing step
        h_trial = h.copy()
        num_perturb = int(n_points * perturb_rate)
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            h_trial[i] += np.random.normal(0, 0.05)
        h_trial = np.clip(h_trial, 0, 1)
        h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
        
        # Compute C5 for trial
        corr_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
        c5_trial = np.max(corr_trial) * dx
        
        # Accept or reject based on SA
        if c5_trial < best_c5 or np.random.rand() < np.exp(-(c5_trial - best_c5) / T):
            h = h_trial
            c5_current = c5_trial
        else:
            c5_current = np.max(np.correlate(h, 1 - h, mode='full')) * dx

        # Gradient descent step
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5_current) / eps
        
        # Update with gradient descent
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept the improvement
        if c5_new < c5_current:
            h = h_new
            c5_current = c5_new
            lr *= 1.01
        else:
            lr *= 0.95
        
        lr = np.clip(lr, 0.001, 0.02)
        T *= alpha
        
        # Track best
        if c5_current < best_c5:
            best_c5 = c5_current
            best_h = h.copy()
    
    # Final evaluation
    final_corr = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
