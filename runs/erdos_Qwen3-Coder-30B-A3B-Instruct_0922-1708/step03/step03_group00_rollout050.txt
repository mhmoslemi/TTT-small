import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    max_no_improvement = 50
    
    # PSO parameters
    n_particles = 20
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    personal_bests = []
    personal_best_values = []
    
    # Start with the initial solution
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    personal_bests.append(h.copy())
    personal_best_values.append(float('inf'))
    
    # Generate additional random particles
    for _ in range(n_particles - 1):
        h_rand = np.random.rand(n_points)
        h_rand = np.clip(h_rand, 0, 1)
        h_rand = h_rand * (n_points / 2) / np.sum(h_rand)
        particles.append(h_rand)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
        personal_bests.append(h_rand.copy())
        personal_best_values.append(float('inf'))
    
    # Update the global best with the initial solution
    global_best_idx = 0
    global_best = h.copy()
    global_best_value = float('inf')
    
    # PSO iteration
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        iteration += 1
        
        # Evaluate all particles
        for i, particle in enumerate(particles):
            correlation = np.correlate(particle, 1 - particle, mode='full')
            c5_value = np.max(correlation) * dx
            
            if c5_value < personal_best_values[i]:
                personal_best_values[i] = c5_value
                personal_bests[i] = particle.copy()
                
            if c5_value < global_best_value:
                global_best_value = c5_value
                global_best = particle.copy()
                no_improvement_count = 0  # reset counter
                
        if global_best_value < best_c5:
            best_c5 = global_best_value
            best_h = global_best.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.rand(2)
            particle = particles[i]
            vel = velocities[i]
            p_best = personal_bests[i]
            g_best = global_best
            
            # Update velocity
            new_vel = (w * vel + 
                       c1 * r1 * (p_best - particle) + 
                       c2 * r2 * (g_best - particle))
            
            # Clip velocity
            new_vel = np.clip(new_vel, -v_max, v_max)
            
            # Update position
            new_pos = particle + new_vel
            
            # Apply constraints
            new_pos = np.clip(new_pos, 0, 1)
            new_pos = new_pos * (n_points / 2) / np.sum(new_pos)
            
            particles[i] = new_pos
            velocities[i] = new_vel
        
        # Occasionally try a small random perturbation
        if iteration % 100 == 0:
            for i in range(n_particles):
                particles[i] = np.clip(particles[i] + np.random.normal(0, 0.001, n_points), 0, 1)
                particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
        
        no_improvement_count += 1
    
    # Refine with gradient descent on the best solution
    h = best_h.copy()
    epsilon = 1e-4
    max_refine_iter = 500
    refine_iter = 0
    last_c5 = best_c5
    
    while refine_iter < max_refine_iter and time.time() - start_time < budget_s - 1:
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Estimate gradient using finite differences
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += epsilon
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / epsilon
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad)**0.5 * 1e-2)
        h_new = h - lr * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        refine_iter += 1
        
        # Check for convergence
        new_c5 = np.max(np.correlate(h, 1 - h, mode='full') * dx)
        if abs(new_c5 - last_c5) < 1e-8:
            break
        last_c5 = new_c5
    
    correlation = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    if final_c5 < best_c5:
        best_c5 = final_c5
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
