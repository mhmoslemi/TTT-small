import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for efficiency
    lags = list(range(1, n_points))
    
    # Use a more refined optimization approach
    iteration = 0
    
    # Implement simulated annealing approach to escape local optima
    current_h = h.copy()
    current_c5 = float('inf')
    temperature = 1.0
    cooling_rate = 0.995
    min_temp = 1e-6
    
    while time.time() - start_time < budget_s - 5:
        iteration += 1
        if iteration % 10 == 0:
            # Recompute the current solution
            correlation = np.correlate(current_h, 1 - current_h, mode='full')
            current_c5 = np.max(correlation) * dx
            
            if current_c5 < best_c5:
                best_c5 = current_c5
                best_h = current_h.copy()
            
        # Perturb the solution
        h_perturbed = current_h.copy()
        
        # Pick a random index to modify
        idx = np.random.randint(0, n_points)
        # Modify the value
        delta = np.random.normal(0, 0.05)
        h_perturbed[idx] += delta
        
        # Clip to [0,1]
        h_perturbed = np.clip(h_perturbed, 0, 1)
        
        # Rescale to maintain constraint
        h_perturbed = h_perturbed * (n_points / 2) / np.sum(h_perturbed)
        
        # Compute new overlap
        correlation = np.correlate(h_perturbed, 1 - h_perturbed, mode='full')
        new_c5 = np.max(correlation) * dx
        
        # Accept or reject based on simulated annealing
        if new_c5 < current_c5:
            current_h = h_perturbed
        else:
            # Accept with probability based on temperature
            if np.random.rand() < np.exp(-(new_c5 - current_c5) / temperature):
                current_h = h_perturbed
                
        # Decrease temperature
        temperature *= cooling_rate
        if temperature < min_temp:
            temperature = min_temp
            
        # Occasionally try the constrained optimization approach
        if iteration % 50 == 0:
            # Solve a simpler convex optimization
            try:
                hvars = cp.Variable(n_points)
                constraints = [hvars >= 0, hvars <= 1, cp.sum(hvars) == n_points / 2]
                
                # Approximate with a single overlap constraint that is likely to be the maximum
                # This is a crude approximation to keep computations fast
                objective = cp.Minimize(cp.max(cp.conv(hvars, cp.reshape(1 - hvars, (len(hvars), 1)))))
                prob = cp.Problem(objective, constraints)
                
                # Try MOSEK first, then fallback to ECOS if necessary
                try:
                    prob.solve(solver=cp.MOSEK, verbose=False)
                except:
                    try:
                        prob.solve(solver=cp.ECOS, verbose=False)
                    except:
                        pass
                        
                if prob.status == cp.OPTIMAL:
                    h_new = hvars.value
                    if h_new is not None:
                        h_new = np.clip(h_new, 0, 1)
                        h_new = h_new * (n_points / 2) / np.sum(h_new)
                        
                        correlation = np.correlate(h_new, 1 - h_new, mode='full')
                        c5_new = np.max(correlation) * dx
                        
                        if c5_new < best_c5:
                            best_h = h_new
                            best_c5 = c5_new
            except:
                pass
    
    # Final refinement at the end
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
