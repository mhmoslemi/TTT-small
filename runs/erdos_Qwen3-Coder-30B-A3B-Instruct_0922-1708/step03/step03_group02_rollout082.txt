import numpy as np
import cvxpy as cp
import time
from scipy.optimize import minimize_scalar

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using parent or uniform distribution
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Heuristic to improve convergence by alternating optimization strategies
    strategy = "convex"  # switch between "convex" and "gradient"
    
    while time.time() - start_time < budget_s - 1:
        # Evaluate current solution
        correlation = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(correlation) * dx
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
        
        # Alternate strategies to avoid local minima
        if strategy == "convex":
            # Try convex relaxation approach
            try:
                # Create a simpler convex optimization
                h_var = cp.Variable(n_points)
                t = cp.Variable()
                
                # Constraints
                constraints = [
                    h_var >= 0,
                    h_var <= 1,
                    cp.sum(h_var) == n_points / 2,
                    t >= 0
                ]
                
                # Approximate the max_k integral using McCormick envelopes
                # Simplified version - still aiming to optimize the max_k term
                for k in range(n_points):
                    # For small values of k, collect terms more precisely
                    if k < n_points // 2:
                        expr = 0
                        for i in range(n_points - k):
                            expr += h_var[i] * (1 - h_var[i + k])
                        constraints.append(t >= expr * dx)
                
                # Objective
                prob = cp.Problem(cp.Minimize(t), constraints)
                prob.solve(solver=cp.ECOS, verbose=False)
                
                if prob.status == cp.OPTIMAL:
                    h = np.array(h_var.value).reshape(-1)
                    # Ensure exact constraint
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    # Fallback to gradient approach
                    strategy = "gradient"
                    
            except Exception:
                strategy = "gradient"

        else:
            # Classic gradient descent
            grad = np.zeros_like(h)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                corr_orig = np.correlate(h, 1 - h, mode='full') 
                c5_orig = np.max(corr_orig) * dx
                
                grad[i] = (c5_plus - c5_orig) / eps
            
            # Update with learning rate adapted based on gradient magnitude
            lr = 0.005 / (1 + np.linalg.norm(grad) * 0.1)
            h_new = h - lr * grad
            
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
            # Switch strategy occasionally
            if np.random.rand() < 0.1:
                strategy = "convex"

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
