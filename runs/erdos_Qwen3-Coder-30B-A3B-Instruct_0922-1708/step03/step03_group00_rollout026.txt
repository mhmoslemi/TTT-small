import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def evaluate_h(h, dx):
    """Evaluate the C5 constant for a given step function h"""
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def project_h_to_feasible(h, n_points):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2"""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_h(h, dx)
    
    # PSO parameters
    n_particles = 20
    max_iter = 200
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    vel_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    pbest = []
    pbest_vals = []
    
    # Start with the best solution as first particle
    particles.append(h.copy())
    velocities.append(np.zeros(n_points))
    pbest.append(h.copy())
    pbest_vals.append(best_c5)
    
    # Add random particles
    for i in range(n_particles - 1):
        h_random = np.random.rand(n_points)
        h_random = project_h_to_feasible(h_random, n_points)
        particles.append(h_random)
        velocities.append(np.random.rand(n_points) * 0.01)
        pbest.append(h_random.copy())
        pbest_vals.append(evaluate_h(h_random, dx))
    
    # Global best
    global_best_idx = np.argmin(pbest_vals)
    global_best = pbest[global_best_idx].copy()
    global_best_val = pbest_vals[global_best_idx]
    
    start_time = time.time()
    no_improve_count = 0
    no_improve_limit = 50
    
    iteration = 0
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        for i in range(n_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (
                inertia * velocities[i] +
                c1 * r1 * (pbest[i] - particles[i]) +
                c2 * r2 * (global_best - particles[i])
            )
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -vel_max, vel_max)
            
            # Update position
            particles[i] += velocities[i]
            
            # Project to feasible region
            particles[i] = project_h_to_feasible(particles[i], n_points)
            
            # Evaluate
            c5 = evaluate_h(particles[i], dx)
            
            # Update personal best
            if c5 < pbest_vals[i]:
                pbest[i] = particles[i].copy()
                pbest_vals[i] = c5
                
                # Update global best
                if c5 < global_best_val:
                    global_best = particles[i].copy()
                    global_best_val = c5
                    no_improve_count = 0
                else:
                    no_improve_count += 1
        
        # Early termination if no improvement
        if no_improve_count >= no_improve_limit:
            break
    
    # Final refinement with gradient descent
    best_h = global_best.copy()
    best_c5 = global_best_val
    
    # Small gradient descent on the best result
    max_grad_iter = 50
    for i in range(max_grad_iter):
        # Compute correlation
        correlation = np.correlate(best_h, 1 - best_h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient estimation
        grad = np.zeros(n_points)
        eps = 1e-4
        
        for j in range(n_points):
            h_plus = best_h.copy()
            h_plus[j] += eps
            h_plus = project_h_to_feasible(h_plus, n_points)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[j] = (c5_plus - c5) / eps
        
        # Adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        best_h_new = best_h - lr * grad
        
        # Project and rescale
        best_h_new = project_h_to_feasible(best_h_new, n_points)
        
        # Check improvement
        new_c5 = evaluate_h(best_h_new, dx)
        if new_c5 < best_c5:
            best_h = best_h_new
            best_c5 = new_c5
        else:
            break
    
    return (best_h.tolist(), best_c5, n_points)
