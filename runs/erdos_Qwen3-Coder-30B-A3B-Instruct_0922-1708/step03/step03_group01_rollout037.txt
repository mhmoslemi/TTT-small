import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős constant C₅ for given step function h."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    stagnation_count = 0
    stagnation_threshold = 100
    aspiration_prob = 0.1
    
    start_time = time.time()
    iteration = 0
    
    # Hash function for storing solutions in tabu list
    def hash_solution(sol):
        # Convert to tuple of integers for hashing
        integer_sol = (sol * 10000).astype(int)
        return hashlib.md5(integer_sol.tobytes()).hexdigest()
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        candidates = []
        
        # Generate multiple candidate solutions
        for k in range(50):
            # Create a perturbed version of current solution
            h_candidate = h.copy()
            
            # Select random indices to modify
            indices_to_modify = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbations
            for i in indices_to_modify:
                delta = np.random.uniform(-step_size, step_size)
                h_candidate[i] += delta
            
            # Ensure h_candidate stays within bounds and sum is correct
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Check if this candidate is in tabu list
            candidate_hash = hash_solution(h_candidate)
            if candidate_hash not in tabu_list:
                c5 = evaluate_erdos_solution(h_candidate)
                candidates.append((h_candidate.copy(), c5))
        
        if not candidates:
            # If tabu list is too large, clear it to avoid memory issues
            tabu_list.clear()
            continue
            
        # Select best candidate that's not tabu
        best_candidate = min(candidates, key=lambda x: x[1])
        h_new, c5_new = best_candidate
        
        # Update global best if we improved
        if c5_new < best_c5:
            best_c5 = c5_new
            best_h = h_new.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
        
        # Manage TABU list
        old_hash = hash_solution(h)
        tabu_list.add(old_hash)
        if len(tabu_list) > tabu_length:
            tabu_list.pop()
        
        # Accept worse solution with low probability (aspiration)
        if c5_new > best_c5 and np.random.random() < aspiration_prob:
            h = h_new.copy()
        else:
            # Set new current solution
            h = h_new.copy()
        
        # Increase step size if stagnating
        if stagnation_count > stagnation_threshold:
            step_size *= 1.2
            stagnation_count = 0
            # Clear tabu list to improve diversity
            tabu_list.clear()
    
    # Final re-evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
