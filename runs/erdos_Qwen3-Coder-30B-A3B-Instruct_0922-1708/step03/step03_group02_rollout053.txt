import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured pattern
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured starting point
        h = np.zeros(n_points)
        # Create two spikes
        idx1 = slice(int(0.2/dx), int(0.4/dx))
        idx2 = slice(int(1.2/dx), int(1.4/dx))
        h[idx1] = 1.0
        h[idx2] = 1.0
        h = h / np.sum(h) * (n_points / 2)

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters
    temp = 0.1
    cooling_rate = 0.995
    lr = 0.01
    perturb_rate = 0.1
    
    def compute_c5(h):
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def project_to_feasible(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        return h
    
    def get_gradient(h, eps=1e-5):
        grad = np.zeros_like(h)
        base_c5 = compute_c5(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = project_to_feasible(h_plus)
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - base_c5) / eps
        return grad
    
    # Main optimization loop
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_candidate = h.copy()
        # Randomly perturb 10% of the values
        indices = np.random.choice(n_points, int(perturb_rate * n_points), replace=False)
        for i in indices:
            delta = np.random.uniform(-0.1, 0.1)
            h_candidate[i] += delta
        h_candidate = project_to_feasible(h_candidate)
        c5_candidate = compute_c5(h_candidate)
        
        # Accept based on SA criterion
        if c5_candidate < compute_c5(h) or np.random.rand() < math.exp(-(c5_candidate - compute_c5(h)) / temp):
            h = h_candidate
        
        # Gradient Descent Step
        grad = get_gradient(h)
        h_new = h - lr * grad
        h_new = project_to_feasible(h_new)
        c5_new = compute_c5(h_new)
        
        # Update learning rate
        if c5_new < compute_c5(h):
            lr *= 1.01
        else:
            lr *= 0.95
        lr = np.clip(lr, 0.001, 0.02)
        
        h = h_new
        
        # Update temperature
        temp *= cooling_rate
        
        # Update best
        if compute_c5(h) < best_c5:
            best_c5 = compute_c5(h)
            best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
