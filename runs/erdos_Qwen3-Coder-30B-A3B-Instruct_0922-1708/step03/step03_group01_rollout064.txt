import numpy as np
import time
from hashlib import md5

def evaluate_erdos_solution(h):
    """Evaluate the C5 bound for given h."""
    dx = 2.0 / len(h)
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * dx

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from parent or uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_length = 50
    max_perturbation = 0.3
    perturbation_step = 0.1
    perturbation_count = max(1, int(0.2 * n_points))
    stagnation_count = 0
    last_improvement = 0
    aspiration_prob = 0.1
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Generate several candidate solutions
        candidates = []
        for k in range(50):
            # Create perturbed solution
            h_candidate = h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbation
            for i in indices:
                delta = np.random.uniform(-perturbation_step, perturbation_step)
                h_candidate[i] += delta
            
            # Project to feasible space
            h_candidate = np.clip(h_candidate, 0, 1)
            h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
            
            # Hash the candidate to check TABU
            candidate_hash = md5(h_candidate.tobytes()).hexdigest()
            
            # Skip if in TABU list
            if candidate_hash in tabu_list:
                continue
                
            # Evaluate
            c5 = evaluate_erdos_solution(h_candidate)
            candidates.append((h_candidate, c5, candidate_hash))
        
        if not candidates:
            # Reset tabu list if all candidates same
            tabu_list.clear()
            continue
            
        # Sort candidates by c5
        candidates.sort(key=lambda x: x[1])
        best_candidate, candidate_c5, candidate_hash = candidates[0]
        
        # Check if better than current best
        if candidate_c5 < best_c5:
            best_h = best_candidate.copy()
            best_c5 = candidate_c5
            last_improvement = 0
            
            # Add previous h to tabu list
            h_hash = md5(h.tobytes()).hexdigest()
            tabu_list.add(h_hash)
            if len(tabu_list) > tabu_length:
                tabu_list.pop()
                
            h = best_candidate.copy()
        else:
            # Aspiration criterion
            if np.random.random() < aspiration_prob:
                h = best_candidate.copy()
                h_hash = md5(h.tobytes()).hexdigest()
                tabu_list.add(h_hash)
                if len(tabu_list) > tabu_length:
                    tabu_list.pop()
            
            last_improvement += 1
        
        # Adaptive perturbation
        if last_improvement > 100:
            if perturbation_step < max_perturbation:
                perturbation_step *= 1.2
                last_improvement = 0
                if len(tabu_list) > 0:
                    tabu_list.clear()
        
        # Remove oldest from tabu list if necessary
        if len(tabu_list) > tabu_length:
            tabu_list.pop()
    
    # Final evaluation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
