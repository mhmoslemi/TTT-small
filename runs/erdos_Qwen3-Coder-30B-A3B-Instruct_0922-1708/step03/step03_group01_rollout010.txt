import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant C5 for given h."""
    n_points = len(h)
    dx = 2.0 / n_points
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize with parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5

    # Initialize tracking variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(h)
    current_h = h.copy()
    
    # TABU search parameters
    tabu_list = set()
    tabu_list_length = 50
    step_size = 0.1
    stagnation_count = 0
    stagnation_threshold = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Generate candidates
        candidates = []
        num_candidates = 50
        
        for _ in range(num_candidates):
            # Create a new candidate
            candidate_h = current_h.copy()
            perturbation_count = max(1, int(0.2 * n_points))
            
            # Select indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbations
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Project to feasible region
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h *= scale_factor
            
            # Check if this candidate is in TABU list
            candidate_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            if candidate_hash not in tabu_list:
                c5 = evaluate_erdos_solution(candidate_h)
                candidates.append((candidate_h.copy(), c5, candidate_hash))
        
        # Select best candidate not in TABU list, or any if none available
        if candidates:
            best_candidate = min(candidates, key=lambda x: x[1])
            candidate_h, candidate_c5, candidate_hash = best_candidate
            
            # Accept if better or with aspiration criterion
            if candidate_c5 < best_c5:
                best_h = candidate_h.copy()
                best_c5 = candidate_c5
                stagnation_count = 0
            else:
                # Aspiration criterion
                if np.random.random() < 0.1:
                    # Accept even if worse to escape local optima
                    stagnation_count = 0
                else:
                    # If no improvement, increase step size if stagnation
                    stagnation_count += 1
                    if stagnation_count > stagnation_threshold:
                        step_size *= 1.2
                        tabu_list.clear()  # Clear to allow diversity
                        stagnation_count = 0
            
            # Update TABU list
            current_hash = hashlib.md5(current_h.tobytes()).hexdigest()
            tabu_list.add(current_hash)
            
            # Maintain TABU list size
            if len(tabu_list) > tabu_list_length:
                tabu_list.pop()
            
            # Update current solution
            current_h = candidate_h.copy()
        else:
            # If no candidates generated, take a random perturbation
            current_h = h.copy()
            indices = np.random.choice(n_points, size=10, replace=False)
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                current_h[i] += delta
            current_h = np.clip(current_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(current_h)
            current_h *= scale_factor
            
            # Update best if needed
            c5 = evaluate_erdos_solution(current_h)
            if c5 < best_c5:
                best_h = current_h.copy()
                best_c5 = c5
                stagnation_count = 0
            else:
                stagnation_count += 1
                if stagnation_count > stagnation_threshold:
                    step_size *= 1.2
                    tabu_list.clear()
                    stagnation_count = 0
                        
    return (best_h.tolist(), best_c5, n_points)
