import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    # Compute the overlap integral C_5
    correlation = np.correlate(h, 1 - h, mode='full')
    return np.max(correlation) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU search parameters
    tabu_list = set()
    tabu_list_size = 50
    stagnation_count = 0
    max_stagnation = 100
    perturbation_count = max(1, int(0.2 * n_points))
    step_size = 0.1
    
    start_time = time.time()
    current_h = h.copy()
    current_c5 = best_c5
    
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Generate candidates by perturbing current solution
        candidates = []
        num_candidates = 50
        
        for _ in range(num_candidates):
            # Create a new candidate by perturbing the current solution
            candidate_h = current_h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Apply perturbation
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Project to [0,1]
            candidate_h = np.clip(candidate_h, 0, 1)
            
            # Rescale to meet integral constraint
            scaling = (n_points / 2) / np.sum(candidate_h)
            candidate_h *= scaling
            
            # Check if this candidate is in TABU list
            h_hash = hashlib.md5(candidate_h.tobytes()).hexdigest()
            if h_hash in tabu_list:
                continue
                
            # Evaluate candidate
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            candidates.append((candidate_h.copy(), candidate_c5))
        
        # If we have no valid candidates, just create a new random one
        if not candidates:
            candidate_h = np.random.rand(n_points)
            candidate_h = np.clip(candidate_h, 0, 1)
            scaling = (n_points / 2) / np.sum(candidate_h)
            candidate_h *= scaling
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            candidates = [(candidate_h.copy(), candidate_c5)]
            
        # Find the best candidate
        best_candidate_h, best_candidate_c5 = min(candidates, key=lambda x: x[1])
        
        # Check if we should accept this
        if best_candidate_c5 < best_c5:
            # We found a new best solution
            best_c5 = best_candidate_c5
            best_h = best_candidate_h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1

        # Aspiration criteria - occasionally accept worse solutions for escape
        if np.random.rand() < 0.1 and best_candidate_c5 < current_c5 + 1e-8:
            # Accept even if worse
            pass

        # Update TABU list
        current_h = best_candidate_h.copy()
        current_c5 = best_candidate_c5

        # Update tabu list
        h_hash = hashlib.md5(current_h.tobytes()).hexdigest()
        if len(tabu_list) >= tabu_list_size:
            # Remove oldest entry (in practice, this is a simple set, so order is not guaranteed)
            tabu_list.pop()
        tabu_list.add(h_hash)
        
        # Increase perturbation if stagnant
        if stagnation_count >= max_stagnation:
            step_size *= 1.2
            stagnation_count = 0
            # Clear tabu list occasionally to maintain diversity
            tabu_list.clear()
            
        # Check if we have time to continue
        if time.time() - start_time > budget_s - 2:
            break
            
    return (best_h.tolist(), best_c5, n_points)
