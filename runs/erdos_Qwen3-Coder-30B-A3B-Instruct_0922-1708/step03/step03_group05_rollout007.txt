import numpy as np
import time
from scipy.signal import correlate

def evaluate_h(h, dx):
    """Evaluate the C5 bound for given h."""
    correlation = correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def project_h(h, n_points, dx):
    """Project h to satisfy constraints: 0 <= h <= 1 and sum(h) = n_points/2."""
    h = np.clip(h, 0, 1)
    h_sum = np.sum(h)
    if h_sum > 0:
        h = h * (n_points / 2) / h_sum
    return h

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize swarm
    num_particles = 50
    w = 0.7
    c1, c2 = 1.5, 1.5
    v_max = 0.1
    
    # Initialize particles
    particles = []
    velocities = []
    
    # Use initial_h_values if available
    if 'initial_h_values' in globals():
        init_h = np.array(initial_h_values)
        if len(init_h) != n_points:
            x_orig = np.linspace(0, 2, len(init_h))
            x_target = np.linspace(0, 2, n_points)
            init_h = np.interp(x_target, x_orig, init_h)
        init_h = init_h / np.sum(init_h) * (n_points / 2)
        
        # Seed 10% with initial solution
        for i in range(num_particles // 10):
            particles.append(init_h.copy())
            velocities.append(np.random.uniform(-v_max, v_max, n_points))
        remaining = num_particles - num_particles // 10
    else:
        remaining = num_particles
    
    # Initialize remaining particles randomly
    for _ in range(remaining):
        h = np.random.rand(n_points)
        h = project_h(h, n_points, dx)
        particles.append(h)
        velocities.append(np.random.uniform(-v_max, v_max, n_points))
    
    # Best positions and values
    p_best = [p.copy() for p in particles]
    p_best_c5 = [evaluate_h(p, dx) for p in particles]
    
    g_best_idx = np.argmin(p_best_c5)
    g_best = p_best[g_best_idx].copy()
    g_best_c5 = p_best_c5[g_best_idx]
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Update all particles
        for i in range(num_particles):
            # Update velocity
            r1, r2 = np.random.rand(2)
            velocities[i] = (w * velocities[i] + 
                           c1 * r1 * (p_best[i] - particles[i]) + 
                           c2 * r2 * (g_best - particles[i]))
            
            # Clip velocity
            velocities[i] = np.clip(velocities[i], -v_max, v_max)
            
            # Update position
            particles[i] = particles[i] + velocities[i]
            
            # Project to feasible region
            particles[i] = project_h(particles[i], n_points, dx)
            
            # Evaluate
            c5 = evaluate_h(particles[i], dx)
            
            # Update personal best
            if c5 < p_best_c5[i]:
                p_best[i] = particles[i].copy()
                p_best_c5[i] = c5
                
                # Update global best
                if c5 < g_best_c5:
                    g_best = particles[i].copy()
                    g_best_c5 = c5
    
    return (g_best.tolist(), g_best_c5, n_points)
