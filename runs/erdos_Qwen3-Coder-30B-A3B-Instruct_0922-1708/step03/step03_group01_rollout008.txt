import numpy as np
import time
from scipy import signal

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    population = []
    
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h.copy())
    
    # Fill population with random individuals
    while len(population) < pop_size:
        h = np.random.rand(n_points)
        h = h * (n_points / 2) / np.sum(h)
        h = np.clip(h, 0, 1)
        population.append(h.copy())
    
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 100:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            correlation = signal.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
        
        # Sort by fitness (lower is better)
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        # Keep top 40% as elites
        elite_size = int(0.4 * pop_size)
        new_population = population[:elite_size]
        
        # Generate offspring through crossover
        while len(new_population) < pop_size:
            # Select two parents (tournament selection)
            parent1_idx = np.random.randint(0, elite_size)
            parent2_idx = np.random.randint(0, elite_size)
            parent1 = population[parent1_idx]
            parent2 = population[parent2_idx]
            
            # Crossover: blend points
            alpha = np.random.rand()
            child = alpha * parent1 + (1 - alpha) * parent2
            
            # Mutate
            if np.random.rand() < 0.2:  # 20% mutation rate
                num_mutation = int(0.05 * n_points)  # 5% of points
                mutation_indices = np.random.choice(n_points, num_mutation, replace=False)
                for i in mutation_indices:
                    noise = np.random.normal(0, 0.1)
                    child[i] += noise
                    child[i] = np.clip(child[i], 0, 1)
            
            # Rescale to maintain constraint
            child = child * (n_points / 2) / np.sum(child)
            
            new_population.append(child)
        
        population = new_population
        
        # Early stopping if no improvement
        if generation > 1 and fitness[0] >= best_c5:
            no_improvement += 1
        else:
            no_improvement = 0
    
    return (best_h.tolist(), best_c5, n_points)
