import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from initial_h_values if available, else uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Normalize and ensure constraints
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    while (high - low > tolerance) and (time.time() - start_time < budget_s - 1):
        mid = (low + high) / 2.0
        
        # Create CVX problem
        h_vars = cp.Variable(n_points)
        
        # Objective to minimize C5
        obj = cp.Minimize(mid)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints for all shifts
        for k in range(n_points):
            # Compute sum h[i](1 - h[i+k]) for shift k
            # Using numpy-style indexing with wraparound
            overlap_sum = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap_sum += h_vars[i] * (1 - h_vars[j])
            constraints.append(overlap_sum <= mid / dx)
        
        # Formulate and solve the problem
        prob = cp.Problem(obj, constraints)
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == 'optimal':
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Compute actual C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                actual_c5 = np.max(correlation) * dx
                
                if actual_c5 < best_c5:
                    best_c5 = actual_c5
                    best_h = h_sol.copy()
                
                high = mid
            else:
                low = mid
        except:
            low = mid
    
    # Final computation of C5 for best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
