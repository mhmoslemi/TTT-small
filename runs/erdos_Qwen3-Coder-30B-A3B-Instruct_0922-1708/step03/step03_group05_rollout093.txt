import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using initial_h_values if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Solve convex relaxation
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed <= 1, h_relaxed >= 0, cp.sum(h_relaxed) == n_points/2]
        
        # Formulate the overlap constraints
        overlap_expressions = []
        for lag in range(n_points):
            expr = 0
            for i in range(n_points - lag):
                expr += h_relaxed[i] * (1 - h_relaxed[i+lag])
            overlap_expressions.append(expr)
        
        objective = cp.Minimize(cp.max(cp.hstack(overlap_expressions)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == 'optimal':
                h_relaxed_val = h_relaxed.value
            else:
                h_relaxed_val = h
        except:
            h_relaxed_val = h
            
        # Round and adjust
        h_rounded = (h_relaxed_val >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        
        # Adjust to ensure sum = n_points / 2
        if current_sum > n_points / 2:
            # Need to flip some 1's to 0's
            diff = int(current_sum - n_points / 2)
            flip_indices = np.argsort(h_relaxed_val[h_rounded == 1])[:diff]
            flip_indices = np.where(h_rounded == 1)[0][flip_indices]
            h_rounded[flip_indices] = 0
        elif current_sum < n_points / 2:
            # Need to flip some 0's to 1's
            diff = int(n_points / 2 - current_sum)
            flip_indices = np.argsort(-h_relaxed_val[h_rounded == 0])[:diff]
            flip_indices = np.where(h_rounded == 0)[0][flip_indices]
            h_rounded[flip_indices] = 1
            
        # Check this solution
        correlation = np.correlate(h_rounded, 1 - h_rounded, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h_rounded.copy()
            
        # Slightly modify h for next iteration
        h = h_rounded.copy()
        perturbation = 0.01 * np.random.randn(n_points)
        h = np.clip(h + perturbation, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
