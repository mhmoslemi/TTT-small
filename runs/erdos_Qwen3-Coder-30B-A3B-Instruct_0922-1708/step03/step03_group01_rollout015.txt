import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search on C5
    low, high = 0.38, 0.4
    tolerance = 1e-4
    
    while high - low > tolerance and time.time() - start_time < budget_s - 1:
        c5_guess = (low + high) / 2.0
        
        # Create optimization variables
        h_vars = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add overlap constraints
        for k in range(n_points):
            # Compute the convolution constraint manually for this k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_vars[i] * (1 - h_vars[j])
            constraints.append(expr <= c5_guess / dx)
        
        # Objective: minimize C5 (though it's just a parameter in constraints)
        prob = cp.Problem(cp.Minimize(0), constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                h_sol = h_vars.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Compute actual C5 with this h_sol
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                computed_c5 = np.max(correlation) * dx
                
                if computed_c5 < best_c5:
                    best_c5 = computed_c5
                    best_h = h_sol.copy()
                
                # Update search bounds
                high = c5_guess
            else:
                low = c5_guess
        except:
            low = c5_guess
            
    return (best_h.tolist(), best_c5, n_points)
