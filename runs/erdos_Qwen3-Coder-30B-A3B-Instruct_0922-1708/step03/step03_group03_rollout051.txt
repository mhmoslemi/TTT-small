import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Barker-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
    else:
        # Create a Barker-like initialization
        h = np.zeros(n_points)
        # Alternate pattern approximating Barker code properties
        for i in range(n_points):
            if i % 4 < 2:
                h[i] = 1.0
            else:
                h[i] = 0.0
                
    # Normalize to unit integral
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    def objective_and_gradient(h_vec):
        # Reshape for optimization
        h = h_vec.copy()
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Gradient computation (finite difference approximation)
        grad = np.zeros_like(h)
        eps = 1e-6
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps
            
        return c5, grad
    
    # Optimization using L-BFGS-B
    bounds = [(0, 1) for _ in range(n_points)]
    
    # Initial optimization
    x0 = h.copy()
    
    # Run a few iterations of optimization
    opt_result = minimize(
        objective_and_gradient,
        x0,
        method='L-BFGS-B',
        bounds=bounds,
        jac=True,
        options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
    )
    
    if opt_result.success:
        h_opt = opt_result.x
        h_opt = np.clip(h_opt, 0, 1)
        h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
        
        correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_h = h_opt
            best_c5 = c5

    # Additional search strategies
    # Try a few restarts with different initialization
    for restart in range(5):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Perturb initialization
        h_init = h.copy() + np.random.normal(0, 0.02, n_points)
        h_init = np.clip(h_init, 0, 1)
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        
        opt_result = minimize(
            objective_and_gradient,
            h_init,
            method='L-BFGS-B',
            bounds=bounds,
            jac=True,
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if opt_result.success:
            h_opt = opt_result.x
            h_opt = np.clip(h_opt, 0, 1)
            h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
            
            correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
            c5 = np.max(correlation) * dx
            
            if c5 < best_c5:
                best_h = h_opt
                best_c5 = c5
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
