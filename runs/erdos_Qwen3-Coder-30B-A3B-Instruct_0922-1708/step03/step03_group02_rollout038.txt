import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init)
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init)
    
    # Fill remaining particles with variations
    for _ in range(n_particles - 1):
        h = h_init + np.random.normal(0, 0.05, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h)
    
    # Initialize velocities
    for _ in range(n_particles):
        velocities.append(np.random.uniform(-0.1, 0.1, n_points))
    
    # Initialize best positions
    personal_best = [p.copy() for p in particles]
    personal_best_scores = [float('inf')] * n_particles
    global_best = particles[0].copy()
    global_best_score = float('inf')
    
    # PSO parameters
    omega = 1.0
    c1 = c2 = 2.0
    max_velocity = 0.2
    
    start_time = time.time()
    step = 0
    
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure constraints
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute overlap integral
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_scores[i]:
                personal_best_scores[i] = c5
                personal_best[i] = h.copy()
            
            # Update global best
            if c5 < global_best_score:
                global_best_score = c5
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            v = omega * velocities[i] + \
                c1 * r1 * (personal_best[i] - particles[i]) + \
                c2 * r2 * (global_best - particles[i])
            
            # Clamp velocity
            v = np.clip(v, -max_velocity, max_velocity)
            velocities[i] = v
            
            # Update position
            particles[i] = particles[i] + v
            
        # Periodic reset to maintain diversity
        if step % 100 == 0:
            for i in range(n_particles):
                if np.random.random() < 0.1:
                    particles[i] = np.random.uniform(0.3, 0.7, n_points)
                    particles[i] = particles[i] * (n_points / 2) / np.sum(particles[i])
    
    # Final verification
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
