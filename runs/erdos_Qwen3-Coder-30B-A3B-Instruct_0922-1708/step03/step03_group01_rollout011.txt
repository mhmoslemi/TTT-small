import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdos constant for given step function h."""
    # h is assumed to be normalized so that sum(h) = n_points/2
    n_points = len(h)
    dx = 2.0 / n_points
    # Compute the overlap integral
    correlation = np.correlate(h, 1 - h, mode='full')
    c5 = np.max(correlation) * dx
    return c5

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        h = h * (n_points / 2) / np.sum(h)
    
    # Initialize variables
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU search settings
    tabu_list = set()
    tabu_length = 50
    perturbation_count = int(0.2 * n_points)
    step_size = 0.1
    stagnation_count = 0
    max_stagnation = 100
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1:
        # Generate candidates by perturbing current solution
        candidates = []
        
        for k in range(50):
            # Copy current solution and perturb
            candidate_h = h.copy()
            
            # Select random indices to perturb
            indices = np.random.choice(n_points, size=perturbation_count, replace=False)
            
            # Perturb selected values
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
            
            # Project to feasible region
            candidate_h = np.clip(candidate_h, 0, 1)
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h = candidate_h * scale_factor
            
            # Evaluate candidate
            c5 = evaluate_erdos_solution(candidate_h)
            candidates.append((candidate_h.copy(), c5))
            
        # Find best candidate not in tabu list
        valid_candidates = [(h_candidate, c5) for h_candidate, c5 in candidates 
                            if hashlib.md5(h_candidate.tobytes()).hexdigest() not in tabu_list]
        
        if not valid_candidates:
            # If all candidates are tabu, allow aspiration
            valid_candidates = candidates
            
        # Find best valid candidate
        best_candidate_h, best_candidate_c5 = min(valid_candidates, key=lambda x: x[1])
        
        # Check if better than current best
        if best_candidate_c5 < best_c5:
            # Update best solution
            best_h = best_candidate_h.copy()
            best_c5 = best_candidate_c5
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Update tabu list
        h_hash = hashlib.md5(h.tobytes()).hexdigest()
        tabu_list.add(h_hash)
        if len(tabu_list) > tabu_length:
            tabu_list.pop()
            
        # Update solution
        h = best_candidate_h.copy()
        
        # Adapt step size if stuck
        if stagnation_count >= max_stagnation:
            step_size *= 1.2
            stagnation_count = 0
            tabu_list.clear()  # Clear tabu list to allow restarting
        
        # Ensure solution remains valid
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
    # Final validation
    final_c5 = evaluate_erdos_solution(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
