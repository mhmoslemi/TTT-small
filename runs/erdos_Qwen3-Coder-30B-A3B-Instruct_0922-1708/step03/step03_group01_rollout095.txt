            best_c5 = c5

            # Adaptive step size decrease
            alpha *= 0.95
            patience = 0
            
        # Compute gradient analytically for the maximum overlap
        max_shift_idx = np.argmax(correlation)
        shift = max_shift_idx - (len(h) - 1)
            
        grad = np.zeros_like(h)
        for i in range(len(h)):
            j1 = (i + shift) % len(h)
            j2 = (i - shift) % len(h)
            grad[i] = dx * ((1 - h[j1]) - h[j2])
            
        # Update h
        h_new = h - alpha * grad

        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

    correlation = np.correlate(best_h, 1 - best_h, mode='full')
