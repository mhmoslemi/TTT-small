import numpy as np
import time
from scipy.signal import correlate

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize population
    pop_size = 50
    elite_rate = 0.2
    cross_rate = 0.7
    mut_rate = 0.1
    tournament_size = 5
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
    else:
        h_init = np.ones(n_points) * 0.5
        
    # Create initial population
    population = []
    population.append(h_init)
    for _ in range(pop_size - 1):
        h = h_init + np.random.normal(0, 0.1, n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        population.append(h)
    
    best_h = population[0].copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    generation = 0
    no_improvement_count = 0
    max_no_improvement = 100
    
    while time.time() - start_time < budget_s - 1 and no_improvement_count < max_no_improvement:
        generation += 1
        
        # Evaluate fitness
        fitness = []
        for h in population:
            corr = correlate(h, 1 - h, mode='full')
            c5 = np.max(corr) * dx
            fitness.append(c5)
            
        # Track best solution
        min_fitness_idx = np.argmin(fitness)
        if fitness[min_fitness_idx] < best_c5:
            best_c5 = fitness[min_fitness_idx]
            best_h = population[min_fitness_idx].copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
            
        # Selection with tournament
        selected = []
        for _ in range(pop_size):
            tournament = np.random.choice(pop_size, tournament_size)
            winner_idx = tournament[np.argmin([fitness[i] for i in tournament])]
            selected.append(population[winner_idx].copy())
            
        # Create new population
        new_population = []
        
        # Elitism
        elites = [population[i] for i in np.argsort(fitness)[:int(elite_rate * pop_size)]]
        new_population.extend(elites)
        
        # Crossover and mutation
        while len(new_population) < pop_size:
            if np.random.random() < cross_rate and len(selected) >= 2:
                # Crossover
                p1, p2 = np.random.choice(len(selected), 2, replace=False)
                point = np.random.randint(1, n_points - 1)
                child1 = np.concatenate([selected[p1][:point], selected[p2][point:]])
                child2 = np.concatenate([selected[p2][:point], selected[p1][point:]])
                
                # Mutation
                if np.random.random() < mut_rate:
                    mutate_idx = np.random.randint(0, n_points)
                    child1[mutate_idx] += np.random.normal(0, 0.1)
                if np.random.random() < mut_rate:
                    mutate_idx = np.random.randint(0, n_points)
                    child2[mutate_idx] += np.random.normal(0, 0.1)
                    
                # Ensure feasibility
                child1 = np.clip(child1, 0, 1)
                child1 = child1 * (n_points / 2) / np.sum(child1)
                new_population.append(child1)
                
                if len(new_population) < pop_size:
                    child2 = np.clip(child2, 0, 1)
                    child2 = child2 * (n_points / 2) / np.sum(child2)
                    new_population.append(child2)
            else:
                # Pure mutation
                parent = np.random.choice(selected)
                child = parent.copy()
                mutate_indices = np.random.choice(n_points, int(mut_rate * n_points), replace=False)
                for i in mutate_indices:
                    child[i] += np.random.normal(0, 0.1)
                child = np.clip(child, 0, 1)
                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
                
        population = new_population[:pop_size]
    
    # Final verification
    corr = correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(corr) * dx
    
    return (best_h.tolist(), final_c5, n_points)
