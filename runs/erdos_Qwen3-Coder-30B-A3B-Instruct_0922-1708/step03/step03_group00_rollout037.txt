import numpy as np
import time
from scipy.fft import fft, ifft

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
    else:
        h = np.ones(n_points) * 0.5
        
    # Normalize
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Spectral optimization
    n_freq = 50
    freqs = np.arange(-n_freq, n_freq + 1)
    learning_rate = 0.001
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute overlap using FFT
        h_fft = fft(h)
        one_minus_h_fft = 1 - h_fft
        overlap_fft = h_fft * one_minus_h_fft
        c5 = np.max(np.abs(overlap_fft)) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Project h back to feasible set
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Compute gradients in frequency domain
        if iteration % 10 == 0:
            # Use finite differences for gradient
            grad_fft = np.zeros_like(h_fft, dtype=complex)
            eps = 1e-5
            
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                h_plus_fft = fft(h_plus)
                overlap_plus_fft = h_plus_fft * (1 - h_plus_fft)
                c5_plus = np.max(np.abs(overlap_plus_fft)) * dx
                grad_fft += (c5_plus - c5) / eps * ifft(1 - h_fft)
                
            grad_fft = grad_fft / n_points
            
            # Update with learning rate
            h_fft -= learning_rate * grad_fft
            h = np.real(ifft(h_fft))
            
            learning_rate *= 0.999
            
        else:
            # Minor gradient-based update
            h_fft = fft(h)
            h_fft -= learning_rate * np.conj(h_fft) * (h_fft * (1 - h_fft))
            h = np.real(ifft(h_fft))
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
        # Occasionally escape local optima
        if iteration % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h += noise
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    return (best_h.tolist(), best_c5, n_points)
