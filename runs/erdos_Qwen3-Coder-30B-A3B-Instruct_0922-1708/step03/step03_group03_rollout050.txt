import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create modified Barker-like initialization
        h = np.zeros(n_points)
        # Place pulses to minimize overlap
        pulse_width = n_points // 5
        for i in range(5):
            start = i * pulse_width
            end = min(start + pulse_width, n_points)
            h[start:end] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Define objective function and its gradient
    def objective_and_gradient(h_vec):
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        max_corr = np.max(correlation)
        
        # Approximate gradient with finite differences
        eps = 1e-6
        grad = np.zeros_like(h_vec)
        orig = max_corr
        
        for i in range(n_points):
            h_plus = h_vec.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            max_plus = np.max(corr_plus)
            grad[i] = (max_plus - orig) / eps
            
        return max_corr, grad
    
    # Optimization with restarts
    max_restarts = 5
    restart_count = 0
    
    while time.time() - start_time < budget_s - 1 and restart_count < max_restarts:
        # Ensure starting point satisfies constraints
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        # Use L-BFGS-B to minimize the maximum overlap
        bounds = [(0, 1) for _ in range(n_points)]
        
        try:
            result = minimize(
                lambda x: objective_and_gradient(x)[0],
                h,
                method='L-BFGS-B',
                bounds=bounds,
                jac=lambda x: objective_and_gradient(x)[1],
                options={'maxiter': 1000, 'gtol': 1e-8}
            )
            
            if result.success:
                h_opt = result.x
                h_opt = np.clip(h_opt, 0, 1)
                h_opt = h_opt * (n_points / 2) / np.sum(h_opt)
                
                correlation = np.correlate(h_opt, 1 - h_opt, mode='full')
                c5_opt = np.max(correlation) * dx
                
                if c5_opt < best_c5:
                    best_c5 = c5_opt
                    best_h = h_opt.copy()
        except:
            pass
        
        # Restart with a new random perturbation
        h = best_h.copy()
        noise = np.random.normal(0, 0.01, n_points)
        h = h + noise
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        
        restart_count += 1
    
    # Final evaluation of best solution
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
