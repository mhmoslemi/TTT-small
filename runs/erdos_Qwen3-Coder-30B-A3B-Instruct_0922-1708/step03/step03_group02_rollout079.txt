import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a structured construction
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Create a structured initial guess
        h = np.zeros(n_points)
        # Place steps in strategic locations
        ind1 = int(0.2 * n_points)
        ind2 = int(0.4 * n_points)
        ind3 = int(1.2 * n_points)
        ind4 = int(1.4 * n_points)
        h[ind1:ind2] = 1.0
        h[ind3:ind4] = 1.0
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Parameters for the hybrid approach
    temp = 0.1
    alpha = 0.995
    lr = 0.01
    pert_rate = 0.1
    
    while time.time() - start_time < budget_s - 1:
        # Simulated Annealing Step
        h_candidate = h.copy()
        num_perturb = max(1, int(n_points * pert_rate))
        indices = np.random.choice(n_points, num_perturb, replace=False)
        for i in indices:
            h_candidate[i] += np.random.normal(0, 0.05)
        h_candidate = np.clip(h_candidate, 0, 1)
        h_candidate = h_candidate * (n_points / 2) / np.sum(h_candidate)
        
        # Compute C5 for candidate
        corr = np.correlate(h_candidate, 1 - h_candidate, mode='full')
        c5_candidate = np.max(corr) * dx
        
        # Accept or reject based on SA
        if c5_candidate < best_c5 or np.random.rand() < np.exp(-(c5_candidate - best_c5) / temp):
            h = h_candidate.copy()
            c5 = c5_candidate
        
        # Gradient Descent Step
        eps = 1e-5
        grad = np.zeros_like(h)
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            grad[i] = (c5_plus - c5) / eps
        
        # Update with gradient
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        corr_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr_new) * dx
        
        # Accept the update if it improves the objective
        if c5_new < c5:
            h = h_new.copy()
            c5 = c5_new
        
        # Adaptive learning rate
        if c5_new < best_c5:
            lr *= 1.01
        else:
            lr *= 0.95
        lr = np.clip(lr, 0.001, 0.02)
        
        # Update temperature
        temp *= alpha
        
        # Update best
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()

    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
