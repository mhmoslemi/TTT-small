import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize h from provided parent or uniform
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Iterative convex relaxation with McCormick envelopes
    max_iter = 100
    for iter_num in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Solve convex relaxation
        h_var = cp.Variable(n_points)
        t = cp.Variable()
        
        # McCormick bounds (linear approximations)
        # z[i,j] ~ h[i] * (1 - h[j])
        constraints = [
            cp.sum(h_var) == n_points / 2,
            h_var >= 0,
            h_var <= 1,
            t >= 0
        ]
        
        # Compute overlap integrals via McCormick relaxation
        # We'll form a convex problem by relaxing h[i]*(1-h[j]) terms
        for k in range(n_points):  # lag values
            # For each lag, we sum over valid indices
            expr = 0
            for i in range(n_points - k):
                # Use McCormick relaxation:
                # z[i] = h[i] * (1 - h[i+k])
                # Lower bound: z[i] >= h[i] + h[i+k] - 1
                # Upper bounds: z[i] <= h[i], z[i] <= 1 - h[i+k] 
                # z[i] >= 0
                expr += h_var[i] * (1 - h_var[i + k])
            # Use convex relaxation of the sum term
            # This is a proxy for the actual bilinear terms
            constraints.append(t >= expr * dx)
        
        # Solve with ECOS for efficiency
        prob = cp.Problem(cp.Minimize(t), constraints)
        try:
            prob.solve(solver=cp.ECOS, max_iters=500, abstol=1e-4, reltol=1e-4)
            if prob.status == cp.OPTIMAL:
                h_sol = h_var.value
                if h_sol is not None:
                    h = np.clip(h_sol, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                    
                    # Compute actual C5
                    correlation = np.correlate(h, 1 - h, mode='full')
                    c5 = np.max(correlation) * dx
                    
                    if c5 < best_c5:
                        best_c5 = c5
                        best_h = h.copy()
        except:
            pass
            
        # Apply small random perturbations to escape local minima
        if iter_num % 5 == 0:
            h = np.clip(h + np.random.normal(0, 0.01, n_points), 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
