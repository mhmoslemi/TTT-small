import numpy as np
from scipy.optimize import minimize
import time
import math

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # PSO parameters
    n_particles = 20
    w = 0.7  # inertia weight
    c1 = 2.0  # cognitive parameter
    c2 = 2.0  # social parameter
    max_velocity = 0.1
    
    # Initialize particles
    particles = []
    particle_fitness = []
    pbest = []
    pbest_fitness = []
    
    for _ in range(n_particles):
        if 'initial_h_values' in globals():
            ph = np.array(initial_h_values)
            if len(ph) != n_points:
                x_orig = np.linspace(0, 2, len(ph))
                x_new = np.linspace(0, 2, n_points)
                ph = np.interp(x_new, x_orig, ph)
            ph = np.clip(ph, 0, 1)
            ph = ph * (n_points / 2) / np.sum(ph)
        else:
            ph = np.random.uniform(0, 1, n_points)
            ph = ph * (n_points / 2) / np.sum(ph)
        particles.append(ph)
        fitness = np.max(np.correlate(ph, 1 - ph, mode='full')) * dx
        particle_fitness.append(fitness)
        pbest.append(ph.copy())
        pbest_fitness.append(fitness)
        
        if fitness < best_c5:
            best_c5 = fitness
            best_h = ph.copy()
    
    # Global best
    gbest_idx = np.argmin(pbest_fitness)
    gbest = pbest[gbest_idx].copy()
    
    # PSO iteration
    no_improve_iter = 0
    max_no_improve_iter = 50
    
    while time.time() - start_time < budget_s - 1 and no_improve_iter < max_no_improve_iter:
        iteration += 1
        
        for i in range(n_particles):
            particle = particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = w * (particle - pbest[i]) + c1 * r1 * (pbest[i] - particle) + c2 * r2 * (gbest - particle)
            
            # Clip velocity
            v = np.clip(v, -max_velocity, max_velocity)
            
            # Update particle
            new_particle = particle + v
            
            # Apply constraints
            new_particle = np.clip(new_particle, 0, 1)
            new_particle = new_particle * (n_points / 2) / np.sum(new_particle)
            
            # Evaluate fitness
            fitness = np.max(np.correlate(new_particle, 1 - new_particle, mode='full')) * dx
            
            # Update particle
            particles[i] = new_particle.copy()
            
            # Update personal best
            if fitness < pbest_fitness[i]:
                pbest[i] = new_particle.copy()
                pbest_fitness[i] = fitness
                
                # Update global best
                if fitness < best_c5:
                    gbest = new_particle.copy()
                    best_c5 = fitness
                    best_h = new_particle.copy()
                    no_improve_iter = 0
                else:
                    no_improve_iter += 1
            else:
                no_improve_iter += 1
                
            # Occasionally try a small random perturbation to escape local optima
            if iteration % 100 == 0:
                random_pert = np.random.normal(0, 0.001, n_points)
                new_particle = particles[i] + random_pert
                new_particle = np.clip(new_particle, 0, 1)
                new_particle = new_particle * (n_points / 2) / np.sum(new_particle)
                
                # Evaluate fitness
                fitness = np.max(np.correlate(new_particle, 1 - new_particle, mode='full')) * dx
                
                # Update particle
                particles[i] = new_particle.copy()
                
                # Update personal best
                if fitness < pbest_fitness[i]:
                    pbest[i] = new_particle.copy()
                    pbest_fitness[i] = fitness
                    
                    # Update global best
                    if fitness < best_c5:
                        gbest = new_particle.copy()
                        best_c5 = fitness
                        best_h = new_particle.copy()
                        no_improve_iter = 0
                    else:
                        no_improve_iter += 1
                else:
                    no_improve_iter += 1

    return (best_h.tolist(), best_c5, n_points)
