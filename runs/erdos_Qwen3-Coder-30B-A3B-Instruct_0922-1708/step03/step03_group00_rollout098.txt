from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):

    n_points = 100
    dx = 2.0 / n_points

    best_h = h.copy()
    best_c5 = float('inf')

    # Genetic Algorithm approach

    generations = 300
    elite_size = population_size // 5
    mutation_rate = 0.1
    crossover_rate = 0.8
    
    # Initialize population
    population = []
    if 'initial_h_values' in globals():
        population.append(h.copy())
        for _ in range(population_size - 1):
            pert = h + np.random.normal(0, 0.05, n_points)
            pert = np.clip(pert, 0, 1)
            pert = pert * (n_points / 2) / np.sum(pert)
            population.append(pert)
    else:
        for _ in range(population_size):
            h_init = np.random.rand(n_points)
            h_init = np.clip(h_init, 0, 1)
            h_init = h_init * (n_points / 2) / np.sum(h_init)
            population.append(h_init)
    
    stagnation_count = 0
    best_global_c5 = float('inf')
    
    for gen in range(generations):
        if time.time() - start_time > budget_s - 5:
            break
            
        # Evaluate fitness
        fitness = []
        for individual in population:
            correlation = np.correlate(individual, 1 - individual, mode='full')
            c5 = np.max(correlation) * dx
            fitness.append(c5)
            
        # Sort population by fitness
        sorted_indices = np.argsort(fitness)
        population = [population[i] for i in sorted_indices]
        fitness = [fitness[i] for i in sorted_indices]
        
        if fitness[0] < best_global_c5:
            best_global_c5 = fitness[0]
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        if stagnation_count > 100:
            break
            
        # Update best solution so far
        if fitness[0] < best_c5:
            best_c5 = fitness[0]
            best_h = population[0].copy()
            
        # Selection: Keep top elite_size individuals
        new_population = population[:elite_size]

        while len(new_population) < population_size:

            tournament_size = population_size // 10
            tournament_indices = np.random.choice(population_size, tournament_size, replace=False)
            tournament_fitness = [fitness[i] for i in tournament_indices]
            winner_idx = tournament_indices[np.argmin(tournament_fitness)]
            
            if np.random.rand() < crossover_rate:
                # Select second parent
                tournament_indices2 = np.random.choice(population_size, tournament_size, replace=False)
                tournament_fitness2 = [fitness[i] for i in tournament_indices2]
                winner_idx2 = tournament_indices2[np.argmin(tournament_fitness2)]
                
                # Blend crossover
                parent1 = population[winner_idx]
                parent2 = population[winner_idx2]
                alpha = np.random.rand()
                child = alpha * parent1 + (1 - alpha) * parent2
                
                # Mutation
                if np.random.rand() < mutation_rate:
                    mutate_indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
                    for i in mutate_indices:
                        child[i] += np.random.normal(0, 0.01)
                else:
                    child = parent1
                
                # Ensure feasibility
                child = np.clip(child, 0, 1)

                child = child * (n_points / 2) / np.sum(child)
                new_population.append(child)
        
        population = new_population

    h = best_h.copy()
    for _ in range(100):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - best_c5) / eps
            
        # Update using gradient descent
        lr = 0.001
        h_new = h - lr * grad
        
        # Apply constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Evaluate new solution
        corr = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(corr) * dx
        
        if c5_new < best_c5:
            h = h_new
            best_c5 = c5_new
        else:
            # Early stopping if no improvement
            break

    return (h.tolist(), best_c5, n_points)
