import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturb_std = 0.1
    max_anneal_steps = 2000
    
    # Nelder-Mead related
    simplex_size = 5
    max_nm_steps = 1000
    
    # Annealing phase
    steps = 0
    while steps < max_anneal_steps and time.time() - start_time < budget_s - 2:
        # Perturb randomly
        h_new = h.copy()
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        h_new[indices] += np.random.normal(0, perturb_std, len(indices))
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new value
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Accept or reject
        if c5_new < best_c5:
            h = h_new.copy()
            best_c5 = c5_new
            best_h = h.copy()
        elif np.random.rand() < np.exp(-(c5_new - best_c5) / T):
            h = h_new.copy()
            
        T *= alpha
        steps += 1

    # Nelder-Mead refinement
    def objective(h_vals):
        h = np.array(h_vals)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx

    def constraint(h_vals):
        return np.sum(h_vals) - n_points / 2

    # Generate initial simplex
    simplex = [best_h.copy()]
    for i in range(simplex_size - 1):
        h_pert = best_h + np.random.normal(0, 0.05, n_points)
        h_pert = np.clip(h_pert, 0, 1)
        h_pert = h_pert * (n_points / 2) / np.sum(h_pert)
        simplex.append(h_pert)

    # Nelder-Mead loop
    nm_steps = 0
    while nm_steps < max_nm_steps and time.time() - start_time < budget_s - 1:
        # Evaluate simplex
        values = [objective(s) for s in simplex]
        indices = np.argsort(values)
        simplex = [simplex[i] for i in indices]
        values = [values[i] for i in indices]
        
        # Reflect
        x_med = np.mean(simplex[:-1], axis=0)
        x_refl = 2 * x_med - simplex[-1]
        
        # Ensure feasibility by clipping
        x_refl = np.clip(x_refl, 0, 1)
        x_refl = x_refl * (n_points / 2) / np.sum(x_refl)
        
        f_refl = objective(x_refl)
        
        if values[0] <= f_refl < values[-2]:
            simplex[-1] = x_refl.copy()
        elif f_refl < values[0]:
            # Expand
            x_exp = 2 * x_refl - x_med
            x_exp = np.clip(x_exp, 0, 1)
            x_exp = x_exp * (n_points / 2) / np.sum(x_exp)
            f_exp = objective(x_exp)
            if f_exp < f_refl:
                simplex[-1] = x_exp.copy()
            else:
                simplex[-1] = x_refl.copy()
        else:
            # Contract
            x_con = 0.5 * (simplex[0] + simplex[-1])
            x_con = np.clip(x_con, 0, 1)
            x_con = x_con * (n_points / 2) / np.sum(x_con)
            f_con = objective(x_con)
            if f_con < values[-1]:
                simplex[-1] = x_con.copy()
            else:
                # Shrink
                for i in range(1, len(simplex)):
                    simplex[i] = simplex[0] + 0.5 * (simplex[i] - simplex[0])
                    simplex[i] = np.clip(simplex[i], 0, 1)
                    simplex[i] = simplex[i] * (n_points / 2) / np.sum(simplex[i])
        
        nm_steps += 1
    
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
