import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a Frank-Zadoff-like sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate Frank-Zadoff-like sequence with minimal correlation
        h = np.zeros(n_points)
        for i in range(n_points):
            # Use a sinusoidal pattern that's nearly constant but with small
            # variation to avoid total uniformity which can trap optimization
            h[i] = 0.5 + 0.3 * np.sin(2 * np.pi * i / n_points * 10)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Phase 1: Binary search with QCQP feasibility checks
    low, high = 0.38, 0.4
    while high - low > 1e-4 and time.time() - start_time < budget_s * 0.7:
        C_mid = (low + high) / 2
        # Formulate QCQP feasibility problem for this C_mid
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        # Add overlap constraints for all shifts k from 0 to n_points-1
        for k in range(n_points):
            shift = np.roll(1 - h_var, k)
            overlap = cp.sum(cp.multiply(h_var, shift))
            constraints.append(overlap <= C_mid / dx)
        # QCQP feasibility check
        prob = cp.Problem(cp.Minimize(0), constraints)
        try:
            prob.solve(solver=cp.MOSEK, verbose=False)
            if prob.status in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
                high = C_mid
                # Save the current h_var if it's in a feasible region
                current_h = h_var.value
                if current_h is not None:
                    # Compute c5 for this solution to update best
                    corr = np.correlate(current_h, 1 - current_h, mode='full')
                    c5_computed = np.max(corr) * dx
                    if c5_computed < best_c5:
                        best_c5 = c5_computed
                        best_h = current_h.copy()
            else:
                low = C_mid
        except Exception:
            low = C_mid

    # Phase 2: Gradient-based refinement
    h = best_h.copy()
    max_iter = 1000
    alpha = 0.01
    patience = 0
    patience_limit = 100
    last_c5 = float('inf')

    for iteration in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Compute current overlaps
        correlations = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlations) * dx
        max_shift = np.argmax(correlations)
        
        # Analytical gradient for the max shift
        grad = np.zeros(n_points)
        for i in range(n_points):
            # Adding 1 to handle wrap-around properly
            k = max_shift % n_points
            if (i + k) < n_points:
                grad[i] += 1 - h[i + k]  # From term h[i] * (1 - h[i+k])
            else:
                grad[i] += 1 - h[i + k - n_points]
            if (i - k) >= 0:
                grad[i] -= h[i - k]  # From term h[i-k] * (1 - h[i])
            else:
                grad[i] -= h[i - k + n_points]

        # Update with gradient descent
        h_new = h - alpha * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new

        # Update best if this is better
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            patience = 0
        else:
            patience += 1
            if patience > patience_limit:
                break

    # Downsample to 100 points
    n_final = 100
    dx_final = 2.0 / n_final
    h_final = np.interp(np.linspace(0, 2, n_final), np.linspace(0, 2, n_points), best_h)
    h_final = np.clip(h_final, 0, 1)
    h_final = h_final * (n_final / 2) / np.sum(h_final)

    # Final computation of C5
    corr_final = np.correlate(h_final, 1 - h_final, mode='full')
    best_c5_final = np.max(corr_final) * dx_final

    return (h_final.tolist(), best_c5_final, n_final)
