import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    def compute_c5(h_vals):
        # Compute the overlap integral using correlation
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(correlation) * dx
    
    def objective_and_gradient(h_vals):
        # Compute overlap integral
        c5 = compute_c5(h_vals)
        
        # Compute the gradient analytically for the max overlap
        # Find the shift that maximizes the overlap term
        correlation = np.correlate(h_vals, 1 - h_vals, mode='full')
        max_k = np.argmax(correlation)
        
        # Gradient for the maximum overlap term
        grad = np.zeros_like(h_vals)
        for i in range(len(h_vals)):
            # i-th element contributes to correlation with (i-k)th element
            # for k at position max_k (roll index)
            k = max_k - len(h_vals) + 1
            if i + k >= 0 and i + k < len(h_vals):
                grad[i] += (1 - h_vals[i + k]) * dx
            if i - k >= 0 and i - k < len(h_vals):
                grad[i] += -h_vals[i - k] * dx
                
        return c5, grad
    
    # Define constraints and bounds
    bounds = [(0, 1) for _ in range(n_points)]
    cons = {'type': 'eq', 'fun': lambda x: np.sum(x) * dx - 1.0}
    
    # Use SLSQP solver for optimization
    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Optimize using scipy.minimize
        res = minimize(
            lambda x: compute_c5(x),
            h,
            method='SLSQP',
            bounds=bounds,
            constraints=cons,
            jac=lambda x: objective_and_gradient(x)[1],
            options={'maxiter': 100, 'ftol': 1e-8, 'gtol': 1e-8}
        )
        
        if res.success:
            h = res.x
            c5 = compute_c5(h)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()
                
            # Occasionally restart with a different initialization to avoid local optima
            if iteration % 20 == 0:
                # Inject some randomness
                h = h * 0.9 + 0.1 * np.random.rand(n_points)
                h = np.clip(h, 0, 1)
                h = h * (n_points / 2) / np.sum(h)
                
        else:
            # Fallback to gradient descent if optimization fails
            c5, grad = objective_and_gradient(h)
            h = h - 1e-4 * grad
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
            if c5 < best_c5:
                best_c5 = c5
                best_h = h.copy()

    return (best_h.tolist(), best_c5, n_points)
