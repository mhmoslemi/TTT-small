import numpy as np
import time
from scipy.optimize import minimize

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # PSO Parameters
    n_particles = 20
    inertia = 0.7
    c1 = 2.0
    c2 = 2.0
    v_max = 0.1
    
    # Initialize particles
    particles = []
    particle_fitness = []
    personal_best = []
    personal_best_fitness = []
    global_best = h.copy()
    global_best_fitness = float('inf')
    
    for i in range(n_particles):
        if i == 0:
            particle = h.copy()
        else:
            # Perturb the initial solution or create random one with same integral
            particle = h + np.random.normal(0, 0.05, n_points)
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
        particles.append(particle)
        fitness = np.max(np.correlate(particle, 1 - particle, mode='full')) * dx
        particle_fitness.append(fitness)
        personal_best.append(particle.copy())
        personal_best_fitness.append(fitness)
        
        if fitness < global_best_fitness:
            global_best = particle.copy()
            global_best_fitness = fitness
    
    # PSO loop
    iteration = 0
    no_improvement = 0
    
    while time.time() - start_time < budget_s - 1 and no_improvement < 50:
        iteration += 1
        
        # Update all particles
        for i in range(n_particles):
            particle = particles[i]
            
            # Update velocity
            r1, r2 = np.random.rand(2)
            v = (inertia * (particle - particle) + 
                 c1 * r1 * (personal_best[i] - particle) + 
                 c2 * r2 * (global_best - particle))
            
            # Clamp velocity
            v = np.clip(v, -v_max, v_max)
            
            # Update position
            particle = particle + v
            
            # Clip and rescale
            particle = np.clip(particle, 0, 1)
            particle = particle * (n_points / 2) / np.sum(particle)
            
            # Update particle
            particles[i] = particle
            
            # Evaluate fitness
            fitness = np.max(np.correlate(particle, 1 - particle, mode='full')) * dx
            
            # Update personal best
            if fitness < personal_best_fitness[i]:
                personal_best[i] = particle.copy()
                personal_best_fitness[i] = fitness
                
                if fitness < global_best_fitness:
                    global_best = particle.copy()
                    global_best_fitness = fitness
                    no_improvement = 0  # Reset no improvement counter
        
        no_improvement += 1
    
    # Local refinement using gradient descent on best particle
    h = global_best.copy()
    max_iter = 200
    
    for k in range(max_iter):
        if time.time() - start_time > budget_s - 1:
            break
            
        # Evaluate current value
        corr = np.correlate(h, 1 - h, mode='full')
        current_c5 = np.max(corr) * dx
        
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps = 1e-4
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - current_c5) / eps
        
        # Update with adaptive learning rate
        lr = 0.01 / (1 + np.linalg.norm(grad) * 1e-3)
        h_new = h - lr * grad
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final evaluation
    corr = np.correlate(h, 1 - h, mode='full')
    final_c5 = np.max(corr) * dx
    
    if final_c5 < global_best_fitness:
        result_h = h
        result_c5 = final_c5
    else:
        result_h = global_best
        result_c5 = global_best_fitness
    
    return (result_h.tolist(), result_c5, n_points)
