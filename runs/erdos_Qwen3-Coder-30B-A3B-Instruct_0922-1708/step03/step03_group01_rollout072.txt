import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or Frank-Zadoff sequence
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        # Generate a Frank-Zadoff-like sequence with low cross-correlation
        h = np.zeros(n_points)
        for i in range(n_points):
            h[i] = 0.5 * (1 + np.sin(2 * np.pi * i * 0.318))
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    alpha = 0.01
    max_iter = 2000
    
    # Locally refine via gradient descent
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute the gradient for the maximum overlap shift
        max_shift_idx = np.argmax(correlation)
        max_shift = max_shift_idx - (len(h) - 1)
        
        # Build the gradient vector
        grad = np.zeros_like(h)
        for i in range(len(h)):
            # Contribution from h[i] * (1 - h[i + max_shift])
            if (i + max_shift) % len(h) < len(h):
                grad[i] -= dx
            # Contribution from (1 - h[i]) * h[i - max_shift]
            if (i - max_shift) % len(h) >= 0:
                grad[i] += dx
                
        # Update step
        h_new = h - alpha * grad
        
        # Project to [0,1] and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Update
        h = h_new
        
        # Adaptive step size adjustment
        if iteration % 100 == 0:
            alpha *= 0.99
            if alpha < 1e-6:
                alpha = 1e-6
    
    # Downsample to final resolution
    if n_points > 100:
        h_final = np.zeros(100)
        for i in range(100):
            h_final[i] = np.mean(h[2*i:2*i+2])
        h_final = h_final * 50 / np.sum(h_final)
        n_points_final = 100
    else:
        h_final = h
        n_points_final = n_points
        
    # Final evaluation
    correlation_final = np.correlate(h_final, 1 - h_final, mode='full')
    c5_final = np.max(correlation_final) * (2.0 / n_points_final)
    
    return (h_final.tolist(), c5_final, n_points_final)
