import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Binary search for the minimum C5
    low = 0.38
    high = 0.4
    
    while high - low > 1e-4 and (time.time() - start_time < budget_s - 1):
        c_mid = (low + high) / 2.0
        # Formulate the QCQP problem
        h_var = cp.Variable(n_points)
        
        # Constraints
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Overlap constraints
        for k in range(n_points):
            # Compute overlap for shift k
            expr = 0
            for i in range(n_points):
                j = (i + k) % n_points
                expr += h_var[i] * (1 - h_var[j])
            constraints.append(expr <= c_mid / dx)
        
        # Objective
        objective = cp.Minimize(c_mid)
        
        # Solve the problem
        prob = cp.Problem(objective, constraints)
        
        try:
            prob.solve(solver=cp.ECOS, verbose=False)
            if prob.status == cp.OPTIMAL:
                h_sol = h_var.value
                h_sol = np.clip(h_sol, 0, 1)
                h_sol = h_sol * (n_points / 2) / np.sum(h_sol)
                
                # Evaluate the real C5
                correlation = np.correlate(h_sol, 1 - h_sol, mode='full')
                real_c5 = np.max(correlation) * dx
                
                if real_c5 < best_c5:
                    best_c5 = real_c5
                    best_h = h_sol.copy()
                
                high = c_mid
            else:
                low = c_mid
        except:
            low = c_mid
    
    # Final evaluation
    if best_c5 == float('inf'):
        # If no better solution found, use current h
        correlation = np.correlate(h, 1 - h, mode='full')
        best_c5 = np.max(correlation) * dx
        best_h = h.copy()
    
    return (best_h.tolist(), best_c5, n_points)
