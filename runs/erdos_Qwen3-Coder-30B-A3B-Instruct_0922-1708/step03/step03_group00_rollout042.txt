from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 50
    dx = 2.0 / n_points

    if 'initial_h_values' in globals():

            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Simulated annealing parameters
    T = 0.1
    alpha = 0.99
    epsilon = 0.05
    max_iterations = 500
    patience = 100
    
    start_time = time.time()
    iteration = 0
    no_improvement_count = 0
    
    # Annealing loop
    while iteration < max_iterations and time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            no_improvement_count = 0
        else:
            no_improvement_count += 1
        
        # Generate neighbor
        h_new = h + np.random.normal(0, epsilon, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)

        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject
        delta = c5_new - c5
        if delta < 0 or np.random.random() < np.exp(-delta / T):
            h = h_new

        T *= alpha

            h = best_h.copy()
            no_improvement_count = 0

        # Anneal learning rate
        if refine_iter % 50 == 0:
