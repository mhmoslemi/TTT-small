import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Start with a block function
        h = np.zeros(n_points)
        h[:n_points//2] = 2.0
        
    h = np.clip(h, 0, 1)
    h = h * (n_points / 2) / np.sum(h)
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0

    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute the overlap integral
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Use gradient-free optimization for exploration
        # Try to improve with a simple local search
        h_trial = h.copy()
        improvement = False
        
        for _ in range(10):
            if time.time() - start_time > budget_s - 1:
                break
            i = np.random.randint(0, n_points)
            delta = np.random.uniform(-0.1, 0.1)
            old_val = h_trial[i]
            h_trial[i] = np.clip(old_val + delta, 0, 1)
            # Rescale
            h_trial = h_trial * (n_points / 2) / np.sum(h_trial)
            
            correlation_trial = np.correlate(h_trial, 1 - h_trial, mode='full')
            c5_trial = np.max(correlation_trial) * dx
            
            if c5_trial < c5:
                h = h_trial.copy()
                c5 = c5_trial
                improvement = True
                break
                
        if not improvement:
            # Perturb all points slightly
            h_new = h + np.random.normal(0, 0.01, n_points)
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final evaluation
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
