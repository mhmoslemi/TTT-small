import numpy as np
from scipy.optimize import minimize
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    eps = 1e-5
    last_improvement = 0
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient for a specific shift k
    def compute_gradient_for_shift(h_array, k):
        grad = np.zeros_like(h_array)
        for i in range(n_points):
            idx = (i - k) % n_points
            grad[i] = (1 - h_array[idx]) * dx
        return grad
    
    def projected_step(h_array, delta):
        h_new = h_array + delta
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        return h_new
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            last_improvement = iteration
            
        # Find the maximizing shift
        correlation = np.correlate(h, 1 - h, mode='full')
        k_max = np.argmax(correlation) - (n_points - 1)
        
        # Compute gradient for this shift
        grad = compute_gradient_for_shift(h, k_max)
        
        # Update h with projected gradient descent
        h_new = projected_step(h, -lr * grad)
        
        h = h_new
        
        # Adjust learning rate and periodically restart if no improvement
        if iteration % 100 == 0:
            if iteration - last_improvement > 500:
                # Reset with initial values
                if 'initial_h_values' in globals():
                    h = np.array(initial_h_values)
                    if len(h) != n_points:
                        x_orig = np.linspace(0, 2, len(h))
                        x_new = np.linspace(0, 2, n_points)
                        h = np.interp(x_new, x_orig, h)
                    h = np.clip(h, 0, 1)
                    h = h * (n_points / 2) / np.sum(h)
                else:
                    h = np.ones(n_points) * 0.5
                last_improvement = iteration
            else:
                lr *= 0.95
            
    return (best_h.tolist(), best_c5, n_points)
