import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    random.seed(seed)
    
    n_points = 50
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Simulated Annealing phase
    T = 0.1
    alpha = 0.99
    eps = 0.05
    
    iterations = 0
    stagnation_count = 0
    max_stagnation = 100
    
    while time.time() - start_time < budget_s * 0.8 and stagnation_count < max_stagnation:
        iterations += 1
        
        # Compute current C5
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            stagnation_count = 0
        else:
            stagnation_count += 1
            
        # Generate neighbor
        h_new = h + np.random.normal(0, eps, n_points)
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute new C5
        correlation_new = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation_new) * dx
        
        # Accept or reject
        if c5_new < c5 or np.random.random() < np.exp(-(c5_new - c5) / T):
            h = h_new.copy()
            
        # Cool down
        T *= alpha
        
        # Occasionally reset to best seen
        if iterations % 50 == 0 and np.random.random() < 0.3:
            h = best_h.copy()
            
    # Gradient descent refinement on best solution
    h = best_h.copy()
    lr = 0.001
    decay_rate = 0.95
    ref_iterations = 0
    
    while time.time() - start_time < budget_s - 1:
        ref_iterations += 1
        correlation = np.correlate(h, 1 - h, mode='full')
        c5 = np.max(correlation) * dx
        
        # Compute gradient with finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-5
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
            c5_plus = np.max(corr_plus) * dx
            
            grad[i] = (c5_plus - c5) / eps_fd
            
        # Update step
        h_new = h - lr * grad
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Decay learning rate
        if ref_iterations % 50 == 0:
            lr *= decay_rate
            
        # Occasionally perturb to escape
        if ref_iterations % 100 == 0:
            noise = np.random.normal(0, 0.001, n_points)
            h = np.clip(h + noise, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            
    # Final evaluation of best solution
    final_correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(final_correlation) * dx
    
    # Final check
    if final_c5 < best_c5:
        best_h = best_h.copy()
        best_c5 = final_c5
        
    # Refine at higher resolution if time allows
    if time.time() - start_time < budget_s - 5:
        n_points = 100
        dx = 2.0 / n_points
        h_refined = np.interp(np.linspace(0, 2, n_points), np.linspace(0, 2, len(best_h)), best_h)
        h_refined = np.clip(h_refined, 0, 1)
        h_refined = h_refined * (n_points / 2) / np.sum(h_refined)
        
        # Gradient descent refinement at higher resolution
        h = h_refined.copy()
        lr = 0.001
        iterations = 0
        while time.time() - start_time < budget_s - 1 and iterations < 150:
            iterations += 1
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            grad = np.zeros_like(h)
            eps_fd = 1e-5
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps_fd
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                
                corr_plus = np.correlate(h_plus, 1 - h_plus, mode='full')
                c5_plus = np.max(corr_plus) * dx
                
                grad[i] = (c5_plus - c5) / eps_fd
                
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            h = h_new
            if iterations % 50 == 0:
                lr *= 0.95
                
        # Final evaluation
        final_correlation = np.correlate(h, 1 - h, mode='full')
        final_c5 = np.max(final_correlation) * dx
        
        if final_c5 < best_c5:
            best_h = h.copy()
            best_c5 = final_c5

    return (best_h.tolist(), best_c5, n_points)
