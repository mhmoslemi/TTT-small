import numpy as np
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize particles
    n_particles = 40
    particles = []
    velocities = []
    
    if 'initial_h_values' in globals():
        h_init = np.array(initial_h_values)
        if len(h_init) != n_points:
            x_orig = np.linspace(0, 2, len(h_init))
            x_target = np.linspace(0, 2, n_points)
            h_init = np.interp(x_target, x_orig, h_init)
        h_init = h_init / np.sum(h_init) * (n_points / 2)
        particles.append(h_init.copy())
    else:
        h_init = np.ones(n_points) * 0.5
        h_init = h_init * (n_points / 2) / np.sum(h_init)
        particles.append(h_init.copy())
    
    # Add random particles
    for _ in range(n_particles - len(particles)):
        h = np.random.uniform(0.4, 0.6, n_points)
        h = h * (n_points / 2) / np.sum(h)
        particles.append(h.copy())
    
    # Initialize velocities
    for _ in range(n_particles):
        v = np.random.uniform(-0.1, 0.1, n_points)
        velocities.append(v.copy())
    
    # PSO parameters
    w = 1.0  # inertia weight
    c1 = c2 = 2.0  # acceleration coefficients
    max_vel = 0.2  # maximum velocity
    
    # Track personal bests and global best
    personal_bests = [p.copy() for p in particles]
    personal_best_values = [float('inf')] * n_particles
    global_best = particles[0].copy()
    global_best_value = float('inf')
    
    start_time = time.time()
    
    step = 0
    while time.time() - start_time < budget_s - 1:
        step += 1
        
        # Evaluate all particles
        for i in range(n_particles):
            h = particles[i]
            
            # Ensure bounds and normalization
            h = np.clip(h, 0, 1)
            h = h * (n_points / 2) / np.sum(h)
            particles[i] = h
            
            # Compute C5
            correlation = np.correlate(h, 1 - h, mode='full')
            c5 = np.max(correlation) * dx
            
            # Update personal best
            if c5 < personal_best_values[i]:
                personal_best_values[i] = c5
                personal_bests[i] = h.copy()
            
            # Update global best
            if c5 < global_best_value:
                global_best_value = c5
                global_best = h.copy()
        
        # Update velocities and positions
        for i in range(n_particles):
            r1, r2 = np.random.random(2)
            h = particles[i]
            v = velocities[i]
            
            # Compute new velocity
            if i == 0:  # Use global best for first particle to increase diversity
                v = w * v + c1 * r1 * (personal_bests[i] - h) + c2 * r2 * (global_best - h)
            else:
                v = w * v + c1 * r1 * (personal_bests[i] - h) + c2 * r2 * (global_best - h)
            
            # Clip velocity
            v = np.clip(v, -max_vel, max_vel)
            velocities[i] = v
            
            # Update position
            h_new = h + v
            
            # Ensure bounds and normalization
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            
            particles[i] = h_new
        
        # Occasionally reset particles to avoid local minima
        if step % 100 == 0:
            for i in range(1, n_particles):  # Don't reset the global best particle
                if np.random.random() < 0.1:
                    h = np.random.uniform(0.4, 0.6, n_points)
                    h = h * (n_points / 2) / np.sum(h)
                    particles[i] = h.copy()
                    velocities[i] = np.random.uniform(-0.1, 0.1, n_points)
        
        # Adapt inertia weight to balance exploration and exploitation
        if step % 1000 == 0 and w > 0.6:
            w = max(0.6, w - 0.01)
    
    # Final verification
    correlation = np.correlate(global_best, 1 - global_best, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (global_best.tolist(), final_c5, n_points)
