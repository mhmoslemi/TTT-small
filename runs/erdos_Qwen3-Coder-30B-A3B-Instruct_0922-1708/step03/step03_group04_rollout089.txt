import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to desired resolution
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    lr = 0.01
    
    # Helper function to compute overlap integral
    def compute_c5(h_array):
        correlation = np.correlate(h_array, 1 - h_array, mode='full')
        return np.max(correlation) * dx
    
    # Helper function to compute gradient for the maximum overlap
    def compute_gradient(h_array):
        # Compute all correlations
        corr = np.correlate(h_array, 1 - h_array, mode='full')
        max_k = np.argmax(corr)
        
        # The full correlation is of size 2*n_points-1, so we need to map k to a valid shift
        # k in [0, 2*n_points-2], and max_k is the index of maximum
        
        # The shift value corresponds to: max_k - (n_points - 1)
        shift = max_k - (n_points - 1)
        
        # When shift is negative, it means it wraps around
        # To compute gradient, d/dh[j] of the maximum correlation:
        # it's the derivative of the sum h[i]*h[i-shift], or h[i]*h[i+shift]
        # which equals the derivative of h[i]*h[i-shift] = h[i-shift] or h[i+shift]
        
        grad = np.zeros_like(h_array)
        # For each position j, if i = j + shift (considering modular arithmetic)
        # we have contribution to the derivative. So we compute it:
        # grad[j] = -[the contribution from h[j-shift] to the overlap term]
        # but also considering that if i = j-shift, then we take contribution from h[i]
        # It's just the identity with rotation.
        
        # Compute gradient contribution from this shift
        for j in range(len(h_array)):
            # If shift is positive, h[j] contributes to h[j-shift]
            # If shift is negative, h[j] contributes to h[j+shift] (wrapping around)
            # More carefully, for correl[h[i], 1-h[i+shift]]:
            # the term is h[i] * (1 - h[i + shift]) for all i
            # So for that term to involve h[j], we need i = j and i + shift = j
            # That is: i = j, shift = 0, so grad[i] = -h[i] 
            # For i + shift = j: i = j - shift, so derivatives depend on ...
            # Actually better way:
            
            # For the maximum correlation term, there's one certain most relevant shift
            # We'll compute the sensitivity of the maximum to changes in h
            
            # Let's do it carefully.
            idx1 = (j - shift) % n_points
            idx2 = (j + shift) % n_points
            grad[j] = -h[idx1] - h[idx2] if shift != 0 else -2 * h[j]
        return grad, max_k
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        current_c5 = compute_c5(h)
        
        if current_c5 < best_c5:
            best_c5 = current_c5
            best_h = h.copy()
            
        # Compute gradient for current state
        grad, k_max = compute_gradient(h)
        
        # Update step
        h_new = h - lr * grad
        
        # Project and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
        
        # Adaptive learning rate
        if iteration % 100 == 0:
            lr *= 0.95
            
        # Occasionally random restart
        if iteration % 200 == 0:
            if 'initial_h_values' in globals():
                h0 = np.array(initial_h_values)
                if len(h0) != n_points:
                    x_orig = np.linspace(0, 2, len(h0))
                    x_new = np.linspace(0, 2, n_points)
                    h0 = np.interp(x_new, x_orig, h0)
                h0 = np.clip(h0, 0, 1)
                h0 = h0 * (n_points / 2) / np.sum(h0)
                h = h0
            else:
                h = np.ones(n_points) * 0.5
                
    return (best_h.tolist(), best_c5, n_points)
