import numpy as np
from scipy.optimize import minimize
import time
import random

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    # Start timing
    start_time = time.time()
    
    # Simulated Annealing parameters
    T = 0.1
    alpha = 0.995
    perturbation_scale = 0.1
    
    # Nelder-Mead parameters
    max_nelder_mead_steps = 1000
    
    # Phase 1: Simulated Annealing
    sa_steps = 0
    max_sa_steps = 2000
    
    while sa_steps < max_sa_steps and time.time() - start_time < budget_s - 10:
        # Make a perturbed copy
        h_new = h.copy()
        # Perturb 10% of elements
        indices = np.random.choice(n_points, size=int(0.1 * n_points), replace=False)
        for i in indices:
            h_new[i] += np.random.normal(0, perturbation_scale)
        
        # Clip and rescale
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        # Compute C5
        correlation = np.correlate(h_new, 1 - h_new, mode='full')
        c5_new = np.max(correlation) * dx
        
        # Compute current C5
        correlation_old = np.correlate(h, 1 - h, mode='full')
        c5_old = np.max(correlation_old) * dx
        
        # Accept or reject
        if c5_new < c5_old or np.random.rand() < np.exp(-(c5_new - c5_old) / T):
            h = h_new
            if c5_new < best_c5:
                best_c5 = c5_new
                best_h = h.copy()
        
        T *= alpha
        sa_steps += 1
    
    # Phase 2: Nelder-Mead optimization
    # Create initial simplex by perturbing the best solution
    def compute_c5(h):
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
        correlation = np.correlate(h, 1 - h, mode='full')
        return np.max(correlation) * dx
    
    def nelder_mead_step(vertices, f_values, alpha=1.0, gamma=2.0, rho=0.5, sigma=0.5):
        # Sort vertices by function values
        indices = np.argsort(f_values)
        vertices = vertices[indices]
        f_values = f_values[indices]
        
        # Centroid (excluding worst point)
        centroid = np.mean(vertices[:-1], axis=0)
        
        # Reflection
        reflected = centroid + alpha * (centroid - vertices[-1])
        f_reflected = compute_c5(reflected)
        
        if f_values[0] <= f_reflected < f_values[-2]:
            # Accept reflection
            vertices[-1] = reflected
            f_values[-1] = f_reflected
        
        elif f_reflected < f_values[0]:
            # Expansion
            expanded = centroid + gamma * (reflected - centroid)
            f_expanded = compute_c5(expanded)
            
            if f_expanded < f_reflected:
                vertices[-1] = expanded
                f_values[-1] = f_expanded
            else:
                vertices[-1] = reflected
                f_values[-1] = f_reflected
        
        else:
            # Contraction
            if f_reflected < f_values[-1]:
                # Outside contraction
                contracted = centroid + rho * (reflected - centroid)
            else:
                # Inside contraction
                contracted = centroid + rho * (vertices[-1] - centroid)
                
            f_contracted = compute_c5(contracted)
            
            if f_contracted < f_values[-1]:
                vertices[-1] = contracted
                f_values[-1] = f_contracted
            else:
                # Shrink
                for i in range(1, len(vertices)):
                    vertices[i] = vertices[0] + sigma * (vertices[i] - vertices[0])
        
        return vertices, f_values
    
    # Initialize simplex
    simplex_size = min(5, n_points)
    vertices = [best_h.copy()]
    for _ in range(simplex_size - 1):
        pert = np.random.normal(0, 0.05, n_points)
        vertex = best_h + pert
        vertex = np.clip(vertex, 0, 1)
        vertex = vertex * (n_points / 2) / np.sum(vertex)
        vertices.append(vertex)
    
    vertices = np.array(vertices)
    f_values = np.array([compute_c5(v) for v in vertices])
    
    # Nelder-Mead iterations
    for step in range(max_nelder_mead_steps):
        if time.time() - start_time > budget_s - 1:
            break
            
        vertices, f_values = nelder_mead_step(vertices, f_values)
        current_best = min(f_values)
        if current_best < best_c5:
            best_c5 = current_best
            # Find the vertex with the lowest value
            best_idx = np.argmin(f_values)
            best_h = vertices[best_idx].copy()
    
    # Final check
    correlation = np.correlate(best_h, 1 - best_h, mode='full')
    final_c5 = np.max(correlation) * dx
    
    return (best_h.tolist(), final_c5, n_points)
