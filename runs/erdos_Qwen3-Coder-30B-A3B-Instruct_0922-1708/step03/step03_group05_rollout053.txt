import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Use higher resolution
    n_points = 200
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        h = np.ones(n_points) * 0.5
        
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    iteration = 0
    
    # Adam optimizer variables
    m = np.zeros(n_points)
    v = np.zeros(n_points)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-7
    lr = 0.001
    
    # Precompute all lags and indices for efficiency
    max_lag = n_points - 1
    
    def project_h(h_vec):
        # Clip to bounds and rescale to integral constraint
        h_vec = np.clip(h_vec, 0, 1)
        h_vec = h_vec * (n_points / 2) / np.sum(h_vec)
        return h_vec
    
    def compute_c5(h_vec):
        # Compute correlation and find max
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        c5 = np.max(correlation) * dx
        return c5
    
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Compute current C5
        c5 = compute_c5(h)
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Compute gradient using finite differences
        grad = np.zeros_like(h)
        eps_fd = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps_fd
            h_plus = project_h(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps_fd
            
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * grad**2
        m_hat = m / (1 - beta1**iteration)
        v_hat = v / (1 - beta2**iteration)
        h_new = h - lr * m_hat / (np.sqrt(v_hat) + eps)
        
        # Project and rescale
        h_new = project_h(h_new)
        h = h_new
        
        # Apply Simulated Annealing with decreasing probability
        if np.random.rand() < max(0.1 * np.exp(-iteration/100), 0.01):
            # Random perturbation
            pert_mag = 0.05
            perturbation = np.random.randn(n_points) * pert_mag
            h = project_h(h + perturbation)
    
    # Final check
    final_c5 = compute_c5(best_h)
    
    return (best_h.tolist(), final_c5, n_points)
