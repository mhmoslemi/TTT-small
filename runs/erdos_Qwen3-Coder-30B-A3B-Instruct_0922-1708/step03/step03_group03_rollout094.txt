import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize using the provided parent or a uniform distribution
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            # Interpolate to target resolution
            h_orig = np.array(initial_h_values)
            x_orig = np.linspace(0, 2, len(h_orig))
            x_target = np.linspace(0, 2, n_points)
            h = np.interp(x_target, x_orig, h_orig)
        h = h / np.sum(h) * (n_points / 2)
    else:
        # Initialize with a step function
        h = np.ones(n_points) * 0.5
        
    start_time = time.time()
    
    def compute_c5(h_vec):
        correlation = np.correlate(h_vec, 1 - h_vec, mode='full')
        return np.max(correlation) * dx
    
    best_c5 = compute_c5(h)
    best_h = h.copy()
    
    # Binary search on C5 upper bound
    low = 0.0
    high = 0.38085
    tolerance = 1e-5
    
    def is_feasible(c_target, h_ref):
        # Focus on top 10% most critical lags
        correlation = np.correlate(h_ref, 1 - h_ref, mode='full')
        idxs = np.argsort(-correlation)
        top_k = max(5, int(0.1 * len(idxs)))
        critical_lags = idxs[:top_k]
        
        h_vars = cp.Variable(n_points)
        constraints = [
            h_vars >= 0,
            h_vars <= 1,
            cp.sum(h_vars) == n_points / 2
        ]
        
        # Add constraint for each critical lag
        for k in critical_lags:
            # Ensure that overlap for lag k is bounded by c_target
            # Approximating the sum as a linear constraint
            # This assumes a linearization around current h_ref
            coeff = np.zeros(n_points)
            for i in range(n_points):
                j = (i + k) % n_points
                coeff[i] = 1 - h_ref[j]
            # Add constraint sum_{i=0}^{n-1} coeff[i] * h_vars[i] <= c_target * dx
            constraints.append(cp.sum(coeff * h_vars) <= c_target * dx)
            
        # Try to solve it using MOSEK
        try:
            objective = cp.Minimize(0)  # Dummy objective
            prob = cp.Problem(objective, constraints)
            prob.solve(solver=cp.MOSEK, verbose=False)
            return prob.status == cp.OPTIMAL
        except:
            return False
    
    # Main binary search loop
    while high - low > tolerance and (time.time() - start_time) < budget_s - 5:
        c_test = (low + high) / 2
        if is_feasible(c_test, h):
            high = c_test
        else:
            low = c_test
            
    # Refine using gradient-based optimization with the final target
    c_final = high
    c = c_final
    tolerance = 1e-6
    
    # Local optimization using gradient descent on a refined version
    max_iter = 1000
    lr = 0.01
    
    for iter_num in range(max_iter):
        if (time.time() - start_time) > budget_s - 5:
            break
            
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
            
        # Small perturbation to compute gradient
        grad = np.zeros_like(h)
        eps = 1e-5
        
        for i in range(n_points):
            h_plus = h.copy()
            h_plus[i] += eps
            h_plus = np.clip(h_plus, 0, 1)
            h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
            
            c5_plus = compute_c5(h_plus)
            grad[i] = (c5_plus - c5) / eps
            
        # Update with gradient
        h_new = h - lr * grad
        
        # Re-normalize and clip
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        
        h = h_new
    
    # Final verification
    final_c5 = compute_c5(best_h)
    if final_c5 < best_c5:
        best_c5 = final_c5
    
    return (best_h.tolist(), best_c5, n_points)
