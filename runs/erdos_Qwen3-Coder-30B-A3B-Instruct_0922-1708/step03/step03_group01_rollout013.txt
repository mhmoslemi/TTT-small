import numpy as np
import time
import hashlib

def evaluate_erdos_solution(h):
    """Evaluate the Erdős overlap integral for given h"""
    # Use np.correlate for convolution with mode="full"
    correlation = np.correlate(h, 1 - h, mode='full')
    # The maximum overlap integral C_5
    return np.max(correlation) * (2.0 / len(h))

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    # Initialize parameters
    n_points = 100
    dx = 2.0 / n_points
    max_iterations = 10000
    tabu_list_size = 50
    stagnation_threshold = 100
    step_size = 0.1
    
    # Initialize h from existing construction if available, otherwise uniform
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        if len(h) != n_points:
            x_orig = np.linspace(0, 2, len(h))
            x_new = np.linspace(0, 2, n_points)
            h = np.interp(x_new, x_orig, h)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    # Initialize best solution
    best_h = h.copy()
    best_c5 = evaluate_erdos_solution(best_h)
    
    # TABU list storage
    tabu_list = set()
    tabu_list.add(hashlib.md5(h.tobytes()).hexdigest())
    
    current_h = h.copy()
    current_c5 = best_c5
    
    iteration = 0
    stagnant_count = 0
    
    start_time = time.time()
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iterations:
        iteration += 1
        candidate_best_c5 = float('inf')
        candidate_best_h = None
        
        # Generate multiple candidate solutions
        num_candidates = 50
        for _ in range(num_candidates):
            # Select random indices to perturb
            num_perturb = max(1, int(0.2 * n_points))
            indices = np.random.choice(n_points, size=num_perturb, replace=False)
            
            # Create candidate h by perturbing selected indices
            candidate_h = current_h.copy()
            
            # Add noise to selected indices
            for i in indices:
                delta = np.random.uniform(-step_size, step_size)
                candidate_h[i] += delta
                
            # Clip values to [0,1] range
            candidate_h = np.clip(candidate_h, 0, 1)
            
            # Normalize to sum to n_points/2
            scale_factor = (n_points / 2) / np.sum(candidate_h)
            candidate_h *= scale_factor
            
            # Evaluate this candidate
            candidate_c5 = evaluate_erdos_solution(candidate_h)
            
            # Hash the candidate for tabu check
            hash_val = hashlib.md5(candidate_h.tobytes()).hexdigest()
            
            # Accept if not in tabu list and better than previous best
            if hash_val not in tabu_list and candidate_c5 < candidate_best_c5:
                candidate_best_c5 = candidate_c5
                candidate_best_h = candidate_h.copy()
                
        # If we found a candidate
        if candidate_best_h is not None:
            # Check if it's better than current best
            if candidate_best_c5 < best_c5:
                best_c5 = candidate_best_c5
                best_h = candidate_best_h.copy()
                stagnant_count = 0  # reset stagnation counter
            else:
                stagnant_count += 1
                
            # Add current_h to tabu list
            tabu_list.add(hashlib.md5(current_h.tobytes()).hexdigest())
            
            # If tabu list is too large, remove oldest entry
            if len(tabu_list) > tabu_list_size:
                # Remove oldest entry (simple approach, just take first)
                tabu_list.pop()
                
            # Update current solution
            current_h = candidate_best_h.copy()
            current_c5 = candidate_best_c5
            
            # Occasionally accept worse solutions to escape local minima
            if np.random.rand() < 0.05 and candidate_best_c5 > best_c5:
                # Accept even if worse based on probability
                tabu_list.add(hashlib.md5(current_h.tobytes()).hexdigest())
                current_h = candidate_best_h.copy()
                current_c5 = candidate_best_c5
                
        # Increase step size if we've been stagnant
        if stagnant_count > stagnation_threshold:
            step_size *= 1.2
            stagnant_count = 0
            # Empty tabu list to allow more exploration
            tabu_list.clear()
            
    return (best_h.tolist(), best_c5, n_points)
