import numpy as np
from scipy.optimize import minimize
import time
import cvxpy as cp

def frank_zadoff_sequence(n):
    """Generate a Frank-Zadoff sequence adapted for minimal cross-correlation."""
    seq = np.zeros(n)
    for i in range(n):
        seq[i] = np.sin(2 * np.pi * i * (i + 1) / (n + 1))
    return (seq + 1) / 2

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    
    # Use initial construction if available
    if 'initial_h_values' in globals():
        h = np.array(initial_h_values)
        n_points = len(h)
        dx = 2.0 / n_points
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)
    else:
        n_points = 200
        dx = 2.0 / n_points
        h = frank_zadoff_sequence(n_points)
        h = np.clip(h, 0, 1)
        h = h * (n_points / 2) / np.sum(h)

    best_h = h.copy()
    best_c5 = float('inf')
    
    # Phase 1: Binary search with QCQP
    low = 0.38
    high = 0.4
    tolerance = 1e-4
    
    while time.time() - start_time < budget_s * 0.6 and (high - low) > tolerance:
        c_mid = (low + high) / 2
        # QCQP feasibility check
        h_var = cp.Variable(n_points)
        constraints = [
            h_var >= 0,
            h_var <= 1,
            cp.sum(h_var) == n_points / 2
        ]
        
        # Add overlap constraints for all k
        for k in range(n_points):
            # The overlap for shift k: sum(h[i] * (1 - h[i+k]))
            # This is convex in h for each fixed k
            # However, as we are looking for the max, we include a constraint
            # that ensures max of such overlaps is <= c_mid / dx
            overlap = 0
            for i in range(n_points):
                j = (i + k) % n_points
                overlap += h_var[i] * (1 - h_var[j])
            overlap *= dx
            constraints.append(overlap <= c_mid)
        
        # We don't actually solve, just use this as a "can we do it?"
        # Instead, we'll try to directly optimize in Phase 2 with this as target
        # If we use qcqp approach correctly it would be more principled, but testing shows
        # doing a direct smooth optimization on this isn't ideal
        # Try to just increase resolution with gradient seekers
    
        if high - low < tolerance:
            break
        low = c_mid
    
    # Phase 2: Refinement using gradient descent
    iteration = 0
    alpha = 0.01
    max_iter = 1000
    
    while time.time() - start_time < budget_s - 1 and iteration < max_iter:
        iteration += 1
        
        # Compute overlap integral efficiently
        overlaps = []
        for k in range(n_points):
            shift = np.roll(1 - h, k)
            overlap = np.dot(h, shift)
            overlaps.append(overlap)
        c5 = np.max(overlaps) * dx
        
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Compute gradient analytically for max overlap
        max_k = np.argmax(overlaps)
        grad = np.zeros_like(h)
        for i in range(n_points):
            j = (i + max_k) % n_points
            grad[i] = (1 - h[j]) - h[(i - max_k) % n_points]
        grad *= dx
        
        # Update with gradient descent
        h_new = h - alpha * grad
        
        # Project to constraints
        h_new = np.clip(h_new, 0, 1)
        h_new = h_new * (n_points / 2) / np.sum(h_new)
        h = h_new
        
        # Adaptive step size
        if iteration % 50 == 0:
            alpha = max(0.0001, alpha * 0.95)
    
    # Downsample to 100 points if needed
    if n_points > 100:
        h_final = np.interp(np.linspace(0, 2, 100), np.linspace(0, 2, n_points), h)
        h_final = np.clip(h_final, 0, 1)
        h_final = h_final * (100 / 2) / np.sum(h_final)
        h = h_final
        n_points = 100
    
    # Final evaluation
    overlaps = []
    for k in range(n_points):
        shift = np.roll(1 - h, k)
        overlap = np.dot(h, shift)
        overlaps.append(overlap)
    final_c5 = np.max(overlaps) * (2.0 / n_points)
    
    return (h.tolist(), final_c5, n_points)
