import numpy as np
import cvxpy as cp
import time

def run(seed=42, budget_s=600, **kwargs):
    np.random.seed(seed)
    
    n_points = 100
    dx = 2.0 / n_points
    
    # Initialize from parent or uniform
    if 'initial_h_values' in globals():
        h_orig = np.array(initial_h_values)
        x_orig = np.linspace(0, 2, len(h_orig))
        x_target = np.linspace(0, 2, n_points)
        h = np.interp(x_target, x_orig, h_orig)
        h = h * (n_points / 2) / np.sum(h)
    else:
        h = np.ones(n_points) * 0.5
    
    best_h = h.copy()
    best_c5 = float('inf')
    
    start_time = time.time()
    
    # Precompute all lags for convolution
    lags = list(range(1, n_points))
    
    def compute_c5(h_vals):
        corr = np.correlate(h_vals, 1 - h_vals, mode='full')
        return np.max(corr) * dx
    
    # Convex relaxation helper
    def solve_convex_relaxation(h_in):
        h_relaxed = cp.Variable(n_points)
        constraints = [h_relaxed >= 0, h_relaxed <= 1, cp.sum(h_relaxed) == n_points / 2]
        
        # For each lag, compute corresponding overlap term
        overlaps = []
        for i in range(n_points):
            if i < n_points - 1:
                overlap = cp.sum(cp.multiply(h_relaxed[i:], 1 - h_relaxed[:n_points-i]))
            else:
                overlap = 0
            overlaps.append(overlap)
        
        objective = cp.Minimize(cp.max(cp.hstack(overlaps)))
        prob = cp.Problem(objective, constraints)
        try:
            prob.solve(solver=cp.ECOS, eps=1e-6)
            if prob.status == cp.OPTIMAL:
                return h_relaxed.value
            else:
                return None
        except:
            return None
    
    # Rounding helper with prioritization
    def round_solution(h_relaxed):
        h_rounded = (h_relaxed >= 0.5).astype(float)
        current_sum = np.sum(h_rounded)
        target_sum = n_points / 2
        
        if current_sum > target_sum:
            # Need to flip some 1s to 0s
            diff = int(current_sum - target_sum)
            indices_to_flip = np.argsort(h_relaxed)[:diff]
            for i in indices_to_flip:
                h_rounded[i] = 0.0
        elif current_sum < target_sum:
            # Need to flip some 0s to 1s
            diff = int(target_sum - current_sum)
            indices_to_flip = np.argsort(h_relaxed)[-diff:]  # picking largest values
            for i in indices_to_flip:
                h_rounded[i] = 1.0
        
        # Final normalization
        h_rounded = h_rounded * (n_points / 2) / np.sum(h_rounded)
        return h_rounded

    iteration = 0
    while time.time() - start_time < budget_s - 1:
        iteration += 1
        
        # Current evaluation
        c5 = compute_c5(h)
        if c5 < best_c5:
            best_c5 = c5
            best_h = h.copy()
        
        # Convex relaxation
        h_relaxed = solve_convex_relaxation(h)
        if h_relaxed is not None:
            h_new = round_solution(h_relaxed)
            if compute_c5(h_new) < compute_c5(h):
                h = h_new
        else:
            # Fallback to gradient descent
            grad = np.zeros_like(h)
            eps = 1e-4
            for i in range(n_points):
                h_plus = h.copy()
                h_plus[i] += eps
                h_plus = np.clip(h_plus, 0, 1)
                h_plus = h_plus * (n_points / 2) / np.sum(h_plus)
                c5_plus = compute_c5(h_plus)
                grad[i] = (c5_plus - c5) / eps
            
            lr = 0.002
            h_new = h - lr * grad
            h_new = np.clip(h_new, 0, 1)
            h_new = h_new * (n_points / 2) / np.sum(h_new)
            h = h_new
            
    # Final check
    final_c5 = compute_c5(best_h)
    return (best_h.tolist(), final_c5, n_points)
